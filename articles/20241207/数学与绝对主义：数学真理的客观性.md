sigma = 0.2  # 资产价格波动率

# 布莱克-舒尔斯模型公式
def black_scholes(S0, K, T, r, sigma):
    d1 = (np.log(S0 / K) + (r + 0.5 * sigma ** 2) * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    call_price = (S0 * np.exp(-r * T) * norm.cdf(d1) - K * np.exp(-r * T) * norm.cdf(d2))
    return call_price

# 标准正态分布函数
from scipy.stats import norm

# 计算期权价格
