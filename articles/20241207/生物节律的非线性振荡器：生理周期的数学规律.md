```python
import numpy as np
import matplotlib.pyplot as plt

# Lorenz 系统的微分方程
def lorenz_system(x, y, z, sigma=10, rho=28, beta=8/3):
    dxdt = sigma * (y - x)
    dydt = x * (rho - z) - y
    dzdt = x * y - beta * z
    return dxdt, dydt, dzdt

# 使用欧拉方法进行数值求解
def lorenz_integration(x0, y0, z0, dt=0.01, steps=10000):
    x, y, z = x0, y0, z0
    x_values, y_values, z_values = [x0], [y0], [z0]

    for _ in range(steps):
        dxdt, dydt, dzdt = lorenz_system(x, y, z)
        x, y, z = x + dxdt * dt, y + dydt * dt, z + dzdt * dt
        x_values.append(x)
        y_values.append(y)
        z_values.append(z)

    return x_values, y_values, z_values

# 求解并绘制 Lorenz 流轨图
x_values, y_values, z_values = lorenz_integration(1, 1, 1)
plt.plot(x_values, y_values, label='Lorenz Trajectory')
plt.xlabel('X')
plt.ylabel('Y')
plt.title('Lorenz System Trajectory')
plt.legend()
plt.show()
```
