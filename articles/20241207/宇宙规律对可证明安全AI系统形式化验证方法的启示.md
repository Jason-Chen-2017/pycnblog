```python
def calculate_energy(mass, velocity):
    return 0.5 * mass * velocity**2

def is_converged(prev_energy, current_energy, threshold):
    return abs(prev_energy - current_energy) < threshold

def optimize():
    mass = 1.0
    velocity = 1.0
    prev_energy = calculate_energy(mass, velocity)
    while not is_converged(prev_energy, current_energy, 0.01):
        prev_energy = current_energy
        velocity *= 0.9  # 更新速度
        current_energy = calculate_energy(mass, velocity)
    print("Optimized velocity:", velocity)
```
