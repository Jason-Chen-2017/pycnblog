  ```python
  import numpy as np
  import matplotlib.pyplot as plt

  def simulate_blackhole(M_bh, r_min, r_max, N):
      # 模拟中间质量黑洞的引力场
      rs = np.linspace(r_min, r_max, N)
      fs = np.zeros_like(rs)
      for i, r in enumerate(rs):
          m = 1.989e30  # 单位：千克
          G = 6.6743e-11  # 单位：牛顿·平方米²/千克²
          f = G * M_bh * m / r**2
          fs[i] = f
      return rs, fs

  def plot_simulation(rs, fs):
      plt.plot(rs, fs)
      plt.xlabel('Distance (m)')
      plt.ylabel('Force (N)')
      plt.title('Gravitational Force of a Intermediate-mass Black Hole')
      plt.show()

  if __name__ == '__main__':
      M_bh = 10**4 * 1.989e30
      r_min = 1e9
      r_max = 1e13
      N = 1000
      rs, fs = simulate_blackhole(M_bh, r_min, r_max, N)
      plot_simulation(rs, fs)
  ```
