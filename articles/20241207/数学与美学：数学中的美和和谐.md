```python
import numpy as np

class MathModel:
    def __init__(self, equation, variables):
        self.equation = equation
        self.variables = variables

    def evaluate(self, values):
        return eval(self.equation, {"math": np, **values})

class LinearModel(MathModel):
    def __init__(self, a, b):
        equation = f"{a}x + {b} = 0"
        variables = ["x"]
        super().__init__(equation, variables)

class QuadraticModel(MathModel):
    def __init__(self, a, b, c):
        equation = f"{a}x^2 + {b}x + {c} = 0"
        variables = ["x"]
        super().__init__(equation, variables)
```

2. **美学分析模块**：

```python
def analyze_model(model):
    # 分析模型的美学特征，如对称性、简洁性等
    # 这里以对称性为例进行简单分析
    terms = model.equation.split()
    is_symmetric = all([term == term[::-1] for term in terms if term.isalpha()])
    return is_symmetric

def get_model_aesthetics(model):
    is_symmetric = analyze_model(model)
    aesthetics = "对称" if is_symmetric else "非对称"
    return aesthetics
```

3. **可视化模块**：

```python
import matplotlib.pyplot as plt

def plot_model(model, values=None):
    if isinstance(model, LinearModel):
        x = np.linspace(values[0] - 5, values[0] + 5, 100)
        y = -model.a * x - model.b
        plt.plot(x, y, label=f"{model.equation}")
    elif isinstance(model, QuadraticModel):
        x = np.linspace(values[0] - 10, values[0] + 10, 100)
        y = model.a * x**2 + model.b * x + model.c
        plt.plot(x, y, label=f"{model.equation}")
    plt.xlabel('x')
    plt.ylabel('y')
    plt.title(f"Model: {model.equation}")
    plt.legend()
    plt.show()
```
