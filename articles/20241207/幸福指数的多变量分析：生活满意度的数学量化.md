import numpy as np
import pandas as pd
from sklearn.decomposition import PCA
from sklearn.cluster import KMeans
from sklearn.metrics import silhouette_score
import matplotlib.pyplot as plt

# 数据收集
def data_collection():
    # 这里可以使用问卷调查等方式收集数据
    # 本例中使用已收集的样本数据
    data = pd.read_csv('happiness_data.csv')
    return data

# 数据清洗
def data_cleaning(data):
    # 去除缺失值、异常值等
    # 数据预处理
    data = data.dropna()
    return data

# 主成分分析
def principal_component_analysis(data):
    pca = PCA(n_components=2)
    principal_components = pca.fit_transform(data)
    return principal_components

# 聚类分析
def kmeans_clustering(data, n_clusters=3):
    kmeans = KMeans(n_clusters=n_clusters)
    clusters = kmeans.fit_predict(data)
    return clusters

# 结果分析
def result_analysis(clusters):
    silhouette_avg = silhouette_score(data, clusters)
    print(f"Silhouette Score: {silhouette_avg}")
    
    # 可视化结果
    plt.scatter(data[:, 0], data[:, 1], c=clusters, cmap='viridis')
    plt.xlabel('Principal Component 1')
    plt.ylabel('Principal Component 2')
    plt.title('K-Means Clustering')
    plt.show()

# 主函数
def main():
    data = data_collection()
    data = data_cleaning(data)
    principal_components = principal_component_analysis(data)
    clusters = kmeans_clustering(principal_components)
    result_analysis(clusters)

if __name__ == '__main__':
    main()
```
