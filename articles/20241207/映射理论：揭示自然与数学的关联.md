import cvxpy as cp

x = cp.Variable()

# 构建线性规划模型
supply = 100  # 供给量
demand = 50   # 需求量
price = 10    # 价格
profit = x * price

# 添加约束
constraints = [x >= supply, x <= demand]

# 求解线性规划
prob = cp.Problem(cp.Minimize(profit), constraints)
