**源代码实现：**  
以下是一个简单的思维链算法实现示例，用于对考古数据进行预处理和关系网络构建。

```python
import numpy as np
import pandas as pd
from sklearn import preprocessing
from sklearn.cluster import KMeans
import networkx as nx
import matplotlib.pyplot as plt

# 数据预处理
def preprocess_data(data):
    # 数据清洗
    data = data.dropna()
    # 数据整合
    data = data.reset_index(drop=True)
    # 数据特征提取
    data['features'] = data.apply(lambda row: extract_features(row), axis=1)
    return data

# 特征提取
def extract_features(row):
    return [row['feature1'], row['feature2'], row['feature3']]

# 关系网络构建
def build_relation_network(data):
    # 数据输入
    features = data['features'].values
    # 数据标准化
    features = preprocessing.scale(features)
    # K均值聚类
    kmeans = KMeans(n_clusters=5)
    kmeans.fit(features)
    # 构建关系网络
    G = nx.Graph()
    for i in range(features.shape[0]):
        G.add_node(i, label=kmeans.labels_[i])
    for i in range(features.shape[0]):
        for j in range(i+1, features.shape[0]):
            if np.linalg.norm(features[i] - features[j]) < 0.5:
                G.add_edge(i, j)
    return G

# 主函数
def main():
    # 加载数据
    data = pd.read_csv('archaeological_data.csv')
    # 数据预处理
    data = preprocess_data(data)
    # 构建关系网络
    G = build_relation_network(data)
    # 绘制关系网络图
    nx.draw(G, with_labels=True)
    plt.show()

if __name__ == '__main__':
    main()
```

**代码解读：**  
1. **数据预处理：** 对原始数据进行清洗、整合和特征提取，以生成可分析的数据集。
2. **关系网络构建：** 利用K均值聚类方法对数据进行聚类，构建关系网络。
