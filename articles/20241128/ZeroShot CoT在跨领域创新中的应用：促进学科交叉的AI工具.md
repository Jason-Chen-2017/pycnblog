similarity = torch.cosine_similarity(hidden_states[0], hidden_states[1], dim=1)

# 6. 转移概念

print(f"Transferred concept index: {transferred_concept}")

# 7. 解读转移的概念
if transferred_concept == 0:
    print("The concept from text1 is more similar to text2.")
else:
    print("The concept from text2 is more similar to text1.")
