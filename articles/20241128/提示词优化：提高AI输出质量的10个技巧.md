```python
def adjust_word_length(input_prompt, max_length):
    """
    调整提示词长度，使其不超过max_length
    """
    words = input_prompt.split()
    if len(words) > max_length:
        words = words[:max_length]
        adjusted_prompt = " ".join(words)
    else:
        adjusted_prompt = input_prompt
    return adjusted_prompt
```

**实战2：提升关键词密度**

```python
from collections import Counter

def increase_keyword_density(prompt: str, target_density: float) -> str:
    words = prompt.split()
    word_counts = Counter(words)
    total_words = len(words)
    target_word_count = int(total_words * target_density)
    most_common_words = [word for word, count in word_counts.most_common() if count > 1]
    for word in most_common_words:
        if word_counts[word] > 1:
            words.extend([word] * (target_word_count - total_words))
    adjusted_prompt = " ".join(words)
    return adjusted_prompt

input_prompt = "请问你最喜欢的水果是什么？"
adjusted_prompt = increase_keyword_density(input_prompt, 0.8)
print("调整后的提示词：", adjusted_prompt)
```

```python
import random

def add_diversity_to_prompt(prompt: str, additional_prompts: List[str]) -> str:
    words = prompt.split()
    random_prompt = random.choice(additional_prompts)
    random_words = random_prompt.split()
    for word in random_words:
        if word not in words:
            words.append(word)
    adjusted_prompt = " ".join(words)
    return adjusted_prompt

input_prompt = "请问你最喜欢的水果是什么？"
additional_prompts = [
    "你喜欢哪种水果？",
    "你最爱的水果是哪个？",
    "请告诉我你喜欢的水果。"
]
adjusted_prompt = add_diversity_to_prompt(input_prompt, additional_prompts)
print("调整后的提示词：", adjusted_prompt)
```
