# 提示词设计项目：生成关于环境保护的文本

import random
from nltk.tokenize import word_tokenize

# 提示词库
prompts = [
    "环境保护与可持续发展",
    "地球的未来与环境保护",
    "保护环境，从我做起",
    "环境保护的重要性",
    "可持续发展的挑战与机遇"
]

# 选择随机提示词
selected_prompt = random.choice(prompts)

# 生成文本
def generate_text(prompt):
    words = word_tokenize(prompt)
    text = " ".join(words)
    return text

generated_text = generate_text(selected_prompt)
print(generated_text)

# 代码解读：
# 1. 导入必需的库。
# 2. 定义提示词库。
# 3. 从提示词库中随机选择一个提示词。
