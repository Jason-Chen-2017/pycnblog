#### 4.1 算法原理

##### 4.1.1 决策树算法
决策树是一种常用的分类算法，通过一系列的决策规则对数据进行分类。下面是一个简单的决策树算法的实现。

```python
class DecisionTreeClassifier:
    def __init__(self, criterion='gini', max_depth=None):
        self.criterion = criterion
        self.max_depth = max_depth
    
    def fit(self, X, y):
        self.tree_ = self._build_tree(X, y)
    
    def _build_tree(self, X, y):
        # 判断是否满足停止条件
        if self._is_stop_condition(X, y):
            return self._leaf_node(y)
        
        # 计算最优划分
        best_split = self._find_best_split(X, y)
        
        # 划分数据
        left subtree = self._build_tree(best_split['left'], y[best_split['left']])
        right subtree = self._build_tree(best_split['right'], y[best_split['right']])
        
        return {
            'feature': best_split['feature'],
            'threshold': best_split['threshold'],
            'left': left subtree,
            'right': right subtree
        }
    
    def _is_stop_condition(self, X, y):
        # 判断数据是否纯净
        if len(set(y)) == 1:
            return True
        # 判断特征是否不足
        if X.shape[1] == 0:
            return True
        # 判断最大深度
        if self.max_depth is not None and self.max_depth <= 0:
            return True
        return False
    
    def _leaf_node(self, y):
        return {
            'label': max(y, key=y.count)
        }
    
    def _find_best_split(self, X, y):
        best_split = None
        max_impurity = -1
        
        for feature in X.columns:
            for threshold in X[feature].unique():
                left = X[X[feature] <= threshold]
                right = X[X[feature] > threshold]
                impurity = self._calculate_impurity(y[left], y[right])
                
                if impurity > max_impurity:
                    max_impurity = impurity
                    best_split = {
                        'feature': feature,
                        'threshold': threshold,
                        'left': left.index,
                        'right': right.index
                    }
        
        return best_split
    
    def _calculate_impurity(self, y_left, y_right):
        p0 = len(y_left) / (len(y_left) + len(y_right))
        p1 = len(y_right) / (len(y_left) + len(y_right))
        return p0 * p0 + p1 * p1
```
