    if User.objects.filter(username=username).exists():
        return HttpResponse('注册失败：用户名已存在', status=400)
    
    user = User(username=username, password=password)
    user.save()
    
    return HttpResponse('注册成功', status=200)

# 用户登录
def login(request):
    username = request.POST.get('username')
    password = request.POST.get('password')
    
    if not is_valid_username(username) or not is_valid_password(password):
        return HttpResponse('登录失败：用户名或密码不正确', status=400)
    
    user = User.objects.filter(username=username).first()
    if not user:
        return HttpResponse('登录失败：用户不存在', status=400)
    
    if user.password != password:
        return HttpResponse('登录失败：密码不正确', status=400)
    
    return HttpResponse({'user_id': user.id}, status=200)
