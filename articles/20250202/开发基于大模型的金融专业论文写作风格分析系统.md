##### Writing Style Analysis

```python
from sklearn.cluster import KMeans
import numpy as np

# Extract sentence embeddings from the trained model
def extract_sentence_embeddings(model, texts):
    model.eval()
    embeddings = []
    with torch.no_grad():
        for text in texts:
            inputs = tokenizer(text, return_tensors='pt', truncation=True, padding=True)
            inputs = {key: value.to(device) for key, value in inputs.items()}
            outputs = model(**inputs)
            embeddings.append(outputs.last_hidden_state.mean(dim=1).cpu().numpy())
    return np.array(embeddings)

# Perform K-means clustering on sentence embeddings to identify writing styles
def analyze_writing_style(embeddings, num_clusters=3):
    kmeans = KMeans(n_clusters=num_clusters, random_state=42)
    kmeans.fit(embeddings)
    return kmeans.labels_

# Analyze writing styles of the test data
test_embeddings = extract_sentence_embeddings(model, test_data['preprocessed_text'].tolist())
writing_styles = analyze_writing_style(test_embeddings)

# Print the results
for i, style in enumerate(writing_styles):
    print(f"Test Text {i+1}: Writing Style {style}")
```
