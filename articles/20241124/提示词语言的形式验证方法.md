### 项目实战示例

以下是一个简单的Python脚本示例，展示了如何使用NLTK和朴素贝叶斯分类器进行情感分析。

```python
import nltk
from nltk.tokenize import word_tokenize
from nltk.corpus import stopwords
from nltk.classify import NaiveBayesClassifier
from nltk.corpus import movie_reviews

# 数据准备
nltk.download('movie_reviews')
nltk.download('stopwords')
stop_words = set(stopwords.words('english'))

# 分词与去除停用词
def preprocess(text):
    tokens = word_tokenize(text)
    return [word.lower() for word in tokens if word.isalpha() and word not in stop_words]

# 加载和准备训练数据
def load_data():
    fileids = movie_reviews.fileids()
    data = []
    for fileid in fileids:
        text = movie_reviews.raw(fileid).lower()
        label = 'positive' if fileid.startswith('pos') else 'negative'
        words = preprocess(text)
        data.append((words, label))
    return data

data = load_data()

# 训练模型
classifier = NaiveBayesClassifier.train(data)

# 预测
def predict(text):
    words = preprocess(text)
    return classifier.classify(words)

# 示例
print(predict("This movie was amazing!"))
print(predict("I hated this movie!"))

```
