**后端代码（app.py）：**

```python
from flask import Flask, request, jsonify

app = Flask(__name__)

@app.route('/generate_prompt', methods=['POST'])
def generate_prompt():
    text = request.form['text']
    prompt = generate_prompt_function(text)
    return jsonify({'prompt': prompt})

def generate_prompt_function(text):
    # 简单的提示词生成算法
    keywords = text.split()
    prompt = " ".join(keywords[:5]) + "。"
    return prompt

if __name__ == '__main__':
    app.run(debug=True)
```
