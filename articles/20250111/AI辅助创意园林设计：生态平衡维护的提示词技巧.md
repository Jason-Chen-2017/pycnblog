  \[ \frac{dZ}{dt} = -\beta Z + \alpha P\frac{Z}{Z + K} \]

```python
        child1 = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
        child2 = np.concatenate((parent2[:crossover_point], parent1[crossover_point:]))
    else:
        child1, child2 = parent1, parent2
    
    # 变异操作
    if np.random.rand() < mutation_rate:
        mutation_point = np.random.randint(0, num_variables)
        child1[mutation_point] = np.random.rand()
        child2[mutation_point] = np.random.rand()
    
    return child1, child2

# 遗传算法主循环
for generation in range(generations):
    # 计算适应度
    fitness_scores = np.array([fitness_function(individual) for individual in population])
    
    # 选择操作
    selected_indices = np.random.choice(range(population_size), size=population_size, p=fitness_scores/np.sum(fitness_scores))
    selected_population = population[selected_indices]
    
    # 生成下一代种群
    next_population = []
    for _ in range(population_size // 2):
        parent1, parent2 = selected_population[np.random.choice(range(population_size), size=2, replace=False)]
        child1, child2 = genetic_operations(parent1, parent2)
        next_population.extend([child1, child2])
    
    population = np.array(next_population)

# 输出最优解
best_individual = population[np.argmax(fitness_scores)]
print("最优解：", best_individual)
print("适应度：", fitness_scores.max())
```
