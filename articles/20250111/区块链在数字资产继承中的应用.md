### 5.2 核心代码实现

在本项目中，我们将使用Solidity语言编写智能合约，实现数字资产的登记、继承设置和执行。以下是核心代码的实现：

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract DigitalAssetInheritance {
    address public owner;
    mapping(uint256 => address) public assets;
    mapping(uint256 => address) public heirs;
    mapping(uint256 => bool) public isRegistered;

    constructor() {
        owner = msg.sender;
    }

    function registerAsset(uint256 assetId, address heir) external {
        require(msg.sender == owner, "Only owner can register assets");
        require(!isRegistered[assetId], "Asset already registered");
        assets[assetId] = heir;
        isRegistered[assetId] = true;
    }

    function setHeir(uint256 assetId, address heir) external {
        require(msg.sender == owner, "Only owner can set heirs");
        require(isRegistered[assetId], "Asset not registered");
        heirs[assetId] = heir;
    }

    function executeInheritance() external {
        require(msg.sender == owner, "Only owner can execute inheritance");
        for (uint256 assetId = 0; assetId < assets.length; assetId++) {
            if (isRegistered[assetId] && heirs[assetId] != address(0)) {
                assets[assetId](heirs[assetId]);
            }
        }
    }
}
```

在这个智能合约中，我们定义了以下变量和方法：

- `owner`：所有者的地址。
- `assets`：存储所有数字资产的映射。
- `heirs`：存储所有继承人的映射。
- `isRegistered`：标记数字资产是否已注册。

方法包括：

- `registerAsset`：注册数字资产和继承人。
- `setHeir`：设置数字资产的继承人。
- `executeInheritance`：执行数字资产的继承。
