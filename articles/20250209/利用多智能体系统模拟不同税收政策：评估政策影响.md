#### 5.2 粒子群优化算法的实现
```python
import random

class Particle:
    def __init__(self, n, bounds):
        self.n = n
        self.bounds = bounds
        self.position = [random.uniform(b[0], b[1]) for b in bounds]
        self.velocity = [0.0 for _ in range(n)]
        self.best = self.position.copy()
        self.pbest = self.position.copy()

def optimize(fitness_func, n_particles, n_dimensions, bounds, max_iter=100):
    particles = [Particle(n_dimensions, bounds) for _ in range(n_particles)]
    global_best = particles[0].position.copy()

    for _ in range(max_iter):
        for particle in particles:
            fitness = fitness_func(particle.position)
            if fitness < min([fitness_func(g.position) for g in particles]):
                global_best = particle.position.copy()
            particle.velocity = [1 - (abs(particle.best[i] - particle.pbest[i])) * 0.1 for i in range(n_dimensions)]
            particle.position = [particle.position[i] + particle.velocity[i] for i in range(n_dimensions)]
        print(f"迭代 {_}: 全局最优值为 {fitness_func(global_best)}")
    return global_best
```

#### 5.3 遗传算法的实现
```python
import random

def fitness(individual):
    return sum(individual)  # 示例：计算个体适应度

def crossover(parent1, parent2):
    return parent2.copy(), parent1.copy()  # 示例：简单的单点交叉

def mutate(individual):
    idx = random.randint(0, len(individual)-1)
    individual[idx] = 1 - individual[idx]  # 示例：翻转突变

def genetic_algorithm(pop_size, gen_limit, fitness_func):
    population = [[random.randint(0,1) for _ in range(5)] for _ in range(pop_size)]
    for gen in range(gen_limit):
        population = sorted(population, key=lambda x: fitness_func(x), reverse=True)
        new_population = population[:pop_size//2]
        for i in range(pop_size//2):
            p1 = new_population[i]
            p2 = new_population[pop_size//2 + i]
            child1, child2 = crossover(p1, p2)
            mutate(child1)
            mutate(child2)
            new_population += [child1, child2]
        population = new_population
    return population[0]
```
