        rates = [initial_rate]
        for _ in range(100):
            # 模拟利率变化
            new_rate = rates[-1] + np.random.normal(0, 0.01)
            rates.append(new_rate)
        plt.plot(rates)
        plt.xlabel('时间')
        plt.ylabel('利率')
        plt.show()

    simulate_interest_rate()
