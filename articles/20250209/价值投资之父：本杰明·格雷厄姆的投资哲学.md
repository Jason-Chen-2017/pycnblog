```python
def calculate_intrinsic_value(cash_flows, discount_rate):
    """
    计算公司的内在价值（现金流折现法）
    """
    present_value = 0
    for year in range(len(cash_flows)):
        cash_flow = cash_flows[year]
        present_value += cash_flow / (1 + discount_rate) ** (year + 1)
    return present_value

# 示例数据
cash_flows = [100, 120, 150, 200]
discount_rate = 0.1

intrinsic_value = calculate_intrinsic_value(cash_flows, discount_rate)
print(f"内在价值: {intrinsic_value}")
```
