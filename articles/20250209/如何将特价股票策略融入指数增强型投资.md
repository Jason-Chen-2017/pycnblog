```python
import pandas as pd

def screen_cheap_stocks(data, threshold=0.5):
    # 计算PE、PB等指标
    data['PE'] = data['price'] / data['eps']
    data['PB'] = data['price'] / data['book_value']
    # 筛选PE < threshold且PB < threshold的股票
    cheap Stocks = data[(data['PE'] < threshold) & (data['PB'] < threshold)]
    return cheap_stocks
```

### 5.2.2 投资组合优化算法
```python
import numpy as np

def optimize_portfolio(stocks, returns, risk_free_rate=0.02):
    # 计算协方差矩阵
    cov_matrix = np.cov(returns)
    # 计算最优权重
    weights = np.linalg.inv(cov_matrix) * np.ones(len(stocks))
    weights = weights / np.sum(weights)
    return weights
```
