```python
import pandas as pd
import requests

def get_stock_data(ticker):
    url = f"https://query1.finance.yahoo.com/finance/quote/{ticker}/key-statistics"
    response = requests.get(url)
    data = response.json()
    pe = data['data']['概要']['市盈率-TTM']
    pb = data['data']['概要']['市净率-LF']
    return {'PE': pe, 'PB': pb}

def screen_stocks(tickers):
    results = []
    for ticker in tickers:
        data = get_stock_data(ticker)
        if float(data['PB']) < 1.5 and float(data['PE']) < 15:
            results.append(ticker)
    return results

tickers = ['AAPL', 'MSFT', 'GOOGL', 'AMZN', 'FB']
low_pb_stocks = screen_stocks(tickers)
print("低市净率股票:", low_pb_stocks)
```
