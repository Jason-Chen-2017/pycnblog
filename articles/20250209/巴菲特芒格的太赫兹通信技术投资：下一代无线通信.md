def generate_signal(frequency, duration):
    # 生成正弦信号
    t = np.linspace(0, duration, 1000)
    signal = np.sin(2 * np.pi * frequency * t)
    return signal

def modulate_signal(signal, carrier_frequency):
    # 调制信号
    modulated_signal = signal * np.sin(2 * np.pi * carrier_frequency * np.linspace(0, 1, len(signal)))
    return modulated_signal

# 示例调用

modulated_signal = modulate_signal(signal, 5000)
