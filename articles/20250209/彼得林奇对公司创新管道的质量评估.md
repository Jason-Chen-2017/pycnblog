import numpy as np
import pandas as pd

def calculate_weights(matrix):
    # 计算权重向量
    # matrix是判断矩阵
    # 返回权重向量
    n = len(matrix)
    # 计算特征向量
    feature_vector = matrix.sum(axis=1)
    # 计算权重
    weights = feature_vector / feature_vector.sum()
    return weights

def ahp_assessment(matrix):
    n = len(matrix)
    # 计算一致性检验
    CI = (np.trace(matrix) - n) / (n - 1)
    CR = CI / np.max(np.linalg.eigvals(matrix))
    if CR < 0.1:
        # 一致性检验通过
        return calculate_weights(matrix)
    else:
        # 需要重新构造判断矩阵
        return None

# 示例判断矩阵
matrix = np.array([
    [1, 2, 3],
    [2, 1, 4],
    [3, 4, 1]
])

# 计算权重
weights = ahp_assessment(matrix)
print("权重向量为:", weights)
