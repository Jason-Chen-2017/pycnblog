  ```python
  import pandas as pd
  import numpy as np

  def calculate_intrinsic_value/assets：
      net_assets = assets['net_asset_value']
      shares = assets['shares_outstanding']
      iv = net_assets / shares
      return iv

  def is_net_net_stock(stock, iv):
      price = stock['price']
      return price < iv

  # 示例数据
  data = {
      'company': ['Company A', 'Company B', 'Company C'],
      'net_asset_value': [1000000000, 800000000, 600000000],
      'shares_outstanding': [10000000, 8000000, 6000000],
      'price': [50, 40, 30]
  }

  df = pd.DataFrame(data)
  df['iv'] = df.apply(lambda x: calculate_intrinsic_value(x['net_asset_value'], x['shares_outstanding']), axis=1)
  df['is_net_net'] = df.apply(lambda x: is_net_net_stock(x['price'], x['iv']), axis=1)
  ```
