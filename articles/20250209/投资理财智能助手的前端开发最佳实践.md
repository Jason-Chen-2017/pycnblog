    function calculate_volatility(returns) {  
        const avg = returns.reduce((a, b) => a + b, 0) / returns.length;  
        const variance = returns.reduce((a, b) => a + (b - avg)**2, 0) / returns.length;  
        return Math.sqrt(variance);  
    }
