#### 5.2 核心代码实现

```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# 假设我们有两支股票的收益率数据
returns = pd.DataFrame({
    'stock1': [0.1, 0.2, 0.15, 0.05, 0.2],
    'stock2': [0.05, 0.1, 0.15, 0.2, 0.1]
})

# 计算协方差矩阵
cov_matrix = returns.cov()

# 计算投资组合的夏普比率
def calculate_sharpe_ratio(weights, returns, cov_matrix):
    portfolio_return = (weights * returns.mean()).sum()
    portfolio_volatility = np.sqrt(weights.T.dot(cov_matrix).dot(weights))
    return (portfolio_return - returns.mean()).std() / portfolio_volatility

# 等权重分配
weights = np.array([0.5, 0.5])
sharpe_ratio = calculate_sharpe_ratio(weights, returns, cov_matrix)
print(sharpe_ratio)

# 可视化投资组合
plt.pie(weights, labels=['stock1', 'stock2'], autopct='%1.1f%%')
plt.show()
```
