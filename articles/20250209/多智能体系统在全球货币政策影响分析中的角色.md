import numpy as np
import matplotlib.pyplot as plt
from networkx import DiGraph

class Agent:
    def __init__(self, id):
        self.id = id
        self.state = 0
        self.neighbors = []
    
    def update_state(self, neighbors_state):
        self.state = np.mean(neighbors_state)

class MAS:
    def __init__(self, num_agents):
        self.agents = [Agent(i) for i in range(num_agents)]
        self.network = DiGraph()
    
    def build_network(self, connections):
        for connection in connections:
            self.network.add_edge(connection[0], connection[1])
    
    def simulate(self, steps):
        for _ in range(steps):
            states = [agent.state for agent in self.agents]
            for agent in self.agents:
                neighbor_states = [self.agents[neighbor].state for neighbor in agent.neighbors]
                agent.update_state(neighbor_states)
        return [agent.state for agent in self.agents]

# 创建多智能体系统实例

class Agent货币政策:

        self.id = id
        self利率 = 初始利率
        self.neighbors = []
    
    def update_rate(self, 外部影响):
        self利率 = self利率 + 外部影响

class MAS货币政策:
    def __init__(self, num_agents):
        self.agents = [Agent货币政策(i, 0.05) for i in range(num_agents)]
        self.network = DiGraph()
    
    def build_network(self, connections):
        for connection in connections:
            self.network.add_edge(connection[0], connection[1])
    
    def simulate(self, steps,政策变化):
        for _ in range(steps):
            for agent in self.agents:
                if agent.id == 政策变化['central_bank']:
                    agent.update_rate(政策变化['base_rate_change'])
                else:
                    external_influence = 0
                    for neighbor in agent.neighbors:
                        external_influence += self.agents[neighbor].利率
                    agent.update_rate(external_influence / len(agent.neighbors))
        return [agent利率 for agent in self.agents]
