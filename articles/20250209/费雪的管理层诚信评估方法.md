def calculate_financial_indices(revenue, net_profit, total_assets):
    roe = (net_profit / total_assets) * 100
    roa = (net_profit / revenue) * 100
    return roe, roa

def calculate_non_financial_indices(innovation, employee_satisfaction):
    return innovation, employee_satisfaction

def fisher_integrity_score(roe, roa, innovation, satisfaction):
    financial_score = (roe + roa) / 2
    non_financial_score = (innovation + satisfaction) / 2
    total_score = (financial_score + non_financial_score) / 2
    return total_score

# 示例
revenue = 500000
net_profit = 50000
total_assets = 250000
innovation = 80
satisfaction = 70

roe, roa = calculate_financial_indices(revenue, net_profit, total_assets)
financial_score = (roe + roa) / 2
innovation_score, satisfaction_score = calculate_non_financial_indices(innovation, satisfaction)
non_financial_score = (innovation_score + satisfaction_score) / 2
total_score = (financial_score + non_financial_score) / 2

print(f"总评分：{total_score}%")
