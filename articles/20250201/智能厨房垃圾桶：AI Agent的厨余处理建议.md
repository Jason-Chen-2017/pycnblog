```python
import cv2
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier

# 数据预处理
def preprocess_image(image):
    # 转换为灰度图像
    gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    # 进行二值化处理
    _, binary = cv2.threshold(gray, 128, 255, cv2.THRESH_BINARY_INV + cv2.THRESH_OTSU)
    return binary

# 特征提取
def extract_features(image):
    # 使用哈希特征提取器
    hasher = cv2.xfeatures2d.SIFT_create()
    keypoints, descriptors = hasher.detectAndCompute(image, None)
    return descriptors

# 训练分类器
def train_classifier(descriptors, labels):
    X_train, X_test, y_train, y_test = train_test_split(descriptors, labels, test_size=0.2, random_state=42)
    classifier = RandomForestClassifier(n_estimators=100, random_state=42)
    classifier.fit(X_train, y_train)
    accuracy = classifier.score(X_test, y_test)
    return classifier, accuracy

# 垃圾识别与分类
def classify_waste(image, classifier):
    processed = preprocess_image(image)
    features = extract_features(processed)
    prediction = classifier.predict([features])
    return prediction

# 示例代码
if __name__ == '__main__':
    image = cv2.imread('example.jpg')
    classifier = train_classifier(extract_features(image), 'organic')
    print(classify_waste(image, classifier))
```

- 决策树分类器：$$ g(x) = \prod_{i=1}^{n} p_i(x) $$
