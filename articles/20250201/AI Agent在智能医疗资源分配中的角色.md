while heap:
    time, department = heapq.heappop(heap)
    schedule_resource(department, time)

# 模型评估
response_time = calculate_response_time()
resource_utilization = calculate_resource_utilization()
print(f'Response Time: {response_time}, Resource Utilization: {resource_utilization}')
