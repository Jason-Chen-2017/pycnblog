- **Python代码示例**：
  ```python
  import pandas as pd
  import numpy as np
  from sklearn.ensemble import RandomForestClassifier
  import matplotlib.pyplot as plt

  # 加载数据
  data = pd.read_csv('market_data.csv')

  # 数据预处理
  data['return'] = data['close'].pct_change()

  # 划分训练集和测试集
  train_data = data[:int(len(data)*0.8)]
  test_data = data[int(len(data)*0.8):]

  # 特征工程
  features = ['open', 'high', 'low', 'volume', 'return']
  X_train = train_data[features]
  y_train = train_data['bubble']

  X_test = test_data[features]
  y_test = test_data['bubble']

  # 模型训练
  model = RandomForestClassifier()
  model.fit(X_train, y_train)

  # 模型预测
  predictions = model.predict(X_test)

  # 结果分析
  accuracy = np.mean(predictions == y_test)
  print(f"Model accuracy: {accuracy:.2f}")

  # 可视化
  plt.scatter(test_data.index, predictions, c='green', label='Predicted Bubble')
  plt.scatter(test_data.index, y_test, c='red', label='Actual Bubble')
  plt.legend()
  plt.show()
  ```
