- 跨部门协作和系统集成

- 自动化操作：通过提示词触发自动化流程，如发送提醒、分配任务等。

### 1.2.3 提示词对提高客户关系管理效率的影响

```python
import pandas as pd
from sklearn.cluster import KMeans
from sklearn.metrics.pairwise import cosine_similarity

# 数据准备
data = pd.DataFrame({
    'user': [1, 1, 2, 2],
    'item': ['Python入门', 'Python进阶', 'Java入门', 'Java进阶'],
    'rating': [5, 4, 5, 4]
})

# 构建用户-物品矩阵
user_item_matrix = data.pivot(index='user', columns='item', values='rating').fillna(0)

# 使用K-means聚类构建用户画像
kmeans = KMeans(n_clusters=2, random_state=0).fit(user_item_matrix)
user_clusters = kmeans.labels_

# 计算用户相似度
user_similarity = cosine_similarity(user_item_matrix)

# 生成推荐列表
def generate_recommendations(user_id, similarity_matrix, clusters):
    similar_users = similarity_matrix[user_id].argsort()[::-1]
    similar_users = similar_users[similar_users != user_id]

    recommendations = []
    for user in similar_users:
        if user_clusters[user] != user_clusters[user_id]:
            continue
        recommendations.extend([item for item in user_item_matrix.index if item not in data[data.user == user_id].item.values])

    return recommendations[:5]

# 测试
print(generate_recommendations(1, user_similarity, user_clusters))
```
