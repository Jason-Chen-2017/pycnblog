def filter_prompts(prompts, threshold=0.5):

    for prompt in prompts:
        if prompt_score(prompt) > threshold:
            filtered_prompts.append(prompt)
    return filtered_prompts

    synonyms = get_synonyms(prompt)  # 获取与prompt相关的同义词

    for synonym in synonyms:
        enhanced_prompt = enhanced_prompt.replace(prompt, synonym)
    return enhanced_prompt
