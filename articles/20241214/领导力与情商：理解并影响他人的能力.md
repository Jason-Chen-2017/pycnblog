```python
# Python代码示例：情感分析
import nltk
from nltk.corpus import movie_reviews

nltk.download('movie_reviews')

def sentiment_analysis(text):
    # 文本预处理
    tokens = nltk.word_tokenize(text)
    tagged = nltk.pos_tag(tokens)
    
    # 特征提取
    features = nltk.FreqDist(tag for word, tag in tagged)
    
    # 分类器训练
    classifier = nltk.classify.train([(list(features.keys()) for features in movie_reviews.fileids('pos')), 'positive'],
                                      [(list(features.keys()) for features in movie_reviews.fileids('neg')), 'negative'])
    
    # 输出结果
    return classifier.classify(features)

text = "这是一部非常有趣的电影，情节紧凑，演员表现出色。"
print(sentiment_analysis(text))
```

### 6.2 情感识别算法

情感识别算法的核心在于如何从文本中提取情感信息。以下是一个简单的情感识别算法的Python代码示例：

```python
# Python代码示例：情感识别
from textblob import TextBlob

def sentiment_recognition(text):
    analysis = TextBlob(text)
    if analysis.sentiment.polarity > 0:
        return "正面情感"
    elif analysis.sentiment.polarity == 0:
        return "中性情感"
    else:
        return "负面情感"

text = "我非常不喜欢这部电影，剧情无聊，演员表演差。"
print(sentiment_recognition(text))
```

```python
# Python代码示例：情感分析
import nltk
from nltk.corpus import movie_reviews

nltk.download('movie_reviews')

def sentiment_analysis(text):
    # 文本预处理
    tokens = nltk.word_tokenize(text)
    tagged = nltk.pos_tag(tokens)
    
    # 特征提取
    features = nltk.FreqDist(tag for word, tag in tagged)
    
    # 分类器训练
    classifier = nltk.classify.train([(list(features.keys()) for features in movie_reviews.fileids('pos')), 'positive'],
                                      [(list(features.keys()) for features in movie_reviews.fileids('neg')), 'negative'])
    
    # 输出结果
    return classifier.classify(features)

text = "这是一部非常有趣的电影，情节紧凑，演员表现出色。"
print(sentiment_analysis(text))
```

### 情感识别算法

```python
# Python代码示例：情感识别
from textblob import TextBlob

def sentiment_recognition(text):
    analysis = TextBlob(text)
    if analysis.sentiment.polarity > 0:
        return "正面情感"
    elif analysis.sentiment.polarity == 0:
        return "中性情感"
    else:
        return "负面情感"

text = "我非常不喜欢这部电影，剧情无聊，演员表演差。"
print(sentiment_recognition(text))
```
