```python
    'relationships': ['relationship1', 'relationship2', 'relationship3']
```

```python
# 提示词生成函数
def generate_prompts(data):
    entities = list(set(data['entity']))
    relationships = list(set(data['relationship']))
    prompts = {
        'entities': entities,
        'relationships': relationships
    }
    return prompts

prompts = generate_prompts(preprocessed_data)
```

##### 5.2.3 提示词优化

```python
# 提示词优化函数
def optimize_prompts(prompts):
    # 优化提示词，例如增加提示词的多样性、减少冗余等
    optimized_prompts = prompts.copy()
    optimized_prompts['entities'] = [entity + '_variant' for entity in optimized_prompts['entities']]
    optimized_prompts['relationships'] = [relationship + '_variant' for relationship in optimized_prompts['relationships']]
    return optimized_prompts

optimized_prompts = optimize_prompts(prompts)
```
