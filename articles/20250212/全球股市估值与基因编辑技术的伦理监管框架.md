```python
import requests
import pandas as pd

def get_stock_data(ticker):
    url = f"https://query1.finance.yahoo.com/ws/api/v3/finance/sticker/{ticker}"
    response = requests.get(url)
    data = response.json()
    return pd.DataFrame(data)

# 示例：获取基因编辑公司GRTX的股票数据
data = get_stock_data("GRTX")
print(data)
```

### 5.2.2 基因编辑技术评估模块

```python
class GeneEditTechnologyEvaluator:
    def __init__(self, data):
        self.data = data

    def calculate_efficiency(self):
        # 示例：计算技术效率
        return sum(self.data['efficiency']) / len(self.data['efficiency'])

    def calculate_safety(self):
        # 示例：计算技术安全性
        return sum(self.data['safety']) / len(self.data['safety'])

    def get_rating(self):
        efficiency = self.calculate_efficiency()
        safety = self.calculate_safety()
        return {
            'efficiency': efficiency,
            'safety': safety,
            'rating': (efficiency + safety) / 2
        }
```
