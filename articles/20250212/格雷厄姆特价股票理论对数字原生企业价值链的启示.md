     ```python
     import pandas as pd
     from sklearn.linear_model import LinearRegression

     # 示例数据：数字企业的用户增长率与股票价格
     data = {'增长率': [10, 20, 30, 40, 50], '股票价格': [100, 120, 130, 150, 180]}
     df = pd.DataFrame(data)

     # 构建线性回归模型
     model = LinearRegression()
     model.fit(df[['增长率']], df['股票价格'])

     # 预测股票价格
     predicted_price = model.predict([[25]])
     print(predicted_price)
     ```
