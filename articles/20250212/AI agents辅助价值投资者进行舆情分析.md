```python
from gensim import corpora, models

def perform_lda(corpus):
    dictionary = corpora.Dictionary(corpus)
    bow = [dictionary.doc2bow(text) for text in corpus]
    lda = models.LdaModel(bow, num_topics=5, id2word=dictionary)
    return lda
```
