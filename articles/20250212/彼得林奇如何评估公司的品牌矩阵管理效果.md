import pandas as pd
import numpy as np

def calculate_brand_matrix(brand_data):
    # 计算品牌权重
    brand_weights = brand_data.mean(axis=1)
    # 构建品牌矩阵
    brand_matrix = brand_data.values
    return brand_matrix, brand_weights

# 示例数据
brand_data = pd.DataFrame({
    '品牌A': [8, 7, 6],
    '品牌B': [5, 6, 7],
    '品牌C': [3, 4, 5]
})

matrix, weights = calculate_brand_matrix(brand_data)
print("品牌矩阵:\n", matrix)
print("品牌权重:\n", weights)
