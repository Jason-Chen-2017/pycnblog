```python
import numpy as np
import matplotlib.pyplot as plt

# 信号生成
fs = 1000
ts = 1/fs
t = np.arange(0, 1, ts)
carrier_freq = 100e9  # 太赫兹频率

# QAM调制
bits = np.random.randint(0, 2, 100)
QAM_modulation = 2*bits - 1
symbol_period = 1/(carrier_freq/1000)

# 信号传输
def modulate(bits, carrier_freq, ts):
    modulated = []
    for bit in bits:
        modulated.append(bit * np.cos(2*np.pi*carrier_freq*t))
    return modulated

modulated_signal = modulate(bits, carrier_freq, ts)

# 信号解调
def demodulate(signal, carrier_freq, ts):
    demodulated = []
    for i in range(len(signal)):
        correlation = np.correlate(signal[i:], np.cos(2*np.pi*carrier_freq*t), mode='valid')
        demodulated.append(np.argmax(correlation))
    return demodulated

demodulated_bits = demodulate(modulated_signal, carrier_freq, ts)
```
