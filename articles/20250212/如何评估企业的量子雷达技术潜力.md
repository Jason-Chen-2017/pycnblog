```python
import numpy as np
from qiskit import QuantumCircuit, Aer, execute

# 量子态的表示
def quantum_state(n=1):
    qc = QuantumCircuit(n)
    qc.h(range(n))  # 应用哈达玛门
    return qc

# 量子纠缠实现
def quantum_entanglement(n=2):
    qc = QuantumCircuit(n)
    qc.h(0)
    qc.csx(0, 1)
    return qc

# 量子态测量
def measure(qc):
    backend = Aer.get_backend('qasm_simulator')
    result = execute(qc, backend).result()
    return result.get_counts()

# 示例代码
qc = quantum_entanglement(2)
print(qc.draw())
result = measure(qc)
print(result)
```
