        # 模拟传感器数据采集

        self.rain_data = data.tolist()

# 数据处理模块
class DataProcessor:
    def __init__(self):
        self.data = []
    
    def process_data(self, data):
        # 数据清洗与预处理
        processed_data = [x for x in data if x > -2]
        return processed_data

# 优化算法模块
class Optimizer:
    def __init__(self):
        self.rain_data = []
    
    def optimize(self, data):
        # 简单的优化算法示例
        optimal_point = max(data)
        return optimal_point

# 系统控制模块
class SystemController:
    def __init__(self):
        self.data_processor = DataProcessor()
        self.optimizer = Optimizer()
        self.rain_sensor = RainSensor()
    
    def run_system(self):
        raw_data = self.rain_sensor.collect_data()
        processed_data = self.data_processor.process_data(raw_data)
        optimal_point = self.optimizer.optimize(processed_data)
