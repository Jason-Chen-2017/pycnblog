```python
# 风险评估模型
def risk_assessment(data):
    model = LinearRegression()
    model.fit(data[['stock', 'fund']], data['bond'])
    return model.predict(data[['stock', 'fund']])

# 风险评分计算
def calculate_risk_score(prediction):
    return np.mean(prediction)
```

### 5.2.3 资产配置模型实现

```python
import numpy as np

# 资产配置模型
def asset_allocation(data):
    # 均值-方差优化
    covariance_matrix = np.cov(data.T)
    returns = data.mean()
    
    # 约束条件
    n = len(returns)
    ones = np.ones(n)
    A = ones.reshape(1, n)
    b = np.array([1.0])
    
    # 优化问题
    from scipy.optimize import minimize
    def objective(weights):
        return np.dot(weights.T, np.dot(covariance_matrix, weights))
    
    def constraint(weights):
        return np.dot(A, weights) - b
    
    # 求解
    weight = minimize(objective, np.ones(n)/n, 
                      constraints=[{'type': 'eq', 'fun': constraint}]).x
    return weight
```
