### 4.3 仿生技术算法的实现

```python
import random

def fitness(x):
    return x**2

def evolve(population_size, mutation_rate):
    population = [random.random() for _ in range(population_size)]
    for _ in range(100):
        fitness_list = [fitness(x) for x in population]
        parents = [x for x in population if fitness(x) > max(fitness_list)/2]
        new_population = []
        for _ in range(population_size):
            p1 = random.choice(parents)
            p2 = random.choice(parents)
            child = (p1 + p2)/2
            if random.random() < mutation_rate:
                child += random.gauss(0, 0.1)
            new_population.append(child)
        population = new_population
    return population

result = evolve(10, 0.1)
print(result)
```
