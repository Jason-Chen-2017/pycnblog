import pandas as pd

# 创新文化指数计算
def calculate_innovation_culture_score(data):
    weights = [0.4, 0.3, 0.2, 0.1]
    scores = []
    for row in data.itertuples():
        score = sum(w * v for w, v in zip(weights, row[1:]))
        scores.append(score)
    return scores

# 示例数据
data = pd.DataFrame({
    '创新意识': [8, 7, 9],
    '创新机制': [7, 8, 6],
    '创新氛围': [6, 5, 7],
    '创新激励': [5, 6, 4]
})

print(scores)
