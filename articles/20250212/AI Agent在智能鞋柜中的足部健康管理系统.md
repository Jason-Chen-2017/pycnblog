# 示例代码：数据预处理
import pandas as pd
import numpy as np

def preprocess_data(data):
    # 去除缺失值
    data = data.dropna()
    # 标准化处理
    data[['压力', '温度', '湿度']] = data[['压力', '温度', '湿度']].apply(lambda x: (x - x.mean()) / x.std())
    return data

# 示例代码：机器学习模型训练
from sklearn.svm import SVC
from sklearn.model_selection import train_test_split

X = data[['压力', '温度', '湿度']]
y = data['健康状态']

X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2)
model = SVC()
model.fit(X_train, y_train)

### 11.1 数据预处理公式
$$
\text{标准化处理：} \quad z = \frac{x - \mu}{\sigma}
$$

其中，$\mu$ 表示数据的均值，$\sigma$ 表示数据的标准差。

### 11.2 健康评估公式
$$
\text{健康评估分数：} \quad S = w_1 \cdot x_1 + w_2 \cdot x_2 + \cdots + w_n \cdot x_n
$$

其中，$w_i$ 表示各健康指标的权重，$x_i$ 表示各健康指标的值。
