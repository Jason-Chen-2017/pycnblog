### 5.2.1 数据预处理
```python
import pandas as pd
import numpy as np

def preprocess_data(stock_data, medical_data):
    # 数据清洗
    stock_data.dropna(inplace=True)
    medical_data.dropna(inplace=True)
    
    # 合并数据
    merged_data = pd.merge(stock_data, medical_data, on='date')
    return merged_data

# 示例数据加载
stock_data = pd.read_csv('em_stock.csv')
medical_data = pd.read_csv('medical.csv')

# 数据预处理
merged_data = preprocess_data(stock_data, medical_data)
```

### 5.2.2 互动机制实现
```python
from sklearn.tree import DecisionTreeRegressor
from sklearn.model_selection import train_test_split

def stock_valuation_model(features, target):
    X = features
    y = target
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2)
    model = DecisionTreeRegressor()
    model.fit(X_train, y_train)
    return model

# 示例应用
features = merged_data[['market_cap', 'return_on_equity']]
target = merged_data['stock_price']
```
