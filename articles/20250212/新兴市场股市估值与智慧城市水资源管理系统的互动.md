```python
import pandas as pd
import numpy as np

def calculate_valuation(stock_data):
    stock_data['市盈率'] = stock_data['股价'] / stock_data['每股收益']
    return stock_data

# 示例数据
data = {'股价': [100, 200, 300], '每股收益': [10, 20, 30}
stock_data = pd.DataFrame(data)
print(calculate_valuation(stock_data))
```

#### 5.2.2 水资源管理代码
```python
from sklearn.linear_model import LinearRegression

def optimize_water_management(sensor_data):
    model = LinearRegression()
    model.fit(sensor_data[['流量', '时间']], sensor_data['水量'])
    return model.predict(sensor_data[['流量', '时间']])

# 示例数据
data = {'流量': [10, 20, 30], '时间': [1, 2, 3], '水量': [50, 60, 70}
sensor_data = pd.DataFrame(data)
print(optimize_water_management(sensor_data))
```
