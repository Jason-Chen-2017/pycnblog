### 4.2 层次分析法的实现代码

```python
def ahp(weight_matrix, alternatives):
    n = len(weight_matrix)
    # 计算一致性检查
    consistency = (n - 1) * sum(weight_matrix[i][i] for i in range(n))
    # 计算特征向量
    eigenvector = [sum(weight_matrix[i] / weight_matrix[i][i] for i in range(n)) for _ in range(n)]
    # 计算一致性比率
    cr = (max(eigenvector) - min(eigenvector)) / sum(eigenvector)
    return cr

# 示例：权重矩阵
weight_matrix = [
    [1, 2, 3],
    [2, 1, 1],
    [1, 3, 1]
]
cr = ahp(weight_matrix, ['A', 'B', 'C'])
print(f"一致性比率：{cr}")
```

    for copyright in copyrights:

    return value
