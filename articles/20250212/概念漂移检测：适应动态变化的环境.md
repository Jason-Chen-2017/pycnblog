  ```python
  import numpy as np
  from sklearn.metrics import accuracy_score

  def detect_drift(current_data, previous_data, threshold=0.05):
      # 计算KL散度
      current_dist = np.bincount(current_data, minlength=len(previous_dist))
      kl_div = np.sum(np.where(previous_dist != 0, previous_dist * np.log(previous_dist / current_dist), 0))
      if kl_div > threshold:
          return True
      else:
          return False

  # 示例数据
  previous_data = np.random.normal(0, 1, 100)
  current_data = np.random.normal(1, 1, 100)
  threshold = 0.1

  # 检测漂移
  if detect_drift(current_data, previous_data, threshold):
      print("检测到概念漂移！")
  else:
      print("未检测到概念漂移。")
  ```
