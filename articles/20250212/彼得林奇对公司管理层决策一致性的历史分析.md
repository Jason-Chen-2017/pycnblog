```python
import pandas as pd
from sklearn.metrics import accuracy_score

def calculate_consistency_score(decision_logs):
    consistent_decisions = 0
    total_decisions = len(decision_logs)
    for log in decision_logs:
        if log['decision'] == log['previous_decision']:
            consistent_decisions += 1
    return consistent_decisions / total_decisions * 100

# 示例数据
decision_logs = [
    {'decision': 'buy', 'previous_decision': 'buy'},
    {'decision': 'hold', 'previous_decision': 'hold'},
    {'decision': 'sell', 'previous_decision': 'sell'},
    {'decision': 'buy', 'previous_decision': 'sell'}
]

score = calculate_consistency_score(decision_logs)
print(f"一致性评分：{score}%")
```
