      matched_rules = [rule for rule in rules if matches(rule, inputs)]
      if not matched_rules:
          return default_decision()
      # 解决冲突，选择优先级最高的规则
      matched_rules.sort(key=lambda x: x.priority)
      return matched_rules[0].action
