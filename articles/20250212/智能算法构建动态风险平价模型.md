import numpy as np

def dynamic_risk_parity(returns, window=20):
    # 计算协方差矩阵
    cov_matrix = np.cov(returns.T)
    # 计算权重
    inv_cov = np.linalg.inv(cov_matrix)
    weights = inv_cov / np.sum(inv_cov, axis=0)
    return weights
