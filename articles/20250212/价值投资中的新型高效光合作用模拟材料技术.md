```python
def calculate_efficiency(material_params):
    band_gap = material_params['band_gap']
    surface_area = material_params['surface_area']
    efficiency = band_gap * surface_area / 100
    return efficiency

material_params = {
    'band_gap': 1.5,
    'surface_area': 200
}

result = calculate_efficiency(material_params)
print(f"计算效率为: {result}%")
```

```python
    'band_gap': [1.5, 1.6, 1.7],
    'surface_area': [200, 250, 300]
```
