```python
import numpy as np
import open3d as o3d

def read_point_cloud(file_path):
    # 读取点云数据
    point_cloud = o3d.io.read_point_cloud(file_path)
    return point_cloud

def downsample_point_cloud(point_cloud, voxel_size=0.05):
    # 点云降噪
    point_cloud_down = point_cloud.voxel_down_sample(voxel_size)
    return point_cloud_down

# 使用示例
point_cloud = read_point_cloud("input.pcd")
point_cloud_down = downsample_point_cloud(point_cloud)
o3d.visualization.draw([point_cloud_down])
```
