- **5.2.2 模型训练代码**
  ```python
  import torch
  import torch.nn as nn
  import torch.optim as optim

  # 定义模型
  class InvestmentReportGenerator(nn.Module):
      def __init__(self, input_dim, hidden_dim, output_dim):
          super(InvestmentReportGenerator, self).__init__()
          self.embedding = nn.Embedding(input_dim, hidden_dim)
          self.decoder = nn.LSTM(hidden_dim, hidden_dim, 1)
          self.fc = nn.Linear(hidden_dim, output_dim)

      def forward(self, x):
          embedded = self.embedding(x)
          output, (h, c) = self.decoder(embedded, None)
          output = self.fc(output[:, -1])
          return output

  # 初始化模型
  model = InvestmentReportGenerator(input_dim, hidden_dim, output_dim)
  # 定义损失函数
  criterion = nn.CrossEntropyLoss()
  # 定义优化器
  optimizer = optim.Adam(model.parameters(), lr=0.001)
  ```
