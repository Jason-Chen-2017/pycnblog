#### 5.3 代码示例
```python
import pandas as pd

def calculate_roe(net_income, average_equity):
    return net_income / average_equity

def calculate_ev(pe, revenue, operating_margin):
    return pe * revenue * operating_margin * 2.5

# 示例数据
net_income = 100
average_equity = 500
pe = 15
revenue = 500
operating_margin = 0.2

roe = calculate_roe(net_income, average_equity)
ev = calculate_ev(pe, revenue, operating_margin)

print(f"ROE: {roe}")
print(f"EV: {ev}")
```
