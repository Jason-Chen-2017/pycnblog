def calculate pricing_power_index(data):
    # 计算毛利率
    gross_margin = data['毛利率']
    # 计算品牌影响力
    brand_influence = data['品牌影响力']
    # 计算市场占有率
    market_share = data['市场占有率']
    
    # 定价权指数计算
    pp_index = (gross_margin * brand_influence * market_share) / (data['行业平均毛利率'] * data['行业平均市场占有率'])
    return pp_index
