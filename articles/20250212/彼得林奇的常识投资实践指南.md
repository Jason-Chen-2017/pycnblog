import pandas as pd

# 数据清洗
def data_cleaning(df):
    # 删除缺失值
    df.dropna(inplace=True)
    # 处理异常值
    df = df[df['收盘价'].between(df['收盘价'].quantile(0.05), df['收盘价'].quantile(0.95))]
    return df

# 示例数据
data = {'股票代码': ['AAPL', 'MSFT', 'GOOGL'], 
        '收盘价': [150, 300, 120]}
df = pd.DataFrame(data)
df_cleaned = data_cleaning(df)
print(df_cleaned)

### 6.2 核心实现

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

def compute_return(data):
    return data['收盘价'].pct_change().cumsum()

data = {'股票代码': ['AAPL', 'MSFT', 'GOOGL'], 
        '收盘价': [150, 300, 120]}
df = pd.DataFrame(data)
returns = compute_return(df)
plt.plot(returns)
plt.title('投资回报率')
plt.show()

## 第7章: 投资系统的最佳实践与注意事项

### 7.1 最佳实践 Tips

1. **长期投资**：避免频繁交易，降低交易成本。
2. **分散投资**：不要将所有资金投入单一资产。
3. **持续学习**：关注市场动态，更新投资策略。

### 7.2 小结

彼得林奇的“常识投资”策略强调基本面分析和长期投资，通过系统的数据处理和模型构建，投资者可以更有效地做出投资决策。

### 7.3 注意事项

- **市场波动**：市场波动可能导致短期亏损，需耐心持有。
- **风险管理**：设置止损点，避免重大损失。
