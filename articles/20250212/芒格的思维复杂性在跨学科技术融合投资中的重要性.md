def m_g_complexity_analysis(data):
    # 数据预处理
    preprocessed_data = data.dropna().std()
    # 多维度分析
    importances = preprocessed_data.sort_values(ascending=False)
    # 投资策略生成
    strategy = [importances.index[0], importances.iloc[0]]
    return strategy
