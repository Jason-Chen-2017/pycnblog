```python
class InstitutionInvestor:
    def __init__(self, id, capital):
        self.id = id
        self.capital = capital
        self.strategy = "momentum"
    
    def make_decision(self, market_data):
        if self.strategy == "momentum":
            return self.momentum_strategy(market_data)
        else:
            return self.value_strategy(market_data)
    
    def momentum_strategy(self, market_data):
        # 简单动量策略：价格上涨则买入，反之卖出
        if market_data['price'] > market_data['price_prev']:
            return 'buy'
        else:
            return 'sell'
```

- 交易模拟模块：
```python
class TradingSimulator:
    def __init__(self, market_data, investors):
        self.market_data = market_data
        self.investors = investors
    
    def simulate(self):
        for investor in self.investors:
            decision = investor.make_decision(self.market_data)
            self.execute_trade(investor, decision)
    
    def execute_trade(self, investor, decision):
        if decision == 'buy':
            investor.capital -= self.market_data['price']
            # 更新市场数据
            self.market_data['price'] += 0.01
        elif decision == 'sell':
            investor.capital += self.market_data['price']
            # 更新市场数据
            self.market_data['price'] -= 0.01
```
