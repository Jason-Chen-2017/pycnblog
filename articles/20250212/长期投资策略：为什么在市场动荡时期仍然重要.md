import numpy as np
import pandas as pd

def portfolio_optimization(returns, risk_free_rate):

    Cov_matrix = returns.cov()
    mu = returns.mean().values
    # 使用均值-方差优化模型
    from pypkg import portfoliooptimizer
    port = portfoliooptimizer.Portfolio(mu, Cov_matrix)
    port.minimize_variance(target_return=0.05)
    weights = port.get_weights()
    return weights

def rebalance_portfolio(current_weights, target_weights, returns):
    # 计算当前投资组合的市场价值
    current_value = sum(current_weights * returns.mean())
    # 计算目标投资组合的市场价值
    target_value = sum(target_weights * returns.mean())
    # 计算再平衡所需的买卖金额
    buy_amount = target_weights - current_weights
    sell_amount = current_weights - target_weights
    return buy_amount, sell_amount
