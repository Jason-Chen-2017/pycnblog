```python
import numpy as np

def calculate_portfolio_return(portfolio_weights, expected_returns):
    return np.dot(portfolio_weights, expected_returns)

def calculate_portfolio_risk(portfolio_weights, covariance_matrix):
    return np.sqrt(np.dot(portfolio_weights.T, np.dot(covariance_matrix, portfolio_weights)))

# 示例数据
portfolio_weights = [0.4, 0.3, 0.3]
expected_returns = [0.1, 0.12, 0.08]
covariance_matrix = np.array([[0.02, 0.01, 0.01],
                              [0.01, 0.03, 0.01],
                              [0.01, 0.01, 0.02]])

return_ = calculate_portfolio_return(portfolio_weights, expected_returns)
risk = calculate_portfolio_risk(portfolio_weights, covariance_matrix)

print(f"预期收益率：{return_}%")
print(f"投资组合风险：{risk}%")
```

    return current_weights
