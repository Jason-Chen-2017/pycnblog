def calculate_transparency_score(records):
    criteria = ['全面性', '准确性', '及时性']
    scores = []
    for record in records:
        score = 0
        for c in criteria:
            if c in record:
                score += 1
        scores.append(score)
    transparency_score = sum(scores) / len(scores)
    return transparency_score

# 示例数据

    "财务报告准确，信息全面",
    "管理层会议记录及时发布",
    "公告信息不完整"
]
