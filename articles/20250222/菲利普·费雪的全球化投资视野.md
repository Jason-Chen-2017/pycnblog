### 附录A：数学模型与公式

#### A.1 成长股筛选的数学模型
$$ \text{ROE} = \frac{\text{净利润}}{\text{股东权益}} $$
$$ \text{增长率} = \frac{\text{今年净利润} - \text{去年净利润}}{\text{去年净利润}} \times 100\% $$

#### A.2 投资组合优化的数学模型
$$ \text{风险调整后收益} = \text{收益} - \text{风险} \times \text{风险溢价} $$

#### B.1 成长股筛选代码
```python
import pandas as pd

# 加载数据
df = pd.read_csv('stock_data.csv')

# 计算ROE和增长率
df['ROE'] = df['净利润'] / df['股东权益']
df['增长率'] = ((df['净利润'] - df['净利润'].shift(1)) / df['净利润'].shift(1)) * 100

# 筛选成长股
成长股 = df[(df['ROE'] > 0.15) & (df['增长率'] > 20)]
```

#### B.2 投资组合优化代码
```python
import numpy as np

# 定义风险函数
def calculate_risk(weights, cov_matrix):
    return np.sqrt(np.dot(weights.T, np.dot(cov_matrix, weights)))

# 定义收益函数
def calculate_return(weights, returns):
    return np.dot(weights.T, returns)

# 最优化问题
from scipy.optimize import minimize

def optimize_portfolio(returns, cov_matrix, method='min_var'):
    n = returns.shape[1]
    if method == 'min_var':
        # 最小化方差
        def objective(weights):
            return calculate_risk(weights, cov_matrix)
        constraints = [{'type': 'eq', 'fun': lambda w: np.sum(w) - 1}]
    elif method == 'max_sharpe':
        # 最大化夏普比率
        def objective(weights):
            return -calculate_return(weights, returns) / calculate_risk(weights, cov_matrix)
        constraints = [{'type': 'eq', 'fun': lambda w: np.sum(w) - 1}, {'type': 'eq', 'fun': lambda w: np.mean(returns.T.dot(w)) - 0.05}]
    else:
        return None

    result = minimize(objective, np.ones(n) / n, method='SLSQP', constraints=constraints)
    return result.x
```
