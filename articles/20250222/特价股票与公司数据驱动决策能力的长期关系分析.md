```python
import pandas as pd
import numpy as np

# 创建示例数据
data = pd.DataFrame({
    'date': pd.date_range('2020-01-01', '2020-01-31', 31),
    'price': np.random.uniform(100, 200, 31),
    'volume': np.random.randint(1000, 10000, 31)
})

# 检查缺失值
print(data.isnull().sum())

# 处理缺失值：填充法
data_filled = data.fillna(method='ffill')

# 处理异常值：使用z-score方法检测异常
z = np.abs((data['price'] - data['price'].mean()) / data['price'].std())
data['price'] = data['price'][z < 1.5]
```
