import numpy as np

def dcfvaluation(cash_flows, discount_rate):
    present_value = sum(cash_flows / (1 + discount_rate) ** i for i in range(len(cash_flows)))
    return present_value

# 示例数据
cash_flows = [100, 200, 300, 400]
discount_rate = 0.1

valuation = dcfvaluation(cash_flows, discount_rate)
print(f"DCF 估值：{valuation}")
