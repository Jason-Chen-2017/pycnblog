def calculate_intrinsic_value(asset_price, earnings_growth_rate, discount_rate):
    # 计算内在价值
    return asset_price / (1 + earnings_growth_rate) / (1 + discount_rate)

def greaham_strategy(asset_data, margin_of_safety):
    selected_assets = []
    for asset in asset_data:
        intrinsic_value = calculate_intrinsic_value(asset['price'], asset['earnings_growth'], asset['discount_rate'])
        if asset['price'] < intrinsic_value * (1 - margin_of_safety):
            selected_assets.append(asset)
    return selected_assets

# 示例数据
asset_data = [
    {'price': 100, 'earnings_growth': 0.1, 'discount_rate': 0.05},
    {'price': 80, 'earnings_growth': 0.15, 'discount_rate': 0.03}
]

    function calculateIntrinsicValue(uint assetPrice, uint earningsGrowth, uint discountRate) public pure returns (uint) {
        return assetPrice / (1 + earningsGrowth) / (1 + discountRate);
    }

    function selectAssets(address[] memory assets, uint marginOfSafety) public returns (address[] memory) {
        address[] memory selected = new address[](assets.length);
        for (uint i = 0; i < assets.length; i++) {
            uint intrinsicValue = calculateIntrinsicValue(getAssetPrice(assets[i]), getEarningsGrowth(assets[i]), getDiscountRate(assets[i]));
            if (getAssetPrice(assets[i]) < intrinsicValue * (1 - marginOfSafety)) {
                selected[i] = assets[i];
            }
        }
        return selected;
    }
}
