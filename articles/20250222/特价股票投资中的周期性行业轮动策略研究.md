```python
# 示例代码：计算行业轮动信号
def calculate_signals(data, window=12):
    # 计算市盈率
    pe = data['price'] / data['eps']
    # 计算动量指标
    momentum = data['price'].rolling(window).mean()
    # 生成信号
    signals = (pe < pe.rolling(window).quantile(0.2)) & (momentum > momentum.rolling(window).quantile(0.8))
    return signals
```
