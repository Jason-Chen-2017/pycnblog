    ```python
    import numpy as np
    from scipy.cluster.hierarchy import fcluster
    from scipy.spatial.distance import pdist, squareform

    # 示例数据矩阵
    X = np.array([[2, 3], [3, 5], [4, 7]])

    # 计算距离矩阵
    distances = pdist(X)
    distances_matrix = squareform(distances)

    # 应用层次分析法进行聚类
    clusters = fcluster(distances_matrix, t=0.5, criterion='distance')

    print(clusters)
    ```
