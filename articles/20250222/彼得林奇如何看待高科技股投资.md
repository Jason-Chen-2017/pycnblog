import pandas as pd
import numpy as np

def calculate_growth_rate(revenue_data):
    growth_rates = []
    for i in range(1, len(revenue_data)):
        growth_rate = (revenue_data[i] - revenue_data[i-1]) / revenue_data[i-1]
        growth_rates.append(growth_rate)
    return growth_rates

# 示例数据：某高科技公司过去5年的收入数据（单位：百万美元）
revenue = [100, 120, 150, 180, 200]

growth_rates = calculate_growth_rate(revenue)
print("收入增长率：", growth_rates)

# 计算平均增长率
average_growth = np.mean(growth_rates)
print("平均收入增长率：", average_growth)
