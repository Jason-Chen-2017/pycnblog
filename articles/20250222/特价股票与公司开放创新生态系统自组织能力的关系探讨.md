import pandas as pd
import numpy as np

def calculate_rsi(data, period=14):
    # 计算RSI指标
    diff = data['收盘价'] - data['开盘价']
    up = diff.where(diff > 0, 0)
    down = -diff.where(diff < 0, 0)
    rs = up.rolling(period).mean() / down.rolling(period).mean()
    rsi = 100 - 100 * (1 - rs)
    return rsi

data = pd.read_csv('stock_data.csv')
rsi = calculate_rsi(data)
print(rsi)

import networkx as nx

def optimize_ecosystem(partners, resources):
    # 构建合作伙伴网络
    G = nx.Graph()
    G.add_nodes_from(partners)
    for i in range(len(partners)):
        for j in range(i+1, len(partners)):
            G.add_edge(partners[i], partners[j])
    # 计算网络中心性
    centrality = nx.betweenness_centrality(G)
    # 选择中心性最高的合作伙伴
    top_partners = sorted(partners, key=lambda x: centrality[x], reverse=True)[:3]
    return top_partners

partners = ['供应商A', '供应商B', '客户C', '开发者D']
top_partners = optimize_ecosystem(partners, resources)
print(top_partners)
