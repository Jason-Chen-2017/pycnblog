## 第7章: 项目实战与案例分析

### 7.1 项目实战

#### 7.1.1 环境安装与配置
安装必要的软件和库，如Python、NumPy、Scipy等。

#### 7.1.2 系统核心实现源代码
```python
import numpy as np

def objective_function(x):
    return (x[0] - 3)**2 + (x[1] - 4)**2

def constraint_func(x):
    return x[0] * x[1] >= 12

def optimize():
    # 初始化种群
    population = np.random.rand(20, 2)
    for _ in range(50):
        # 计算适应度
        fitness = np.array([objective_function(x) for x in population])
        # 进行选择
        selected = fitness.argsort()[:10]
        # 进行交叉
        crossed = np.zeros((20, 2))
        for i in range(10):
            for j in range(10, 20):
                crossed[j] = np.where(np.random.rand(2) < 0.5, population[selected[i]], population[selected[i]])
        # 进行变异
        mutated = crossed + np.random.rand(20, 2) * 0.1
        # 更新种群
        population = mutated
    # 评估结果
    best = population[np.argmin([objective_function(x) for x in population])]
    print("最优解为:", best)
    print("最优解适应度为:", objective_function(best))

optimize()
```
