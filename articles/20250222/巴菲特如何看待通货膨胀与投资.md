    from scipy.optimize import minimize

    def optimize_portfolio(weights, returns, inflation_rate):
        # 定义目标函数
        def objective(weights):
            return -weights.dot(returns.dot(weights.T))
        # 约束条件
        constraints = [
            {'type': 'eq', 'fun': lambda w: np.sum(w) - 1},
            {'type': 'ineq', 'fun': lambda w: w >= 0}
        ]
        result = minimize(objective, np.ones(len(weights)) / len(weights), constraints)
        return result.x
