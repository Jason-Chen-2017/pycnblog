import pandas as pd
import numpy as np

# 示例：计算市盈率
def calculate_P_E(stock_price, earnings_per_share):
    return stock_price / earnings_per_share

# 示例数据
stock_prices = [100, 90, 80, 70, 60]
earnings_per_share = 10

# 计算市盈率
pe_ratio = [calculate_P_E(price, earnings_per_share) for price in stock_prices]
print("市盈率:", pe_ratio)
