def asset_allocation(weights, returns):
    # 计算资产配置的期望收益
    expected_return = sum(w * r for w, r in zip(weights, returns))
    return expected_return

# 示例权重和收益
weights = [0.4, 0.3, 0.2, 0.1]
returns = [0.08, 0.05, 0.07, 0.03]
allocation_return = asset_allocation(weights, returns)
print(f"资产配置的期望收益为：{allocation_return}")
```

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

def calculate_expected_return(weights, returns):
    return np.dot(weights, returns)

def calculate_risk(weights, covariance_matrix):
    return np.sqrt(np.dot(weights.T, np.dot(covariance_matrix, weights)))

# 示例数据
returns = pd.Series([0.08, 0.05, 0.07, 0.03])
covariance_matrix = pd.DataFrame({
    'A': [0.02, 0.01, 0.01, 0.005],
    'B': [0.01, 0.03, 0.02, 0.01],
    'C': [0.01, 0.02, 0.05, 0.01],
    'D': [0.005, 0.01, 0.01, 0.02]
})

weights = np.array([0.4, 0.3, 0.2, 0.1])
expected_return = calculate_expected_return(weights, returns)
risk = calculate_risk(weights, covariance_matrix.values)

print(f"期望收益：{expected_return}")
print(f"风险：{risk}")
```
