import numpy as np

# 创建量子电路
qc = QuantumCircuit(2, 2)
qc.h(0)
qc.cx(0, 1)
qc.measure([0, 1], [1, 0])

# 执行量子电路

result = execute(qc, backend).result()
counts = result.get_counts()
print(counts)
