def logistic_regression(data, alpha, iterations):
    X = [[1, x[0]] for x in data]
    y = [x[1] for x in data]
    theta = [0, 0]
    
    for _ in range(iterations):
        h = numpy.dot(X, theta)
        errors = y - h
        theta[0] -= alpha * (1/numpy.sum(h * errors))
        theta[1] -= alpha * (1/numpy.sum(X * errors))
    
    return theta
