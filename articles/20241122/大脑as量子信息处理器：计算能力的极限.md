```python
# 导入所需库
import qiskit

# 初始化量子比特
quantum_bits = qiskit.QuantumCircuit(5)

# 应用量子门创建连接
quantum_bits.h(0)  # 初始化第一个量子比特为超置态
for i in range(4):
    quantum_bits.cx(0, i+1)  # 应用控制非门

# 执行量子测量
quantum_bits.measure_all()

# 运行量子计算
backend = qiskit.Aer.get_backend('qasm_simulator')
result = qiskit.execute(quantum_bits, backend).result()

# 输出结果
print(result.get_counts(quantum_bits))
```
