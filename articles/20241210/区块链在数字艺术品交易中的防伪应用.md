### 4.4 数学模型和数学公式

数字签名算法通常基于公钥加密体制，以下是一个基本的数学模型：

$$
\begin{aligned}
& \text{私钥} \: sk = (p, q, d), \\
& \text{公钥} \: pk = (n, e), \\
& \text{消息} \: m, \\
& \text{签名} \: sig = (r, s), \\
& \text{步骤}: \\
& (1) \: H(m) \: \text{为消息的哈希值}, \\
& (2) \: \text{选择随机数} \: k, \\
& (3) \: r = (k * n) \mod p, \\
& (4) \: s = (d * (H(m) + r * e)) \mod n, \\
& (5) \: \text{签名} \: sig = (r, s).
\end{aligned}
$$

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract DigitalArtContract {
    mapping(address => mapping(bytes32 => bool)) public signatures;

    function signArtwork(bytes32 artworkHash) public {
        require(signatures[msg.sender][artworkHash] == false, "Artwork already signed");
        signatures[msg.sender][artworkHash] = true;
    }

    function verifySignature(bytes32 artworkHash, bytes32 signature) public view returns (bool) {
        bytes32 messageHash = keccak256(abi.encodePacked(artworkHash));
        return recoverSignature(messageHash, signature) == msg.sender;
    }

    function recoverSignature(bytes32 messageHash, bytes32 signature) public pure returns (address) {
        (bytes32 r, bytes32 s, uint8 v) = splitSignature(signature);
        return ecrecover(messageHash, v, r, s);
    }

    function splitSignature(bytes32 signature) public pure returns (bytes32 r, bytes32 s, uint8 v) {
        return (signature >> 192, signature >> 128, signature >> 64);
    }
}
```
