1. **创建智能合约：**

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract ArtMarket {
    mapping(uint256 => address) public artIds;
    mapping(address => mapping(uint256 => uint256)) public artistBalances;

    function registerArt(uint256 artId, string memory uri) public {
        require(artIds[artId] == address(0), "Art already registered");
        artIds[artId] = msg.sender;
        artistBalances[msg.sender()][artId] += 1;
    }

    function buyArt(uint256 artId) public payable {
        require(artIds[artId] != address(0), "Art not registered");
        address artist = artIds[artId];
        artistBalances[artist][artId] -= 1;
        artistBalances[msg.sender()][artId] += 1;
        payable(artist).transfer(msg.value);
    }
}
```

2. **部署智能合约：**

```bash
truffle init
truffle migrate --network development
```

const web3 = new Web3('http://localhost:8545');
const ArtMarket = new web3.eth.contract(artMarketABI);
const artMarketInstance = ArtMarket.at(artMarketAddress);

// 提交艺术品

    console.log(result);
