def analyze_language(material):
    limitations = "自然、热爱、情感"
    return limitations

# 语言限制判断
# 根据分析结果，判断语言限制
def judge_limitations(limitations):
    if "情感" in limitations:
        return "情感限制"
    else:
        return "无情感限制"

# 调整语言表达
# 根据判断结果，调整语言表达
def adjust_expression(expression, limitation):
