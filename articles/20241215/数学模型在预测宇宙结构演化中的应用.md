import numpy as np
import pandas as pd
from sklearn.preprocessing import StandardScaler
from sklearn.linear_model import LinearRegression

# 读取观测数据
data = pd.read_csv('observation_data.csv')

# 数据预处理
def preprocess_data(data):
    # 数据清洗
    data = data.dropna()
    # 数据归一化
    scaler = StandardScaler()
    normalized_data = scaler.fit_transform(data[['redshift', 'distance']])
    return normalized_data

# 特征提取
def extract_features(data):
    features = data[['redshift']]
    return features

# 模型训练
def train_model(features, labels):
    model = LinearRegression()
    model.fit(features, labels)
    return model

# 预测膨胀速度
def predict_expansion_speed(model, new_data):
    normalized_data = preprocess_data(new_data)
    features = extract_features(normalized_data)
    prediction = model.predict(features)
    return prediction

# 输出结果
def output_results(prediction):
    print(f'预测的膨胀速度：{prediction}')

# 主函数
def main():
    # 加载数据
    data = pd.read_csv('observation_data.csv')
    # 分割数据集
    features = extract_features(data)
    labels = data['distance']
    # 训练模型
    model = train_model(features, labels)
    # 预测膨胀速度
    new_data = pd.DataFrame({'redshift': [0.1]})
    prediction = predict_expansion_speed(model, new_data)
    # 输出结果
    output_results(prediction)

if __name__ == '__main__':
    main()
