```python
import numpy as np
import matplotlib.pyplot as plt
from mermaid import Mermaid

def calculate_time_complexity(n, Lambda):
    # 计算时间复杂度
    T_n = n * Lambda * np.exp(-Lambda * n)
    return T_n

def plot_time_complexity(Lambdas, n):
    # 绘制时间复杂度曲线
    T_values = [calculate_time_complexity(n, Lambda) for Lambda in Lambdas]
    plt.plot(Lambdas, T_values, label='Time Complexity')
    plt.xlabel('Lambda')
    plt.ylabel('T(n)')
    plt.title('Time Complexity as a Function of Lambda')
    plt.legend()
    plt.show()

# 示例数据
Lambdas = np.linspace(0.01, 1, 100)
n = 1000

# 绘制时间复杂度曲线
plot_time_complexity(Lambdas, n)
```
