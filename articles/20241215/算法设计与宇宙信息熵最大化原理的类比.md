$$
H(X) = -\sum_{i} p(x_i) \log_2 p(x_i)

```python
import numpy as np
import matplotlib.pyplot as plt

# 信息熵计算函数
def entropy(p):
    return -np.sum(p * np.log2(p))

# 搜索算法实现
def search_algorithm(data, k):
    # 初始化概率分布
    p = np.ones(len(data)) / len(data)
    # 计算初始信息熵
    H = entropy(p)
    # 存储最佳信息熵和索引
    best_entropy = H
    best_index = 0
    # 进行k次迭代
    for _ in range(k):
        # 根据当前概率分布重新计算信息熵
        new_p = np.log(data / np.sum(data))
        new_p = np.exp(new_p)
        new_entropy = entropy(new_p)
        # 如果新信息熵更高，更新概率分布
        if new_entropy > best_entropy:
            best_entropy = new_entropy
            best_index = _
            p = new_p
    return p, best_index

# 数据示例
data = np.array([1, 2, 3, 4, 5])

# 执行搜索算法
p, best_index = search_algorithm(data, 5)

# 打印结果
print(f"Optimized probability distribution: {p}")
print(f"Best index: {best_index}")

# 绘制信息熵变化图
H_values = [entropy(p) for p in data]
plt.plot(data, H_values, marker='o')
plt.xlabel('Index')
plt.ylabel('Entropy')
plt.title('Entropy Change during Search Algorithm')
plt.show()
```
