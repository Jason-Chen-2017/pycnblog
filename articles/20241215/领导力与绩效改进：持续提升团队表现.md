```python
# Python script to simulate transformational leadership
class Team:
    def __init__(self, members):
        self.members = members

    def motivate_members(self, leader_style):
        if leader_style == 'transformational':
            print("Transformational leadership is applied.")
            for member in self.members:
                print(f"{member} is motivated to perform better.")
        else:
            print("Traditional leadership style is applied.")

team = Team(['Alice', 'Bob', 'Charlie'])
team.motivate_members('transformational')
```
