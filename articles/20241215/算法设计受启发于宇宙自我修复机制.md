import numpy as np
import matplotlib.pyplot as plt

# 定义目标函数
def objective_function(x):
    return sum(x**2)

# 定义约束条件
def constraint_function(x):
    return np.sum(x) - 100

# 初始化参数
n = 10
x = np.random.uniform(-10, 10, n)

# 模拟演化过程
for i in range(1000):
    # 交互和碰撞

    x = np.clip(x, -10, 10)  # 约束条件

    alpha = 0.1
    x = x - alpha * (objective_function(x) - 100)

    # 优化目标调整
    if i % 100 == 0:
        x = x - 0.01 * (objective_function(x) - 100)

# 输出最优解
print("最优解：", x)
print("目标函数值：", objective_function(x))
