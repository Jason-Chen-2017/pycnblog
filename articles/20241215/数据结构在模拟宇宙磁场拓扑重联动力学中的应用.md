import numpy as np
import matplotlib.pyplot as plt

def simulate_magnetic_field(A, omega, phi, t_max=100, dt=0.1):
    t = np.arange(0, t_max, dt)
    phi_t = np.zeros_like(t)
    for i in range(len(omega)):
        phi_t += A[i] * np.sin(omega[i] * t + phi[i])
    return t, phi_t

def plot_magnetic_field(t, phi_t):
    plt.plot(t, phi_t)
    plt.xlabel('Time (s)')
    plt.ylabel('Magnetic Field (T)')
    plt.title('Magnetic Field Simulation')
    plt.show()

if __name__ == '__main__':
    A = [1, 2, 3]
    omega = [1, 2, 3]
    phi = [0, np.pi/2, np.pi]
    t, phi_t = simulate_magnetic_field(A, omega, phi)
    plot_magnetic_field(t, phi_t)
```

import matplotlib.pyplot as plt

def simulate_magnetic_field(A, omega, phi, t_max=100, dt=0.1):
    t = np.arange(0, t_max, dt)
    phi_t = np.zeros_like(t)
    for i in range(len(omega)):
        phi_t += A[i] * np.sin(omega[i] * t + phi[i])
    return t, phi_t

def plot_magnetic_field(t, phi_t):
    plt.plot(t, phi_t)
    plt.xlabel('Time')
    plt.ylabel('Magnetic Field')
    plt.title('Magnetic Field Simulation')
    plt.show()

if __name__ == '__main__':
    A = [1, 2, 3]
    omega = [1, 2, 3]
    phi = [0, np.pi/2, np.pi]
    t, phi_t = simulate_magnetic_field(A, omega, phi)
    plot_magnetic_field(t, phi_t)
```
