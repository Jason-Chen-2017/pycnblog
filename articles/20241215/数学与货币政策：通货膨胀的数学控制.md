$$
\Delta P = \frac{\Delta Q}{Q} - \frac{\Delta M}{M}

其中，$\Delta P$ 表示价格变动率，$\Delta Q$ 表示需求变动率，$\Delta M$ 表示货币供应量变动率，$Q$ 和 $M$ 分别为价格和货币供应量的初始水平。

IS-LM模型：
$$

\end{aligned}
$$
其中，$Y$ 表示国民收入，$C$ 和 $I$ 分别为消费和投资函数，$T$ 为税收，$r$ 为利率，$L$ 为货币需求函数，$M$ 为货币供应量。

$$

$$

$$

公开市场操作策略：
$$

其中，$M_t$ 表示第 $t$ 期的货币供应量，$M^*$ 表示目标货币供应量，$\beta$ 为调节参数。

$$

    cursor.execute('''INSERT INTO inflation_data (inflation_rate, unemployment_rate, monetary_supply, interest_rate) VALUES (?, ?, ?, ?)''', tuple(data.values()))

# 预测模型模块
def predict_inflation(data):
    # 加载数据
    X = np.array([[d['inflation_rate'], d['unemployment_rate'], d['monetary_supply']] for d in data])
    y = np.array([d['interest_rate'] for d in data])
    
    # 建立线性回归模型
    model = LinearRegression()
    model.fit(X, y)
    
    # 预测
    predicted_rate = model.predict(np.array([[3.0, 0.05, 1000.0]]))
    
    return predicted_rate[0]

# 策略制定模块
def make_policy(predicted_rate, target_rate):
    # 计算利率调整值
    adjustment = target_rate - predicted_rate
    
    # 制定利率、货币供应量和准备金率调整策略
    new_interest_rate = predicted_rate + adjustment
    new_monetary_supply = 1000.0 + adjustment * 0.1
    new_reserve_rate = 0.1 + adjustment * 0.05
    
    return new_interest_rate, new_monetary_supply, new_reserve_rate

# 策略执行模块
def execute_policy(new_interest_rate, new_monetary_supply, new_reserve_rate):
    # 更新数据库中的政策参数
    conn = sqlite3.connect('inflation_control.db')
    cursor = conn.cursor()
    
    cursor.execute('''UPDATE inflation_data SET interest_rate = ?, monetary_supply = ?, reserve_rate = ? WHERE id = 1''', (new_interest_rate, new_monetary_supply, new_reserve_rate))
    
    # 提交并关闭连接
    conn.commit()
    conn.close()
