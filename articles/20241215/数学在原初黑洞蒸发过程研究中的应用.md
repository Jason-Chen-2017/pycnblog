```python

import matplotlib.pyplot as plt
from scipy.integrate import odeint
from mermaidpy import Mermaid

# 黑洞蒸发速率模型
def evaporation_rate(M, t, sigma, c, k_B, T):
    return -sigma * c * M * T**3 / (3 * k_B * T**4)

# 温度变化模型
def T_equation(T, t, M, sigma, c, k_B):
    return evaporation_rate(M, t, sigma, c, k_B, T) / M

# 初始条件
M0 = 1e5 * 1.98847e30  # 初始质量（太阳质量的100万倍）
T0 = 1e8  # 初始温度（单位：K）
t_final = 1e7  # 时间终点（单位：秒）
t_steps = 1e4  # 时间步数

# 时间数组
t = np.linspace(0, t_final, t_steps)

# 解微分方程
T = odeint(T_equation, T0, t, args=(M0, sigma, c, k_B))

# 绘制温度随时间的变化
plt.plot(t / (1e9), T)
plt.xlabel('Time (Gyrs)')
plt.ylabel('Temperature (K)')
plt.title('Black Hole Evaporation')
plt.show()

# 输出Mermaid流程图
mermaid = Mermaid()
mermaid.add_diagram("""
graph TD
    A[Initial State] --> B[Evaporation Rate]
    B --> C[Temperature Change]
    C --> D[ODE Solver]
    D --> E[Time Evolution]
    E --> F[Plot Result]
""")
mermaid.save('black_hole_evaporation.mmd')
```
