def bubble_sort(arr):
    n = len(arr)
    for i in range(n):
        for j in range(0, n-i-1):
            if arr[j] > arr[j+1]:
                arr[j], arr[j+1] = arr[j+1], arr[j]
    return arr

arr = [64, 34, 25, 12, 22, 11, 90]

```python
import numpy as np
import matplotlib.pyplot as plt

def calculate_entropy(data):
    probabilities = np.bincount(data) / len(data)
    entropy = -np.sum(probabilities * np.log2(probabilities))
    return entropy

def calculate_entropy_rate(data, time_interval):
    initial_entropy = calculate_entropy(data[:-1])
    final_entropy = calculate_entropy(data[1:])
    entropy_rate = (final_entropy - initial_entropy) / time_interval
    return entropy_rate

# 生成随机数据
data = np.random.randint(0, 10, size=100)

# 计算信息熵增长率
time_interval = 1
entropy_rate = calculate_entropy_rate(data, time_interval)

print("Entropy rate:", entropy_rate)

# 可视化展示
plt.hist(data, bins=10, edgecolor='black')
plt.xlabel('Value')
plt.ylabel('Frequency')
plt.title('Data Distribution')
plt.show()
```
