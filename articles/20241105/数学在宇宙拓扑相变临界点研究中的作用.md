```python
import numpy as np

# 定义函数
def function(x):
    return x**2

# 计算导数
def derivative(f, x, h=1e-5):
    return (f(x+h) - f(x-h)) / (2*h)

# 计算积分
def integral(f, a, b, n=1000):
    h = (b-a) / n
    return sum(f(x) * h for x in np.linspace(a, b, n))

# 计算均值和方差
def mean_variance(data):
    mu = np.mean(data)
    sigma = np.std(data)
    return mu, sigma

# 使用实例
x = np.linspace(0, 1, 100)
y = function(x)

# 计算导数
dydx = derivative(function, x[0])

# 计算积分
I = integral(function, 0, 1)

# 计算均值和方差
mu, sigma = mean_variance(y)

print(f"导数：{dydx}")
print(f"积分：{I}")
print(f"均值：{mu}")
print(f"方差：{sigma}")
```

```python
import numpy as np
import matplotlib.pyplot as plt

# 定义函数
def f(x):
    return x**2

# 计算导数
def derivative(f, x, h=1e-5):
    return (f(x+h) - f(x-h)) / (2*h)

# 计算积分
def integral(f, a, b, n=1000):
    h = (b-a) / n
    return sum(f(x) * h for x in np.linspace(a, b, n))

# 计算均值和方差
def mean_variance(data):
    mu = np.mean(data)
    sigma = np.std(data)
    return mu, sigma

# 使用实例
x = np.linspace(0, 1, 100)
y = f(x)

# 计算导数
dydx = derivative(f, x[0])

# 计算积分
I = integral(f, 0, 1)

# 计算均值和方差
mu, sigma = mean_variance(y)

print(f"导数：{dydx}")
print(f"积分：{I}")
print(f"均值：{mu}")
print(f"方差：{sigma}")

# 绘制函数图像
plt.plot(x, y)
plt.xlabel('x')
plt.ylabel('f(x)')
plt.title('函数f(x)的图像')
plt.show()
```
