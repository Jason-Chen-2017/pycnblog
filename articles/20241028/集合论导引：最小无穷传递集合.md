```python
import networkx as nx
from networkx.algorithms import tree

def find_minimal_infinite_transitive_set(G):
    # 构建传递闭包图
    T = tree.transitive_closure(G)
    
    # 寻找包含所有节点的最小无穷传递集合
    minimal_set = None
    for node in T.nodes():
        if nx.is_tree(T.subgraph(T.nodes_of_degree(2)[0])))
        minimal_set = node
    
    return minimal_set

# 创建图
G = nx.Graph()
G.add_nodes_from([1, 2, 3, 4, 5])
G.add_edges_from([(1, 2), (2, 3), (3, 4), (4, 5)])

# 寻找最小无穷传递集合
minimal_set = find_minimal_infinite_transitive_set(G)

# 打印结果
print(f"最小无穷传递集合：{minimal_set}")
```

```python
    for i in range(2, int(n**0.5) + 1):
```
