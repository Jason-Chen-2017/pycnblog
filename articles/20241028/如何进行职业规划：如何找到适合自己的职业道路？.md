**核心算法原理讲解**

- **自我评估算法（SWOT分析法）：**

  function SWOTAnalysis(personality, strengths, weaknesses, opportunities, threats) {
      // 对个人性格、优势、劣势、机会和威胁进行评估
      // 返回一个综合评分
      let score = 0;
      
      // 性格评估
      score += personality.score;
      
      // 强势评估
      score += strengths.score;
      
      // 劣势评估
      score -= weaknesses.score;
      
      // 机会评估
      score += opportunities.score;
      
      // 威胁评估
      score -= threats.score;
      
      return score;
  }
