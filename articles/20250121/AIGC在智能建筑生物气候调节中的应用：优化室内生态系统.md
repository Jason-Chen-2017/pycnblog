import pandas as pd
import cv2

model = tf.keras.Sequential([

    tf.keras.layers.Dense(1)
])

# 编译模型
model.compile(optimizer='adam', loss='mse')

# 训练模型
model.fit(data[['temp', 'humidity', 'air_quality']], data['target'], epochs=100)

# 测试模型
test_data = pd.DataFrame({
