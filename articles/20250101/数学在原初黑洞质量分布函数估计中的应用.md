```python
import numpy as np
from scipy.stats import norm
import matplotlib.pyplot as plt

# 数据预处理
def preprocess_data(data):
    cleaned_data = np.array(data)
    cleaned_data = cleaned_data[~np.isnan(cleaned_data)]
    return cleaned_data

# 模型训练
def train_model(data):
    model = norm
    params = model.fit(data)
    return model, params

# 模型评估
def evaluate_model(model, data):
    predicted_data = model.pdf(data, *params)
    fit = predicted_data.sum()
    return fit

# 结果输出
def output_results(model, params, data):
    fitted_data = model.pdf(data, *params)
    cumulative_data = np.cumsum(fitted_data)
    plt.plot(data, fitted_data, 'ro', data, cumulative_data, 'b-')
    plt.xlabel('Mass')
    plt.ylabel('Probability')
    plt.show()

# 主函数
def main():
    data = np.random.normal(0, 1, 1000)
    cleaned_data = preprocess_data(data)
    model, params = train_model(cleaned_data)
    fit = evaluate_model(model, cleaned_data)
    output_results(model, params, cleaned_data)

if __name__ == "__main__":
    main()
```
