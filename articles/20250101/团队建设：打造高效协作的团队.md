class TeamMember:
    def __init__(self, name, role, responsibilities):
        self.name = name
        self.role = role
        self.responsibilities = responsibilities

team_members = [
    TeamMember("Alice", "Project Manager", ["规划项目进度", "协调团队成员"]),
    TeamMember("Bob", "Developer", ["编码实现", "单元测试"]),
    TeamMember("Charlie", "QA Engineer", ["测试用例编写", "缺陷追踪"]),
]

# 打印团队成员信息
for member in team_members:
    print(f"{member.name} ({member.role}): {', '.join(member.responsibilities)}")
