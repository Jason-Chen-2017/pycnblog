```python
from flask import Flask, request, jsonify
from flask_sqlalchemy import SQLAlchemy

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'mysql+pymysql://username:password@localhost/emotion Regulation'
db = SQLAlchemy(app)

class EmotionState(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    leader_id = db.Column(db.Integer, nullable=False)
    state = db.Column(db.String(50), nullable=False)
    timestamp = db.Column(db.DateTime, nullable=False)

@app.route('/emotion_state', methods=['POST'])
def add_emotion_state():
    data = request.get_json()
    state = EmotionState(
        leader_id=data['leader_id'],
        state=data['state'],
        timestamp=data['timestamp']
    )
    db.session.add(state)
    db.session.commit()
    return jsonify({'status': 'success'}), 201

@app.route('/emotion_state', methods=['GET'])
def get_emotion_state():
    leader_id = request.args.get('leader_id')
    states = EmotionState.query.filter_by(leader_id=leader_id).all()
    return jsonify([{'id': state.id, 'state': state.state, 'timestamp': state.timestamp} for state in states])

if __name__ == '__main__':
    db.create_all()
    app.run(debug=True)
```
