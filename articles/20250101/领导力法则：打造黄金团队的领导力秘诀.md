# 领导力反馈系统

class LeadershipFeedback:
    def __init__(self, leader, members):
        self.leader = leader
        self.members = members

    def collect_feedback(self):
        feedbacks = []
        for member in self.members:
            feedback = member.give_feedback(self.leader)
            feedbacks.append(feedback)
        return feedbacks

    def analyze_feedback(self):
        # 分析反馈，评估领导力表现
        feedbacks = self.collect_feedback()
        if all(feedback == "Positive" for feedback in feedbacks):
            return "Highly effective leader"
        elif any(feedback == "Negative" for feedback in feedbacks):
            return "Some areas for improvement"
        else:
            return "Average effectiveness"
