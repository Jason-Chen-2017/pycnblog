import numpy as np

def friedmann_equation(H, a):
    return H * a

def lemaître_equation(a, da, G, rho, p):
    return -G * (rho + 3 * p / np.c**2)

a_final = calculate_expansion(H0, a0, da, G, rho, p, N)
print("最终宇宙尺度:", a_final)

def calculate_expansion(H0, a0, da, G, rho, p, N):
    a = a0
    times = []
    scales = []
    for _ in range(N):
        H = friedmann_equation(H0, a)
        da = lemaître_equation(a, da, G, rho, p)
        a += da
        times.append(_ * da)
        scales.append(a)
    return times, scales

H0 = 100  # 哈勃参数
a0 = 1    # 宇宙尺度
da = 0.1  # 宇宙膨胀速率
G = 6.674 * 10**-11  # 引力常数
rho = 1   # 宇宙总密度

N = 1000  # 迭代次数

times, scales = calculate_expansion(H0, a0, da, G, rho, p, N)
