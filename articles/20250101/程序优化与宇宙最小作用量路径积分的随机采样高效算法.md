import numpy as np
import matplotlib.pyplot as plt
from mpi4py import MPI

# 参数设置
N = 1000  # 采样点数量
t1 = 0
t2 = 10
m = 1
V = lambda x: 0.5 * x**2

# 初始化随机数生成器
np.random.seed(42)

# 蒙特卡洛采样
def monte_carlo_sampler(N, t1, t2, m, V):
    t = np.linspace(t1, t2, N)
    q_samples = np.random.uniform(t1, t2, N)
    p_samples = np.random.uniform(-m, m, N)
    
    # 计算作用量
    I_samples = np.zeros(N)
    for i in range(N):
        q = q_samples[i]
        p = p_samples[i]
        L = 0.5 * p**2 - V(q)
        for dt in t[1:]:
            L += m * (p / np.sqrt(2 * np.pi * m * dt)) * np.exp(-0.5 * (q - q_samples[i])**2 / (m * dt))
        I_samples[i] = L
    
    return I_samples

# 并行计算
comm = MPI.Comm.world()
rank = comm.Get_rank()
size = comm.Get_size()

# 分布式计算
def parallel_monte_carlo_sampler(N, t1, t2, m, V):
    comm.Barrier()
    if rank == 0:
        q_samples = np.random.uniform(t1, t2, N)
        p_samples = np.random.uniform(-m, m, N)
        I_samples = np.zeros(N)
    else:
        q_samples = None
        p_samples = None
        I_samples = None
    
    comm.Bcast(q_samples, root=0)
    comm.Bcast(p_samples, root=0)
    
    if rank != 0:
        I_samples = monte_carlo_sampler(N, t1, t2, m, V)
    
    return I_samples

# 主函数
if __name__ == "__main__":
    comm = MPI.Comm.world()
    rank = comm.Get_rank()
    
    I_samples = parallel_monte_carlo_sampler(N, t1, t2, m, V)
    
    if rank == 0:
        # 统计分析
        I_mean = np.mean(I_samples)
        I_std = np.std(I_samples)
        print(f"I_mean: {I_mean}, I_std: {I_std}")
        
        # 可视化
        plt.hist(I_samples, bins=50, alpha=0.5)
        plt.xlabel("Integral Value")
        plt.ylabel("Frequency")
        plt.title("Histogram of Integral Values")
        plt.show()
