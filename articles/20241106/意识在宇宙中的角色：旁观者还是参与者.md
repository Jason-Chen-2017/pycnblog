```python
class QuantumComputer:
    def __init__(self, qubits):
        self.qubits = qubits
        self.registers = [0] * qubits

    def hadamard(self, qubit_index):
        # Hadamard gate
        self.registers[qubit_index] = 1 / math.sqrt(2) * (1 + self.registers[qubit_index])

    def cnot(self, control, target):
        # CNOT gate
        if self.registers[control]:
            self.registers[target] = 1 - self.registers[target]

    def measure(self):
        # Measure all qubits
        for i in range(len(self.registers)):
            self.registers[i] = int(random.random() > 0.5)

    def run_algorithm(self):
        # Run quantum algorithm
        self.hadamard(0)
        self.cnot(0, 1)
        self.measure()
```
