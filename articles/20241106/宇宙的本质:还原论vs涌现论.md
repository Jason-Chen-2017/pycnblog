def simulate_interaction(stars):

            star_i = stars[i]
            star_j = stars[j]
            distance = calculate_distance(star_i, star_j)
            if distance < 10:
                star_i.change_speed(-random.uniform(0.1, 0.5))
                star_j.change_speed(random.uniform(0.1, 0.5))

def calculate_distance(star1, star2):
    return ((star1.x - star2.x) ** 2 + (star1.y - star2.y) ** 2) ** 0.5

def main():
    stars = [
        Star(1.0, 10.0),
        Star(2.0, 20.0),
        Star(3.0, 30.0),
    ]
    for _ in range(100):
        simulate_interaction(stars)
    print(f"The final positions of the stars are: {stars}")
