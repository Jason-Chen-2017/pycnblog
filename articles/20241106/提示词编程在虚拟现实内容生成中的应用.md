import spacy
import numpy as np
import matplotlib.pyplot as plt

# 加载spaCy模型
nlp = spacy.load("en_core_web_sm")

# 用户输入自然语言指令
user_input = "create a sphere with radius 2 and green color"

# 处理自然语言指令
doc = nlp(user_input)
for token in doc:
    print(token.text, token.lemma_, token.pos_, token.dep_, token.head.text)

# 生成Unity脚本代码
def generate_unity_code(doc):
    code = ""
    for token in doc:
        if token.dep_ == "nsubj":
            code += f"CreateSphere({token.head.text}, {token.head.head.text});\n"
        elif token.dep_ == "amod":
            code += f"SetMaterialColor({token.head.text}, Color.green);\n"
    return code

unity_code = generate_unity_code(doc)
print(unity_code)

# 渲染场景
import Unity

unity = Unity.Unity()
unity.load_scene("Scene.unity")
unity.run_script(unity_code)
