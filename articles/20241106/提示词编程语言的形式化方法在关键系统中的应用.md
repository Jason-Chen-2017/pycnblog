- **可验证性**：形式化方法允许开发者对程序进行形式化验证，从而确保程序满足特定的属性要求。

```python
# elevator.py

from pyke import state_machine

class ElevatorSystem(state_machine.StateMachine):
    states = [
        ('idle', 'initial'),
        ('moving_up', 'moving'),
        ('moving_down', 'moving'),
        ('stopped', 'final'),
    ]

    transitions = [
        ('idle', 'moving_up', ['call_up']),
        ('idle', 'moving_down', ['call_down']),
        ('moving_up', 'stopped', ['stop']),
        ('moving_down', 'stopped', ['stop']),
        ('moving_up', 'moving_up', ['move_up']),
        ('moving_down', 'moving_down', ['move_down']),
        ('stopped', 'moving_up', ['call_up']),
        ('stopped', 'moving_down', ['call_down']),
    ]

    properties = [
        ('state', 'is_state'),
        ('direction', 'is_direction'),
    ]

    def is_state(s):
        return s in self.states

    def is_direction(d):
        return d in ['up', 'down']

def verify_system():
    system = ElevatorSystem()
    system.modelCheck(['initial'], 'stopped')
    print("System verified successfully!")

if __name__ == '__main__':
    verify_system()
```
