```python
def train_model(data, model, optimizer, loss_function, epochs):
    for epoch in range(epochs):
        for batch in data:
            optimizer.zero_grad()
            outputs = model(batch)
            loss = loss_function(outputs, batch)
            loss.backward()
            optimizer.step()
    return model
```
