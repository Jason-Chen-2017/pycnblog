```python
# 导入必要的库
import tensorflow as tf
import tensorflow_text as tf_text
from tensorflow.keras.layers import Embedding, LSTM, Dense

# 加载文本数据
text_data = tf_text.open("text_data.txt", "r").read()

# 预处理文本数据
tokenizer = tf_text.WordPieceTokenizer(vocab_size=10000)
tokenized_text = tokenizer.tokenize(text_data)

# 构建语言模型
model = tf.keras.Sequential([
    Embedding(vocab_size=10000, embedding_dim=256),
    LSTM(512),
    Dense(vocab_size=10000, activation='softmax')
])

# 编译模型
model.compile(optimizer='adam', loss='sparse_categorical_crossentropy', metrics=['accuracy'])

# 训练模型
model.fit(tokenized_text, epochs=10, batch_size=32)

# 生成文本
generated_text = model.predict([tokenizer.encode("Hello, how are you?")])
print(tokenizer.decode(generated_text))
```
