```python
import math

# 欧几里得几何距离计算
def euclidean_distance(x1, y1, x2, y2):
    return math.sqrt((x2 - x1)**2 + (y2 - y1)**2)

# 罗巴切夫斯基几何距离计算
def Lobachevsky_distance(x1, y1, x2, y2):
    return math.asin(math.sqrt((x2 - x1)**2 + (y2 - y1)**2))

# 主函数

    x1, y1 = 1, 2
    x2, y2 = 4, 6

    print("欧几里得几何距离：", euclidean_distance(x1, y1, x2, y2))
    print("罗巴切夫斯基几何距离：", Lobachevsky_distance(x1, y1, x2, y2))

if __name__ == "__main__":
    main()
```
