```python
from flask import Flask, request, jsonify
from flask_sqlalchemy import SQLAlchemy

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///screens_time.db'
db = SQLAlchemy(app)

class ScreenTime(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    user_id = db.Column(db.Integer, nullable=False)
    screen_time = db.Column(db.Integer, nullable=False)
    date = db.Column(db.Date, nullable=False)

@app.route('/api/record_screen_time', methods=['POST'])
def record_screen_time():
    data = request.get_json()
    user_id = data['user_id']
    screen_time = data['screen_time']
    date = data['date']
    
    new_screen_time = ScreenTime(user_id=user_id, screen_time=screen_time, date=date)
    db.session.add(new_screen_time)
    db.session.commit()
    
    return jsonify({'message': 'Screen time recorded successfully'})

if __name__ == '__main__':
    app.run(debug=True)
```
