```python
# user_management.py

import sqlite3
import hashlib
from flask import Flask, request, jsonify

app = Flask(__name__)

# 数据库连接
def get_db_connection():
    conn = sqlite3.connect('lifemeaningbuilder.db')
    conn.row_factory = sqlite3.Row
    return conn

# 注册用户
@app.route('/register', methods=['POST'])
def register():
    username = request.form['username']
    password = request.form['password']
    # 加密密码
    encrypted_password = hashlib.sha256(password.encode()).hexdigest()

    conn = get_db_connection()
    cursor = conn.cursor()

    # 检查用户名是否已存在
    cursor.execute("SELECT * FROM users WHERE username=?", (username,))
    user = cursor.fetchone()
    if user:
        conn.close()
        return jsonify({'error': '用户名已存在'})

    # 插入新用户
    cursor.execute("INSERT INTO users (username, password) VALUES (?, ?)", (username, encrypted_password))
    conn.commit()
    conn.close()

    return jsonify({'message': '注册成功'})

# 登录用户
@app.route('/login', methods=['POST'])
def login():
    username = request.form['username']
    password = request.form['password']
    # 加密密码
    encrypted_password = hashlib.sha256(password.encode()).hexdigest()

    conn = get_db_connection()
    cursor = conn.cursor()

    # 检查用户名和密码是否匹配
    cursor.execute("SELECT * FROM users WHERE username=? AND password=?", (username, encrypted_password))
    user = cursor.fetchone()
    if user:
        conn.close()
        return jsonify({'message': '登录成功'})
    else:
        conn.close()
        return jsonify({'error': '用户名或密码错误'})

# 自我认知测试
@app.route('/self_awareness', methods=['POST'])
def self_awareness():
    user_id = request.form['user_id']
    self_awareness = request.form['self_awareness']

    conn = get_db_connection()
    cursor = conn.cursor()

    # 插入自我认知测试结果
    cursor.execute("INSERT INTO self_awareness_tests (user_id, self_awareness) VALUES (?, ?)", (user_id, self_awareness))
    conn.commit()
    conn.close()

    return jsonify({'message': '自我认知测试成功'})

if __name__ == '__main__':
    app.run(debug=True)
```
