```python
def reduce_device_usage(target_time, daily_reduction):
    current_time = 180  # 初始使用时间为3小时
    while current_time > target_time:
        current_time -= daily_reduction
        print(f"今日设备使用时间：{current_time}分钟")
    return current_time

target_time = 90  # 目标使用时间为1.5小时
daily_reduction = 5  # 每天减少5分钟

print("开始实施科技禁欲计划：")
current_time = reduce_device_usage(target_time, daily_reduction)

```

$$
\text{新使用时间} = \text{旧使用时间} - \text{每天减少的时间}
$$

- ...
