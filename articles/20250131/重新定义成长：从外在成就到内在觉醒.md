```python
# app.py

from flask import Flask, request, jsonify
from models import Diary, Mood, Reflection, ValueSystem
from database import init_db

app = Flask(__name__)
init_db()

@app.route('/diary', methods=['POST'])
def add_diary():
    content = request.form['content']
    diary = Diary(content=content)
    db.session.add(diary)
    db.session.commit()
    return jsonify({'id': diary.id})

@app.route('/diary', methods=['GET'])
def get_diaries():
    diaries = Diary.query.all()
    return jsonify([{'id': diary.id, 'content': diary.content} for diary in diaries])

@app.route('/mood', methods=['GET'])
def get_mood():
    mood = Mood.query.first()
    return jsonify({'mood': mood.mood})

@app.route('/mood', methods=['POST'])
def update_mood():
    new_mood = request.form['mood']
    mood = Mood.query.first()
    mood.mood = new_mood
    db.session.commit()
    return jsonify({'status': 'success'})

@app.route('/reflection', methods=['POST'])
def add_reflection():
    content = request.form['content']
    reflection = Reflection(content=content)
    db.session.add(reflection)
    db.session.commit()
    return jsonify({'id': reflection.id})

@app.route('/reflection', methods=['GET'])
def get_reflections():
    reflections = Reflection.query.all()
    return jsonify([{'id': reflection.id, 'content': reflection.content} for reflection in reflections])

@app.route('/valuesystem', methods=['POST'])
def add_valuesystem():
    value = request.form['value']
    value_system = ValueSystem(value=value)
    db.session.add(value_system)
    db.session.commit()
    return jsonify({'id': value_system.id})

@app.route('/valuesystem', methods=['GET'])
def get_valuesystem():
    value_systems = ValueSystem.query.all()
    return jsonify([{'id': value_system.id, 'value': value_system.value} for value_system in value_systems])

if __name__ == '__main__':
    app.run(debug=True)
```
