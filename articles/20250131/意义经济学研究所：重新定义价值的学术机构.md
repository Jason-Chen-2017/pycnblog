```python
import pandas as pd

import matplotlib.pyplot as plt

# 读取数据

# 计算平均价值感知
average_value_perception = data['value_perception'].mean()

# 计算平均情感体验
average_emotional_experience = data['emotional_experience'].mean()

# 绘制价值感知与情感体验散点图
plt.scatter(data['value_perception'], data['emotional_experience'])
plt.xlabel('价值感知')
plt.ylabel('情感体验')
plt.title('价值感知与情感体验关系')
plt.show()
```
