```python
import nltk

def keyword_filtering(user_input):
    # 根据关键词过滤信息
    keywords = ["焦虑", "抑郁", "压力"]
    if any(keyword in user_input for keyword in keywords):
        return "过滤掉负面信息。"
    else:
        return "保留正面信息。"

def sentiment_analysis(user_input):
    # 进行情感分析
    tokens = nltk.word_tokenize(user_input)
    pos_tags = nltk.pos_tag(tokens)
    positive_words = ["开心", "快乐", "愉快"]
    negative_words = ["焦虑", "抑郁", "烦恼"]
    positive_count = sum(tag in positive_words for word, tag in pos_tags)
    negative_count = sum(tag in negative_words for word, tag in pos_tags)
    if positive_count > negative_count:
        return "情感分析结果：正面。"
    else:
        return "情感分析结果：负面。"

```
