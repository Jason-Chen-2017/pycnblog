```python
# 水质监测数据处理脚本

import csv
from datetime import datetime

# 读取CSV文件
def read_data(file_path):
    with open(file_path, 'r') as file:
        reader = csv.DictReader(file)
        data = [row for row in reader]
    return data

# 数据处理
def process_data(data):
    processed_data = []
    for row in data:
        timestamp = datetime.strptime(row['timestamp'], '%Y-%m-%d %H:%M:%S')
        ph_value = float(row['ph_value'])
        dissolved_oxygen = float(row['dissolved_oxygen'])
        temperature = float(row['temperature'])
        
        # 数据预处理
        if ph_value >= 6 and ph_value <= 8.5 and dissolved_oxygen >= 6 and temperature >= 10 and temperature <= 30:
            processed_data.append({
                'timestamp': timestamp,
                'ph_value': ph_value,
                'dissolved_oxygen': dissolved_oxygen,
                'temperature': temperature
            })
    return processed_data

# 数据分析
def analyze_data(processed_data):
    total_ph_value = 0
    total_dissolved_oxygen = 0
    total_temperature = 0
    for data in processed_data:
        total_ph_value += data['ph_value']
        total_dissolved_oxygen += data['dissolved_oxygen']
        total_temperature += data['temperature']
    
    average_ph_value = total_ph_value / len(processed_data)
    average_dissolved_oxygen = total_dissolved_oxygen / len(processed_data)
    average_temperature = total_temperature / len(processed_data)
    
    print("平均pH值：", average_ph_value)
    print("平均溶解氧：", average_dissolved_oxygen)
    print("平均温度：", average_temperature)

# 主函数
def main():
    file_path = 'water_quality_data.csv'
    data = read_data(file_path)
    processed_data = process_data(data)
    analyze_data(processed_data)

if __name__ == '__main__':
    main()
```

```python
from flask import Flask, jsonify, request
from flask_sqlalchemy import SQLAlchemy

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'mysql+pymysql://username:password@localhost/db_name'
db = SQLAlchemy(app)

class User(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    username = db.Column(db.String(80), unique=True, nullable=False)
    # ... 其他用户属性

class Activity(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(120), nullable=False)
    start_time = db.Column(db.DateTime, nullable=False)
    end_time = db.Column(db.DateTime, nullable=False)
    # ... 其他活动属性

@app.route('/users', methods=['GET', 'POST'])
def handle_users():
    if request.method == 'GET':
        users = User.query.all()
        return jsonify([user.username for user in users])
    elif request.method == 'POST':
        user = User(username=request.json['username'])
        db.session.add(user)
        db.session.commit()
        return jsonify({'message': 'User added successfully.'})

@app.route('/activities', methods=['GET', 'POST'])
def handle_activities():
    if request.method == 'GET':
        activities = Activity.query.all()
        return jsonify([{'name': activity.name, 'start_time': activity.start_time, 'end_time': activity.end_time} for activity in activities])
    elif request.method == 'POST':
        activity = Activity(
            name=request.json['name'],
            start_time=request.json['start_time'],
            end_time=request.json['end_time']
        )
        db.session.add(activity)
        db.session.commit()
        return jsonify({'message': 'Activity added successfully.'})

if __name__ == '__main__':
    db.create_all()
    app.run(debug=True)
```
