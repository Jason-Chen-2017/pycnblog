def parse_prompt(prompt):

        else:
            print("Invalid action")
    else:
        print("Invalid prompt")

if __name__ == "__main__":

import re

def parse_prompt(prompt):
    # 解析提示词
    pattern = re.compile(r"^(?P<action>\w+)\s+(?P<args>.+)$")
    match = pattern.match(prompt)
    if match:
        return match.group('action'), match.group('args')
    else:
        return None, None

# prompt_executor.py
def execute_action(action, args):
    # 执行语义
    if action == 'add':
        return int(args[0]) + int(args[1])
    elif action == 'subtract':
        return int(args[0]) - int(args[1])
    else:
        return None

# main.py
import sys

def main():
    prompt = sys.argv[1]
    action, args = parse_prompt(prompt)
    if action:
        result = execute_action(action, args)
        if result:
            print(result)
        else:
            print("Invalid action")
    else:
        print("Invalid prompt")

if __name__ == "__main__":
    main()
