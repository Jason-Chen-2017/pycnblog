           )

           self.decoder = nn.Sequential(
               nn.Linear(256, 128),
               nn.ReLU(),
               nn.Unflatten(1, (128, 1, 1)),

               nn.ReLU(),

               nn.ReLU(),
               nn.ConvTranspose2d(32, 16, 4, 2),
               nn.ReLU(),

           )
