from flask import Flask, request, jsonify
import ply.lex as lex

app = Flask(__name__)

# 语法规则定义
tokens = ('PROMPT')
t_PROMPT = r'[A-Za-z0-9]+'

# 语法解析器
lexer = lex.lex()

# 输入模块
@app.route('/input', methods=['POST'])
def input():
    prompt = request.json['prompt']
    lexer.input(prompt)
    while True:
        token = lexer.token()
        if not token:
            break
        # 检测异常
        if token.type != 'PROMPT':
            return jsonify({'status': 'error', 'result': '语法错误', 'error': token.value})
    # 输出结果
    return jsonify({'status': 'success', 'result': '今天天气晴朗。'})

if __name__ == '__main__':
    app.run(debug=True)
