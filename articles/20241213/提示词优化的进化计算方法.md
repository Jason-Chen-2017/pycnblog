import random

# 初始种群生成
def initial_population(size, chromosome_length):
    population = []
    for _ in range(size):
        individual = [random.randint(0, 1) for _ in range(chromosome_length)]
        population.append(individual)
    return population

# 适应度评估
def fitness_function(individual):
    # 这里是一个简单的适应度函数，实际应用中需要根据具体问题设计
    fitness = sum(individual)
    return fitness

# 选择操作
def selection(population, fitnesses, k):
    selected = random.choices(population, weights=fitnesses, k=k)
    return selected

# 交叉操作
def crossover(parent1, parent2):
    crossover_point = random.randint(1, len(parent1) - 1)
    child1 = parent1[:crossover_point] + parent2[crossover_point:]
    child2 = parent2[:crossover_point] + parent1[crossover_point:]
    return child1, child2

# 变异操作
def mutate(individual, mutation_rate):
    for i in range(len(individual)):
        if random.random() < mutation_rate:
            individual[i] = 1 - individual[i]
    return individual

# 进化计算主函数
def evolutionary_computation(pop_size, chrom_len, gen_num, mutation_rate):
    population = initial_population(pop_size, chrom_len)
    for _ in range(gen_num):
        fitnesses = [fitness_function(ind) for ind in population]
        new_population = []
        for _ in range(pop_size // 2):
            parent1, parent2 = selection(population, fitnesses, 2)
            child1, child2 = crossover(parent1, parent2)
            new_population.extend([mutate(child1, mutation_rate), mutate(child2, mutation_rate)])
        population = new_population
    best_individual = max(population, key=fitness_function)
    return best_individual

# 测试进化计算算法
best_solution = evolutionary_computation(100, 100, 1000, 0.05)
print("Best solution:", best_solution)

### 3.3 进化计算算法

#### 进化计算算法原理

进化计算算法通过模拟生物进化过程，实现对问题的优化。其主要步骤包括：

1. **初始种群生成**：随机生成初始种群。
2. **适应度评估**：计算每个个体的适应度值。
3. **选择操作**：根据适应度值，选择优秀的个体。
4. **交叉操作**：将选择的个体进行交叉，生成新的个体。
5. **变异操作**：对新生成的个体进行变异，增加种群的多样性。
6. **迭代优化**：重复步骤2-5，直至达到优化目标。

import random

# 初始种群生成
def initial_population(size, chromosome_length):
    population = []
    for _ in range(size):
        individual = [random.randint(0, 1) for _ in range(chromosome_length)]
        population.append(individual)
    return population

# 适应度评估
def fitness_function(individual):
    # 这里是一个简单的适应度函数，实际应用中需要根据具体问题设计
    fitness = sum(individual)
    return fitness

# 选择操作
def selection(population, fitnesses, k):
    selected = random.choices(population, weights=fitnesses, k=k)
    return selected

# 交叉操作
def crossover(parent1, parent2):
    crossover_point = random.randint(1, len(parent1) - 1)
    child1 = parent1[:crossover_point] + parent2[crossover_point:]
    child2 = parent2[:crossover_point] + parent1[crossover_point:]
    return child1, child2

# 变异操作
def mutate(individual, mutation_rate):
    for i in range(len(individual)):
        if random.random() < mutation_rate:
            individual[i] = 1 - individual[i]
    return individual

# 进化计算主函数
def evolutionary_computation(pop_size, chrom_len, gen_num, mutation_rate):
    population = initial_population(pop_size, chrom_len)
    for _ in range(gen_num):
        fitnesses = [fitness_function(ind) for ind in population]
        new_population = []
        for _ in range(pop_size // 2):
            parent1, parent2 = selection(population, fitnesses, 2)
            child1, child2 = crossover(parent1, parent2)
            new_population.extend([mutate(child1, mutation_rate), mutate(child2, mutation_rate)])
        population = new_population
    best_individual = max(population, key=fitness_function)
    return best_individual

# 测试进化计算算法
best_solution = evolutionary_computation(100, 100, 1000, 0.05)
print("Best solution:", best_solution)
