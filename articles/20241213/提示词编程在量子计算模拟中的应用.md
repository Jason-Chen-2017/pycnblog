```python
import qiskit
from qiskit import QuantumCircuit, execute, Aer

# 创建量子电路
circuit = QuantumCircuit(2)

# 添加量子门
circuit.h(0)
circuit.cx(0, 1)

# 运行模拟器
simulator = Aer.get_backend('qasm_simulator')
result = execute(circuit, simulator, shots=1024).result()

# 分析结果
counts = result.get_counts(circuit)
print(counts)

# 画出量子电路
circuit.draw()
```
