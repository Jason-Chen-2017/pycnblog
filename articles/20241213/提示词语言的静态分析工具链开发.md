```python
# Python代码示例
class Lexer:
    def __init__(self, source_code):
        self.source_code = source_code
        self.tokens = []

    def tokenize(self):
        state = 'INIT'
        token = ''
        for char in self.source_code:
            if state == 'INIT':
                if char.isalpha():
                    state = 'IDENTIFIER'
                    token += char
                elif char.isdigit():
                    state = 'NUMBER'
                    token += char
                else:
                    self.tokens.append((state, token))
                    state = 'INIT'
                    token = ''
            # 其他状态的处理...

        if token:
            self.tokens.append((state, token))

        return self.tokens

lexer = Lexer(source_code)
tokens = lexer.tokenize()
print(tokens)
```

```python

```

```python
# Python代码示例
class Parser:
    def __init__(self, tokens):
        self.tokens = tokens
        self.current_token = self.tokens.pop(0)

    def parse(self):
        ast = self.parse_program()
        return ast

    def parse_program(self):
        program = {}
        while self.current_token:
            if self.current_token[0] == 'KEYWORD' and self.current_token[1] == 'def':
                function = self.parse_function()
                program[function['name']] = function
            self.current_token = self.tokens.pop(0)
        return program

    def parse_function(self):
        function = {}
        function['name'] = self.current_token[1]
        self.current_token = self.tokens.pop(0)
        function['arguments'] = self.parse_arguments()
        function['body'] = self.parse_block()
        return function

    def parse_arguments(self):
        arguments = []
        while self.current_token[0] == 'IDENTIFIER':
            arguments.append(self.current_token[1])
            self.current_token = self.tokens.pop(0)
        return arguments

    def parse_block(self):
        block = []
        while self.current_token[0] != 'KEYWORD' or self.current_token[1] != 'return':
            block.append(self.current_token)
            self.current_token = self.tokens.pop(0)
        return block

tokens = [['KEYWORD', 'def'], ['IDENTIFIER', 'add'], ['IDENTIFIER', 'a'], ['IDENTIFIER', 'b'], ['KEYWORD', 'return'], ['IDENTIFIER', 'a'], ['OPERATOR', '+'], ['IDENTIFIER', 'b']]
parser = Parser(tokens)
ast = parser.parse()
print(ast)
```


```python
# Python代码示例
import re

# 词法分析器实现
class Lexer:
    def __init__(self, source_code):
        self.source_code = source_code
        self.tokens = []

    def tokenize(self):
        # 正则表达式匹配词法单元
        pattern = r'\b\w+\b|\S'
        for token in re.findall(pattern, self.source_code):
            self.tokens.append(token)
        return self.tokens

# 语法分析器实现
class Parser:
    def __init__(self, tokens):
        self.tokens = tokens
        self.current_token = self.tokens[0]

    def parse(self):
        # 根据词法单元构建抽象语法树
        ast = self.parse_program()
        return ast

    def parse_program(self):
        ast = {'type': 'Program', 'body': []}
        while self.current_token:
            ast['body'].append(self.parse_statement())
            self.current_token = self.tokens.pop(0)
        return ast

    def parse_statement(self):
        # 分析单个语句
        if self.current_token == 'def':
            self.tokens.pop(0)
            return self.parse_function_definition()
        else:
            return self.parse_expression()

    def parse_function_definition(self):
        # 分析函数定义
        name = self.tokens.pop(0)
        self.tokens.pop(0)  # pop '(':)
        arguments = self.tokens.pop(0)
        self.tokens.pop(0)  # pop ')':)
        body = self.parse_block()
        return {'type': 'FunctionDefinition', 'name': name, 'arguments': arguments, 'body': body}

    def parse_block(self):
        # 分析代码块
        block = []
        while self.current_token and self.current_token != '}':
            block.append(self.parse_statement())
            self.tokens.pop(0)  # pop ';':)
        return block

    def parse_expression(self):
        # 分析表达式
        return self.parse_binary_expression()

    def parse_binary_expression(self):
        # 分析二元表达式
        left = self.parse_unary_expression()
        while self.current_token in ['+', '-']:
            operator = self.current_token
            self.tokens.pop(0)  # pop operator:):
            right = self.parse_unary_expression()
            return {'type': 'BinaryExpression', 'operator': operator, 'left': left, 'right': right}
        return left

    def parse_unary_expression(self):
        # 分析一元表达式
        if self.current_token == '+':
            self.tokens.pop(0)  # pop '+':)
            return self.parse_unary_expression()
        elif self.current_token == '-':
            self.tokens.pop(0)  # pop '-':)
            return {'type': 'UnaryExpression', 'operator': '-', 'operand': self.parse_unary_expression()}
        else:
            return self.parse_primary_expression()

    def parse_primary_expression(self):
        # 分析基本表达式
        token = self.current_token
        self.tokens.pop(0)  # pop token:)
        return {'type': 'PrimaryExpression', 'value': token}

# AST构建实现
class ASTBuilder:
    def __init__(self):
        self.ast = None

    def build_ast(self, tokens):
        parser = Parser(tokens)
        self.ast = parser.parse()
        return self.ast

    def print_ast(self, ast, indent=0):
        if ast['type'] == 'Program':
            for statement in ast['body']:
                self.print_statement(statement, indent)
        elif ast['type'] == 'FunctionDefinition':
            self.print_function_definition(ast, indent)
        elif ast['type'] == 'BinaryExpression':
            self.print_binary_expression(ast, indent)
        elif ast['type'] == 'UnaryExpression':
            self.print_unary_expression(ast, indent)
        elif ast['type'] == 'PrimaryExpression':
            self.print_primary_expression(ast, indent)

    def print_statement(self, statement, indent):
        print(' ' * indent + statement['type'])

    def print_function_definition(self, function, indent):
        print(' ' * indent + f'Function {function["name"]}')
        print(' ' * (indent + 4) + f'Arguments: {function["arguments"]}')
        print(' ' * (indent + 4) + 'Body:')
        for stmt in function['body']:
            self.print_statement(stmt, indent + 4)

    def print_binary_expression(self, expr, indent):
        print(' ' * indent + f'BinaryExpression {expr["operator"]} ({expr["left"]["value"]}, {expr["right"]["value"]})')
        self.print_statement(expr["left"], indent + 4)
        self.print_statement(expr["right"], indent + 4)

    def print_unary_expression(self, expr, indent):
        print(' ' * indent + f'UnaryExpression {expr["operator"]} ({expr["operand"]["value"]})')
        self.print_statement(expr["operand"], indent + 4)

    def print_primary_expression(self, expr, indent):
        print(' ' * indent + f'PrimaryExpression {expr["value"]}')

# 使用示例
source_code = """
def add(a, b): return a + b
"""
tokens = Lexer(source_code).tokenize()
ast_builder = ASTBuilder()
ast = ast_builder.build_ast(tokens)
ast_builder.print_ast(ast)
```
