```python
import re
from mermaid import Mermaid

def parse_prompt_language(prompt):
    # 使用正则表达式解析输入的提示词语言
    pattern = r"(\d+)\s*+\+(\s*\d+)"
    matches = re.findall(pattern, prompt)
    return [int(match[0]), int(match[1])] if matches else None

def construct_semantic_network(expression):
    # 构建形式化语义网
    memird = Mermaid()
    memird.add_node("A", "输入：{}".format(expression[0]))
    memird.add_node("B", "输入：{}".format(expression[1]))
    memird.add_edge("A", "B", label="加法运算")
    memird.add_node("C", "输出：{}".format(expression[0] + expression[1]))
    return memird.render()

def prove_equivalence(expression1, expression2):
    # 使用匹配算法验证两个表达式的等价性
    semantic_network1 = construct_semantic_network(expression1)
    semantic_network2 = construct_semantic_network(expression2)
    return semantic_network1 == semantic_network2

# 示例输入
prompt1 = "3 + 4"
prompt2 = "4 + 3"

# 解析输入
expression1 = parse_prompt_language(prompt1)
expression2 = parse_prompt_language(prompt2)

# 构建形式化语义网
semantic_network1 = construct_semantic_network(expression1)
semantic_network2 = construct_semantic_network(expression2)

# 验证等价性
is_equivalent = prove_equivalence(expression1, expression2)

print("语义网1：", semantic_network1)
print("语义网2：", semantic_network2)
print("等价性结果：", is_equivalent)
```
