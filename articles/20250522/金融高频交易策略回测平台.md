### 5.2.2 策略实现代码
```python
def mean_reversion_strategy(prices, sma_window=20, std_window=20):
    sma = prices.rolling(sma_window).mean()
    std = prices.rolling(std_window).std()
    signals = pd.DataFrame(index=prices.index)
    signals['signal'] = 0
    signals['signal'][sma > prices] = 1
    signals['signal'][sma < prices] = -1
    return signals
```

### 5.2.3 回测与分析代码
```python
import backtrader as bt

class MeanReversionStrategy(bt.Strategy):
    def __init__(self):
        self.position = None
        self.data = self.datas[0]
        self.sma = bt.indicators.SimpleMovingAverage(
            self.data.close, period=20
        )
        self.std = bt.indicators.StandardDeviation(
            self.data.close, period=20
        )

    def next(self):
        if self.sma > self.data.close:
            self.position.close()
            self.position = self.buy()
        elif self.sma < self.data.close:
            self.position.close()
            self.position = self.sell()

# 回测设置
cerebro = bt.Cerebro()
cerebro.addstrategy(MeanReversionStrategy)
data = bt.feeds.PandasData(dataname=data)
cerebro.adddata(data)
cerebro.broker.setcash(100000.0)
cerebro.run()
```
