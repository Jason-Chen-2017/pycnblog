```python
def cultural_fusion_assessment(strategies):
    weights = [0.4, 0.3, 0.3]
    scores = []
    for strategy in strategies:
        score = sum(weights[i] * strategy[i] for i in range(len(weights)))
        scores.append(score)
    return scores

strategies = [
    [0.8, 0.7, 0.6],
    [0.6, 0.8, 0.7],
    [0.7, 0.6, 0.8]
]

print(cultural_fusion_assessment(strategies))
```
