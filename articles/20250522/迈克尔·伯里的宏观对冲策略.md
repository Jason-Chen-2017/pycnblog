### 5.2 代码实现

```python
import pandas as pd
import numpy as np

def generate_signal(data):
    # 计算指标的Z-score
    mu = data.mean()
    sigma = data.std()
    z_scores = (data - mu) / sigma
    # 生成信号
    signal = np.where(z_scores > 1, 1, np.where(z_scores < -1, -1, 0))
    return signal

# 示例数据
data = pd.DataFrame({'GDP': [2, 3, 4, 1, 2], 'CPI': [1, 2, 3, 2, 1]})
signal = generate_signal(data)
print(signal)
```
