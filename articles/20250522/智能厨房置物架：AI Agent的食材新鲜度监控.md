```python
# 示例代码：简单分类模型
import numpy as np
from sklearn import svm

# 数据集示例
X = np.array([[1, 2], [3, 4], [5, 6], [7, 8]])
y = np.array([0, 1, 0, 1])

# 创建和支持向量分类器
clf = svm.SVC()
clf.fit(X, y)

# 预测新数据
new_data = np.array([[2, 3]])
print(clf.predict(new_data))  # 输出结果
```
