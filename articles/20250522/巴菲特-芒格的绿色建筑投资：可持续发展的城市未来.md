#### 6.4 算法实现代码

```python
def calculate_score(building_data):
    weights = {
        'energy_efficiency': 0.4,
        'environmental_impact': 0.3,
        'economic_benefit': 0.2,
        'social_impact': 0.1
    }
    score = 0
    for key in building_data:
        score += building_data[key] * weights[key]
    return score

# 示例数据
building = {
    'energy_efficiency': 85,
    'environmental_impact': 20,
    'economic_benefit': 70,
    'social_impact': 60
}

print(calculate_score(building))
```

```python
import pandas as pd

def main():
    # 示例数据
    data = {
        'building': ['Building A', 'Building B', 'Building C'],
        'energy_efficiency': [85, 70, 90],
        'environmental_impact': [20, 25, 15],
        'economic_benefit': [70, 60, 80],
        'social_impact': [60, 50, 70]
    }
    df = pd.DataFrame(data)
    
    # 计算评分
    weights = {'energy_efficiency': 0.4, 'environmental_impact': 0.3, 
               'economic_benefit': 0.2, 'social_impact': 0.1}
    df['score'] = df.apply(lambda x: sum(x[i] * weights[i] for i in weights), axis=1)
    
    print(df)

if __name__ == "__main__":
    main()
```
