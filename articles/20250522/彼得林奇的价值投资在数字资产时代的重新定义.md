#### 3.3 Python代码实现
```python
import pandas as pd
import numpy as np

def calculate_intrinsic_value(data):
    # 计算自由现金流
    free_cash_flow = data['free_cash_flow']
    # 计算加权平均资本成本
    wacc = data['wacc']
    # 计算终值
    terminal_value = free_cash_flow / (wacc - data['growth_rate'])
    # 计算内在价值
    intrinsic_value = (free_cash_flow + terminal_value) / (1 + wacc)
    return intrinsic_value

# 示例数据
data = {
    'free_cash_flow': 100,
    'wacc': 0.1,
    'growth_rate': 0.05
}

iv = calculate_intrinsic_value(data)
print(f"内在价值: {iv}")
```
