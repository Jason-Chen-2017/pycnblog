```python
# 示例代码：投资者行为分类模型训练

import pandas as pd
from sklearn.model import DecisionTreeClassifier
from sklearn.metrics import accuracy_score

# 数据加载
data = pd.read_csv('investor_behavior.csv')

# 特征提取与标签分离
X = data.drop('strategy_type', axis=1)
y = data['strategy_type']

# 模型训练
model = DecisionTreeClassifier()
model.fit(X, y)

# 预测与评估
predicted = model.predict(X)
print("模型准确率:", accuracy_score(y, predicted))
```
