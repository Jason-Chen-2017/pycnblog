#### 3.3 Python代码实现
```python
import pandas as pd

def calculate_sentiment(data):
    # 计算市场情绪指数
    sentiment = data['price'].pct_change().mean() * 100
    return sentiment

def inverse_strategy(data):
    # 逆向投资策略实现
    sentiment = calculate_sentiment(data)
    if sentiment < -5:
        return "买入"
    elif sentiment > 5:
        return "卖出"
    else:
        return "观望"

# 示例数据
data = pd.DataFrame({
    'price': [100, 98, 95, 90, 85],
    'volume': [1000, 800, 600, 400, 200]
})

result = inverse_strategy(data)
print(result)  # 输出: "买入"
```
