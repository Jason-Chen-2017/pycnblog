  ```python
  import pandas as pd
  import yfinance as yf

  # 下载数据
  def get_data(ticker, start, end):
      data = yf.download(ticker, start=start, end=end)
      return data

  # 计算市盈率
  def calculate_pe(data):
      pe = data['Close'] / data['EPS']
      return pe

  # 筛选低市盈率股票
  def screen_low_pe(tickers, start, end, threshold=15):
      low_pe_stocks = []
      for ticker in tickers:
          data = get_data(ticker, start, end)
          if not data.empty:
              pe = calculate_pe(data)
              if pe.mean() < threshold:
                  low_pe_stocks.append(ticker)
      return low_pe_stocks

  # 示例运行
  tickers = ['AAPL', 'MSFT', 'GOOGL']
  start = '2022-01-01'
  end = '2022-12-31'
  threshold = 15

  low_pe_stocks = screen_low_pe(tickers, start, end, threshold)
  print(f"低市盈率股票：{low_pe_stocks}")
  ```
