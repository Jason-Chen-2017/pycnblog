#### 核心代码实现

```python
import numpy as np
import pandas as pd

def graham_value_estimate(fcf, discount_rate, growth_rate, perpetuity_growth=0):
    """
    计算企业内在价值的现金流折现模型（DCF）。
    
    参数：
    - fcf: 自由现金流（单位：万元）
    - discount_rate: 折现率（年化）
    - growth_rate: 前期增长率（通常为5年）
    - perpetuity_growth: 永续期增长率
    
    返回：
    - 内在价值（亿元）
    """
    # 前期现金流增长阶段
    periods = 5  # 假设增长期为5年
    cash_flows = []
    for i in range(periods):
        cf = fcf * (1 + growth_rate/100)**i
        cash_flows.append(cf)
    
    # 永续期现金流
    terminal_value = cash_flows[-1] * (1 + perpetuity_growth/100) / (discount_rate - perpetuity_growth/100)
    cash_flows.append(terminal_value)
    
    # 计算现值
    present_values = [cf / (1 + discount_rate/100)**i for i, cf in enumerate(cash_flows)]
    total_pv = sum(present_values)
    
    return total_pv / 10000  # 转换为亿元

# 示例数据
fcf = 10000  # 自由现金流：10,000万元
discount_rate = 10  # 折现率：10%
growth_rate = 5  # 前期增长率：5%
perpetuity_growth = 2  # 永续期增长率：2%

# 计算内在价值
iv = graham_value_estimate(fcf, discount_rate, growth_rate, perpetuity_growth)
print(f"企业的内在价值为：{iv:.2f}亿元")
```
