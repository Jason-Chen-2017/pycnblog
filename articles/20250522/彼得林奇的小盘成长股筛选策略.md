import pandas as pd
import numpy as np

def calculate_growth_rate(data):
    return (data - data.shift(1)) / data.shift(1)

def peter_lynch_strategy(data):
    growth_rate = calculate_growth_rate(data['净利润'])
    pe_ratio = data['股价'] / data['每股净利润']
    score = (growth_rate + pe_ratio) / 2
    return score

# 示例数据
data = pd.DataFrame({
    '股价': [10, 15, 20, 25],
    '每股净利润': [1, 1.5, 2, 2.5],
    '营业收入': [100, 120, 140, 160],
    '净利润': [50, 60, 70, 80]
})

score = peter_lynch_strategy(data)
print(score)
