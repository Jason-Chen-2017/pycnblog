```python
import pandas as pd

def calculate_cultural_difficulty(cultural_data):
    weights = cultural_data['权重']
    differences = cultural_data['差异程度']
    cultural_difficulty = sum(weights[i] * differences[i] for i in range(len(weights)))
    return cultural_difficulty

# 示例数据
cultural_data = pd.DataFrame({
    '文化因素': ['价值观', '沟通方式', '管理风格'],
    '权重': [0.4, 0.3, 0.3],
    '差异程度': [0.6, 0.5, 0.4]
})

result = calculate_cultural_difficulty(cultural_data)
print(f"文化差异指数: {result}")
```
