```python
import numpy as np

def prospect_theory_utility(gains, reference_point, risk_attitude):
    # gains: 收益或损失
    # reference_point: 参照点
    # risk_attitude: 风险态度（0表示风险中性，正值表示风险厌恶，负值表示风险寻求）
    def probability_weight(p):
        return p ** (1/0.8) if p > 0 else (-p) ** (1/0.8)
    
    adjusted_gains = gains - reference_point
    probabilities = [0.5, 0.5]
    
    utility = 0
    for i in range(len(probabilities)):
        p = probabilities[i]
        weighted_p = probability_weight(p)
        utility += weighted_p * (adjusted_gains[i] ** risk_attitude)
    return utility

gains = [100, -50]
reference_point = 0
risk_attitude = 1.5  # 风险厌恶

result = prospect_theory_utility(gains, reference_point, risk_attitude)
print("Prospect Theory Utility:", result)
```

```python
import pandas as pd

def behavioral_bias_analysis(decision_data):
    # decision_data: 包含投资决策数据的 DataFrame
    # 分析过度自信和损失厌恶的影响
    confidence_levels = decision_data['confidence'].values
    loss_aversion_scores = decision_data['loss_aversion'].values

    # 过度自信校正
    adjusted_confidence = confidence_levels * 0.8
    # 损失厌恶校正
    adjusted_loss_aversion = loss_aversion_scores * 0.9

    return adjusted_confidence, adjusted_loss_aversion

# 示例数据
data = {'confidence': [0.9, 0.8, 0.7], 'loss_aversion': [0.8, 0.9, 0.6]}
df = pd.DataFrame(data)
adj_confidence, adj_loss = behavioral_bias_analysis(df)
print("Adjusted Confidence:", adj_confidence)
print("Adjusted Loss Aversion:", adj_loss)
```

```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

def behavioral_investment_strategy(portfolio, market_data):
    # portfolio: 投资组合数据
    # market_data: 市场数据
    # 基于 prospect theory 的投资策略
    reference_point = portfolio.mean()
    gains = market_data - reference_point
    
    # 计算概率加权
    def probability_weight(p):
        return p ** (1/0.8) if p > 0 else (-p) ** (1/0.8)
    
    probabilities = [0.5, 0.5]
    utility = 0
    for i in range(len(probabilities)):
        p = probabilities[i]
        weighted_p = probability_weight(p)
        utility += weighted_p * (gains[i] ** 1.5)
    
    return utility

# 示例数据
portfolio = pd.Series([100, 200, 300])
market_data = pd.Series([150, 250, 350])
result = behavioral_investment_strategy(portfolio, market_data)
print("Behavioral Investment Strategy Utility:", result)
```
