```python
import pandas as pd
import numpy as np

def calculate_intrinsic_value(financial_data, discount_rate):
    cash_flows = financial_data['现金流'].values
    present_value = sum(cash_flows / (1 + discount_rate)**i for i in range(len(cash_flows)))
    return present_value

# 示例数据
financial_data = pd.DataFrame({
    '现金流': [100, 150, 200],
    '折扣率': [0.05, 0.05, 0.05]
})

# 计算内在价值
intrinsic_value = calculate_intrinsic_value(financial_data, 0.05)
print(f"内在价值: {intrinsic_value}")
```
