### 5.3.2 数据清洗代码
```python
import pandas as pd

def clean_data(data):
    # 删除缺失值
    data.dropna(inplace=True)
    # 处理异常值
    data = data[(data['Price'] > 0) & (data['Earnings'] > 0)]
    return data

clean_data = clean_data(data)
```

### 5.3.3 估值计算代码
```python
import pandas as pd

def calculate_valuation(data):
    data['PE'] = data['Price'] / data['Earnings']
    data['PB'] = data['Price'] / data['Book Value']
    data['Dividend Yield'] = (data['Dividend'] / data['Price']) * 100
    return data

valuated_data = calculate_valuation(clean_data)
```

### 5.3.4 结果分析代码
```python
import matplotlib.pyplot as plt

def analyze_results(data):
    plt.figure(figsize=(10, 6))
    data['PE'].hist(bins=20, alpha=0.5)
    plt.title('PE Distribution')
    plt.xlabel('PE')
    plt.ylabel('Frequency')
    plt.show()

analyze_results(valuated_data)
```
