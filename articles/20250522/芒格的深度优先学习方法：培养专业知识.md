import numpy as np
import pandas as pd

# 示例数据
X = np.array([[2, 3], [5, 4], [6, 7], [8, 9]])
y = np.array([0, 1, 1, 0])

# 构建决策树模型
model = DecisionTreeClassifier()
model.fit(X, y)

# 提取模型决策路径
def get_decision_path(model, X):
    return model.predict(X)

# 输出决策路径
print(get_decision_path(model, X))
