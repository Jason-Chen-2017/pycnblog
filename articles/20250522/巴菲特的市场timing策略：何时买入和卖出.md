import numpy as np
import matplotlib.pyplot as plt

# 示例代码：计算内在价值
def calculate_intrinsic_value(price, growth_rate, discount_rate):
    return price / (1 + growth_rate) / (1 - discount_rate)

# 示例代码：计算安全边际
def calculate_margin_of_safety(intrinsic_value, market_price):
    return (intrinsic_value - market_price) / intrinsic_value * 100

# 示例代码：可视化市场趋势
