### 代码示例：

```python
import networkx as nx
import numpy as np
import pandas as pd

def dijkstra(graph, start_node):
    distances = {node: np.inf for node in graph.nodes()}
    distances[start_node] = 0
    visited = set()
    while visited != graph.nodes():
        min_dist = np.inf
        current_node = None
        for node in graph.nodes():
            if node not in visited and distances[node] < min_dist:
                min_dist = distances[node]
                current_node = node
        if current_node is None:
            break
        visited.add(current_node)
        for neighbor in graph.neighbors(current_node):
            if distances[neighbor] > min_dist + graph[current_node][neighbor]['weight']:
                distances[neighbor] = min_dist + graph[current_node][neighbor]['weight']
    return distances

graph = nx.Graph()
graph.add_nodes_from(['A', 'B', 'C', 'D'])
graph.add_edge('A', 'B', weight=5)
graph.add_edge('A', 'C', weight=10)
graph.add_edge('B', 'D', weight=3)
graph.add_edge('C', 'D', weight=8)
distances = dijkstra(graph, 'A')
print(distances)
```
