def calculate_success_probability(s, n):
    return s / n

def calculate_roi(profits, costs):
    return profits / costs

# 示例数据
s = 5
n = 10
profits = 100
costs = 50

# 调用函数
success_prob = calculate_success_probability(s, n)
roi = calculate_roi(profits, costs)

print(f"投资回报率为：{roi}")
