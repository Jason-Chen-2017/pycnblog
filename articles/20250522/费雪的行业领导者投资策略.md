  ```python
  def calculate_growth(income_growth, profit_growth):
      return (income_growth + profit_growth) / 2

  # 示例数据
  company_data = {
      "Apple": {"income_growth": 0.15, "profit_growth": 0.20},
      "Microsoft": {"income_growth": 0.12, "profit_growth": 0.18}
  }

  # 计算成长性
  for company, data in company_data.items():
      growth = calculate_growth(data["income_growth"], data["profit_growth"])
      print(f"{company}: 成长性 = {growth}")
  ```

- 5.2 核心实现
  ```python
  import pandas as pd

  def calculate_growth(income, profit):
      return (income + profit) / 2

  # 示例数据
  data = {
      "公司": ["Apple", "Microsoft", "Google"],
      "收入增长率": [0.15, 0.12, 0.10],
      "利润增长率": [0.20, 0.18, 0.15]
  }

  df = pd.DataFrame(data)
  df["成长性"] = df.apply(lambda x: calculate_growth(x["收入增长率"], x["利润增长率"]), axis=1)
  print(df)
  ```
