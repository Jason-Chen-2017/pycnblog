```javascript
// index.js
const web3 = new Web3("http://127.0.0.1:8545");
const contractAddress = '合约地址';
const contractABI = [/* ...合约ABI ... */];
const creditRatingContract = new web3.eth.Contract(contractABI, contractAddress);

// 设置信用评分
async function setCreditRating(companyAddress, rating) {
    const account = web3.eth.accounts[0];
    const tx = {
        from: account,
        to: contractAddress,
        data: creditRatingContract.methods.setCreditRating(companyAddress, rating).encodeABI(),
    };
    await web3.eth.sendTransaction(tx);
}

// 获取信用评分
async function getCreditRating(companyAddress) {
    const rating = await creditRatingContract.methods.getCreditRating(companyAddress).call();
    return rating;
}
```
