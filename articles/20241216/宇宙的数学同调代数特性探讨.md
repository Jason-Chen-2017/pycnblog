    返回：
    同调群字典，键为整数，值为同调类列表。

    homology_groups = nx.homology(G, edge_labels=True)
    return homology_groups

def main():

    # 添加节点和边

    homology_groups = calculate_homology(G)

    # 输出同调群
    for n, group in homology_groups.items():
        print(f"H{n}: {group}")

if __name__ == "__main__":
    main()
