这意味着，\( i \)、\( j \)和\( k \)的乘积等于-1，与复数中的\( i \)的性质相似。四元数的乘法规则相对复杂，但其定义和性质使得其在表示和计算三维旋转时具有独特优势。

\[ q_1 \cdot q_2 = q_2 \cdot q_1 \]

\[ (q_1 \cdot q_2) \cdot q_3 = q_1 \cdot (q_2 \cdot q_3) \]

\[ q^{-1} = \frac{\bar{q}}{\|q\|^2} \]

import numpy as np

class Quaternion:
    def __init__(self, w, x, y, z):
        self.w = w
        self.x = x
        self.y = y
        self.z = z

    def __repr__(self):
        return f"({self.w}, {self.x}i, {self.y}j, {self.z}k)"

    def __mul__(self, other):
        w1, x1, y1, z1 = self.w, self.x, self.y, self.z
        w2, x2, y2, z2 = other.w, other.x, other.y, other.z
        return Quaternion(
            w1*w2 - x1*x2 - y1*y2 - z1*z2,
            w1*x2 + x1*w2 + y1*z2 - z1*y2,
            w1*y2 - x1*z2 + y1*w2 + z1*x2,
            w1*z2 + x1*y2 - y1*x2 + z1*w2
        )

    def conjugate(self):
        return Quaternion(self.w, -self.x, -self.y, -self.z)

    def norm(self):
        return np.sqrt(self.w**2 + self.x**2 + self.y**2 + self.z**2)

    def inverse(self):
        return Quaternion(self.w / self.norm(), -self.x / self.norm(), -self.y / self.norm(), -self.z / self.norm())

# 示例

q2 = Quaternion(1, 0, 0, 0)

def rotate_vector(q, v):
    """
    使用四元数旋转一个三维向量。
    
    参数：
    - q：旋转四元数。

    返回：
