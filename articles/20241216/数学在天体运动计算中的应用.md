def nbody_ana_noe(x, v, dt, G, m):

    # 更新速度
    v = v + a * dt

    k1 = G * m / x**2
    # 计算k2
    k2 = G * m / (x + 0.5 * k1 * dt)**2
    # 计算k3
    k3 = G * m / (x + 0.5 * k2 * dt)**2
    # 计算k4
    k4 = G * m / (x + k3 * dt)**2
