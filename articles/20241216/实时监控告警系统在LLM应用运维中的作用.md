    output = subprocess.check_output(["top", "-b", "-n", "1"], universal_newlines=True)

        if "Cpu(s)" in line:
            return float(line.split()[9])

def get_memory_usage():
    output = subprocess.check_output(["free", "-m"], universal_newlines=True)
    for line in output.split('\n'):
        if "Mem" in line:
