```python
# app.py

from flask import Flask, request, jsonify
from flask_sqlalchemy import SQLAlchemy

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///universe_thought_experiment.db'
db = SQLAlchemy(app)

class ThoughtExperiment(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(50))
    description = db.Column(db.Text)
    purpose = db.Column(db.Text)
    method = db.Column(db.Text)

@app.route('/thought_experiments', methods=['GET', 'POST'])
def thought_experiments():
    if request.method == 'GET':
        experiments = ThoughtExperiment.query.all()
        return jsonify([experiment.to_dict() for experiment in experiments])
    elif request.method == 'POST':
        data = request.json
        new_experiment = ThoughtExperiment(
            name=data['name'],
            description=data['description'],
            purpose=data['purpose'],
            method=data['method']
        )
        db.session.add(new_experiment)
        db.session.commit()
        return jsonify(new_experiment.to_dict()), 201

if __name__ == '__main__':
    db.create_all()
    app.run(debug=True)
```
