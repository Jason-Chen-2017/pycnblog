```python
// 提示词生成算法伪代码
function generatePrompt(data, keywords):
    for each article in data:
        for each keyword in keywords:
            if keyword in article:
                generatePrompt = "请问关于" + keyword + "在文章中的具体内容是什么？"
                return generatePrompt
    return "没有找到相关内容"

// 提示词筛选算法伪代码
function filterPrompt(promptList, threshold):
    filteredList = []
    for each prompt in promptList:
        if length(prompt) > threshold:
            filteredList.append(prompt)
    return filteredList
```
