```python
def extract_features(data):
    # 统计特征
    mean = np.mean(data)
    variance = np.var(data)
    std = np.std(data)
    
    # 时序特征
    window_size = 10
    window_data = data[-window_size:]
    mean_window = np.mean(window_data)
    variance_window = np.var(window_data)
    std_window = np.std(window_data)
    
    # 空间特征
    clusters = KMeans(n_clusters=2).fit(data)
    cluster_labels = clusters.labels_
    cluster0_data = data[cluster_labels == 0]
    cluster1_data = data[cluster_labels == 1]
    mean_cluster0 = np.mean(cluster0_data)
    mean_cluster1 = np.mean(cluster1_data)
    
    return mean, variance, std, mean_window, variance_window, std_window, mean_cluster0, mean_cluster1
```

```python
    # 监控性能指标
    performance_metrics = []
    while True:
        start_time = time.time()
        # 运行模型
        output = model.predict(input_data)
        end_time = time.time()
        # 计算性能指标
        performance = end_time - start_time
        performance_metrics.append(performance)
        # 输出性能指标
        print("Performance:", performance)
        # 休眠一段时间
        time.sleep(1)
```
