```python
# 伪代码：知识付费收益计算算法

def calculate_income(content_quality, marketing_strategy, user_base):
    """
    计算知识付费项目的收入

    :param content_quality: 内容质量评分（0-10分）
    :param marketing_strategy: 营销策略评分（0-10分）
    :param user_base: 用户基数（人）
    :return: 收入（元）
    """
    base_income_per_user = 10  # 每个用户的基础收入
    quality_bonus = content_quality * 0.1  # 内容质量奖金
    strategy_bonus = marketing_strategy * 0.1  # 营销策略奖金

    total_income = (base_income_per_user + quality_bonus + strategy_bonus) * user_base
    return total_income
```
