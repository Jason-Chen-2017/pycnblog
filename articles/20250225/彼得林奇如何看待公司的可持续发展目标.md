#### 6.2.1 可持续发展评分系统
以下代码示例展示如何计算企业的可持续发展指数：

```python
import numpy as np
import pandas as pd

def calculate_sustainability_score(company_data):
    # 环境保护评分
    environmental_score = company_data['carbon_emissions'] / company_data['revenue'] * 100
    # 社会责任评分
    social_score = company_data['employee_welfare'] / company_data['total_assets'] * 100
    # 经济效益评分
    economic_score = company_data['net_profit'] / company_data['revenue'] * 100
    
    # 计算综合评分
    sustainability_score = np.mean([environmental_score, social_score, economic_score])
    return sustainability_score

# 示例数据
company_data = {
    'carbon_emissions': 100,
    'revenue': 1000,
    'employee_welfare': 80,
    'total_assets': 500,
    'net_profit': 200
}

score = calculate_sustainability_score(company_data)
print(f"可持续发展指数: {score}")
```
