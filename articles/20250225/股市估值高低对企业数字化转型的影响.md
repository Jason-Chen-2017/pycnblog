        '数字化转型投入': [x1, x2, ..., xn],
        '股市估值': [y1, y2, ..., yn]
    })

    # 模型训练
    model = LinearRegression()
    model.fit(df[['数字化转型投入']], df['股市估值'])

    # 预测与评估
    predictions = model.predict(df[['数字化转型投入]])
