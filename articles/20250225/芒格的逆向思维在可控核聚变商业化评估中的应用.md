def calculate_technical_feasibility(technology_maturity, cost_efficiency):
    return (technology_maturity + cost_efficiency) / 2

def calculate_economic_feasibility(market_potential, economic_benefit, investment_cost):
    return (market_potential * economic_benefit) / investment_cost

    return (environmental_impact + social_benefit) / risk_factor
