  ```python
  import numpy as np
  import matplotlib.pyplot as plt

  # 示例数据：海洋能源开发成本与收益模型
  def energy_profit(invest, cost, revenue):
      return revenue - cost * invest

  invest_range = np.linspace(0, 100, 100)
  profit = []
  for invest in invest_range:
      revenue = 2 * invest
      cost = invest * 0.5
      profit.append(energy_profit(invest, cost, revenue))

  plt.figure(figsize=(8, 6))
  plt.plot(invest_range, profit, 'b-', linewidth=2)
  plt.title('海洋能源开发投资回报模型')
  plt.xlabel('投资金额 (百万美元)')
  plt.ylabel('利润 (百万美元)')
  plt.grid(True)
  plt.show()
  ```
