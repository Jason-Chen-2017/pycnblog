优点：数据不可篡改、透明度高、去中心化。缺点：技术复杂度高、初期投入大、监管挑战大。

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract SupplyChain {
    event ProductCreated(uint256 productId, string name);
    event ProductTransferred(uint256 productId, address from, address to);

    struct Product {
        uint256 id;
        string name;
        address owner;
        uint256 price;
    }

    mapping(uint256 => Product) products;

    function createProduct(string memory name, uint256 price) public {
        uint256 productId =/products.length + 1;
        products[productId] = Product(productId, name, msg.sender, price);
        emit ProductCreated(productId, name);
    }

    function transferProduct(uint256 productId, address recipient) public {
        Product storage product = products[productId];
        require(product.owner == msg.sender, "Not the owner");
        product.owner = recipient;
        emit ProductTransferred(productId, msg.sender, recipient);
    }
}
```
