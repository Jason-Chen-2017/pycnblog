#### 2.2.3 彼得·林奇的价值投资方法
彼得·林奇通过分析企业的财务数据、行业地位和竞争优势，选择具有成长潜力的公司进行投资。他强调长期持有，避免短期波动的影响。

```python
import pandas as pd
import numpy as np
from sklearn.linear_model import LinearRegression
import matplotlib.pyplot as plt

# 加载数据
data = pd.read_csv('market_data.csv')

# 计算收益和风险
returns = data.pct_change().dropna()
risk = returns.std()

# 优化资产配置
def optimize_portfolio(returns, n_assets):
    # 确定权重的范围
    bounds = [(0, 1) for _ in range(n_assets)]
    # 约束条件：权重之和为1
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - 1}
    # 定义目标函数：最小化风险
    def objective(weights):
        return np.dot(weights, returns.mean(axis=1)) * weights.T.dot(returns.cov())
    # 使用SLSQP求解器优化
    from scipy.optimize import minimize
    result = minimize(objective, np.array([1/n_assets]*n_assets), 
                      bounds=bounds, constraints=constraints, method='SLSQP')
    return result.x

# 应用优化算法
n_assets = len(data.columns)
weights = optimize_portfolio(returns, n_assets)
print("最优权重:", weights)
```
