### 6.3.4 系统动力学模型实现

```python
import matplotlib.pyplot as plt

# 系统动力学模型参数
def system_dynamics_model(initial_conditions, parameters, steps):
    results = [initial_conditions]
    for _ in range(steps-1):
        current = results[-1]
        next_val = current * (1 + parameters['growth_rate']) + parameters['feedback']
        results.append(next_val)
    return results

# 参数设置
parameters = {
    'growth_rate': 0.1,
    'feedback': 0.5
}
initial_conditions = 100
steps = 10

# 模型运行
results = system_dynamics_model(initial_conditions, parameters, steps)
plt.plot(results)
plt.title('System Dynamics Model Results')
plt.xlabel('Time Steps')
plt.ylabel('Value')
plt.show()
```
