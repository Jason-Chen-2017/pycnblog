```python
import pulp

# 创建线性规划问题
prob = pulp.LpProblem("Resource Allocation Problem", pulp.LpMinimize)

# 定义变量
x1 = pulp.LpVariable("x1", cat="Integer")
x2 = pulp.LpVariable("x2", cat="Integer")

# 定义目标函数
prob += 10 * x1 + 20 * x2

# 定义约束条件
prob += x1 + x2 <= 100
prob += 2 * x1 + x2 <= 150
prob += x1 >= 0
prob += x2 >= 0

# 解线性规划问题
prob.solve()

# 输出结果
print("Optimal solution:")
print("x1 = ", pulp.value(x1))
print("x2 = ", pulp.value(x2))
print("Objective value = ", pulp.value(prob.objective))
```
