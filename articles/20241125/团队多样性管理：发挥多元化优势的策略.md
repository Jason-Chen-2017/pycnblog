```python
def diversity_index(group):
    """
    Calculate the diversity index of a group.
    
    Args:
    group (list): A list of members in the group.
    
    Returns:
    float: The diversity index.
    """
    unique_members = set(group)
    total_members = len(group)
    diversity_index = 1 - (len(unique_members) / total_members)
    return diversity_index

# Example usage
group = ["Alice", "Bob", "Charlie", "Alice", "Dave"]
print("Diversity Index:", diversity_index(group))
```
