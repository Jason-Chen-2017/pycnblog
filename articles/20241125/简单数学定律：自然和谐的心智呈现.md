# 线性回归模型建立
model = np.polyfit(data.index.to_julian_date(), data['sales'], 1)
x = np.arange(data.index.to_julian_date().min(), data.index.to_julian_date().max() + 1)

predicted_sales = model[0] * data.index.to_julian_date().values + model[1]

window_size = 30  # 设定窗口大小为30天
rolling_mean_sales = data['sales'].rolling(window=window_size).mean()

marginal_cost = np.gradient(rolling_mean_sales)

# 库存优化策略
inventory_levels = data['inventory'].values
new_inventory_levels = []
for i in range(len(inventory_levels)):
