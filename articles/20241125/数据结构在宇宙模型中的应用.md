#include <iostream>
#include <vector>
#include <cmath>

const double G = 6.67430e-11;
const double c = 299792458;

struct Particle {
    double mass;
    std::vector<double> position;
    std::vector<double> velocity;
};

double distance(const std::vector<double>& pos1, const std::vector<double>& pos2) {
    double result = 0;
    for (size_t i = 0; i < pos1.size(); ++i) {
        result += (pos1[i] - pos2[i]) * (pos1[i] - pos2[i]);
    }
    return sqrt(result);
}

void black_hole_collision_simulation(int num_particles, double time_step) {
    std::vector<Particle> particles(num_particles);
    
    // 初始化粒子参数
    for (auto& particle : particles) {
        particle.mass = 1.0;
        particle.position = {rand() % 100 - 50, rand() % 100 - 50, rand() % 100 - 50};
        particle.velocity = {rand() % 10 - 5, rand() % 10 - 5, rand() % 10 - 5};
    }
    
    for (int step = 0; step < time_step; ++step) {
        std::vector<std::vector<std::pair<int, int>>> interactions;
        
        // 计算粒子间的相互作用
        for (size_t i = 0; i < particles.size(); ++i) {
            for (size_t j = i + 1; j < particles.size(); ++j) {
                double distance = distance(particles[i].position, particles[j].position);
                if (distance < 10) {
                    interactions.push_back({i, j});
                }
            }
        }

            particle.position += particle.velocity * time_step;
            particle.velocity = {0, 0, 0};

            int i = interaction.first;
            int j = interaction.second;
