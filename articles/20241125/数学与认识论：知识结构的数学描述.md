import sympy as sp
import pandas as pd

knowledge = sp.Symbol('knowledge')

knowledge_structure = sp.Eq(knowledge, belief + experience)

df = pd.DataFrame(data)

# 模型求解
df['knowledge'] = knowledge_structure.subs(dict(zip([belief, experience], df[['belief', 'experience']]))).evaluate()

# 输出结果
print(df)
