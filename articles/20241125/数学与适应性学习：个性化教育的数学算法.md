#### 线性回归伪代码

# 线性回归伪代码

def linear_regression(X, y):
    # 数据预处理
    X_mean = np.mean(X, axis=0)
    y_mean = np.mean(y)
    X = X - X_mean
    y = y - y_mean
    
    # 计算权重
    w = np.linalg.inv(X.T @ X) @ X.T @ y
    
    # 预测
    y_pred = X @ w
    
    # 评估模型
    mse = np.mean((y_pred - y)**2)
    
    return w, y_pred, mse
```

这个伪代码展示了线性回归算法的完整实现过程，包括数据预处理、权重计算、预测和模型评估。

def decision_tree(X, y, depth=0, max_depth=10):

        return np.mean(y)
    
    # 选择最优特征
    best_feature, best_score = find_best_split(X, y)
    
    # 划分节点
    if best_feature is not None:
        left_indices = X[:, best_feature] < best_score
        right_indices = ~left_indices
        
        # 递归生成子树
        left_y = y[left_indices]
        right_y = y[right_indices]
        
        left_value = decision_tree(X[left_indices], left_y, depth+1, max_depth)
        right_value = decision_tree(X[right_indices], right_y, depth+1, max_depth)
        
        return {
            'feature': best_feature,
            'threshold': best_score,
            'left': left_value,
            'right': right_value
        }
    else:
        return np.mean(y)

def find_best_split(X, y):
    # 计算每个特征的信息增益
    gains = []
    for feature in range(X.shape[1]):
        thresholds = np.unique(X[:, feature])
        for threshold in thresholds:
            left_indices = X[:, feature] < threshold
            right_indices = ~left_indices
            gain = info_gain(y[left_indices], y[right_indices])
            gains.append((gain, feature, threshold))
    
    # 选择最大信息增益的特征和阈值
    best_gain, best_feature, best_threshold = max(gains)
    return best_feature, best_threshold
```

这个伪代码展示了决策树算法的核心实现过程，包括选择最优特征、划分节点和递归生成子树。

# 神经网络伪代码

def neural_network(X, y, hidden_layer_sizes, activation='sigmoid', learning_rate=0.1, epochs=1000):
    # 初始化权重和偏置
    W1, b1 = initialize_weights(X.shape[1], hidden_layer_sizes[0])
    W2, b2 = initialize_weights(hidden_layer_sizes[0], hidden_layer_sizes[1])
    output_layer_weights, output_layer_bias = initialize_weights(hidden_layer_sizes[1], y.shape[1])
    
    # 前向传播
    for epoch in range(epochs):
        z1, a1 = forward_propagation(X, W1, b1, activation=activation)
        z2, a2 = forward_propagation(a1, W2, b2, activation=activation)
        z3, a3 = forward_propagation(a2, output_layer_weights, output_layer_bias)
        
        # 反向传播
        dZ3 = a3 - y
        dW3 = (a2.T @ dZ3) / X.shape[0]
        db3 = np.sum(dZ3, axis=0) / X.shape[0]
        
        dA2 = W3.T @ dZ3
        dZ2 = dA2 * (1 - npderiv(activation(a2)))
        dW2 = (a1.T @ dZ2) / X.shape[0]
        db2 = np.sum(dZ2, axis=0) / X.shape[0]
        
        dA1 = W2.T @ dZ2
        dZ1 = dA1 * (1 - npderiv(activation(a1)))
        dW1 = (X.T @ dZ1) / X.shape[0]
        db1 = np.sum(dZ1, axis=0) / X.shape[0]
        
        # 更新权重和偏置
        W1 -= learning_rate * dW1
        b1 -= learning_rate * db1
        W2 -= learning_rate * dW2
        b2 -= learning_rate * db2
        output_layer_weights -= learning_rate * dW3
        output_layer_bias -= learning_rate * db3
        
        # 计算误差
        mse = np.mean((a3 - y)**2)
        
        if epoch % 100 == 0:
            print(f"Epoch {epoch}: MSE = {mse}")
    
    return W1, b1, W2, b2, output_layer_weights, output_layer_bias
