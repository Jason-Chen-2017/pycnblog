#### 第2章 算法优化基本原理

##### 2.1 优化算法的基本概念

优化算法是指通过迭代过程，逐步寻找最优解的算法。优化算法通常分为以下几种类型：

1. **局部优化算法**：这类算法通过在当前解的邻域内寻找最优解来迭代优化。
2. **全局优化算法**：这类算法通过在整个搜索空间中搜索最优解来迭代优化。
3. **动态优化算法**：这类算法在运行过程中根据环境变化动态调整优化策略。

##### 2.2 优化算法的分类

根据优化目标的不同，优化算法可以分为以下几类：

1. **无约束优化算法**：这类算法在无任何约束条件下寻找最优解。
2. **有约束优化算法**：这类算法在存在约束条件下寻找最优解。
3. **组合优化算法**：这类算法用于解决离散优化问题，如旅行商问题、背包问题等。

##### 2.3 优化算法的性能评估

优化算法的性能评估是衡量算法优劣的重要标准。常用的性能评估指标包括：

1. **求解时间**：算法找到最优解所需的时间。
2. **求解精度**：算法找到的最优解与真实最优解之间的误差。
3. **鲁棒性**：算法在不同输入条件下保持稳定性和效率的能力。

import numpy as np

def fitness_function(x):
    # 定义适应度函数
    return np.sum((x - x_best) ** 2)

def select_parents(pop, fitness):
    # 选择父母
    indices = np.argsort(fitness)
    parents = pop[indices[:2]]
    return parents

def crossover(parents):
    # 交叉操作
    child = parents[0].copy()
    crossover_point = np.random.randint(1, len(child) - 1)
    child[crossover_point:] = parents[1][crossover_point:]
    return child

def mutate(child):
    # 变异操作
    for i in range(len(child)):
        if np.random.rand() < mutation_rate:
            child[i] += np.random.randn() * mutation_step
    return child

def genetic_algorithm(pop_size, mutation_rate, mutation_step):
    # 初始化种群
    pop = np.random.rand(pop_size, n_features)
    x_best = None
    f_best = float('inf')

    # 迭代过程

        fitness = fitness_function(pop)
        parents = select_parents(pop, fitness)
        child = crossover(parents)
        child = mutate(child)
        fitness = fitness_function(child)

import numpy as np

def fitness_function(x):
    # 定义适应度函数
    return np.sum((x - x_best) ** 2)

def update_position(x, v, pBest, gBest, w, c1, c2, r1, r2):
    # 更新位置
    x_new = x + w * v + c1 * r1 * (pBest - x) + c2 * r2 * (gBest - x)
    return x_new

def update_velocity(v, w, c1, c2, r1, r2, pBest, gBest):
    # 更新速度
    v_new = w * v + c1 * r1 * (pBest - x) + c2 * r2 * (gBest - x)
    return v_new

def particle_swarm_optimization(pop_size, w, c1, c2, max_iterations):
    # 初始化粒子群
    pos = np.random.rand(pop_size, n_features)
    velocity = np.zeros((pop_size, n_features))
    pBest = pos.copy()
    gBest = pos[0].copy()
    f_best = fitness_function(gBest)

    # 迭代过程
    for _ in range(max_iterations):
        for i in range(pop_size):
            fitness = fitness_function(pos[i])
            if fitness < fitness_function(pBest[i]):
                pBest[i] = pos[i]
            if fitness < f_best:
                gBest = pos[i]
                f_best = fitness

        for i in range(pop_size):
            velocity = update_velocity(velocity[i], w, c1, c2, r1, r2, pBest[i], gBest)
            pos[i] = update_position(pos[i], velocity[i], pBest[i], gBest, w, c1, c2, r1, r2)

    return gBest

# 运行算法
best_solution = particle_swarm_optimization(pop_size=100, w=0.5, c1=1.5, c2=1.5, max_iterations=100)
print("Best solution:", best_solution)
