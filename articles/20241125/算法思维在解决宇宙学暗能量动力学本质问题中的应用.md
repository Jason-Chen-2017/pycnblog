```python
import numpy as np
import matplotlib.pyplot as plt

# 定义宇宙膨胀模型函数
def expand_universe(Lambda, w, a_0, H_0, n_steps):
    M_pl = 1e-5  # 引力常数
    a = np.zeros(n_steps)
    H = np.zeros(n_steps)
    a[0] = a_0
    H[0] = H_0
    for i in range(1, n_steps):
        a[i] = a[i-1] * (1 + H[i-1])
        H[i] = Lambda / (3 * M_pl**2 * a[i]**3)
    return a, H

# 设置参数
Lambda = 0.69
w = -1
a_0 = 1
H_0 = 100
n_steps = 100

# 运行宇宙膨胀模型

# 绘制宇宙膨胀率随时间的变化图

plt.xlabel('宇宙尺度因子 a')
plt.ylabel('宇宙膨胀率 H')
