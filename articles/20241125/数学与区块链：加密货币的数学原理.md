### 1.1.2 区块链的核心特点

区块链具有以下几个核心特点：

1. **去中心化**：区块链技术通过去中心化的方式，避免了传统系统中的单点故障问题，提高了系统的可靠性和抗攻击能力。
2. **透明性**：区块链上的所有交易都是公开透明的，每个节点都可以验证和查看交易记录，保证了数据的真实性。
3. **不可篡改**：区块链上的数据一旦被记录，就无法被篡改或删除，确保了数据的持久性和完整性。
4. **安全性**：区块链技术采用了多种密码学算法，确保了数据的安全性和隐私性。

#### 核心概念与联系

区块链技术的核心概念包括：

- **分布式账本**：区块链本质上是一个分布式数据库，所有节点都维护一份完整的账本记录。
- **区块**：区块是区块链的基本单元，包含一定数量的交易记录，并具有时间戳等信息。
- **链**：多个区块按照时间顺序链接在一起，形成区块链。

#### 核心算法原理讲解

区块链技术中的关键算法包括：

- **工作量证明（PoW）**：用于确定下一个区块的生成者，通过计算耗费大量计算资源的难题来证明工作量，确保区块链的安全性和一致性。
- **共识算法**：用于确保区块链上所有节点对于区块的添加达成一致，如PoW、权益证明（PoS）等。

// Blockchain.go
package main

import (
    "crypto/sha256"
    "encoding/hex"
    "math"
    "time"
)

// Block represents each block in the blockchain
type Block struct {
    Index     int
    Timestamp string
    Transactions []string
    PrevHash  string
    Hash      string
}

// Blockchain represents the chain of blocks
type Blockchain struct {
    chain []*Block
    difficulty int
}

// NewBlockchain creates a new blockchain with the genesis block
func NewBlockchain(difficulty int) *Blockchain {
    genesisBlock := CreateGenesisBlock()
    bc := &Blockchain{[]*Block{genesisBlock}, difficulty}
    return bc
}

// CreateGenesisBlock creates the genesis block
func CreateGenesisBlock() *Block {
    return &Block{
        Index:       0,
        Timestamp:   time.Now().Format("02/01/2006 15:04:05"),
        Transactions: []string{},
        PrevHash:    "",
        Hash:       "",
    }
}

// AddBlock adds a new block to the blockchain
func (bc *Blockchain) AddBlock(transactions []string) {
    prevBlock := bc.chain[len(bc.chain)-1]
    newBlock := &Block{
        Index:       prevBlock.Index + 1,
        Timestamp:   time.Now().Format("02/01/2006 15:04:05"),
        Transactions: transactions,
        PrevHash:    prevBlock.Hash,
        Hash:       "",
    }

    newBlock.Hash = calculateHash(newBlock)

    bc.chain = append(bc.chain, newBlock)
}

// calculateHash calculates the SHA256 hash of a block
func calculateHash(block *Block) string {
    record := string(block.Index) + block.Timestamp + strings.Join(block.Transactions, "") + block.PrevHash
    hash := sha256.Sum256([]byte(record))
    return hex.EncodeToString(hash[:])
}

// validProof works to find a hash that has the required 0's before the hash
func validProof(prevHash string, proof int, difficulty int) bool {
    hash := sha256.Sum256([]byte(prevHash + string(proof)))
    fmt.Printf("Hash: %x\n", hash)
    return hex.EncodeToString(hash[:])[:difficulty] == strings.Repeat("0", difficulty)
}

// mineBlock finds a valid proof of work for a block
func (bc *Blockchain) mineBlock(transactions []string) *Block {
    prevBlock := bc.chain[len(bc.chain)-1]

    proof := 0
    start := time.Now()
    for !validProof(prevBlock.Hash, proof, bc.difficulty) {
        proof++
    }
    end := time.Now()

    newBlock := &Block{
        Index:       prevBlock.Index + 1,
        Timestamp:   end.Format("02/01/2006 15:04:05"),
        Transactions: transactions,
        PrevHash:    prevBlock.Hash,
        Hash:       calculateHash(newBlock),
    }

    bc.AddBlock(newBlock.Transactions)

    return newBlock
}

// BlockchainMain is the main function to run the blockchain
func BlockchainMain() {
    difficulty := 4
    bc := NewBlockchain(difficulty)

    transactions := []string{"0->1", "2->3"}
    newBlock := bc.mineBlock(transactions)

    for i := 0; i < len(bc.chain); i++ {
        fmt.Printf("Block %d\n", i)
        fmt.Println.bc.chain[i].String())
    }
}
```
