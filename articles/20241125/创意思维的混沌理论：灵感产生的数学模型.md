def logistic_map(r, x0):
    x = x0
    for _ in range(100):
        x = r * x * (1 - x)
    return x

r_values = np.linspace(2.5, 4, 1000)
x0 = 0.1
x_values = [logistic_map(r, x0) for r in r_values]
