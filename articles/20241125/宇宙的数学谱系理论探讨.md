```python
import numpy as np

# 定义常数
G = 6.674 * 10**-11  # 万有引力常数
M = 5.972 * 10**24  # 地球质量
R = 6.371 * 10**6  # 地球半径

# 计算密度变化
def density_change(r):
    return M * (1 - r / R)

# 计算引力势能
def potential_energy(r):
    return G * M / r

# 计算总能量
def total_energy(m, r):
    return 0.5 * m * np.square(np.linalg.norm(r)) + potential_energy(r)

# 测试代码

m = 5.972 * 10**24  # 地球质量

# 输出结果

print("引力势能：", potential_energy(r))
print("总能量：", total_energy(m, r))
```
