```python
# 线性规划示例代码
import numpy as np
from scipy.optimize import linprog

# 目标函数系数
c = [-1, -2]  # 目标是最小化-z = -x1 - 2x2

# 约束条件系数矩阵
A = [[1, 1], [2, 2]]  # 约束条件 Ax <= b

# 约束条件常数向量
b = [4, 6]  # 约束条件 b

# 边界条件
x0 = [0, 0]  # x0 为非负变量
x1 = [2, 3]  # x1 为非负变量

# 求解线性规划问题
result = linprog(c, A_ub=A, b_ub=b, bounds=(x0, x1), method='highs')

# 输出结果
print("最优解:", result.x)
print("目标值:", -result.fun)
```
