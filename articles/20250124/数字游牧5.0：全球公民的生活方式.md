```python
import sqlite3
from sqlite3 import Error

def create_connection():
    conn = None
    try:
        conn = sqlite3.connect(':memory:')
        print(sqlite3.version)
    except Error as e:
        print(e)
    return conn

def create_table(conn):
    try:
        query = """ CREATE TABLE IF NOT EXISTS tasks (
                    id integer PRIMARY KEY,
                    title text NOT NULL,
                    description text,
                    completed boolean
                ); """
        cursor = conn.cursor()
        cursor.execute(query)
    except Error as e:
        print(e)

def insert_task(conn, task):
    try:
        cursor = conn.cursor()
        query = """INSERT INTO tasks(title, description, completed) VALUES(?,?,?);"""
        cursor.execute(query, task)
        conn.commit()
        return cursor.lastrowid
    except Error as e:
        print(e)

def get_all_tasks(conn):
    try:
        cursor = conn.cursor()
        query = "SELECT * FROM tasks"
        cursor.execute(query)
        rows = cursor.fetchall()
        return rows
    except Error as e:
        print(e)

def update_task(conn, task):
    try:
        cursor = conn.cursor()
        query = """UPDATE tasks set title=?, description=?, completed=? WHERE id=?;"""
        cursor.execute(query, task)
        conn.commit()
    except Error as e:
        print(e)

def delete_task(conn, task_id):
    try:
        cursor = conn.cursor()
        query = "DELETE FROM tasks WHERE id=?;"
        cursor.execute(query, (task_id,))
        conn.commit()
    except Error as e:
        print(e)

def main():
    conn = create_connection()
```

```Dockerfile
FROM python:3.8

RUN apt-get update && apt-get install -y \
    postgresql-client \
    redis-server \
    build-essential

WORKDIR /app

COPY requirements.txt .

RUN pip install -r requirements.txt

COPY . .

CMD ["python", "manage.py", "runserver", "0.0.0.0:8000"]
```

接着，创建一个名为`docker-compose.yml`的文件，内容如下：

```yaml
version: '3.8'

services:
  web:
    build: .
    command: gunicorn task_manager.wsgi:application --bind 0.0.0.0:8000
    volumes:
      - .:/app
    ports:
      - "8000:8000"
    depends_on:
      - db
      - redis

  db:
    image: postgres:12
    environment:
      POSTGRES_DB: task_manager
      POSTGRES_USER: admin
      POSTGRES_PASSWORD: admin
    volumes:
      - postgres_data:/var/lib/postgresql/data/

  redis:
    image: redis:6
    volumes:
      - redis_data:/data/

volumes:
  postgres_data:
  redis_data:
```

```bash
docker-compose up --build
```

   ```json
       "password": "password123"
   ```

   ```json
   {
       "email": "jane.doe@example.com",
       "password": "password123"
   }
   ```

   ```json
       "title": "学习Python",
       "description": "每天学习一小时",
       "completed": false,
       "priority": 1,
       "category_id": 1,
       "tag_ids": [1, 2]
   }
   ```

#### **步骤 6：更新任务**

更新任务，发送一个PUT请求到`/tasks/{id}/`接口，请求头包含`Authorization: Bearer {token}`，请求体如下：

```json
{
    "title": "学习Python进阶",
    "description": "每天学习两小时",
    "completed": false,
    "priority": 1,
    "category_id": 1,
    "tag_ids": [1, 2]
}
```

#### **步骤 7：删除任务**

删除任务，发送一个DELETE请求到`/tasks/{id}/`接口，请求头包含`Authorization: Bearer {token}`。
