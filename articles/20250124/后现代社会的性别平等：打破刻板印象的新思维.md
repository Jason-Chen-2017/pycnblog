from flask import Flask, request, jsonify

from flask_migrate import Migrate

app = Flask(__name__)

db = SQLAlchemy(app)
migrate = Migrate(app, db)

    user = User(username=username, gender=gender)
    db.session.add(user)
    db.session.commit()
    return jsonify({'message': 'User created successfully'})

    posts = Post.query.all()
    return jsonify({'posts': [{'id': post.id, 'content': post.content, 'user': user.username} for post in posts for user in User.query.filter(User.id == post.user_id)]})

if __name__ == '__main__':
    app.run(debug=True)
