#### 2. 系统核心实现

以下是数字nomad部落集会组织系统的核心实现代码，包括用户登录、注册、提交偏好、获取集会安排等功能：

```python
# app.py

from flask import Flask, request, jsonify
from flask_sqlalchemy import SQLAlchemy

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///nomad.db'
db = SQLAlchemy(app)

class User(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    username = db.Column(db.String(80), unique=True, nullable=False)
    password = db.Column(db.String(120), nullable=False)
    preferences = db.Column(db.String(120))

class Preference(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    user_id = db.Column(db.Integer, db.ForeignKey('user.id'), nullable=False)
    preference = db.Column(db.String(120))

class Meeting(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    title = db.Column(db.String(120), nullable=False)
    start_time = db.Column(db.DateTime, nullable=False)
    end_time = db.Column(db.DateTime, nullable=False)
    venue = db.Column(db.String(120))

@app.route('/register', methods=['POST'])
def register():
    username = request.form['username']
    password = request.form['password']
    user = User(username=username, password=password)
    db.session.add(user)
    db.session.commit()
    return jsonify({'status': 'success'})

@app.route('/login', methods=['POST'])
def login():
    username = request.form['username']
    password = request.form['password']
    user = User.query.filter_by(username=username, password=password).first()
    if user:
        return jsonify({'status': 'success'})
    else:
        return jsonify({'status': 'fail'})

@app.route('/submit_preference', methods=['POST'])
def submit_preference():
    user_id = request.form['user_id']
    preference = request.form['preference']
    preference = Preference(user_id=user_id, preference=preference)
    db.session.add(preference)
    db.session.commit()
    return jsonify({'status': 'success'})

@app.route('/get_meetings', methods=['GET'])
def get_meetings():
    user_id = request.args.get('user_id')
    preferences = Preference.query.filter_by(user_id=user_id).all()
    meetings = Meeting.query.all()
    meeting_prefs = {}
    for preference in preferences:
        meeting_prefs[preference.id] = preference.preference
    recommended_meetings = []
    for meeting in meetings:
        if meeting_prefs.get(meeting.id, False):
            recommended_meetings.append(meeting)
    return jsonify({'meetings': recommended_meetings})

if __name__ == '__main__':
    db.create_all()
    app.run(debug=True)
```

- 用户注册：通过`/register`接口接收用户名和密码，将用户信息存储到数据库中。
- 用户登录：通过`/login`接口接收用户名和密码，验证用户信息，返回登录状态。

- 获取集会安排：通过`/get_meetings`接口接收用户ID，获取用户偏好，推荐符合偏好的集会信息。
