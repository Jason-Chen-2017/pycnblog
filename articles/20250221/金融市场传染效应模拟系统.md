```python
nx.draw(G, with_labels=True, node_color='red', edge_color='blue', alpha=0.8)
plt.show()

# 初始化感染状态
initial_infection = [0, 1, 0, 0, 1]
state = np.array(initial_infection)

# 传播概率
beta = 0.3

# 模拟传播过程
for t in range(10):
    new_infection = np.zeros_like(state)
    for i in range(len(state)):
        if state[i] == 1:
            for neighbor in G.neighbors(i):
                if state[neighbor] == 0:
                    new_infection[neighbor] = 1
    state += new_infection
    print(f"Time step {t}: {state}")
```
