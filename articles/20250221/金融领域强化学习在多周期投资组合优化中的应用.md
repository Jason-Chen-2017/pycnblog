### 3.4.1 Python代码实现
```python
import numpy as np
import gym
from gym import spaces
from gym.utils import seeding

class PortfolioEnv(gym.Env):
    def __init__(self, n_assets=3):
        super(PortfolioEnv, self).__init__()
        self.n_assets = n_assets
        self.observation_space = spaces.Box(low=0, high=1, shape=(self.n_assets,))
        self.action_space = spaces.Box(low=0, high=1, shape=(self.n_assets,))
        self.current_obs = None

    def _take_action(self, action):
        # 计算投资组合的收益
        return np.sum(action * self.current_obs)

    def reset(self):
        # 初始化环境
        self.current_obs = np.random.rand(self.n_assets,)
        return self.current_obs

    def step(self, action):
        # 执行动作并返回新的状态、奖励、终止标志
        reward = self._take_action(action)
        done = False
        return self.current_obs, reward, done, {}
```

from gym import wrappers

def train_agent(env, num_episodes=1000):
    agent = DQNAgent(env)
    for episode in range(num_episodes):
        state = env.reset()
        total_reward = 0
        while True:
            action = agent.act(state)
            next_state, reward, done, _ = env.step(action)
            agent.remember(state, action, reward, next_state, done)
            agent.replay()
            state = next_state
            total_reward += reward
            if done:
                break
        print(f"Episode {episode}, Total Reward: {total_reward}")
