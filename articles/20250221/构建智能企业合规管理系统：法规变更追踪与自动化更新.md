```python
import requests
from bs4 import BeautifulSoup

response = requests.get(url)

text = soup.get_text()
```

#### 5.2.2 实体识别
```python
from nltk import word_tokenize, pos_tag
from nltk.chunk import ne_chunk

tokens = word_tokenize(text)
tags = pos_tag(tokens)
chunks = ne_chunk(tags)
```
