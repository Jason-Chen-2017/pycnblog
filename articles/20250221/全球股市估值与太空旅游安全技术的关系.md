```python
import pandas as pd
import numpy as np

def calculate_market_valuation(safety_technology_score):
    return 100 * safety_technology_score

# 示例数据
data = {
    "公司名称": ["公司A", "公司B"],
    "安全技术评分": [0.8, 0.9]
}

df = pd.DataFrame(data)
df["市场估值"] = df["安全技术评分"].apply(calculate_market_valuation)
print(df)
```
