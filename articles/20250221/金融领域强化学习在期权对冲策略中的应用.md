  import numpy as np
  import gym
  from gym import spaces
  from gym.utils import seeding

  class OptionTradingEnv(gym.Env):
      def __init__(self):
          self.state = None
          self.action_space = spaces.Discrete(3)  # 0: sell, 1: hold, 2: buy
          self.observation_space = spaces.Box(low=0, high=1, shape=(2,))
          self.seed()

      def seed(self, seed=None):
          self.np_random, seed = seeding.np_random(seed)
          return seed

      def reset(self):
          # 初始化状态，如资产价格和波动率
          self.state = self.np_random.uniform(low=0, high=1, size=(2,))
          return self.state

      def step(self, action):
          # 根据动作更新状态并返回奖励
          # action 0: sell, action 1: hold, action 2: buy
          # 假设波动率影响收益，这里简化处理
          if action == 0:
              reward = -self.state[0] * self.state[1]
          elif action == 1:
              reward = 0
          else:
              reward = self.state[0] * self.state[1]
          # 更新状态，简化为随机波动
          self.state = self.np_random.uniform(low=0, high=1, size=(2,))
          return self.state, reward, False, {}
  ```
