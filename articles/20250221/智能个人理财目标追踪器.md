from sklearn.linear_model import LinearRegression

# 示例数据
X = np.array([[1], [2], [3], [4]])

# 模型训练
model = LinearRegression()
model.fit(X, y)

# 预测
print(model.predict([[5]]))  # 输出: [[5.5]]
```

- 3.2.2 决策树实现

# 模型训练

# 预测
print(model.predict([[1, 1]]))  # 输出: [0]
```
