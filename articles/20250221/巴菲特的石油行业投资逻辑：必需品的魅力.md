def calculate_intrinsic_value(financial_data):
    # 计算内在价值
    gdp_growth = financial_data['gdp_growth']
    roe = financial_data['roe']
    book_value = financial_data['book_value']
    # 内在价值公式
    intrinsic_value = (book_value * (1 + gdp_growth / 100)) * roe
    return intrinsic_value
