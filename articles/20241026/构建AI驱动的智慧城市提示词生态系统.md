    forests = []
    for _ in range(n_estimators):
        sample = sample_data(data)
        tree = build_decision_tree(sample, all_features)
        forests.append(tree)
    return forests

// 样本数据抽样函数
function sample_data(data, sample_size):
    indices = random.sample(range(len(data)), sample_size)
    return [data[i] for i in indices]
```

// 前向传播函数
function forwardPropagation(input, weights, biases, activation_functions):
    layer_outputs = [input]
    for layer, activation_function in zip(layers, activation_functions):
        output = activation_function(dot_product(weights[layer], layer_outputs[-1]) + biases[layer])
        layer_outputs.append(output)
    return layer_outputs[-1]

// 反向传播函数
function backwardPropagation(output, expected_output, weights, biases, activation_functions):
    d_weights = []
    d_biases = []
    for layer in reversed(layers):
        d_output = output - expected_output
        d_input = d_output * activation_function_derivative(output)
        d_weights.append(d_input.dot(previous_layer))
        d_biases.append(d_input)
        output = previous_layer
    d_weights.reverse()
    d_biases.reverse()
    return d_weights, d_biases
```
