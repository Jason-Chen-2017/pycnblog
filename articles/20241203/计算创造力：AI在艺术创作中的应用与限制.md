### 数学模型和数学公式

在艺术创作中，生成对抗网络（GAN）是一种常用的算法，其核心包括生成器和判别器。生成器的数学模型可以用以下公式表示：

$$ G(z) = \mu_G(z) + \sigma_G(z) \odot \text{sigmoid}(\text{W}_G \cdot z + b_G) $$

其中，$z$ 是从先验分布中抽取的噪声向量，$G(z)$ 是生成器生成的样本。$\mu_G(z)$ 和 $\sigma_G(z)$ 分别是生成器的均值和方差，$W_G$ 和 $b_G$ 是生成器的权重和偏置，$\odot$ 表示逐元素乘积，$\text{sigmoid}$ 是一个激活函数。

判别器的数学模型如下：

$$ D(x) = \text{sigmoid}(\text{W}_D \cdot x + b_D) $$
$$ D(G(z)) = \text{sigmoid}(\text{W}_D \cdot G(z) + b_D) $$

其中，$x$ 是真实数据，$G(z)$ 是生成器生成的样本，$\text{W}_D$ 和 $b_D$ 是判别器的权重和偏置。

# gan_model.py

import numpy as np
import tensorflow as tf
from tensorflow.keras.models import Model
from tensorflow.keras.layers import Input, Dense, Reshape, Flatten
from tensorflow.keras.optimizers import Adam

# 生成器模型
def build_generator(z_dim):
    input_z = Input(shape=(z_dim,))
    x = Dense(128, activation='relu')(input_z)
    x = Dense(256, activation='relu')(x)
    x = Dense(512, activation='relu')(x)
    x = Dense(1024, activation='relu')(x)
    x = Dense(128 * 128 * 3, activation='tanh')(x)
    x = Reshape((128, 128, 3))(x)
    generator = Model(inputs=input_z, outputs=x)
    return generator

# 判别器模型
def build_discriminator(image_shape):
    image = Input(shape=image_shape)
    x = Flatten()(image)
    x = Dense(1024, activation='relu')(x)
    x = Dense(512, activation='relu')(x)
    x = Dense(256, activation='relu')(x)
    x = Dense(128, activation='relu')(x)
    validity = Dense(1, activation='sigmoid')(x)
    discriminator = Model(inputs=image, outputs=validity)
    return discriminator

# GAN模型
def build_gan(generator, discriminator):
    discriminator.trainable = False
    gan_output = discriminator(generator(z))
    gan = Model(inputs=generator.input, outputs=gan_output)
    return gan

# 模型编译和训练
def compile_and_train(generator, discriminator, gan, z_dim, batch_size, epochs):
    discriminator_optimizer = Adam(0.0001)
    generator_optimizer = Adam(0.0001)
    gan_optimizer = Adam(0.0002)

    discriminator.compile(loss='binary_crossentropy', optimizer=discriminator_optimizer)
    gan.compile(loss='binary_crossentropy', optimizer=gan_optimizer)

    for epoch in range(epochs):
        for _ in range(batch_size):
            z = np.random.normal(size=(1, z_dim))
            real_images = np.random.choice(real_images, size=batch_size)
            fake_images = generator.predict(z)

            # 训练判别器
            d_loss_real = discriminator.train_on_batch(real_images, np.ones((batch_size, 1)))
            d_loss_fake = discriminator.train_on_batch(fake_images, np.zeros((batch_size, 1)))
            d_loss = 0.5 * np.add(d_loss_real, d_loss_fake)

            # 训练生成器
            g_loss = gan.train_on_batch(z, np.ones((batch_size, 1)))

        print(f"Epoch {epoch}, Generator Loss: {g_loss}, Discriminator Loss: {d_loss}")

# 主程序
if __name__ == '__main__':
    z_dim = 100
    image_shape = (128, 128, 3)
    batch_size = 64
    epochs = 100

    generator = build_generator(z_dim)
    discriminator = build_discriminator(image_shape)
    gan = build_gan(generator, discriminator)

    compile_and_train(generator, discriminator, gan, z_dim, batch_size, epochs)

    # 生成图像
