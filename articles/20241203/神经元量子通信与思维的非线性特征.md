import numpy as np
import matplotlib.pyplot as plt

# 定义非线性动力学模型
def nonlinear_dynamics(x, y, a, b, c):
    dxdt = a * (y - x)
    dydt = b * x - c * y
    return dxdt, dydt

# 初始化参数
a = 0.1
b = 0.2
c = 0.3

# 设置时间步长和模拟时间
dt = 0.01
t_max = 100

# 初始化变量
x = np.zeros(t_max)
y = np.zeros(t_max)

# 模拟非线性动力学过程
for t in range(1, t_max):
    dxdt, dydt = nonlinear_dynamics(x[t-1], y[t-1], a, b, c)
    x[t] = x[t-1] + dxdt * dt
    y[t] = y[t-1] + dydt * dt

# 绘制结果
plt.plot(x, y)
plt.xlabel('x')
plt.ylabel('y')
plt.title('Nonlinear Dynamics')
plt.show()
