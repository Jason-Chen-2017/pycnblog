import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

def calculate_value_market(market_data):
    # 筛选本地市场数据
    local_data = market_data[market_data['region'] == '本地']
    # 计算市盈率与市净率
    local_data['市盈率'] = local_data['股价'] / local_data['每股收益']
    local_data['市净率'] = local_data['股价'] / local_data['每股净资产']
    # 计算投资价值
    investment_value = local_data['市盈率'] * local_data['市净率']
    return investment_value

# 示例数据
market_data = pd.DataFrame({
    '公司': ['本地公司1', '本地公司2', '外地公司1'],
    '股价': [50, 60, 70],
    '每股收益': [5, 6, 7],
    '每股净资产': [10, 12, 14],
    'region': ['本地', '本地', '外地']
})

investment_value = calculate_value_market(market_data)
print(investment_value)
