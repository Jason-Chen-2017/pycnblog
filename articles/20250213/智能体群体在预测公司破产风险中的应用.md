class Agent:
    def __init__(self, id):

            self.decision = '非破产'

### 4.2.2 群体决策类
```python
class GroupDecision:
    def __init__(self, agents):
        self.agents = agents

    def 收集数据(self, data):
        for agent in self.agents:
            agent.感知环境(data)

    def 处理数据(self):
        decisions = [agent.决策() for agent in self.agents]
        from collections import Counter
        result = Counter(decisions)
        return result

    def 输出预测(self):
        result = self.处理数据()
        return result.most_common(1)[0][0]
```
