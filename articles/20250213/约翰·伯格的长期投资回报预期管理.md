  ```python
  import numpy as np
  from scipy.optimize import minimize

  def portfolio_optimization(returns, weights):
      def objective(weights):
          return np.dot(weights.T @ returns @ weights)
      constraints = {'sum(weights) == 1', 'weights >= 0'}
      result = minimize(objective, weights, constraints)
      return result.x
  ```
