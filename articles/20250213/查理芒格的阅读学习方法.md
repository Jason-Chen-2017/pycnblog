```python
# 知识点类
class KnowledgePoint:
    def __init__(self, id, content):
        self.id = id
        self.content = content
        self.related_points = []

# 知识框架类
class KnowledgeFramework:
    def __init__(self):
        self.points = {}

    def add_point(self, point):
        self.points[point.id] = point

    def connect_points(self, point1, point2):
        point1.related_points.append(point2)
        point2.related_points.append(point1)

# 用户类
class User:
    def __init__(self, name):
        self.name = name
        self.knowledge_framework = KnowledgeFramework()

# 示例用法
user = User("Charlie")
point1 = KnowledgePoint(1, "投资的基本原则")
point2 = KnowledgePoint(2, "心理学与行为经济学")
user.knowledge_framework.add_point(point1)
user.knowledge_framework.add_point(point2)
user.knowledge_framework.connect_points(point1, point2)
```
