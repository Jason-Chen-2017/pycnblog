#### 5.2.2 模型训练代码
```python
from tensorflow.keras import layers, Model

def build_model():
    # 编码器部分
    encoder_input = layers.Input(shape=(None, 100))
    encoder = layers.LSTM(128, return_sequences=True)(encoder_input)
    # 解码器部分
    decoder_input = layers.Input(shape=(None, 100))
    decoder = layers.LSTM(128, return_sequences=True)(decoder_input)
    # 注意力机制
    attention = layers.Attention()([encoder, decoder])
    # 输出层
    output = layers.Dense(50, activation='softmax')(attention)
    model = Model([encoder_input, decoder_input], output)
    return model
```
