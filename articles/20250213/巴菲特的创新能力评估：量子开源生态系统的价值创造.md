### 3.2 Python代码实现

```python
import numpy as np

def quantum_add(a, b):
    # 创建量子叠加态
    q_a = np.array([a, b]) / np.sqrt(2)
    q_b = np.array([b, a]) / np.sqrt(2)
    
    # 量子门操作
    cnot = np.array([[1, 0], [0, 1], [0, 1], [1, 0]])
    
    # 应用CNOT门
    result = np.dot(cnot, q_a)
    result = np.dot(cnot, q_b)
    
    # 测量结果
    return result

# 示例
print(quantum_add(1, 0))
```
