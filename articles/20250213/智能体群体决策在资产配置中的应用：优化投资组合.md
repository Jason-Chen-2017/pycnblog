```python
import numpy as np

def pso_optimizer(n_particles, dimensions, bounds, max_iter):
    # 初始化粒子群
    particles = np.random.uniform(bounds[0], bounds[1], (n_particles, dimensions))
    # 全局最优解
    global_best = np.zeros(dimensions)
    # 粒子最优解
    personal_best = np.zeros((n_particles, dimensions))
    # 计算初始适应度
    fitness = np.zeros(n_particles)
    
    for i in range(max_iter):
        for j in range(n_particles):
            # 计算适应度
            fitness[j] = calculate_fitness(particles[j])
            # 更新个人最优
            if fitness[j] > calculate_fitness(personal_best[j]):
                personal_best[j] = particles[j]
        # 更新全局最优
        if np.mean(fitness) > calculate_fitness(global_best):
            global_best = np.mean(personal_best, axis=0)
        # 更新速度和位置
        particles += np.random.rand(n_particles, dimensions) * (personal_best - particles)
```

```python
import numpy as np

def calculate_fitness(position):
    # 计算适应度函数，此处简化为收益
    return np.sum(position)

def pso_optimizer(n_particles, dimensions, bounds, max_iter):
    particles = np.random.uniform(bounds[0], bounds[1], (n_particles, dimensions))
    personal_best = particles.copy()
    global_best = particles.mean(axis=0)
    
    for _ in range(max_iter):
        fitness = np.apply_along_axis(calculate_fitness, 1, particles)
        if np.max(fitness) > calculate_fitness(global_best):
            global_best = personal_best[np.argmax(fitness)]
        # 更新速度和位置
        velocity = np.random.rand(n_particles, dimensions)
        particles += velocity
    return global_best

# 示例调用
n_particles = 10
dimensions = 3
bounds = (0, 1)
max_iter = 100
result = pso_optimizer(n_particles, dimensions, bounds, max_iter)
print("最优解:", result)
```
