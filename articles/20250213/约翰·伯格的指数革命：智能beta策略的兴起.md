```python
import numpy as np
import pandas as pd

def optimize_weights(factor_data, risk_aversion):
    # 计算协方差矩阵
    covariance = factor_data.cov()
    # 构建目标函数
    def objective(weights):
        return 0.5 * weights.T @ covariance @ weights
    # 约束条件：权重之和为1
    constraints = [{'type': 'eq', 'fun': lambda w: np.sum(w) - 1}]
    # 使用最小二乘法求解
    from scipy.optimize import minimize
    result = minimize(objective, np.ones(len(factor_data.columns)) / len(factor_data.columns), constraints=constraints)
    return result.x

# 示例数据
factor_data = pd.DataFrame({
    'MKT': [1, 2, 3, 4, 5],
    'SMB': [2, 3, 4, 5, 6],
    'HML': [3, 4, 5, 6, 7]
})
weights = optimize_weights(factor_data, 1)
print(weights)
```
