    ```python
    import pandas as pd
    from sklearn.ensemble import RandomForestRegressor

    def predict_growth_potential(data):
        # 数据预处理
        data = data.dropna()
        features = data[['revenue_growth', 'profit_margin', 'market_share']]
        target = data['growth_potential']

        # 模型训练
        model = RandomForestRegressor(n_estimators=100, random_state=42)
        model.fit(features, target)

        return model

    # 示例数据
    sample_data = pd.DataFrame({
        'revenue_growth': [0.2, 0.3, 0.1],
        'profit_margin': [0.15, 0.2, 0.1],
        'market_share': [0.3, 0.4, 0.2],
        'growth_potential': [0.7, 0.8, 0.6]
    })

    # 训练模型
    model = predict_growth_potential(sample_data)
    ```
