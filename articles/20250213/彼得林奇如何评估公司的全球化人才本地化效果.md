      global_weight = 0.6
      local_weight = 0.4
      global_score = calculate_global_score(global_skills)
      local_score = calculate_local_score(local_market_knowledge)
      total_score = global_weight * global_score + local_weight * local_score
      return total_score

  def calculate_global_score(global_skills):
      score_map = {"leadership": 0.8, "cross-cultural_communication": 0.7}
      total = sum(score_map[skill] for skill in global_skills)
      return total / len(global_skills)

  def calculate_local_score(local_market_knowledge):
      score_map = {"advanced": 0.9, "moderate": 0.7, "basic": 0.5}
      return score_map[local_market_knowledge]
