```python
def depth_first_search(graph, start, goal):
    stack = [(start, [])]
    while stack:
        node, path = stack.pop()
        if node == goal:
            return path + [node]
        for neighbor in graph[node]:
            new_path = path + [node]
            stack.append((neighbor, new_path))
    return None
```
