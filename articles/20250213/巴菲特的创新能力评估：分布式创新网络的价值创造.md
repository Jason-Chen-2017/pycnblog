#### 4.3 算法实现的Python代码
```python
def calculate_innovation_capacity(actors, resources, processes, results):
    # 计算创新主体活跃度
    actor_score = sum([actor['活跃度'] for actor in actors])
    # 计算创新资源利用率
    resource_score = sum([resource['利用率'] for resource in resources])
    # 计算创新过程效率
    process_score = sum([process['效率'] for process in processes])
    # 计算创新成果质量
    result_score = sum([result['质量'] for result in results])
    # 综合评分
    total_score = (actor_score + resource_score + process_score + result_score) / 4
    return total_score

# 示例数据
actors = [{'活跃度': 0.8}, {'活跃度': 0.7}]
resources = [{'利用率': 0.75}, {'利用率': 0.8}]
processes = [{'效率': 0.85}, {'效率': 0.9}]
results = [{'质量': 0.9}, {'质量': 0.85}]

# 调用算法
total_score = calculate_innovation_capacity(actors, resources, processes, results)
print(f"创新能力综合评分为: {total_score}")
```
