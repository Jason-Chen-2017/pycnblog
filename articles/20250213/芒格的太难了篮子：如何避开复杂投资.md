def calculate_complexity(risk_factors):
    weights = [0.3, 0.2, 0.25, 0.15, 0.1]
    scores = [8, 7, 6, 5, 4]
    total = 0
    for i in range(len(risk_factors)):
        total += weights[i] * scores[i] if risk_factors[i] else 0
    return total

# 示例：风险因素包括市场波动、行业风险、公司治理、财务状况、宏观经济
risk_factors = [True, True, True, False, True]
complexity_score = calculate_complexity(risk_factors)
print(f"复杂性评分：{complexity_score}")
