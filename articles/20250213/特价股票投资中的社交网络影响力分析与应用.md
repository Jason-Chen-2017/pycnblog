```python
import numpy as np
from collections import deque

def influence_spread(graph, initial_nodes):
    n = len(graph)
    infected = np.zeros(n, dtype=int)
    for node in initial_nodes:
        infected[node] = 1
    queue = deque(initial_nodes)
    while queue:
        u = queue.popleft()
        for v in graph[u]:
            if infected[v] == 0:
                infected[v] = 1
                queue.append(v)
    return infected
```
