### 第3章: 对抗学习的数学模型与算法原理

#### 3.1 对抗学习的数学模型
- 3.1.1 生成器的损失函数
$$ L_G = \mathbb{E}_{z \sim p_z} [\log D(G(z))] $$
- 3.1.2 判别器的损失函数
$$ L_D = \mathbb{E}_{x \sim p_x} [\log(1 - D(G(z)))] + \mathbb{E}_{z \sim p_z} [\log D(G(z))] $$
- 3.1.3 综合损失函数
$$ L = L_G + L_D $$

#### 3.3 对抗学习的Python实现示例
```python
import torch
import torch.nn as nn

class Generator(nn.Module):
    def __init__(self, input_dim, hidden_dim, output_dim):
        super(Generator, self).__init__()
        self.fc1 = nn.Linear(input_dim, hidden_dim)
        self.fc2 = nn.Linear(hidden_dim, output_dim)
        self.relu = nn.ReLU()
        self.sigmoid = nn.Sigmoid()

    def forward(self, x):
        x = self.relu(self.fc1(x))
        x = self.sigmoid(self.fc2(x))
        return x

class Discriminator(nn.Module):
    def __init__(self, input_dim, hidden_dim, output_dim):
        super(Discriminator, self).__init__()
        self.fc1 = nn.Linear(input_dim, hidden_dim)
        self.fc2 = nn.Linear(hidden_dim, output_dim)
        self.relu = nn.ReLU()
        self.sigmoid = nn.Sigmoid()

    def forward(self, x):
        x = self.relu(self.fc1(x))
        x = self.sigmoid(self.fc2(x))
        return x

# 初始化模型和优化器
generator = Generator(input_dim=100, hidden_dim=256, output_dim=1)
discriminator = Discriminator(input_dim=1, hidden_dim=256, output_dim=1)
```

```python
# 继续完成之前定义的Generator和Discriminator类
# 初始化模型和优化器
generator = Generator(input_dim=100, hidden_dim=256, output_dim=1)
discriminator = Discriminator(input_dim=1, hidden_dim=256, output_dim=1)

# 定义优化器
optimizer_G = torch.optim.Adam(generator.parameters(), lr=0.0002)
optimizer_D = torch.optim.Adam(discriminator.parameters(), lr=0.0002)

# 定义损失函数
criterion = nn.BCELoss()

# 训练过程
for epoch in range(100):
    for _ in range(100):
        # 训练判别器
        optimizer_D.zero_grad()
        real_data = torch.randn(128, 1)
        fake_data = generator(torch.randn(128, 100)).detach()
        d_real = discriminator(real_data)
        d_fake = discriminator(fake_data)
        loss_D = (torch.mean(torch.log(d_real)) + torch.mean(torch.log(1 - d_fake))) * (-1)
        loss_D.backward()
        optimizer_D.step()

        # 训练生成器
        optimizer_G.zero_grad()
        d_fake_new = discriminator(generator(torch.randn(128, 100)))
        loss_G = torch.mean(torch.log(1 - d_fake_new))
        loss_G.backward()
        optimizer_G.step()
```
