  ```python
  import pandas as pd
  import requests

  # 获取历史股价数据
  def get_stock_data(ticker, start_date, end_date):
      url = f'https://query1.finance.yahoo.com/v7/finance/d/consolidated?'
      params = {
          'symbol': ticker,
          'period1': start_date,
          'period2': end_date
      }
      response = requests.get(url, params=params)
      data = response.json()
      df = pd.DataFrame(data['chart']['result'][0]['prices'])
      return df
  ```
