```python
def value_investment():
    import numpy as np
    import pandas as pd
    from sklearn.cluster import KMeans

    # 数据采集
    data = pd.read_csv('investment_data.csv')
    # 特征提取
    features = data[['revenue', 'profit', 'growth']]
    # 模型训练
    kmeans = KMeans(n_clusters=3)
    clusters = kmeans.fit_predict(features)
    # 决策优化
    optimal_portfolio = np.zeros(len(clusters))
    optimal_portfolio[clusters == 2] = 1  # 假设2为优质投资
    return optimal_portfolio

value_investment()
```
