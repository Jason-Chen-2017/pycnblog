  ```python
  import networkx as nx
  import matplotlib.pyplot as plt

  def emotional_spread(G, beta):
      infected = set()
      initial_infected = list(G.nodes())[0]
      infected.add(initial_infected)
      G.nodes[initial_infected]['state'] = 'infected'

      while True:
          new_infected = []
          for node in G.nodes():
              if node not in infected and G.nodes[node]['state'] == 'susceptible':
                  infected_neighbors = [n for n in G.neighbors(node) if G.nodes[n]['state'] == 'infected']
                  if len(infected_neighbors) > 0:
                      prob_infection = beta * len(infected_neighbors)
                      if prob_infection >= 1:
                          new_infected.append(node)
          if not new_infected:
              break
          for node in new_infected:
              infected.add(node)
              G.nodes[node]['state'] = 'infected'

          nx.draw(G, node_color=['green' if node in infected else 'red' for node in G.nodes()], with_labels=True)
          plt.show()
  ```
