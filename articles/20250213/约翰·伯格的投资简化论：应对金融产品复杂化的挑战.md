      # 简化投资组合优化的实现
      # 计算期望收益和协方差矩阵
      mu = data.mean()
      sigma = data.cov()
      # 计算最优权重
      w = np.linalg.solve(sigma, mu) / np.sum(np.linalg.solve(sigma, mu))
