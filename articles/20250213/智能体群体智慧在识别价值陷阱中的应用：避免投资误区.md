import numpy as np
from sklearn.cluster import KMeans

# 示例数据
X = np.array([[1, 2], [1.5, 2.5], [3, 3], [4, 4]])

# 聚类分析
kmeans = KMeans(n_clusters=2).fit(X)
clusters = kmeans.labels_

# 判断是否为价值陷阱
def is_value_trap(clusters):
    return len(set(clusters)) > 1

result = is_value_trap(clusters)
print("是否为价值陷阱:", result)
