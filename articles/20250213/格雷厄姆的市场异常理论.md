  ```python
  import pandas as pd
  import numpy as np
  from sklearn.ensemble import IsolationForest

  def detect_anomaly(data):
      model = IsolationForest(n_estimators=100, contamination=0.05)
      model.fit(data)
      anomalies = model.predict(data)
      return np.where(anomalies == -1)[0]
  ```
