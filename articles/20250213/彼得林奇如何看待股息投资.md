def calculate_dividend_yield(stock_price, dividend):
    return (dividend / stock_price) * 100

def calculate_dividend_growth(dividends):
    growth = []
    for i in range(1, len(dividends)):
        growth.append((dividends[i] - dividends[i-1]) / dividends[i-1])
    return growth

def calculate_dividend_yield_ratio(dividends, investment):
    return (sum(dividends) / investment) * 100

# 示例
stock_price = 50
dividend = 2.5
dividends = [2, 2.5, 3]

yield = calculate_dividend_yield(stock_price, dividend)
growth = calculate_dividend_growth(dividends)
yield_ratio = calculate_dividend_yield_ratio(dividends, 1000)

print(f"股息率: {yield}%")
print(f"股息增长率: {growth}")
print(f"股息收益率: {yield_ratio}%")
