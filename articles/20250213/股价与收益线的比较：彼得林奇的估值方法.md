```python
import pandas as pd
import matplotlib.pyplot as plt

# 数据加载
df = pd.read_csv('stock_data.csv')

# 计算相对市盈率
industry_avg_pe = df['行业平均市盈率'].mean()
df['相对市盈率'] = df['市盈率'] / industry_avg_pe

# 绘制收益线
plt.figure(figsize=(10, 6))
plt.plot(df['日期'], df['收益'], label='收益线')
plt.plot(df['日期'], df['股价'], label='股价线')
plt.xlabel('日期')
plt.ylabel('数值')
plt.legend()
plt.title('收益线与股价线对比图')
plt.show()
```
