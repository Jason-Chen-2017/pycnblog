def wavelength_to_color(wavelength):
    if wavelength < 450:
        return '紫色'
    elif 450 <= wavelength < 495:
        return '蓝色'
    elif 495 <= wavelength < 560:
        return '绿色'
    else:
        return '红色'

print(quantum_dot_display())
