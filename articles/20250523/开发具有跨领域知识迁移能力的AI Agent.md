import tensorflow as tf
from tensorflow.keras import layers

def contrastive_model(input_dim):
    base_cnn = layers.Conv2D(32, (3,3), activation='relu', input_shape=(input_dim, input_dim, 1))
    base_cnn = layers.MaxPooling2D((2,2))(base_cnn)
    base_cnn = layers.Conv2D(64, (3,3), activation='relu')(base_cnn)
    base_cnn = layers.MaxPooling2D((2,2))(base_cnn)
    base_cnn = layers.Flatten()(base_cnn)
    base_cnn = layers.Dense(256, activation='relu')(base_cnn)
    base_cnn = layers.Dense(128, activation='relu')(base_cnn)
    return base_cnn

def contrastive_loss(y_true, y_pred):
    margin = 1.0
    loss = tf.reduce_mean(tf.square(y_true - y_pred))
    return loss
