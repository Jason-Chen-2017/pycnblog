```python
class AI-Agent:
    def __init__(self):
        self.tasks = 0
        self.time = 0

    def track_efficiency(self):
        if self.time == 0:
            return 0
        return self.tasks / self.time

# 实例化
agent = AI-Agent()
agent.tasks = 10
agent.time = 2
print(f"效率为: {agent.track_efficiency()}")
```
