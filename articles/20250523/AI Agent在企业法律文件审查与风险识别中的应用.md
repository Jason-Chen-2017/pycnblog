```python
import spacy
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.svm import SVC

nlp = spacy.load("en_core_web_sm")

def preprocess(text):
    return [token.text for token in nlp(text)]

def classify_documents(documents, labels):
    X = [" ".join(preprocess(doc)) for doc in documents]
    y = labels
    vectorizer = TfidfVectorizer()
    X_vec = vectorizer.fit_transform(X)
    clf = SVC()
    clf.fit(X_vec, y)
    return clf, vectorizer

# 示例数据
documents = ["违约条款", "合规条款", "其他条款"]
labels = [0, 1, 2]
model, vectorizer = classify_documents(documents, labels)

def predict_new_document(new_doc):
    X_new = [" ".join(preprocess(new_doc))]
    X_new_vec = vectorizer.transform(X_new)
    prediction = model.predict(X_new_vec)[0]
    return prediction

new_document = "本合同约定的履行期限已届满，甲方应立即支付剩余款项。"
print(predict_new_document(new_document))
```
