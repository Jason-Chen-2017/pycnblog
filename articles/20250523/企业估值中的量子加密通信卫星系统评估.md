import random

def generate_key():
    key = random.getrandbits(128)
    return key

def encrypt(plaintext, key):
    ciphertext = plaintext ^ key
    return ciphertext

def decrypt(ciphertext, key):
    plaintext = ciphertext ^ key
    return plaintext
