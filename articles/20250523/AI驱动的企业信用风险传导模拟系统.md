class GNNModel(tf.keras.Model):
    def __init__(self, input_dim, output_dim):
        super(GNNModel, self).__init__()
        self.gcn_layer = GCNLayer(input_dim, output_dim)
        self.dropout = layers.Dropout(0.5)
        self.dense = layers.Dense(output_dim, activation='sigmoid')

    def call(self, inputs):
        x = self.gcn_layer(inputs)
        x = self.dropout(x)
        x = self.dense(x)
        return x

# 定义图卷积层
class GCNLayer(layers.Layer):
    def __init__(self, input_dim, output_dim):
        super(GCNLayer, self).__init__()
        self.kernel = tf.keras.layers.Dense(output_dim, input_dim)
        self.bias = tf.keras.layers.Dense(output_dim)

    def call(self, inputs):
        A = inputs[0]  # 邻接矩阵
        X = inputs[1]  # 特征矩阵
        X = tf.matmul(A, X) + self.bias(X)
        return X
