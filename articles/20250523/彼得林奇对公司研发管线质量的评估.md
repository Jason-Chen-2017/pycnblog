  ```python
      return risk_premium + risk_free_rate

  def net_present_value(cash_flows, discount_rate):
      npv = 0
      for year, cash_flow in enumerate(cash_flows):
          discount_factor = (1 + discount_rate) ** year
          npv += cash_flow / discount_factor
      return npv
  ```
