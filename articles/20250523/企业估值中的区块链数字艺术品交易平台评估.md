pragma solidity ^0.8.0;

contract NFTMarket {
    // 定义NFT的结构体
    struct NFT {
        uint256 tokenId;
        address owner;
        uint256 price;
        string metadata;
    }

    // 存储NFT信息
    mapping(uint256 => NFT) public nfts;

    // 事件：NFT被购买
    event NFTBought(uint256 tokenId, address buyer);

    // 函数：创建NFT
    function createNFT(uint256 tokenId, string memory metadata, uint256 price) public {
        NFT memory newNFT = NFT(tokenId, msg.sender, price, metadata);
        nfts[tokenId] = newNFT;
    }

    // 函数：购买NFT
    function buyNFT(uint256 tokenId) public payable {
        NFT memory nft = nfts[tokenId];
        require(nft.owner != msg.sender, "You already own this NFT");
        require(msg.value >= nft.price, "Insufficient funds");

        // 转账
        payable(nft.owner).transfer(msg.value);
        nft.owner = msg.sender;
        nfts[tokenId] = nft;

        emit NFTBought(tokenId, msg.sender);
    }
}
