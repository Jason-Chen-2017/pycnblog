  ```python
  import spacy

  nlp = spacy.load("en_core_web_sm")

  def identify_intent(text):
      doc = nlp(text)
      keywords = ['天气', '查询', '预报']
      for token in doc:
          if token.text in keywords:
              return token.text
      return '其他'
  ```
