  import pandas as pd
  import numpy as np

  def calculate_residual_income(net_income, cost_of_equity, equity_value):
      residual_income = net_income - (cost_of_equity * equity_value)
      return residual_income

  # 示例数据
  data = {
      '净利润': [100, 200, 150],
      '股权资本成本': [0.1, 0.08, 0.09],
      '股权公允价值': [500, 600, 700]
  }
  df = pd.DataFrame(data)

  # 计算剩余收益
  df['剩余收益'] = df.apply(lambda x: calculate_residual_income(x['净利润'], x['股权资本成本'], x['股权公允价值']), axis=1)

  print(df)
  ```
