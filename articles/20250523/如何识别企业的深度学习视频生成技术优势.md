#### 3.2 生成器与判别器的数学模型
- 生成器的损失函数：$$L_G = -\log(P(G(x)))$$
- 判别器的损失函数：$$L_D = -\log(D(x)) - \log(1 - D(G(x)))$$

import torch
import torch.nn as nn
import torch.optim as optim

# 定义生成器和判别器
class Generator(nn.Module):
    def __init__(self):
        super(Generator, self).__init__()
        self.main = nn.Sequential(
            nn.ConvTranspose2d(100, 512, 4, 2, 1),
            nn.BatchNorm2d(512),
            nn.ReLU(True),
            # 更多层...
        )

class Discriminator(nn.Module):
    def __init__(self):
        super(Discriminator, self).__init__()
        self.main = nn.Sequential(
            nn.Conv2d(3, 64, 4, 2, 1),
            nn.LeakyReLU(0.2),
            # 更多层...
        )

# 初始化模型
generator = Generator()
discriminator = Discriminator()

# 定义损失函数和优化器
criterion = nn.BCELoss()
optimizer_G = optim.Adam(generator.parameters(), lr=0.0002)
optimizer_D = optim.Adam(discriminator.parameters(), lr=0.0002)

# 训练过程
for epoch in range(num_epochs):
    for i, data in enumerate(dataloader):
        # 更新判别器
        optimizer_D.zero_grad()
        real = data
        fake = generator(data)
        # 计算判别器的损失
        loss_D = criterion(discriminator(real).detach(), torch.ones_like(discriminator(real))) + \
                 criterion(discriminator(fake).detach(), torch.zeros_like(discriminator(fake)))
        loss_D.backward()
        optimizer_D.step()

        # 更新生成器
        optimizer_G.zero_grad()
        loss_G = criterion(discriminator(fake), torch.ones_like(discriminator(fake)))
        loss_G.backward()
        optimizer_G.step()
```
