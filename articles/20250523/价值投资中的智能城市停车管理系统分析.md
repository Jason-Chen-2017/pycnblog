### 3.3 算法实现的Python代码示例

#### 3.3.1 遗传算法实现

```python
def fitness(x):
    return -x**2 + 4*x + 5

def genetic_algorithm(population_size, generations):
    population = [random.uniform(0, 10) for _ in range(population_size)]
    for _ in range(generations):
        fitness_values = [fitness(x) for x in population]
        selected = [x for x in population if fitness(x) > random.uniform(0, 1)]
        selected = sorted(selected, key=lambda x: -fitness(x))[:population_size//2]
        population = [x for x in population if fitness(x) < random.uniform(0, 1)]
        for i in range(len(population)//2):
            x1 = population[2*i]
            x2 = population[2*i+1]
            new_x1 = x1 * (1 - random.random() / 2) + x2 * (random.random() / 2)
            new_x2 = x2 * (1 - random.random() / 2) + x1 * (random.random() / 2)
            if fitness(new_x1) > fitness(new_x2):
                population[2*i] = new_x1
                population[2*i+1] = new_x1
            else:
                population[2*i] = new_x2
                population[2*i+1] = new_x2
    return max(fitness(x) for x in population)
```
