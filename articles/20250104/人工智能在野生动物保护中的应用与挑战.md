#### 2. Python源代码实现

```python
import cv2
import numpy as np

# 读取图像
image = cv2.imread('animal.jpg')

# 预处理图像
gray_image = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
blurred_image = cv2.GaussianBlur(gray_image, (5, 5), 0)

# 特征提取
SIFT = cv2.SIFT_create()
keypoints, descriptors = SIFT.detectAndCompute(blurred_image, None)

# 训练模型
model = cv2.xgboost.XGBClassifier(use_label_encoder=False, eval_metric='logloss')
model.fit(descriptors, keypoints)

# 模型评估
predictions = model.predict(descriptors)
accuracy = np.mean(predictions == keypoints)
print(f'Accuracy: {accuracy:.2f}')

# 输出结果
cv2.drawKeypoints(image, keypoints, image)
cv2.imshow('Tracked Animal', image)
cv2.waitKey(0)
cv2.destroyAllWindows()
```

2. **特征提取**：使用SIFT算法检测关键点和计算描述子。
3. **训练模型**：利用XGBoost分类器对描述子进行分类训练。
4. **模型评估**：通过准确率等指标评估模型性能。
5. **输出结果**：绘制检测到的关键点，实现动物追踪。

数学模型包括：

- **SIFT算法**：关键点检测和描述子计算。

```python
import cv2
import numpy as np
import xgboost as xgb

# 动物追踪模块
def track_animal(image_path):
    image = cv2.imread(image_path)
    gray_image = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    blurred_image = cv2.GaussianBlur(gray_image, (5, 5), 0)
    SIFT = cv2.SIFT_create()
    keypoints, descriptors = SIFT.detectAndCompute(blurred_image, None)
    model = cv2.xgboost.XGBClassifier(use_label_encoder=False, eval_metric='logloss')
    model.fit(descriptors, keypoints)
    predictions = model.predict(descriptors)
    accuracy = np.mean(predictions == keypoints)
    cv2.drawKeypoints(image, keypoints, image)
    cv2.imshow('Tracked Animal', image)
    cv2.waitKey(0)
    cv2.destroyAllWindows()
    return accuracy

# 栖息地监测模块
def monitor_habitat(image_path):
    image = cv2.imread(image_path)
    gray_image = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    blurred_image = cv2.GaussianBlur(gray_image, (5, 5), 0)
    SIFT = cv2.SIFT_create()
    keypoints, descriptors = SIFT.detectAndCompute(blurred_image, None)
    model = cv2.xgboost.XGBRegressor(use_label_encoder=False, eval_metric='mse')
    model.fit(descriptors, keypoints)
    predictions = model.predict(descriptors)
    cv2.imshow('Monitored Habitat', image)
    cv2.waitKey(0)
    cv2.destroyAllWindows()
    return predictions

# 非法交易打击模块
def combat_illegal_trade(data_path):
    data = np.loadtxt(data_path)
    X = data[:, :-1]
    y = data[:, -1]
    model = cv2.xgboost.XGBClassifier(use_label_encoder=False, eval_metric='logloss')
    model.fit(X, y)
    predictions = model.predict(X)
    return predictions

# 测试代码
if __name__ == '__main__':
    track_animal('animal.jpg')
    monitor_habitat('habitat.jpg')
    combat_illegal_trade('trade_data.txt')
```
