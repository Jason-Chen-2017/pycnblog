data = pd.read_csv('data.csv')

predictions = model.predict(X_test)
example_indices = np.where(predictions == 1)[0]

# 示例可视化
for idx in example_indices:
    plt.scatter(X_test[idx, 0], X_test[idx, 1], c='r', label='Predicted 1')
plt.scatter(X_test[~example_indices, 0], X_test[~example_indices, 1], c='b', label='Predicted 0')
plt.xlabel('Feature 1')
plt.ylabel('Feature 2')
plt.legend()
plt.show()

# 用户交互
print("Please review the following examples and provide feedback:")
for idx in example_indices:

user_feedback = input("Do you agree with the prediction? (yes/no): ")

if user_feedback.lower() == 'no':
