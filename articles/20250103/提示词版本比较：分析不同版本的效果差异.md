以下Python源代码实现了一个简单的提示词版本比较算法：

```python
import numpy as np

# 提取提示词版本
def get_prompt_versions():
    return ["基础版本", "优化版本", "高级版本"]

# 评估效果
def evaluate_prompt(prompt, task):
    # 示例：根据任务类型和提示词内容计算效果得分
    if "基础版本" in prompt:
        return 1
    elif "优化版本" in prompt:
        return 2
    else:
        return 3

# 主函数
def main():
    versions = get_prompt_versions()
    best_version = None
    best_score = 0

    for version in versions:
        score = evaluate_prompt(version, "文本生成任务")
        if score > best_score:
            best_score = score
            best_version = version

    print("最优版本：", best_version)

if __name__ == "__main__":
    main()
```
