```python
import openai

def assess_consulting_ability(consulting_content):
    response = openai.Completion.create(
        engine="text-davinci-002",
        prompt=consulting_content,
        max_tokens=50,
        n=1,
        stop=None,
        temperature=0.5,
    )
    return response.choices[0].text.strip()
```
