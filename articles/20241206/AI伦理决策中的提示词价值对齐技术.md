def value_alignment(prompt_words, target_value):
    value_adjustments = []
    for word in prompt_words:
        adjustment = target_value - get_value(word)
        value_adjustments.append(adjustment)
    
    total_adjustment = sum(value_adjustments)
    num_of_prompt_words = len(prompt_words)
    
    value_alignment_score = total_adjustment / num_of_prompt_words
