```python
import nltk
from nltk.tokenize import word_tokenize
from nltk.corpus import stopwords
from nltk.tag import pos_tag

def extract_domain_knowledge(knowledge_source):
    # 读取知识源
    with open(knowledge_source, 'r', encoding='utf-8') as f:
        knowledge = f.read()

    # 预处理
    tokens = word_tokenize(knowledge)
    tokens = [token.lower() for token in tokens if token.lower() not in stopwords.words('english')]
    pos_tags = pos_tag(tokens)

    # 构建领域本体
    domain_ontology = {}
    for word, tag in pos_tags:
        if tag.startswith('NN'):
            domain_ontology[word] = []
        elif tag.startswith('VB'):
            domain_ontology[word] = '动词'

    return domain_ontology

domain_knowledge = extract_domain_knowledge('domain_knowledge.txt')
print(domain_knowledge)
```
