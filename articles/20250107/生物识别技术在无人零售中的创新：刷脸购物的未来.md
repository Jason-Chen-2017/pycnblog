import cv2
import face_recognition
import numpy as np

# 人脸检测
def detect_face(image):
    # 加载图像
    image = cv2.imread(image)
    # 转换为灰度图像
    gray_image = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    # 使用Haar级联分类器进行人脸检测
    faces = face_recognition.face_locations(image, model='cascade')
    return faces, image

# 人脸识别
def recognize_face(faces, known_faces, known_labels):
    # 遍历所有检测到的人脸
    for (top, right, bottom, left), label in zip(faces, known_labels):
        # 裁剪人脸区域
        face_image = image[left:right, top:bottom]
        # 人脸编码
        encoding = face_recognition.face_encodings(face_image)[0]
        # 与已知的脸进行匹配
        matches = face_recognition.compare_faces(known_faces, encoding)
        if True in matches:
            return label
    return "未知"

# 商品识别
def recognize_goods(image, known_goods, known_scores):
    # 遍历所有检测到的商品
    for (top, right, bottom, left), score in zip(known_goods, known_scores):
        # 裁剪商品区域
        good_image = image[top:bottom, left:right]
        # 商品编码
        encoding = cv2.imencode('.jpg', good_image)[1].tobytes()
        # 与已知的商品进行匹配
        matches = cv2.matchTemplate(image, good_image, cv2.TM_CCOEFF_NORMED)
        if score > matches.mean():
            return label
    return "未知"

# 支付
def pay(label, price):
    # 根据标签和价格进行支付
    if label == "会员":
        return price * 0.9  # 会员享受9折优惠
    else:

    # 识别商品
    label = recognize_goods(image, known_goods, known_scores)
    # 支付
    price = pay(label, price)
