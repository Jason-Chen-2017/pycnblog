##### 4.2.1 数据库模型设计

首先，我们需要设计数据库模型。以下是Facility和Network两个模型的设计：

```python
from flask_sqlalchemy import SQLAlchemy

db = SQLAlchemy()

class Facility(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(100), nullable=False)
    status = db.Column(db.String(50), nullable=False)
    network_id = db.Column(db.Integer, db.ForeignKey('network.id'), nullable=False)

class Network(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(100), nullable=False)
```

##### 4.2.2 系统接口实现

接下来，我们实现系统接口。以下是Facility和Network两个接口的实现：

```python
from flask import Flask, request, jsonify
from models import db, Facility, Network

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'postgresql://username:password@localhost/dbname'
db.init_app(app)

@app.route('/facilities', methods=['GET', 'POST'])
def facility_handler():
    if request.method == 'GET':
        facilities = Facility.query.all()
        return jsonify([{'id': f.id, 'name': f.name, 'status': f.status, 'network_id': f.network_id} for f in facilities])
    elif request.method == 'POST':
        data = request.get_json()
        new_facility = Facility(name=data['name'], status=data['status'], network_id=data['network_id'])
        db.session.add(new_facility)
        db.session.commit()
        return jsonify({'message': 'Facility added successfully.'})

@app.route('/networks', methods=['GET', 'POST'])
def network_handler():
    if request.method == 'GET':
        networks = Network.query.all()
        return jsonify([{'id': n.id, 'name': n.name} for n in networks])
    elif request.method == 'POST':
        data = request.get_json()
        new_network = Network(name=data['name'])
        db.session.add(new_network)
        db.session.commit()
        return jsonify({'message': 'Network added successfully.'})

if __name__ == '__main__':
    app.run(debug=True)
```

class Facility(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(100), nullable=False)

##### 4.2.2 系统接口实现

接下来，我们实现系统接口。以下是Facility和Network两个接口的实现：

```python
from flask import Flask, request, jsonify
from models import db, Facility, Network

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'postgresql://username:password@localhost/dbname'
db.init_app(app)

@app.route('/facilities', methods=['GET', 'POST'])
def facility_handler():
    if request.method == 'GET':
        facilities = Facility.query.all()
        return jsonify([{'id': f.id, 'name': f.name, 'status': f.status, 'network_id': f.network_id} for f in facilities])
    elif request.method == 'POST':
        data = request.get_json()
        new_facility = Facility(name=data['name'], status=data['status'], network_id=data['network_id'])
        db.session.add(new_facility)
        db.session.commit()
        return jsonify({'message': 'Facility added successfully.'})

@app.route('/networks', methods=['GET', 'POST'])
def network_handler():
    if request.method == 'GET':
        networks = Network.query.all()
        return jsonify([{'id': n.id, 'name': n.name} for n in networks])
    elif request.method == 'POST':
        data = request.get_json()
        new_network = Network(name=data['name'])
        db.session.add(new_network)
        db.session.commit()
        return jsonify({'message': 'Network added successfully.'})

if __name__ == '__main__':
    app.run(debug=True)
```
