#### 1.1.1 问题背景：数字艺术品拍卖的挑战

In recent years, the digital art market has surged in popularity, with the advent of Non-Fungible Tokens (NFTs). However, in the digital art market, where authenticity can be easily falsified and ownership transferred anonymously, this trust mechanism is inadequate.

The primary issue in digital art auctions is the lack of trust. Without a reliable trust mechanism, buyers may be hesitant to participate in auctions, fearing fraud or counterfeit artworks. Some of the key trust-related problems in digital art auctions include:

1. **Authenticity**: Ensuring that the artwork being auctioned is genuinely created by the claimed artist.
2. **Provenance**: Verifying the ownership history of the artwork, including previous sales and exhibitions.
3. **Ownership**: Confirming that the seller has the legal right to sell the artwork.
4. **Transparency**: Ensuring that the auction process is transparent, with no hidden fees or transactions.

Blockchain technology offers a potential solution to these trust-related problems. By leveraging its core properties of decentralization, transparency, and immutability, blockchain can enhance the trust and transparency in digital art auctions.

2. **Transparency**: All transactions and records on the blockchain are transparent and accessible to everyone. This transparency ensures that all participants can verify the authenticity and ownership of artworks.
3. **Immutability**: Once data is recorded on the blockchain, it cannot be altered or deleted. This immutability ensures that the provenance and ownership history of artworks are preserved and cannot be tampered with.

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

import "@openzeppelin/contracts/token/ERC721/ERC721.sol";
import "@openzeppelin/contracts/security/Pausable.sol";
import "@openzeppelin/contracts/access/Ownable.sol";

contract ArtAuction is ERC721, Pausable, Ownable {
    struct Auction {
        uint256 startTimestamp;
        uint256 endTimestamp;
        uint256 reservePrice;
        address highestBidder;
        uint256 highestBid;
    }

    mapping(uint256 => Auction) public auctions;
    mapping(address => mapping(uint256 => uint256)) public bids;

    event AuctionCreated(uint256 tokenId, uint256 startTimestamp, uint256 endTimestamp, uint256 reservePrice);
    event BidPlaced(uint256 tokenId, address bidder, uint256 bid);
    event AuctionEnded(uint256 tokenId, address highestBidder, uint256 highestBid);

    constructor() ERC721("Digital Art", "ART") {}

    function createAuction(uint256 tokenId, uint256 startTimestamp, uint256 endTimestamp, uint256 reservePrice) external onlyOwner {
        require(auctions[tokenId].highestBidder == address(0), "Auction already exists");
        require(startTimestamp < endTimestamp, "Invalid auction times");
        require(reservePrice > 0, "Invalid reserve price");

        auctions[tokenId] = Auction({
            startTimestamp: startTimestamp,
            endTimestamp: endTimestamp,
            reservePrice: reservePrice,
            highestBidder: address(0),
            highestBid: 0
        });

        _mint(msg.sender, tokenId);
        emit AuctionCreated(tokenId, startTimestamp, endTimestamp, reservePrice);
    }

    function bid(uint256 tokenId) external payable {
        require(auctions[tokenId].startTimestamp <= block.timestamp, "Auction has not started");
        require(auctions[tokenId].endTimestamp > block.timestamp, "Auction has ended");
        require(msg.value > auctions[tokenId].highestBid, "Bid too low");

        uint256 newBid = msg.value;
        address sender = msg.sender;

        bids[sender][tokenId] = newBid;

        if (newBid > auctions[tokenId].highestBid) {
            auctions[tokenId].highestBid = newBid;
            auctions[tokenId].highestBidder = sender;
        }

        emit BidPlaced(tokenId, sender, newBid);
    }

    function endAuction(uint256 tokenId) external {
        require(auctions[tokenId].endTimestamp <= block.timestamp, "Auction has not ended");

        uint256 highestBid = auctions[tokenId].highestBid;
        address highestBidder = auctions[tokenId].highestBidder;

        require(highestBidder != address(0), "No bids received");

        _transferFrom(auctions[tokenId].highestBidder, owner(), tokenId);
        payable(owner()).transfer(highestBid);

        emit AuctionEnded(tokenId, highestBidder, highestBid);
    }
}
