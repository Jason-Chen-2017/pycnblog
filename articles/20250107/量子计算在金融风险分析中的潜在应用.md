##### 3.2.1.1 Python 源代码

# 导入量子计算库
from qiskit import QuantumCircuit, execute, Aer
from qiskit.optimize import NumPyOptimizer
from qiskit.aqua.components.operators import WeightedPauliOperator
from qiskit.aqua.algorithms import VariationalQuantumRefiningAlgorithm

# 定义投资组合参数
assets = {'A': {'weight': 0.5, 'return': 0.1, 'variance': 0.04},
          'B': {'weight': 0.3, 'return': 0.08, 'variance': 0.03},
          'C': {'weight': 0.2, 'return': 0.05, 'variance': 0.02}}

# 构建权重 Pauli 作用器
 pauli_list = []
 for asset, params in assets.items():
     pauli = WeightedPauliOperator(
         paulis={'pauli': 'II', 'coeff': -params['variance']},
         weights={'weights': [params['weight']]}
     )
     pauli_list.append(pauli)

# 定义 VQE 算法
vqra = VariationalQuantumRefiningAlgorithm(
    optimization_problem=NumPyOptimizer(),
    ansatz='Ry', # 参考Ansatz为旋转门
    objective_function=WeightedPauliOperator(pauli_list)
)

# 执行 VQE 算法
result = vqra.solve()

# 输出最优解
print(result)
