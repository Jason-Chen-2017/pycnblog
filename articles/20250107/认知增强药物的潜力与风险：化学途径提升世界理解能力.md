```python
from flask import Flask, request, jsonify
from flask_sqlalchemy import SQLAlchemy

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'mysql+pymysql://username:password@localhost/db_name'
db = SQLAlchemy(app)

class Patient(db.Model):
    patient_id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(100))
    age = db.Column(db.Integer)

class Drug(db.Model):
    drug_id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(100))
    dosage = db.Column(db.Integer)

class Experiment(db.Model):
    experiment_id = db.Column(db.Integer, primary_key=True)
    patient_id = db.Column(db.Integer, db.ForeignKey('patient.patient_id'))
    drug_id = db.Column(db.Integer, db.ForeignKey('drug.drug_id'))
    start_date = db.Column(db.Date)
    end_date = db.Column(db.Date)
    result = db.Column(db.Text)

@app.route('/patients', methods=['POST'])
def add_patient():
    data = request.json
    new_patient = Patient(name=data['name'], age=data['age'])
    db.session.add(new_patient)
    db.session.commit()
    return jsonify({'patient_id': new_patient.patient_id})

@app.route('/drugs', methods=['POST'])
def add_drug():
    data = request.json
    new_drug = Drug(name=data['name'], dosage=data['dosage'])
    db.session.add(new_drug)
    db.session.commit()
    return jsonify({'drug_id': new_drug.drug_id})

@app.route('/experiments', methods=['POST'])
def add_experiment():
    data = request.json
    new_experiment = Experiment(patient_id=data['patient_id'], drug_id=data['drug_id'], start_date=data['start_date'], end_date=data['end_date'], result=data['result'])
    db.session.add(new_experiment)
    db.session.commit()
    return jsonify({'experiment_id': new_experiment.experiment_id})

if __name__ == '__main__':
    db.create_all()
    app.run(debug=True)
```
