import numpy as np

def lorenz_system(x, y, z, sigma=10, rho=28, beta=8/3):
    dxdt = sigma * (y - x)
    dydt = x * (rho - z) - y
    dzdt = x * y - beta * z
    return dxdt, dydt, dzdt

# 初始化参数
sigma = 10
rho = 28
beta = 8/3
x0, y0, z0 = 1, 1, 1

# 计算流
t = np.linspace(0, 100, 1000)
x, y, z = np.meshgrid(x0, y0, z0)
dxdt, dydt, dzdt = lorenz_system(x, y, z)

print(f"x(t) = {x}, y(t) = {y}, z(t) = {z}")
