import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from mermaid import Mermaid

# 数据处理
def preprocess_data(file_path):
    data = pd.read_csv(file_path)
    # 填充缺失值
    data.fillna(method='ffill', inplace=True)
    # 过滤异常值
    data = data[(data['velocity'] > -100) & (data['velocity'] < 100)]
    return data

# 信息扩散模拟
def simulate_diffusion(data, diffusion_coefficient, time_steps):
    # 初始化信息分布
    I = np.zeros_like(data['velocity'])
    # 迭代计算信息扩散
    for _ in range(time_steps):
        I += diffusion_coefficient * np.random.normal(size=I.shape)
    return I

# 熵变化模拟
def simulate_entropy(data, time_steps):
    entropy_history = []
    for _ in range(time_steps):
        p = data['velocity'].value_counts() / len(data)
        entropy = -np.sum(p * np.log2(p))
        entropy_history.append(entropy)
    return entropy_history

# 结构演化模拟
def simulate_structure(data, time_steps):
    # 模拟结构演化，如星系合并
    for _ in range(time_steps):
        # 假设结构演化规则
        data['velocity'] += np.random.normal(size=data['velocity'].shape)
    return data

# 主函数
def main():
    file_path = 'universe_data.csv'
    data = preprocess_data(file_path)
    diffusion_coefficient = 0.1
    time_steps = 100

    # 模拟信息扩散
    I_simulation = simulate_diffusion(data['velocity'], diffusion_coefficient, time_steps)

    # 模拟熵变化
    entropy_history = simulate_entropy(data['velocity'], time_steps)

    # 模拟结构演化
    data_simulation = simulate_structure(data['velocity'], time_steps)

    # 绘制结果
    plt.figure(figsize=(12, 6))
    plt.subplot(1, 3, 1)
    plt.plot(I_simulation)
    plt.title('Information Diffusion')
    plt.xlabel('Time Steps')
    plt.ylabel('Information')

    plt.plot(entropy_history)
    plt.title('Entropy Change')
    plt.xlabel('Time Steps')
    plt.ylabel('Entropy')

    plt.plot(data_simulation['velocity'])
    plt.title('Structure Evolution')
    plt.xlabel('Time Steps')
    plt.ylabel('Velocity')

if __name__ == '__main__':
    main()

   $$D(P||Q) = \sum_{i} p(x_i) \log_2 \frac{p(x_i)}{q(x_i)}$$
   - 衡量两个概率分布的差异。

4. **结构熵**：
   $$H(G) = -\sum_{i} p(G_i) \log_2 p(G_i)$$
   - 衡量系统结构的无序程度。
