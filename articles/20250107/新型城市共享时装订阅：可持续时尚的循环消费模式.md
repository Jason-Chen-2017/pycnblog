```python
import random

# 用户需求
user_demand = {
    'style': 'Casual',
    'brand': 'Nike',
    'size': 'M'
}

# 库存信息
inventory = [
    {'style': 'Casual', 'brand': 'Nike', 'size': 'M'},
    {'style': 'Formal', 'brand': 'Adidas', 'size': 'L'},
    {'style': 'Casual', 'brand': 'Puma', 'size': 'S'}
]

# 算法优化
def allocate_clothing(user_demand, inventory):
    suitable_clothing = [
        item for item in inventory
        if item['style'] == user_demand['style']
        and item['brand'] == user_demand['brand']
        and item['size'] == user_demand['size']
    ]
    return random.choice(suitable_clothing)

# 反馈调整
def adjust_demand(suitable_clothing, user_demand):
    if suitable_clothing:
        user_demand['size'] = suitable_clothing['size']
    else:
        user_demand['style'] = 'Formal'

# 主程序
while True:
    clothing_allocated = allocate_clothing(user_demand, inventory)
    if clothing_allocated:
        print(f'Allocated: {clothing_allocated}')
        adjust_demand(clothing_allocated, user_demand)
    else:
        print('No suitable clothing available.')
        break
```
