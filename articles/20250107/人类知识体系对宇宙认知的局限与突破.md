以下是基于Python和Scikit-learn实现的核心代码：

```python
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score

# 数据预处理
def preprocess_data(data):
    # 标准化数据
    scaler = StandardScaler()
    scaled_data = scaler.fit_transform(data)
    return scaled_data

# 特征提取
def extract_features(data):
    # 使用pandas提取特征
    features = data[['feature1', 'feature2', 'feature3']]
    return features

# 模型训练
def train_model(train_data, train_labels):
    # 使用随机森林分类器进行训练
    model = RandomForestClassifier(n_estimators=100)
    model.fit(train_data, train_labels)
    return model

# 预测结果
def predict_results(model, test_data):
    # 进行预测
    predictions = model.predict(test_data)
    return predictions

# 评估模型
def evaluate_model(predictions, test_labels):
    # 计算准确率
    accuracy = accuracy_score(test_labels, predictions)
    return accuracy

# 主函数
def main():
    # 加载数据
    data = pd.read_csv('cosmic_data.csv')
    labels = data['label']

    # 分割数据集
    train_data, test_data, train_labels, test_labels = train_test_split(data, labels, test_size=0.2, random_state=42)

    # 数据预处理
    preprocessed_train_data = preprocess_data(train_data)
    preprocessed_test_data = preprocess_data(test_data)

    # 特征提取
    features = extract_features(preprocessed_train_data)

    # 模型训练
    model = train_model(features, train_labels)

    # 预测结果
    predictions = predict_results(model, preprocessed_test_data)

    # 评估模型
    accuracy = evaluate_model(predictions, test_labels)
    print(f"Model accuracy: {accuracy:.2f}")

if __name__ == "__main__":
    main()
```
