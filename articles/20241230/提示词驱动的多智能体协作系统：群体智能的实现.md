以下是系统核心实现的源代码：

```python
import tensorflow as tf
import keras
import redis

# 智能体类
class Agent:
    def __init__(self, id):
        self.id = id
        self.model = keras.Sequential([
            keras.layers.Dense(64, activation='relu', input_shape=(784,)),
            keras.layers.Dense(10, activation='softmax')
        ])

    def train(self, x, y):
        self.model.compile(optimizer='adam', loss='categorical_crossentropy', metrics=['accuracy'])
        self.model.fit(x, y, epochs=5)

    def predict(self, x):
        return self.model.predict(x)

# 控制层类
class Controller:
    def __init__(self):
        self.agents = []
        self.redis_client = redis.StrictRedis(host='localhost', port=6379, db=0)

    def register_agent(self, agent):
        self.agents.append(agent)
        self.redis_client.lpush('agents', agent.id)

    def assign_task(self, task):
        agent_id = self.redis_client.rpop('agents')
        agent = self.agents[agent_id]
        agent.train(task.x, task.y)
        return agent.predict(task.x)

# 主函数
if __name__ == '__main__':
    controller = Controller()
    agent1 = Agent(1)
    agent2 = Agent(2)
    controller.register_agent(agent1)
    controller.register_agent(agent2)
    task = Task(x=[1, 2, 3], y=[0, 1, 0])
    prediction = controller.assign_task(task)
    print(prediction)
```
