# 预测

# 评估模型
score = regressor.score(X_test, y_test)
print(f'Model Score: {score:.2f}')

# 可视化结果
plt.scatter(y_test, y_pred)
plt.xlabel('Actual Performance')
plt.ylabel('Predicted Performance')
plt.title('Decision Tree Regression')
plt.show()
