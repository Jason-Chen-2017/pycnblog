import spacy
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier

# 1. 自然语言处理（NLP）
nlp = spacy.load("en_core_web_sm")

# 2. 数据预处理
def preprocess_text(text):
    doc = nlp(text)
    tokens = [token.lemma_ for token in doc if not token.is_punct]
    return " ".join(tokens)

# 3. 生成指令
def generate_command(prompt):
    processed_prompt = preprocess_text(prompt)
    # 假设指令列表
    commands = ["查询交通流量", "分析环境污染", "预测人口增长"]
    vectorizer = TfidfVectorizer()
    command_vectors = vectorizer.fit_transform(commands)
    prompt_vector = vectorizer.transform([processed_prompt])
    closest_command = commands[command_vectors @ prompt_vector.argmax()]
    return closest_command

# 4. 数据挖掘与模型训练
def train_model(X, y):
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2)
    model = RandomForestClassifier()
    model.fit(X_train, y_train)
    print("模型训练准确率：", model.score(X_test, y_test))

# 示例
prompt = "如何减少交通拥堵？"
command = generate_command(prompt)
print("生成的指令：", command)

#### 3.4 数学模型与公式讲解

在机器学习模型中，常用的数学模型包括特征提取、模型训练和预测等。以下是一些关键的数学公式和解释：

1. **TF-IDF公式**：

$$
\text{TF-IDF}(w_d,t) = \frac{f(w_d,t)}{N} \log \left( \frac{N}{n(w_d)} \right)
$$

其中，$w_d$是词汇，$t$是文档，$f(w_d,t)$是词汇在文档中出现的频率，$N$是文档总数，$n(w_d)$是包含词汇的文档数量。

2. **余弦相似度**：

$$
\text{Cosine Similarity}(A, B) = \frac{A \cdot B}{\|A\| \|B\|}
$$

其中，$A$和$B$是两个向量，$\cdot$表示点积，$\|\|$表示向量的模。

3. **随机森林分类器损失函数**：

$$
L(y, \hat{y}) = -\sum_{i=1}^{n} y_i \log(\hat{y}_i) + (1 - y_i) \log(1 - \hat{y}_i)
$$

其中，$y$是真实标签，$\hat{y}$是预测标签。

这些数学模型和公式为机器学习算法的实现提供了理论基础，使得系统能够有效地处理和分析城市数据，生成准确的规划结果。
