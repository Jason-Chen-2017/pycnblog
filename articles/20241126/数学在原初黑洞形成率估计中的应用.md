import numpy as np
from scipy.integrate import solve_ivp

def density_evolution(t, y):
    rho = y[0]
    d_rho_dt = -rho / (4 * np.pi * G * t**2)
    return [d_rho_dt]

G = 6.674 * 10**-11  # 引力常数
initial_density = 1e12  # 初始密度
time_step = 1e9  # 时间步长

t_span = (0, time_step)
y0 = [initial_density]

sol = solve_ivp(density_evolution, t_span, y0, method='RK45')

plt.plot(sol.t, sol.y[0])
plt.xlabel('Time (s)')
plt.ylabel('Density')
plt.title('Density Evolution')
plt.show()
