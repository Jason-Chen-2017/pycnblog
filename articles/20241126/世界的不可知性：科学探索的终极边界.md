### 核心算法原理讲解

以宇宙学中的黑洞为例，黑洞的形成和性质是宇宙探索中的重要课题。以下是一个简单的Python程序，模拟黑洞的形成过程：

```python
import numpy as np

def simulate_blackhole(mass, radius):
    # 计算黑洞的引力场
    g = 6.674 * 10**-11  # 万有引力常数
    r = radius  # 黑洞半径
    M = mass  # 物体质量
    
    # 计算引力
    F = g * M * mass / r**2
    
    return F

# 模拟一个质量为10个太阳质量，半径为3公里的黑洞
mass = 10 * 1.989 * 10**30  # 太阳质量
radius = 3000  # 米

# 计算引力
force = simulate_blackhole(mass, radius)
print(f"The gravitational force is: {force} N")

```

```python
# 黑洞模拟器：模拟黑洞的引力场

import numpy as np
import matplotlib.pyplot as plt

def simulate_blackhole(mass, radius, steps=1000, r_min=100, r_max=5000):
    # 计算引力场
    g = 6.674 * 10**-11
    positions = np.linspace(r_min, r_max, steps)
    forces = g * mass**2 / positions**2
    
    # 绘制引力场
    plt.plot(positions, forces)
    plt.xlabel('Distance (m)')
    plt.ylabel('Force (N)')
    plt.title('Gravitational Field of a Black Hole')
    plt.grid(True)
    plt.show()

# 模拟一个质量为10个太阳质量，半径为3公里的黑洞
mass = 10 * 1.989 * 10**30
radius = 3000
simulate_blackhole(mass, radius)

```
