# 导入相关库
import numpy as np
import matplotlib.pyplot as plt

# 定义二次函数
def quadratic_function(a, b, c):
    x = np.linspace(0, 10, 100)
    y = a * x**2 + b * x + c
    return x, y

# 计算并绘图
def plot_function(a, b, c):
    x, y = quadratic_function(a, b, c)
    plt.plot(x, y)
    plt.xlabel('Time (s)')
    plt.ylabel('Position (m)')
    plt.title('Quadratic Motion')
    plt.show()

# 示例参数
a = 1
b = 0
c = 0

# 运行绘图
plot_function(a, b, c)
```
