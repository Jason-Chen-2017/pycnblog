import numpy as np
import matplotlib.pyplot as plt

# 宇宙膨胀模型参数
H_0 = 70  # 哈勃常数（公里/秒/百万秒差距）
a_0 = 1   # 初始尺度因子

    return 1 / (1 + z + (H_0 * z)**-1)

def cosmic_time(H_0, z):
    return np.log(1 + z) / H_0

# 绘制宇宙膨胀模型

    a = np.linspace(0, 1, 1000)
    t = cosmic_time(H_0, a)

    plt.title('宇宙膨胀模型')

plot_universe(H_0, z)
