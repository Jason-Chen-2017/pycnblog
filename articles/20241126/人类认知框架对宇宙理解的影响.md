  ```python
  import numpy as np
  import matplotlib.pyplot as plt

  def calculate_mass_density(data):
      # 计算质量密度
      mass_density = data[:, 2] / data[:, 3]
      return mass_density

  def plot_mass_density(mass_density):
      # 绘制质量密度图
      plt.plot(mass_density)
      plt.xlabel('Distance (Mpc)')
      plt.ylabel('Mass Density (M_sun/Mpc^3)')
      plt.title('Mass Density Profile')
      plt.show()

  # 读取观测数据

  # 绘制质量密度图
  plot_mass_density(mass_density)
  ```
