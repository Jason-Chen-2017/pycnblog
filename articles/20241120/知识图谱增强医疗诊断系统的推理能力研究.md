```
# 导入相关库
import json
import pandas as pd
from py2neo import Graph

# 连接到Neo4j数据库
graph = Graph("bolt://localhost:7687", auth=("neo4j", "password"))

# 构建疾病实体
def build_disease_entity(disease_name):
    query = """
    CREATE (d:Disease {name: $disease_name})
    """
    graph.run(query, disease_name=disease_name)

# 构建症状实体
def build_symptom_entity(symptom_name):
    query = """
    CREATE (s:Symptom {name: $symptom_name})
    """
    graph.run(query, symptom_name=symptom_name)

# 构建关系
def build_relationship(disease_name, symptom_name):
    query = """
    MATCH (d:Disease {name: $disease_name}), (s:Symptom {name: $symptom_name})
    CREATE (d)-[:HAS_SYMPTOM]->(s)
    """
    graph.run(query, disease_name=disease_name, symptom_name=symptom_name)

# 搜索疾病
def search_disease(symptom_names):
    query = """
    MATCH (s:Symptom)-[:HAS_SYMPTOM]->(d:Disease)
    WHERE s.name IN $symptom_names
    RETURN d
    """
    results = graph.run(query, symptom_names=symptom_names)
    diseases = [result["d"]["name"] for result in results]
    return diseases
```
