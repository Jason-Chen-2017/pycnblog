```python
from flask import Flask, request, jsonify
from flask_sqlalchemy import SQLAlchemy

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'mysql+pymysql://username:password@localhost/philosophy'
db = SQLAlchemy(app)

class Philosophy(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    title = db.Column(db.String(255))
    content = db.Column(db.Text)
    author = db.Column(db.String(255))
    topic = db.Column(db.String(255))

@app.route('/api/knowledge', methods=['GET'])
def search_knowledge():
    topic = request.args.get('topic')
    philosophies = Philosophy.query.filter_by(topic=topic).all()
    return jsonify([{'title': p.title, 'content': p.content, 'author': p.author} for p in philosophies])

@app.route('/api/knowledge', methods=['POST'])
def add_knowledge():
    data = request.get_json()
    new_philosophy = Philosophy(
        title=data['title'],
        content=data['content'],
        author=data['author'],
        topic=data['topic']
    )
    db.session.add(new_philosophy)
    db.session.commit()
    return jsonify({'message': 'Knowledge added successfully.'})

if __name__ == '__main__':
    db.create_all()
    app.run(debug=True)
```
