```python
import numpy as np

# 设置随机种子以保证结果的可重复性
np.random.seed(0)

# 生成模拟数据集
X = np.random.rand(100, 1) # 输入特征，100个样本，每个样本一个特征
y = 2 + 3 * X + np.random.randn(100, 1) # 输出目标值，每个样本的目标值是2+3x+随机噪声

# 初始化模型的权重和偏置
w = np.random.rand(1) # 权重
b = np.random.rand(1) # 偏置

# 定义线性回归模型的前向传播函数
def forward(X, w, b):
    z = X * w + b
    return z

# 定义线性回归模型的损失函数
def loss(y_true, y_pred):
    return np.mean((y_true - y_pred)**2)

# 定义线性回归模型的反向传播函数
def backward(dz, w, b):
    dw = dz * X
    db = dz
    return dw, db

# 定义训练过程
def train(X, y, w, b, epochs=1000, learning_rate=0.01):
    for epoch in range(epochs):
        # 前向传播
        z = forward(X, w, b)
        
        # 计算损失
        loss_val = loss(y, z)
        
        # 反向传播
        dz = 2 * (y - z)
        dw, db = backward(dz, w, b)
        
        # 更新权重和偏置
        w -= learning_rate * dw
        b -= learning_rate * db
        
        # 输出当前epoch的损失
        if epoch % 100 == 0:
            print(f"Epoch {epoch}: Loss = {loss_val}")

# 训练模型
train(X, y, w, b)

# 输出最终模型参数
print(f"Final weights: {w}, bias: {b}")

# 预测
X_new = np.array([[0.5]])
z_new = forward(X_new, w, b)
print(f"Prediction for new input {X_new}: {z_new}")
```

- **模型初始化**：初始化模型的权重（`w`）和偏置（`b`）为随机值。
