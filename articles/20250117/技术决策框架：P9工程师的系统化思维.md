# database.py
class Database:
    def __init__(self, db_name):
        self.db_name = db_name
        self.conn = None

    def connect(self):
        self.conn = psycopg2.connect(
            dbname=self.db_name,
            user="username",
            password="password",
            host="localhost"
        )

    def close(self):
        self.conn.close()

    def execute_query(self, query):
        cursor = self.conn.cursor()
        cursor.execute(query)
        result = cursor.fetchall()
        cursor.close()
        return result
```

        result = cursor.fetchall()
