3. **动态演化**：整体论认为，系统的整体行为是一个动态演化的过程，而不是静态的平衡状态。

然而，整体论也存在一些局限性。首先，整体论可能难以具体描述系统组成部分之间的相互作用，从而无法提供明确的机制。其次，整体论的模型通常较为复杂，难以进行精确的数学描述和计算。

    return 1 / (1 + np.exp(-x))

# 定义神经网络前向传播
def forward(x):
    hidden = sigmoid(np.dot(x, weights_input_hidden) + bias_hidden)
    output = sigmoid(np.dot(hidden, weights_hidden_output) + bias_output)
    return output

# 定义神经网络反向传播
def backward(x, y):
    output_error = y - forward(x)
    d_output = output_error * (1 - forward(x))

    d_hidden = hidden_error * (1 - sigmoid(np.dot(x, weights_input_hidden) + bias_hidden))
    
    d_weights_input_hidden = x.T.dot(d_hidden)
    d_bias_hidden = d_hidden
    d_weights_hidden_output = hidden.T.dot(d_output)
    d_bias_output = d_output
    
    return d_weights_input_hidden, d_weights_hidden_output, d_bias_hidden, d_bias_output

# 定义训练函数
def train(x, y, epochs):

        d_weights_input_hidden, d_weights_hidden_output, d_bias_hidden, d_bias_output = backward(x, y)
        weights_input_hidden += d_weights_input_hidden
        weights_hidden_output += d_weights_hidden_output

        if epoch % 100 == 0:
            print(f"Epoch {epoch}: Loss = {np.mean((forward(x) - y) ** 2)}")

x = np.array([[0, 0], [0, 1], [1, 0], [1, 1]])
y = np.array([[0], [1], [1], [0]])

# 训练神经网络
