class SharedTool:
    def __init__(self, name, condition):
        self.name = name
        self.condition = condition

class Participant:
    def __init__(self, name, tool_list):
        self.name = name
        self.tool_list = tool_list

class SharedPlatform:
    def __init__(self):
        self.tool_list = []
        self.participant_list = []

    def add_shared_tool(self, tool):
        if self.check_tool_condition(tool):
            self.tool_list.append(tool)
            print(f"{tool.name} 已添加至共享平台。")
        else:
            print(f"{tool.name} 未满足条件，未添加。")

    def check_tool_condition(self, tool):
        return tool.condition == "可共享"

    def connect_participant(self, participant):
        self.participant_list.append(participant)
        print(f"{participant.name} 已连接到共享平台。")

# 示例代码

participant = Participant("张三", [tool])

platform.connect_participant(participant)
