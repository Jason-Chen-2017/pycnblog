#### 3.1.2 Python源代码实现

```python
import pandas as pd

# 数据脱敏函数
def deidentify_data(data):
    # 使用k-anonymity算法进行数据脱敏
    # 这里以k=3为例
    k = 3
    data['k_anonymized'] = data.groupby('category').nunique().clip(lower=k)
    return data

# 数据恢复函数
def reidentify_data(data, original_data):
    # 使用同态加密算法进行数据恢复
    # 这里以同态加密函数为f(x)=x^2为例
    original_values = original_data['original_column'].values
    encrypted_values = [x**2 for x in original_values]
    return encrypted_values

# 示例数据
data = pd.DataFrame({
    'category': ['A', 'B', 'C', 'A', 'B'],
    'original_column': [1, 2, 3, 4, 5]
})

# 脱敏数据
deidentified_data = deidentify_data(data)
print("脱敏数据：")
print(deidentified_data)

# 恢复数据
reidentified_data = reidentify_data(data, data)
print("\n恢复数据：")
print(reidentified_data)
```

```python
import pandas as pd
from sklearn import preprocessing
import cryptography

def main():
    # 示例数据
    data = pd.DataFrame({
        'category': ['A', 'B', 'C', 'A', 'B'],
        'original_column': [1, 2, 3, 4, 5]
    })

    # 数据脱敏
    k_anonymized = preprocessing.KBinsDiscretizer(n_bins=3, strategy='uniform').fit_transform(data[['original_column']])
    data['k_anonymized'] = k_anonymized

    # 数据恢复
    encrypted_data = [x**2 for x in data['original_column']]
    data['encrypted_data'] = encrypted_data

    # 数据存储
    data.to_csv('deidentified_data.csv', index=False)

    # 数据访问
    accessed_data = pd.read_csv('deidentified_data.csv')
    print("访问的脱敏数据：")
    print(accessed_data)

    # 数据恢复
    original_data = accessed_data['original_column'].apply(lambda x: int(x**0.5))
    print("\n恢复的原始数据：")
    print(original_data)

if __name__ == '__main__':
    main()
```
