```python
from flask import Flask, request
import sqlite3

app = Flask(__name__)

@app.route('/submit', methods=['POST'])
def submit():
    data = request.json
    con = sqlite3.connect('participants.db')
    cur = con.cursor()
    cur.execute("INSERT INTO participants (name, email) VALUES (?, ?)",
                (data['name'], data['email']))
    con.commit()
    con.close()
    return 'Success'

if __name__ == '__main__':
    app.run(debug=True)
```
