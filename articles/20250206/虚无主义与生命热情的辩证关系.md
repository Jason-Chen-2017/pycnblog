### 4.3 Python代码实现

```python
def nihilism_assessment(nihilism_level):
    if nihilism_level >= 70:
        return "高"
    elif 30 <= nihilism_level < 70:
        return "中"
    else:
        return "低"

def strategy_recommendation(nihilism_level):
    assessment = nihilism_assessment(nihilism_level)
    if assessment == "高":
        return "强烈建议寻找意义"
    elif assessment == "中":
        return "建议探索热情"
    else:
        return "保持现状"

n_level = 65
recommendation = strategy_recommendation(n_level)
print(f"推荐策略：{recommendation}")
```

from flask import Flask, request, jsonify

@app.route('/api/assess', methods=['POST'])
