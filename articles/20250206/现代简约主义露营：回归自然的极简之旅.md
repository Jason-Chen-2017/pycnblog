import numpy as np

def calculate_footprint(ground_area, tent_size):
    """
    计算露营区域的生态足迹
    :param ground_area: 地面面积，平方米
    :param tent_size: 帐篷尺寸，平方米
    :return: 生态足迹，千克CO2/年
    """
    # 每平方米每年平均CO2排放量（假设）
