import networkx as nx

def spread_influence(graph, initial_node):
    influence = {node: 0 for node in graph.nodes}
    influence[initial_node] = 1
    for node in graph.nodes:
        for neighbor in graph.neighbors(node):
            if influence[neighbor] < influence[node]:
                influence[neighbor] = influence[node] * 0.8
    return influence

# 创建示例图
