- **代码示例**：哈夫曼编码实现：

  ```python
  import heapq

  class Node:
      def __init__(self, value):
          self.value = value
          self.left = None
          self.right = None

  def build_huffman_tree(nodes):
      heap = []
      for node in nodes:
          heapq.heappush(heap, (node.value, node))
      while len(heap) > 1:
          min1 = heapq.heappop(heap)
          min2 = heapq.heappop(heap)
          merged = Node(min1[0] + min2[0])
          merged.left = min1[1]
          merged.right = min2[1]
          heapq.heappush(heap, (merged.value, merged))
      return heap[0][1]

  data = [4, 5, 6]
  nodes = [Node(i) for i in data]
  huffman_tree = build_huffman_tree(nodes)
  ```

  from cryptography.hazmat.primitives import padding
  from cryptography.hazmat.primitives.asymmetric import padding as asym_padding
  from cryptography.hazmat.primitives.asymmetric.rsa import RSAPublicKey

  def encrypt_message(message, public_key):
      cipher = asym_padding.RSAPadding(
          asymmetric=False,
          mode=asym_padding.PKCS1v1_5
      )
      encrypted = public_key.encrypt(message, cipher)
      return encrypted

  public_key = RSAPublicKey.load_pkcs1_der("public.pem")
  message = "保护城市记忆"
  encrypted = encrypt_message(message.encode(), public_key)
