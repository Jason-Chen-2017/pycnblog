```python
def life_narrative_analysis(data):
    # 数据筛选
    filtered_data = [item for item in data if item['重要性'] > 0.5]
    # 数据组织
    organized_data = {
        '早期经历': [],
        '中期发展': [],
        '后期成就': []
    }
    for item in filtered_data:
        if item['时间'] < 2010:
            organized_data['早期经历'].append(item)
        elif 2010 <= item['时间'] < 2020:
            organized_data['中期发展'].append(item)
        else:
            organized_data['后期成就'].append(item)
    # 叙事生成
    narrative = ""
    for phase in ['早期经历', '中期发展', '后期成就']:
        narrative += f"### {phase}\n"
        for event in organized_data[phase]:
            narrative += f"- {event['事件描述']}\n"
    return narrative
```

$$

$$

```python
class LifeNarrativeSystem:
    def __init__(self):
        self.events = []

    def add_event(self, event):
        self.events.append(event)

    def generate_narrative(self):
        narrative = "### 生命叙事\n"
        for event in self.events:
            narrative += f"- {event['时间']}: {event['事件描述']}\n"
        return narrative

# 示例使用
system = LifeNarrativeSystem()
system.add_event({'时间': 2015, '事件描述': '完成第一个项目'})
system.add_event({'时间': 2020, '事件描述': '发布开源项目'})
print(system.generate_narrative())
```
