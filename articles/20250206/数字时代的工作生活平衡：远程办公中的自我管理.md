```python
def time_management(tasks):
    sorted_tasks = sorted(tasks, key=lambda x: x['priority'])
    return sorted_tasks

tasks = [
    {'name': '完成报告', 'priority': 2},
    {'name': '处理邮件', 'priority': 1},
    {'name': '参加会议', 'priority': 3}
]
print(time_management(tasks))  # 输出按优先级排序的任务
```
