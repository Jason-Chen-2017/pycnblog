```python
def preprocess(text):
    text = text.lower()
    text = re.sub(r'[^\w]', ' ', text)
    return text.split()

def encode(input_tokens, vocab):
    return [vocab.index(token) for token in input_tokens]

input_text = "我喜欢音乐节"
tokens = preprocess(input_text)
vocab = ["我", "喜欢", "音乐节"]
input_ids = encode(tokens, vocab)
```
