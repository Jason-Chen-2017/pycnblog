### 6.2 核心代码实现  
以下是一个简单的温室环境监控系统的Python代码示例：  

```python
import time
import Adafruit_DHT

# 设置传感器类型和GPIO引脚
SENSOR_TYPE = Adafruit_DHT.DHT22
SENSOR_PIN = 4

# 初始化传感器
def get_sensor_data():
    humidity, temperature = Adafruit_DHT.read_retry(SENSOR_TYPE, SENSOR_PIN)
    return temperature, humidity

# 数据存储
def store_data(temperature, humidity):
    # 连接到数据库
    import mysql.connector
    cnx = mysql.connector.connect(user='admin', password='password', host='localhost')
    cursor = cnx.cursor()
    query = "INSERT INTO greenhouse_data (temperature, humidity) VALUES (%s, %s)"
    cursor.execute(query, (temperature, humidity))
    cnx.commit()
    cursor.close()
    cnx.close()

# 主程序
while True:
    temp, hum = get_sensor_data()
    print(f"Temperature: {temp}C, Humidity: {hum}%")
    store_data(temp, hum)
    time.sleep(60)
```
