def identify_memory(city_data):
    memory = []
    for event in city_data['历史事件']:
        if event['重要性'] > 0.7:
            memory.append(event)
    return memory

city_data = {
    '历史事件': [
        {'名称': '城市成立', '重要性': 0.9},
        {'名称': '工业革命', '重要性': 0.8},
        {'名称': '文化庆典', '重要性': 0.6}
    ]
}

result = identify_memory(city_data)
print(result)
