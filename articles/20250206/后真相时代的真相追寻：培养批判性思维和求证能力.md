#### 5.4.2 核心代码实现

```python
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.svm import SVC
from sklearn.pipeline import Pipeline
import numpy as np

# 数据预处理
def preprocess(text):
    # 文本清洗
    text = text.lower()
    return text

# 特征提取和模型训练
def train_model(X, y):
    vectorizer = TfidfVectorizer(preprocessor=preprocess)
    model = SVC()
    pipe = Pipeline([('vectorizer', vectorizer), ('model', model)])
    pipe.fit(X, y)
    return pipe

# 预测
def predict(pipe, text):
    return pipe.predict([text])[0]

# 示例数据
X = [
    "This is a true news.",
    "This is a fake news.",
    "Fake news example."
]
y = ['true', 'fake', 'fake']

# 训练模型
model = train_model(X, y)

# 测试
print(predict(model, "This is a fake news."))  # 输出 'fake'
print(predict(model, "This is a true news."))  # 输出 'true'
```
