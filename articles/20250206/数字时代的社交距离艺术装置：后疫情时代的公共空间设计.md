import numpy as np

def calculate_social_distance(people_positions):
    distances = []
    for i in range(len(people_positions)):
        for j in range(i+1, len(people_positions)):
            distance = np.sqrt((people_positions[i][0]-people_positions[j][0])**2 + (people_positions[i][1]-people_positions[j][1])**2)
            distances.append(distance)
    return distances

# 示例数据
