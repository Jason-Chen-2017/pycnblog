    # 检查用户名是否已存在
    if check_username_exists(username):
        return "用户名已存在"
    # 验证邮箱和密码的格式
    if not is_valid_email(email) or not is_valid_password(password):
        return "信息格式错误"
    # 创建新账户
    create_user_account(username, email, password)
    return "账户创建成功"

# 登录流程
def login_user(username, password):
    # 验证用户名和密码

    # 登录成功，返回用户ID
