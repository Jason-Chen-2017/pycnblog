### 5.2 系统核心实现源代码

以下是智能门把手手部卫生提醒系统的核心实现源代码，包括摄像头捕获、图像预处理、图像识别和提醒功能：

```python
import cv2
import numpy as np
import tensorflow as tf

# 初始化摄像头
cap = cv2.VideoCapture(0)

# 加载预训练的图像识别模型
model = tf.keras.models.load_model('hand_recognition_model.h5')

while True:
    # 采集图像
    ret, frame = cap.read()

    # 图像预处理
    frame = cv2.resize(frame, (224, 224))
    frame = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)

    # 图像识别
    features = model.predict(np.expand_dims(frame, axis=0))

    # 判断手部是否干净
    if features[0][0] < 0.5:
        # 触发提醒
        remind_to_wash_hands()
    else:
        # 无需提醒
        print("Hand is clean.")

    # 显示图像
    cv2.imshow('Frame', frame)

    # 按下'q'键退出循环
    if cv2.waitKey(1) & 0xFF == ord('q'):
        break

# 释放摄像头
cap.release()
cv2.destroyAllWindows()

def remind_to_wash_hands():
    print("Please wash your hands.")
```
