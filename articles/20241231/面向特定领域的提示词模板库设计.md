### 4.2 算法实现

以下是使用Python编写的提示词生成算法实现：

```python
import jieba
from collections import defaultdict

class PromptTemplateGenerator:
    def __init__(self, template_file):
        self.template_file = template_file
        self.templates = self.load_templates()

    def load_templates(self):
        templates = {}
        with open(self.template_file, 'r', encoding='utf-8') as f:
            for line in f:
                key, value = line.strip().split(':')
                templates[key] = value
        return templates

    def generate_prompt(self, text):
        words = jieba.lcut(text)
        prompts = []
        for word in words:
            if word in self.templates:
                prompts.append(self.templates[word])
        return prompts

if __name__ == "__main__":
    template_file = 'templates.txt'
    generator = PromptTemplateGenerator(template_file)
    text = "计算机编程是人工智能领域的重要分支。"
    prompts = generator.generate_prompt(text)
    print(prompts)
```
