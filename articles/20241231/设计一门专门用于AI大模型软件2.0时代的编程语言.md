```python
import numpy as np

def parallel_matrix_multiplication(A, B):
    n = A.shape[0]
    P = A.reshape(n, -1)
    Q = B.reshape(n, -1)
    R = np.zeros((n, n))

    for i in range(n):
        for j in range(n):
            R[i][j] = np.dot(P[i], Q[j])

    return R
```
