```python
import pandas as pd
import nltk
from nltk.corpus import stopwords

# 加载金融市场数据
market_data = pd.read_csv('market_data.csv')

# 数据预处理
nltk.download('stopwords')
stop_words = set(stopwords.words('english'))
market_data['cleaned_text'] = market_data['text'].apply(lambda x: ' '.join([word for word in x.split() if word not in stop_words]))
```

##### 4.2.2 数据分析与挖掘

```python
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.cluster import KMeans

# TF-IDF向量表示
tfidf_vectorizer = TfidfVectorizer()
tfidf_matrix = tfidf_vectorizer.fit_transform(market_data['cleaned_text'])

# K-Means聚类
kmeans = KMeans(n_clusters=3)
kmeans.fit(tfidf_matrix)
market_data['cluster'] = kmeans.predict(tfidf_matrix)
```

##### 4.2.3 投资策略生成

```python
# 基于聚类结果生成投资策略
def generate_investment_strategy(clusters):
    strategy = {}
    for cluster in clusters:
        # 对每个cluster中的数据计算平均值
        cluster_data = market_data[market_data['cluster'] == cluster]
        strategy[cluster] = cluster_data.mean().tolist()
    return strategy

strategy = generate_investment_strategy(kmeans.labels_)
```
