```python
import torchvision

for epoch in range(num_epochs):

        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()
```

```python
import mxnet as mx
from mxnet import gluon, init

# 定义模型
net = gluon.nn.Sequential()
with net.name_scope():
    net.add(gluon.nn.Conv2D(64, 3, padding=1))
    net.add(gluon.nn.Activation('relu'))
    net.add(gluon.nn.MaxPool2D(2))
    net.add(gluon.nn.Conv2D(128, 3, padding=1))
    net.add(gluon.nn.Activation('relu'))
    net.add(gluon.nn.MaxPool2D(2))
    net.add(gluon.nn.Flatten())
    net.add(gluon.nn.Dense(10))

# 初始化模型参数
net.initialize(init.Xavier(), ctx=mx.gpu(0))

# 定义损失函数和优化器
softmax_loss = gluon.loss.SoftmaxCrossEntropyLoss()
trainer = gluon.Trainer(net.collect_params(), 'adam', {'learning_rate': 0.001})

# 训练模型
for epoch in range(num_epochs):
    for data in train_data:
        data = data.as_in_context(mx.gpu(0))
        labels = labels.as_in_context(mx.gpu(0))
        with mx.autograd.record():
            outputs = net(data)
            loss = softmax_loss(outputs, labels)
        loss.backward()
        trainer.step(batch_size)
```
