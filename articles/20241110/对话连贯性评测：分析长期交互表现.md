### 5.1 伪代码实现

```python
def evaluate_coherence(dialog):
    # 加载预训练模型
    model = load_pretrained_model()

    # 分句处理
    sentences = split_dialog_into_sentences(dialog)

    # 评估每对相邻句子的连贯性
    coherence_scores = []
    for i in range(len(sentences) - 1):
        score = model.evaluate_similarity(sentences[i], sentences[i+1])
        coherence_scores.append(score)

    # 计算整体连贯性得分
    total_score = sum(coherence_scores) / len(coherence_scores)
    return total_score
```
