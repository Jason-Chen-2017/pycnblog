```python
import numpy as np
import heapq
import json

def count_frequencies(data):
    # 计算提示词的频率
    frequencies = {}
    for word in data:
        frequencies[word] = frequencies.get(word, 0) + 1
    return frequencies

def build_huffman_tree(frequencies):
    # 构建霍夫曼树
    heap = [[weight, [symbol, ""]] for symbol, weight in frequencies.items()]
    heapq.heapify(heap)
    while len(heap) > 1:
        lo = heapq.heappop(heap)
        hi = heapq.heappop(heap)
        for pair in lo[1:]:
            pair[1] = '0' + pair[1]
        for pair in hi[1:]:
            pair[1] = '1' + pair[1]
        heapq.heappush(heap, [lo[0] + hi[0]] + lo[1:] + hi[1:])
    return heap[0]

def generate_encoding_table(huffman_tree):
    # 生成编码表
    encoding_table = {}
    for pair in huffman_tree[1:]:
        symbol, code = pair
        encoding_table[symbol] = code
    return encoding_table

def encode_data(data, encoding_table):
    # 编码数据
    encoded_data = ""
    for word in data:
        encoded_data += encoding_table[word]
    return encoded_data

def decode_data(encoded_data, encoding_table):
    # 解码数据
    reverse_encoding_table = {v: k for k, v in encoding_table.items()}
    decoded_data = ""
    buffer = ""
    for bit in encoded_data:
        buffer += bit
        if buffer in reverse_encoding_table:
            decoded_data += reverse_encoding_table[buffer]
            buffer = ""
    return decoded_data

def huffman_encoding(data):
    # 计算提示词的频率
    frequencies = count_frequencies(data)
    # 构建霍夫曼树
    huffman_tree = build_huffman_tree(frequencies)
    # 生成编码表
    encoding_table = generate_encoding_table(huffman_tree)
    # 编码数据
    encoded_data = encode_data(data, encoding_table)
    return encoded_data

def huffman_decoding(encoded_data, encoding_table):
    # 解码数据
    decoded_data = decode_data(encoded_data, encoding_table)
    return decoded_data

# 示例
data = ["this", "is", "a", "test", "of", "huffman", "encoding"]
encoding_table = generate_encoding_table(count_frequencies(data))
encoded_data = huffman_encoding(data)
decoded_data = huffman_decoding(encoded_data, encoding_table)

print("Encoded data:", encoded_data)
print("Decoded data:", decoded_data)
```
