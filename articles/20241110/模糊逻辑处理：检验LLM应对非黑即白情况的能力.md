```python
import pandas as pd
from transformers import AutoTokenizer, AutoModelForSequenceClassification
from torch import nn

# 加载预训练模型
tokenizer = AutoTokenizer.from_pretrained("bert-base-uncased")
model = AutoModelForSequenceClassification.from_pretrained("bert-base-uncased")

# 准备数据集
data = pd.read_csv("non_binary_dataset.csv")
X = data["text"]
y = data["label"]

# 数据预处理
inputs = tokenizer(X.tolist(), padding=True, truncation=True, return_tensors="pt")

# 训练模型
optimizer = torch.optim.Adam(model.parameters(), lr=1e-5)
criterion = nn.CrossEntropyLoss()

for epoch in range(10):
    optimizer.zero_grad()
    outputs = model(**inputs)
    loss = criterion(outputs.logits, y)
    loss.backward()
    optimizer.step()

    if (epoch + 1) % 10 == 0:
        print(f"Epoch [{epoch + 1}/{10}], Loss: {loss.item()}")

# 评估模型
with torch.no_grad():
    outputs = model(**inputs)
    predictions = torch.argmax(outputs.logits, dim=1)
    accuracy = (predictions == y).float().mean()
    print(f"Test Accuracy: {accuracy.item()}")

```

```python
import numpy as np

# 模糊集合的定义
def triangular_fuzzy_number(alpha, beta, gamma):
    return lambda x: max(0, min(1, (alpha - x) * (beta - x) * (gamma - x)))

# 模糊逻辑规则库
rules = {
    "if rain then wet": triangular_fuzzy_number(0, 0.2, 0.5),
    "if sun then dry": triangular_fuzzy_number(0.8, 1, 1),
    "if snow then cold": triangular_fuzzy_number(0, 0.5, 1)
}

# 模糊推理
def fuzzy_inference(facts):
    conclusions = {}
    for rule, consequence in rules.items():
        antecedents = rule.split(" ")[1:]
        antecedent_values = [facts.get(ant, 0) for ant in antecedents]
        rule_value = np.min(antecedent_values)
        conclusions[rule] = rule_value * consequence(0.5)  # 以0.5为例
    return max(conclusions.values())

# 输入事实
facts = {"rain": 0.3, "sun": 0.6, "snow": 0.1}

# 进行模糊推理
conclusion = fuzzy_inference(facts)
print(f"Conclusion: {'wet' if conclusion >= 0.5 else 'not wet'}")
```

在这个案例中，我们定义了一个模糊逻辑规则库，并使用模糊推理函数来处理一个简单的事实集。通过输入事实，我们可以得到一个结论，指示当前环境的湿度状态。这种模糊逻辑处理方法可以很好地应用于非黑即白情况，特别是在环境监测和决策支持系统中。
