def decision_tree_learning(data):
    if isPure(data):
        return majorityClass(data)
    else:
        best_attribute = getBestAttribute(data)
        node = Node(best_attribute)
        for value in uniqueAttributeValues(data[best_attribute]):
            subset = filterData(data, best_attribute, value)
            node.addChild(value, decision_tree_learning(subset))
        return node
