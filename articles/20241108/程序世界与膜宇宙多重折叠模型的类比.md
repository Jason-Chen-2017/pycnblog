```python
import numpy as np
import matplotlib.pyplot as plt

def fold_membrane(membrane, angle):
    """
    将二维膜沿给定角度折叠。

    参数:
    membrane: NumPy数组，表示二维膜。
    angle: 折叠角度（以弧度为单位）。

    返回:
    折叠后的膜。
    """
    # 创建折叠后的膜
    new_membrane = np.zeros_like(membrane)
    
    # 计算折叠后的位置
    for i in range(membrane.shape[0]):
        for j in range(membrane.shape[1]):
            new_i = int(np.floor(i * np.cos(angle) - j * np.sin(angle)))
            new_j = int(np.floor(i * np.sin(angle) + j * np.cos(angle)))
            new_membrane[new_i, new_j] = membrane[i, j]
    
    return new_membrane

def plot_membrane(membrane):
    """
    绘制二维膜。

    参数:
    membrane: NumPy数组，表示二维膜。
    """
    plt.imshow(membrane, origin='lower', cmap='gray')
    plt.colorbar()
    plt.xlabel('X')
    plt.ylabel('Y')
    plt.title('Membrane')
    plt.show()

# 初始化膜
initial_membrane = np.random.rand(10, 10)

# 折叠膜
folded_membrane = fold_membrane(initial_membrane, np.pi / 4)

# 绘制折叠前后的膜
plt.figure(figsize=(10, 5))
plt.subplot(1, 2, 1)
plot_membrane(initial_membrane)
plt.subplot(1, 2, 2)
plot_membrane(folded_membrane)
plt.show()
```
