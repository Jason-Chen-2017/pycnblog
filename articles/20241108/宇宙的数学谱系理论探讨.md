### 6.2.3 源代码详细实现

以下是一个简单的Python代码示例，用于模拟宇宙中两个星系的相互运动。这个示例将基于牛顿的万有引力定律。

```python
import numpy as np
import matplotlib.pyplot as plt

# 定义星系的质量、初始位置和速度
mass1 = 1.0
mass2 = 0.5
pos1 = np.array([1.0, 0.0])
pos2 = np.array([-1.0, 0.0])
vel1 = np.array([0.0, 1.0])
vel2 = np.array([0.0, -1.0])

# 时间步长
dt = 0.01

# 运动模拟
steps = 1000
positions = [[pos1.copy(), pos2.copy()] for _ in range(steps)]

for i in range(1, steps):
    # 计算引力
    force = np.array([0.0, 0.0])
    for pos in [pos1, pos2]:
        diff = pos - pos1 if pos == pos2 else pos - pos2
        force += (mass2 if pos == pos1 else mass1) * diff / np.linalg.norm(diff)**3

    # 更新速度和位置
    vel1 += force * dt / mass1
    vel2 += force * dt / mass2
    pos1 += vel1 * dt
    pos2 += vel2 * dt
    positions[i][0] = pos1.copy()
    positions[i][1] = pos2.copy()

# 绘制星系运动轨迹
plt.figure(figsize=(8, 6))
for pos in positions:
    plt.plot(pos[0][0], pos[0][1], 'ro')
    plt.plot(pos[1][0], pos[1][1], 'bo')
plt.xlabel('X坐标')
plt.ylabel('Y坐标')
plt.title('星系相互运动模拟')
plt.show()
```
