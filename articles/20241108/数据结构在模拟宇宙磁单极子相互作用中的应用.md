import numpy as np
import matplotlib.pyplot as plt

def initialize_particles(num_particles, position_range):
    positions = np.random.uniform(-position_range, position_range, (num_particles, 3))
    velocities = np.random.uniform(-0.1, 0.1, (num_particles, 3))
    return positions, velocities

def calculate_forces(positions):
    num_particles = len(positions)
    forces = np.zeros((num_particles, 3))
    for i in range(num_particles):
        for j in range(num_particles):
            if i != j:
                distance = np.linalg.norm(positions[i] - positions[j])
                force = -np.cross(positions[i], positions[j]) / (distance**3)
                forces[i] += force
    return forces

def update_positions(positions, velocities, forces, delta_t):
    velocities += forces / 10 * delta_t
    positions += velocities * delta_t
    return positions

def simulate(num_steps, delta_t, position_range, num_particles):
    positions, velocities = initialize_particles(num_particles, position_range)
    forces = np.zeros((num_particles, 3))

    for step in range(num_steps):
        forces = calculate_forces(positions)
        positions = update_positions(positions, velocities, forces, delta_t)

        # 存储数据
        with open(f'data_{step}.txt', 'w') as f:
            for i in range(num_particles):
                f.write(f'{positions[i][0]}, {positions[i][1]}, {positions[i][2]}, {forces[i][0]}, {forces[i][1]}, {forces[i][2]}\n')

    return positions

num_steps = 1000
delta_t = 0.01
position_range = 10
num_particles = 10

positions = simulate(num_steps, delta_t, position_range, num_particles)

# 绘制磁单极子分布图

plt.xlabel('X Position')

plt.title('Magnetic Monopole Distribution')
plt.show()

for i in range(num_particles):
    for j in range(num_particles):
        if i != j:

            potential_energy += -np.log(distance)
