```python
import numpy as np
import matplotlib.pyplot as plt

def calculate_energy(masses):
    return 0.5 * (masses[0] + masses[1]) * np.pi * np.sqrt(3)

def create_particles(energy):
    particles = []
    for _ in range(int(energy)):
        particles.append({'mass': 1, 'velocity': np.random.randn()})
    return particles

def simulate_annihilation(positions, velocities, masses):
    particles = create_particles(calculate_energy(masses))
    positions = [p['position'] for p in particles]
    velocities = [p['velocity'] for p in particles]
    return positions, velocities

positions = [np.random.randn() for _ in range(1000)]
velocities = [np.random.randn() for _ in range(1000)]
masses = [1 for _ in range(1000)]

positions, velocities = simulate_annihilation(positions, velocities, masses)

plt.scatter(positions, velocities)
plt.xlabel('Position')
plt.ylabel('Velocity')
plt.show()
```
