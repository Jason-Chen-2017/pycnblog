```python
import random

# 定义团队成员类
class TeamMember:
    def __init__(self, name, skills, knowledge):
        self.name = name
        self.skills = skills
        self.knowledge = knowledge
        self.contribution = 0

    def evaluate_performance(self):
        self.contribution = random.uniform(0.5, 1.0)

# 定义团队管理类
class TeamManagement:
    def __init__(self, team_members, objectives):
        self.team_members = team_members
        self.objectives = objectives
        self.resource_allocation = {}

    def optimize_team_performance(self):
        for member in self.team_members:
            member.evaluate_performance()
            self.resource_allocation[member.name] = member.contribution

    def evaluate_team_performance(self):
        total_contribution = sum(self.resource_allocation.values())
        print(f"Total team contribution: {total_contribution}")

# 创建团队成员
members = [
    TeamMember("Alice", ["Python", "Data Analysis"], ["Machine Learning", "Data Science"]),
    TeamMember("Bob", ["Java", "Database Management"], ["Web Development", "System Architecture"]),
    TeamMember("Charlie", ["C++", "Embedded Systems"], ["Networking", "System Integration"])
]

# 创建团队管理实例
team_management = TeamManagement(members, ["Project Completion", "Client Satisfaction"])

# 优化团队绩效
team_management.optimize_team_performance()

# 评估团队绩效
team_management.evaluate_team_performance()
```
