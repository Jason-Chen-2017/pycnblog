```python
import matplotlib.pyplot as plt
import pandas as pd

def visualize_spiral_galaxy():
    # 读取星系数据
    data = pd.read_csv('spiral_galaxy.csv')
    
    # 绘制星系分布图
    plt.scatter(data['x'], data['y'])
    plt.xlabel('X坐标')
    plt.ylabel('Y坐标')
    plt.title('星系分布图')
    plt.show()

def visualize_black_hole_event_horizon():
    # 读取黑洞事件视界数据
    data = pd.read_csv('black_hole_event_horizon.csv')
    
    # 绘制黑洞事件视界图
    plt.scatter(data['x'], data['y'])
    plt.xlabel('X坐标')
    plt.ylabel('Y坐标')
    plt.title('黑洞事件视界图')
    plt.show()

def visualize_big_bang():
    # 读取宇宙大爆炸数据
    data = pd.read_csv('big_bang.csv')
    
    # 绘制宇宙大爆炸图
    plt.scatter(data['t'], data['a'])
    plt.xlabel('时间')
    plt.ylabel('宇宙膨胀因子')
    plt.title('宇宙大爆炸图')
    plt.show()

# 测试可视化工具
visualize_spiral_galaxy()
visualize_black_hole_event_horizon()
visualize_big_bang()

```
