# 提示词生成算法示例
import random

def generate_prompt(civilization):
    # 根据文明特点生成提示词
    if civilization == "X":
        return "尊敬的外星文明X，我们希望与您进行友好交流，您对此有何看法？"
    elif civilization == "Y":
        return "尊敬的外星文明Y，我们注意到您对我们的科技非常感兴趣，您是否有任何合作建议？"

# 提示词优化算法示例
def optimize_prompt(prompt, feedback):
    # 根据用户反馈优化提示词
    if "感谢" in feedback:
        return prompt.replace("希望", "期待")
    elif "不清楚" in feedback:
        return prompt.replace("有何看法", "请详细说明")

# 示例应用
prompt = generate_prompt("X")
print(prompt)
feedback = "感谢您的回答，但我们需要更详细的讨论。"
optimized_prompt = optimize_prompt(prompt, feedback)
print(optimized_prompt)
