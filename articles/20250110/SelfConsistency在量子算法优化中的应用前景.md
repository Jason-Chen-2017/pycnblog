import numpy as np
from qiskit import QuantumCircuit, Aer, execute
from qiskit.visualization import plot_bloch_vector
from scipy.linalg import eig

def qPCA(data, n_components):
    # 初始化量子系统
    n_qubits = data.shape[1]
    qc = QuantumCircuit(n_qubits)

    # 计算主成分
    U, S, V = np.linalg.svd(data.T)
    u = U[:, :n_components]
    
    # 量子操作
    for i in range(n_qubits):
        for j in range(n_components):
            qc.h(i)
            qc.rz(np.arccos(u[j][i]), i)
    
    # 量子测量
    qc.measure_all()

    # 执行量子程序
    backend = Aer.get_backend('qasm_simulator')
    job = execute(qc, backend, shots=1024)
    result = job.result()

    # 输出低维数据
    counts = result.get_counts(qc)
    state_counts = [int(counts[str(k)]) for k in range(len(counts))]
    state_vector = np.array(state_counts) / np.linalg.norm(state_counts)
    return np.dot(u, state_vector)

# 示例数据
data = np.array([[1, 2], [3, 4], [5, 6], [7, 8]])

# 运行QPCA算法
n_components = 2
result = qPCA(data, n_components)
print("低维数据：", result)
