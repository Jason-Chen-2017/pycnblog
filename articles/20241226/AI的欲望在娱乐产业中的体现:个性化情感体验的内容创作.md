#### 1.4.1.2 Python代码实现

```python
import numpy as np

def matrix_factorization(R, K, iter_num=100, lambda_=0.1):
    N, M = R.shape
    H = np.random.rand(K, N)
    V = np.random.rand(K, M)
    for i in range(iter_num):
        for k in range(K):
            for n in range(N):
                if R[n] > 0:
                    e = R[n] - np.dot(H[k], V[n])
                    dH = -2 * e * V[n]
                    dV = -2 * e * H[k]
                    if e >= 0:
                        dH[k] = dH[k] + lambda_ * H[k]
                        dV[n] = dV[n] + lambda_ * V[n]
        H = H - 0.01 * dH
        V = V - 0.01 * dV
    return np.dot(H, V)

R = np.array([[5, 3, 0, 1],
              [4, 0, 0, 2],
              [1, 1, 0, 5],
              [1, 0, 0, 4],
              [5, 4, 9, 0]])
K = 2
iter_num = 100
lambda_ = 0.1

R_pred = matrix_factorization(R, K, iter_num, lambda_)
print(R_pred)
```
