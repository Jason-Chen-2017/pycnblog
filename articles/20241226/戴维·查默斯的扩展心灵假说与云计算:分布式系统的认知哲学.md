#### Python源代码

接下来，我们将这个算法用Python实现。代码展示了节点初始化、任务分配、状态更新和结束条件等过程。

```python
import threading

# 假设我们有一个任务队列
task_queue = ["任务1", "任务2", "任务3"]

# 定义计算节点的类
class ComputeNode:
    def __init__(self, task):
        self.task = task
        self.completed = False

    def execute_task(self):
        print(f"节点 {threading.current_thread().name} 开始执行任务：{self.task}")
        # 模拟任务执行
        time.sleep(2)
        self.completed = True
        print(f"节点 {threading.current_thread().name} 完成任务：{self.task}")

    def is_completed(self):
        return self.completed

# 初始化计算节点
nodes = [ComputeNode(task) for task in task_queue]

# 分配任务到节点
threads = []
for node in nodes:
    thread = threading.Thread(target=node.execute_task, name=f"节点 {nodes.index(node)}")
    threads.append(thread)
    thread.start()

# 等待所有任务完成
for thread in threads:
    thread.join()

# 检查所有任务是否完成
all_completed = all(node.is_completed() for node in nodes)
print("所有任务完成：", all_completed)
```

在这个代码中，我们创建了一个`ComputeNode`类，每个节点独立执行任务，并通过线程模拟并行处理。代码展示了如何初始化节点、分配任务、执行任务以及检查任务完成情况。

import threading
import time

class ComputeNode:
    def __init__(self, task_id, controller_ip):
        self.task_id = task_id
        self.completed = False
        self.controller_ip = controller_ip

    def execute_task(self):
        print(f"Node {self.task_id} 开始执行任务...")
        time.sleep(2)  # 模拟任务执行时间
        self.completed = True
        print(f"Node {self.task_id} 完成任务。")

    def is_completed(self):
        return self.completed

    def send_status(self):
        # 模拟节点状态发送到控制层
        print(f"Node {self.task_id} 发送状态：{self.completed}")
        # 实际应用中，应通过网络发送状态到控制层

# 创建计算节点
nodes = [
    ComputeNode(0, "controller_ip"),
    ComputeNode(1, "controller_ip"),
    ComputeNode(2, "controller_ip"),
]

# 启动计算节点线程
node_threads = []
for node in nodes:
    thread = threading.Thread(target=node.execute_task, args=(node,))
    node_threads.append(thread)
    thread.start()

# 等待所有节点任务完成
for thread in node_threads:
    thread.join()
```

**控制层（Controller.py）**

import threading
import time
from ComputeNode import ComputeNode

class Controller:
    def __init__(self):
        self.nodes = []

    def add_node(self, node):
        self.nodes.append(node)

    def monitor_nodes(self):
        while True:
            for node in self.nodes:
                if not node.is_completed():
                    node.send_status()
                    time.sleep(1)  # 模拟状态检查时间
                else:
                    print(f"Node {node.task_id} 已完成任务。")
                    break
            else:
                print("所有节点任务已完成。")
                break

    def start_monitoring(self):
        monitoring_thread = threading.Thread(target=self.monitor_nodes)
        monitoring_thread.start()

# 创建控制层
controller = Controller()

# 添加计算节点
for node in nodes:
    controller.add_node(node)

# 启动监控线程
controller.start_monitoring()
```
