    return gender_role

def shape_essence(freedom, gender_role):
    if freedom:
        if gender_role == "男":
            essence = "自由选择的男性"
        elif gender_role == "女":
            essence = "自由选择的女性"
        else:
            essence = "非标准性别角色"
    else:
        essence = "受限的个人本质"
    return essence

    if essence == "自由选择的男性" or essence == "自由选择的女性":
        subjectivity = "高主观性"
    else:
        subjectivity = "低主观性"
    return subjectivity

    if subjectivity == "高主观性":
        absurdity = "存在有其意义"

        absurdity = "存在充满荒谬"
    return absurdity

    if absurdity == "存在有其意义":
        meaning = "有意义的生活"
    else:
        meaning = "探索生命的意义"

from flask import Flask, request, jsonify
from flask_sqlalchemy import SQLAlchemy

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'mysql+pymysql://username:password@localhost/gender_perspective'
db = SQLAlchemy(app)

class User(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    username = db.Column(db.String(80), unique=True, nullable=False)

class Article(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    title = db.Column(db.String(120), nullable=False)
    content = db.Column(db.Text, nullable=False)
    user_id = db.Column(db.Integer, db.ForeignKey('user.id'), nullable=False)

@app.route('/register', methods=['POST'])
def register():
    username = request.form['username']
    if User.query.filter_by(username=username).first():
        return jsonify({'error': '用户已存在'})
    new_user = User(username=username)
    db.session.add(new_user)
    db.session.commit()
    return jsonify({'message': '注册成功'})

@app.route('/login', methods=['POST'])
def login():
    username = request.form['username']
    user = User.query.filter_by(username=username).first()
    if not user:
        return jsonify({'error': '用户不存在'})
    # 这里应添加密码验证逻辑
    return jsonify({'message': '登录成功'})

@app.route('/articles', methods=['POST'])
def create_article():
    title = request.form['title']
    content = request.form['content']
    user_id = request.form['user_id']
    new_article = Article(title=title, content=content, user_id=user_id)
    db.session.add(new_article)
    db.session.commit()
    return jsonify({'message': '文章发表成功'})

@app.route('/articles', methods=['GET'])
def get_articles():
    articles = Article.query.all()
    return jsonify([{'id': article.id, 'title': article.title, 'content': article.content, 'user_id': article.user_id} for article in articles])

if __name__ == '__main__':
    db.create_all()
    app.run(debug=True)
