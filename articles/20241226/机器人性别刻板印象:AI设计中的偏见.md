```python
from flask import Flask, request, jsonify
from models import User, Robot, Task

app = Flask(__name__)

# 初始化数据库连接
db.init_app(app)

@app.route('/users', methods=['POST'])
def create_user():
    data = request.get_json()
    user = User(name=data['name'])
    db.session.add(user)
    db.session.commit()
    return jsonify(user.id)

@app.route('/robots', methods=['POST'])
def create_robot():
    data = request.get_json()
    robot = Robot(name=data['name'], user_id=data['user_id'])
    db.session.add(robot)
    db.session.commit()
    return jsonify(robot.id)

@app.route('/tasks', methods=['POST'])
def create_task():
    data = request.get_json()
    task = Task(name=data['name'], robot_id=data['robot_id'])
    db.session.add(task)
    db.session.commit()
    return jsonify(task.id)

if __name__ == '__main__':
    app.run(debug=True)
```
