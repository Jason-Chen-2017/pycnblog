#### 7.1.2 系统核心实现源代码

以下是系统核心实现的源代码。

```python
import numpy as np
import matplotlib.pyplot as plt

# 暗能量演化方程
def dark_energy_evolution(E0, rho, c, t):
    return E0 * (1 - rho * c**2 * t / 3)

# 暗能量随时间变化
def dark_energy_variation(E0, rho, c, t_max):
    t = np.linspace(0, t_max, 1000)
    E = dark_energy_evolution(E0, rho, c, t)
    return t, E

# 绘制暗能量随时间变化曲线
def plot_dark_energy_variation(t, E):
    plt.plot(t, E)
    plt.xlabel('Time')
    plt.ylabel('Dark Energy')
    plt.title('Dark Energy Variation over Time')
    plt.show()

# 测试
E0 = 1.0  # 初始暗能量
rho = 0.01  # 暗能量密度
c = 299792458  # 光速
t_max = 1000  # 时间上限

t, E = dark_energy_variation(E0, rho, c, t_max)
plot_dark_energy_variation(t, E)
```
