from scipy import spatial

def identify_phase_transition(data):
    # 拓扑空间分析

    # 相变识别

# 拓扑空间分析
def analyze_topology(data):
    # 计算点之间的距离
    distances = spatial.distance.pdist(data)
    # 构建邻接矩阵
    adj_matrix = spatial.distance.squareform(distances)
