```python
import numpy as np
from scipy.optimize import linprog

# 定义目标函数
objective = lambda x: -x[0] - x[1]

# 定义约束条件
constraints = [
    [1, 1, '<=', 10],  # 总成本不超过10
    [0.5, 0.5, '<=', 5],  # 总营养素不超过5
]

# 定义变量范围
x0 = np.array([0, 0])

# 求解最优解
result = linprog(objective, constraints=constraints, x0=x0)

# 输出结果
if result.success:
    print(f"最优解：{result.x}")
else:
    print("无法求解最优解")
```
