    expansion_rates = calculate_expansion_rate(distances, velocities)

if __name__ == '__main__':

import numpy as np
import matplotlib.pyplot as plt
import pandas as pd

def calculate_expansion_rate(distance, velocity):
    return distance / velocity

def visualize_expansion_rate(data):
    distances = data['distance']
    velocities = data['velocity']
    expansion_rates = calculate_expansion_rate(distances, velocities)

    plt.scatter(distances, expansion_rates)
    plt.xlabel('Distance (light-years)')
    plt.ylabel('Expansion Rate (light-years/km/s)')
    plt.title('Cosmic Expansion Rate')
    plt.show()

if __name__ == '__main__':
    data = pd.DataFrame({
        'distance': [100, 200, 300],
        'velocity': [10, 20, 30]
    })

    visualize_expansion_rate(data)
```
