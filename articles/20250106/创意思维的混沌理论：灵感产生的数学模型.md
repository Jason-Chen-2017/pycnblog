import numpy as np
import matplotlib.pyplot as plt

def lorenz_attractor(s, r, b):
    x, y, z = s
    dx = (r * (y - x)) * np.cos(z)
    dy = (x * (r - z)) * np.sin(z) - (b * y)
    dz = (x * y) * np.sin(z) + (x * z) * np.cos(z)
    return np.array([dx, dy, dz])

def generate_inspirations(num_steps, s0, r, b):
    s = s0
    inspirations = []
    for _ in range(num_steps):
        s += lorenz_attractor(s, r, b)
        if np.random.rand() < 0.1:  # 灵感触发条件
            inspirations.append(s.copy())
    return inspirations

s0 = np.array([1.0, 1.0, 1.0])
r = 28
b = 2.667
num_steps = 10000

inspirations = generate_inspirations(num_steps, s0, r, b)
