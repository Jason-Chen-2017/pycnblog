**3.2.2 Python源代码详解**

```python
import numpy as np

def attention_allocation(tasks):
    resources = 100  # 假设总资源为100
    task需求的 = np.array(tasks)  # 将任务需求转换为numpy数组
    
    allocation = np.zeros(len(tasks))  # 初始化资源分配数组
    
    for i, demand in enumerate(task需求的):
        if demand <= resources:
            allocation[i] = demand
            resources -= demand
        else:
            allocation[i] = resources
    
    return allocation

tasks = [30, 20, 10]  # 任务需求示例
allocation = attention_allocation(tasks)
print("资源分配：", allocation)
```

```python
import numpy as np
import matplotlib.pyplot as plt

def emotional_intensity(stimuli, alpha=0.5, beta=0.3):
    emotional_state = 0
    intensity = []

    for stimulus in stimuli:
        emotional_state = alpha * emotional_state + beta * stimulus
        intensity.append(emotional_state)

    return intensity

stimuli = [5, 3, -2, 1, -4]  # 外部刺激示例
```

社会影响力模型是一个描述个体在社会网络中影响他人行为的数学模型。模型中，个体的行为受到其直接联系者的影响，同时也会影响其联系者的行为。以下是一个简单的社会影响力模型的mermaid流程图：


**5.2.2 Python源代码详解**
```python
import numpy as np
import matplotlib.pyplot as plt

def social_influence(initial_state, neighbors, influence_prob):
    state = np.array(initial_state)
    n = len(state)

    for t in range(10):  # 运行10个时间步
        new_state = np.zeros(n)
        for i in range(n):
            influence = np.random.random()
            if influence < influence_prob:
                new_state[i] = state[neighbors[i]]
        
        state = new_state

    return state

initial_state = [1, 0, 0, 1, 0]  # 初始状态示例
neighbors = [1, 3, 4, 0, 2]  # 邻居索引示例
influence_prob = 0.5  # 影响概率示例
state = social_influence(initial_state, neighbors, influence_prob)
plt.plot(state)
plt.xlabel('Time')
plt.ylabel('State')
plt.title('Social Influence Model')
plt.show()
```

**5.2.3 数学模型和公式**
社会影响力模型可以用以下数学公式表示：
$$
s_i(t+1) = \sum_{j=1}^{n} p_{ij} \cdot s_j(t)
$$
其中，$s_i(t)$是时间$t$个体$i$的状态，$s_j(t)$是时间$t$个体$j$的状态，$p_{ij}$是影响概率，$n$是总个体数。

```python
import numpy as np

def attention_allocation(tasks):
    resources = 100  # 假设总资源为100
    task需求的 = np.array(tasks)  # 将任务需求转换为numpy数组
    
    allocation = np.zeros(len(tasks))  # 初始化资源分配数组
    
    for i, demand in enumerate(task需求的):
        if demand <= resources:
            allocation[i] = demand
            resources -= demand
        else:
            allocation[i] = resources
    
    return allocation

tasks = [30, 20, 10]  # 任务需求示例
allocation = attention_allocation(tasks)
print("资源分配：", allocation)
```

**4.2.2 Python源代码详解**
```python
import numpy as np
import matplotlib.pyplot as plt

def emotional_intensity(stimuli, alpha=0.5, beta=0.3):
    emotional_state = 0
    intensity = []

    for stimulus in stimuli:
        emotional_state = alpha * emotional_state + beta * stimulus
        intensity.append(emotional_state)

    return intensity

stimuli = [5, 3, -2, 1, -4]  # 外部刺激示例
intensity = emotional_intensity(stimuli)
plt.plot(intensity)
plt.xlabel('Time')
plt.ylabel('Emotional Intensity')
plt.title('Emotional Intensity Model')
plt.show()
```
