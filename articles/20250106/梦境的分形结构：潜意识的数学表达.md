```python
import numpy as np
import matplotlib.pyplot as plt
from scipy import ndimage

def fractal_dimension(Z, threshold=0.1):
    Z = (Z - Z.min()) / (Z.max() - Z.min())
    label_im, num_labels = ndimage.label(Z > threshold)
    sizes = ndimage.sum(Z, label_im, range(num_labels+1))
    if num_labels:
        max_size = np.max(sizes)
        fractal_dimension = np.log(max_size) / np.log(len(sizes))
    else:
        fractal_dimension = 0
    return fractal_dimension

# 生成分形图像
image = np.random.rand(100, 100)
dimension = fractal_dimension(image)
print(f"Fractal Dimension: {dimension}")

# 绘制分形图像
plt.imshow(image, cmap='gray')
plt.title(f"Fractal Image with Dimension: {dimension}")
plt.show()
```
