**models.py**：定义系统模型
```python
from django.db import models

class Employee(models.Model):
    name = models.CharField(max_length=100)
    email = models.EmailField(unique=True)
    training_modules = models.ManyToManyField('TrainingModule')

class TrainingModule(models.Model):
    description = models.TextField()
    duration = models.DurationField()

class SelfAssessment(models.Model):
    employee = models.ForeignKey(Employee, on_delete=models.CASCADE)
    questions = models.JSONField()
    scoring = models.JSONField()

class Feedback(models.Model):
    employee = models.ForeignKey(Employee, on_delete=models.CASCADE)
    comments = models.TextField()
    rating = models.IntegerField()
    timestamp = models.DateTimeField(auto_now_add=True)
```

**views.py**：定义系统视图
```python
from django.http import HttpResponse
from .models import Employee, TrainingModule, SelfAssessment, Feedback

def training_modules(request):
    modules = TrainingModule.objects.all()
    return HttpResponse({'modules': list(modules.values('id', 'description', 'duration'))})

def self_assessment(request, employee_id):
    employee = Employee.objects.get(id=employee_id)
    assessment = SelfAssessment.objects.get(employee=employee)
    return HttpResponse({'questions': assessment.questions, 'scoring': assessment.scoring})

def submit_feedback(request, employee_id):
    employee = Employee.objects.get(id=employee_id)
    comments = request.POST.get('comments')
    rating = request.POST.get('rating')
    Feedback.objects.create(employee=employee, comments=comments, rating=rating)
    return HttpResponse('Feedback submitted successfully')
```
