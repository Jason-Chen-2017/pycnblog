def f(t, x, y):
    # 牛顿法中的目标函数
    return x - t**2 * y

def df(t, x, y):
    # 目标函数的导数
    return 1 - 2 * t * y

def newton_method(t0, x, y, tol=1e-6):
    # 牛顿法求解碰撞时间
    t = t0
    while abs(f(t, x, y)) > tol:
        t = t - f(t, x, y) / df(t, x, y)
    return t

# 示例参数
