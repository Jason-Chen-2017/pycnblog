class CelestialBody:
    def __init__(self, mass, position, velocity):
        self.mass = mass
        self.position = position
        self.velocity = velocity

# 引力计算
def gravity_force(body1, body2):
    G = 6.674 * 10**-11  # 引力常数
    distance = np.linalg.norm(body2.position - body1.position)
    force_magnitude = G * body1.mass * body2.mass / distance**2
    force_direction = (body2.position - body1.position) / distance
    return force_magnitude * force_direction

# 时间步进
def time_step(bodies, dt):
    new_bodies = []
    for body in bodies:
        total_force = np.zeros(3)
        for other_body in bodies:
            if body != other_body:
                force = gravity_force(body, other_body)
                total_force += force
        new_velocity = body.velocity + total_force / body.mass * dt
        new_position = body.position + new_velocity * dt
        new_bodies.append(CelestialBody(body.mass, new_position, new_velocity))
    return new_bodies

# 模拟演化
def simulate(bodies, time_max, dt):
    times = []
    positions = []
    for t in np.arange(0, time_max, dt):
        times.append(t)
        positions.append([body.position for body in bodies])
        bodies = time_step(bodies, dt)
    return times, positions

# 初始化天体
