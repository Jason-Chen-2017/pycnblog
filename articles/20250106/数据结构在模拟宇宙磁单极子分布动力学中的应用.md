### Python源代码

```python
import numpy as np
import matplotlib.pyplot as plt

def init_particles(num_particles, dim):
    # 初始化磁单极子
    positions = np.random.rand(num_particles, dim) * 1000
    velocities = np.random.rand(num_particles, dim) * 10
    return positions, velocities

def update_particles(positions, velocities, magnetic_field):
    # 更新磁单极子位置
    new_positions = positions + velocities
    return new_positions

def calculate_magnetic_field(positions):
    # 计算磁场
    field = np.zeros_like(positions)
    for i, p1 in enumerate(positions):
        for j, p2 in enumerate(positions):
            if i != j:
                distance = np.linalg.norm(p1 - p2)
                field[i] += (1 / distance) * (p2 - p1)
    return field

def main():
    # 主函数
    num_particles = 100
    dim = 3
    iterations = 1000
    positions, velocities = init_particles(num_particles, dim)
    for _ in range(iterations):
        magnetic_field = calculate_magnetic_field(positions)
        velocities = update_particles(positions, velocities, magnetic_field)
        positions = update_particles(positions, velocities, magnetic_field)
    plt.scatter(*positions.T)
    plt.show()

if __name__ == "__main__":
    main()
```

### 数学模型与公式

磁单极子分布模拟的数学模型基于磁单极子之间的相互作用力。设磁单极子位置为 \( \mathbf{r}_i \)，则相互作用力为：

$$ F_i = \frac{1}{4\pi\epsilon_0} \frac{\mu_0}{|\mathbf{r}_i - \mathbf{r}_j|^3} (\mathbf{r}_i - \mathbf{r}_j) $$

其中，\( \epsilon_0 \) 是真空电容率，\( \mu_0 \) 是真空磁导率。

### 举例说明

假设有10个磁单极子，其位置分别为 \( \mathbf{r}_1, \mathbf{r}_2, ..., \mathbf{r}_{10} \)。首先计算每个磁单极子之间的相互作用力，然后根据这些力更新磁单极子的位置。通过迭代计算，最终得到磁单极子的稳定分布。

import numpy as np
import matplotlib.pyplot as plt

class ParticleDistributionSimulation:
    def __init__(self, particle_data):
        self.particle_data = particle_data
    
    def execute_simulation(self):
        positions = self.particle_data['positions']
        velocities = self.particle_data['velocities']
        iterations = 1000
        
        for _ in range(iterations):
            magnetic_field = self.calculate_magnetic_field(positions)
            velocities = self.update_particles(positions, velocities, magnetic_field)
            positions = self.update_particles(positions, velocities, magnetic_field)
        
        plt.scatter(*positions.T)
        plt.show()

    def calculate_magnetic_field(self, positions):
        field = np.zeros_like(positions)
        for i, p1 in enumerate(positions):
            for j, p2 in enumerate(positions):
                if i != j:
                    distance = np.linalg.norm(p1 - p2)
                    field[i] += (1 / distance**3) * (p2 - p1)
        return field

    def update_particles(self, positions, velocities, magnetic_field):
        new_positions = positions + velocities
        return new_positions

def main():
    num_particles = 100
    dim = 3
    
    positions = np.random.rand(num_particles, dim) * 1000
    velocities = np.random.rand(num_particles, dim) * 10
    
    particle_data = {'positions': positions, 'velocities': velocities}
    simulation = ParticleDistributionSimulation(particle_data)
    simulation.execute_simulation()

if __name__ == "__main__":
    main()
