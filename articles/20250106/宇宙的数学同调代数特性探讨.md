def preprocess_data(data):
    # 清洗数据
    data = data.replace([np.inf, -np.inf], np.nan)
    data = data.dropna()

    homology_info = []
    for d in data:
        homology_info.append(homology(d))
    return homology_info

    return len(d) - 1

# 数据可视化

    # 加载数据

    # 数据预处理
    data = preprocess_data(data)
    
    # 同调信息提取
    homology_info = extract_homology_info(data)
    
    # 数据可视化
    visualize_data(homology_info)
