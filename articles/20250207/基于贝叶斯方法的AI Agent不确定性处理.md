- 5.3.2 贝叶斯推理模块代码

  ```python
  import numpy as np
  from scipy.stats import norm

  class BayesianInference:
      def __init__(self, prior_mean, prior_var, likelihood_mean, likelihood_var):
          self.prior_mean = prior_mean
          self.prior_var = prior_var
          self.likelihood_mean = likelihood_mean
          self.likelihood_var = likelihood_var

      def update(self, evidence):
          # 置信度更新
          precision = 1 / self.prior_var + 1 / self.likelihood_var
          mean = (self.prior_mean / self.prior_var + evidence / self.likelihood_var) / precision
          var = 1 / precision
          return norm.rvs(mean, np.sqrt(var), size=len(evidence))
  ```
