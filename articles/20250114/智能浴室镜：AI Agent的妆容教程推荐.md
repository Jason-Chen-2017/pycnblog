    faces = face_cascade.detectMultiScale(gray, scaleFactor=1.1, minNeighbors=5)
    for (x, y, w, h) in faces:
        cv2.rectangle(image, (x, y), (x+w, y+h), (255, 0, 0), 2)
    return image

def recommend_makeup(image):
    # 妆容教程推荐
    face = detect_face(image)
    face = cv2.resize(face, (128, 128))
    face = face / 255.0
    face = np.expand_dims(face, axis=0)
    face = np.expand_dims(face, axis=-1)
    prediction = makeup_model.predict(face)
    return prediction

def monitor_health(image):
    # 健康监测
    face = detect_face(image)
    face = cv2.resize(face, (128, 128))
    face = face / 255.0
    face = np.expand_dims(face, axis=0)
    face = np.expand_dims(face, axis=-1)
    prediction = health_model.predict(face)
    return prediction

def main():
    # 主函数
    camera = cv2.VideoCapture(0)
    while True:
        ret, frame = camera.read()
        if not ret:
            break
        frame = detect_face(frame)
        makeup_prediction = recommend_makeup(frame)
        health_prediction = monitor_health(frame)
        cv2.imshow('Smart Bathroom Mirror', frame)
        if cv2.waitKey(1) & 0xFF == ord('q'):
            break
    camera.release()
    cv2.destroyAllWindows()

if __name__ == '__main__':
    main()
