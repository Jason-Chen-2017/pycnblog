##### 3.2 AI Agent节水算法的Python源代码

以下是AI Agent节水算法的Python源代码示例：

```python
import pandas as pd
from sklearn.ensemble import RandomForestRegressor

# 数据采集
data = pd.read_csv('water_usage_data.csv')

# 数据处理
X = data[['flow_rate', 'time']]
y = data['water_consumption']

# 模型训练
model = RandomForestRegressor()
model.fit(X, y)

# 节水策略生成
def generate_saving_strategy(flow_rate, time):
    water_consumption = model.predict([[flow_rate, time]])
    optimized_flow_rate = flow_rate * (1 - water_consumption / max(water_consumption))
    return optimized_flow_rate

# 策略执行
def execute_strategy(optimized_flow_rate):
    print(f"Optimized flow rate: {optimized_flow_rate} L/min")
```

```python
import time
import serial
from sklearn.ensemble import RandomForestRegressor

# 模型训练
def train_model(data):
    X = data[['flow_rate', 'time']]
    y = data['water_consumption']
    model = RandomForestRegressor()
    model.fit(X, y)
    return model

# 节水策略生成
def generate_saving_strategy(model, flow_rate, time):
    water_consumption = model.predict([[flow_rate, time]])
    optimized_flow_rate = flow_rate * (1 - water_consumption / max(water_consumption))
    return optimized_flow_rate

# 策略执行
def execute_strategy(optimized_flow_rate):
    print(f"Optimized flow rate: {optimized_flow_rate} L/min")

# 设备连接
ser = serial.Serial('COM3', 9600)

# 模型训练
data = pd.read_csv('water_usage_data.csv')
model = train_model(data)

# 策略执行
while True:
    line = ser.readline()
    if line:
        flow_rate, time = line.decode('utf-8').split(',')
        optimized_flow_rate = generate_saving_strategy(model, float(flow_rate), float(time))
        execute_strategy(optimized_flow_rate)
```
