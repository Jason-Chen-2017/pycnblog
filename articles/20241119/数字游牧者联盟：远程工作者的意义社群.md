# 数字游牧者联盟知识共享代码示例

class KnowledgeShare:
    def __init__(self, members):
        self.members = members
        self.knowledge_base = {}

    def add_knowledge(self, member, knowledge):
        if member in self.members:
            self.knowledge_base[member] = knowledge
            print(f"{member} contributed knowledge: {knowledge}")
        else:
            print(f"{member} is not a member of the alliance.")

    def share_knowledge(self):
        for member, knowledge in self.knowledge_base.items():
            print(f"Sharing knowledge from {member}: {knowledge}")

# 创建联盟和成员
alliance = KnowledgeShare(["Alice", "Bob", "Charlie"])

# 添加知识
alliance.add_knowledge("Alice", "Python programming")
alliance.add_knowledge("Bob", "Machine Learning")
alliance.add_knowledge("Charlie", "Blockchain")

# 分享知识
alliance.share_knowledge()
