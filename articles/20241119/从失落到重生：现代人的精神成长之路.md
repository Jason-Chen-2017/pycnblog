#### 源代码详细实现和代码解读

1. **后端代码实现**：

   ```javascript
   // server.js
   const express = require('express');
   const mongoose = require('mongoose');
   const SpiritState = require('./models/SpiritState');

   const app = express();
   const port = 3000;

   // 连接数据库
   mongoose.connect('mongodb://localhost:27017/spirit-growth-system', {
       useNewUrlParser: true,
       useUnifiedTopology: true,
   });

   // 解析JSON请求体
   app.use(express.json());

   // 添加精神状态数据
   app.post('/spirit-state', async (req, res) => {
       const { state, description } = req.body;
       try {
           const newSpiritState = new SpiritState({
               state,
               description,
           });
           await newSpiritState.save();
           res.status(201).json(newSpiritState);
       } catch (error) {
           res.status(500).json({ message: 'Error adding spirit state' });
       }
   });

   // 获取所有精神状态数据
   app.get('/spirit-states', async (req, res) => {
       try {
           const spiritStates = await SpiritState.find();
           res.status(200).json(spiritStates);
       } catch (error) {
           res.status(500).json({ message: 'Error fetching spirit states' });
       }
   });

   app.listen(port, () => {
       console.log(`Server running on port ${port}`);
   });
   ```

   - 代码解读：
     - 使用 Express 创建 Web 服务器。
     - 连接 MongoDB 数据库。
     - 提供添加和获取精神状态数据的 API 接口。
