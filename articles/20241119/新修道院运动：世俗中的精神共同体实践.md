import math

def calculate_member_growth(initial_members, growth_rate, years):
    """
    计算共同体成员数量的增长
    :param initial_members: 初始成员数量
    :param growth_rate: 成长率
    :param years: 年数
    :return: 成员数量
    """
    member_growth = initial_members * (1 + growth_rate) ** years
    return math.ceil(member_growth)

# 示例
