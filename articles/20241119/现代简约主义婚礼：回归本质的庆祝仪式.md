```python
# 简约主义婚礼预算分配算法
def budget_allocation(total_budget, budget_items):
    """
    根据各项预算比例，合理分配总预算。
    
    :param total_budget: 总预算
    :param budget_items: 预算项目列表，格式为{'项目名称': 比例}
    :return: 分配后的预算字典
    """
    allocated_budget = {}
    total_ratio = sum(budget_items.values())
    
    for item, ratio in budget_items.items():
        allocated_budget[item] = (total_budget * ratio) / total_ratio
    
    return allocated_budget
```
