```python
# 科技禁食月参与记录器

import datetime
from collections import defaultdict

# 定义参与记录器类
class TechFastRecord:
    def __init__(self):
        self.participants = defaultdict(list)

    def add_participant(self, name, start_date, end_date):
        """
        添加一个参与者的记录。
        :param name: 参与者姓名。
        :param start_date: 参与开始日期。
        :param end_date: 参与结束日期。
        """
        self.participants[name].append((start_date, end_date))

    def get_participant_dates(self, name):
        """
        获取特定参与者的参与日期。
        :param name: 参与者姓名。
        :return: 参与日期列表。
        """
        return self.participants[name]

    def calculate_participation_duration(self, name):
        """
        计算特定参与者的参与时长。
        :param name: 参与者姓名。
        :return: 参与时长（以天为单位）。
        """
        dates = self.participants[name]
        total_days = 0
        for start, end in dates:
            total_days += (end - start).days
        return total_days

# 使用示例
record = TechFastRecord()
record.add_participant('Alice', datetime.datetime(2023, 4, 1), datetime.datetime(2023, 4, 30))
record.add_participant('Bob', datetime.datetime(2023, 5, 1), datetime.datetime(2023, 5, 15))

# 输出参与记录
for name, dates in record.participants.items():
    print(f"{name}的参与日期：{dates}")

# 输出参与时长
for name, duration in record.participants.items():
    print(f"{name}的参与时长：{record.calculate_participation_duration(name)}天")
```
