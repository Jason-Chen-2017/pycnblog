```python
import numpy as np

def calculate_weights(matrix):
    # 计算归一化矩阵
    normalized_matrix = matrix / matrix.sum(axis=1, keepdims=True)
    # 计算权重
    weights = normalized_matrix.mean(axis=0)
    return weights

# 示例矩阵
matrix = np.array([[0.2, 0.3, 0.5],
                   [0.4, 0.2, 0.4],
                   [0.1, 0.5, 0.4]])

weights = calculate_weights(matrix)
print("权重向量为:", weights)
```
