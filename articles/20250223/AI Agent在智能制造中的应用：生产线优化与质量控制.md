```python
# 示例代码：基于遗传算法的生产调度优化
import numpy as np
import random

def fitness(schedule, machines, tasks):
    # 计算调度的总时间
    max_time = 0
    for i in range(len(schedule)):
        task_time = tasks[i]
        machine_time = schedule[i] + task_time
        if machine_time > max_time:
            max_time = machine_time
    return -max_time

def mutate(schedule, tasks):
    # 随机选择一个位置进行变异
    pos = random.randint(0, len(schedule)-1)
    schedule[pos] = random.randint(0, len(tasks)-1)
    return schedule

def genetic_algorithm(tasks, machines, population_size=100, generations=50):
    # 初始化种群
    population = []
    for _ in range(population_size):
        schedule = [random.randint(0, len(tasks)-1) for _ in range(len(tasks))]
        population.append(schedule)
    
    for _ in range(generations):
        # 计算适应度
        fitness_values = [fitness(s, machines, tasks) for s in population]
        # 选择最优个体
        selected = [population[i] for i in range(len(population)) if fitness_values[i] == max(fitness_values)]
        # 交叉和变异
        new_population = []
        for _ in range(population_size):
            parent1 = selected[random.randint(0, len(selected)-1)]
            parent2 = selected[random.randint(0, len(selected)-1)]
            child = parent1.copy()
            # 单点交叉
            point = random.randint(1, len(parent1)-1)
            for i in range(point, len(parent1)):
                child[i] = parent2[i]
            child = mutate(child, tasks)
            new_population.append(child)
        population = new_population
    
    return population[0]

# 示例数据
tasks = [3, 5, 2, 8, 4]
machines = 2

best_schedule = genetic_algorithm(tasks, machines)
print("最佳调度方案:", best_schedule)
```
