- 支持向量机（SVM）的应用：  
$$
f(x) = \sum_{i=1}^{n} w_i x_i + b
$$

$$
V = \frac{E}{k} \times (1 + g)^{-n}

import numpy as np
from sklearn.svm import SVC
from sklearn.metrics import accuracy_score

# 数据加载
data = np.loadtxt('medical_data.csv', delimiter=',')

y = data[:, -1]

# 模型训练
model = SVC()
model.fit(X, y)

# 验证
