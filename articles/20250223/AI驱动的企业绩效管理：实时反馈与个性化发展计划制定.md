### 3.3 算法实现代码示例
```python
import pandas as pd
from sklearn.linear_model import LinearRegression

# 数据采集与预处理
data = pd.read_csv('performance_data.csv')
X = data[['任务完成度', '团队协作能力']]
y = data['绩效评分']

# 模型训练
model = LinearRegression()
model.fit(X, y)

# 反馈生成
def generate_feedback(employee_data):
    predicted_score = model.predict(employee_data)[0]
    if predicted_score > 85:
        return "你的绩效表现非常优秀，继续保持！"
    elif predicted_score > 70:
        return "你的绩效表现良好，建议在以下方面进行改进：..."
    else:
        return "你的绩效表现需要提升，请关注以下方面：..."

# 示例使用
employee_data = {'任务完成度': 80, '团队协作能力': 85}
print(generate_feedback(employee_data))
```
