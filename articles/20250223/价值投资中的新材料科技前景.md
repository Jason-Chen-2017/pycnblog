def calculate_intrinsic_value(revenue, growth_rate, discount_rate, cost_of_equity):
    # 计算自由现金流
    FCFF = revenue * (1 - growth_rate)
    # 折现
    present_value = FCFF / (1 + discount_rate)
    # 计算内在价值
