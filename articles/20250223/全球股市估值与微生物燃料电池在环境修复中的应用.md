```python
# 示例代码：粒子群优化算法
import numpy as np

def fitness_function(position):
    # 定义适应度函数
    return np.sum(position**2)

def particle_swarm_optimization(n_particles, n_iterations):
    # 初始化粒子位置和速度
    positions = np.random.rand(n_particles, 2)
    velocities = np.zeros((n_particles, 2))
    best_positions = positions.copy()
    best_fitness = np.inf
    
    for _ in range(n_iterations):
        fitness = np.apply_along_axis(fitness_function, 1, positions)
        if fitness.min() < best_fitness:
            best_positions = positions.copy()
            best_fitness = fitness.min()
        
        # 更新速度和位置
        velocities = 0.5 * velocities + 0.3 * np.random.rand(n_particles, 2)
        positions = positions + velocities
    
    return best_positions, best_fitness

# 示例调用
best_positions, best_fitness = particle_swarm_optimization(100, 50)
print("最优位置:", best_positions)
print("最优适应度:", best_fitness)
```
