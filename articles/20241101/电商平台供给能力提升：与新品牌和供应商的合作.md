# 实战案例：新品牌电商平台项目启动
class NewBrandProject:
    def __init__(self, brand_name, platform):
        self.brand_name = brand_name
        self.platform = platform
        self.suppliers = []

    def add_supplier(self, supplier):
        self.suppliers.append(supplier)

    def start_project(self):
        print(f"项目启动：{self.brand_name}在{self.platform}上启动新品牌项目")
        for supplier in self.suppliers:
            supplier.prepare_production()

    def display_project_info(self):
        print(f"品牌：{self.brand_name}")
        print(f"平台：{self.platform}")
        print(f"供应商：{', '.join([s.name for s in self.suppliers])}")
