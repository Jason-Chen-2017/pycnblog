```python
def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def forward(x, W, b):
    z = np.dot(x, W) + b
    h = sigmoid(z)
    return h

def backward(h, y, W, b):
    delta = (h - y) * h * (1 - h)
    dW = np.dot(x.T, delta)
    db = np.sum(delta)
    return dW, db

# 初始化参数

b = np.zeros((1,))

# 训练数据

y = np.array([[0], [1], [1], [0]])

# 前向传播
h = forward(x, W, b)

# 反向传播
dW, db = backward(h, y, W, b)

# 更新参数
W -= learning_rate * dW
b -= learning_rate * db
```
