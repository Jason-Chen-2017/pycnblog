# 数据预处理类
class DataPreprocessor:
    def preprocess_data(self, data: List[dict]) -> pd.DataFrame:
        df = pd.DataFrame(data)
        df['interaction_time'] = df['interaction'].apply(lambda x: x['time'])
        df['question_length'] = df['interaction'].apply(lambda x: len(x['question']))
        return df

# 认知负荷评估类
class CognitiveLoadAssessor:
    def assess_load(self, df: pd.DataFrame) -> float:
        total_time = df['interaction_time'].sum()
        total_length = df['question_length'].sum()
        cognitive_load = total_length / total_time
        return cognitive_load

class PromptAdjuster:
    def __init__(self, max_question_length: int = 50):
        self.max_question_length = max_question_length

        adjusted_prompts = []
        for prompt in prompts:
            if len(prompt) <= self.max_question_length:
                adjusted_prompts.append(prompt)
        return adjusted_prompts
