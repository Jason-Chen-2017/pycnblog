```python
import jieba

from nltk.tokenize import word_tokenize

def preprocess_text(text):
    # 分词
    words = jieba.lcut(text)
    # 去除停用词
    stop_words = set(stopwords.words('chinese'))
    words = [word for word in words if word not in stop_words]
    # 词性标注
    tagged_words = nltk.pos_tag(words)
```

```python
from sklearn.feature_extraction.text import TfidfVectorizer

def extract_query_terms(text):
    # 预处理文本
    preprocessed_text = preprocess_text(text)
    # 构建TF-IDF向量器
    vectorizer = TfidfVectorizer()
    # 训练模型
    X = vectorizer.fit_transform([text])
    # 提取高TF-IDF的词语
    query_terms = vectorizer.get_feature_names_out()[X.toarray()[0].argsort()[::-1][:5]]
    return query_terms
```

```python
from sklearn.naive_bayes import MultinomialNB
from sklearn.model_selection import train_test_split

def optimize_query_terms(text, labels):
    # 预处理文本
    preprocessed_text = preprocess_text(text)
    # 构建TF-IDF向量器
    vectorizer = TfidfVectorizer()
    # 训练模型
    X = vectorizer.fit_transform([text])
    y = labels
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2)

    clf.fit(X_train, y_train)
    # 预测优化后的提示词
    optimized_query_terms = clf.predict(X_test)
    return optimized_query_terms
```
