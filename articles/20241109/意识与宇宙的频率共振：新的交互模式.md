```python
import numpy as np
from scipy.fft import fft, fftfreq

def spectral_analysis(EEG_data, dT):
    n = len(EEG_data)
    freqs = fftfreq(n, dT)
    fft_data = fft(EEG_data)
    amplitudes = np.abs(fft_data / n)
    return freqs, amplitudes

# 示例数据
EEG_data = np.random.rand(1000)
dT = 1/100  # 采样率

# 执行频谱分析
freqs, amplitudes = spectral_analysis(EEG_data, dT)

# 绘制频谱图
import matplotlib.pyplot as plt

plt.plot(freqs, amplitudes)
plt.xlabel('Frequency (Hz)')
plt.ylabel('Amplitude')
plt.title('EEG Spectral Analysis')
plt.show()
```
