   F = G \frac{m_1 m_2}{r^2}

   v = v_0 + at

### 代码实现与解读

在本节中，我们将通过一个具体实例展示如何使用Python实现天体轨迹模拟算法。以下是天体轨迹模拟的完整代码，包括必要的函数和数据结构。

```python
import numpy as np

# 万有引力常数
G = 6.674 * 10**-11  # 单位：牛顿·米²/千克²

# 初始化天体类
class CelestialBody:
    def __init__(self, mass, position, velocity):
        self.mass = mass  # 质量（千克）
        self.position = np.array(position)  # 位置（米）
        self.velocity = np.array(velocity)  # 速度（米/秒）

# 计算引力
def calculate_gravity(celestial_body1, celestial_body2):
    distance_vector = celestial_body2.position - celestial_body1.position
    distance = np.linalg.norm(distance_vector)
    force_magnitude = G * celestial_body1.mass * celestial_body2.mass / distance**2
    force_vector = force_magnitude * distance_vector / distance
    return force_vector

# 更新位置和速度
def update_position_velocity(celestial_bodies, delta_t):
    for i in range(len(celestial_bodies)):
        for j in range(i+1, len(celestial_bodies)):
            force_vector = calculate_gravity(celestial_bodies[i], celestial_bodies[j])
            celestial_bodies[i].velocity += force_vector / celestial_bodies[i].mass * delta_t
            celestial_bodies[j].velocity -= force_vector / celestial_bodies[j].mass * delta_t

# 模拟天体运动
def simulate(celestial_bodies, delta_t, total_time):
    time = 0
    while time < total_time:
        update_position_velocity(celestial_bodies, delta_t)
        time += delta_t
        # 打印当前时间的天体位置
        for body in celestial_bodies:
            print(f"{time:.2f}秒：天体位置 {body.position}")

# 初始化两个天体
sun = CelestialBody(1.989 * 10**30, [0, 0], [0, 0])  # 太阳
earth = CelestialBody(5.972 * 10**24, [1.496 * 10**11, 0], [0, 29.784])  # 地球

# 模拟地球绕太阳的轨道
simulate([sun, earth], 1, 365.25 * 24 * 3600)

```
