def depression_model(S, C, B, delta_t):
    # 计算情绪状态的改变
    delta_S = 0.1 * (C - S) * delta_t

    # 计算行为状态的改变

    # 更新系统状态
    S = S + delta_S
    C = C + delta_C
    B = B + delta_B
    
    return S, C, B
