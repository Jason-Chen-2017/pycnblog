import matplotlib.pyplot as plt

def preprocess_data(microwave_data):
    # 去噪和异常值检测
    cleaned_data = np.abs(microwave_data)  # 去除负值
    cleaned_data = np.mean(cleaned_data, axis=1)  # 平均处理
    cleaned_data = (cleaned_data - np.mean(cleaned_data)) / np.std(cleaned_data)  # 归一化
    return cleaned_data

def detect_temperature_anomaly(pixel, thresholds):
    temperature_difference = abs(pixel[1] - pixel[0])
    if temperature_difference > thresholds['max_temperature_difference']:
        return True
    else:
        return False

def mark_bubble(pixel, bubble_structures):
    bubble_structures.append(pixel)
    return bubble_structures

# 加载宇宙微波背景辐射数据

# 设置阈值

# 数据预处理

# 检测泡沫结构
bubble_structures = []
for pixel in preprocessed_data:
    if detect_temperature_anomaly(pixel, thresholds):
        mark_bubble(pixel, bubble_structures)

# 输出泡沫结构
print(bubble_structures)

# 可视化泡沫结构
plt.scatter(*zip(*microwave_data), c='blue')
plt.scatter(*zip(*[p for p in bubble_structures]), c='red')
plt.show()
