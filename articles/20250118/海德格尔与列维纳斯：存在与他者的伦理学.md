```python
# 导入必要的库
import mermaid
import matplotlib.pyplot as plt
import numpy as np

# 定义函数，用于绘制Mermaid图表
def draw_mermaid_chart(chart_type, data):
    """
    绘制Mermaid图表，并返回图表的Markdown代码。
    """
    chart = mermaid.Mermaid()
    chart.set_chart_type(chart_type)
    chart.set_chart_data(data)
    chart.render()
    return chart.get_markdown()

# 定义函数，用于比较哲学思想的异同点
def compare_thoughts(heidegger, levinas):
    """
    比较海德格尔与列维纳斯的哲学思想，并绘制ER图和类图。
    """
    # 创建ER图
    er_chart_data = """
    ERDiagram {
        Heidegger ||--|{ Levinas }||>：异同点
        Heidegger --|{ 伦理学 }||>：伦理学
        Levinas --|{ 人本主义 }||>：人本主义
    }
    """
    er_chart_markdown = draw_mermaid_chart('ERDiagram', er_chart_data)
    
    # 创建类图
    class_chart_data = """
    class Heidegger {
        +existentialism
        +Dasein
    }
    class Levinas {
        +humanism
        +The Other
    }
    """
    class_chart_markdown = draw_mermaid_chart('ClassDiagram', class_chart_data)
    
    # 返回Markdown代码
    return er_chart_markdown, class_chart_markdown

# 测试函数
heidegger = {
    'existentialism': '强调个体存在的自主性和自由',
    'Dasein': '存在的个体性'
}
levinas = {
    'humanism': '强调人与人之间的关系和他者的关怀',
    'The Other': '他者是绝对的，不可还原为物或工具'
}

er_chart_markdown, class_chart_markdown = compare_thoughts(heidegger, levinas)

# 打印Markdown代码
print("ER图Markdown代码：")
print(er_chart_markdown)
print("\n类图Markdown代码：")
print(class_chart_markdown)

# 使用matplotlib绘制ER图和类图
plt.figure(figsize=(8, 4))
plt.subplot(1, 2, 1)
plt.markdown(er_chart_markdown)
plt.title("ER图")
plt.subplot(1, 2, 2)
plt.markdown(class_chart_markdown)
plt.title("类图")
plt.tight_layout()
plt.show()
```
