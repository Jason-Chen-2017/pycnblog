import numpy as np
import matplotlib.pyplot as plt

# 参数设置
G = 6.67430e-11  # 万有引力常数
M_A = 1e12       # 星系A的质量
M_B = 1e12       # 星系B的质量
d0 = 1e13        # 初始距离
t_max = 1e8      # 模拟时间
dt = 1e3         # 时间步长

# 初始位置
x_A = [0, 0]
y_A = [0, 0]
x_B = [d0, 0]
y_B = [0, 0]

# 模拟过程
positions = []
for t in np.arange(0, t_max, dt):
    # 计算引力
    dx = x_B[0] - x_A[0]
    dy = y_B[0] - y_A[0]
    distance = np.sqrt(dx**2 + dy**2)
    F = G * M_A * M_B / distance**2
    
    # 更新位置
    x_A[0] += F[0] * dt
    y_A[0] += F[1] * dt
    x_B[0] += F[2] * dt
    y_B[0] += F[3] * dt
    
    # 保存位置
    positions.append([x_A[0], y_A[0], x_B[0], y_B[0]])

# 绘制结果
plt.plot([p[0] for p in positions], [p[1] for p in positions], label='星系A')
plt.plot([p[2] for p in positions], [p[3] for p in positions], label='星系B')
plt.legend()
plt.show()
