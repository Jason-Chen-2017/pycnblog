#### 最短路径算法

最短路径算法用于计算从源节点到目标节点的最短路径。在宇宙模型中，最短路径算法可以用于计算星系之间的最短距离。

```python
def shortest_path(graph, source, target):
    # 初始化路径长度
    distances = {node: float('infinity') for node in graph}
    distances[source] = 0
    
    # 初始化前驱节点
    predecessors = {node: None for node in graph}
    
    # 暂存已访问节点
    visited = set()
    
    while target not in visited:
        # 找到未访问节点中距离最短的节点
        current = min((distance, node) for node, distance in distances.items() if node not in visited)[1]
        
        # 访问当前节点
        visited.add(current)
        
        # 更新相邻节点的距离
        for neighbor, weight in graph[current].items():
            if neighbor not in visited:
                new_distance = distances[current] + weight
                if new_distance < distances[neighbor]:
                    distances[neighbor] = new_distance
                    predecessors[neighbor] = current
    
    # 构建路径
    path = []
    current = target
    while current is not None:
        path.insert(0, current)
        current = predecessors[current]
    
    return path
```

```python
from collections import defaultdict

def max_flow(graph, source, sink):
    # 初始化流量网络
    flow_network = defaultdict(dict)
    for node in graph:
        for neighbor, capacity in graph[node].items():
            flow_network[node][neighbor] = 0
    
    # 求最大流
    while True:
        # 求增广路径
        path = find_augmenting_path(graph, source, sink)
        if path is None:
            break
        
        # 在增广路径上调整流量
        min_capacity = float('infinity')
        for i in range(len(path) - 1):
            u, v = path[i], path[i + 1]
            min_capacity = min(min_capacity, flow_network[u][v])
        
        for i in range(len(path) - 1):
            u, v = path[i], path[i + 1]
            flow_network[u][v] += min_capacity
            flow_network[v][u] -= min_capacity
    
    # 返回最大流量
    return sum(flow_network[source][v] for v in flow_network[source])

def find_augmenting_path(graph, source, sink):
    # 使用DFS寻找增广路径
    stack = [(source, [])]
    visited = set()
    
    while stack:
        current, path = stack.pop()
        
        if current == sink:
            return path
        
        visited.add(current)
        
        for neighbor, capacity in graph[current].items():
            if neighbor not in visited and capacity > 0:
                stack.append((neighbor, path + [current]))
    
    return None
```

#### 最小生成树算法

最小生成树算法用于从图中生成一个包含所有节点的最小树。在宇宙模型中，最小生成树算法可以用于生成星系的引力网络。

```python
from heapq import heappop, heappush

def min_spanning_tree(graph):
    # 初始化最小生成树
    mst = defaultdict(dict)
    
    # 使用优先队列存储边
    edges = []
    
    # 构造优先队列
    for node in graph:
        for neighbor, weight in graph[node].items():
            if neighbor not in mst:
                heappush(edges, (weight, node, neighbor))
    
    # 选择最小权重边，构建最小生成树
    while edges:
        weight, u, v = heappop(edges)
        if v not in mst:
            mst[u][v] = weight
            mst[v][u] = weight
    
    return mst
```

```python
def calculate_gravity(x, y):
```
