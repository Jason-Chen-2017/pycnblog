import numpy as np
from scipy.integrate import solve_ivp
import matplotlib.pyplot as plt

# 模型参数
rho = 1000  # 海水密度 (kg/m^3)
nu = 1e-6  # 动力粘度 (m^2/s)
g = 9.81  # 重力加速度 (m/s^2)

# 离散化网格
dx = 10  # 网格间距 (m)
x = np.arange(0, 1000, dx)  # 网格节点

# 初始条件
u0 = np.zeros_like(x)  # 初始流速 (m/s)
p0 = np.zeros_like(x)  # 初始压力 (Pa)

# 纳维-斯托克斯方程
def ns_eqns(t, y):
    u, p = y
    du_dt = -1 / (rho * dx) * (np.gradient(p) - nu * np.gradient(u, x))
    dp_dt = -g * rho
    return [u, du_dt, p, dp_dt]

# 求解
t = np.linspace(0, 100, 1000)  # 时间步
sol = solve_ivp(ns_eqns, [t[0], t[-1]], [u0, p0], t_eval=t, method='RK45')

# 可视化
plt.figure(figsize=(10, 5))
plt.plot(sol.t, sol.y[0], label='流速 (m/s)')
plt.plot(sol.t, sol.y[2], label='压力 (Pa)')
plt.xlabel('时间 (s)')
plt.ylabel('值')
plt.legend()
plt.title('深海流体力学模型仿真结果')
plt.show()
