```python
import numpy as np
from scipy.integrate import odeint

def reaction_rate(y, t, k):
    A = y
    dA_dt = -k * A
    return dA_dt

A0 = 1.0  # Initial concentration of A
t = np.linspace(0, 10, 1000)  # Time points
k = 0.1  # Rate constant
solution = odeint(reaction_rate, A0, t, args=(k,))

plt.plot(t, solution)
plt.xlabel('Time (s)')
plt.ylabel('Concentration of A')
plt.title('First-Order Decay')
plt.show()
```
