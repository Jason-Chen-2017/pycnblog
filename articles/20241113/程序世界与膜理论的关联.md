$$

#### 5.3 源代码实现

以下是膜计算程序的核心源代码，包括膜类的定义、状态更新函数和可视化函数。

```python
import numpy as np
import matplotlib.pyplot as plt

class Membrane:
    def __init__(self, size):
        self.size = size
        self.cells = np.zeros((size, size), dtype=int)

    def update(self, neighbors_func):
        new_cells = np.zeros((self.size, self.size), dtype=int)
        for i in range(self.size):
            for j in range(self.size):
                neighbors = neighbors_func(i, j)
                new_state = np.mean(neighbors)
                new_cells[i, j] = new_state
        self.cells = new_cells

    def visualize(self):
        plt.imshow(self.cells, cmap='gray')
        plt.colorbar()
        plt.show()

def neighbors_func(i, j):
    neighbors = []
    for di in [-1, 0, 1]:
        for dj in [-1, 0, 1]:
            if di == 0 and dj == 0:
                continue
            ni, nj = i + di, j + dj
            if 0 <= ni < self.size and 0 <= nj < self.size:
                neighbors.append(self.cells[ni, nj])
    return neighbors

# 实例化膜
membrane = Membrane(100)

# 运行膜计算程序
for _ in range(100):
    membrane.update(neighbors_func)
    membrane.visualize()
```
