```python
# 创建量子电路
qc = QuantumCircuit(2)

# 应用 Hadamard 门产生初始态
qc.h(0)
qc.h(1)

# 应用控制非门，实现纠缠
qc.cx(0, 1)

# 测量量子比特
qc.measure_all()

# 执行量子电路
backend = Aer.get_backend('qasm_simulator')

# 输出测量结果
counts = result.get_counts(qc)
print(counts)

# 绘制 Bloch 多矢量图
statevector = qc.get_statevector()

```

   ```python

   ```
   应用 Hadamard 门对两个量子比特进行初始化，生成叠加态。
