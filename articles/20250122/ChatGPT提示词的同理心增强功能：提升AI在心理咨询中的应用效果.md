nlp = spacy.load("en_core_web_sm")

# 函数：情感分析
def sentiment_analysis(text):
    doc = nlp(text)
    sentiments = []
    for token in doc:
        sentiments.append(token.sentiment.polarity)
    return sum(sentiments) / len(sentiments)

# 函数：上下文理解
def context_understanding(text):
    doc = nlp(text)
    entities = doc.ents
    context = ""
    for entity in entities:
        if entity.label_ == "PERSON":
            context += entity.text + " "
    return context

# 函数：生成同理心增强回复
def generate_response(prompt):
    context = context_understanding(prompt)
    sentiment = sentiment_analysis(prompt)
    response = ChatGPT.generate_response(prompt, sentiment, context)
    return response

# ChatGPT类：生成回复

    def __init__(self):
        self.model = openai.Completion.create(
