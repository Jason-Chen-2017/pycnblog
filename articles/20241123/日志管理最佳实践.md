def preprocess_log_data(log_data):
    # 清洗日志数据
    cleaned_data = []
    for data in log_data:
        if not data.endswith('\n'):
            data += '\n'
        cleaned_data.append(data.strip())
    # 格式化日志数据
    formatted_data = []
    for data in cleaned_data:
        fields = data.split(',')
        formatted_data.append({
            'timestamp': fields[0],
            'level': fields[1],
            'message': fields[2:]
        })
    return formatted_data

以下是一个简单的日志收集器示例：

```python
import os
import time
import json
from fluent import Events

def collect_logs(source_path, logstash):
    while True:
        files = [f for f in os.listdir(source_path) if f.endswith('.log')]
        for file in files:
            with open(os.path.join(source_path, file), 'r') as f:
                lines = f.readlines()
                for line in lines:
                    log_entry = {
                        'timestamp': time.strftime('%Y-%m-%d %H:%M:%S', time.localtime()),
                        'level': 'INFO',
                        'message': line.strip()
                    }
                    logstash.emit('log-event', log_entry)
        time.sleep(60)

if __name__ == '__main__':
    logstash = Events()
    logstash.config_source(type='file', source='fluent.conf')
    logstash.start()
    collect_logs('/var/log', logstash)
```
