```python
def merge_conflicts(old_content, new_content):
    # 初始化合并结果
    result = ""
    
    # 比较新旧内容，找到不同之处
    for i in range(len(old_content)):
        if old_content[i] != new_content[i]:
            # 选择新内容
            result += new_content[i]
        else:
            # 保留旧内容
            result += old_content[i]
    
    return result
```
