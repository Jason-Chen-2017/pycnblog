import numpy as np

def classify(data):
    # 初始化参数
    left = 0
    right = len(data) - 1
    
    # 循环过程
    while left < right:
        # 分析左边的元素
        if data[left] > data[right]:
            # 交换元素
            data[left], data[right] = data[right], data[left]
        
        # 移动指针
        left += 1
        right -= 1
    
    # 返回排序后的数据
    return data

# 测试代码
