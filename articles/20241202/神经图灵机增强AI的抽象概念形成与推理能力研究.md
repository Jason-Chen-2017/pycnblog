import torch
import torch.nn as nn
import torch.optim as optim
import numpy as np

class NeuralTuringMachine(nn.Module):
    def __init__(self, input_size, memory_size, read_heads, hidden_size):
        super(NeuralTuringMachine, self).__init__()
        
        # 定义读写头
        self.read_heads = nn.ModuleList([
            nn.Linear(input_size, hidden_size) for _ in range(read_heads)
        ])
        
        # 定义控制单元
        self.controller = nn.Linear(input_size + hidden_size * read_heads, hidden_size)
        
        # 定义存储
        self.memory = nn.Parameter(torch.randn(memory_size, hidden_size))
        
        # 定义输出层
        self.output_layer = nn.Linear(hidden_size, 1)
        
    def forward(self, x):
        read_modes = []
        for read_head in self.read_heads:
            hidden = read_head(x)
            read_modes.append(hidden)
        
        controller_input = torch.cat(read_modes, dim=1)
        controller_output = self.controller(controller_input)
        
        read_addresses = []
        for i, read_mode in enumerate(read_modes):
            read_address = torch.sigmoid(read_mode + controller_output)
            read_addresses.append(read_address)
        
        read_data = [self.memory[addr] for addr in read_addresses]
        read_data = torch.cat(read_data, dim=1)
        
        output = self.output_layer(read_data)
        return output
    
    def update_memory(self, write_address, write_data):
        self.memory[write_address] = write_data
        
def train_ntm(ntm, x_train, y_train, epochs, lr):
    optimizer = optim.Adam(ntm.parameters(), lr=lr)
    criterion = nn.MSELoss()
    
    for epoch in range(epochs):
        optimizer.zero_grad()
        output = ntm(x_train)
        loss = criterion(output, y_train)
        loss.backward()
        optimizer.step()
        
        if epoch % 100 == 0:
            print(f'Epoch {epoch+1}/{epochs}, Loss: {loss.item()}')
            
def main():
    # 参数设置
    input_size = 10
    memory_size = 100
    read_heads = 2
    hidden_size = 20
    epochs = 1000
    lr = 0.001
    
    # 初始化NTM模型
    ntm = NeuralTuringMachine(input_size, memory_size, read_heads, hidden_size)
    
    # 准备训练数据
    x_train = torch.randn(100, input_size)
    y_train = torch.randn(100, 1)
    
    # 训练模型
    train_ntm(ntm, x_train, y_train, epochs, lr)
    
    # 测试模型

    output = ntm(x_test)
    print(output)

if __name__ == '__main__':
    main()

   - `forward` 方法实现前向传播，首先计算每个读写头的隐藏层输出，然后通过控制单元生成读写地址和读写模式。
