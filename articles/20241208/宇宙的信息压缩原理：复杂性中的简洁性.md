以下是一个简单的信息压缩算法——Huffman编码的Mermaid流程图：

```mermaid
    A[初始化] --> B[构建频率表]
    B --> C[构建Huffman树]
```

Huffman编码的Python实现：

```python
import heapq
import collections

def huffman_encode(text):
    # 计算频率表
    frequency = collections.Counter(text)
    
    # 构建频率堆
    heap = [[weight, [symbol, ""]] for symbol, weight in frequency.items()]
    heapq.heapify(heap)
    
    # 构建Huffman树
    while len(heap) > 1:
        lo = heapq.heappop(heap)
        hi = heapq.heappop(heap)
        for pair in lo[1:]:
            pair[1] = '0' + pair[1]
        for pair in hi[1:]:
            pair[1] = '1' + pair[1]
        heapq.heappush(heap, [lo[0] + hi[0]] + lo[1:] + hi[1:])
    
    # 生成编码表
    code = dict([(symbol, code) for pair in heap[0][1:] for symbol, code in [pair]])
    
    # 编码文本
    encoded_text = ''.join([code[symbol] for symbol in text])
    
    return encoded_text

def huffman_decode(encoded_text, code):
    # 解码文本
    decoded_text = []
    current_code = ""
    
    for bit in encoded_text:
        current_code += bit
        if current_code in code.values():
            decoded_text.append(code[current_code])
            current_code = ""
    
    return ''.join(decoded_text)

text = "hello world"
encoded_text = huffman_encode(text)
print(f"Encoded text: {encoded_text}")

decoded_text = huffman_decode(encoded_text, {v: k for k, v in huffman_encode(text).items()})
print(f"Decoded text: {decoded_text}")
```

```python
# 导入Huffman编码库
import pyhuffman

def compress_data(data):
    # 创建Huffman编码器
    encoder = pyhuffman.HuffmanEncoder()
    # 编码数据
    encoded_data = encoder.encode(data)
    return encoded_data

def decompress_data(encoded_data):
    # 创建Huffman解码器
    decoder = pyhuffman.HuffmanDecoder()
    # 解码数据
    decoded_data = decoder.decode(encoded_data)
    return decoded_data

# 测试
text = "hello world"
encoded_text = compress_data(text)
print(f"Encoded text: {encoded_text}")

decoded_text = decompress_data(encoded_text)
print(f"Decoded text: {decoded_text}")
```
