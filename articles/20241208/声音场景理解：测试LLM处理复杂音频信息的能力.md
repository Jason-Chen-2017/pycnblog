```python
import numpy as np
from scipy.signal import butter, lfilter
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Conv2D, MaxPooling2D, Flatten, Dense
from tensorflow.keras.optimizers import Adam
from tensorflow.keras.losses import CategoricalCrossentropy
from tensorflow.keras.metrics import Accuracy
from librosa import power_to_db, stft

# 4.2.1 数据预处理
def preprocess_audio(audio_signal, sample_rate):
    # 滤波去噪
    b, a = butter(5, 10/(0.5*sample_rate), 'low')
    filtered_signal = lfilter(b, a, audio_signal)
    # 归一化
    normalized_signal = filtered_signal / np.max(np.abs(filtered_signal))
    return normalized_signal

# 4.2.2 特征提取
def extract_features(audio_signal, sample_rate):
    # 短时傅里叶变换
    f, t, Z = stft(audio_signal, nperseg=1024, nfft=2048)
    # 计算功率谱
    power_spectrum = power_to_db(np.abs(Z), ref=np.max)
    return power_spectrum

# 4.2.3 模型训练
def train_model(x_train, y_train, epochs=10, batch_size=32):
    # 构建模型
    model = Sequential([
        Conv2D(32, (3, 3), activation='relu', input_shape=(None, None, 1)),
        MaxPooling2D((2, 2)),
        Flatten(),
        Dense(128, activation='relu'),
        Dense(10, activation='softmax')
    ])

    # 编译模型
    model.compile(optimizer=Adam(), loss=CategoricalCrossentropy(), metrics=[Accuracy()])

    # 训练模型
    model.fit(x_train, y_train, epochs=epochs, batch_size=batch_size)
    return model

# 4.2.4 模型预测
def predict(model, audio_signal, sample_rate):
    # 特征提取
    power_spectrum = extract_features(audio_signal, sample_rate)
    # 展平特征
    flattened_spectrum = power_spectrum.flatten()
    # 预测
    prediction = model.predict(flattened_spectrum.reshape(1, -1))
    return prediction.argmax()
```
