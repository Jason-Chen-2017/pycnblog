import pandas as pd
import numpy as np
import requests
from bs4 import BeautifulSoup
import sqlite3

# 数据采集
def collect_data(user_id):
    # 社交媒体API请求代码，这里用占位符表示
    response = requests.get(f'https://api.socialmedia.com/user/{user_id}')
    data = response.json()
    return data

# 数据处理
def process_data(data):
    processed_data = pd.DataFrame(data)
    # 数据清洗和处理
    processed_data['post_time'] = pd.to_datetime(processed_data['post_time'])
    processed_data.set_index('post_time', inplace=True)
    return processed_data

# 风险评估
def assess_risk(data):
    # 这里简化为统计发帖频率和点赞数
    frequency = data.shape[0] // (24 * 3600)  # 每天的发帖频率
    likes = data['likes'].sum()  # 点赞总数
    risk_score = frequency * likes
    return risk_score

# 数据存储
def store_data(data, user_id):
    conn = sqlite3.connect('addiction.db')
    cursor = conn.cursor()
    cursor.execute('''CREATE TABLE IF NOT EXISTS user_data (user_id INTEGER PRIMARY KEY, data TEXT)''')
    data_json = data.to_json()
    cursor.execute("INSERT INTO user_data (user_id, data) VALUES (?, ?)", (user_id, data_json))
    conn.commit()
    conn.close()

# 主程序
def main():
    user_id = input("请输入您的用户ID：")
    data = collect_data(user_id)
    processed_data = process_data(data)

#### 2.22.6 算法原理讲解

逻辑回归是一种广泛使用的概率预测模型，适用于分类问题。它的核心思想是通过线性模型估计每个特征对目标变量的影响，从而预测目标变量的概率。

在数字成瘾风险评估中，逻辑回归模型可以帮助我们估计每个用户活动特征对成瘾风险的贡献。模型的输出是一个概率值，表示用户成瘾的风险。具体来说，算法原理如下：

1. **线性模型**：逻辑回归模型通过一个线性函数将特征转换为概率。公式为：
   $$ 
   \text{logit}(p) = \log\left(\frac{p}{1-p}\right) = \beta_0 + \beta_1 x_1 + \beta_2 x_2 + \ldots + \beta_n x_n 
   $$
   其中，$p$ 是成瘾风险的概率，$x_1, x_2, \ldots, x_n$ 是特征值，$\beta_0, \beta_1, \beta_2, \ldots, \beta_n$ 是模型的参数。

2. **特征选择**：通过选择对成瘾风险评估有显著影响的特征，可以简化模型，提高预测准确性。特征选择方法（如卡方检验）可以帮助我们识别这些特征。

3. **模型训练**：使用训练数据集，通过最小化损失函数（如交叉熵损失），调整模型的参数，从而训练出一个能够准确预测成瘾风险的模型。

4. **风险评估**：通过将用户的数字媒介活动数据输入训练好的模型，我们可以得到一个概率值，表示用户成瘾的风险。这个概率值可以帮助我们评估用户的成瘾风险，并为用户提供个性化的建议。

通过上述步骤，我们可以构建一个基于机器学习的数字成瘾风险评估模型，帮助用户了解自己的成瘾风险，从而采取相应的措施来减少风险。### 附录E：系统分析与架构设计
