            # 遍历手部标记点
            for i, landmark in enumerate(hand_landmarks.landmark):
                # 获取标记点的坐标
                h, w, c = image.shape
                x = int(landmark.x * w)
                y = int(landmark.y * h)
                
                # 屏蔽电子产品
                if i == 0:
                    cv2.circle(image, (x, y), 10, (0, 0, 255), -1)
                    cv2.putText(image, 'Screen Off', (x+50, y-50), cv2.FONT_HERSHEY_SIMPLEX, 1, (0, 0, 255), 2)
