以下是一个简单的Flask应用程序，用于实现跨领域对话系统：

```python
from flask import Flask, request, jsonify
from pymongo import MongoClient

app = Flask(__name__)

# 连接MongoDB数据库
client = MongoClient('mongodb://localhost:27017/')
db = client['cross_disciplinary_dialogue']

# 用户表
users_collection = db['users']
# 话题表
topics_collection = db['topics']
# 帖子表
posts_collection = db['posts']

@app.route('/users', methods=['POST'])
def create_user():
    user_data = request.json
    users_collection.insert_one(user_data)
    return jsonify({'status': 'success', 'message': 'User created successfully.'})

@app.route('/topics', methods=['POST'])
def create_topic():
    topic_data = request.json
    topics_collection.insert_one(topic_data)
    return jsonify({'status': 'success', 'message': 'Topic created successfully.'})

@app.route('/topics', methods=['GET'])
def get_topics():
    topics = topics_collection.find()
    return jsonify({'status': 'success', 'topics': list(topics)})

@app.route('/topics/<topic_id>/posts', methods=['POST'])
def create_post(topic_id):
    post_data = request.json
    post_data['topic_id'] = topic_id
    posts_collection.insert_one(post_data)
    return jsonify({'status': 'success', 'message': 'Post created successfully.'})

@app.route('/topics/<topic_id>/posts', methods=['GET'])
def get_posts(topic_id):
    posts = posts_collection.find({'topic_id': topic_id})
    return jsonify({'status': 'success', 'posts': list(posts)})

if __name__ == '__main__':
    app.run(debug=True)
```
