def backward_propagation(error, weights, activations):
    delta = error * derivatives(activations[-1])
    deltas = [delta]
    for layer in reversed(range(len(activations) - 1)):
        delta = weights[layer + 1].transpose().dot(delta)
        delta = delta * derivatives(activations[layer])
        deltas.append(delta)
    return deltas[::-1]
