import time

def breath_control(duration, ratio=1:4):

    end_time = start_time + duration

        # 吸气
        print("深呼吸...")
        time.sleep(ratio[0])
        
        # 呼气
        print("慢慢呼气...")
        time.sleep(ratio[1])
