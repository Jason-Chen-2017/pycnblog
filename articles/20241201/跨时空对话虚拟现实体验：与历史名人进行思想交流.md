```csharp
using UnityEngine;

public class VRChat : MonoBehaviour
{
    public Material chatBubbleMaterial;
    public Transform chatBubblePrefab;

    private Queue<string> chatMessages = new Queue<string>();

    void Update()
    {
        if (Input.GetKeyDown(KeyCode.Space))
        {
            string message = "Hello, World!";
            chatMessages.Enqueue(message);
            DisplayChatMessage(message);
        }
    }

    void DisplayChatMessage(string message)
    {
        GameObject chatBubble = Instantiate(chatBubblePrefab);
        chatBubble.GetComponent<MeshRenderer>().material = chatBubbleMaterial;
        Text chatText = chatBubble.transform.GetChild(0).GetComponent<Text>();
        chatText.text = message;
    }
}
```
