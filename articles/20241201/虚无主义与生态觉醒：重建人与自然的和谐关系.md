```python
import numpy as np

# 遗传算法的编码与解码
def encode_chromosomes(length):
    return np.random.randint(2, size=length)

def decode(chromosome):
    return sum(chromosome) == 1

# 遗传算法的适应度函数
def fitness_function(chromosome):
    return decode(chromosome)

# 遗传算法的交叉与变异操作
def crossover(parent1, parent2):
    crossover_point = np.random.randint(0, len(parent1))
    child1 = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
    child2 = np.concatenate((parent2[:crossover_point], parent1[crossover_point:]))
    return child1, child2

def mutate(chromosome, mutation_rate):
    for i in range(len(chromosome)):
        if np.random.random() < mutation_rate:
            chromosome[i] = 1 - chromosome[i]
    return chromosome

# 遗传算法的主循环
def genetic_algorithm(population_size, generations, mutation_rate):
    population = encode_chromosomes(population_size)
    for generation in range(generations):
        fitness_scores = np.array([fitness_function(chromosome) for chromosome in population])
        sorted_population = population[fitness_scores.argsort()[::-1]]
        next_population = []
        for _ in range(int(population_size / 2)):
            parent1, parent2 = sorted_population[:2]
            child1, child2 = crossover(parent1, parent2)
            next_population.extend([mutate(child1, mutation_rate), mutate(child2, mutation_rate)])
        population = next_population
    return population
```
