def preprocess_text(text):
    # 去除特殊字符和标点符号
    text = re.sub('[^a-zA-Z0-9\s]+', '', text)
    # 分词
    tokens = tokenizer.tokenize(text)
    # 大小写转换
    tokens = [token.lower() for token in tokens]
    # 去除停用词
    tokens = [token for token in tokens if token not in stop_words]
    # 词干提取
    tokens = [lemmatizer.lemmatize(token) for token in tokens]
    # 字符串拼接
    processed_text = ' '.join(tokens)
    return processed_text

# 情感标记
def emotion_labeling(text):
    inputs = tokenizer(text, return_tensors='pt', padding=True, truncation=True)
    outputs = model(**inputs)
    logits = outputs.logits
    predicted_labels = torch.argmax(logits, dim=-1).flatten()
    return predicted_labels.tolist()

# 一致性评估
def consistency_evaluation(predictions, ground_truth):
    consistency = torch.equal(predictions, ground_truth).float()
    return consistency.mean()

# 结果分析
def analyze_results(predictions, ground_truth):
    accuracy = (predictions == ground_truth).float().mean()
    f1 = f1_score(ground_truth, predictions, average='weighted')
    return accuracy, f1

- **预训练模型**：如GPT、BERT等，先在大量无标注的文本数据上进行预训练，然后通过微调适应特定任务。
- **任务特定模型**：在预训练模型的基础上，进一步微调以适应特定任务，如问答系统、文本生成等。

- **强大的文本生成能力**：LLM能够生成连贯、自然的文本，广泛应用于文本生成、摘要、翻译等领域。
- **灵活的可扩展性**：LLM可以通过微调快速适应不同的任务和数据集。

在自然语言处理中，LLM的应用场景非常广泛，包括但不限于：

- **文本分类**：自动对文本进行情感分类、主题分类等。
- **问答系统**：通过对话生成回答用户的问题。
- **文本生成**：自动生成文章、故事、摘要等。
- **机器翻译**：将一种语言的文本翻译成另一种语言。

情境变量是影响情感表达的上下文因素。它们在不同的情境下可能对情感标记产生不同的影响。以下是一个情境变量对比表格：

**3.3 评估指标**

评估情感稳定性测试的指标主要包括：

def preprocess_text(text):
    # 去除特殊字符和标点符号
    text = re.sub('[^a-zA-Z0-9\s]+', '', text)
    # 分词
    tokens = tokenizer.tokenize(text)
    # 大小写转换
    tokens = [token.lower() for token in tokens]
    # 去除停用词
    tokens = [token for token in tokens if token not in stop_words]
    # 词干提取
    tokens = [lemmatizer.lemmatize(token) for token in tokens]
    # 字符串拼接
    processed_text = ' '.join(tokens)
    return processed_text

# 情感标记
def emotion_labeling(text):
    inputs = tokenizer(text, return_tensors='pt', padding=True, truncation=True)
    outputs = model(**inputs)
    logits = outputs.logits
    predicted_labels = torch.argmax(logits, dim=-1).flatten()
    return predicted_labels.tolist()

# 一致性评估
def consistency_evaluation(predictions, ground_truth):
    consistency = torch.equal(predictions, ground_truth).float()
    return consistency.mean()

# 结果分析
def analyze_results(predictions, ground_truth):
    accuracy = (predictions == ground_truth).float().mean()
    f1 = f1_score(ground_truth, predictions, average='weighted')
    return accuracy, f1
```
