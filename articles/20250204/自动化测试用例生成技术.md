```python
class RuleBasedTestGenerator:
    def __init__(self, rules):

    def generate_test_case(self, input_data):
        for rule in self.rules:
            if rule.matches(input_data):
                return rule.test_case
        return None

# 示例规则库
rules = [
    Rule('rule1', lambda data: data == 'value1', 'test_case1'),
    Rule('rule2', lambda data: data == 'value2', 'test_case2'),
    # ...
]

generator = RuleBasedTestGenerator(rules)
test_cases = [generator.generate_test_case(data) for data in input_data]
```
