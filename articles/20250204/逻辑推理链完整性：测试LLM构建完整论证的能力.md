            result = self.execute_case(case)
            print(f"Test case {case}: {result}")

# 逻辑推理链构建
class LogicalReasoningChain:
    def __init__(self, premises, rules, conclusion):
        self.premises = premises
        self.rules = rules
        self.conclusion = conclusion

    def build_chain(self):
        chain = []
        for premise in self.premises:
            rule = self.rules.get(premise)
            if rule:
                chain.append((premise, rule))
        return chain

    def is_valid(self):
        for premise, rule in self.chain:
            if not rule.is_valid(premise):
                return False
        return True

# 评估报告生成
class AssessmentReport:
    def __init__(self, cases):
        self.cases = cases

    def generate_report(self):
        report = ""
        for case in self.cases:
            report += f"Test case {case}: {'Valid' if case.is_valid() else 'Invalid'}\n"
        return report
