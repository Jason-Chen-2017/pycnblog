#### 4.2.1 核心代码展示

以下是一个简单的 Python 代码示例，展示了如何使用 TensorFlow 和 PyTorch 分别训练和评估一个文本分类模型：

```python
import tensorflow as tf
import torch
from tensorflow.keras.preprocessing.text import Tokenizer
from tensorflow.keras.preprocessing.sequence import pad_sequences
from sklearn.model_selection import train_test_split
from sklearn.metrics import accuracy_score, recall_score, precision_score, f1_score

# TensorFlow 代码示例
def train_tf_model(text_data, labels):
    # 数据预处理
    tokenizer = Tokenizer(num_words=10000)
    tokenizer.fit_on_texts(text_data)
    sequences = tokenizer.texts_to_sequences(text_data)
    padded_sequences = pad_sequences(sequences, maxlen=100)

    # 划分训练集和验证集
    X_train, X_val, y_train, y_val = train_test_split(padded_sequences, labels, test_size=0.2)

    # 创建模型
    model = tf.keras.Sequential([
        tf.keras.layers.Embedding(10000, 16),
        tf.keras.layers.GlobalAveragePooling1D(),
        tf.keras.layers.Dense(1, activation='sigmoid')
    ])

    # 编译模型
    model.compile(optimizer='adam', loss='binary_crossentropy', metrics=['accuracy'])

    # 训练模型
    model.fit(X_train, y_train, epochs=10, validation_data=(X_val, y_val))

    # 评估模型
    predictions = model.predict(X_val)
    predictions = (predictions > 0.5)
    print("TensorFlow Accuracy:", accuracy_score(y_val, predictions))
    print("TensorFlow Recall:", recall_score(y_val, predictions))
    print("TensorFlow Precision:", precision_score(y_val, predictions))
    print("TensorFlow F1 Score:", f1_score(y_val, predictions))

# PyTorch 代码示例
def train_pytorch_model(text_data, labels):
    # 数据预处理
    tokenizer = torch.utils.text.Tokenizer()
    tokenizer.fit_on_texts(text_data)
    sequences = tokenizer.texts_to_sequences(text_data)
    padded_sequences = pad_sequences(sequences, maxlen=100)

    # 划分训练集和验证集
    X_train, X_val, y_train, y_val = train_test_split(padded_sequences, labels, test_size=0.2)

    # 创建模型
    model = torch.nn.Sequential(
        torch.nn.Embedding(10000, 16),
        torch.nn.GlobalAveragePooling1D(),
        torch.nn.Linear(100, 1)
    )

    # 编译模型
    criterion = torch.nn.BCEWithLogitsLoss()
    optimizer = torch.optim.Adam(model.parameters())

    # 训练模型
    for epoch in range(10):
        for inputs, labels in zip(X_train, y_train):
            optimizer.zero_grad()
            outputs = model(inputs)
            loss = criterion(outputs, labels)
            loss.backward()
            optimizer.step()

    # 评估模型
    with torch.no_grad():
        predictions = model(X_val)
        predictions = (predictions > 0.5)
        print("PyTorch Accuracy:", accuracy_score(y_val.tolist(), predictions.tolist()))
        print("PyTorch Recall:", recall_score(y_val.tolist(), predictions.tolist()))
        print("PyTorch Precision:", precision_score(y_val.tolist(), predictions.tolist()))
        print("PyTorch F1 Score:", f1_score(y_val.tolist(), predictions.tolist()))

# 测试代码
text_data = ["This is a sample text.", "Another sample text.", "Third sample text."]
labels = [0, 1, 0]
train_tf_model(text_data, labels)
train_pytorch_model(text_data, labels)
```

4. **模型训练**：
   - 使用 TensorFlow 进行模型训练，使用验证集进行性能评估。
