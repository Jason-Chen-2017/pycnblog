from sklearn.preprocessing import StandardScaler
from sklearn.decomposition import PCA

class DataProcessor:
    def __init__(self):
        self.scaler = StandardScaler()
        self.pca = PCA(n_components=2)

    def clean_data(self, data):
        # 数据清洗
        data = np.array(data)
        data = data[data != 0]  # 去除无效数据
        return data

    def extract_features(self, data):
        # 特征提取
        data = self.scaler.fit_transform(data)
        data = self.pca.fit_transform(data)
        return data

    def fuse_data(self, hr_data, light_data):
        # 数据融合
        fused_data = np.concatenate((hr_data, light_data), axis=1)
        return fused_data

# 示例

fused_data = data_processor.fuse_data(extracted_hr_data, extracted_light_data)
