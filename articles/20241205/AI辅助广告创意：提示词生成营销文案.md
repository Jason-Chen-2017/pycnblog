$$
h_t = \text{sigmoid}(W_h \cdot [h_{t-1}, x_t] + b_h)

import tensorflow as tf
from tensorflow.keras.preprocessing.sequence import pad_sequences
from tensorflow.keras.layers import Embedding, LSTM, Dense
from tensorflow.keras.models import Sequential
from tensorflow.keras.optimizers import Adam
from tensorflow.keras.callbacks import EarlyStopping
from nltk.tokenize import word_tokenize
from gensim.models import Word2Vec

# 数据预处理
def preprocess_data(data):
    tokenized_data = [word_tokenize(text) for text in data]
    padded_data = pad_sequences(tokenized_data, maxlen=MAX_LENGTH, padding='post', truncating='post')
    return padded_data

# 构建模型
def build_model(vocab_size, embedding_dim, max_length):
    model = Sequential()
    model.add(Embedding(vocab_size, embedding_dim, input_length=max_length))
    model.add(LSTM(128, return_sequences=True))
    model.add(Dense(1, activation='sigmoid'))
    model.compile(optimizer=Adam(learning_rate=0.001), loss='binary_crossentropy', metrics=['accuracy'])
    return model

# 训练模型
def train_model(model, sequences, labels):
    early_stopping = EarlyStopping(monitor='val_loss', patience=5)
    model.fit(sequences, labels, epochs=EPOCHS, batch_size=BATCH_SIZE, callbacks=[early_stopping], validation_split=0.2)

# 生成提示词
def generate_prompt(model, input_text, max_length):
    tokenized_input = word_tokenize(input_text)
    padded_input = pad_sequences([tokenized_input], maxlen=max_length, padding='post', truncating='post')
    prediction = model.predict(padded_input)
    if prediction > 0.5:
        return input_text + "，你可能会感兴趣："
    else:
        return input_text + "，了解更多："

# 加载数据
data = ["智能手表具有健康监测功能，可以实时跟踪心率、睡眠质量等数据。", "想了解如何选择适合自己的智能手表吗？"]
sequences = preprocess_data(data)
labels = [1, 0]

# 构建和训练模型
model = build_model(vocab_size=len(sequences[0]), embedding_dim=EMBEDDING_DIM, max_length=MAX_LENGTH)
train_model(model, sequences, labels)

# 测试生成提示词
input_text = "智能手表具有健康监测功能，可以实时跟踪心率、睡眠质量等数据。"

print(prompt)
