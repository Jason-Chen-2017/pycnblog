##### 3.2 Python源代码讲解

```python
# 实体类定义
class Entity:
    def __init__(self, name, attributes):
        self.name = name
        self.attributes = attributes

# 关系类定义
class Relationship:
    def __init__(self, entities, type):
        self.entities = entities
        self.type = type

# 领域模型构建函数
def build_domain_model(entities, relationships):
    model = {}
    for entity in entities:
        model[entity.name] = entity
    for relationship in relationships:
        model[relationship.type].add(relationship)
    return model

# 实例化实体和关系
entity1 = Entity("Customer", {"name": "John Doe", "age": 30})
entity2 = Entity("Order", {"order_id": 1001, "status": "pending"})
relationship = Relationship([entity1, entity2], "Places")

# 构建领域模型
domain_model = build_domain_model([entity1, entity2], [relationship])

# 输出领域模型
print(domain_model)
```

##### 5.2 系统核心实现源代码

```python
from sqlalchemy import create_engine, Column, Integer, String, ForeignKey
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker, relationship

Base = declarative_base()

class Customer(Base):
    __tablename__ = 'customers'
    id = Column(Integer, primary_key=True)
    name = Column(String)

class Book(Base):
    __tablename__ = 'books'
    id = Column(Integer, primary_key=True)
    title = Column(String)

class Order(Base):
    __tablename__ = 'orders'
    id = Column(Integer, primary_key=True)
    customer_id = Column(Integer, ForeignKey('customers.id'))
    status = Column(String)

# 建立数据库连接
engine = create_engine('sqlite:///books.db')
Base.metadata.create_all(engine)
Session = sessionmaker(bind=engine)
session = Session()

# 创建用户
session.add(Customer(name="John Doe"))
session.commit()

# 创建书籍
session.add(Book(title="1984"))
session.commit()

# 创建订单
session.add(Order(customer_id=1, status="pending"))
session.commit()
```

##### 5.3 代码应用解读与分析

通过上述代码，我们创建了三个实体类：Customer、Book和Order。每个类都对应一个数据库表，并且通过外键关联起来，实现了领域建模的基本功能。
