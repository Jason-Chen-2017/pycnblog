**增强现实开发：** 增强现实开发是指创建能够将虚拟信息叠加到现实场景中的应用程序。这通常涉及图像识别、传感器数据处理和实时交互等复杂技术。

```python
import cv2
import numpy as np
from PyQt5.QtWidgets import QApplication, QWidget, QVBoxLayout, QPushButton
from PyQt5.QtGui import QIcon
from google.cloud import speech

class ARSystem(QWidget):
    def __init__(self):
        super().__init__()
        self.init_ui()

    def init_ui(self):
        self.setWindowTitle('提示词编程增强现实系统')
        self.setGeometry(100, 100, 800, 600)

        layout = QVBoxLayout()

        # 创建按钮和对应的提示词
        self.buttons = {}
        for keyword, action in self.actions().items():
            button = QPushButton(QIcon('icon.png'), keyword)
            button.clicked.connect(action)
            self.buttons[keyword] = button
            layout.addWidget(button)

        self.setLayout(layout)

    def actions(self):
        return {
            '前进': self.move_forward,
            '后退': self.move_backward,
            '左转': self.turn_left,
            '右转': self.turn_right
        }

    def move_forward(self):
        print('向前移动')

    def move_backward(self):
        print('向后移动')

    def turn_left(self):
        print('向左转')

    def turn_right(self):
        print('向右转')

    def recognize_speech(self):
        client = speech.SpeechClient()
        audio = client.record_audio_from_mic()
        print('正在识别语音...')
        transcription = client.recognize(audio)
        print('识别结果：', transcription.results[0].alternatives[0].transcript)

if __name__ == '__main__':
    app = QApplication([])
    ar_system = ARSystem()
    ar_system.recognize_speech()
    ar_system.show()
    app.exec_()
```
