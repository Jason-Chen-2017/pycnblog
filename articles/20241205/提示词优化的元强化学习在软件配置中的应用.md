        super().__init__()
        self.obs_ENCODING = layers.Dense(units=64, activation='relu')(tf.keras.Input(shape=obs_shape))
        self.action_ENCODING = layers.Dense(units=64, activation='relu')(tf.keras.Input(shape=action_shape))
        self.combined = layers.Concatenate()([self.obs_ENCODING, self.action_ENCODING])
        self.q_values = layers.Dense(units=action_shape[0], activation=None)(self.combined)

    def call(self, inputs, training=False):
        obs, action = inputs
        return self.q_values(tf.expand_dims(action, -1))

# 训练模型
def train_model(model, env, epochs, batch_size, learning_rate):
    obs_shape = env.observation_space.shape
    action_shape = env.action_space.shape

    # 训练数据
    obs = np.empty((0, *obs_shape))
    actions = np.empty((0,))
    rewards = np.empty((0,))
    dones = np.empty((0,))

    for epoch in range(epochs):
        state = env.reset()
        done = False
        while not done:
            action = model.predict(state)[0]
            next_state, reward, done, _ = env.step(action)
            obs = np.vstack((obs, state))
            actions = np.append(actions, action)
            rewards = np.append(rewards, reward)
            dones = np.append(dones, done)
            state = next_state

        # 更新模型
        model.fit([obs, actions], rewards + gamma * dones * (1 - dones), batch_size=batch_size, epochs=1, verbose=0)

# 模型评估
def evaluate_model(model, env, steps):
    obs = env.reset()
    done = False
    total_reward = 0
    while not done and steps > 0:
        action = model.predict(obs)[0]
        next_state, reward, done, _ = env.step(action)
        obs = next_state
        total_reward += reward
        steps -= 1
    return total_reward
