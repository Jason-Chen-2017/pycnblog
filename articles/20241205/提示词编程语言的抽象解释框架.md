```python
class Operation:
    def parse(self, expression):
        # 解析运算式
        pass

    def calculate(self, expression):
        # 计算结果
        pass

class Addition(Operation):
    def calculate(self, expression):
        a, b = expression.split('+')
        return int(a) + int(b)

class Subtraction(Operation):
    def calculate(self, expression):
        a, b = expression.split('-')
        return int(a) - int(b)

class Multiplication(Operation):
    def calculate(self, expression):
        a, b = expression.split('*')
        return int(a) * int(b)

class Division(Operation):
    def calculate(self, expression):
        a, b = expression.split('/')
        return int(a) / int(b)

class Calculator:
    def __init__(self):
        self.operations = {
            '+': Addition(),
            '-': Subtraction(),
            '*': Multiplication(),
            '/': Division()
        }

    def calculate(self, expression):
        operation_key = expression[0]
        operation = self.operations[operation_key]
        return operation.calculate(expression)

# 测试代码
calculator = Calculator()
expression = "3 + 5"
result = calculator.calculate(expression)
print(result)  # 输出：8
```
