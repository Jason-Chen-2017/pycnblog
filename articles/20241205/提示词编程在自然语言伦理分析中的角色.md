# 提示词生成类
class KeywordGenerator:
    def __init__(self, keyword_manager, preprocessor):
        self.keyword_manager = keyword_manager
        self.preprocessor = preprocessor

    def generate_keywords(self, text):
        clean_text = self.preprocessor.clean_text(text)
        tokens = self.preprocessor.tokenize_text(clean_text)
        tagged_tokens = self.preprocessor.tag_words(tokens)
        
        # 从标注后的词性中选择名词和动词作为提示词
        keywords = [token for token, tag in tagged_tokens if tag.startswith(('NN', 'VB'))]
        return keywords
