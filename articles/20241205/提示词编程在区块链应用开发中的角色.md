### 第1章：区块链与提示词编程概述

#### 1.1 区块链技术简介

区块链是一种去中心化的分布式账本技术，通过加密算法和时间戳技术确保数据的完整性和安全性。区块链的基本原理包括区块、链式结构和共识算法等。每一个区块都包含一定数量的交易记录，并通过密码学方法与前一个区块连接，形成一条不断延伸的链。

#### 1.2 提示词编程的概念

提示词编程（Prompt Programming）是一种通过提示词（Prompt）来引导模型生成文本或其他输出的编程方法。提示词可以是一个简单的提示，如“编写一个关于区块链的介绍”，也可以是一个复杂的提示，包含多个条件和上下文信息。

contract Voting {
    mapping(address => bool) public hasVoted;
    string[] public options;

    constructor(string[] memory _options) {
        options = _options;
    }

    function vote(string memory option) public {
        require(!hasVoted[msg.sender], "已经投票");
        require(containsOption(option), "无效的投票选项");

        hasVoted[msg.sender] = true;
        emit Voted(msg.sender, option);
    }

    function containsOption(string memory option) public view returns (bool) {
        for (uint256 i = 0; i < options.length; i++) {
            if (keccak256(abi.encodePacked(options[i])) == keccak256(abi.encodePacked(option))) {
                return true;
            }
        }
        return false;
    }

    event Voted(address voter, string option);
}
