import numpy as np
import tensorflow as tf
from tensorflow.keras.models import load_model

# 加载预训练的深度学习模型
model = load_model('model.h5')

# 生成对抗样本
def generate_adversarial_sample(image, model, epsilon=0.1, alpha=0.01, num_iterations=100):
    original_image = image.copy()
    image = tf.convert_to_tensor(image, dtype=tf.float32)
    image = tf.expand_dims(image, 0)

    # 计算模型的预测结果
    with tf.GradientTape() as tape:
        predictions = model(image)
        loss = tf.keras.losses.categorical_crossentropy(predictions, image)

    # 计算梯度
    gradients = tape.gradient(loss, image)

    # 计算对抗样本
    sign gradients = tf.sign(gradients)
    adversarial_image = image + alpha * epsilon * sign gradients
    adversarial_image = tf.clip_by_value(adversarial_image, 0, 1)

    return adversarial_image.numpy()

# 生成对抗样本并评估模型
image = np.random.rand(1, 28, 28)
adversarial_image = generate_adversarial_sample(image, model)
predicted_label = model.predict(adversarial_image)

print("Original label:", np.argmax(image))
print("Predicted label:", np.argmax(predicted_label))
