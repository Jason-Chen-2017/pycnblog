# 定义相似度计算函数
def similarity_score(path1, path2):
    # 计算余弦相似度
    dot_product = np.dot(path1, path2)
    norm_product = np.linalg.norm(path1) * np.linalg.norm(path2)
    return dot_product / norm_product

# 定义概率计算函数
def probability_score(similarity_scores):
    # 计算匹配概率
    probabilities = np.exp(similarity_scores)
    probabilities /= np.sum(probabilities)
    return probabilities

# 定义路径重组函数
def path_recombination(paths, probabilities):
    # 根据匹配概率，选择具有最大匹配概率的子路径进行重组
    selected_paths = []
    for path, probability in zip(paths, probabilities):
        if probability > 0.5:
            selected_paths.append(path)
    return selected_paths
