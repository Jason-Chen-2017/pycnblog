```python
import random

class Particle:
    def __init__(self, dimensions):
        self.position = [random.random() for _ in range(dimensions)]
        self.velocity = [0]*dimensions
        self.pbest = self.position.copy()
        self.best = self.position.copy()

def pso_optimization(dimensions, population_size, max_iterations):
    particles = [Particle(dimensions) for _ in range(population_size)]
    global_best = particles[0].best.copy()

    for _ in range(max_iterations):
        for particle in particles:
            # Update personal best
            if evaluate(particle.position) > evaluate(particle.pbest):
                particle.pbest = particle.position.copy()
            # Update global best
            if evaluate(particle.position) > evaluate(global_best):
                global_best = particle.position.copy()
            # Update velocity and position
            for i in range(dimensions):
                inertia = 0.8
                c1 = 1.5
                c2 = 1.5
                r1 = random.random()
                r2 = random.random()
                particle.velocity[i] = inertia * particle.velocity[i] + c1 * r1 * (particle.pbest[i] - particle.position[i]) + c2 * r2 * (global_best[i] - particle.position[i])
                particle.position[i] += particle.velocity[i]
    return global_best
```
