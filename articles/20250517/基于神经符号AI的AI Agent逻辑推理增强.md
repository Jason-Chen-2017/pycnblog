import tensorflow as tf
import numpy as np

class NeuralSymbolicModel:
    def __init__(self, input_size, hidden_size, output_size):
        self.input_size = input_size
        self.hidden_size = hidden_size
        self.output_size = output_size

        self.weights = {
            'w1': tf.Variable(tf.random.normal([input_size, hidden_size])),
            'w2': tf.Variable(tf.random.normal([hidden_size, output_size]))
        }

        self.biases = {
            'b1': tf.Variable(tf.zeros([hidden_size])),
            'b2': tf.Variable(tf.zeros([output_size]))
        }

    def call(self, inputs):
        layer1 = tf.nn.relu(tf.matmul(inputs, self.weights['w1']) + self.biases['b1'])
        output = tf.matmul(layer1, self.weights['w2']) + self.biases['b2']
        return output

model = NeuralSymbolicModel(input_size=10, hidden_size=20, output_size=5)
