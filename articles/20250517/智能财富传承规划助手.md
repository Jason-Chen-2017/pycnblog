  $$ \text{max} \sum_{i=1}^{n} w_i r_i - \lambda \sum_{i=1}^{n} w_i^2 $$

```python
import numpy as np
from sklearn.tree import DecisionTreeRegressor

def optimize_asset_allocation(users, assets):
    # 数据预处理
    X = []
    y = []
    for user in users:
        for asset in assets:
            X.append([user['age'], asset['value']])
            y.append(user['risk_tolerance'])
    # 模型训练
    model = DecisionTreeRegressor()
    model.fit(X, y)
    # 预测与优化
    predicted_risk = model.predict(X)
    return predicted_risk
```
