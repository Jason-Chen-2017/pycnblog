## 5.2 系统核心实现源代码

### 5.2.1 智能合约实现
以下是一个简单的供应链溯源智能合约示例（Solidity语言）：

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract SupplyChain {
    // 事件：记录产品上架
    event ProductListed(
        uint256 indexed productId,
        string name,
        uint256 quantity,
        uint256 price
    );

    // 事件：记录产品运输
    event ProductShipped(
        uint256 indexed productId,
        address indexed supplier,
        address indexed transporter
    );

    // 事件：记录产品交付
    event ProductDelivered(
        uint256 indexed productId,
        address indexed retailer,
        address indexed customer
    );

    // 结构体：产品信息
    struct Product {
        string name;
        uint256 quantity;
        uint256 price;
        address supplier;
        address transporter;
        address retailer;
    }

    // 映射：产品ID到产品信息
    mapping(uint256 => Product) products;

    // 函数：上架产品
    function listProduct(uint256 productId, string memory name, uint256 quantity, uint256 price) public {
        Product memory product = Product({
            name: name,
            quantity: quantity,
            price: price,
            supplier: msg.sender,
            transporter: address(0),
            retailer: address(0)
        });
        products[productId] = product;
        emit ProductListed(productId, name, quantity, price);
    }

    // 函数：标记产品运输
    function shipProduct(uint256 productId, address transporter) public {
        Product storage product = products[productId];
        require(product.supplier == msg.sender, "Only supplier can ship product");
        product.transporter = transporter;
        emit ProductShipped(productId, product.supplier, transporter);
    }

    // 函数：标记产品交付
    function deliverProduct(uint256 productId, address retailer) public {
        Product storage product = products[productId];
        require(product.transporter == msg.sender, "Only transporter can deliver product");
        product.retailer = retailer;
        emit ProductDelivered(productId, retailer, msg.sender);
    }
}
```

### 5.2.2 后端实现

以下是一个简单的后端接口实现（基于Node.js和Express）：

```javascript
const express = require('express');
const bodyParser = require('body-parser');
const Web3 = require('web3');
const {ganacheProvider, abi, contractAddress} = require('./config');

// 初始化Express应用
const app = express();
app.use(bodyParser.json());

// 连接ganache测试网络
const web3 = new Web3(ganacheProvider);

// 初始化智能合约实例
const contract = new web3.eth.Contract(abi.abi, contractAddress);

// 定义API接口
app.get('/products', async (req, res) => {
    const products = await contract.methods.getAllProducts().call();
    res.json(products);
});

app.post('/listProduct', async (req, res) => {
    const {productId, name, quantity, price} = req.body;
    try {
        const result = await contract.methods.listProduct(productId, name, quantity, price).send();
        res.json({status: 'success', transactionHash: result.transactionHash});
    } catch (error) {
        res.status(500).json({status: 'error', message: error.message});
    }
});

app.put('/shipProduct', async (req, res) => {
    const {productId, transporter} = req.body;
    try {
        const result = await contract.methods.shipProduct(productId, transporter).send();
        res.json({status: 'success', transactionHash: result.transactionHash});
    } catch (error) {
        res.status(500).json({status: 'error', message: error.message});
    }
});

app.put('/deliverProduct', async (req, res) => {
    const {productId, retailer} = req.body;
    try {
        const result = await contract.methods.deliverProduct(productId, retailer).send();
        res.json({status: 'success', transactionHash: result.transactionHash});
    } catch (error) {
        res.status(500).json({status: 'error', message: error.message});
    }
});

// 监听端口
app.listen(3000, () => {
    console.log('Server is running on http://localhost:3000');
});
```

### 5.2.3 前端实现（简化）

```javascript
import React, { useState, useEffect } from 'react';
import axios from 'axios';

function SupplyChainTracker() {
    const [products, setProducts] = useState([]);

        fetch('/products')
            .then(res => res.json())

        try {
            const response = await axios.post('/listProduct', {
                productId: productId,
                name: name,
                quantity: quantity,
                price: price
            });
            console.log('Product listed:', response.data);
        } catch (error) {
            console.error('Error listing product:', error);
        }
    };

    const shipProduct = async (productId, transporter) => {
        try {
            const response = await axios.put('/shipProduct', {
                productId: productId,
                transporter: transporter
            });
            console.log('Product shipped:', response.data);
        } catch (error) {
            console.error('Error shipping product:', error);
        }
    };

    const deliverProduct = async (productId, retailer) => {
        try {
            const response = await axios.put('/deliverProduct', {
                productId: productId,
                retailer: retailer
            });
            console.log('Product delivered:', response.data);
        } catch (error) {
            console.error('Error delivering product:', error);
        }
    };

    return (
        <div>
            <h1>供应链溯源平台</h1>
            <button onClick={() => listProduct(1, '苹果', 100, 10)}>上架产品</button>
            <button onClick={() => shipProduct(1, '物流公司')}>标记运输</button>
            <button onClick={() => deliverProduct(1, '零售商')}>标记交付</button>
            <h2>产品信息:</h2>
            <ul>
                {products.map((product, index) => (
                    <li key={index}>
                        ID: {product.id}, 名称: {product.name}, 数量: {product.quantity}, 价格: {product.price}
                    </li>
                ))}
            </ul>
        </div>
    );
}

export default SupplyChainTracker;
```

## 5.4 实际案例分析

假设我们有一个苹果供应链项目：

1. **供应商**：种植苹果的农户，向加工厂供应原材料。
2. **生产商**：负责加工苹果，生产苹果汁。
3. **物流商**：负责运输苹果汁到零售商。
4. **零售商**：将苹果汁卖给消费者。

通过平台，我们可以实现以下操作：
- 供应商上架苹果原料，记录数量、价格等信息。
- 生产商加工后，标记产品为半成品。
- 物流商运输半成品到零售商，记录运输信息。
- 零售商接收产品，标记为已交付。

消费者可以通过平台查询苹果汁的溯源信息，包括种植、加工、运输等各环节的信息。
