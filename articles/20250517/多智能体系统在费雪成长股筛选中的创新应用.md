import pandas as pd
import numpy as np

def fisher_growth_stock_screening(data):
    # 数据预处理
    data = data.dropna()
    data['增长率'] = (data['营收'] / data['营收'].shift(1)) - 1
    # 多智能体协作筛选
    selected = data[data['增长率'] > 0.1]
    return selected

# 示例数据
data = pd.DataFrame({
    '营收': [100, 120, 150, 180],
    '净利润': [20, 25, 30, 35]
})
result = fisher_growth_stock_screening(data)
print(result)
