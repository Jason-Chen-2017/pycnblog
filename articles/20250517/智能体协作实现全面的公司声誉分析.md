def compute_reputation(intelligence_agents):
    reputation_scores = []
    for agent in intelligence_agents:
        score = 0.0
        for factor in agent.factors:
            score += agent.weights[factor] * agent.factor_scores[factor]
        reputation_scores.append(score)
    return reputation_scores

# 示例用法
class Agent:
    def __init__(self, id, factors, weights, factor_scores):
