#### 3.2.2 Python实现代码

```python
import numpy as np

def expected_return(weights, returns):
    return np.dot(weights.T, returns)

def portfolio_risk(weights, covariance_matrix):
    return np.sqrt(np.dot(weights.T @ covariance_matrix @ weights))

# 示例数据
returns = np.array([0.1, 0.2, 0.15])
covariance_matrix = np.array([[0.02, 0.01, 0.015],
                              [0.01, 0.03, 0.02],
                              [0.015, 0.02, 0.035]])
weights = np.array([0.3, 0.5, 0.2])

# 计算预期回报和风险
er = expected_return(weights, returns)
risk = portfolio_risk(weights, covariance_matrix)

print(f"预期回报: {er}")
print(f"投资组合风险: {risk}")
```

import numpy as np
import pandas as pd

# 示例数据加载
data = pd.read_csv('asset_returns.csv')
returns = data.iloc[:, 1:].values

# 计算协方差矩阵
covariance_matrix = np.cov(returns.T)

# 定义优化目标函数
def optimize_portfolio(weights, returns, covariance_matrix):
    return -expected_return(weights, returns)

# 使用 scipy.optimize 进行优化
from scipy.optimize import minimize

weights = np.array([1/3, 1/3, 1/3])
result = minimize(optimize_portfolio, weights, args=(returns, covariance_matrix))
optimized_weights = result.x

print(f"优化后的投资组合权重: {optimized_weights}")
