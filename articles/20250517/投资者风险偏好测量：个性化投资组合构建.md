#### 4.2.2 Python代码实现
```python
import numpy as np
from scipy.optimize import minimize

def portfolio_optimization(returns, covariance, risk_preference):
    n = len(returns)
    # 定义目标函数
    def objective(weights):
        return np.dot(weights.T, covariance @ weights)
    
    # 约束条件
    constraints = [{'type': 'eq', 'fun': lambda w: np.sum(w) - 1.0}]
    
    # 初始权重
    initial_weights = np.array([1.0/n]*n)
    
    # 最优化问题
    result = minimize(objective, initial_weights, constraints=constraints)
    
    # 计算风险调整后的收益
    adjusted_return = np.dot(result.x, returns) - risk_preference * objective(result.x)
    
    return result.x, adjusted_return

# 示例数据
returns = np.array([0.1, 0.2, 0.15])
covariance = np.array([[0.04, 0.01, 0.005],
                       [0.01, 0.09, 0.002],
                       [0.005, 0.002, 0.05]])
risk_preference = 0.5

weights, adjusted_return = portfolio_optimization(returns, covariance, risk_preference)
print("最优权重:", weights)
print("风险调整后的收益:", adjusted_return)
```
