import pandas as pd
from sklearn.linear_model import LinearRegression

# 数据预处理
data = pd.read_csv('pension_data.csv')
X = data[['age', 'salary']]
y = data['pension_gap']

# 模型训练
model = LinearRegression()
model.fit(X, y)

# 模型预测
new_data = pd.DataFrame({'age': [40], 'salary': [50000]})
prediction = model.predict(new_data)
print(prediction)
