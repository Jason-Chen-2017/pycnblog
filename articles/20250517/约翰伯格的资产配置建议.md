- **3.3 算法实现代码**
  ```python
  import numpy as np

  def portfolio_optimization(returns, weights):
      covariance_matrix = returns.cov()
      risk = np.sqrt(weights.T @ covariance_matrix @ weights)
      return risk
  ```

  import pandas as pd
  import numpy as np
  import matplotlib.pyplot as plt

  # 示例数据
  returns = pd.DataFrame({
      '股票': [0.05, 0.03, 0.04, 0.02, 0.06],
      '债券': [0.02, 0.03, 0.02, 0.04, 0.01],
      '现金': [0.01, 0.01, 0.01, 0.01, 0.01]
  })

  def calculate_portfolio_risk(weights):
      covariance_matrix = returns.cov()
      return np.sqrt(weights.T @ covariance_matrix @ weights)

  # 示例权重
  weights = [0.4, 0.3, 0.3]
  risk = calculate_portfolio_risk(weights)
  print(f"风险：{risk:.2f}")
