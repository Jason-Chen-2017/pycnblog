### 5.2 系统核心实现

```python
import numpy as np
from sklearn.tree import DecisionTreeRegressor

def train_model(X, y):
    model = DecisionTreeRegressor()
    model.fit(X, y)
    return model

def main():
    # 示例数据
    X = np.array([[1], [2], [3], [4]])
    y = np.array([2, 4, 6, 8])
    model = train_model(X, y)
    print(model.predict([[5]]))  # 输出：array([10])

if __name__ == "__main__":
    main()
```
