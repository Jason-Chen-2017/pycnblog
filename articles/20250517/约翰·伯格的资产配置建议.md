# 资产配置优化的Python代码示例
import numpy as np
import pandas as pd

# 示例数据
returns = np.array([[0.1, 0.2, 0.15],
                    [0.05, 0.1, 0.08],
                    [0.2, 0.15, 0.25]])

# 计算方差和协方差矩阵
cov_matrix = np.cov(returns)

# 简单的优化模型实现
def optimize_portfolio(weights, returns, cov_matrix):
    # 计算投资组合的收益和风险
    portfolio_return = np.dot(weights, returns.mean())
    portfolio_risk = np.sqrt(np.dot(weights.T, np.dot(cov_matrix, weights)))
    return -portfolio_return, -portfolio_risk

# 示例权重分配
weights = [0.3, 0.3, 0.4]
print(f"最优收益：{-optimize_portfolio(weights, returns, cov_matrix)[0]}")
print(f"最优风险：{-optimize_portfolio(weights, returns, cov_matrix)[1]}")
