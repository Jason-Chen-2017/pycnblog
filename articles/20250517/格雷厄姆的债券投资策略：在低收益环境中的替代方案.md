```python
import numpy as np
import pandas as pd

def calculate_duration(bond_price, coupon_rate, face_value, yield_to_maturity, years):
    cash_flows = [coupon_rate * face_value] * years + [face_value]
    discounted_cash_flows = [cf / (1 + yield_to_maturity) ** (i+1) for i, cf in enumerate(cash_flows)]
    duration = sum(cash_flows[i] * i for i, cf in enumerate(cash_flows)) / sum(discounted_cash_flows)
    return duration

# 示例数据
bond_price = 100
coupon_rate = 0.05
face_value = 100
yield_to_maturity = 0.03
years = 10

duration = calculate_duration(bond_price, coupon_rate, face_value, yield_to_maturity, years)
print(f"债券的久期为：{duration}")
```
