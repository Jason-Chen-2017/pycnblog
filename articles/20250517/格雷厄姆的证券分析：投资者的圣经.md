```python
import pandas as pd
import numpy as np

def calculate_intrinsic_value(earnings, growth, discount_rate):
    return earnings * (1 + growth) / discount_rate

# 示例数据
data = {
    '公司': ['苹果', '微软', '谷歌'],
    '收益': [100000, 150000, 120000],
    '增长率': [0.05, 0.03, 0.04],
    '折现率': [0.1, 0.1, 0.1]
}

df = pd.DataFrame(data)
df['内在价值'] = df.apply(lambda x: calculate_intrinsic_value(x['收益'], x['增长率'], x['折现率']), axis=1)
print(df)
```
