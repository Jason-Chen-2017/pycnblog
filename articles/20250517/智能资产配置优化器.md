    ```python
    def mutate(individual):
        # 突变操作
        idx = np.random.randint(len(individual))
        individual[idx] = 1 - individual[idx]
        return individual

    def crossover(parent1, parent2):
        # 单点交叉
        idx = np.random.randint(len(parent1))
        return parent1[:idx] + parent2[idx:], parent2[:idx] + parent1[idx:]

    class GAOptimizer:
        def __init__(self, assets, pop_size=100, generations=100):
            self.assets = assets
            self.pop_size = pop_size
            self.generations = generations
            self.population = np.random.random((pop_size, len(assets)))

        def optimize(self):
            for _ in range(self.generations):
                fitness_scores = np.array([fitness(ind) for ind in self.population])
                selected = self.population[fitness_scores.argsort()[-self.pop_size//2:]]
                # 交叉
                new_population = []
                for i in range(0, selected.shape[0], 2):
                    p1, p2 = crossover(selected[i], selected[i+1])
                    new_population.append(mutate(p1))
                    new_population.append(mutate(p2))
                self.population = np.array(new_population)
            best = self.population[0]
            return best
    ```
