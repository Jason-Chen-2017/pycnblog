from neo4j import GraphDatabase

# 连接Neo4j数据库
driver = GraphDatabase.driver('neo4j://localhost:7687', auth=('user', 'password'))

# 创建节点和关系
def create_node(tx, label, properties):
    tx.run("CREATE (:{} {{}})".format(label), properties)

def create_relation(tx, start_label, start_id, relation, end_label, end_id):
    tx.run("MATCH (a:{} {{id: {}}}), (b:{} {{id: {}}}) "
            "CREATE (a)-[r:{} {{}}]->(b)".format(start_label, start_id, end_label, end_id, relation), properties)

with driver.session() as session:
    session.write_transaction(create_node, 'City', {'name': '北京', 'country': '中国'})
    session.write_transaction(create_relation, 'City', '北京', 'capital_of', 'Country', '中国')
```
