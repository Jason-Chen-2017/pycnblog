# 定义优化目标函数
def portfolio_return(weights, returns):
    return np.dot(weights.T, returns)

def portfolio_risk(weights, returns):
    return np.sqrt(np.dot(weights.T, np.dot(returns.cov(), weights)))

# 定义约束条件
def optimization_constraints(weights):
    return (weights - 1) # 约束条件：权重之和为1

# 最优化问题
def optimize_portfolio(returns, method='max_return'):
    n = len(returns.columns)
    init_weights = np.array([1/n] * n)
    
    if method == 'max_return':
        result = minimize(lambda w: -portfolio_return(w, returns), init_weights, 
                          constraints={'type': 'eq', 'fun': lambda w: np.sum(w) - 1})
    elif method == 'min_risk':
        result = minimize(portfolio_risk, init_weights, 
                          constraints={'type': 'eq', 'fun': lambda w: np.sum(w) - 1})
