```python
from django.shortcuts import render, redirect
from django.contrib.auth.decorators import login_required
from .models import Resource, Transaction

@login_required
def create_order(request, resource_id):
    user = request.user
    resource = Resource.objects.get(id=resource_id)
    # 创建交易记录
    transaction = Transaction(user=user, resource=resource)
    transaction.save()
    return redirect('order_confirmation', transaction.id)

def order_confirmation(request, transaction_id):
    transaction = Transaction.objects.get(id=transaction_id)
    return render(request, 'order_confirmation.html', {'transaction': transaction})
```
