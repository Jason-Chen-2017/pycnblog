```python
import pandas as pd
import numpy as np

def portfolio_optimization(stocks, returns, covariance):
    # 计算最优权重
    n = len(stocks)
    # 使用均值-方差优化
    mean_return = returns.mean()
    cov_matrix = covariance
    # 计算最优权重
    inv_cov = np.linalg.inv(cov_matrix)
    weights = inv_cov @ mean_return
    weights = weights / weights.sum()
    return weights
```

```python
import numpy as np
import pandas as pd
from scipy.optimize import minimize

def portfolio_variance(weights, covariance_matrix):
    return (weights.T @ covariance_matrix @ weights).item()

def optimize_portfolio(covariance_matrix, returns, method='min_var'):
    n = covariance_matrix.shape[0]
    if method == 'min_var':
        # 最小方差优化
        def objective(weights):
            return portfolio_variance(weights, covariance_matrix)
        constraints = [{'type': 'eq', 'fun': lambda w: np.sum(w) - 1}]
    elif method == 'max_ret':
        # 最大收益优化
        def objective(weights):
            return -returns.T @ weights
        constraints = [{'type': 'eq', 'fun': lambda w: np.sum(w) - 1},
                        {'type': 'eq', 'fun': lambda w: (weights.T @ covariance_matrix @ weights).item() - 1}]
    else:
        raise ValueError("Invalid optimization method")
    bounds = [(0, 1) for _ in range(n)]
    result = minimize(objective, np.ones(n)/n, method='SLSQP', bounds=bounds, constraints=constraints)
    return result.x

# 示例数据
returns = np.array([0.1, 0.2, 0.15])
covariance = np.array([[0.02, 0.01, 0.01],
                       [0.01, 0.03, 0.02],
                       [0.01, 0.02, 0.03]])

weights = optimize_portfolio(covariance, returns, method='min_var')
print("最小方差权重:", weights)
```
