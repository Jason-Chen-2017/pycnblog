def get_stock_dividend_yield(ticker):

import pandas as pd
import numpy as np

def optimize_portfolio(tickers, weights):
    # 计算投资组合的预期收益率
    expected_returns = np.array(weights) * 100
    # 计算投资组合的风险
    risk = np.sqrt(np.dot(weights, weights.T) * 100)
    return expected_returns, risk

# 示例代码
tickers = ['AAPL', 'MSFT', 'GOOGL', 'AMZN', 'FB']
weights = [0.2, 0.3, 0.2, 0.2, 0.1]
expected_returns, risk = optimize_portfolio(tickers, weights)
print(f"预期收益率：{expected_returns}%，风险：{risk}%")
