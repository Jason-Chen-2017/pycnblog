### 7.2 核心实现源代码
```python
import pandas as pd
import matplotlib.pyplot as plt

# 数据加载
data = pd.read_csv('global_market_valuations.csv')

# 宏观经济指标分析
def calculate_valuation_metrics(data):
    metrics = {}
    metrics['平均市盈率'] = data['PE'].mean()
    metrics['平均市净率'] = data['PB'].mean()
    metrics['平均股息率'] = data['Div Yield'].mean()
    return metrics

# 可视化函数
def visualize_valuations(data, country):
    plt.figure(figsize=(10, 6))
    plt.plot(data['日期'], data[country], label=f'{country}股市估值')
    plt.title(f'{country}股市估值趋势')
    plt.xlabel('日期')
    plt.ylabel('估值')
    plt.legend()
    plt.show()

# 主函数
def main():
    metrics = calculate_valuation_metrics(data)
    print("估值指标分析结果：")
    print(f"平均市盈率：{metrics['平均市盈率']}")
    print(f"平均市净率：{metrics['平均市净率']}")
    print(f"平均股息率：{metrics['平均股息率']}")
    
    # 可视化
    country = '中国'
    visualize_valuations(data, country)

if __name__ == "__main__":
    main()
```
