  ```python
  import numpy as np
  from sklearn.preprocessing import MinMaxScaler

  def calculate_entropy(X):
      # 计算每列的熵
      n = X.shape[1]
      entropy = 0
      for i in range(n):
          # 计算每列的概率分布
          p = np.mean(X[:, i] == X[:, i])
          if p == 0:
              entropy += 0
          else:
              entropy += -p * np.log2(p)
      return entropy
  ```

  ```python
  import numpy as np
  import matplotlib.pyplot as plt

  def quantum_chaos_simulation(n_qubits=2):
      # 初始化量子态
      q = np.zeros(2**n_qubits, dtype=np.complex_)
      q[0] = 1.0

      # 定义量子门
      def hadamard():
          return np.array([[1, 1], [1, -1]]) / np.sqrt(2)

      # 应用哈达玛门
      H = hadamard()
      q = np.dot(q, H)

      # 绘制量子态的幅度
      plt.figure(figsize=(8, 6))
      plt.bar(range(2**n_qubits), np.abs(q))
      plt.title("Quantum State Amplitudes")
      plt.show()

  quantum_chaos_simulation(2)
  ```
