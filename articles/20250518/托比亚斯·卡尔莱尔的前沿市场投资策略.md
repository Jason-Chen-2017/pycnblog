```python
import pandas as pd
import numpy as np
from sklearn.linear_model import LinearRegression

def preprocess(market_data):
    # 数据预处理
    processed_data = market_data.dropna()
    return processed_data

def predict_trend(processed_data):
    # 市场趋势预测
    model = LinearRegression()
    X = processed_data.index.values.reshape(-1, 1)
    y = processed_data['close'].values
    model.fit(X, y)
    predicted_prices = model.predict(X)
    return predicted_prices

def optimize_portfolio(portfolio, predicted_prices):
    # 投资组合优化
    # 假设 portfolio 是一个字典，包含资产及其权重
    # predicted_prices 是预测的价格数组
    # 这里简化为等权重分配
    optimized_portfolio = {}
    for asset in portfolio:
        optimized_portfolio[asset] = predicted_prices / np.sum(predicted_prices)
    return optimized_portfolio

def manage_risk(optimized_portfolio, risk_parameters):
    # 风险管理
    # 这里假设风险参数包括 VaR 和最大回撤
    # 优化后的投资组合的风险调整
    return optimized_portfolio * risk_parameters['risk_adjustment']

def calculate_investment_strategy(market_data, portfolio, risk_parameters):
    processed_data = preprocess(market_data)
    predicted_prices = predict_trend(processed_data)
    optimized_portfolio = optimize_portfolio(portfolio, predicted_prices)
    managed_strategy = manage_risk(optimized_portfolio, risk_parameters)
    return managed_strategy
```
