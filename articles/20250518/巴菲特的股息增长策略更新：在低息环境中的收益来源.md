#### 5.2 核心实现代码
```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# 示例数据：假设股息增长率和股息率
def calculate_stock_value(dividend_growth_rate, required_return):
    initial_dividend = 1  # 假设初始股息为1元
    future_dividend = initial_dividend * (1 + dividend_growth_rate)
    stock_value = future_dividend / (required_return - dividend_growth_rate)
    return stock_value

# 输入参数
required_return = 0.08  # 8%
growth_rates = [0.02, 0.03, 0.04]  # 2%, 3%, 4%

# 计算股票价值
results = []
for gr in growth_rates:
    value = calculate_stock_value(gr, required_return)
    results.append((gr, value))

# 可视化
gr_values = [result[0] for result in results]
value_values = [result[1] for result in results]

plt.plot(gr_values, value_values)
plt.xlabel('股息增长率 (%)')
plt.ylabel('股票价值')
plt.title('股息增长率与股票价值关系')
plt.show()
```
