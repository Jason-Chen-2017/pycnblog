#### 4.1.2 核心算法实现

import pandas as pd
import numpy as np

def calculate_value(company_financials):
    E = company_financials['Earnings']
    k = company_financials['Required_Return']
    V = E / k
    return V

# 示例数据
data = {
    'Earnings': [100, 200, 300],
    'Required_Return': [0.1, 0.1, 0.1]
}

df = pd.DataFrame(data)
value = calculate_value(df)
print(value)
