### 附录: 巴菲特市场情绪指标的Python实现代码

```python
import pandas as pd
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.svm import SVC
from sklearn.metrics import accuracy_score
from nltk.sentiment import SentimentIntensityAnalyzer

# 示例数据加载
data = pd.DataFrame({
    'text': [' Buffett is very optimistic about the market future ',
             'The market is showing signs of weakness'],
    'label': [1, 0]
})

# 文本预处理
def preprocess(text):
    return text.lower().strip()

data['text_processed'] = data['text'].apply(preprocess)

# 特征提取
tfidf = TfidfVectorizer(max_features=500)
X = tfidf.fit_transform(data['text_processed'])

# 情绪分类模型训练
model = SVC()
model.fit(X, data['label'])

# 情绪指标计算
sia = SentimentIntensityAnalyzer()
compound_scores = data['text_processed'].apply(lambda x: sia.polarity_scores(x)['compound'])
data['compound_score'] = compound_scores

# 指标与标签对比
accuracy = accuracy_score(data['label'], model.predict(X))
print(f"Accuracy: {accuracy}")
```
