- **影响力传播代码实现**：
  ```python
  import networkx as nx

  def influence_spread(graph, source):
      nx.set_node_attributes(graph, 'influence', {node: 0 for node in graph.nodes()})
      graph.nodes[source]['influence'] = 1
      while True:
          new_influence = {}
          for node in graph.nodes():
              new_influence[node] = graph.nodes[node]['influence']
              for neighbor in graph.neighbors(node):
                  weight = graph.get_edge_data(node, neighbor).get('weight', 1)
                  new_influence[neighbor] += new_influence[node] * weight
          if new_influence == graph.nodes['influence']:
              break
          graph.nodes[node]['influence'] = new_influence[node]
      return graph
  ```
