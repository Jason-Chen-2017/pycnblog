## 3.3 Python实现
```python
import pandas as pd
import numpy as np
from sklearn.linear_model import LinearRegression
from sklearn.metrics import r2_score

# 数据预处理
data = pd.read_csv('data.csv')
X = data[['生物降解塑料技术进步指数']]
y = data['股市估值']

# 模型训练
model = LinearRegression()
model.fit(X, y)

# 结果分析
predictions = model.predict(X)
r_squared = r2_score(y, predictions)
print(f'R平方值: {r_squared}')
print(f'回归系数: {model.coef_}')
print(f'截距: {model.intercept_}')
```

```python
import pandas as pd
import numpy as np
from sklearn.linear_model import LinearRegression
from sklearn.metrics import r2_score

# 数据加载
data = pd.read_csv('data.csv')

# 特征和目标变量分离
X = data[['生物降解塑料技术进步指数']]
y = data['股市估值']

# 模型训练
model = LinearRegression()
model.fit(X, y)

# 预测与评估
predictions = model.predict(X)
r_squared = r2_score(y, predictions)
print(f"模型的R平方值为: {r_squared}")
print(f"回归系数为: {model.coef_}")
print(f"截距为: {model.intercept_}")
```
