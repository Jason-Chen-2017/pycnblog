```python
# 示例代码：用户粘性与盈利能力的分析
def calculate_user_sticky(users):
    return users['留存率'].mean() / users['获取成本'].mean()

def calculate_profitability(revenue, cost):
    return revenue - cost

# 示例数据
users = {
    '留存率': [0.8, 0.75, 0.9],
    '获取成本': [50, 60, 40]
}
revenue = 1000
cost = 200

sticky_score = calculate_user_sticky(users)
profitability = calculate_profitability(revenue, cost)
print(f"用户粘性得分: {sticky_score}")
print(f"盈利能力: {profitability}")
```
