import requests
import pandas as pd

def get_crypto_data(api_key):
    url = "https://api.coinmarketcap.com/v2/ticker/?limit=10"
    headers = {
        "X-CMC-pro PublicKey": api_key
    }
    response = requests.get(url, headers=headers)
    data = response.json()
    df = pd.DataFrame(data)
    return df

# 示例用法

crypto_data = get_crypto_data(api_key)

### 6.2.2 风险评估模型的实现

#### 6.2.2.1 风险评估模型

```python
def calculate_volatility(prices):
    returns = prices.pct_change()
    volatility = returns.std() * 100
    return volatility

# 示例用法
prices = crypto_data['price']
volatility = calculate_volatility(prices)
print(f'价格波动率: {volatility:.2f}%')
```

#### 6.2.2.2 基于波动率的风险评分

```python
def risk_score(volatility, market_cap):
    if volatility < 10 and market_cap > 1000000000:
        return '低风险'
    elif 10 <= volatility < 20 and market_cap > 500000000:
        return '中等风险'
    else:
        return '高风险'

# 示例用法
market_cap = crypto_data['market_cap']
risk = risk_score(volatility, market_cap)
print(f'风险等级: {risk}')
```

### 6.2.3 投资策略的实现

#### 6.2.3.1 基于价值投资的策略

```python
def value_investment_strategy(data, threshold=0.2):
    # 计算市盈率或其他估值指标
    # 这里简化为基于价格波动的判断
    buy_signals = data[data['price'].pct_change().mean() > threshold]
    return buy_signals

# 示例用法
buy_signals = value_investment_strategy(crypto_data)
print('买入信号币种:', buy_signals['name'].tolist())
```

#### 6.2.3.2 基于技术分析的策略

```python
def technical_analysis_strategy(data, window=20):
    # 简单的均线策略
    data['SMA'] = data['price'].rolling(window).mean()
    buy_signals = data[data['price'] < data['SMA']]
    return buy_signals

# 示例用法
buy_signals = technical_analysis_strategy(crypto_data)
print('技术分析买入信号币种:', buy_signals['name'].tolist())
```
