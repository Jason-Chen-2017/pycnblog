### 6.2 核心代码实现

#### 6.2.1 意图识别模型训练
```python
from spacy.util import minibatch
import spacy

# 加载Spacy模型
nlp = spacy.load("en_core_web_sm")

# 创建训练数据
train_data = [
    ("I want to book a flight to Paris", "book_flight"),
    ("Can you help me with my order?", "help_order"),
    ("I need to change my ticket", "change_ticket"),
    ("Where is my package?", "track_package")
]

# 定义训练函数
def train_model(nlp, train_data):
    # 初始化训练者
    optimizer = nlp.optimize_with_sgd(nlp.Vocab.getOrCreate)
    # 创建批次
    batches = minibatch(train_data, batch_size=2)
    # 开始训练
    for batch, labels in batches:
        texts, labels = batch
        docs = [nlp(text) for text in texts]
        for doc, label in zip(docs, labels):
            for token in doc:
                token.get("lemma_")
            doc.cats = {label: 1.0}
        nlp.update([doc for doc, _ in zip(docs, labels)], sgd=optimizer, losses={})
    return nlp

# 开始训练
nlp = train_model(nlp, train_data)
```

#### 6.2.2 实体识别与信息抽取
```python
def extract_entities(doc):
    entities = []
    for ent in doc.ents:
        entities.append({
            "text": ent.text,
            "start": ent.start,
            "end": ent.end,
            "label": ent.label_
        })
    return entities

# 使用模型进行实体识别
doc = nlp("My flight is from New York to London on 15th October.")
print(extract_entities(doc))
```

#### 6.3.2 对话系统实现

```python
from transformers import pipeline

# 初始化意图分类器
classifier = pipeline("text-classification", model="snunlp/kobert-base-cls-102582")

# 定义对话处理函数
def process_query(query):
    # 分类意图
    result = classifier(query)
    intent = result[0]['label']
    # 根据意图生成响应
    responses = {
        "book_flight": "Sure, let's find you a flight to Paris.",
        "help_order": "I'll help you with your order. Please provide the order number.",
        "change_ticket": "To change your ticket, please contact our support team.",
        "track_package": "Your package is on its way. Expected delivery date is [date]."
    }
    return responses.get(intent, "I'm sorry, I don't understand your request.")
```
