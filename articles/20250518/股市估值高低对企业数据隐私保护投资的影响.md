def calculate_risk_score(factors):
    weights = [0.4, 0.3, 0.2, 0.1]
    risk_factors = [factors[i] * weights[i] for i in range(len(weights))]
    return sum(risk_factors)

# 示例数据
factors = [0.8, 0.6, 0.7, 0.5]
print("风险评分:", calculate_risk_score(factors))
