def calculate_stock_value(cash_flows, discount_rate):
    present_value = sum(flow / (1 + discount_rate) ** t for t, flow in enumerate(cash_flows))
    return present_value

# 示例数据
cash_flows = [100, 150, 200]
discount_rate = 0.05

# 计算股票价值
stock_value = calculate_stock_value(cash_flows, discount_rate)
print(f"股票的现值为：{stock_value}")
```

def calculate_stock_value(cash_flows, discount_rate):
    return sum(flow / (1 + discount_rate) ** t for t, flow in enumerate(cash_flows))

# 示例公司数据
company_data = {
    'name': 'XYZ公司',
    'cash_flows': [100, 150, 200],
    'discount_rate': 0.05
}

# 计算股票价值
stock_value = calculate_stock_value(company_data['cash_flows'], company_data['discount_rate'])
print(f"{company_data['name']}的股票现值为：{stock_value}")

# 可视化现金流与股票价值
plt.figure(figsize=(10, 6))
plt.plot([2020, 2021, 2022], company_data['cash_flows'], label='现金流')
plt.title('公司现金流与股票价值')
plt.xlabel('年份')
plt.ylabel('金额')
plt.legend()
plt.show()
```
