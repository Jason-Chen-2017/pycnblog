```python
import numpy as np
import matplotlib.pyplot as plt

# 数据生成
np.random.seed(42)
population = np.random.normal(50, 10, 1000)
gdp = np.random.normal(100, 20, 1000)
pe_ratio = np.random.normal(15, 5, 1000)

# 数据可视化
plt.figure(figsize=(10, 6))
plt.scatter(population, pe_ratio, c=gdp, cmap='viridis', s=10)
plt.colorbar(label='GDP')
plt.xlabel('Population')
plt.ylabel('PE Ratio')
plt.title('城市化与股市估值关系')
plt.show()
```
