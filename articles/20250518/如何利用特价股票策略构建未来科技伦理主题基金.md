```python
import pandas as pd
from scipy.optimize import minimize

def portfolio_optimization(weights, returns):
    # 目标函数：最大化收益
    def objective(x):
        return -sum(x * returns)
    # 约束条件
    constraints = [{'type': 'eq', 'fun': lambda x: sum(x) - 1}]
    # 边界条件
    bounds = [(0, 1) for _ in range(len(weights))]
    # 优化
    result = minimize(objective, weights, method='SLSQP', bounds=bounds, constraints=constraints)
    return result.x

# 示例数据
returns = pd.Series([0.1, 0.2, 0.15])
weights = [1/3]*3
optimal_weights = portfolio_optimization(weights, returns)
print(optimal_weights)
```
