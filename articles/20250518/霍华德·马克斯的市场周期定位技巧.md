#### 6.1 Python代码实现
```python
import pandas as pd
import numpy as np
from statsmodels.tsa.arima_model import ARIMA

# 示例数据加载
data = pd.read_csv('market_data.csv')

# 模型训练
model = ARIMA(data, order=(5, 1, 0))
model_fit = model.fit()

# 周期检测
acf = model_fit.arima_resid.autocorr(lags=20)
print(acf)
```

```python
# 市场周期定位代码
def detect_cycle(data):
    # 数据预处理
    diff_data = data.diff().dropna()
    # 模型训练
    model = ARIMA(diff_data, order=(5, 1, 0))
    model_fit = model.fit()
    # 周期检测
    acf = model_fit.arima_resid.autocorr(lags=20)
    return acf

# 示例运行
data = pd.read_csv('market_data.csv')
result = detect_cycle(data['index'])
print(result)
```
