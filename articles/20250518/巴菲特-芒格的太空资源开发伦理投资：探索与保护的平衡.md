def calculate_investment_value(cf, r, g):
    return cf / (r - g)

def calculate_ethical_score(value, risk_coefficient):
    return value * (1 - risk_coefficient)

# 示例数据
cf = 100000000
r = 0.08
g = 0.05
risk_coefficient = 0.2

value = calculate_investment_value(cf, r, g)
ethical_value = calculate_ethical_score(value, risk_coefficient)
print(f"投资价值：{value}")
print(f"伦理调整后价值：{ethical_value}")
