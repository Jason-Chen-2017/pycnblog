#### 3.3 量子优越性验证的Python实现  
以下是一个简单的量子线路模拟代码示例：

```python
import random
import numpy as np

def generate_random_circuit(n_qubits, depth):
    circuit = []
    for _ in range(depth):
        qubit = random.randint(0, n_qubits-1)
        circuit.append(qubit)
    return circuit

def simulate_quantum_state(circuit):
    n = len(circuit)
    state = np.zeros(2**n, dtype=np.complex128)
    state[0] = 1
    for qubit in circuit:
        state = apply_quantum_gate(state, qubit)
    return state

def apply_quantum_gate(state, qubit):
    n = len(state)
    gate = np.zeros((n, n), dtype=np.complex128)
    gate_index = 1 << qubit
    gate[gate_index] = [1, 0]
    gate[gate_index + 1] = [0, 1]
    return np.dot(gate, state)

# 示例
circuit = generate_random_circuit(2, 3)
state = simulate_quantum_state(circuit)
print(state)
```

#### 5.2 核心代码实现  
```python
from qiskit import Aer, QuantumCircuit, execute
from qiskit.visualization import plot_histogram

def verify_quantum_supremacy(n_qubits, shots):
    circuit = QuantumCircuit(n_qubits, n_qubits)
    for _ in range(20):
        circuit.h(random.randint(0, n_qubits-1))
    circuit.measure(range(n_qubits), range(n_qubits))
    
    backend = Aer.get_backend('qasm_simulator')
    result = execute(circuit, backend, shots=shots).result()
    counts = result.get_counts()
    return counts

# 示例
counts = verify_quantum_supremacy(2, 1000)
plot_histogram(counts)
```
