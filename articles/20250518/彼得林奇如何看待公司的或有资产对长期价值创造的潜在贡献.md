import pandas as pd
import numpy as np

def 计算概率(数据):

    概率 = 成功案例数 / 总案例数
    return 概率

数据 = pd.DataFrame({
    '资产': ['A', 'B', 'C', 'D'],
    '结果': ['成功', '失败', '成功', '成功']
})

概率 = 计算概率(数据)
价值 = 概率 * 100
print(价值)
