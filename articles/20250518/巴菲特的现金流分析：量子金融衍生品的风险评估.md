#### 7.2.1 现金流计算代码
```python
import numpy as np
import pandas as pd

def calculate_cash_flow(cash_flows, discount_rate):
    npv = np.npv(discount_rate, cash_flows)
    return npv

# 示例数据
cash_flows = [-100, 200, 300, -50]
discount_rate = 0.1

result = calculate_cash_flow(cash_flows, discount_rate)
print(f"净现值: {result}")
```

#### 7.2.2 风险评估代码
```python
import numpy as np
import matplotlib.pyplot as plt

def calculate VaR(cash_flows, num_simulations=10000):
    # 假设市场波动率为20%
    market_volatility = 0.2
    # 生成模拟数据
    simulations = np.random.normal(0, market_volatility, num_simulations)
    # 计算损益
    profits_loss = cash_flows + simulations
    # 排序并计算VaR
    sorted_pl = np.sort(profits_loss)
    VaR = sorted_pl[int(0.05 * num_simulations)]
    return VaR

# 示例数据
cash_flows = [100, 200, 300, 400]
VaR = calculate_VaR(cash_flows)
print(f"风险价值 (VaR): {VaR}")

# 可视化
plt.hist(simulations, bins=50)
plt.xlabel('回报率')
plt.ylabel('频率')
plt.title('回报率分布')
plt.show()
```
