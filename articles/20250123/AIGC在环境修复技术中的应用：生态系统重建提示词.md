#### 3.2 Python源代码讲解
```python
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.linear_model import LinearRegression

# 数据收集
data = pd.read_csv('environment_data.csv')

# 数据预处理
X = data.iloc[:, :-1].values
y = data.iloc[:, -1].values
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=0)

# 特征提取
scaler = StandardScaler()
X_train = scaler.fit_transform(X_train)
X_test = scaler.transform(X_test)

# 模型训练
model = LinearRegression()
model.fit(X_train, y_train)

# 模型评估
score = model.score(X_test, y_test)
print('模型评分：', score)
```

#### 7.2 系统核心实现源代码
以下是一个简单的AIGC生态系统重建系统实现示例：
```python
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.linear_model import LinearRegression

# 数据收集
data = pd.read_csv('environment_data.csv')

# 数据预处理
X = data.iloc[:, :-1].values
y = data.iloc[:, -1].values
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=0)

# 特征提取
scaler = StandardScaler()
X_train = scaler.fit_transform(X_train)
X_test = scaler.transform(X_test)

# 模型训练
model = LinearRegression()
model.fit(X_train, y_train)

# 模型评估
score = model.score(X_test, y_test)
print('模型评分：', score)

# 生成提示词
def generate_prompt_words(data):
    prompt_words = []
    for row in data:
        prompt_words.append(f"土壤湿度：{row[0]}, 温度：{row[1]}, 预测存活率：{model.predict([[row[0], row[1]])[0][0]]}")
    return prompt_words

# 实施生态系统重建
data_to_reconstruct = pd.read_csv('to_reconstruct_data.csv')
prompt_words = generate_prompt_words(data_to_reconstruct)
for word in prompt_words:
    print(word)
```
