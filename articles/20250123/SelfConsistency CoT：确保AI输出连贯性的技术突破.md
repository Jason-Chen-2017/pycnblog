    face_encodings = face_recognition.face_encodings(image, face_locations)
    return face_encodings

def recognize_faces(face_encodings, known_face_encodings, known_face_names):
    # 人脸识别
    results = []
    for face_encoding in face_encodings:
        distances = []
        for known_face_encoding in known_face_encodings:
            distance = cosine_similarity(face_encoding.reshape(1, -1), known_face_encoding.reshape(1, -1))
            distances.append(distance[0][0])
        min_distance = min(distances)
        if min_distance < 0.5:
            index = distances.index(min_distance)
            results.append(known_face_names[index])
        else:
            results.append("未知")
    return results

    # 加载已知人脸数据
    known_face_encodings = []
    known_face_names = []
    # ...（加载已知人脸数据）

    face_locations = detect_faces(image)

    # 提取人脸特征
    face_encodings = extract_face_features(image, face_locations)

    # 识别人脸
    results = recognize_faces(face_encodings, known_face_encodings, known_face_names)
