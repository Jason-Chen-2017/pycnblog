```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# 定义风险评估函数
def risk_assessment(credit_score, loan_amount, market_risk_factor, k=1):
    return 1 / (1 + np.exp(-k * credit_score * loan_amount * market_risk_factor))

# 定义数据输入接口
def input_data():
    credit_score = float(input("请输入信用评分："))
    loan_amount = float(input("请输入贷款金额："))
    market_risk_factor = float(input("请输入市场风险因素："))
    return credit_score, loan_amount, market_risk_factor

# 定义结果展示接口
def display_results(credit_score, loan_amount, market_risk_factor, risk_score):
    print(f"信用评分：{credit_score}")
    print(f"贷款金额：{loan_amount}")
    print(f"市场风险因素：{market_risk_factor}")
    print(f"风险评估结果：{risk_score}")

# 主函数
def main():
    credit_score, loan_amount, market_risk_factor = input_data()
    risk_score = risk_assessment(credit_score, loan_amount, market_risk_factor)
    display_results(credit_score, loan_amount, market_risk_factor, risk_score)

if __name__ == "__main__":
    main()
```
