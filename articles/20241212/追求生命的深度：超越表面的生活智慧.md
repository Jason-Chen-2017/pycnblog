def calculate_ms(p, d, beta=1.0):
    ms = 1 / (1 + math.exp(-beta * (p - d)))
    return ms

def calculate_ea(c, t):
    ea = c / t
    return ea

    id = 1 / (1 + math.exp(-beta * (i - r)))
