```python
from flask import Flask, request, jsonify
from sqlalchemy import create_engine, Column, Integer, String, DateTime
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker

app = Flask(__name__)

# 数据库连接
engine = create_engine('postgresql://username:password@localhost:5432/challenge_db')
Session = sessionmaker(bind=engine)
session = Session()

# 定义数据库模型
Base = declarative_base()

class ChallengeGoal(Base):
    __tablename__ = 'challenge_goals'
    id = Column(Integer, primary_key=True)
    title = Column(String)
    description = Column(String)
    created_at = Column(DateTime, default=DateTime.now())

class ChallengePlan(Base):
    __tablename__ = 'challenge_plans'
    id = Column(Integer, primary_key=True)
    goal_id = Column(Integer, ForeignKey('challenge_goals.id'))
    title = Column(String)
    description = Column(String)
    created_at = Column(DateTime, default=DateTime.now())

# API接口：创建目标
@app.route('/api/goals', methods=['POST'])
def create_goal():
    data = request.get_json()
    new_goal = ChallengeGoal(
        title=data['title'],
        description=data['description']
    )
    session.add(new_goal)
    session.commit()
    return jsonify({'message': 'Goal created successfully', 'goal': new_goal}), 201

# API接口：创建计划
@app.route('/api/plans', methods=['POST'])
def create_plan():
    data = request.get_json()
    new_plan = ChallengePlan(
        goal_id=data['goal_id'],
        title=data['title'],
        description=data['description']
    )
    session.add(new_plan)
    session.commit()
    return jsonify({'message': 'Plan created successfully', 'plan': new_plan}), 201

if __name__ == '__main__':
    Base.metadata.create_all(engine)
    app.run(debug=True)
```

#### 系统核心实现

以下是一个简单的Python代码示例，用于实现新克制主义挑战管理系统的一部分功能，如目标设定和计划管理。

```python
from flask import Flask, request, jsonify
from sqlalchemy import create_engine, Column, Integer, String, DateTime
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker

app = Flask(__name__)

# 数据库连接
engine = create_engine('postgresql://username:password@localhost:5432/challenge_db')
Session = sessionmaker(bind=engine)
session = Session()

# 定义数据库模型
Base = declarative_base()

class ChallengeGoal(Base):
    __tablename__ = 'challenge_goals'
    id = Column(Integer, primary_key=True)
    title = Column(String)
    description = Column(String)
    created_at = Column(DateTime, default=DateTime.now())

class ChallengePlan(Base):
    __tablename__ = 'challenge_plans'
    id = Column(Integer, primary_key=True)
    goal_id = Column(Integer, ForeignKey('challenge_goals.id'))
    title = Column(String)
    description = Column(String)
    created_at = Column(DateTime, default=DateTime.now())

# API接口：创建目标
@app.route('/api/goals', methods=['POST'])
def create_goal():
    data = request.get_json()
    new_goal = ChallengeGoal(
        title=data['title'],
        description=data['description']
    )
    session.add(new_goal)
    session.commit()
    return jsonify({'message': 'Goal created successfully', 'goal': new_goal}), 201

# API接口：创建计划
@app.route('/api/plans', methods=['POST'])
def create_plan():
    data = request.get_json()
    new_plan = ChallengePlan(
        goal_id=data['goal_id'],
        title=data['title'],
        description=data['description']
    )
    session.add(new_plan)
    session.commit()
    return jsonify({'message': 'Plan created successfully', 'plan': new_plan}), 201

if __name__ == '__main__':
    Base.metadata.create_all(engine)
    app.run(debug=True)
```
