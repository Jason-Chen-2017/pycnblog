1. **情感劳动权益评估指数（ELRI）**：

   $$ELRI = \frac{1}{n} \sum_{i=1}^{n} w_i \cdot E_i$$

3. **情感劳动权益保障度（ELRD）**：

```python
# 数据库连接模块
import pymysql

# 情感分析模块
from textblob import TextBlob

# 权益评估模块
def assess_rights(employee_data):
    # 根据情感分析结果评估权益保障水平
    sentiment = TextBlob(employee_data['comment']).sentiment
    if sentiment.polarity > 0:
        return 'High'
    elif sentiment.polarity < 0:
        return 'Low'
    else:
        return 'Medium'

# 支持与干预模块
def provide_support(employee_id, support_type):
    # 根据员工ID和支援类型提供支援
    if support_type == 'Psychological':
        # 提供心理支持
        print(f"Employee {employee_id} is provided with psychological support.")
    elif support_type == 'Training':
        # 提供培训机会
        print(f"Employee {employee_id} is provided with training opportunities.")
    elif support_type == 'Promotion':
        # 提供晋升机会
        print(f"Employee {employee_id} is provided with promotion opportunities.")

# 主函数
def main():
    # 连接数据库
    connection = pymysql.connect(host='localhost', user='root', password='password', database='emotional_rights')

    # 创建员工情感数据表
    with connection.cursor() as cursor:
        cursor.execute('''CREATE TABLE IF NOT EXISTS employee_data (
            id INT AUTO_INCREMENT PRIMARY KEY,
            name VARCHAR(255),
            comment TEXT,
            sentiment VARCHAR(255)
        )''')
        connection.commit()

    # 提交员工情感数据
    employee_data = {
        'name': 'Alice',
        'comment': 'I am very happy with my work environment.',
    }
    with connection.cursor() as cursor:
        cursor.execute('''INSERT INTO employee_data (name, comment) VALUES (%s, %s)''', (employee_data['name'], employee_data['comment']))
        connection.commit()

    # 分析员工情感数据
    with connection.cursor() as cursor:
        cursor.execute('''SELECT * FROM employee_data''')
        for row in cursor:
            employee_id = row[0]
            employee_name = row[1]
            employee_comment = row[2]
            sentiment = TextBlob(employee_comment).sentiment
            with connection.cursor() as cursor:
                cursor.execute('''UPDATE employee_data SET sentiment=%s WHERE id=%s''', (sentiment.polarity, employee_id))
                connection.commit()

            # 评估权益保障水平
            rights = assess_rights({'id': employee_id, 'comment': employee_comment})
            print(f"{employee_name} has {rights} rights.")

            # 提供支援
            provide_support(employee_id, 'Psychological')

    # 关闭数据库连接
    connection.close()

if __name__ == '__main__':
    main()
```
