  Z = \int e^{-S[x(t)]} dt

  其中，$S[x(t)]$ 是量子系统的作用量。

import numpy as np
import scipy.integrate as spi
import matplotlib.pyplot as plt

# 定义分子动力学模拟的函数
def molecular_dynamics(t, y, m, f):
    r = y[:3]
    v = y[3:6]
    a = f(r)
    return np.concatenate([v, a * m])

# 定义分子间的相互作用力
def potential(r):
    # 示例：Lennard-Jones势
    epsilon = 1.0
    sigma = 1.0
    r2 = np.sum(r**2)
    r6 = r2**3
    return 4 * epsilon * (r6 - 1)

# 设置参数
m = 1.0  # 分子的质量
t0 = 0  # 初始时间
tf = 10  # 结束时间
dt = 0.1  # 时间步长
y0 = np.array([0, 0, 0, 1, 0, 0])  # 初始位置和速度

# 进行分子动力学模拟
t, y = spi.odeint(molecular_dynamics, y0, [t0, tf], args=(m, potential))

# 绘制结果
plt.plot(t, y[:, 0], label='x')
plt.plot(t, y[:, 1], label='y')
plt.plot(t, y[:, 2], label='z')
plt.legend()
plt.show()

import numpy as np
from qiskit import QuantumCircuit, Aer, execute
from qiskit.visualization import plot_bloch_vector

# 定义量子蒙特卡洛模拟的函数
def quantum_monte_carlo(qc, shots=1000):
    backend = Aer.get_backend("statevector_simulator")
    result = execute(qc, backend, shots=shots).result()
    statevector = result.get_statevector()
    return statevector

# 定义量子态的函数
def quantum_state(vector):
    qc = QuantumCircuit(2)
    qc.initialize(vector, 0)
    return qc

# 设置参数
state = np.array([1/np.sqrt(2), 1/np.sqrt(2)], dtype=complex)

# 进行量子蒙特卡洛模拟
qc = quantum_state(state)
statevector = quantum_monte_carlo(qc)

# 绘制结果
plot_bloch_vector(statevector[0])
plt.show()
```
