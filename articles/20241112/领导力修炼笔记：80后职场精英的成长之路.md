        strategy = "激励型"

        strategy = "辅导型"
    
    # 根据团队目标调整策略
    if team_objective == "创新":
        strategy = "变革型"
    elif team_objective == "效率":
        strategy = "交易型"
    
    # 根据组织文化调整策略

        strategy = "民主型"
    elif organization_culture == "保守":
