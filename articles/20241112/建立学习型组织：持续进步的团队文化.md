# 学习型组织代码示例

import numpy as np
import matplotlib.pyplot as plt

# 知识共享函数
def knowledge_share(learning_rate, sharing_factor):
    return learning_rate * sharing_factor

# 持续改进函数
def continuous_improvement(improvement_rate, feedback_loop):
    return improvement_rate * feedback_loop

# 初始化参数
learning_rate = 0.1
sharing_factor = 0.8
improvement_rate = 0.05
feedback_loop = 1

# 知识共享

print(f"知识共享后知识水平：{knowledge}")

# 持续改进
improvement = continuous_improvement(improvement_rate, feedback_loop)
print(f"持续改进后效果：{improvement}")

# 绘制知识共享和持续改进曲线
plt.plot([knowledge], label='知识共享')
plt.plot([improvement], label='持续改进')
plt.xlabel('时间')
plt.ylabel('效果')
plt.legend()
plt.show()
