# 共同愿景收集程序
def collect_opinions(team_members):
    opinions = []
    for member in team_members:
        opinion = input(f"{member}, 请描述你对团队目标的看法：")
        opinions.append(opinion)
    return opinions

def calculate_common_vision(opinions, num_members):
    return sum(opinions) / num_members

# 应用案例分析与详细讲解剖析
