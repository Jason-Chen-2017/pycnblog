```python
def create_task(task_name, task_description):
    """创建任务"""
    task = {
        'name': task_name,
        'description': task_description,
        'status': '未开始'
    }
    tasks.append(task)
    return task
```

##### 任务分配
```python
def assign_task(task_id, assignee_id):
    """分配任务"""
    for task in tasks:
        if task['id'] == task_id:
            task['assignee_id'] = assignee_id
            task['status'] = '已分配'
            return task
    return None
```

##### 任务进度跟踪
```python
def update_task_progress(task_id, progress):
    """更新任务进度"""
    for task in tasks:
        if task['id'] == task_id:
            task['progress'] = progress

    return None
```
