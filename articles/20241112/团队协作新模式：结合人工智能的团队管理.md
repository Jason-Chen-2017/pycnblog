function DecisionTree(data, attributes):
    if (data is homogeneous):
        return majority class of data
    else if (attributes is empty):
        return majority class of data
    else:
        best_attribute = ChooseBestAttribute(data, attributes)
        decision_tree = Node(attribute = best_attribute)
        attributes_without_best = attributes - {best_attribute}
        for each value v in best_attribute:
            subset_data = data where attribute = v
            decision_tree.children[v] = DecisionTree(subset_data, attributes_without_best)
        return decision_tree

// 神经网络算法伪代码
function NeuralNetwork(input, weights, bias, activation_function):
    z = sum(input * weights) + bias
    a = activation_function(z)
    return a
