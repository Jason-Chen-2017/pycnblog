```python
import pandas as pd

# 读取员工绩效数据
def read_performance_data(file_path):
    data = pd.read_csv(file_path)
    return data

# 计算绩效得分
def calculate_performance_score(data):
    scores = []
    for _, row in data.iterrows():
        score = 0
        for metric, value in row[1:]:
            score += (value / 100) * 0.5  # 权重分配
        scores.append(score)
    data['performance_score'] = scores
    return data

# 主函数
def main():
    file_path = 'performance_data.csv'
    data = read_performance_data(file_path)
    data = calculate_performance_score(data)
    data.plot(kind='bar')
    plt.xlabel('员工姓名')
    plt.ylabel('绩效得分')
    plt.title('员工绩效得分分布')
    plt.show()

if __name__ == '__main__':
    main()
```
