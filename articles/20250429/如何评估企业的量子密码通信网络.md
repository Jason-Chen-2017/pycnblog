import random

def generate_qubit():
    # 生成随机数决定基和量子态
    basis = random.choice(['Z', 'X'])
    if basis == 'Z':
        qubit = random.choice([0, 1])
    else:
        qubit = random.choice([0, 1])
    return qubit, basis

def send_qubits(alice_basis, bob_basis, alice_qubits):
    # 检测基的一致性
    same_basis = [i for i in range(len(alice_basis)) if alice_basis[i] == bob_basis[i]]
    # 提取一致的基上的量子态
    key = [alice_qubits[i] for i in same_basis]
    return key
