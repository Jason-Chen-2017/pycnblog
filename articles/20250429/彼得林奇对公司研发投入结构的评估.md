- **5.1.2 Python 实现代码**

  import numpy as np

  def calculate_weights(matrix):
      # 计算一致性检验
      n = len(matrix)
      # 计算权重向量
      weights = np.dot(matrix, np.ones(n)) / np.sum(matrix, axis=0)
      return weights

  # 示例判断矩阵
  matrix = np.array([
      [1, 3, 2],
      [1/3, 1, 1/2],
      [1/2, 2, 1]
  ])

  weights = calculate_weights(matrix)
  print("权重向量为：", weights)

##### 7.2 核心代码实现

- 层次分析法的实现：
  ```python
  def ahp(criteria, alternatives, matrix):
      # 计算权重
      weight = np.array(matrix).mean(axis=0)
      weight /= weight.sum()
      
      # 计算每个方案的得分
      scores = np.dot(matrix, weight)
      return scores

  # 示例数据
  criteria = ['资金分配', '项目优先级', '人员配置']
  alternatives = ['方案1', '方案2', '方案3']
  matrix = [
      [1, 3, 2],
      [1/3, 1, 1/2],
      [1/2, 2, 1]
  ]

  # 计算得分
  scores = ahp(criteria, alternatives, matrix)
  print("得分：", scores)
  ```
