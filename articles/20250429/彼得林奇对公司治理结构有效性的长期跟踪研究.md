```python
import pandas as pd
import numpy as np

class Company:
    def __init__(self, name):
        self.name = name
        self.governance_index = 0.0

    def calculate_governance_index(self, board_score, executive_score, shareholder_score):
        self.governance_index = (board_score + executive_score + shareholder_score) / 3

class Board:
    def __init__(self, members):
        self.members = members

    def evaluate_effectiveness(self):
        # 假设评分标准：独立董事比例、高管背景等
        independent_directors = sum(1 for m in self.members if m.is_independent)
        effectiveness = (independent_directors / len(self.members)) * 0.5 + 0.1
        return effectiveness

class ExecutiveTeam:
    def __init__(self, members):
        self.members = members

    def evaluate_effectiveness(self):
        # 假设评分标准：高管平均任期、历史业绩等
        tenure_avg = sum(m.tenure for m in self.members) / len(self.members)
        effectiveness = (tenure_avg / 10) * 0.3 + 0.2
        return effectiveness

# 示例数据
class Employee:
    def __init__(self, name, is_independent=False, tenure=5):
        self.name = name
        self.is_independent = is_independent
        self.tenure = tenure

board = Board([
    Employee("Director 1", is_independent=True),
    Employee("Director 2", is_independent=True),
    Employee("Director 3", is_independent=False)
])
executive_team = ExecutiveTeam([
    Employee("CEO", is_independent=False, tenure=15),
    Employee("CFO", is_independent=False, tenure=10)
])
shareholder_score = 0.8  # 假设股东权利得分为0.8

company = Company("示例公司")
board_score = board.evaluate_effectiveness()
executive_score = executive_team.evaluate_effectiveness()
company.calculate_governance_index(board_score, executive_score, shareholder_score)
print(f"{company.name}的公司治理指数为：{company.governance_index}")
```
