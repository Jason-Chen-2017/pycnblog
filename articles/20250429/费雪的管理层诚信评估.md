import pandas as pd
from sklearn import linear_model

# 数据预处理
data = pd.read_csv('企业数据.csv')
data.dropna(inplace=True)

# 特征提取
X = data[['利润率', '市场份额', '现金流']]
y = data['诚信评分']

# 模型训练
model = linear_model.LinearRegression()
model.fit(X, y)

# 模型预测
predictions = model.predict(X)
print('预测评分:', predictions)
