```python
import numpy as np
import pandas as pd

def scenario_analysis(stock_data, scenarios):
    results = []
    for scenario in scenarios:
        adjusted_data = stock_data * (1 + scenario)
        results.append(adjusted_data.mean())
    return results
```

#### 7.2 蒙特卡洛模拟法的Python实现
```python
import numpy as np
import matplotlib.pyplot as plt

def monte_carlo_simulation(stock_data, iterations=1000):
    np.random.seed(42)
    simulated_prices = []
    for _ in range(iterations):
        drift = stock_data.mean()
        volatility = stock_data.std()
        sim_prices = drift + np.random.normal(0, volatility, len(stock_data))
        simulated_prices.append(sim_prices)
    return simulated_prices
```
