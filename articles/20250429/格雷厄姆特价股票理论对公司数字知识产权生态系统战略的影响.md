```python
from django.db import models

class IntellectualProperty(models.Model):
    IP_TYPE_CHOICES = [
        ('pat', '专利'),
        ('cop', '版权'),
        ('mar', '商标'),
    ]
    id = models.CharField(max_length=50, primary_key=True)
    type = models.CharField(max_length=3, choices=IP_TYPE_CHOICES)
    description = models.TextField()
    creation_date = models.DateTimeField(auto_now_add=True)
    owning_organization = models.ForeignKey('Organization', on_delete=models.CASCADE)
    protected = models.BooleanField(default=False)

    def __str__(self):
        return f"{self.id} {self.type}"

class Organization(models.Model):
    id = models.CharField(max_length=50, primary_key=True)
    name = models.CharField(max_length=100)
    type = models.CharField(max_length=50)
    responsible_person = models.CharField(max_length=100)

    def __str__(self):
        return self.name

class ProtectionMeasure(models.Model):
    id = models.CharField(max_length=50, primary_key=True)
    measure_type = models.CharField(max_length=50)
    coverage = models.TextField()
    creation_date = models.DateTimeField(auto_now_add=True)
    associated_ip = models.ForeignKey(IntellectualProperty, on_delete=models.CASCADE)
    status = models.CharField(max_length=20, default='pending')

    def __str__(self):
        return f"{self.id} {self.measure_type}"

class ValueMetric(models.Model):
    id = models.CharField(max_length=50, primary_key=True)
    name = models.CharField(max_length=100)
    calculation_formula = models.TextField()
    calculation_frequency = models.CharField(max_length=50)

    def __str__(self):
        return self.name

class ValueAssessment(models.Model):
    id = models.CharField(max_length=50, primary_key=True)
    assessment_result = models.FloatField(default=0)
    assessment_date = models.DateTimeField(auto_now_add=True)
    associated_metric = models.ForeignKey(ValueMetric, on_delete=models.CASCADE)

    def __str__(self):
        return f"{self.id} {self.assessment_result}"
```

$$ V = \sum_{i=1}^{n} w_i \cdot v_i $$

### 7.3.2 投资决策模型
$$ I = \sum_{j=1}^{m} p_j \cdot c_j $$
