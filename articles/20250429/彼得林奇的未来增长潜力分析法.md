  - 核心代码示例：
    ```python
    import numpy as np
    import pandas as pd

    def growth_potential_score(company_data):
        # 竞争优势评分
        competitive_advantage = company_data['market_position']
        # 财务状况评分
        financial_health = company_data['profit_margin']
        # 市场机会评分
        market_opportunity = company_data['growth_market']
        # 综合评分
        score = competitive_advantage * 0.4 + financial_health * 0.3 + market_opportunity * 0.3
        return score

    # 示例数据
    data = {
        'market_position': 0.8,
        'profit_margin': 0.6,
        'growth_market': 0.7
    }
    print(growth_potential_score(data))  # 输出结果
    ```

    - 竞争优势：0.4（权重）
    - 财务状况：0.3（权重）

    ```python
    data = pd.read_csv('company_data.csv')

    data.dropna(inplace=True)
    # 数据标准化
    from sklearn.preprocessing import StandardScaler
    scaler = StandardScaler()
    scaled_data = scaler.fit_transform(data[['market_position', 'profit_margin', 'growth_market']])
    ```

  - 模型训练：
    ```python
    from sklearn.linear_model import LinearRegression

    model = LinearRegression()
    model.fit(scaled_data, data['growth_potential'])
    ```

    ```python
    new_company = {
        'market_position': 0.8,
        'profit_margin': 0.6,
        'growth_market': 0.7

    scaled_new_company = scaler.transform([list(new_company.values())])
    predicted_score = model.predict(scaled_new_company)
    print(f'预测的增长潜力评分：{predicted_score[0]}')
    ```
