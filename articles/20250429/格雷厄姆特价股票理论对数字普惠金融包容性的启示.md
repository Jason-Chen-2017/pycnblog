def calculate_intrinsic_value(revenue, yield_rate):
    return revenue / yield_rate

def calculate_margin_of_safety(intrinsic_value, safety_factor):
    return intrinsic_value * safety_factor

# 示例代码
revenue = 1000000  # 年收入
yield_rate = 0.10   # 收益率
intrinsic_value = calculate_intrinsic_value(revenue, yield_rate)
safety_margin = calculate_margin_of_safety(intrinsic_value, 0.20)

print(f"内在价值: {intrinsic_value}")
print(f"安全边际: {safety_margin}")
