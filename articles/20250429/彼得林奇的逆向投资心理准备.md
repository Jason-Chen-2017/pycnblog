```python
import pandas as pd
import numpy as np

# 示例代码：计算恐慌指数
def calculate_fear_index(data, window=20):
    data['SMA'] = data['Close'].rolling(window).mean()
    data['Volatility'] = data['Close'].rolling(window).std()
    data['FearIndex'] = (data['Close'] - data['SMA']) / data['Volatility']
    return data['FearIndex']

fear_index = calculate_fear_index(market_data)
print(fear_index.head())
```
