#### 3.2 算法实现代码

```python
import pandas as pd
import numpy as np

def calculate_volatility(data):
    returns = data.pct_change()
    volatility = returns.std() * np.sqrt(252)
    return volatility

# 示例数据
data = pd.read_csv('stock_data.csv')
volatility = calculate_volatility(data)
print(f"波动率: {volatility}")
```
