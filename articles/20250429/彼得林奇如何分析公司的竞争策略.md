def competitive_edge_analysis(enterprise, competitors):
    industry_avg = sum(enterprise.share + [c.share for c in competitors]) / (1 + len(competitors))
    edge_index = enterprise.share / industry_avg
    return edge_index

## 4.2 核心实现源代码

### 4.2.1 竞争优势指数计算
```python
import pandas as pd

def calculate_edge_index(enterprise_share, industry_avg):
    return enterprise_share / industry_avg

# 示例数据
data = {
    '企业': ['A', 'B', 'C'],
    '市场份额': [0.25, 0.15, 0.1],
    '行业平均': [0.2, 0.2, 0.2]
}

df = pd.DataFrame(data)

# 计算竞争优势指数
edge_indices = []
for i in range(len(df)):
    edge = calculate_edge_index(df.iloc[i]['市场份额'], df.iloc[i]['行业平均'])
    edge_indices.append(edge)

df['竞争优势指数'] = edge_indices
print(df)
```
