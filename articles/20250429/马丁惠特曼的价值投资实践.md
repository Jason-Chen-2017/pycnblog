```python
import numpy as np
from scipy.optimize import minimize

def portfolio_optimization(returns):
    # 定义目标函数
    def objective(weights):
        return -weights.T.dot(returns.mean()).dot(weights)
    
    # 约束条件
    constraints = [{'type': 'eq', 'fun': lambda w: sum(w) - 1}]
    
    # 求解优化问题
    result = minimize(objective, np.ones(len(returns.columns)) / len(returns.columns), 
                     constraints=constraints)
    
    return result.x

# 示例数据
returns = np.array([[0.1, 0.2, 0.15], 
                    [0.05, 0.1, 0.08], 
                    [0.2, 0.15, 0.25]])
weights = portfolio_optimization(returns)
print(weights)
```
