# 色彩搭配算法
def color_mixture(color1, color2, ratio):
    color1_hex = color1.lstrip('#')
    color2_hex = color2.lstrip('#')
    
    color1_rgb = tuple(int(color1_hex[i:i+2], 16) for i in (1, 3, 5))
    color2_rgb = tuple(int(color2_hex[i:i+2], 16) for i in (1, 3, 5))
    
    mixed_rgb = [
        int(color1_rgb[i] * (1 - ratio) + color2_rgb[i] * ratio)
        for i in range(3)
    ]
    
    return '#{:02x}{:02x}{:02x}'.format(*mixed_rgb)
