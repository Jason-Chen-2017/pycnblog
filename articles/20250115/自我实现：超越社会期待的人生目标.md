def analyze_state(user_id):
    # 连接数据库
    conn = connect_database()
    cursor = conn.cursor()

    cursor.execute("SELECT goals.goal_name, goals.goal_description, evaluations.score FROM goals JOIN evaluations ON goals.user_id = evaluations.user_id WHERE goals.user_id = %s", (user_id,))
