**5.1.2 Python代码示例**

以下是一个简单的Python代码示例，展示了如何实现数字干扰过滤算法：

```python
import re

# 用户操作历史
user_history = [
    '浏览了社交媒体',
    '收到了一封电子邮件',
    '打开了在线视频',
    '阅读了一篇文章'
]

# 过滤规则
filter_rules = [
    '社交媒体',
    '电子邮件',
    '在线视频'
]

# 干扰识别
def identify_interference(user_history, filter_rules):
    interference = []
    for activity in user_history:
        if re.search('|'.join(filter_rules), activity):
            interference.append(activity)
    return interference

# 实时过滤
def filter_interference(activity, filter_rules):
    return re.search('|'.join(filter_rules), activity) is not None

# 用户反馈
def user_feedback(interference_detected):
    if interference_detected:
        print("干扰应用被过滤：", interference_detected)
    else:
        print("无干扰应用")
```

```python
# interference_filter.py

import re

# 数字干扰过滤算法
class InterferenceFilter:
    def __init__(self, filter_rules):
        self.filter_rules = filter_rules

    def identify_interference(self, activity):
        for rule in self.filter_rules:
            if re.search(rule, activity):
                return True
        return False

# 示例过滤规则
filter_rules = ['社交媒体', '电子邮件', '在线视频']

# 实例化干扰过滤对象
filter = InterferenceFilter(filter_rules)

# 识别干扰
interference_detected = filter.identify_interference('浏览了社交媒体')
print("干扰应用被过滤：", interference_detected)
```
