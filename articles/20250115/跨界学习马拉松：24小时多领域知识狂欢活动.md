from flask import render_template, url_for, flash, redirect, request
from flask_login import current_user, login_required
from werkzeug.urls import url_parse
from datetime import datetime
from . import app, db
from .models import User, Topic, Activity

@app.before_request

        current_user.last_login = datetime.utcnow()
        db.session.commit()

@app.route('/')
@app.route('/index')
@login_required

        topic = Topic(name=request.form['name'], description=request.form['description'])
        db.session.add(topic)
        db.session.commit()
        flash('New topic was successfully created!', 'success')
        return redirect(url_for('index'))
    return render_template('new_topic.html', title='New Topic')

@app.route('/activity/new', methods=['GET', 'POST'])
@login_required
def new_activity():
    if request.method == 'POST':
        activity = Activity(name=request.form['name'], description=request.form['description'],
                            start_time=datetime.utcnow(), end_time=datetime.utcnow() + timedelta(hours=1),
                            topic_id=request.form['topic_id'])
        db.session.add(activity)
        db.session.commit()
        flash('New activity was successfully created!', 'success')
        return redirect(url_for('index'))
    topics = Topic.query.all()
    return render_template('new_activity.html', title='New Activity', topics=topics)
