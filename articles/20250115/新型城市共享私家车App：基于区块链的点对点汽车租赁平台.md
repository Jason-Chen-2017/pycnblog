- **对称加密**：对称加密算法使用相同的密钥对数据进行加密和解密。常见的对称加密算法有AES（高级加密标准）。其优点是加密速度快，但缺点是密钥管理复杂。
  
- **非对称加密**：非对称加密算法使用一对密钥（公钥和私钥）进行加密和解密。常见的非对称加密算法有RSA（Rivest-Shamir-Adleman）。其优点是解决了密钥管理问题，但缺点是加密速度较慢。

2. **交易数据加密**：在用户发起租赁请求和支付操作时，交易数据需要加密传输，防止中间人攻击和数据篡改。可以使用对称加密算法进行数据加密，同时使用非对称加密算法进行密钥交换。

1. **密钥生成**：非对称加密算法（如RSA）的密钥生成过程涉及以下步骤：
   - 选择两个大素数 \( p \) 和 \( q \)。
   - 计算 \( n = p \times q \)。
   - 计算 \( \phi(n) = (p-1) \times (q-1) \)。
   - 选择一个整数 \( e \)，满足 \( 1 < e < \phi(n) \)，且 \( e \) 与 \( \phi(n) \) 互质。
   - 计算 \( d \)，满足 \( d \times e \equiv 1 \pmod{\phi(n)} \)。

   对应的LaTeX公式：
   \[
   n = p \times q, \quad \phi(n) = (p-1) \times (q-1), \quad d = e^{-1} \pmod{\phi(n)}
   \]

2. **加密过程**：数据 \( M \) 使用公钥 \( (n, e) \) 进行加密：
   \[
   C = M^e \pmod{n}
   \]

3. **解密过程**：加密数据 \( C \) 使用私钥 \( (n, d) \) 进行解密：
   \[
   M = C^d \pmod{n}

**智能合约**

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract CarRental {
    struct RentalOrder {
        address renter;
        address owner;
        uint256 rentalId;
        uint256 startTime;
        uint256 endTime;
        uint256 price;
        bool isCompleted;
    }

    mapping(uint256 => RentalOrder) public rentalOrders;
    mapping(address => uint256[]) public userRentalOrders;

    event RentalCreated(uint256 rentalId, address owner, address renter, uint256 price);
    event RentalStarted(uint256 rentalId, uint256 startTime);
    event RentalEnded(uint256 rentalId, uint256 endTime);
    event RentalCompleted(uint256 rentalId);

    function createRentalOrder(uint256 rentalId, address owner, uint256 price) external {
        require(!rentalOrders[rentalId].isCompleted, "Rental already completed");
        rentalOrders[rentalId] = RentalOrder({
            renter: msg.sender,
            owner: owner,
            rentalId: rentalId,
            startTime: 0,
            endTime: 0,
            price: price,
            isCompleted: false
        });
        userRentalOrders[owner].push(rentalId);
        emit RentalCreated(rentalId, owner, msg.sender, price);
    }

    function startRental(uint256 rentalId) external {
        require(!rentalOrders[rentalId].isCompleted, "Rental already completed");
        require(rentalOrders[rentalId].renter == msg.sender, "Not the renter");
        rentalOrders[rentalId].startTime = block.timestamp;
        emit RentalStarted(rentalId, rentalOrders[rentalId].startTime);
    }

    function endRental(uint256 rentalId) external {
        require(!rentalOrders[rentalId].isCompleted, "Rental already completed");
        require(rentalOrders[rentalId].owner == msg.sender, "Not the owner");
        rentalOrders[rentalId].endTime = block.timestamp;
        emit RentalEnded(rentalId, rentalOrders[rentalId].endTime);
    }

    function completeRental(uint256 rentalId) external {
        require(!rentalOrders[rentalId].isCompleted, "Rental already completed");
        require(rentalOrders[rentalId].endTime != 0, "Rental not ended");
        rentalOrders[rentalId].isCompleted = true;
        emit RentalCompleted(rentalId);
    }
}
```
