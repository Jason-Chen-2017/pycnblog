]

# 创建朴素贝叶斯分类器

# 构建模型管道
model = make_pipeline(vectorizer, classifier)

# 训练模型
model.fit(texts, labels)

# 测试模型
predictions = model.predict(["这部电影非常精彩。"])

print(predictions)
