        # 更新执行节点状态

# 执行动态任务分配算法

def assign_task():
    while True:
        if tasks:
            task = tasks.pop(0)
            min_load_node = None
            min_load = float('inf')
            for node in nodes:
                load = nodes[node]["load"]
                if load < min_load:
                    min_load = load
                    min_load_node = node
            if min_load_node:
                nodes[min_load_node]["load"] += 1
                requests.post(f'http://{min_load_node}:8080/execute_task', json=task)
        time.sleep(1)

if __name__ == '__main__':
