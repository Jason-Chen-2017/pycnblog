from graphframes import GraphFrame

# 实体抽取
def extract_entities(question):
    # 使用NLP技术提取实体
    entities = nlp(question).ents
    return entities

# 关系抽取
def extract_relations(entities):
    # 确定实体之间的关系
    relations = []
    for entity in entities:
        if entity.label_ == 'PERSON':
            relations.append(('张三', '工作于'))
    return relations

# 知识图谱构建
def build_knowledge_graph(entities, relations):
    vertices = [entity.text for entity in entities]
    edges = relations
    graph = GraphFrame(vertices, edges)
    return graph

# 提示词工程设计
def select_prompt(question_type, context):
    if question_type == 'WORKPLACE':
        return "请回答关于张三的工作地点的问题。"
    else:
        return "请回答您的问题。"

# 主函数
def main():
    question = "张三的工作地点是哪里？"
    entities = extract_entities(question)
    relations = extract_relations(entities)
    graph = build_knowledge_graph(entities, relations)
    prompt = select_prompt('WORKPLACE', question)
    print(prompt)

if __name__ == "__main__":
    main()
