import openai

def generate_story(prompt, model_engine='text-davinci-002', max_tokens=100):
    response = openai.Completion.create(
        engine=model_engine,
        prompt=prompt,
        max_tokens=max_tokens,
        n=1,
        stop=None,
        temperature=0.5
    )
    return response.choices[0].text.strip()
