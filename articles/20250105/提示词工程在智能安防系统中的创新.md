def collect_videos(camera_id, video_path):

import cv2
import numpy as np

def preprocess_video(video_path):
    cap = cv2.VideoCapture(video_path)

**报警通知模块**：

import smtplib
from email.mime.text import MIMEText
from email.header import Header

def notify_alarm(address, password, recipients, message):
    subject = '智能安防系统报警通知'
    content = MIMEText(message, 'plain', 'utf-8')
    content['Subject'] = Header(subject, 'utf-8')
    
    server = smtplib.SMTP('smtp.example.com', 587)
    server.starttls()
    server.login(address, password)
    
    for recipient in recipients:
        content['To'] = Header(recipient, 'utf-8')
        server.sendmail(address, [recipient], content.as_string())
    
    server.quit()

# 使用示例
notify_alarm('your_email@example.com', 'your_password', ['recipient1@example.com', 'recipient2@example.com'], '紧急情况：有人在公园打架，请立即处理。')
