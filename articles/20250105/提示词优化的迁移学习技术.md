### 算法原理的数学模型和公式

- **特征筛选**：设训练数据集为D，特征集合为F，筛选关键特征的目标函数为：
  $$ \min_{f \in F} D(f) $$
  其中，$D(f)$为特征f的判别力。

- **特征加权**：设特征权重集合为W，目标函数为：
  $$ \max_{w \in W} \sum_{f \in F} w_f f $$
  其中，$w_f$为特征f的权重。

- **特征重构**：设重构后的特征集合为G，目标函数为：
  $$ \min_{g \in G} ||g - f||^2 $$
  其中，$g$为重构后的特征，$f$为原始特征。

### 系统核心实现

以下是系统核心实现的Python代码：

```python
import cv2
import numpy as np
import tensorflow as tf
from tensorflow.keras.applications import MobileNetV2
from tensorflow.keras.layers import GlobalAveragePooling2D
from tensorflow.keras.models import Model

# 视频处理模块
def video_processing(video_path):
    cap = cv2.VideoCapture(video_path)
    frames = []

    while cap.isOpened():
        ret, frame = cap.read()
        if not ret:
            break

        frame = cv2.resize(frame, (224, 224))
        frames.append(frame)

    cap.release()
    return frames

# 特征优化模块
def feature_optimization(frames):
    # 加载预训练模型
    base_model = MobileNetV2(weights='imagenet', include_top=False)
    x = base_model.input
    x = GlobalAveragePooling2D()(x)

    # 定义迁移学习模型
    model = Model(inputs=x, outputs=x)
    model.compile(optimizer='adam', loss='categorical_crossentropy', metrics=['accuracy'])

    # 训练模型
    model.fit(frames, epochs=10, batch_size=32)

    # 提取特征
    features = model.predict(frames)
    return features

# 分类模块
def classification(features):
    # 加载分类器
    classifier = tf.keras.Sequential([
        tf.keras.layers.Dense(128, activation='relu', input_shape=(7,)),
        tf.keras.layers.Dense(2, activation='softmax')
    ])

    classifier.compile(optimizer='adam', loss='categorical_crossentropy', metrics=['accuracy'])

    # 训练分类器
    classifier.fit(features, epochs=10, batch_size=32)

    # 预测分类结果
    predictions = classifier.predict(features)
    return predictions

# 主程序
def main(video_path):
    frames = video_processing(video_path)
    features = feature_optimization(frames)
    predictions = classification(features)

    # 打印分类结果
    for pred in predictions:
        print(f"分类结果：{pred}")

if __name__ == '__main__':
    video_path = 'path/to/video.mp4'
    main(video_path)
```
