```python
import torch
import torch.nn as nn
import torch.optim as optim

class MemoryEnhancedModel(nn.Module):
    def __init__(self, vocab_size, embedding_dim, hidden_dim, output_dim):
        super(MemoryEnhancedModel, self).__init__()
        self.embedding = nn.Embedding(vocab_size, embedding_dim)
        self.memory_network = nn.GRU(embedding_dim, hidden_dim, num_layers=1, batch_first=True)
        self.fc = nn.Linear(hidden_dim, output_dim)

    def forward(self, input, memory):
        embedded = self.embedding(input)
        memory, _ = self.memory_network(memory, embedded)
        output = self.fc(memory)
        return output

model = MemoryEnhancedModel(vocab_size=10000, embedding_dim=256, hidden_dim=512, output_dim=1)
optimizer = optim.Adam(model.parameters(), lr=0.001)
```
