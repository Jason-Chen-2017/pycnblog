import nltk
from nltk.tokenize import word_tokenize
from spacy.lang.en import English
from mermaid import Mermaid

# 词法分析
def lexical_analysis(text):
    tokens = word_tokenize(text)
    return tokens

# 语法分析
def syntactic_analysis(tokens):
    nlp = English()
    doc = nlp(tokens)
    syntax_tree = [{"text": token.text, "type": token.pos_} for token in doc]
    return syntax_tree

# 语义分析
def semantic_analysis(syntax_tree):
    semantic_representation = {}
    for item in syntax_tree:
        if item['type'] in ['NOUN', 'VERB', 'ADP']:
            semantic_representation[item['text']] = item['text']
    return semantic_representation

# Mermaid流程图渲染
def render_mermaid(syntax_tree):
    mermaid = Mermaid()
    mermaid.add_node('root', 'Start')
    for item in syntax_tree:
        mermaid.add_edge('root', item['text'])
    mermaid_code = mermaid.get_mermaid_code()
    print(mermaid_code)

# 主函数
def main():
    input_text = "如果明天天气很好，就出去散步。"
    tokens = lexical_analysis(input_text)
    syntax_tree = syntactic_analysis(tokens)
    semantic_representation = semantic_analysis(syntax_tree)
    render_mermaid(syntax_tree)

if __name__ == "__main__":
    main()
