import numpy as np
import tensorflow as tf

# 创建强化学习模型
class ReinforcementLearningAgent:
    def __init__(self, state_size, action_size, learning_rate):
        self.state_size = state_size
        self.action_size = action_size
        self.learning_rate = learning_rate
        
        self.model = self._build_model()
        
    def _build_model(self):
        model = tf.keras.Sequential()
        model.add(tf.keras.layers.Dense(24, activation='relu', input_shape=(self.state_size,)))
        model.add(tf.keras.layers.Dense(24, activation='relu'))
        model.add(tf.keras.layers.Dense(self.action_size, activation='softmax'))
        
        model.compile(loss='mse', optimizer=tf.keras.optimizers.Adam(learning_rate=self.learning_rate))
        return model
    
    def predict(self, state):
        state = state.reshape(1, self.state_size)
        action_values = self.model.predict(state)
        return np.argmax(action_values)
    
    def train(self, states, actions, rewards, new_states, done):
        if done:
            target_q = rewards
        else:
            target_q = rewards + self.model.predict(new_states.reshape(1, self.state_size)) * 0.99
        
        target_f = self.model.predict(states.reshape(1, self.state_size))
        target_f[0][actions] = target_q
        self.model.fit(states, target_f, epochs=1, verbose=0)

# 实例化强化学习模型
