def create_merkle_tree(data):
    if len(data) == 1:
        return [hash(data[0])]
    else:
        left_child = create_merkle_tree(data[:len(data) // 2])
        right_child = create_merkle_tree(data[len(data) // 2:])
        return [hash(left_child[i] + right_child[i]) for i in range(len(left_child))]

def verify_merkle_tree(data, root_hash):
