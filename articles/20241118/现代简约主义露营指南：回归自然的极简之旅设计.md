#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

// 装备类定义
class Equipment {
public:
    string Name;
    int Weight;
    int Volume;
    string Functionality;

    Equipment(string name, int weight, int volume, string functionality) {
        Name = name;
        Weight = weight;
        Volume = volume;
        Functionality = functionality;
    }
};

// 装备选择算法
vector<Equipment> selectEquipment(vector<Equipment> equipments, int weightLimit, int volumeLimit, string functionalityRequirements) {
    vector<Equipment> selectedEquipments;
    for (Equipment equipment : equipments) {
        if (equipment.Weight <= weightLimit && equipment.Volume <= volumeLimit && equipment.Functionality == functionalityRequirements) {
            selectedEquipments.push_back(equipment);
        }
    }
    return selectedEquipments;
}

int main() {
    // 初始化装备列表
    vector<Equipment> equipments;
    equipments.push_back(Equipment("帐篷", 3, 10, "住宿"));
    equipments.push_back(Equipment("睡袋", 1, 5, "住宿"));
    equipments.push_back(Equipment("炊具", 2, 8, "烹饪"));
    equipments.push_back(Equipment("食品", 1, 3, "食品"));

    // 设定限制条件
    int weightLimit = 10;
    int volumeLimit = 20;
    string functionalityRequirements = "住宿";

    // 选择装备
    vector<Equipment> selectedEquipments = selectEquipment(equipments, weightLimit, volumeLimit, functionalityRequirements);

    // 输出选择结果
    cout << "选择的装备：\n";
    for (Equipment equipment : selectedEquipments) {
        cout << equipment.Name << " (" << equipment.Weight << "kg, " << equipment.Volume << "L)\n";
    }

    return 0;
}
