```python
import numpy as np
import matplotlib.pyplot as plt

def quantum_tunneling(E, barrier_height, barrier_width):
    """
    模拟量子隧穿现象。

    参数：
    E -- 粒子能量
    barrier_height -- 势垒高度
    barrier_width -- 势垒宽度
    """
    dx = 0.01
    x = np.arange(0, barrier_width, dx)
    potential = np.zeros_like(x)
    potential[:int(barrier_width/dx)] = barrier_height

    # 初始化波函数
    psi = np.exp(-x**2 / 2) * np.sin(x)
    
    # 计算波函数在势垒后的变化
    psi_barrier = np.fft.ifft(np.fft.fft(psi) * np.exp(-1j * potential * dx))
    
    # 绘制波函数图
    plt.plot(x, psi, label='初态')
    plt.plot(x, psi_barrier.real, label='终态')
    plt.xlabel('Position')
    plt.ylabel('Wavefunction')
    plt.legend()
    plt.show()

# 参数设置
E = 1.0
barrier_height = 2.0
barrier_width = 1.0

# 执行量子隧穿模拟
quantum_tunneling(E, barrier_height, barrier_width)
```
