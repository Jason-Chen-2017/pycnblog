```python
class Stock:
    def __init__(self, code, name, price):
        self.code = code
        self.name = name
        self.price = price

    def get_price(self):
        return self.price

class Trade:
    def __init__(self, stock, quantity):
        self.stock = stock
        self.quantity = quantity

    def buy(self):
        return self.stock.get_price() * self.quantity

    def sell(self):
        return self.stock.get_price() * self.quantity

class Report:
    def __init__(self, trades):
        self.trades = trades

    def generate_report(self):
        total_buy = sum(trade.buy() for trade in self.trades if trade.buy() > 0)
        total_sell = sum(trade.sell() for trade in self.trades if trade.sell() > 0)
        return f"总买入金额：{total_buy}，总卖出金额：{total_sell}"

if __name__ == "__main__":
    stock = Stock("000001", "中国平安", 50)
    trade1 = Trade(stock, 10)
    trade2 = Trade(stock, -5)
    report = Report([trade1, trade2])
    print(report.generate_report())
```
