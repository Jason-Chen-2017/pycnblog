psi = np.array([[1], [0]])

def measure(psi):
    probabilities = np.abs(psi)**2
    result = np.random.choice([0, 1], p=probabilities)
    return result

# 进行量子测量并更新量子态
result = measure(psi)
psi = np.array([[0], [1]] if result else [[1], [0]])

# 绘制量子态
plt.bar([0, 1], np.abs(psi)**2)
plt.xlabel('State')
plt.ylabel('Probability')
plt.title('Quantum State after Measurement')
plt.show()
