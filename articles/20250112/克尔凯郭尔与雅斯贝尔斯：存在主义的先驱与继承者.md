```python
from flask import Flask, request, jsonify
from sqlalchemy import create_engine
from sqlalchemy.orm import sessionmaker

app = Flask(__name__)

# 数据库配置
engine = create_engine('sqlite:///individual_awareness.db')
Session = sessionmaker(bind=engine)
session = Session()

# 添加个体意识数据
@app.route('/add_awareness', methods=['POST'])
def add_awareness():
    data = request.get_json()
    awareness = IndividualAwareness(
        date=data['date'],
        description=data['description']
    )
    session.add(awareness)
    session.commit()
    return jsonify({'status': 'success'})

# 查询个体意识数据
@app.route('/get_awareness', methods=['GET'])
def get_awareness():
    awarenesses = session.query(IndividualAwareness).all()
    return jsonify({'awarenesses': [a.to_dict() for a in awarenesses]})

if __name__ == '__main__':
    app.run(debug=True)
```
