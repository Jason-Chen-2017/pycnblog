```python
# 互动模块示例代码

import socket
import threading

# 实时弹幕
class ChatRoom:
    def __init__(self):
        self.clients = []
        self.lock = threading.Lock()

    def send_message(self, message, client):
        with self.lock:
            for c in self.clients:
                if c != client:
                    c.send(message.encode())

    def add_client(self, client):
        with self.lock:
            self.clients.append(client)

    def remove_client(self, client):
        with self.lock:
            self.clients.remove(client)

# 提问和回答
class QuestionAndAnswer:
    def __init__(self):
        self.questions = []

    def ask_question(self, question, client):
        self.questions.append((question, client))
        print(f"Question asked by {client}: {question}")

    def answer_question(self, question_id, answer):
        for q, client in self.questions:
            if question_id == q['id']:
                q['answer'] = answer
                print(f"Answered question by {client}: {answer}")
                break

# 主函数
if __name__ == "__main__":
    chat_room = ChatRoom()
    qa = QuestionAndAnswer()

    # 启动聊天室
    chat_room_thread = threading.Thread(target=lambda: chat_room.run())
    chat_room_thread.start()

    # 启动问答
    qa_thread = threading.Thread(target=lambda: qa.run())
    qa_thread.start()

    print("Chat room and QA system started.")
```
