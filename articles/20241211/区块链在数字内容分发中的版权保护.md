contract CopyrightProtection {
    // 结构体：版权信息
    struct Copyright {
        string title;
        address creator;
        uint256 creationTime;
    }

    // 字典：版权信息映射
    mapping(string => Copyright) public copyrights;

    // 事件：版权登记
    event CopyrightRegistered(string title, address creator, uint256 creationTime);

        emit CopyrightRegistered(title, msg.sender, block.timestamp);
    }

    // 函数：查询版权
    function queryCopyright(string memory title) public view returns (address creator, uint256 creationTime) {
        Copyright memory copyright = copyrights[title];
        return (copyright.creator, copyright.creationTime);
    }
}
