3. **模型训练**：使用机器学习算法，如协同过滤、基于内容的推荐等，训练推荐模型。

```python
from sklearn.model_selection import train_test_split
from sklearn.metrics.pairwise import cosine_similarity
from scipy.sparse.linalg import svds

# 假设已有观众偏好矩阵R和公共空间特征矩阵S
R = np.array([[1, 0, 1], [1, 1, 0], [0, 1, 1]])
S = np.array([[0.8, 0.2], [0.1, 0.9], [0.5, 0.5]])

# 数据预处理
R_train, R_test = train_test_split(R, test_size=0.2)
S_train, _ = train_test_split(S, test_size=0.2)

# 计算相似度矩阵
similarity = cosine_similarity(R_train, S_train)

# SVD分解
U, sigma, V = svds(similarity, k=2)

# 模型预测
def predict(user_id, public_space_id):
    user_profile = R[user_id]
    space_profile = S[public_space_id]
    user_factor = U[user_id]
    space_factor = V[public_space_id]
```

```python
    start_time = db.Column(db.DateTime, nullable=False)
    end_time = db.Column(db.DateTime, nullable=False)

from flask import render_template, request, redirect, url_for
from app import app, db
from app.models import Artist, Activity

@app.route('/add_activity', methods=['GET', 'POST'])
def add_activity():
    if request.method == 'POST':
        title = request.form['title']
        description = request.form['description']
        start_time = request.form['start_time']
        end_time = request.form['end_time']
        location = request.form['location']
        artist_id = request.form['artist_id']

        new_activity = Activity(
            title=title,
            description=description,
            start_time=start_time,
            end_time=end_time,
            location=location,
            artist_id=artist_id
        )
        db.session.add(new_activity)
        db.session.commit()
        return redirect(url_for('index'))

    return render_template('add_activity.html')
```
