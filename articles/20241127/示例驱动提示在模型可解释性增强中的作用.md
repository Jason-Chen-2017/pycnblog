2. **示例选择**：根据相似度度量结果，选择最相似的示例数据作为匹配示例。示例选择策略可以是基于阈值的方法，也可以是基于聚类的方法。

import torchvision.transforms as transforms

# 提取决策路径
def extract_decision_path(model, images):
    decision_paths = []
    with torch.no_grad():
        for image in images:
            path = []
            for layer in model:
                if isinstance(layer, nn.Conv2d) or isinstance(layer, nn.Linear):
                    output = layer(image)
                    path.append(output)
            decision_paths.append(path)
    return decision_paths

# 可视化决策路径
def visualize_decision_path(decision_paths):
    for i, path in enumerate(decision_paths):
        fig, axes = plt.subplots(1, len(path), figsize=(10, 2.5))
        for j, layer in enumerate(path):
            if j == 0:
                axes[j].imshow(layer.squeeze(0).cpu().numpy(), cmap='gray')
                axes[j].axis('off')
            else:
                axes[j].imshow(layer.squeeze(0).cpu().numpy(), cmap='gray')
                axes[j].axis('off')
        plt.show()

        self.classifier = nn.Linear(768, 1)

        pooled_output = self.dropout(pooled_output)
        logits = self.classifier(pooled_output)
        return logits

# 数据预处理

def preprocess_text(text):
    text = text.lower()
    text = tokenizer.tokenize(text)
    text = ['[CLS]'] + text + ['[SEP]']
    return text

# 加载数据
train_data = []
for line in open('train.txt', 'r', encoding='utf-8'):
    text, label = line.strip().split('\t')
    train_data.append((preprocess_text(text), int(label)))

# 示例数据收集
examples, targets = [], []
for texts, labels in train_loader:
