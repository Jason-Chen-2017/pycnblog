import numpy as np
import matplotlib.pyplot as plt

# 宇宙膨胀模型参数
H0 = 70  #哈勃常数（单位：公里/秒/百万秒差距）
Omega_m = 0.3  # 暗物质密度参数
Omega_lambda = 0.7  # 暗能量密度参数

# 时间步长和演化时间
dt = 1  # 年
t_max = 10  # 亿年

# 初始化宇宙参数
t = np.arange(0, t_max + dt, dt)
a = np.zeros_like(t)  # 宇宙尺度因子
da_dt = np.zeros_like(t)

# 宇宙膨胀方程

# 模拟宇宙膨胀

# 绘制宇宙膨胀图

plt.xlabel('Time (Billions of Years)')
plt.ylabel('Scale Factor')
plt.title('Simulation of the Universe Expansion')
