```python
# Python代码示例
import numpy as np
import matplotlib.pyplot as plt

# 定义Lorenz方程
def lorenz(x, y, z, sigma, rho, beta):
    return sigma * (y - x),
           x * (rho - z) - y,
           x * y - beta * z

# 数值积分
t = np.linspace(0, 100, 10000)
x, y, z = np.array([1.0, 1.0, 1.0]), np.array([0.0, 0.0, 0.0])
for i in range(len(t) - 1):
    dx, dy, dz = lorenz(x[i], y[i], z[i], 10.0, 28.0, 8.0 / 3.0)
    x = np.append(x, x[i] + dx * dt)
    y = np.append(y, y[i] + dy * dt)
    z = np.append(z, z[i] + dz * dt)

# 绘制Lorenz吸引子
plt.plot(x, y, 'b', lw=0.5)
plt.title('Lorenz Attractor')
plt.show()
```
