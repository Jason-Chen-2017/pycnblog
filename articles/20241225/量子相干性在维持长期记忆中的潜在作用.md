```python
import qiskit
import matplotlib.pyplot as plt
import pandas as pd

# 创建量子计算器
qc = qiskit.QuantumCircuit(2)

# 添加量子门
qc.h(0)
qc.cx(0, 1)

# 执行量子计算
result = qc.execute(qiskit.Aer.get_backend('qasm_simulator')).result()

# 获取测量结果
measurements = result.get_counts()

# 可视化数据
plt.bar(measurements.keys(), measurements.values())
plt.xlabel('测量结果')
plt.ylabel('概率')
plt.title('量子计算结果')
plt.show()

# 数据分析
df = pd.DataFrame(measurements)
df['概率'] = df.apply(lambda x: x['0'] + x['1'], axis=1)
print(df)
```
