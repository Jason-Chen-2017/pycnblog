```python
import numpy as np

def gaussian_filter(image, sigma=1.0):
    size = int(2 * sigma + 1)
    filter = np.zeros((size, size))
    for i in range(size):
        for j in range(size):
            filter[i][j] = np.exp(-((i - size // 2)**2 + (j - size // 2)**2) / (2 * sigma**2))

    return np.convolve(image, filter, mode='same')

image = np.random.rand(10, 10)

print(filtered_image)
```
