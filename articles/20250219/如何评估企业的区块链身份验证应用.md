  // SPDX-License-Identifier: MIT
  pragma solidity ^0.8.0;
  
  contract IdentityVerification {
      struct User {
          address payable id;
          bytes32 publicKey;
          mapping(bytes32 => bool) permissions;
      }
      
      function register(bytes32 publicKey) public {
          User storage user = users[msg.sender];
          user.publicKey = publicKey;
          user.permissions[bytes32(uint256(0))] = true;
      }
      
      function verify(bytes32 publicKey) public view returns (bool) {
          User storage user = users[msg.sender];
          return user.publicKey == publicKey;
      }
  }
