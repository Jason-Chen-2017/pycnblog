import pandas as pd
import numpy as np

def reverse思维资产配置(data, target_return):
    # 计算反向指标
    reverse_indicator = data.rolling(window=20).mean().iloc[-1]
    # 计算配置权重
    weights = np.where(reverse_indicator > data.mean(), 0.7, 0.3)
    return weights.dot(data.iloc[-1])

# 示例数据

- `weights = np.where(reverse_indicator > data.mean(), 0.7, 0.3)`：根据反向指标判断权重分配。
