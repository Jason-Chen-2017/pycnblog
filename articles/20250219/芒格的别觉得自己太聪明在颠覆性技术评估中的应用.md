def compute_criterion_weights(criteria_scores):
    # 计算层次分析法中的准则权重
    # 输入：criteria_scores（各准则的得分）
    # 输出：准则权重
    total = sum(criteria_scores)
    weights = [score / total for score in criteria_scores]
    return weights

# 示例
criteria_scores = [0.2, 0.3, 0.5]
weights = compute_criterion_weights(criteria_scores)
print(weights)
