### 5.1 格雷厄姆策略的数学模型
- 内在价值公式：

pragma solidity ^0.8.0;

contract GrahamStrategy {
    // 数据存储
    struct Stock {
        uint price;
        uint dividend;
        uint growthRate;
    }

    // 评估内在价值
    function calculateIntrinsicValue(uint dividend, uint growthRate, uint discountRate) public pure returns (uint) {
        return dividend / (discountRate - growthRate);
    }

    // 判断是否符合条件
    function shouldInvest(uint price, uint intrinsicValue) public pure returns (bool) {
        return price < intrinsicValue;
    }
}
