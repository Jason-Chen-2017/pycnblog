```python
import pandas as pd
from sklearn.linear_model import LinearRegression
from sklearn.metrics import r2_score

# 加载数据
data = pd.read_csv('data.csv')

# 特征选择
X = data[['环境责任评分']]
y = data['股价表现']

# 训练模型
model = LinearRegression()
model.fit(X, y)

# 预测结果
predictions = model.predict(X)

# 评估模型
r2 = r2_score(y, predictions)
print('R平方值:', r2)
```
