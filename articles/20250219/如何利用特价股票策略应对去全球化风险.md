```python
def stock_selection(data):
    # 数据预处理
    data = data.dropna().sort_values('price')
    # 计算价值比
    value_ratio = data['price'] / data['earnings_per_share']
    # 选择价值比最低的股票
    selected = data[value_ratio < value_ratio.quantile(0.1)]
    return selected
```

```python
def calculate_volatility(prices):
    returns = np.log(prices.pct_change().dropna())
    volatility = returns.std()
    return volatility
```
