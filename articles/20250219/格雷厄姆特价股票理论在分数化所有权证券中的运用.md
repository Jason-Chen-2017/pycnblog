## 3.3 算法代码实现
```python
def calculate_intrinsic_value(cash_flows, discount_rate):
    iv = 0
    for t in range(len(cash_flows)):
        iv += cash_flows[t] / (1 + discount_rate) ** t
    return iv

def is_undervalued(current_price, iv, margin):
    return current_price < iv * (1 - margin)

# 示例
cash_flows = [100, 120, 140, 160]
discount_rate = 0.08
margin = 0.2

iv = calculate_intrinsic_value(cash_flows, discount_rate)
print(f"内在价值: {iv}")
print(f"是否低估: {is_undervalued(100, iv, margin)}")
```

## 5.2 核心代码实现
```python
import pandas as pd
import sqlite3

# 数据库连接
conn = sqlite3.connect('stocks.db')

# 数据采集
df = pd.read_sql_query("SELECT * FROM stocks WHERE type='fractional';", conn)

# 评估计算
def apply_grammar(row):
    iv = calculate_intrinsic_value(row['cash_flows'], 0.08)
    return iv

df['intrinsic_value'] = df.apply(apply_grammar, axis=1)

# 结果展示
undervalued_stocks = df[df['price'] < df['intrinsic_value'] * 0.8]
print(undervalued_stocks)
```
