```python
import numpy as np
import pandas as pd

def calculate_wacc(market_value_debt, market_value_equity, cost_debt, cost_equity, tax_rate):
    wacc = (market_value_debt / (market_value_debt + market_value_equity)) * cost_debt * (1 - tax_rate) + \
           (market_value_equity / (market_value_debt + market_value_equity)) * cost_equity
    return wacc

def calculate_eva(net_profit, wacc, capital):
    eva = net_profit - (wacc * capital)
    return eva

# 示例数据
net_profit = 100  # 净利润
capital = 500  # 资本
market_value_debt = 200  # 市场价值债务
market_value_equity = 300  # 市场价值权益
cost_debt = 0.05  # 债务成本
cost_equity = 0.10  # 权益成本
tax_rate = 0.25  # 税率

wacc = calculate_wacc(market_value_debt, market_value_equity, cost_debt, cost_equity, tax_rate)
eva = calculate_eva(net_profit, wacc, capital)

print(f"计算的WACC为：{wacc}")
print(f"计算的EVA为：{eva}")
```
