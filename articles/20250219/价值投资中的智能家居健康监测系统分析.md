### 6.2.2 数据分析模块代码

```python
import pandas as pd
from sklearn import preprocessing

# 数据预处理
def preprocess_data(data):
    # 删除缺失值
    data = data.dropna()
    # 标准化处理
    scaler = preprocessing.StandardScaler()
    data[['temperature', 'humidity']] = scaler.fit_transform(data[['temperature', 'humidity']])
    return data

# 特征提取
def extract_features(data):
    features = data[['temperature', 'humidity']]
    return features
```

### 6.2.3 价值评估模块代码
```python
import pandas as pd
from sklearn.linear_model import LinearRegression

# 价值指标计算
def calculate_value_indices(data):
    # 计算健康指数
    health_index = data['temperature'] * 0.5 + data['humidity'] * 0.3
    return health_index

# 投资价值评估
def assess_investment_value(data):
    model = LinearRegression()
    model.fit(data[['temperature', 'humidity']], data['value'])
    return model.predict(data[['temperature', 'humidity']])
```
