```python
import numpy as np
class Agent:
    def __init__(self, state_size, action_size):
        self.state_size = state_size
        self.action_size = action_size
        self.gamma = 0.99
        self.epsilon = 1.0
        self.qnetwork = np.zeros((state_size, action_size))
    
    def act(self, state):
        if np.random.random() < self.epsilon:
            return np.random.randint(self.action_size)
        else:
            return np.argmax(self.qnetwork[state])
    
    def reward(self, state, action, reward, next_state):
        self.qnetwork[state, action] = reward + self.gamma * np.max(self.qnetwork[next_state])
```

#### 5.2.2 多智能体协作算法实现
```python
import numpy as np
class MultiAgent:
    def __init__(self, num_agents, state_size, action_size):
        self.agents = [Agent(state_size, action_size) for _ in range(num_agents)]
        self.num_agents = num_agents
    
    def step(self, states, actions, rewards, next_states):
        for i in range(self.num_agents):
            self.agents[i].reward(states[i], actions[i], rewards[i], next_states[i])
    
    def get_actions(self, states):
        actions = []
        for i in range(self.num_agents):
            actions.append(self.agents[i].act(states[i]))
        return actions
```
