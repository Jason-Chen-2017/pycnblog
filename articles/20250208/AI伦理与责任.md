```python
def detect_bias(model, data):
    # 计算模型在不同群体中的表现
    groups = data['group'].unique()
    performance = {}
    for group in groups:
        subset = data[data['group'] == group]
        predictions = model.predict(subset)
        accuracy = sum(predictions == subset['label']) / len(subset)
        performance[group] = accuracy
    return performance
```
