### 3.2.1 Python代码实现

```python
import numpy as np

# 定义环境
class Environment:
    def __init__(self):
        self.states = ['bull', 'bear']
        self.actions = ['buy', 'sell']
        self.current_state = 'bull'

    def step(self, action):
        # 这里简化为随机结果
        if action == 'buy':
            reward = 1 if np.random.rand() < 0.6 else -1
        else:
            reward = -1 if np.random.rand() < 0.6 else 1
        next_state = self.states[np.random.randint(0, 2)]
        return next_state, reward

# 定义策略
class Policy:
    def __init__(self, env):
        self.env = env
        self.learning_rate = 0.1

    def act(self, state):
        if state == 'bull':
            return 'buy'
        else:
            return 'sell'

# 强化学习训练
env = Environment()
policy = Policy(env)
for _ in range(100):
    state = env.current_state
    action = policy.act(state)
    next_state, reward = env.step(action)
    # 更新策略
    if next_state == 'bull' and action == 'buy':
        policy.learning_rate += 0.01
```

def optimize_portfolio(weights, returns, risk_aversion=0.01):
    n = len(weights)
    def objective(w):
        return - (w.T @ returns @ w) + risk_aversion * (w.T @ w)
    constraints = [{'type': 'eq', 'fun': lambda w: sum(w) - 1}]
    bounds = [(0, 1) for _ in range(n)]
