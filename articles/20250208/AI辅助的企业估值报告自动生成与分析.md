#### 5.2 核心代码实现
```python
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
import spacy

# 数据预处理
data = pd.read_csv('enterprise_data.csv')
data = data.dropna()
data['label'] = data['label'].astype(int)

# 特征提取
nlp = spacy.load('en_core_web_sm')
features = []
for text in data['text']:
    doc = nlp(text)
    feature = []
    for token in doc:
        feature.append(token.lemma_)
    features.append(feature)

# 模型训练
X_train, X_test, y_train, y_test = train_test_split(features, data['label'], test_size=0.2)
model = LinearRegression()
model.fit(X_train, y_train)

# 预测与报告生成
y_pred = model.predict(X_test)
report = pd.DataFrame({'实际值': y_test, '预测值': y_pred})
report.to_csv('valuation_report.csv', index=False)
```

- 预测与报告生成：生成预测结果并保存为报告文件。
