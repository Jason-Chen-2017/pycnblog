## 3.3 算法实现代码

```python
import random

def objective_function(solution):
    # 计算收益和风险指标
    return -solution['收益'] + solution['风险']  # 最大化收益，最小化风险

def generate_initial_population(size, num_assets):
    # 初始化种群
    population = []
    for _ in range(size):
        solution = {'权重': [random.random() for _ in range(num_assets)}, 
                   '收益': 0, 
                   '风险': 0}
        population.append(solution)
    return population

def evaluate_population(population, data):
    # 计算每个方案的收益和风险
    for solution in population:
        # 假设data包含资产的历史收益和风险数据
        solution['收益'] = sum(solution['权重'][i] * data['收益'][i] for i in range(len(solution['权重'])))
        solution['风险'] = sum(solution['权重'][i] * data['风险'][i] for i in range(len(solution['权重'])))
    return population

def selection(population, k):
    # 选择前k个最优解
    population.sort(key=lambda x: objective_function(x))
    return population[:k]

def crossover(parent1, parent2):
    # 单点交叉
    point = random.randint(1, len(parent1['权重']) - 1)
    new_child = {'权重': parent1['权重'][0:point] + parent2['权重'][point:], 
                  '收益': 0, 
                  '风险': 0}
    return new_child

def mutation(solution, mutation_rate):
    # 随机突变
    for i in range(len(solution['权重'])):
        if random.random() < mutation_rate:
            solution['权重'][i] = random.random()
    return solution

def optimize(num_assets, population_size, generations):
    data = {'收益': [0.1, 0.2, 0.15, 0.08], 
            '风险': [0.05, 0.1, 0.08, 0.12]}
    
    population = generate_initial_population(population_size, num_assets)
    population = evaluate_population(population, data)
    
    for _ in range(generations):
        population = selection(population, int(population_size/2))
        new_population = []
        for i in range(0, len(population), 2):
            parent1 = population[i]
            parent2 = population[i+1]
            child1 = crossover(parent1, parent2)
            child2 = crossover(parent2, parent1)
            child1 = mutation(child1, 0.1)
            child2 = mutation(child2, 0.1)
            new_population.append(child1)
            new_population.append(child2)
        population = new_population + population
        population = evaluate_population(population, data)
    
    optimal_solution = min(population, key=lambda x: objective_function(x))
    return optimal_solution
```

## 3.4 算法的数学模型

### 3.4.1 目标函数
$$ \text{目标函数} = \max (\text{收益}) - \min (\text{风险}) $$

### 3.4.2 约束条件
$$ \sum_{i=1}^{n} w_i = 1 \quad (w_i \geq 0) $$
