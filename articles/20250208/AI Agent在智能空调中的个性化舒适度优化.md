  def fitness(x):
      # 舒适度评价函数
      return np.sum(x)

  def genetic_algorithm(population, fitness_func, num_generations=100):
      for _ in range(num_generations):
          fitness_scores = [fitness(individual) for individual in population]
          # 选择
          selected = [p for p in population if fitness(p) > np.mean(fitness_scores)]
          # 交叉
          if len(selected) > 1:
              parent1 = selected[0]
              parent2 = selected[1]
              child = np.mean([parent1, parent2], axis=0)
              population = [child] + [p for p in population if p not in selected]
          # 变异
          mutation_rate = 0.1
          for i in range(len(population)):
              if np.random.random() < mutation_rate:
                  population[i] = np.random.random() * 2 - 1
      return population[0]
