  import numpy as np
  from scipy.optimize import minimize

  def objective_function(params, constraints):
      # 定义目标函数
      return np.sum((params - constraints['target']) ** 2)

  def constraints_function(params, constraints):
      # 定义约束条件
      return {'type': 'ineq', 'fun': lambda x: x - constraints['min_value']}

  def optimize_portfolio(initial_params, constraints):
      # 使用scipy.optimize进行优化
      result = minimize(objective_function, initial_params, constraints=constraints_function(initial_params, constraints))
      return result.x

  # 示例数据
  initial_params = np.array([0.2, 0.3, 0.5])
  constraints = {
      'target': 0.4,
      'min_value': 0.1
  }
