  ```python
  import numpy as np
  from sklearn.metrics.pairwise import cosine_similarity

  def collaborative_filtering(user_item_matrix):
      similarities = cosine_similarity(user_item_matrix)
      np.fill_diagonal(similarities, 0)
      user_based_recommendations = np.dot(similarities, user_item_matrix)
      return user_based_recommendations
  ```

- **5.3.2 强化学习算法的代码实现**
  ```python
  import numpy as np
  import random

  def q_learning(env, num_episodes=1000, alpha=0.1, gamma=0.99):
      Q = np.zeros(env.observation_space.shape)
      for _ in range(num_episodes):
          state = env.reset()
          while not env.done:
              action = np.argmax(Q[state])
              next_state, reward, done = env.step(action)
              Q[state][action] += alpha * (reward + gamma * np.max(Q[next_state]))
              state = next_state
      return Q
  ```
