#### 4.1.1.2 数学模型解释
- **迁移学习公式**：表示源领域模型与目标领域模型之间的差异。
- **元学习公式**：表示模型在未知领域中的性能。
- **样本少学习公式**：表示模型在数据样本较少的情况下的性能。

import numpy as np

def migration_learning(source_theta, target_theta):
    return 0.5 * np.sum((source_theta - target_theta) ** 2)

def meta_learning(theta):
    return np.sum((theta - target_theta) ** 2)

def sample_sparing_learning(theta, x):
    return 0.5 * np.sum((theta - target_theta) ** 2)

# 示例数据
source_theta = np.array([1, 2, 3])
target_theta = np.array([4, 5, 6])

# 迁移学习计算
migration_score = migration_learning(source_theta, target_theta)

# 元学习计算
meta_score = meta_learning(target_theta)

# 样本少学习计算
sample_sparing_score = sample_sparing_learning(target_theta, x)

print(f"Migration Learning Score: {migration_score}")
print(f"Meta Learning Score: {meta_score}")
