def optimize_prompt(prompt):
    tokens = nltk.word_tokenize(prompt)
    pos_tags = nltk.pos_tag(tokens)

    for i, (word, pos) in enumerate(pos_tags):
        if pos.startswith('NN'):
            return prompt.replace(word, word.capitalize())
