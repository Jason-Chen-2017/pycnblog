```python
def stratified_sampling(users, strata, num_groups):
    groups = [[] for _ in range(num_groups)]
    for stratum in strata:
        stratum_users = [user for user in users if user.stratum == stratum]
        random.shuffle(stratum_users)
        for group in groups:
            group.extend(random.sample(stratum_users, len(group)))
    return groups
```
