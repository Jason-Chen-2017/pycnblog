// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract CopyrightChain {
    // 结构体定义版权信息
    struct Copyright {
        string contentHash;
        address creator;
        uint256 timestamp;
    }

    // 映射版权信息哈希到版权结构体
    mapping(string => Copyright) public copyrights;

    // 添加版权信息
    function registerCopyright(string memory contentHash, string memory creator) public {
        require(copyrights[contentHash].contentHash == "", "Copyright already registered");

    }

    // 查询版权信息
    function getCopyright(string memory contentHash) public view returns (Copyright memory) {
        return copyrights[contentHash];
    }
}
