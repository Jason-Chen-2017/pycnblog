$$
\Delta w_{ij} = \eta \cdot (x_i - y) \cdot y_j

以下是一个简单的源代码实现：

```python
# cognitive_algorithm.py

import torch
import torch.nn as nn
import torch.optim as optim

# 初始化神经网络
class NeuralNetwork(nn.Module):
    def __init__(self):
        super(NeuralNetwork, self).__init__()
        self.layers = nn.Sequential(
            nn.Linear(in_features=10, out_features=1),
            nn.ReLU(),
            nn.Linear(in_features=1, out_features=1)
        )

    def forward(self, x):
        return self.layers(x)

# 创建神经网络实例
neural_network = NeuralNetwork()

# 编译神经网络
optimizer = optim.Adam(neural_network.parameters(), lr=0.001)
criterion = nn.MSELoss()

# 训练神经网络
for epoch in range(100):
    # 生成随机输入和期望输出
    inputs = torch.randn(1, 10)
    expected_output = torch.tensor([1.0])

    # 前向传播
    outputs = neural_network(inputs)

    # 计算损失
    loss = criterion(outputs, expected_output)

    # 反向传播
    optimizer.zero_grad()
    loss.backward()
    optimizer.step()

    # 输出训练进度
    print(f'Epoch {epoch+1}: Loss = {loss.item()}')

# 测试神经网络
inputs = torch.randn(1, 10)
outputs = neural_network(inputs)
print(f'Output: {outputs.item()}')
```
