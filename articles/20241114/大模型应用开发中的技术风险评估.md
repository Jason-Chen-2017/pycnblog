```python
# Sample Code for Data Quality Check
def check_data_quality(data):
    """
    This function checks the quality of the input data.
    
    Parameters:
    - data (DataFrame): The input data to be checked.
    
    Returns:
    - quality (str): The quality status of the data ('High', 'Medium', 'Low').
    """
    missing_values = data.isnull().sum().sum()
    if missing_values > 0.1 * data.shape[0] * data.shape[1]:
        return 'Low'
    elif missing_values > 0.05 * data.shape[0] * data.shape[1]:
        return 'Medium'
    else:
        return 'High'
```
