def train_recommendation_model(data):
    model = NearestNeighbors(n_neighbors=5)
    model.fit(data)
    return model

def recommend_items(model, user_vector, n_recommendations=5):
    distances, indices = model.kneighbors([user_vector], n_neighbors=n_recommendations+1)
    recommendations = [index for index, distance in zip(indices[:,1:], distances[:,1:])]
    return recommendations

if __name__ == "__main__":
