// 伪代码：计算航天器轨道
function calculate_orbit(mass1, mass2, distance, velocity) {
    // 引力常数
    G = 6.6743e-11; 

    // 计算万有引力
    force = G * (mass1 * mass2) / distance^2;

    // 计算向心加速度
    acceleration = force / mass2;

    // 计算轨道半径和速度
    radius = distance / (2 * acceleration);
    velocity = sqrt(G * mass1 / radius);

    return (radius, velocity);
}

import matplotlib.pyplot as plt

G = 6.6743e-11

# 天体参数

m_sun = 1.989e30

# 时间步长

# 运动方程
def update_position_and_velocity(earth, sun):
    r = earth - sun
    distance = np.linalg.norm(r)
    force = G * (m_earth * m_sun) / distance**2
    acceleration = force / m_earth
    earth += acceleration * dt
    return earth

# 模拟行星运动
def simulate_planet():
    earth = np.array([r_earth_sun, 0])
    sun = np.array([0, 0])
    positions = [earth]

    for _ in range(1000):
        earth = update_position_and_velocity(earth, sun)
        positions.append(earth)

    return positions

# 可视化行星轨道
def visualize_orbit(positions):
    x = [pos[0] for pos in positions]
    y = [pos[1] for pos in positions]

    plt.plot(x, y)
    plt.xlabel('X Position (m)')
    plt.ylabel('Y Position (m)')
    plt.title('Earth-Sun Orbit')
    plt.show()

# 模拟并可视化行星轨道
positions = simulate_planet()
visualize_orbit(positions)
