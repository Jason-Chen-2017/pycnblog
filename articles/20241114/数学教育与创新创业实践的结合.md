    prob = pulp.LpProblem("BusinessPlanOptimization", pulp.LpMaximize)

    # Define the variables
    x = pulp.LpVariable("ProductX", cat="Continuous")
    y = pulp.LpVariable("ProductY", cat="Continuous")

    # Objective function
    prob += 3*x + 4*y

    # Constraints
    prob += x + y <= 4
    prob += 2*x + y <= 6
    prob += x >= 0
    prob += y >= 0

    # Solve the problem
    prob.solve()

    # Output the results
    for v in prob.variables():
        print(v.name, "=", v.varValue)
