3. **更新评分**：将每个数据点的熵累加到复杂性评分 CS 中，得到整个宇宙数据的平均熵 AverageEntropy。

5. **返回结果**：最后，返回复杂性评分 CS，该评分反映了宇宙的复杂程度。

```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

def calculate_entropy(data):
    """
    计算数据点的熵
    """
    unique_values, counts = np.unique(data, return_counts=True)
    probabilities = counts / len(data)
    entropy = -np.sum(probabilities * np.log2(probabilities))
    return entropy

def calculate_complexity_score(data):
    """
    计算宇宙数据的复杂性评分
    """
    entropy_values = [calculate_entropy(data[i]) for i in range(len(data))]
    average_entropy = np.mean(entropy_values)
    complexity_score = np.sum(entropy_values)
    if average_entropy > 0.5:
        complexity_score *= 1.5
    return complexity_score

def visualize_complexity_score(score):
    """
    可视化复杂性评分
    """
    plt.bar(['复杂性评分'], [score])
    plt.xlabel('评分')
    plt.ylabel('分数')
    plt.title('宇宙复杂性评分')
    plt.show()

# 加载数据
data = pd.read_csv('universe_data.csv')

# 计算复杂性评分
complexity_score = calculate_complexity_score(data)

# 可视化结果
visualize_complexity_score(complexity_score)
```
