```python
import numpy as np
import matplotlib.pyplot as plt

# 设置宇宙的基本参数
G = 6.67430e-11  # 万有引力常数
M_sun = 1.98847e30  # 太阳质量
R_sun = 6.957e8  # 太阳半径

# 创建星系
class StarSystem:
    def __init__(self, stars):
        self.stars = stars

    def evolve(self, t, dt):
        # 计算万有引力
        forces = []
        for star in self.stars:
            for other_star in self.stars:
                if star != other_star:
                    dx = other_star.position - star.position
                    distance = np.linalg.norm(dx)
                    force = G * star.mass * other_star.mass / distance**2
                    force方向 = dx / distance
                    forces.append(force方向)
        
        # 应用牛顿第二定律
        accelerations = [np.sum(force方向 / star.mass) for star in self.stars]
        
        # 更新星系中的每个星体的位置
        for star in self.stars:
            star.position += star.velocity * dt
            star.velocity += accelerations * dt

# 创建星体
class Star:
    def __init__(self, mass, position, velocity):
        self.mass = mass
        self.position = position
        self.velocity = velocity

# 创建一个包含太阳的星系
solar_system = StarSystem([Star(M_sun, np.array([0, 0]), np.array([0, 0]))])

# 模拟星系演化
time_steps = 1000
dt = 1000  # 每个时间步秒
for step in range(time_steps):
    solar_system.evolve(step, dt)

# 绘制星系
positions = [star.position for star in solar_system.stars]
plt.plot(*zip(*positions))
plt.xlabel('X Position (m)')
plt.ylabel('Y Position (m)')
plt.title('Solar System Evolution')
plt.show()
```

3. **牛顿第二定律**：我们应用牛顿第二定律，将引力转换为加速度。加速度方向指向受力方向，大小与力成正比。
