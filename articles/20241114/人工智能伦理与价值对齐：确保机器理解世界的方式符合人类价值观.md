```python
def value_alignment(values, target_value):
    """
    对输入的值进行价值对齐，使其更接近目标值。
    
    参数：
    values: 输入的值列表
    target_value: 目标值
    
    返回：
    aligned_values: 对齐后的值列表
    """
    aligned_values = []
    
    for value in values:
        aligned_value = value * (target_value / value)
        aligned_values.append(aligned_value)
    
    return aligned_values
```
