\[ S = \frac{S_1 + S_2 + \ldots + S_n}{n} \]

```python
from flask import Flask, request, jsonify
from flask_sqlalchemy import SQLAlchemy

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'mysql://username:password@localhost:3306/detox'
db = SQLAlchemy(app)

class Guest(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(50))
    email = db.Column(db.String(50))

class Activity(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(50))
    description = db.Column(db.String(255))

@app.route('/guests', methods=['POST'])
def add_guest():
    guest = Guest(name=request.json['name'], email=request.json['email'])
    db.session.add(guest)
    db.session.commit()
    return jsonify({'message': 'Guest added successfully.'})

@app.route('/guests', methods=['GET'])
def get_guests():
    guests = Guest.query.all()
    return jsonify({'guests': [guest.name for guest in guests]})

if __name__ == '__main__':
    db.create_all()
    app.run(debug=True)
```
