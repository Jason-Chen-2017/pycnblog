def assess_career_security(security_level):
    if security_level > 7:
        return "固定岗位"
    else:
        return "灵活角色"

def assess_development_path(path clarity):
    if path clarity == "明确":
        return "固定岗位"
    else:
        return "灵活角色"

def choose_job_role(security, path):
    if assess_career_security(security) == "固定岗位" and assess_development_path(path) == "固定岗位":
        return "固定岗位"
    else:
        return "灵活角色"
```

from flask import Flask, request, jsonify
from profile import Profile

app = Flask(__name__)

@app.route('/profile', methods=['GET'])
def get_profile():
    user_id = request.args.get('user_id')
    profile = Profile.get(user_id)
    return jsonify(profile.to_dict())

if __name__ == '__main__':
    app.run(debug=True)
```
