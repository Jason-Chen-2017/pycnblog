### Python源代码实现

下面是一个简单的Python源代码实现，用于计算不同年龄层的数字素养评分：

```python
import numpy as np

# 设置权重参数
weights = {'I': 0.5, 'P': 0.3, 'U': 0.2}

# 设置个体数据
data = [
    {'age': '20-30', 'education': '本科及以上', 'income': '高', 'I': 0.8, 'P': 0.85, 'U': 0.75},

    {'age': '40-50', 'education': '本科及以上', 'income': '高', 'I': 0.6, 'P': 0.5, 'U': 0.55},

]

# 计算数字素养评分
for person in data:
    S_A = weights['I'] * person['I'] + weights['P'] * person['P'] + weights['U'] * person['U']

```

```python
    db.session.commit()
```

```python
@app.route('/exams/<int:exam_id>/submit', methods=['POST'])
@login_required
def submit_exam(exam_id):
    data = request.get_json()
    answers = data['answers']
    exam = Exam.query.get(exam_id)
    score = 0
    for question, answer in zip(exam.questions, answers):
        if question.answer == answer:
            score += 1
    return jsonify({'score': score, 'message': 'Exam submitted successfully.'})

    db.create_all()
    app.run(debug=True)
```
