```python
import numpy as np
import matplotlib.pyplot as plt

# 设置宇宙模型的参数
G = 6.67430e-11  # 万有引力常数
c = 299792458   # 光速

# 初始化宇宙模型
def initialize_universe(N, r_min, r_max, r_step):
    positions = np.mgrid[r_min:r_max:r_step*1j, r_min:r_max:r_step*1j, r_min:r_max:r_step*1j]
    velocities = np.zeros_like(positions)
    masses = np.random.uniform(1, 100, size=positions.shape[0])
    return positions, velocities, masses

# 计算万有引力
def calculate_gravity(positions, velocities, masses):
    forces = np.zeros_like(positions)
    for i in range(positions.shape[0]):
        for j in range(i+1, positions.shape[0]):
            r = positions[i] - positions[j]
            r_norm = np.linalg.norm(r)
            r_hat = r / r_norm
            force = -G * masses[i] * masses[j] * r_hat / r_norm**2
            forces[i] += force
            forces[j] -= force
    return forces

# 更新宇宙模型
def update_universe(positions, velocities, masses, forces, dt):
    velocities += forces * dt
    positions += velocities * dt
    return positions, velocities

# 绘制宇宙模型
def plot_universe(positions, masses, title=""):
    plt.figure(figsize=(8, 8))
    plt.scatter(positions[:, 0], positions[:, 1], c=masses, cmap="viridis", s=10)
    plt.title(title)
    plt.xlabel("X Position")
    plt.ylabel("Y Position")
    plt.colorbar(label="Mass")
    plt.show()

# 运行宇宙模型
N = 100  # 天体数量
r_min = -10  # 宇宙范围
r_max = 10
r_step = 0.1
dt = 0.1

positions, velocities, masses = initialize_universe(N, r_min, r_max, r_step)
forces = calculate_gravity(positions, velocities, masses)

for _ in range(1000):
    positions, velocities = update_universe(positions, velocities, masses, forces, dt)
    forces = calculate_gravity(positions, velocities, masses)

plot_universe(positions, masses, "Final State of the Universe")
```

该代码实现了一个简单的宇宙模型，模拟了天体在万有引力作用下的运动。通过计算天体之间的万有引力，更新天体的位置和速度，最终绘制出宇宙模型的最终状态。这个项目可以帮助我们直观地理解宇宙中的物理现象，如引力作用、天体运动等。
