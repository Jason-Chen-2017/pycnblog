            r_hat = r / r_norm

            dydt[j] += F * r_hat
            dydt[i] -= F * r_hat
    return dydt

    solution = odeint(gravitational_interactions, velocities, t, args=(positions, velocities))

   class CelestialBody:

           self.mass = mass
           self.position = np.array(position)
           self.velocity = np.array(velocity)
           self.acceleration = np.zeros(3)

       def update_position(self, time_step):
           self.position += self.velocity * time_step
           self.velocity += self.acceleration * time_step

       def calculate_force(self, other_body):
           distance = np.linalg.norm(self.position - other_body.position)
           force_magnitude = G * self.mass * other_body.mass / distance**2
           force_direction = (other_body.position - self.position) / distance

   class UniverseSpace:

           self.bodies.append(body)

       def remove_celestial_body(self, body):

       def simulate(self, time_steps, time_step_size):
           def differential_equation(t, y):
               accelerations = np.zeros((len(self.bodies), 3))
               for i, body in enumerate(self.bodies):

               return accelerations
