```python
import numpy as np
import matplotlib.pyplot as plt

# IFS规则
a = np.array([[0.5, 0.5], [-0.5, 0.5]])
b = np.array([[0.5, -0.5], [0.5, 0.5]])
c = np.array([[-0.5, 0.5], [0.5, -0.5]])
d = np.eye(2)

# 初始化参数
x = y = 0.5
max_iterations = 1000
scale = 0.5

# 分形生成函数
def iterated_function_system(x, y, a, b, c, d, scale, max_iterations):
    for _ in range(max_iterations):
        x, y = scale * (a @ np.array([x, y]) + b @ np.array([x, y]), scale * (c @ np.array([x, y]) + d @ np.array([x, y]))
    return x, y

# 生成分形
x, y = iterated_function_system(x, y, a, b, c, d, scale, max_iterations)

# 绘图
plt.scatter(x, y)
plt.xlabel('X')
plt.ylabel('Y')
plt.title('Iterated Function System (IFS) Fractal')
plt.show()
```
