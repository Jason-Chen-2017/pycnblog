def solve_equation(equation):

    # 提取变量和系数

    constant = extract_constant(right)

    # 求解
    solution = (constant - coef) / float(coef)

    # 提取变量和系数
    # 这里简化处理，仅支持 $ax+b=0$ 形式的方程
    # 实际应用中可以使用更复杂的解析方法
    parts = expression.split('x')
    coef = parts[0].strip()
    var = 'x' if len(parts) == 1 else parts[1].strip()
    return var, coef

def extract_constant(expression):
    # 提取常数项
    # 这里简化处理，仅支持数字
    # 实际应用中可以使用更复杂的解析方法
