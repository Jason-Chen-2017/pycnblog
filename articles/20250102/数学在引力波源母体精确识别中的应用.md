def read_data(filename):

def butter_bandpass_filter(data, lowcut, highcut, fs, order=4):
    nyq = 0.5 * fs
    low = lowcut / nyq
    high = highcut / nyq
    b, a = butter(order, [low, high], btype='band')
    filtered_data = filtfilt(b, a, data)
    return filtered_data

def pca_decomposition(data, n_components):
    pca = PCA(n_components=n_components)
    filtered_data = pca.fit_transform(data)
    return filtered_data

# 特征提取
def extract_time_domain_features(data):
    mean = np.mean(data)
    variance = np.var(data)
    autocorr = np.correlate(data, data, mode='full')
    return mean, variance, autocorr

def extract_frequency_domain_features(data, fs):
    fft_data = np.fft.rfft(data)
    fft_freq = np.fft.rfftfreq(len(data), d=1/fs)
    freq_energy = np.abs(fft_data)**2
    return fft_freq, freq_energy

# 模式识别
def train_classifier(features, labels):
    classifier = SVC(kernel='rbf', C=1.0)
    classifier.fit(features, labels)
    return classifier

def predict_and_evaluate(classifier, test_features, test_labels):
    predictions = classifier.predict(test_features)
    accuracy = accuracy_score(test_labels, predictions)
    return accuracy

# 主函数
def main():
    train_data = read_data('train_data.txt')
    test_data = read_data('test_data.txt')
    
    filtered_train_data = butter_bandpass_filter(train_data, lowcut=20, highcut=500, fs=1000)
    filtered_test_data = butter_bandpass_filter(test_data, lowcut=20, highcut=500, fs=1000)
    
    train_features = extract_time_domain_features(filtered_train_data)
    test_features = extract_time_domain_features(filtered_test_data)
    
    classifier = train_classifier(train_features, labels)
    
    accuracy = predict_and_evaluate(classifier, test_features, labels)
