```python
import numpy as np
from sklearn.metrics.pairwise import cosine_similarity

def imagination_score(texts):
    # 假设 texts 是一个包含多个文本的列表
    # 计算文本间的余弦相似度
    similarities = []
    for i in range(len(texts)):
        for j in range(i+1, len(texts)):
            similarity = cosine_similarity([texts[i]], [texts[j]])
            similarities.append(similarity[0][0])
    # 计算平均相似度
    avg_similarity = np.mean(similarities)
    # 计算想象力得分
    IM = 1 / (1 + np.exp(-0.1 * avg_similarity))
    return IM

# 测试数据
texts = [
    "我看到了一片星空，闪烁着无数的光点。",
    "夜晚的星空非常美丽，星星闪烁着迷人的光芒。",
    "我凝视着夜空，感受到了宇宙的无限广阔。",
]

# 计算想象力得分
IM = imagination_score(texts)
print(f"想象力得分: {IM}")
```
