        for i in range(1, len(segments) - 1):
            segments[i] = phi * segments[i - 1]
    return segments

    """优化建筑尺寸以符合黄金比例

    参数:
    width: 建筑宽度
    height: 建筑高度

    返回值:

    segments = golden_ratio_section(height, 2)  # 对高度进行分割
