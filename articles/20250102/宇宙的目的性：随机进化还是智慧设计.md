```python
import numpy as np
import matplotlib.pyplot as plt

# 设置随机种子，确保结果可复现
np.random.seed(42)

# 初始化宇宙参数
num_planets = 100  # 行星数量
num_steps = 100  # 演化步骤

# 生成随机行星数据
planets = np.random.rand(num_planets, 2)  # 行星位置

# 演化步骤
for step in range(num_steps):
    # 计算每个行星的引力相互作用
    forces = np.zeros_like(planets)
    for i in range(num_planets):
        for j in range(num_planets):
            if i != j:
                dx = planets[i, 0] - planets[j, 0]
                dy = planets[i, 1] - planets[j, 1]
                distance = np.sqrt(dx**2 + dy**2)
                force = -6.67430e-11 * (dx / distance)  # 引力公式
                forces[i] += force

    # 更新行星位置
    planets += forces / 1000  # 减速因子，使演化更稳定

# 绘制演化结果
plt.figure(figsize=(10, 6))
plt.scatter(planets[:, 0], planets[:, 1], s=5)
plt.xlabel('X坐标')
plt.ylabel('Y坐标')
plt.title('宇宙演化模拟')
plt.show()
```
