```python
import numpy as np
import matplotlib.pyplot as plt
from mermaid import Mermaid

# 参数设置
time_steps = 100
time_step_size = 0.01

temperature[0] = 1e15

# 模拟过程
for i in range(1, time_steps):
    # 时间演化
    time = i * time_step_size
    
    # 热辐射冷却
    temperature[i] = temperature[i-1] * (1 - 0.001 * time)

# 绘制结果
plt.figure(figsize=(10, 5))
plt.plot(time, temperature, label='Temperature')
plt.xlabel('Time (s)')
plt.ylabel('Temperature (K)')
plt.legend()
plt.title('Simulation of Cosmic Background Radiation Temperature')
plt.show()

# Mermaid流程图
flow = Mermaid('graph TD\n'
              'A[Initial State]\n'
              'B[Expanding Universe]\n'
              'C[Cooling Process]\n'
              'D[Final State]\n'
              'A --> B\n'
              'B --> C\n'
              'C --> D')
flow.save('cosmic_background Radiation_mermaid.md')
```
