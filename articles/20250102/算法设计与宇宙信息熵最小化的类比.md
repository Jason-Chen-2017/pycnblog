2. **构建霍夫曼树**：根据频次分布表，构建霍夫曼树，叶子节点表示字符，路径表示编码。

```python
import heapq
from collections import Counter

def build_huffman_tree(freqs):
    heap = [[weight, [symbol, ""]] for symbol, weight in freqs.items()]
    heapq.heapify(heap)
    while len(heap) > 1:
        lo = heapq.heappop(heap)
        hi = heapq.heappop(heap)
        for pair in lo[1:]:
            pair[1] = '0' + pair[1]
        for pair in hi[1:]:
            pair[1] = '1' + pair[1]
        heapq.heappush(heap, [lo[0] + hi[0]] + lo[1:] + hi[1:])
    return heap[0]

def generate_codes(tree):
    code = ""
    codes = {}
    for symbol, path in tree:
        if path:
            codes[symbol] = code
            generate_codes([symbol, path])
            code = code[:-1]
        else:
            code += "0"
            codes[symbol] = code
            code = ""
    return codes

def compress_data(data, codes):
    return ''.join([codes[c] for c in data])

if __name__ == "__main__":
    data = "this is an example for huffman coding"
    freqs = Counter(data)
    huffman_tree = build_huffman_tree(freqs)
    codes = generate_codes(huffman_tree)
    compressed_data = compress_data(data, codes)
    print("Original data:", data)
    print("Compressed data:", compressed_data)
```

#### 4.2.2 学习算法案例

以线性回归为例，其原理如下：

1. **确定损失函数**：通常使用均方误差（MSE）作为损失函数，表示预测值与实际值之间的差距。
   $$\text{MSE} = \frac{1}{n} \sum_{i=1}^{n} (\hat{y}_i - y_i)^2$$
   其中，$\hat{y}_i$ 为预测值，$y_i$ 为实际值，$n$ 为样本数量。
2. **计算梯度**：计算损失函数关于模型参数的梯度。
   $$\nabla_\theta \text{MSE} = -2 \sum_{i=1}^{n} (\hat{y}_i - y_i) x_i$$
   其中，$\theta$ 为模型参数，$x_i$ 为输入特征。
3. **更新参数**：根据梯度下降方向，更新模型参数。
   $$\theta := \theta - \alpha \nabla_\theta \text{MSE}$$
   其中，$\alpha$ 为学习率。
4. **迭代优化**：重复计算梯度、更新参数的过程，直到达到预设的停止条件。

#### 4.2.3 Python代码实现

```python
import numpy as np

def linear_regression(X, y, alpha, epochs):
    m = X.shape[0]
    theta = np.zeros(X.shape[1])
    X = np.concatenate(([np.ones((m, 1))], X), axis=1)
    for _ in range(epochs):
        y_pred = X @ theta
        error = y - y_pred
        theta -= alpha * X.T @ error / m
    return theta

if __name__ == "__main__":
    X = np.array([[1, 2], [2, 3], [3, 4], [4, 5]])
    y = np.array([3, 4, 5, 6])
    alpha = 0.01
    epochs = 1000
    theta = linear_regression(X, y, alpha, epochs)
    print("Theta:", theta)
```

```python
import heapq
from collections import Counter

def build_huffman_tree(freqs):
    heap = [[weight, [symbol, ""]] for symbol, weight in freqs.items()]
    heapq.heapify(heap)
    while len(heap) > 1:
        lo = heapq.heappop(heap)
        hi = heapq.heappop(heap)
        for pair in lo[1:]:
            pair[1] = '0' + pair[1]
        for pair in hi[1:]:
            pair[1] = '1' + pair[1]
        heapq.heappush(heap, [lo[0] + hi[0]] + lo[1:] + hi[1:])
    return heap[0]

def generate_codes(tree):
    code = ""
    codes = {}
    for symbol, path in tree:
        if path:
            codes[symbol] = code
            generate_codes([symbol, path])
            code = code[:-1]
        else:
            code += "0"
            codes[symbol] = code
            code = ""
    return codes

def compress_data(data, codes):
    return ''.join([codes[c] for c in data])

def decompress_data(compressed_data, codes):
    reverse_codes = {v: k for k, v in codes.items()}
    current_code = ""
    decompressed_data = ""
    for bit in compressed_data:
        current_code += bit
        if current_code in reverse_codes:
            decompressed_data += reverse_codes[current_code]
            current_code = ""
    return decompressed_data
```

```python
import numpy as np

def linear_regression(X, y, alpha, epochs):
    m = X.shape[0]
    theta = np.zeros(X.shape[1])
    X = np.concatenate(([np.ones((m, 1))], X), axis=1)
    for _ in range(epochs):
        y_pred = X @ theta
        error = y - y_pred
        theta -= alpha * X.T @ error / m
    return theta

def predict(X, theta):
    X = np.concatenate(([np.ones((X.shape[0], 1))], X), axis=1)
    return X @ theta
```
