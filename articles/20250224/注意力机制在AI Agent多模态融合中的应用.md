  def attention(query, key, value):
      # 计算注意力权重
      scores = torch.einsum('bdqd,bdkv->bqkv', query, key)
      scores = scores.softmax(dim=-1)
      # 加权求和
      output = torch.einsum('bqkv,bdkv->bqdv', scores, value)
      return output

      # 计算注意力权重

      scores = scores.softmax(dim=-1)
      # 加权求和
      output = torch.einsum('bqkv,bdkv->bqdv', scores, value)
      return output

  # 示例数据
