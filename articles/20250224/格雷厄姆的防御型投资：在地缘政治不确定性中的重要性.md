- 4.1.2 Python代码实现
```python
import numpy as np
from scipy.optimize import minimize

def portfolio_optimization(returns, risks):
    n = len(returns)
    # 定义目标函数
    def objective(weights):
        return np.dot(weights.T, np.dot(returns, weights))
    
    # 定义约束条件
    constraints = [
        {'type': 'eq', 'fun': lambda w: np.sum(w) - 1},
        {'type': 'ineq', 'fun': lambda w: w[i] >= 0 for i in range(n)}
    ]
    
    # 初始化权重
    initial_weights = np.array([1.0/n]*n)
    
    # 求解优化问题
    result = minimize(objective, initial_weights, constraints=constraints)
    return result.x

# 示例数据
returns = np.array([0.1, 0.12, 0.08])
risks = np.array([0.2, 0.15, 0.3])
weights = portfolio_optimization(returns, risks)
print(weights)
```
