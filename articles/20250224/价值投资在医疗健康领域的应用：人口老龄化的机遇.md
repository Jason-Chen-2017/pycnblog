   ```python
   import pandas as pd
   import numpy as np
   import matplotlib.pyplot as plt

   # 数据加载与预处理
   df = pd.read_csv('medical_health_data.csv')
   df = df.dropna()
   df = df[~df['profit'].isin([0])]

   # 计算市盈率
   def calculate_pe_ratio(price, profit):
       return price / profit

   pe_ratio = df['price'].apply(lambda x: calculate_pe_ratio(x, df['profit']))

   # 筛选低市盈率股票
   value_stocks = df[pe_ratio < np.mean(pe_ratio)]

   # 可视化
   value_stocks[['price', 'profit']].plot(kind='bar', figsize=(10,6))
   plt.title('Value Stocks Performance')
   plt.xlabel('Stocks')
   plt.ylabel('Value')
   plt.show()
   ```
