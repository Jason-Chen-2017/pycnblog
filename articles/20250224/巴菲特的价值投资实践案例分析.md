```python
import pandas as pd
import numpy as np
from datetime import datetime

# 示例代码：计算内在价值
def calculate_intrinsic_value(free_cash_flow, discount_rate, growth_rate):
    return free_cash_flow / (discount_rate - growth_rate)

# 示例代码：计算安全边际
def calculate_margin_of_safety(intrinsic_value, market_price):
    return intrinsic_value - market_price

# 数据获取与处理
data = pd.read_csv('stock_data.csv')
data['date'] = pd.to_datetime(data['date'])
data.set_index('date', inplace=True)

# 示例计算
fcf = data['free_cash_flow'].iloc[-4:]
discount_rate = 0.1
growth_rate = 0.05
intrinsic_value = calculate_intrinsic_value(fcf.mean(), discount_rate, growth_rate)
market_price = data['close'].iloc[-1]
margin_of_safety = calculate_margin_of_safety(intrinsic_value, market_price)

print(f"内值: {intrinsic_value:.2f}")
print(f"市场价格: {market_price:.2f}")
print(f"安全边际: {margin_of_safety:.2f}")
```
