import numpy as np

# 定义层次分析法计算权重
def calculate_weights(matrix):
    # 计算权重向量
    eigenvalues, eigenvectors = np.linalg.eig(matrix.T @ matrix)
    weights = eigenvectors[:, 0] / np.sum(eigenvectors[:, 0])
    return weights

# 示例数据矩阵
matrix = np.array([
    [1, 3, 2],
    [2, 1, 4],
    [3, 2, 1]
])

weights = calculate_weights(matrix)
print(weights)
