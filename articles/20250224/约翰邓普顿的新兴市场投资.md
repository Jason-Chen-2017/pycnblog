  returns = data.pct_change().dropna()

  # 使用均值-方差优化
  mu = returns.mean()
  sigma = returns.cov()

  # 计算最优投资组合
  inv = np.linalg.inv(sigma)
  w = inv @ mu
  w = w / w.sum()

  print("最优权重:", w)
