    # 数据预处理

    from sklearn.ensemble import RandomForestRegressor
    # 示例数据加载与处理
    data = pd.read_csv('data.csv')
    features = data[['age', 'income', 'expenses']]
    target = data['gap']
    # 模型训练
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(features, target)
    # 预测
    new_data = pd.DataFrame({'age': 60, 'income': 50000, 'expenses': 30000}, index=[0])
    prediction = model.predict(new_data)
    print(prediction)
  ```

    @app.route('/api/predict', methods=['GET'])
    def predict():
        age = request.args.get('age', type=int)
        income = request.args.get('income', type=int)
        expenses = request.args.get('expenses', type=int)
        prediction = model.predict([[age, income, expenses]])
        return jsonify({'predicted_gap': prediction[0].tolist()})
