```python
import torch

def attention(query, key, value):
    # 点积注意力机制
    scores = torch.bmm(query, key.transpose(-2, -1))
    scores = torch.softmax(scores, dim=-1)
    output = torch.bmm(scores, value)
    return output
```

### 5.2.2 系统功能实现
```python
class AI-Agent:
    def __init__(self):
        self.input = None
        self.output = None
        self.attention_weights = None
    
    def process_input(self, input_sequence):
        # 输入处理
        self.input = input_sequence
    
    def compute_attention(self):
        # 注意力计算
        self.attention_weights = self.input @ self.input.transpose(-2, -1)
        self.attention_weights = torch.softmax(self.attention_weights, dim=-1)
    
    def generate_output(self):
        # 输出生成
        self.output = self.attention_weights @ self.input
        return self.output
```
