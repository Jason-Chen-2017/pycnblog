```python
import pandas as pd
from sklearn.model import LinearRegression
from sklearn.metrics import mean_squared_error

def load_data():
    # 数据加载函数
    data = pd.read_csv('financial_data.csv')
    return data

def preprocess_data(data):
    # 数据预处理函数
    data = data.dropna()
    return data

def train_model(data):
    # 模型训练函数
    X = data[['market_rate', 'customer_score']]
    y = data['price']
    model = LinearRegression()
    model.fit(X, y)
    return model

def optimize_price(model, data):
    # 定价优化函数
    predicted_prices = model.predict(data[['market_rate', 'customer_score']])
    mse = mean_squared_error(data['price'], predicted_prices)
    return predicted_prices, mse

if __name__ == '__main__':
    data = load_data()
    processed_data = preprocess_data(data)
    model = train_model(processed_data)
    optimized_prices, mse = optimize_price(model, processed_data)
    print("优化后的价格:", optimized_prices)
    print("均方误差:", mse)
```
