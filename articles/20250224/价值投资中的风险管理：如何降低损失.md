### 5.2.2 风险评估模块
```python
def calculate_vaR(data, confidence_level=0.95):
    # 计算收益
    returns = data.pct_change().dropna()
    # 计算波动率
    volatility = returns.std()
    # 计算 VaR
    VaR = norm.ppf(1 - confidence_level) * volatility
    return VaR

def calculate_max_drawdown(data):
    # 计算累积收益
    cum_return = (1 + data.pct_change()).cumprod()
    # 计算最大回撤
    max_drawdown = (cum_return.cummax() - cum_return).max()
    return max_drawdown
```

### 5.2.3 投资策略制定模块
```python
import numpy as np

def optimal_portfolio(returns, risk_free_rate=0.02):
    # 计算各资产的协方差矩阵
    covariance_matrix = returns.cov()
    # 计算各资产的预期收益
    expected_returns = returns.mean()
    # 使用拉格朗日乘数法求解最优投资组合
    # 约束条件：sum(w_i) = 1，sum(w_i * r_i) - risk_free_rate = 0
    n = returns.shape[1]
    A = np.array([np.ones(n), np.array(expected_returns)])
    b = np.array([1, risk_free_rate])
    # 求解优化问题
    w = np.linalg.lstsq(A.T @ A, A.T @ b, rcond=None)[0]
    return w
```
