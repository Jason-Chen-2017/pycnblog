```python
import pandas as pd
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

# 加载数据集
data = pd.read_csv("cases.csv")
data["description"] = data["description"].apply(lambda x: x.lower())

# 文本向量化
vectorizer = TfidfVectorizer()
vectors = vectorizer.fit_transform(data["description"])

# 搜索相似案例
def search_cases(query):
    query_vector = vectorizer.transform([query.lower()])
    similarity_scores = cosine_similarity(query_vector, vectors)
    top_cases = data.iloc[similarity_scores.argsort()[0][-10:]]
    return top_cases

# 测试
query = "版权侵权"
cases = search_cases(query)
print(cases)
```
