n_layers = 20
n_units = 1024
dropout_rate = 0.5

# WaveNet 模型定义
def WaveNet(inputs, training=True):
    x = inputs
    for i in range(n_layers):
        x = tf.layers.dense(x, n_units, activation=tf.nn.tanh, name='dense_{}'.format(i))
