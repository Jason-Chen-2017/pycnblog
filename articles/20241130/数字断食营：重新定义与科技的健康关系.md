```python
import time

def reduce_usage(usage_time, reduction):
    while usage_time > reduction:
        usage_time -= reduction
        time.sleep(1)  # 每次减少1小时
        print(f"当前使用时间：{usage_time}小时")
    print("目标使用时间已达到！")
    
# 假设当前使用时间为6小时，目标使用时间为3小时

target_usage_time = 3
```
