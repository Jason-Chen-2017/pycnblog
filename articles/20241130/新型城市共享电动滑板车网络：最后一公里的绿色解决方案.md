import heapq
import math

# 车辆调度算法示例
def vehicle_scheduling(requests, vehicles):
    # 创建一个优先队列，用于存储待调度的车辆
    schedule_queue = []
    for request in requests:
        # 计算每个请求的优先级（目标位置到当前车辆位置的曼哈顿距离）
        priority = abs(request['destination_x'] - request['current_x']) + abs(request['destination_y'] - request['current_y'])
        # 将请求加入优先队列
        heapq.heappush(schedule_queue, (priority, request))
    
    # 调度车辆
    scheduled_vehicles = []
    while schedule_queue:
        _, request = heapq.heappop(schedule_queue)
        # 分配车辆
        vehicle = find_vehicle(vehicles)
        if vehicle:
            scheduled_vehicles.append((vehicle, request))
            # 更新车辆位置
            vehicle['current_x'] = request['destination_x']
            vehicle['current_y'] = request['destination_y']
    
    return scheduled_vehicles

# 找到可用车辆
def find_vehicle(vehicles):
    for vehicle in vehicles:
        if vehicle['available']:
            vehicle['available'] = False
            return vehicle
    return None

# 示例数据
