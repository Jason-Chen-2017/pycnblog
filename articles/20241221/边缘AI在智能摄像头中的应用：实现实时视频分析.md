```python
    compressed_frame = cv2.resize(frame, (320, 240))

    # 将图像输入到网络进行目标检测
    blob = cv2.dnn.blobFromImage(compressed_frame, 1.0, (320, 240), (104.0, 177.0, 123.0))
    net.setInput(blob)
    detections = net.forward()

    # 遍历检测结果
    for i in range(detections.shape[2]):
        confidence = detections[0, 0, i, 2]
        if confidence > 0.5:
            class_id = int(detections[0, 0, i, 1])
            class_name = classes[class_id]
            x_min = int(detections[0, 0, i, 3] * compressed_frame.shape[1])
            y_min = int(detections[0, 0, i, 4] * compressed_frame.shape[0])
            x_max = int(detections[0, 0, i, 5] * compressed_frame.shape[1])
            y_max = int(detections[0, 0, i, 6] * compressed_frame.shape[0])
            cv2.rectangle(compressed_frame, (x_min, y_min), (x_max, y_max), (0, 255, 0), 2)
            cv2.putText(compressed_frame, class_name + " {:.2f}%".format(confidence * 100), (x_min, y_min - 10), cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 2)

    # 显示检测结果
    cv2.imshow('Object Detection', compressed_frame)

    # 按下ESC键退出
    if cv2.waitKey(1) & 0xFF == 27:
        break

# 释放资源
cap.release()
cv2.destroyAllWindows()
```
