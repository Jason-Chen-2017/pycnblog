#### Python Code Snippet

Here's a simplified Python code snippet that demonstrates the resource allocation algorithm:

```python
# Import necessary libraries
import numpy as np

# Define the function for resource allocation
def allocate_resources(demand, availability):
    """
    Allocate resources based on demand and availability.
    
    :param demand: List of demand values for each tool.
    :param availability: List of available tool quantities.
    :return: Allocated resources for each community.
    """
    allocation = np.zeros(len(demand))
    for i in range(len(demand)):
        if availability[i] >= demand[i]:
            allocation[i] = demand[i]
            availability[i] -= demand[i]
        else:
            allocation[i] = availability[i]
            demand[i] -= availability[i]
    return allocation

# Example demand and availability lists
demand = [10, 20, 5, 15]
availability = [12, 18, 7, 10]

# Allocate resources
allocated_resources = allocate_resources(demand, availability)

# Print the result
print("Allocated resources:", allocated_resources)
```

#### Mathematical Models and Formulas

The resource allocation algorithm can be further understood through the lens of mathematical optimization. The objective function aims to minimize the difference between demand and availability:

$$\min_{x} \sum_{i=1}^{n} (d_i - a_i - x_i)^2$$

where:

- \(d_i\) is the demand for the \(i\)-th tool.
- \(a_i\) is the availability of the \(i\)-th tool.
- \(x_i\) is the allocated quantity of the \(i\)-th tool.

#### Example

Consider a scenario where there are four tools (A, B, C, D) with respective demands of [10, 20, 5, 15] and availabilities of [12, 18, 7, 10]. Using the allocation algorithm, we get the following allocated resources:

- Tool A: 10 (demand) - 12 (availability) = 0
- Tool B: 20 (demand) - 18 (availability) = 2
- Tool C: 5 (demand) - 7 (availability) = 0
- Tool D: 15 (demand) - 10 (availability) = 5

The resulting allocation is [0, 2, 0, 5], indicating that Tool A and C are fully available, Tool B is slightly underutilized, and Tool D is fully allocated.

By leveraging these algorithms, the New Urban Tool Sharing Alliance ensures that resources are efficiently allocated to meet the demands of urban communities, thereby maximizing resource utilization and enhancing the overall user experience.

```python
# views.py

from django.contrib.auth.models import User
from django.contrib.auth.hashers import make_password
from rest_framework.response import Response
from rest_framework import status

def register_user(request):
    data = request.data
    username = data.get('username')
    email = data.get('email')
    password = make_password(data.get('password'))

    if User.objects.filter(username=username).exists():
        return Response({'error': 'Username already exists'}, status=status.HTTP_400_BAD_REQUEST)
    if User.objects.filter(email=email).exists():
        return Response({'error': 'Email already exists'}, status=status.HTTP_400_BAD_REQUEST)

    new_user = User.objects.create_user(username=username, email=email, password=password)
    new_user.save()

    return Response({'message': 'User registered successfully'}, status=status.HTTP_201_CREATED)
```

**Booking and Payment Processing:**

```python
# views.py

from rest_framework import status
from rest_framework.response import Response
from .models import Booking, Resource
from .utils import process_payment

def book_resource(request):
    data = request.data
    user_id = data.get('user_id')
    resource_id = data.get('resource_id')
    start_time = data.get('start_time')
    end_time = data.get('end_time')

    user = User.objects.get(id=user_id)
    resource = Resource.objects.get(id=resource_id)

    if not resource.availability_status:
        return Response({'error': 'Resource is not available'}, status=status.HTTP_400_BAD_REQUEST)

    booking = Booking.objects.create(user=user, resource=resource, start_time=start_time, end_time=end_time)
    booking.save()

    payment_details = {
        'amount': resource.booking_price,
        'start_time': start_time,
        'end_time': end_time,
        'user_id': user_id,
        'resource_id': resource_id
    }
    process_payment(payment_details)

    return Response({'message': 'Resource booked successfully'}, status=status.HTTP_201_CREATED)
```
