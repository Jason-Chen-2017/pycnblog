**1.2 区块链的核心技术**

区块链的核心技术主要包括以下几方面：

- **分布式账本**：区块链是一种去中心化的数据库，数据存储在多个节点上，每个节点都保存一份完整的数据副本，从而确保了数据的安全性和一致性。

- **加密算法**：区块链使用非对称加密算法来保证数据的隐私性和安全性。每个用户都有一对密钥，公钥用于加密数据，私钥用于解密数据。

- **共识机制**：共识机制是区块链网络中节点之间达成一致性的算法。常见的共识机制包括工作量证明（PoW）、权益证明（PoS）等。

在`contracts/`文件夹中，创建一个名为`ArtAuction.sol`的文件，编写以下智能合约代码：

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract ArtAuction {
    address public owner;
    address public artist;
    string public artworkTitle;
    string public artworkDescription;
    uint256 public startPrice;
    uint256 public reservePrice;
    uint256 public auctionEnd;
    address public highestBidder;
    uint256 public highestBid;
    bool public auctionEnded;

    constructor(
        string memory _artworkTitle,
        string memory _artworkDescription,
        uint256 _startPrice,
        uint256 _reservePrice,
        uint256 _auctionEnd
    ) {
        owner = msg.sender;
        artist = msg.sender;
        artworkTitle = _artworkTitle;
        artworkDescription = _artworkDescription;
        startPrice = _startPrice;
        reservePrice = _reservePrice;
        auctionEnd = _auctionEnd;
        auctionEnded = false;
    }

    function placeBid() public payable {
        require(!auctionEnded, "Auction has ended");
        require(msg.value > highestBid, "Bid too low");
        if (highestBidder != address(0)) {
            payable(highestBidder).transfer(highestBid);
        }
        highestBidder = msg.sender;
        highestBid = msg.value;
    }

    function endAuction() public {
        require(msg.sender == owner, "Only owner can end the auction");
        require(block.timestamp >= auctionEnd, "Auction has not ended yet");
        auctionEnded = true;
        if (highestBid < reservePrice) {
            revert("Reserve price not met");
        }
        payable(artist).transfer(highestBid);
    }
}
```
