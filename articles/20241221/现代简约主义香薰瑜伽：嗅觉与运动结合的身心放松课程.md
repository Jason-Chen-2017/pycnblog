##### 6.2.2 数学模型实现

```python
import numpy as np

def calculate_body_impact(scent, concentration, body_status, environmental_factors):
    # 香薰成分影响系数
    scent_impact = {
        '薰衣草': 0.8,
        '迷迭香': 0.6,
        '洋甘菊': 0.7,
        '薄荷': 0.9
    }
    
    # 身体状态影响系数
    body_status_impact = {
        '良好': 1.0,
        '一般': 0.8,
        '较差': 0.6
    }
    
    # 环境因素影响系数
    environmental_impact = {
        '良好': 1.0,
        '一般': 0.8,
        '较差': 0.6
    }
    
    # 计算香薰成分对身心放松的影响
    scent_effect = scent_impact[scent] * concentration
    
    # 计算身体状态对身心放松的影响
    body_effect = body_status_impact[body_status]
    
    # 计算环境因素对身心放松的影响
    env_effect = environmental_impact[environmental_factors]
    
    # 计算总影响
    total_impact = scent_effect * body_effect * env_effect
    
    return total_impact

# 示例
scent = '薰衣草'
concentration = 0.1
body_status = '良好'
environmental_factors = '良好'

impact = calculate_body_impact(scent, concentration, body_status, environmental_factors)
print(f'身心放松影响值：{impact}')
```

在上面的代码中，`calculate_body_impact`函数通过一系列的系数，计算香薰成分、身体状态和环境因素对身心放松的影响。
