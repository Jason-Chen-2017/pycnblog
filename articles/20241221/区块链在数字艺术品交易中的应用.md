### 1.1 区块链的定义与特点

#### 1.1.1 区块链的概念

区块链是一种分布式数据库技术，其基本原理是利用块链结构存储数据，每个区块包含一定数量的交易记录，并通过密码学确保数据的完整性和不可篡改性。区块链的特点主要包括去中心化、安全性高、透明度高和不可篡改性。

- **去中心化**：区块链不存在中心化的管理节点，数据存储在所有参与节点的分布式账本中，因此系统具有较高的容错性和抗攻击能力。
- **安全性高**：区块链采用密码学算法确保数据的安全性和完整性，如哈希函数、非同质化代币（NFT）和智能合约等。
- **透明度高**：区块链的所有交易记录都是公开的，任何人都可以查看，提高了交易的透明度。
- **不可篡改性**：区块链的数据一旦被写入，就不可篡改，确保了数据的历史可追溯性。

#### 1.1.2 区块链的核心特点

区块链的核心特点主要体现在以下几个方面：

- **分布式存储**：区块链的数据存储在所有参与节点的分布式账本中，避免了单点故障和数据丢失的风险。
- **共识机制**：区块链采用共识机制确保所有节点对数据的共识，如工作量证明（Proof of Work, PoW）、权益证明（Proof of Stake, PoS）等。
- **智能合约**：智能合约是一种自动执行合约条款的计算机程序，能够实现自动化交易和执行。

区块链的技术架构主要包括以下几个层次：

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

// NFT合约
contract NFT {
    // NFT名称
    string public name;
    // NFT符号
    string public symbol;
    // 总供应量
    uint256 public totalSupply;
    // 创建者地址
    address public creator;
    // 所有者映射
    mapping(uint256 => address) private _owners;
    // 持有者数量映射
    mapping(address => uint256) private _balances;

    // 构造函数
    constructor(string memory _name, string memory _symbol) {
        name = _name;
        symbol = _symbol;
        totalSupply = 1;
        creator = msg.sender;
        _owners[1] = creator;
        _balances[creator] = 1;
    }

    // 创建NFT
    function createNFT() public {
        require(totalSupply < 100, "NFT数量已达到上限");
        totalSupply++;
        _owners[totalSupply] = msg.sender;
        _balances[msg.sender]++;
    }

    // 转移NFT
    function transferNFT(uint256 _tokenId, address _to) public {
        require(_owners[_tokenId] == msg.sender, "您不是NFT的所有者");
        _owners[_tokenId] = _to;
        _balances[msg.sender]--;
        _balances[_to]++;
    }

    // 查询NFT所有者
    function getNFTOwner(uint256 _tokenId) public view returns (address) {
        return _owners[_tokenId];
    }

    // 查询NFT余额
    function getBalance(address _owner) public view returns (uint256) {
        return _balances[_owner];
    }
}
```

```solidity
// 拍卖合约
contract Auction {
    // 艺术品所有者
    address public artist;
    // 艺术品NFT地址
    address public nftAddress;
    // 拍卖开始时间
    uint256 public startTime;
    // 拍卖结束时间
    uint256 public endTime;
    // 当前最高出价
    uint256 public highestBid;
    // 最高出价者地址
    address public highestBidder;

    // 构造函数
    constructor(address _artist, address _nftAddress, uint256 _startTime, uint256 _endTime) {
        artist = _artist;
        nftAddress = _nftAddress;
        startTime = _startTime;
        endTime = _endTime;
    }

    // 出价函数
    function bid() public payable {
        require(block.timestamp >= startTime && block.timestamp <= endTime, "拍卖尚未开始或已结束");
        require(msg.value > highestBid, "出价太低");

        if (highestBidder != address(0)) {
            // 返还上一出价者的出价
            payable(highestBidder).transfer(highestBid);
        }

        highestBid = msg.value;
        highestBidder = msg.sender;
    }

    // 结束拍卖函数
    function endAuction() public {
        require(msg.sender == artist || block.timestamp > endTime, "只有艺术家或拍卖结束前可以结束拍卖");
        
        if (highestBidder != address(0)) {
            // 转移艺术品所有权
            IERC721 nft = IERC721(nftAddress);
            nft.transferFrom(artist, highestBidder, 1);
        }
    }
}
```

该智能合约实现了以下功能：

3. **返还出价**：如果用户出价失败，会返还之前的出价。
4. **结束拍卖**：艺术家或拍卖结束前可以调用`endAuction()`函数结束拍卖，并将艺术品所有权转移给最高出价者。
