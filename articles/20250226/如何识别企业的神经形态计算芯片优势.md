import numpy as np

class Neuron:
    def __init__(self, threshold=1.0):
        self.threshold = threshold
        self.current = 0.0
        self.pulse = 0

    def activate(self, input_current):
        self.current += input_current
        if self.current >= self.threshold:
            self.pulse = 1
            self.current = 0.0
        return self.pulse

class Synapse:
    def __init__(self, weight=0.1):
        self.weight = weight

    def forward(self, pulse_input):
        return pulse_input * self.weight

# 示例网络
neuron = Neuron(threshold=1.5)
synapse = Synapse(weight=0.8)

input_signal = 0.5
output_pulse = neuron.activate(input_signal)
output_signal = synapse.forward(output_pulse)

print("输出信号:", output_signal)
```

- **网络处理流程**：输入信号经过神经元处理后，通过突触传递到下一个神经元。
