# 示例代码：AI Agent的核心逻辑
import numpy as np
from sklearn.neighbors import NearestNeighbors

class AIAgent:
    def __init__(self, 食材特征):
        self.model = NearestNeighbors(n_neighbors=3).fit(食材特征)

    def 推荐食谱(self, query):
        neighbors = self.model.kneighbors(query, return_distance=False)
        return 食材特征.iloc[neighbors[0]]

# 初始化并使用AI Agent
食材特征 = ... # 加载食材特征数据
