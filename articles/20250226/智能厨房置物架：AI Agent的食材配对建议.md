```python
def collaborative_filtering(users, user):
    # 计算用户与其他用户的相似性
    similarities = {}
    for u in users:
        if u != user:
            similarities[u] = cosine_similarity(users[user], users[u])
    # 根据相似性排序并推荐
    recommendations = []
    for u in sorted(similarities, key=lambda x: similarities[x], reverse=True):
        recommendations.extend(users[u])
    return recommendations
```
