```python
import numpy as np

# 数据加载
data = pd.read_csv('customer_data.csv')

# 数据清洗
data = data.dropna()
data = data.replace({'gender': {'M': 0, 'F': 1}})

# 数据标准化
from sklearn.preprocessing import StandardScaler
scaler = StandardScaler()
scaled_data = scaler.fit_transform(data[['age', 'income', 'transaction_amount']])
```

#### 5.2.2 模型训练代码
```python
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score

# 模型训练
model = RandomForestClassifier(n_estimators=100, max_depth=4)
model.fit(scaled_data, data['target'])

# 模型预测
predicted = model.predict(scaled_data)
print("Accuracy:", accuracy_score(data['target'], predicted))
```
