```python
import numpy as np
import matplotlib.pyplot as plt

def model_function(S0, α, β, t):
    S = S0 * np.exp((α - β * S0) * t)
    return S

# 示例运行
S0 = 100
α = 0.05
β = 0.001
t = np.linspace(0, 10, 100)
S = model_function(S0, α, β, t)

plt.plot(t, S)
plt.xlabel('时间（年）')
plt.ylabel('资产价值')
plt.show()
```
