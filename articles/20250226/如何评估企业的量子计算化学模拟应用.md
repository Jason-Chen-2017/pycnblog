  ```python
  from qiskit.circuit import Hadamard, CNOT

  def quantum_fourier_transform(n):
      qc = QuantumCircuit(n)
      for i in range(n):
          qc.apply(Hadamard(), i)
          for j in range(i+1, n):
              qc.apply(CNOT(), j, i)
      return qc
  ```

  ```python
      qc.apply(Hadamard(), 0)
      # 应用条件门
      for i in range(1, n):
          qc.apply(CNOT(), i, 0)
      return qc
  ```
