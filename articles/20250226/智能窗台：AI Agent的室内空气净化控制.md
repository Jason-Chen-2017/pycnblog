```python
import numpy as np
import time

class AirQualitySensor:
    def read_data(self):
        return np.random.uniform(0, 100)

class AIAgent:
    def __init__(self, sensor):
        self.sensor = sensor

    def analyze_data(self, data):
        if data > 70:
            return 'high'
        elif data > 40:
            return 'medium'
        else:
            return 'low'

    def execute_mode(self, mode):
        print(f'执行{mode}模式')

# 初始化系统
sensor = AirQualitySensor()
agent = AIAgent(sensor)

# 获取数据并执行净化
data = sensor.read_data()
mode = agent.analyze_data(data)
agent.execute_mode(mode)
```
