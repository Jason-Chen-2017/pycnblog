```python
import numpy as np
import gym

class WaterResourceEnv(gym.Env):
    def __init__(self):
        # 定义状态和动作空间
        self.observation_space = gym.spaces.Box(low=0, high=100, shape=(1,))
        self.action_space = gym.spaces.Box(low=0, high=10, shape=(1,))
    
    def step(self, action):
        # 状态转移和奖励计算
        reward = action[0] * 0.5  # 示例奖励函数
        next_state = self.state + action[0]
        self.state = next_state
        return next_state, reward, False, {}
    
    def reset(self):
        self.state = np.random.uniform(low=0, high=50)
        return self.state
```
