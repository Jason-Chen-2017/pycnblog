```python
def generate_question(model, tokenizer, max_length=50):
    input_ids = tokenizer.encode("Generate a financial question:", add_special_tokens=True)
    input_ids = input_ids + [tokenizer.eos_token_id]
    input_ids = torch.tensor(input_ids, dtype=torch.long).unsqueeze(0)
    output = model.generate(
        input_ids=input_ids,
        max_length=max_length,
        temperature=0.7,
        top_k=50
    )
    question = tokenizer.decode(output[0], skip_special_tokens=True)
    return question
```
#### 6.2.4 评分函数实现
```python
def score_question(question, answer, model, tokenizer):
    input = f"Question: {question}\nAnswer: {answer}"
    inputs = tokenizer.encode_plus(input, return_tensors='pt', max_length=100, truncation=True)
    outputs = model.generate(input_ids=inputs['input_ids'])
    prediction = tokenizer.decode(outputs[0], skip_special_tokens=True)
    return prediction == answer
```
