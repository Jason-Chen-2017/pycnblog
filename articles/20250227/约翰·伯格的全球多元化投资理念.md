### 附录A：投资组合优化的Python代码实现

```python
import numpy as np
import pandas as pd
from scipy.optimize import minimize

def portfolio Optimization(returns, weights):
    n = len(weights)
    # 定义目标函数（最小化风险）
    def objective(weights):
        return np.dot(weights.T, np.dot(returns, weights))
    
    # 约束条件：权重之和为1，且非负
    constraints = [
        {'type': 'eq', 'fun': lambda w: np.sum(w) - 1},
        {'type': 'ineq', 'fun': lambda w: w[i] >= 0 for i in range(n)}
    ]
    
    # 使用最小二乘法优化
    result = minimize(objective, np.array(weights), constraints=constraints)
    return result.x

# 示例数据
returns = np.array([[0.1, 0.2], [0.3, 0.1]])
weights = np.array([0.5, 0.5])
optimized_weights = portfolio_Optimization(returns, weights)
print("Optimized weights:", optimized_weights)
```
