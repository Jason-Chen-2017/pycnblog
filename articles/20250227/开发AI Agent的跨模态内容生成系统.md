### 3.3.2 算法实现的代码示例
以下是一个多模态生成的代码示例：

```python
import torch
import torch.nn as nn

class MultiModalGenerator(nn.Module):
    def __init__(self, input_size, hidden_size):
        super(MultiModalGenerator, self).__init__()
        self.fc1 = nn.Linear(input_size, hidden_size)
        self.fc2 = nn.Linear(hidden_size, input_size)
    
    def forward(self, x1, x2):
        x = torch.cat((x1, x2), dim=1)
        x = self.fc1(x)
        x = self.fc2(x)
        return x

# 示例数据
input_size = 2
hidden_size = 4
batch_size = 32

generator = MultiModalGenerator(input_size, hidden_size)
x1 = torch.randn(batch_size, input_size//2)
x2 = torch.randn(batch_size, input_size//2)
output = generator(x1, x2)
print(output.size())
```
