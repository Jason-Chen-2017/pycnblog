import pandas as pd
import numpy as np

def calculate_intrinsic_value(cash_flow, discount_rate, growth_rate):
    return cash_flow / (discount_rate - growth_rate)

# 示例数据
data = {
    '公司': ['Company A', 'Company B', 'Company C'],
    '现金流': [100, 150, 200],
    '折现率': [0.1, 0.12, 0.08],
    '增长率': [0.05, 0.03, 0.04]
}

df = pd.DataFrame(data)

# 计算内在价值
intrinsic_values = []
for index, row in df.iterrows():
    iv = calculate_intrinsic_value(row['现金流'], row['折现率'], row['增长率'])
    intrinsic_values.append(iv)

df['内在价值'] = intrinsic_values
print(df)
