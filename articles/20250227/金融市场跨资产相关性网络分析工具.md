### 附录：相关性网络分析工具的Python代码实现

```python
import numpy as np
import networkx as nx
import matplotlib.pyplot as plt

# 示例数据：资产回报矩阵
def generate_asset_returns(n_assets, n_samples):
    return np.random.randn(n_samples, n_assets)

# 计算相关性矩阵
def calculate_correlation_matrix(returns):
    return np.corrcoef(returns.T)

# 构建资产网络
def build_asset_network(correlation_matrix, threshold=0.5):
    n_assets = correlation_matrix.shape[0]
    G = nx.Graph()
    for i in range(n_assets):
        for j in range(i+1, n_assets):
            if correlation_matrix[i, j] > threshold:
                G.add_edge(i, j, weight=correlation_matrix[i, j])
    return G

# 可视化网络
def visualize_network(G):
    plt.figure(figsize=(10, 10))
    nx.draw(G, node_size=300, node_color='red', edge_color='blue', linewidth=1, font_size=10)
    plt.title("Asset Correlation Network")
    plt.show()

# 示例运行
if __name__ == "__main__":
    n_assets = 10
    n_samples = 100
    returns = generate_asset_returns(n_assets, n_samples)
    correlation_matrix = calculate_correlation_matrix(returns)
    G = build_asset_network(correlation_matrix, threshold=0.3)
    visualize_network(G)
```
