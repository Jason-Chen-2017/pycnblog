import numpy as np
from scipy.optimize import minimize

def portfolio_optimization(returns, cov_matrix, risk_aversion):
    n = len(returns)
    # 定义目标函数
    def objective(weights):
        return (weights.T @ cov_matrix @ weights)[0,0] / 2
    # 定义约束条件
    constraints = [
        ('sum_eq', weights @ np.ones(n) - 1 == 0)
    ]
    # 求解优化问题
    result = minimize(objective, np.ones(n)/n, method='SLSQP', 
                     constraints=constraints)
    return result.x

# 示例数据
returns = np.array([0.1, 0.2, 0.15])
cov_matrix = np.array([[0.02, 0.01, 0.01],
                        [0.01, 0.03, 0.02],
                        [0.01, 0.02, 0.03]])
weights = portfolio_optimization(returns, cov_matrix, 1)
print(weights)
```
