import pandas as pd

def calculate_owner_profit(net_profit, dividend):
    return net_profit - dividend

# 示例数据
data = {
    '净利润': [1000000],
    '股东分红': [200000]
}
df = pd.DataFrame(data)
owner_profit = calculate_owner_profit(df['净利润'].values[0], df['股东分红'].values[0])
print(f"所有者收益为：{owner_profit}元")
