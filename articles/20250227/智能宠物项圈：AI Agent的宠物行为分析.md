```python
import numpy as np
import pandas as pd

# 示例代码：从传感器数据中提取特征
def extract_features(data):
    features = []
    for window in data:
        # 提取加速度特征
        acc_features = np.mean(window['acceleration'])
        # 提取心率特征
        heart_rate_features = np.mean(window['heart_rate'])
        features.append([acc_features, heart_rate_features])
    return features
```
