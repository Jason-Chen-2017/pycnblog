```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import pandas_datareader as pdr

# 加载数据
def load_data(tickers, start_date, end_date):
    data = pdr.get_data_yahoo(tickers, start=start_date, end=end_date)
    returns = data['Adj Close'].pct_change().dropna()
    return returns

# 示例代码：加载苹果、微软、谷歌的回报数据

```

#### 6.2.2 计算风险调整后的收益

```python
def compute_adjusted_return(returns, risk_free_rate=0.02):
    # 计算平均回报
    mean_return = returns.mean() * 100
    # 计算标准差
    std_deviation = returns.std() * 100
    # 计算风险调整后的收益
    adjusted_return = mean_return - 3 * std_deviation
    return adjusted_return

# 示例代码：计算各资产的风险调整后收益
risk_free_rate = 0.02
adjusted_returns = compute_adjusted_return(returns)
```
