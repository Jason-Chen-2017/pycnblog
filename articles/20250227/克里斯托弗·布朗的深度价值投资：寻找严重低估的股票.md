```python
import pandas as pd
import numpy as np

def calculate_intrinsic_value(revenue, net_income, growth_rate, discount_rate):
    # 现金流折现模型（DCF）
    present_value = net_income / (1 + discount_rate)
    while revenue > 0:
        revenue *= (1 + growth_rate)
        present_value += revenue / (1 + discount_rate)
    return present_value
```
