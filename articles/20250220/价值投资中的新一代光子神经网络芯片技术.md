#### 6.2.1 光子神经网络芯片的核心代码
```python
import numpy as np

def activate(x):
    return np.tanh(x)

class PhotonNeuralNetwork:
    def __init__(self, input_dim, hidden_dim, output_dim):
        self.W1 = np.random.randn(input_dim, hidden_dim)
        self.W2 = np.random.randn(hidden_dim, output_dim)
        self.b1 = np.zeros((1, hidden_dim))
        self.b2 = np.zeros((1, output_dim))

    def forward(self, x):
        self.z1 = np.dot(x, self.W1) + self.b1
        self.a1 = activate(self.z1)
        self.z2 = np.dot(self.a1, self.W2) + self.b2
        self.a2 = activate(self.z2)
        return self.a2

    def backward(self, x, y, learning_rate):
        delta2 = (y - self.a2) * (1 - self.a2**2)
        dW2 = (1/m) * np.dot(self.a1.T, delta2)
        db2 = np.sum(delta2, axis=0)
        delta1 = delta2.dot(self.W2.T) * (1 - self.a1**2)
        dW1 = (1/m) * np.dot(x.T, delta1)
        db1 = np.sum(delta1, axis=0)
        self.W2 += dW2 * learning_rate
        self.b2 += db2 * learning_rate
        self.W1 += dW1 * learning_rate
        self.b1 += db1 * learning_rate

    def train(self, x, y, learning_rate, epochs):
        for _ in range(epochs):
            self.forward(x)
            self.backward(x, y, learning_rate)

# 示例代码
m = 100
X = np.random.randn(m, 100)
y = np.random.randn(m, 1)
pnn = PhotonNeuralNetwork(100, 50, 1)
pnn.train(X, y, 0.1, 100)
print(pnn.forward(X))
```
