import numpy as np
from sklearn.metrics import r2_score

def value_assessment(resource_quality, resource_rarity):
    return 0.6 * resource_quality + 0.4 * resource_rarity

# 示例数据
resource_quality = np.array([80, 90, 75])
resource_rarity = np.array([95, 85, 90])

# 计算经济价值
value = value_assessment(resource_quality, resource_rarity)

# 评估模型的准确性
print("R² 值：", r2_score(resource_rarity, value))
