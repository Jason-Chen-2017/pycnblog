**5.3.1 数据收集与清洗**

```python
import pandas as pd
from sklearn.preprocessing import StandardScaler

# 读取数据
data = pd.read_csv("forest_fire_data.csv")

# 数据清洗
def clean_data(data):
    # 去除重复数据
    data = data.drop_duplicates()
    # 数据格式转换
    data["date"] = pd.to_datetime(data["date"])
    data = data.sort_values("date")
    # 缺失数据填补
    data["temperature"] = data["temperature"].fillna(data["temperature"].mean())
    data["humidity"] = data["humidity"].fillna(data["humidity"].mean())
    return data

cleaned_data = clean_data(data)

# 特征提取
def extract_features(data):
    # 提取统计特征
    data["temp_range"] = data["temperature"].max() - data["temperature"].min()
    data["temp_avg"] = data["temperature"].mean()
    data["temp_std"] = data["temperature"].std()
    data["humidity_range"] = data["humidity"].max() - data["humidity"].min()
    data["humidity_avg"] = data["humidity"].mean()
    data["humidity_std"] = data["humidity"].std()
    # 提取空间特征
    data["longitude_diff"] = data["longitude"].max() - data["longitude"].min()
    data["latitude_diff"] = data["latitude"].max() - data["latitude"].min()
    # 数据标准化
    scaler = StandardScaler()
    features = scaler.fit_transform(data[["temp_range", "temp_avg", "temp_std", "humidity_range", "humidity_avg", "humidity_std", "longitude_diff", "latitude_diff"]])
    return features

features = extract_features(cleaned_data)
```

```python
import nltk
from nltk.tokenize import word_tokenize
from nltk.corpus import stopwords

# 下载并加载停用词列表
nltk.download("stopwords")
stop_words = set(stopwords.words("english"))

# 生成提示词
def generate_lexicon(data, threshold=0.5):
    lexicon = {}
    for text in data["description"]:
        tokens = word_tokenize(text)
        tokens = [token.lower() for token in tokens if token.isalpha() and token.lower() not in stop_words]
        frequency = nltk.FreqDist(tokens)
        for token, count in frequency.items():
            if count / len(tokens) > threshold:
                lexicon[token] = True
    return lexicon

lexicon = generate_lexicon(cleaned_data)
```
