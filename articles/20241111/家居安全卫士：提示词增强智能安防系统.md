import numpy as np

# 传感器数据
sensor_data = np.array([[1, 2], [3, 4], [5, 6]])

# 提示词
prompt = ["安全", "危险", "安全"]

# 权重
weights = np.array([0.5, 0.5])

# 预测
y = predict(sensor_data, prompt, weights)

# 输出预测结果
print("预测结果：", y)

def predict(sensor_data, prompt, weights):
    # 计算特征值
    features = np.array([1 if p == "危险" else 0 for p in prompt])
    
    # 计算预测值
    y = np.dot(weights, features)
    
    return y
