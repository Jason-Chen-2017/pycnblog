```python
import numpy as np

# 智能体类
class Agent:
    def __init__(self, data_source):
        self.data_source = data_source
        self.esg_score = 0

    def evaluate(self):
        # 根据数据源评估ESG得分
        score = self.data_source.calculate_esg_score()
        self.esg_score += score

    def communicate(self, other_agents):
        # 与其他智能体交换评估结果
        for agent in other_agents:
            self.esg_score += agent.esg_score

    def update(self):
        # 根据评估结果更新自身状态
        if self.esg_score > 0:
            self.esg_score /= len(self.data_source)
        else:
            self.esg_score = 0

# 数据源类
class DataSource:
    def __init__(self):
        self.data = []

    def add_data(self, data):
        self.data.append(data)

    def calculate_esg_score(self):
        # 计算ESG得分
        return np.mean(self.data)

# 初始化智能体和数据源
agent1 = Agent(DataSource())
agent2 = Agent(DataSource())

# 添加数据
agent1.data_source.add_data([0.8, 0.9, 1.0])
agent2.data_source.add_data([0.7, 0.8, 0.9])

# 评估ESG得分
agent1.evaluate()
agent2.evaluate()

# 通信与更新
agent1.communicate([agent2])
agent2.communicate([agent1])

# 更新状态
agent1.update()
agent2.update()

print(f"Agent 1 ESG Score: {agent1.esg_score}")
print(f"Agent 2 ESG Score: {agent2.esg_score}")
```
