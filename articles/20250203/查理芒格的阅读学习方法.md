```python
# app.py

from flask import Flask, request, jsonify
from models import db, User, Book, ReadingPlan, LearningJournal
from sqlalchemy.orm import sessionmaker

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///app.db'
db.init_app(app)

@app.route('/books', methods=['GET'])
def get_books():
    session = sessionmaker(bind=db.engine)()
    books = session.query(Book).all()
    return jsonify([book.to_dict() for book in books])

@app.route('/reading_plans', methods=['POST'])
def create_reading_plan():
    session = sessionmaker(bind=db.engine)()
    new_plan = ReadingPlan.from_dict(request.json)
    session.add(new_plan)
    session.commit()
    return jsonify(new_plan.to_dict()), 201

@app.route('/learning_journals', methods=['POST'])
def create_learning_journal():
    session = sessionmaker(bind=db.engine)()
    new_journal = LearningJournal.from_dict(request.json)
    session.add(new_journal)
    session.commit()
    return jsonify(new_journal.to_dict()), 201

if __name__ == '__main__':
    app.run(debug=True)
```
