def calculate_distribution(land_layer, population_layer):
    # 加载土地和人口数据
    land_data = gpd.read_file(land_layer)
    population_data = gpd.read_file(population_layer)
    
    # 计算人均土地面积
    total_area = land_data.area.sum()
    population = population_data['人口总数'].sum()
    per_person_area = total_area / population
    
    # 分配生态保护区

    # 分配农业生产区
    farm_area = 0.6 * per_person_area
    # 分配基础设施区
    infra_area = 0.2 * per_person_area
    
    # 调整资源分配
    if eco_area + farm_area > per_person_area:
        diff = eco_area + farm_area - per_person_area
        eco_area -= diff * 0.5
        farm_area -= diff * 0.5
    
    # 输出结果
    return eco_area, farm_area, infra_area

# 示例数据路径
