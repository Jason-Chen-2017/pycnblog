### 2.1 类型论的基本概念
   - **核心算法原理讲解**：
     ```plaintext
     // 定义类型
     Type Integer = {value: int};
     Type Float = {value: float};

     // 类型构造
     Function makeInteger(value: int): Integer {
       return {value: value};
     }

     Function makeFloat(value: float): Float {
       return {value: value};
     }

     // 类型等价
     Predicate isEqualInteger(a: Integer, b: Integer): bool {
       return a.value == b.value;
     }

     Predicate isEqualFloat(a: Float, b: Float): bool {
       return a.value == b.value;
     }
     ```
