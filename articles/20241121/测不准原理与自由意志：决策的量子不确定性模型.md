from qiskit import QuantumCircuit, Aer, execute

psi = np.array([0.6, 0.8, 0.1]) / np.linalg.norm(psi)

# 定义哈密顿量
H = np.array([[0, 1, 0],
              [1, 0, 1],
              [0, 1, 0]])

# 定义演化时间
t = 1

# 创建量子电路
qc = QuantumCircuit(3)

# 初始化量子态
qc.initialize(psi)

# 添加哈密顿量
qc.add_hadamard_gate(0)
qc.add_hadamard_gate(1)
qc.add_hadamard_gate(2)

# 演化
qc evolve(H, t)

# 执行电路

result = execute(qc, backend).result()

# 输出量子态
print(state)
