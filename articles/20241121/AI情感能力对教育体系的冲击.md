```pseudo
function sentiment_analysis(text):
    # 初始化情感词典
    sentiment_lexicon = load_sentiment_lexicon()

    # 分词
    words = tokenize(text)

    # 计算情感得分
    sentiment_score = 0
    for word in words:
        sentiment_score += sentiment_lexicon[word]

    # 判断情感极性
    if sentiment_score > 0:
        return "正面"
    elif sentiment_score < 0:
        return "负面"
    else:
        return "中性"
```
