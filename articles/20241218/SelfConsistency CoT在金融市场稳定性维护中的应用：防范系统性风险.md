def data_preprocessing(data):

    cleaned_data = data.dropna()
    # 数据格式化
    formatted_data = cleaned_data.reset_index(drop=True)
    return formatted_data

# 一致性检查
def check_consistency(data):
    # 检查数据一致性
    consistency = data["close"].eq(data["close"].shift(1))
    # 标记异常数据
    anomalies = ~consistency
    return anomalies

# 主函数
def main():
    # 读取金融数据
    data = pd.read_csv("financial_data.csv")
    # 数据预处理
    processed_data = data_preprocessing(data)
    # 一致性检查
    anomalies = check_consistency(processed_data)
    # 异常处理
    if anomalies.any():
        print("警告：检测到异常数据！")
    else:
        print("当前市场稳定。")

# 运行主函数
if __name__ == "__main__":
