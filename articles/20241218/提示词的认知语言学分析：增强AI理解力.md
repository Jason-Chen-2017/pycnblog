import jieba
import jieba.posseg as pseg
from nltk.corpus import wordnet as wn

# 分词与词性标注
def tokenize_and_annotate(text):
    words = jieba.cut(text)
    word_tags = pseg.cut(text)
    return list(zip(words, word_tags))

# 提取提示词
def extract_fragments(words):
    fragments = []
    for word, tag in words:
        if tag in ["v", "n", "a", "d"]:  # 基本词性，可根据实际需求调整
            fragments.append(word)
    return fragments

# 语义分析
def semantic_analysis(word):
    synsets = wn.synsets(word)
    if synsets:
        return synsets[0].definition()
    else:
        return "未知语义"

# 语法分析
def grammatical_analysis(tag):
    if tag == "v":
        return "动词"
    elif tag == "n":
        return "名词"
    elif tag == "a":
        return "形容词"
    elif tag == "d":
        return "副词"
    else:
        return "其他"

# 语用分析
def pragmatic_analysis(word):
    if word in ["因此", "然而", "所以"]:
        return "逻辑关系"
    else:
        return "未知语用"

# 提示词认知语言学分析算法
def cognitive_linguistic_analysis(text):
    words = tokenize_and_annotate(text)
    fragments = extract_fragments(words)
    for word in fragments:
        print(f"词汇: {word}")
        print(f"词性: {words[word][1]}")
        print(f"语义: {semantic_analysis(word)}")
        print(f"语法: {grammatical_analysis(words[word][1])}")
        print(f"语用: {pragmatic_analysis(word)}")
        print()

# 测试
text = "然而，这个结论与实际情况不符，因此我们需要重新考虑这个问题。"
cognitive_linguistic_analysis(text)

import jieba
import jieba.posseg as pseg
from nltk.corpus import wordnet as wn
from collections import defaultdict

# 分词与词性标注
def tokenize_and_annotate(text):
    words = jieba.cut(text)
    word_tags = pseg.cut(text)
    return list(zip(words, word_tags))

# 提取提示词
def extract_fragments(words):
    fragments = []
    for word, tag in words:
        if tag in ["v", "n", "a", "d"]:  # 基本词性，可根据实际需求调整
            fragments.append(word)
    return fragments

# 语义分析
def semantic_analysis(word):
    synsets = wn.synsets(word)
    if synsets:
        return synsets[0].definition()
    else:
        return "未知语义"

# 语法分析
def grammatical_analysis(tag):
    if tag == "v":
        return "动词"
    elif tag == "n":
        return "名词"
    elif tag == "a":
        return "形容词"
    elif tag == "d":
        return "副词"
    else:
        return "其他"

# 语用分析
def pragmatic_analysis(word):
    if word in ["因此", "然而", "所以"]:
        return "逻辑关系"
    else:
        return "未知语用"

# 提示词认知语言学分析算法
def cognitive_linguistic_analysis(text):
    words = tokenize_and_annotate(text)
    fragments = extract_fragments(words)
    for word in fragments:
        print(f"词汇: {word}")
        print(f"词性: {words[word][1]}")
        print(f"语义: {semantic_analysis(word)}")
        print(f"语法: {grammatical_analysis(words[word][1])}")
        print(f"语用: {pragmatic_analysis(word)}")
        print()
