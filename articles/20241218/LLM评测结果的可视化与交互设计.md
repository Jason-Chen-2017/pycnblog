#### 2.2.3 交互设计

```javascript
    <script>
        // 生成交互式散点图
        function create_interactive_scatter_plot(data) {
            const margin = { top: 20, right: 20, bottom: 30, left: 40 };
            const width = 960 - margin.left - margin.right;
            const height = 500 - margin.top - margin.bottom;

            const x = d3.scaleLinear().domain([0, 1]).range([0, width]);
            const y = d3.scaleLinear().domain([0, 1]).range([height, 0]);

            const svg = d3.select("#scatter-plot")
                          .append("svg")
                          .attr("width", width + margin.left + margin.right)
                          .attr("height", height + margin.top + margin.bottom)
                          .append("g")
                          .attr("transform", `translate(${margin.left}, ${margin.top})`);

            // 添加X轴和Y轴
            svg.append("g")
               .attr("transform", `translate(0, ${height})`)
               .call(d3.axisBottom(x));

            svg.append("g")
               .call(d3.axisLeft(y));

            // 添加散点
            svg.selectAll("dot")
               .data(data)
               .enter().append("circle")
               .attr("cx", d => x(d.Accuracy))
               .attr("cy", d => y(d.F1))
               .attr("r", 3.5)
               .style("fill", "blue");

            // 添加交互
            svg.selectAll("circle")
               .on("mouseover", function(event, d) {
                   // 鼠标悬停时显示提示框
                   const tooltip = d3.select("body").append("div")
                                     .attr("class", "tooltip")
                                     .style("opacity", 0);

                   tooltip.html(`Accuracy: ${d.Accuracy}, F1: ${d.F1}`)
                          .style("left", (event.pageX) + "px")
                          .style("top", (event.pageY - 28) + "px");
                   tooltip.transition()
                          .duration(200)
                          .style("opacity", .9);
               })
               .on("mouseout", function(d) {
                   // 鼠标离开时隐藏提示框
                   d3.select(".tooltip").transition()
                                      .duration(500)
                                      .style("opacity", 0);
               });
        }

        // 示例数据
        const data = [
            { Accuracy: 0.8, F1: 0.9 },
            { Accuracy: 0.7, F1: 0.85 },
            { Accuracy: 0.9, F1: 0.95 },
            { Accuracy: 0.6, F1: 0.8 },
            { Accuracy: 0.75, F1: 0.88 }
        ];

        // 调用函数创建交互式散点图
        create_interactive_scatter_plot(data);
    </script>
</body>
</html>
```

   ```javascript
   <html>
   <head>
       <script src="https://d3js.org/d3.v7.min.js"></script>

           const margin = { top: 20, right: 20, bottom: 30, left: 40 };
           const width = 960 - margin.left - margin.right;
           const height = 500 - margin.top - margin.bottom;

           const x = d3.scaleLinear().domain([0, 1]).range([0, width]);
           const y = d3.scaleLinear().domain([0, 1]).range([height, 0]);

           const svg = d3.select("#scatter-plot")
                         .append("svg")
                         .attr("width", width + margin.left + margin.right)
                         .attr("height", height + margin.top + margin.bottom)
                         .append("g")
                         .attr("transform", `translate(${margin.left}, ${margin.top})`);

           // 添加X轴和Y轴
           svg.append("g")
              .attr("transform", `translate(0, ${height})`)
              .call(d3.axisBottom(x));

           svg.append("g")
              .call(d3.axisLeft(y));

           // 添加散点
           svg.selectAll("dot")
              .data(processed_data)
              .enter().append("circle")
              .attr("cx", d => x(d.Accuracy))
              .attr("cy", d => y(d.F1))
              .attr("r", 3.5)
              .style("fill", "blue");

           // 添加交互
           svg.selectAll("circle")
              .on("mouseover", function(event, d) {
                  // 鼠标悬停时显示提示框
                  const tooltip = d3.select("body").append("div")
                                    .attr("class", "tooltip")
                                    .style("opacity", 0);

                         .style("top", (event.pageY - 28) + "px");
                  tooltip.transition()
                         .duration(200)
                         .style("opacity", .9);
              })
              .on("mouseout", function(d) {
                  // 鼠标离开时隐藏提示框
                  d3.select(".tooltip").transition()
                                      .duration(500)
                                      .style("opacity", 0);
              });
       </script>
   </body>
   </html>
   ```
