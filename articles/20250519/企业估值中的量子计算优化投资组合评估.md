from qiskit.circuit import Gate, gates
import numpy as np

def quantumannealing(n_qubits, initial_temp, final_temp, steps):
    qc = QuantumCircuit(n_qubits)
    qc.h(range(n_qubits))
    backend = Aer('statevector_simulator')
    job = execute(qc, backend)
    result = job.result()
    return result
