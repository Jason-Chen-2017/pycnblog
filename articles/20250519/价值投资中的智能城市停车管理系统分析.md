import numpy as np
import pandas as pd
from sklearn.tree import DecisionTreeRegressor

# 数据加载
data = pd.read_csv('parking_data.csv')

# 数据清洗
data.dropna(inplace=True)

# 特征提取
X = data[['人口', '就业密度', '交通便利性']]
y = data['需求']

# 模型训练
model = DecisionTreeRegressor()
model.fit(X, y)

# 预测
print(model.predict(X))
