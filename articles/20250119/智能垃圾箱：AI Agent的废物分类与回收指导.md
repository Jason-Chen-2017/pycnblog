def classify_waste(waste_type):
    if waste_type == "recyclable":
        return "可回收处理"
    elif waste_type == "hazardous":
        return "有害处理"
    elif waste_type == "wet":
        return "湿垃圾处理"
    elif waste_type == "dry":
        return "干垃圾处理"
    else:
        return "未知废物类型"
