```python
import Adafruit_GPIO.SPI as SPI
import Adafruit_MCP3008

mcp = Adafruit_MCP3008.MCP3008(spi=SPI.SpiDev(0, 0))

def collect_environmental_data():
    temperature = mcp.read_adc(0)  # 温度传感器
    humidity = mcp.read_adc(1)  # 湿度传感器
    light = mcp.read_adc(2)  # 光照传感器
    return temperature, humidity, light
```

```python
def control_temperature(humidity):
    if humidity > 70:
        GPIO.output(11, True)  # 打开风扇
    else:
        GPIO.output(11, False)  # 关闭风扇

def control_light(temperature):
    if temperature < 20:
        GPIO.output(10, True)  # 打开照明
    else:
        GPIO.output(10, False)  # 关闭照明

# 初始化GPIO
```
