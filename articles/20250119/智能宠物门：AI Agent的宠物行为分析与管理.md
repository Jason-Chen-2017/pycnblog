#### 1.5.4.2 系统交互

7. **数据存储**

```python
import cv2
from keras.models import load_model

# 加载预训练的宠物识别模型
model = load_model('pet_recognition_model.h5')

# 读取摄像头帧
cap = cv2.VideoCapture(0)

while True:
    ret, frame = cap.read()
    if not ret:
        break
    
    # 对图像进行预处理
    processed_frame = preprocess_frame(frame)
    
    # 使用模型进行宠物识别
    prediction = model.predict(processed_frame)
    pet_name = get_pet_name(prediction)

    # 显示识别结果
    cv2.putText(frame, f'Pet: {pet_name}', (10, 30), cv2.FONT_HERSHEY_SIMPLEX, 1, (0, 0, 255), 2)
    cv2.imshow('Pet Recognition', frame)
    
    if cv2.waitKey(1) & 0xFF == ord('q'):
        break

# 释放摄像头资源
cap.release()
cv2.destroyAllWindows()

# 预处理图像
def preprocess_frame(frame):
    # 进行图像缩放、灰度化、归一化等操作
    processed_frame = cv2.resize(frame, (128, 128))
    processed_frame = cv2.cvtColor(processed_frame, cv2.COLOR_BGR2GRAY)
    processed_frame = processed_frame / 255.0
    return processed_frame

# 获取宠物名称
def get_pet_name(prediction):
    # 根据预测结果返回宠物名称
    if prediction[0] > 0.5:
        return 'Dog'
    else:
        return 'Cat'
```

这段代码首先加载预训练的宠物识别模型，然后通过摄像头读取实时图像，对图像进行预处理后，使用模型进行宠物识别，并将识别结果显示在窗口中。
