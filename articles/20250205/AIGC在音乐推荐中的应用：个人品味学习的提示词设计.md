# 示例代码：基于协同过滤的音乐推荐
import numpy as np

# 用户-歌曲矩阵
user_item_matrix = np.array([[1, 0, 1],
                               [0, 1, 1],
                               [1, 1, 0]])

# 计算用户相似度
user_similarity = np.dot(user_item_matrix, user_item_matrix.T)

# 生成推荐
def recommend(user_id):
    similarities = user_similarity[user_id]
    recommendations = {}
    for i in range(len(similarities)):
        if i != user_id:
            recommendations[i] = similarities[i]
    return recommendations

print(recommend(0))
