def diagnose(symptoms, tests):
    # 症状分析
    symptom_score = 0
    for s in symptoms:
        symptom_score += symptom_weights[s]
    # 检查结果分析
    test_score = 0
    for t in tests:
        test_score += test_weights[t]
    # 综合诊断
    total_score = symptom_score + test_score
    if total_score > threshold:
        return "诊断为疾病"
    else:
        return "无疾病症状"
```

class DiagnoseEngine:
    def __init__(self, symptom_weights, test_weights, threshold):
        self.symptom_weights = symptom_weights
        self.test_weights = test_weights
        self.threshold = threshold

    def diagnose(self, symptoms, tests):
        symptom_score = sum(self.symptom_weights[s] for s in symptoms)
        test_score = sum(self.test_weights[t] for t in tests)
        total_score = symptom_score + test_score
        if total_score > self.threshold:
            return "诊断为疾病"
        else:
            return "无疾病症状"
```
