### 5.1 环境安装

### 5.2 系统核心实现源代码

```python
import openai

def chatgpt_collaboration(api_key):
    openai.api_key = api_key
    while True:
        user_input = input("请输入问题：")
        # 调用ChatGPT API
        response = openai.ChatCompletion.create(
            model="gpt-3.5-turbo",
            messages=[{"role": "user", "content": user_input}]
        )
        chatgpt_output = response.choices[0].message.content
        print("ChatGPT生成结果：", chatgpt_output)
        expert_feedback = input("专家，请输入反馈或优化建议：")
        print("专家反馈：", expert_feedback)
        print("\n")

# 使用示例API密钥
chatgpt_collaboration("your-api-key")

```
