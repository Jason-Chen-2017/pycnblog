  import torch
  import torch.nn as nn
  import torch.nn.functional as F

  class ZeroShotCoT(nn.Module):
      def __init__(self, embed_dim, num_heads):
          super(ZeroShotCoT, self).__init__()
          self.embed_dim = embed_dim
          self.num_heads = num_heads
          self.key_dim = embed_dim // num_heads
          self.value_dim = embed_dim // num_heads
          self.all_head_weights = nn.Parameter(torch.randn(num_heads, 1))

      def forward(self, x, mask=None):
          batch_size, seq_len, _ = x.size()
          x = x.view(batch_size, seq_len, self.num_heads, self.key_dim)
          key = x.permute(0, 1, 2, 3)
          value = x.permute(0, 1, 2, 3)
          
          attention = (key @ value.permute(0, 1, 3, 2)) / torch.sqrt(torch.tensor(self.key_dim))
          attention = F.softmax(attention, dim=-1)
          
          context = (attention @ value).permute(0, 1, 3, 2).view(batch_size, seq_len, self.embed_dim)
          
          head_weights = F.softmax(self.all_head_weights, dim=0)
          context = (context * head_weights).sum(dim=2)
          
          return context
