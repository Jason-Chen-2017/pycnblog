```python
import torch
import torch.nn as nn

class ThinkChainModel(nn.Module):
    def __init__(self, input_size, hidden_size, output_size):
        super(ThinkChainModel, self).__init__()
        self.hidden_size = hidden_size
        self.lstm = nn.LSTM(input_size, hidden_size, batch_first=True)
        self.fc = nn.Linear(hidden_size, output_size)
    
    def forward(self, x):
        out, _ = self.lstm(x)
        out = self.fc(out[:, -1, :])
        return out

# 示例数据
input_size = 4
hidden_size = 10
output_size = 2

model = ThinkChainModel(input_size, hidden_size, output_size)
```
