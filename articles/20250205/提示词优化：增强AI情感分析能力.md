def optimize_prompt(prompt, target_emotion):
    # 初始化候选提示词列表
    candidates = generate_candidates(prompt)
    # 计算每个候选提示词的情感得分
    scores = {}
    for candidate in candidates:
        score = evaluate_emotion(candidate, target_emotion)
        scores[candidate] = score
    # 选择得分最高的提示词
    best_candidate = max(scores, key=lambda x: scores[x])
    return best_candidate
```
