#### 3.2 Python代码示例

def generate_response(prompt):
    # 模拟ChatGPT的响应生成
    import openai
    client = openai.Client(api_key='your_api_key')
    response = client.chat.completions.create(
        model="gpt-3.5-turbo",
        messages=[{"role": "user", "content": prompt}]
    )
    return response.choices[0].message.content

# 示例使用
prompt = "编写一段关于人工智能的简介。"
print(generate_response(prompt))
```

import openai

def optimize_prompt(prompt, iterations=5):
    for i in range(iterations):
        # 调整提示词参数
        response = generate_response(prompt)
        feedback = input(f"迭代{i+1}，生成结果为：{response}\n请输入反馈或调整后的提示词：")
        if feedback:
            prompt = feedback
    return prompt

# 示例优化
initial_prompt = "生成一段关于气候变化的文字。"
optimized_prompt = optimize_prompt(initial_prompt)
print(f"优化后的提示词：{optimized_prompt}")
```
