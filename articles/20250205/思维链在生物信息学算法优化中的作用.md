class MindChain:
    def __init__(self):
        self.nodes = []
        self.edges = []

    def add_node(self, node):
        self.nodes.append(node)

    def add_edge(self, from_node, to_node, weight):
        self.edges.append((from_node, to_node, weight))

    def run(self, input_data):
        # 简单的优化逻辑示例
        optimized = input_data.copy()
        for node in self.nodes:
            optimized = self.apply_rule(optimized, node)
        return optimized

    def apply_rule(self, data, node):
        # 示例优化规则
        return data * node.weight
