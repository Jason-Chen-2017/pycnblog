import numpy as np
import matplotlib.pyplot as plt

def self_consistency(data):
    x = np.mean(data, axis=0)
    while True:
        x_new = np.mean(data, axis=0)
        if np.allclose(x, x_new, rtol=1e-6):
            break
        x = x_new
    return x_new
