    def __init__(self, actions):
        self.actions = actions
        self.Q = defaultdict(int)
    
    def choose_action(self, state):
        max_action = max(self.Q[(state, action)] for action in self.actions)
        return random.choice([action for action in self.actions if self.Q[(state, action)] == max_action])
