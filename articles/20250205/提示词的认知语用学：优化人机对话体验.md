```python
import numpy as np

def cosine_similarity(a, b):
    return np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b))

def entropy(w_i):
    prob = w_i / sum(w_i)
    return -sum(prob * np.log2(prob))

def optimize_prompt(w, w_i, lambda_val):
    similarity = cosine_similarity(w, w_i)
    ent = entropy(w_i)
    return (similarity - lambda_val * ent) / len(w_i)
```

```python
import numpy as np
from sklearn.metrics.pairwise import cosine_similarity

def generate_prompt(w_i):
    # 生成提示词
    return "基于上下文，建议" + w_i

def evaluate_prompt(w, w_i):
    # 计算相似度
    sim = cosine_similarity(w.reshape(1, -1), w_i.reshape(1, -1))[0, 0]
    # 计算熵
    prob = w_i / sum(w_i)
    ent = -sum(prob * np.log2(prob))
    # 计算优化值
    return sim - 0.5 * ent

# 示例
w = np.array([1, 2, 3])
w_i = np.array([2, 3, 4])
print(evaluate_prompt(w, w_i))
```
