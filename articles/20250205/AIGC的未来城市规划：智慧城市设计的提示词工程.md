### 5.2 系统核心实现源代码

```python
from transformers import pipeline

# 初始化提示词生成器
generator = pipeline('text-generation', model='gpt2')

# 提示词生成函数
def generate_prompts(num_prompts=5):
    prompts = []
    for _ in range(num_prompts):
        prompt = generator.generate("设计一个未来的智慧城市", max_length=50)
        prompts.append(prompt[0]['generated_text'])
    return prompts

# 输出结果
print(generate_prompts())
```
