import numpy as np

def monte_carlo_simulation(n_iterations, low, high):
    np.random.seed(42)
    losses = np.random.uniform(low, high, n_iterations)
    expected_loss = np.mean(losses)
    return expected_loss

n_iterations = 10000
low = 0
high = 1000000

result = monte_carlo_simulation(n_iterations, low, high)
print(f"Expected loss: {result}")
