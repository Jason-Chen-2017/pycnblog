```python
def collaborative_filtering(user_item_matrix):
    similarity = cosine_similarity(user_item_matrix)
    recommendation = np.dot(similarity, user_item_matrix)
    return recommendation
```

### 深度学习模型实现

```python
class NeuralNetwork Recommender:
    def __init__(self, input_dim, hidden_dim, output_dim):
        self.weights = np.random.randn(input_dim, hidden_dim)
        self.output_weights = np.random.randn(hidden_dim, output_dim)
    
    def forward(self, input):
        hidden = np.dot(input, self.weights)
        output = np.dot(hidden, self.output_weights)
        return output
```
