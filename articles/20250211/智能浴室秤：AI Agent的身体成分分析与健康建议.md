```python
import numpy as np
from sklearn.linear_model import LinearRegression

# 数据准备
X = np.array([[1], [2], [3], [4]])
y = np.array([2, 4, 5, 4])

# 模型训练
model = LinearRegression()
model.fit(X, y)

# 预测
print(model.predict([[5]]))
```

- 3.3.2 支持向量机模型的Python实现

```python
from sklearn import svm

X = [[1, -1], [-1, 1], [1, 1], [-1, -1]]
y = [1, -1, -1, 1]

# 模型训练
model = svm.SVC()
model.fit(X, y)

# 预测
print(model.predict([[1, 1]]))
```
