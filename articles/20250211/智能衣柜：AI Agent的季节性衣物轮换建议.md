### 5.2.1 时间序列预测代码
```python
import torch
from torch import nn
import numpy as np

class ARIMA(nn.Module):
    def __init__(self, input_size, hidden_size):
        super(ARIMA, self).__init__()
        self.lstm = nn.LSTM(input_size, hidden_size, 1)
        self.linear = nn.Linear(hidden_size, 1)

    def forward(self, x):
        out, _ = self.lstm(x)
        out = self.linear(out)
        return out

# 示例数据
data = torch.randn(10, 1, 1)
model = ARIMA(1, 10)
output = model(data)
print(output)
```

### 5.2.2 协同过滤推荐代码
```python
from sklearn.metrics.pairwise import cosine_similarity

def collaborative_filtering(user_data, user_id):
    similarity = cosine_similarity(user_data, user_data)
    similar_users = np.argsort(similarity[user_id])[::-1]
    recommended_items = []
    for user in similar_users:
        if user != user_id:
            recommended_items.extend(user_data[user])
    return recommended_items

# 示例数据
user_data = {
    0: [1, 2, 3],
    1: [2, 3, 4],
    2: [3, 4, 5],
}
recommended_items = collaborative_filtering(user_data, 0)
print(recommended_items)
```
