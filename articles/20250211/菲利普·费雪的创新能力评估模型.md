def calculate_innovation_capability(tech_score, market_score, team_score):
    # 权重分配
    weights = {
        'tech': 0.4,
        'market': 0.3,
        'team': 0.3
    }
    # 计算综合得分
    total_score = tech_score * weights['tech'] + market_score * weights['market'] + team_score * weights['team']

from sklearn.metrics import accuracy_score

def main():
    # 示例数据
    X = np.array([[85, 70, 80], [70, 85, 90]])
    y = np.array([0.8, 0.9])
    
    # 模型训练
    model = np.linalg.lstsq(np.c_[np.ones(len(X)), X], y, rcond=None)[0]
    
    # 模型预测
    new_data = np.array([[80, 65, 75]])
    predicted_score = model[0] + model[1][0] * 80 + model[1][1] * 65 + model[1][2] * 75
    print(f"预测创新能力得分：{predicted_score}")
    
    # 模型评估
    y_pred = model[0] + model[1][0] * X[:,0] + model[1][1] * X[:,1] + model[1][2] * X[:,2]
