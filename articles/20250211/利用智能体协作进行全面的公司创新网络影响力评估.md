```python
def calculate_influence(nodes, edges):
    # nodes: 创新网络的节点列表
    # edges: 创新网络的边列表
    influence = {}
    for node in nodes:
        influence[node] = 0
    # 计算每个节点的影响力
    for edge in edges:
        source, target = edge
        influence[source] += 1
        influence[target] += 1
    return influence
```

```python
class Agent:
    def __init__(self, id):
        self.id = id
        self.tasks = []
    
    def communicate(self, other_agents):
        for agent in other_agents:
            if agent.id != self.id:
                agent.receive_message(self.send_message())
    
    def send_message(self):
        return f"Agent {self.id} message"
    
    def receive_message(self, message):
        print(f"Agent {self.id} received message: {message}")
```

### 6.2.2 创新网络模块实现

```python
class InnovationNetwork:
    def __init__(self, nodes, edges):
        self.nodes = nodes
        self.edges = edges
    
    def calculate_influence(self):
        influence = {node: 0 for node in self.nodes}
        for edge in self.edges:
            source, target = edge
            influence[source] += 1
            influence[target] += 1
        return influence
```
