```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# 定义费雪文化指数计算函数
def fisher_cultural_index(values, leadership, employee_involvement, org_structure):
    # 权重分别为30%, 25%, 20%, 25%
    weights = [0.3, 0.25, 0.2, 0.25]
    scores = [values, leadership, employee_involvement, org_structure]
    # 计算加权和
    cultural_index = sum(w * s for w, s in zip(weights, scores))
    return cultural_index

# 示例数据
values_score = 85  # 价值观与使命的评分，满分100
leadership_score = 80  # 领导风格评分
employee_score = 85  # 员工参与度评分
org_score = 90  # 组织结构评分

# 计算文化指数
cultural_index = fisher_cultural_index(values_score, leadership_score, employee_score, org_score)
print(f"费雪文化指数: {cultural_index}%")

# 可视化结果
plt.figure(figsize=(10, 6))
plt.bar(['价值观', '领导风格', '员工参与度', '组织结构'], [values_score, leadership_score, employee_score, org_score])
plt.title('企业文化评估指标')
plt.ylabel('评分')
plt.show()
```
