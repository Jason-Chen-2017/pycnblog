```python
import pandas as pd
import numpy as np
from sklearn.linear_model import LinearRegression

# 示例数据
data = {
    '收入': [100, 200, 300, 400],
    '利润': [20, 30, 40, 50]
}
df = pd.DataFrame(data)

# 特征提取
X = df[['收入']].values
y = df['利润'].values

# 模型训练
model = LinearRegression()
model.fit(X, y)

# 预测估值
new_income = 500
predicted_profit = model.predict([[new_income]])[0]
print(f"收入为{new_income}时，预测利润为{predicted_profit}")
```
