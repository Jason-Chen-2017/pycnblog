```python
import numpy as np
import pandas as pd
from sklearn import tree

# 加载数据
data = pd.read_csv('project_progress.csv')
X = data[['task_num', 'resource_num', 'time_spent']]
y = data['progress']

# 训练模型
model = tree.DecisionTreeRegressor()
model.fit(X, y)

# 预测结果
new_data = np.array([[10, 5, 20]])
predicted_progress = model.predict(new_data)
print(predicted_progress)
```
