#### 6.2.2 塞思·卡拉曼模型的实现
```python
def carathesouels_rap(rf, beta, market Risk):
    """
    计算风险调整现值
    """
    # 计算无风险利率和市场风险溢价
    risk_free_rate = rf
    market_risk_premium = beta * (market_Risk - rf)
    
    # 计算风险调整现值
    rap = 1 / (1 + risk_free_rate + market_risk_premium)
    return rap

# 示例计算
rf = 0.02  # 无风险利率
beta = 1.5  # 市场贝塔系数
market_Risk = 0.08  # 市场风险溢价

rap_value = carathesouels_rap(rf, beta, market_Risk)
print(f"风险调整现值: {rap_value}")
```

#### 6.2.3 投资组合优化
```python
import numpy as np
from scipy.optimize import minimize

def portfolio_optimization(returns, weights):
    """
    最小化投资组合风险
    """
    # 计算投资组合的方差
    portfolio_variance = np.dot(weights.T, np.dot(returns.cov(), weights))
    return portfolio_variance

# 示例数据
returns = np.array([0.1, 0.2, 0.15, 0.08])
weights = np.array([0.25, 0.3, 0.25, 0.2])

# 优化过程
result = minimize(portfolio_optimization, weights, args=(returns,))
optimized_weights = result.x
print(f"优化后的权重: {optimized_weights}")
```
