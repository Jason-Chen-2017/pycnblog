#### 6.3.1 简单的投资组合管理代码

```python
class CommodityPortfolio:
    def __init__(self):
        self.assets = {}

    def add_asset(self, asset, quantity):
        self.assets[asset] = quantity

    def remove_asset(self, asset):
        del self.assets[asset]

    def get_asset_value(self, asset, price):
        if asset in self.assets:
            return self.assets[asset] * price
        else:
            return 0

# 示例用法
portfolio = CommodityPortfolio()
portfolio.add_asset('黄金', 10)
portfolio.add_asset('原油', 5)
print(portfolio.get_asset_value('黄金', 1500))  # 输出：15000
print(portfolio.get_asset_value('原油', 100))  # 输出：500
```
