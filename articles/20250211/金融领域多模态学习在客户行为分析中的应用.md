#### 3.3 算法实现代码示例
```python
import torch
import torch.nn as nn

class MultiModal(nn.Module):
    def __init__(self, input_dim1, input_dim2, hidden_dim, output_dim):
        super(MultiModal, self).__init__()
        self.fc1 = nn.Linear(input_dim1, hidden_dim)
        self.fc2 = nn.Linear(input_dim2, hidden_dim)
        self.fc3 = nn.Linear(hidden_dim * 2, output_dim)
        self.relu = nn.ReLU()
        self.dropout = nn.Dropout(0.5)
    
    def forward(self, x1, x2):
        out1 = self.fc1(x1)
        out1 = self.relu(out1)
        out1 = self.dropout(out1)
        
        out2 = self.fc2(x2)
        out2 = self.relu(out2)
        out2 = self.dropout(out2)
        
        out = torch.cat([out1, out2], dim=1)
        out = self.fc3(out)
        return out

# 示例输入
input1 = torch.randn(10, 64)
input2 = torch.randn(10, 64)
model = MultiModal(64, 64, 128, 1)
output = model(input1, input2)
print(output.size())  # 输出：(10, 1)
```

#### 5.2 核心代码实现
```python
import torch
import torch.nn as nn
import torch.optim as optim
import numpy as np
import matplotlib.pyplot as plt

# 数据生成（示例）
class DataGenerator:
    def __init__(self, batch_size, input_dim1, input_dim2, output_dim):
        self.batch_size = batch_size
        self.input_dim1 = input_dim1
        self.input_dim2 = input_dim2
        self.output_dim = output_dim
    
    def generate_data(self):
        x1 = torch.randn(self.batch_size, self.input_dim1)
        x2 = torch.randn(self.batch_size, self.input_dim2)
        y = torch.randint(0, self.output_dim, (self.batch_size,))
        return x1, x2, y

# 模型定义
class MultiModal(nn.Module):
    def __init__(self, input_dim1, input_dim2, hidden_dim, output_dim):
        super(MultiModal, self).__init__()
        self.fc1 = nn.Linear(input_dim1, hidden_dim)
        self.fc2 = nn.Linear(input_dim2, hidden_dim)
        self.fc3 = nn.Linear(hidden_dim * 2, output_dim)
        self.relu = nn.ReLU()
        self.dropout = nn.Dropout(0.5)
    
    def forward(self, x1, x2):
        out1 = self.fc1(x1)
        out1 = self.relu(out1)
        out1 = self.dropout(out1)
        
        out2 = self.fc2(x2)
        out2 = self.relu(out2)
        out2 = self.dropout(out2)
        
        out = torch.cat([out1, out2], dim=1)
        out = self.fc3(out)
        return out

# 训练过程
def train(model, optimizer, criterion, x1, x2, y):
    optimizer.zero_grad()
    outputs = model(x1, x2)
    loss = criterion(outputs, y)
    loss.backward()
    optimizer.step()
    return loss.item()

# 主函数
def main():
    batch_size = 32
    input_dim1 = 64
    input_dim2 = 64
    hidden_dim = 128
    output_dim = 10
    learning_rate = 0.001
    num_epochs = 100
    
    data_generator = DataGenerator(batch_size, input_dim1, input_dim2, output_dim)
    model = MultiModal(input_dim1, input_dim2, hidden_dim, output_dim)
    optimizer = optim.Adam(model.parameters(), lr=learning_rate)
    criterion = nn.CrossEntropyLoss()
    
    for epoch in range(num_epochs):
        x1, x2, y = data_generator.generate_data()
        loss = train(model, optimizer, criterion, x1, x2, y)
        if epoch % 10 == 0:
            print(f'Epoch [{epoch+1}/{num_epochs}], Loss: {loss:.4f}')
    
    # 模型评估
    with torch.no_grad():
        outputs = model(x1, x2)
        _, predicted = torch.max(outputs.data, 1)
        accuracy = (predicted == y).sum().item() / batch_size
        print(f'Accuracy: {accuracy:.4f}')

if __name__ == '__main__':
    main()
```
