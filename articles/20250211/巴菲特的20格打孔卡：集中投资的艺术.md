import pandas as pd

def calculate_profitability(data):
    # 计算盈利能力指标
    data['net_profit_margin'] = data['净利润'] / data['收入']
    return data

def calculate_roe(data):
    # 计算ROE
    data['roe'] = data['净利润'] / data['股东权益']
    return data

def score_company(data):
    # 根据各项指标评分
    data['总得分'] = data['净_profit_margin'] * 0.4 + data['roe'] * 0.4 + data['市场占有率'] * 0.2
    return data

# 示例数据
data = {
    '公司': ['公司A', '公司B', '公司C'],
    '收入': [100, 200, 150],
    '净利润': [10, 20, 15],
    '股东权益': [50, 100, 75],
    '市场占有率': [20, 30, 25]
}

df = pd.DataFrame(data)
df = calculate_profitability(df)
df = calculate_roe(df)
df = score_company(df)
print(df)
