  import numpy as np
  import pandas as pd
  from scipy.optimize import minimize

  def portfolio_optimization(weights, returns, risk_free_rate):
      # 优化目标是最小化风险
      n = len(weights)
      def objective(x):
          return np.dot(x.T, np.dot(returns, x))
      def constraint(x):
          return sum(x) - 1
      # 使用SLSQP求解器
      result = minimize(objective, np.ones(n)/n, method='SLSQP', 
                        constraints={'type': 'eq', 'fun': constraint})
      return result.x

  # 示例数据

  weights = portfolio_optimization([0.5, 0.5], returns, 0.05)
