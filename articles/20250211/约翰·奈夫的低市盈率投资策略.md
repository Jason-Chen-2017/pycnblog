#### 2.2.2 算法实现代码
```python
def calculate_pe_ratio(stock_price, earnings_per_share):
    return stock_price / earnings_per_share

def screen_low_pe_stocks(stocks, max_pe_ratio):
    return [stock for stock in stocks if calculate_pe_ratio(stock['price'], stock['eps']) < max_pe_ratio]

def calculate_margin_of_safety(stock_price, intrinsic_value):
    return (intrinsic_value - stock_price) / intrinsic_value * 100

# 示例
stocks = [
    {'price': 50, 'eps': 10},
    {'price': 40, 'eps': 10},
    {'price': 30, 'eps': 15}
]
max_pe_ratio = 15

low_pe_stocks = screen_low_pe_stocks(stocks, max_pe_ratio)
print(low_pe_stocks)
```
