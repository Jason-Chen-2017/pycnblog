function optimize(process, constraints):
    # 初始化解空间
    solutions = initialize_solution_space(process, constraints)
    best_solution = None
    best_score = -infinity

    while not all_solutions_explored(solutions):
        # 选择下一个解决方案
        next_solution = select_solution(solutions)
        # 计算解决方案的得分
        score = evaluate_solution(next_solution, constraints)
        # 如果当前解决方案得分更高，更新最佳解决方案
        if score > best_score:
            best_solution = next_solution
            best_score = score

        # 更新解空间
        solutions = update_solution_space(solutions, next_solution)

    return best_solution
