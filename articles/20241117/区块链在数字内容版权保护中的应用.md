```solidity
pragma solidity ^0.8.0;

contract CopyrightManagement {
    mapping(address => mapping(string => bool)) public copyrights;
    
    function registerCopyright(string memory workTitle) public {
        require(copyrights[msg.sender][workTitle] == false, "版权已存在");
        copyrights[msg.sender][workTitle] = true;
    }
    
    function transferCopyright(string memory workTitle, address to) public {
        require(copyrights[msg.sender][workTitle] == true, "版权不存在");
        copyrights[msg.sender][workTitle] = false;
        copyrights[to][workTitle] = true;
    }
}
```
