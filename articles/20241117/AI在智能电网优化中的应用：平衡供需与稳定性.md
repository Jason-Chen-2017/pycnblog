    // 初始化Q值表
    Q = initialize_q_table(s)
    // 设置学习率
    alpha = 0.1
    // 设置折扣因子
    gamma = 0.9
    // 设置迭代次数
    num_iterations = 1000
    // 循环迭代
    for i in 1 to num_iterations:

        s_next, r = execute_action(s, a)
