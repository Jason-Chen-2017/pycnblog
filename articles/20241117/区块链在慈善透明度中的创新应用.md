```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract Charity {
    address public donor;
    address public recipient;
    uint public donationAmount;
    bool public isDonated;

    event DonationReceived(address donor, address recipient, uint amount);

    constructor(uint _donationAmount) {
        donor = msg.sender;
        donationAmount = _donationAmount;
        recipient = address(0);
        isDonated = false;
    }

    function donate() public payable {
        require(!isDonated, "Donation already completed");
        require(msg.value == donationAmount, "Incorrect donation amount");

        recipient = msg.sender;
        isDonated = true;
        emit DonationReceived(donor, recipient, donationAmount);
    }
}
```
