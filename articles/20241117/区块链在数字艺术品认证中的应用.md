  ```solidity
          certificateStatus[_uniqueId] = CertificateStatus.Rejected;
      }

      // 查询数字艺术品状态
      function getDigitalArtStatus(string memory _uniqueId) public view returns (CertificateStatus) {
          return certificateStatus[_uniqueId];
      }
  ```

#### 2.1 区块链的数据结构

区块链的数据结构是理解区块链技术的基础。区块链由一系列按时间顺序排列的区块组成，每个区块包含一定数量的交易记录。这些区块通过哈希函数和链接方式相互关联，形成一个不可篡改的分布式账本。

1. **区块（Block）**：区块是区块链的基本组成单位，每个区块包含以下信息：
   - **区块头**：包含版本号、随机数、前一个区块的哈希值等。
   - **交易数据**：包含多个交易记录，如数字艺术品的铸造、交易、认证等。
   - **Merkle树**：用于验证交易数据的完整性，通过哈希树结构存储交易记录的哈希值。

2. **链（Chain）**：区块链由一系列按时间顺序排列的区块组成，每个区块都通过其前一个区块的哈希值与链中的其他区块关联。这种链接方式确保了区块链的不可篡改性。

3. **链表（Linked List）**：从结构上看，区块链类似于一个链表，每个区块包含一个指向下一个区块的指针。这种链表结构使得区块链上的数据在逻辑上保持连续性。

4. **分布式账本（Distributed Ledger）**：区块链通过分布式账本技术，将数据分散存储在多个节点上。这些节点共同维护区块链的状态，确保数据的完整性和一致性。

2. **权益证明（Proof of Stake, PoS）**：PoS算法通过持有代币的数量和持币时间来决定生成新区块的优先级。持有代币越多的节点，其生成新区块的概率越大。以太坊2.0计划采用PoS算法。

4. **权威证明（Proof of Authority, PoA）**：PoA算法基于节点的身份和信誉来决定生成新区块的权力。通常，只有被授权的节点才能参与共识过程。

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract DigitalArtCertificate {
    // 定义数字艺术品结构体
    struct DigitalArt {
        string title;
        string creator;
        string uniqueId;
        bool isCertified;
    }

    // 定义认证状态
    enum CertificateStatus {
        Pending,
        Certified,
        Rejected
    }

    // 定义数字艺术品映射
    mapping(string => DigitalArt) public digitalArts;
    mapping(string => CertificateStatus) public certificateStatus;

    // 添加数字艺术品
    function addDigitalArt(string memory _uniqueId, string memory _title, string memory _creator) public {
        require(digitalArts[_uniqueId].isCertified == false, "Artwork already certified");
        digitalArts[_uniqueId] = DigitalArt(_title, _creator, _uniqueId, false);
        certificateStatus[_uniqueId] = CertificateStatus.Pending;
    }

    // 认证数字艺术品
    function certifyDigitalArt(string memory _uniqueId) public {
        require(certificateStatus[_uniqueId] == CertificateStatus.Pending, "Artwork not pending certification");
        digitalArts[_uniqueId].isCertified = true;
        certificateStatus[_uniqueId] = CertificateStatus.Certified;
    }

    // 拒绝认证数字艺术品
    function rejectDigitalArt(string memory _uniqueId) public {
        require(certificateStatus[_uniqueId] == CertificateStatus.Pending, "Artwork not pending certification");
        digitalArts[_uniqueId].isCertified = false;
        certificateStatus[_uniqueId] = CertificateStatus.Rejected;
    }

    // 查询数字艺术品状态
    function getDigitalArtStatus(string memory _uniqueId) public view returns (CertificateStatus) {
        return certificateStatus[_uniqueId];
    }
}
```
