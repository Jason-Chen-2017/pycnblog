import spacy

nlp = spacy.load("en")

def preprocess(text):
    doc = nlp(text)
    return [token.text for token in doc]

text = "今天会议讨论了下一步的市场策略，最终决定在下季度推出新产品。"
processed = preprocess(text)
print(processed)
```

### 5.2.2 信息抽取实现

from spacy.matcher import Matcher

matcher = Matcher(nlp.vocab)
patterns = [[(0, "NN"), (1, "VB"), (2, "NN")]]
matcher.add("DecisionPoint", patterns)

matches = matcher(doc)
for match_id, start, end in matches:
    print(doc[start:end].text)
```
