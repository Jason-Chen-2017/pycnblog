import pandas as pd
import numpy as np

def analyze_interview_transcript(transcript):
    # 分析访谈记录，提取关键词和主题
    keywords = ['strategy', 'growth', 'risk', 'market']
    transcript_text = ' '.join(transcript)
    word_counts = {}
    for keyword in keywords:
        count = transcript_text.lower().count(keyword)
        word_counts[keyword] = count
    return word_counts

# 示例数据
transcript = [
    "Our strategy is to expand into new markets.",
    "We believe the market has significant growth potential.",
    "We are prepared to handle any risks associated with expansion."
]
result = analyze_interview_transcript(transcript)
print(result)
```

import re

def extract_keywords(text, keywords):
    keyword_counts = {}
    for keyword in keywords:
        keyword_counts[keyword] = len(re.findall(r'\b{}\b'.format(re.escape(keyword)), text.lower()))
    return keyword_counts

# 示例
text = "Our strategy is to expand into new markets and we believe the market has significant growth potential. We are prepared to handle any risks associated with expansion."
keywords = ['strategy', 'growth', 'risk']
result = extract_keywords(text, keywords)
print(result)
```
