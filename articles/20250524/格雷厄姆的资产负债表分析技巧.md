```python
# 读取资产负债表数据
def read_financial_statement():
    df = pd.read_csv('balance_sheet.csv')
    return df

# 数据预处理
def preprocess_data(df):
    df = df.dropna()
    df['ratio'] = df['流动资产'] / df['流动负债']
    return df

# 示例数据
data = {
    '流动资产': [100, 120, 150],
    '流动负债': [60, 70, 80],
    '非流动资产': [50, 60, 70],
    '非流动负债': [20, 25, 30],
    '所有者权益': [70, 80, 90]
}

df = pd.DataFrame(data)
```

#### 7.2.2 财务比率计算与可视化
```python
import matplotlib.pyplot as plt
import seaborn as sns

# 计算流动比率
def calculate流动比率(df):
   流动资产 = df['流动资产']
   流动负债 = df['流动负债']
   流动比率 = 流动资产 / 流动负债
    return 流动比率

# 可视化流动比率
流动比率 = calculate流动比率(df)
plt.figure(figsize=(10,6))
sns.lineplot(x=df.index, y=流动比率, label='流动比率')
plt.title('流动比率变化趋势')
plt.xlabel('季度')
plt.ylabel('流动比率')
plt.show()
```
