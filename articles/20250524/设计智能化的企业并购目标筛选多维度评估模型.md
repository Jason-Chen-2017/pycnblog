  import pandas as pd
  import numpy as np

  # 读取数据
  df = pd.read_csv('target_companies.csv')
  # 删除缺失值
  df.dropna(inplace=True)
  # 标准化处理
  df = (df - df.mean()) / df.std()
  ```

- **6.2.2 权重计算模块**
  ```python
  from sklearn.decomposition import PCA

  # 主成分分析
  pca = PCA(n_components=3)
  principal_components = pca.fit_transform(df)
  # 计算各主成分的权重
  explained_variance = pca.explained_variance_ratio_
  ```

- **6.2.3 模型运行模块**
  ```python
  def calculate_score(weights, features):
      return np.dot(features, weights)
  
  # 调用模型计算评分
  scores = calculate_score(explained_variance, principal_components)
  ```

- **6.2.4 结果展示模块**

  import matplotlib.pyplot as plt

  plt.bar(range(len(scores)), scores)

  plt.show()

- **6.2.1 数据处理模块**
  ```python
  import pandas as pd
  import numpy as np

  # 读取数据
  df = pd.read_csv('target_companies.csv')
  # 删除缺失值
  df.dropna(inplace=True)
  # 标准化处理
  df = (df - df.mean()) / df.std()
  ```

- **6.2.2 权重计算模块**
  ```python
  from sklearn.decomposition import PCA

  # 主成分分析
  pca = PCA(n_components=3)
  principal_components = pca.fit_transform(df)
  # 计算各主成分的权重
  explained_variance = pca.explained_variance_ratio_
  ```

- **6.2.3 模型运行模块**
  ```python
  def calculate_score(weights, features):
      return np.dot(features, weights)
  
  # 调用模型计算评分
  scores = calculate_score(explained_variance, principal_components)
  ```

- **6.2.4 结果展示模块**

  import matplotlib.pyplot as plt

  plt.bar(range(len(scores)), scores)
  plt.xticks(range(len(scores)), ['Company 1', 'Company 2', 'Company 3'])
  plt.title('Target Company Scores')
  plt.show()
