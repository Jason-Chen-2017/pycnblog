#### 5.2 核心代码实现
```python
import tensorflow as tf
from tensorflow.keras import layers

# 多模态数据预处理
def preprocess_data(text, image):
    # 文本处理
    text_seq = tf.keras.preprocessing.text.Tokenizer().texts_to_sequences([text])[0]
    # 图像处理
    image_seq = tf.keras.preprocessing.image.load_img(image, target_size=(224, 224))
    image_seq = tf.keras.preprocessing.image.img_to_array(image_seq)
    return text_seq, image_seq
```
