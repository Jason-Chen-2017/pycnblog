def calculate创新能力_assessment(innovation, feasibility, execution, value_recognition):
    # 计算创意生成能力
    cg = (innovation + feasibility) / 2
    # 计算创意实现能力
    ch = (feasibility + execution) / 2
    # 计算创意价值实现能力
