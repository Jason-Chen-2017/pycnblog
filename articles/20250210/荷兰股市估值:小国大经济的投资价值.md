```python
import pandas as pd
import numpy as np

def calculate_pe_ratio(df):
    pe_ratio = df['Price'] / df['Earnings per Share']
    return pe_ratio

def calculate_pb_ratio(df):
    pb_ratio = df['Price'] / df['Book Value per Share']
    return pb_ratio

def calculate_dcf(df, discount_rate):
    free_cash_flow = df['Free Cash Flow']
    dcf = free_cash_flow / (1 + discount_rate)
    return dcf

# 示例数据
data = {
    'Stock': ['A', 'B', 'C'],
    'Price': [50, 60, 70],
    'Earnings per Share': [5, 6, 7],
    'Book Value per Share': [10, 12, 14],
    'Free Cash Flow': [100, 120, 140],
    'Discount Rate': [0.05, 0.05, 0.05]
}

df = pd.DataFrame(data)

pe = calculate_pe_ratio(df)
pb = calculate_pb_ratio(df)
discount_rate = df['Discount Rate'][0]
dcf = calculate_dcf(df, discount_rate)

print("市盈率:", pe)
print("市净率:", pb)
print("DCF:", dcf)
```
