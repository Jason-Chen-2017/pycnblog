import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

def social_impact_score(financial_return, environmental_impact, social_benefit):
    alpha = 0.4
    beta = 0.3
    gamma = 0.3
    return alpha * financial_return + beta * environmental_impact + gamma * social_benefit

data = pd.DataFrame({
    'Project': ['A', 'B', 'C'],
    'Financial_Return': [0.1, 0.15, 0.12],
    'Environmental_Impact': [0.2, 0.1, 0.15],
    'Social_Benefit': [0.15, 0.2, 0.18]
})

data['Social_Impact_Score'] = data.apply(lambda x: social_impact_score(x['Financial_Return'], x['Environmental_Impact'], x['Social_Benefit']), axis=1)
print(data)
