import pandas as pd
import numpy as np
from sklearn.linear_model import LinearRegression

# 数据预处理
data = pd.read_csv('stock_data.csv')
data = data[['price', 'volume', 'open', 'close']]

# 特征提取
X = data[['open', 'close']]
y = data['price']

# 模型训练
model = LinearRegression()
model.fit(X, y)

# 策略执行
predicted_price = model.predict(X)
# 判断套利机会
profit = predicted_price - data['close']
# 筛选利润为正的股票
opportunities = data[profit > 0]
