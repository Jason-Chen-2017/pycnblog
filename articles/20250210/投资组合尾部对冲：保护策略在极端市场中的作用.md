#### 3.3 算法实现代码
##### 3.3.1 Python代码实现
```python
import numpy as np
import pandas as pd
from scipy.stats import norm

def calculate_var(positions, confidence_level=0.95):
    portfolio_value = np.sum(positions)
    mu = np.mean(positions)
    sigma = np.std(positions)
    var = norm.ppf(1 - confidence_level) * sigma * portfolio_value
    return var

# 示例
positions = [100, 200, 300]
confidence_level = 0.95
var = calculate_var(positions, confidence_level)
print(f"VaR at {confidence_level} confidence level: {var}")
```

def calculate_cvar(positions, confidence_level=0.95):
    portfolio_value = np.sum(positions)
    mu = np.mean(positions)
    sigma = np.std(positions)
    cvar = -norm.pdf(norm.ppf(confidence_level)) * sigma * portfolio_value / (1 - confidence_level)
    return cvar

positions = [100, 200, 300]
