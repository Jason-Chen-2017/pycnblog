```python
def calculate_intrinsic_value(dividend, growth_rate):
    return dividend * (1 + growth_rate) ** (-1)

def graham_strategy(stocks):
    safe_stocks = []
    for stock in stocks:
        pe_ratio = stock.pe_ratio
        intrinsic_value = calculate_intrinsic_value(stock.dividend, stock.growth_rate)
        margin_of_safety = intrinsic_value - stock.price
        if margin_of_safety > 0 and pe_ratio < 15:
            safe_stocks.append(stock)
    return safe_stocks
```
