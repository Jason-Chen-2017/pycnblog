```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# 数据处理
data = pd.read_csv('stock_data.csv')
data = data.dropna()
data['Return'] = data['Close'].pct_change()

# 量化模型构建
def mean_reversion_strategy(data):
    mu = data['Return'].mean()
    sigma = data['Return'].std()
    z = (data['Close'] - mu) / sigma
    return z

z_scores = mean_reversion_strategy(data)

# 策略回测
positions = np.where(z_scores > 1, -1, np.where(z_scores < -1, 1, 0))
returns = data['Return'] * positions.sum()

plt.plot(z_scores, returns, 'r')
plt.xlabel('Z-scores')
plt.ylabel('Returns')
plt.show()
```
