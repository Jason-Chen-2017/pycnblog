def calculate_market_mood(data):
    # 计算平均收益率
    returns = data['close'].pct_change().dropna()
    avg_return = np.mean(returns)
    
    # 计算波动率
    volatility = np.std(returns)
    
    # 市场情绪指标：波动率标准化后的值
    mood_index = (avg_return - np.mean(returns)) / np.std(returns)
