### 5.3 Python实现
```python
import pandas as pd
import numpy as np

def calculate_intrinsic_value(stock_data):
    # 计算市盈率
    stock_data['P/E'] = stock_data['Price'] / stock_data['Earnings']
    # 价值评估
    stock_data['Intrinsic Value'] = stock_data['Earnings'] * (stock_data['P/E'].mean())
    return stock_data

# 示例数据
data = {
    'Stock': ['A', 'B', 'C'],
    'Price': [100, 50, 75],
    'Earnings': [20, 10, 15]
}
df = pd.DataFrame(data)
result = calculate_intrinsic_value(df)
print(result)
```
