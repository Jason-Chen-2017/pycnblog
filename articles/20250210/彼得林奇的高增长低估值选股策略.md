  def calculate_growth_rates(finance_data):
      # 计算净利润增长率
      growth_rates = []
      for i in range(1, len(finance_data)):
          growth = (finance_data[i] - finance_data[i-1]) / finance_data[i-1] * 100
          growth_rates.append(growth)
      return growth_rates

  def calculate_valuation_metrics(stock_data):
      # 计算市盈率
      pe_ratio = stock_data['股价'] / stock_data['每股收益']
      return pe_ratio

  def综合评分(growth_score, valuation_score):

- 净利润增长率
  $$\text{净利润增长率} = \frac{\text{本期净利润} - \text{上期净利润}}{\text{上期净利润}} \times 100\%$$

  $$\text{市盈率} = \frac{\text{股价}}{\text{每股收益}}$$
