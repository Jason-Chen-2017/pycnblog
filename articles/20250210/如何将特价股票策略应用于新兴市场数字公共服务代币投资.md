import pandas as pd
import numpy as np

# 示例数据
data = pd.DataFrame({'price': [100, 95, 90, 85, 80]})

# 计算目标价格
alpha = 0.2
target_price = data['price'] * (1 - alpha)
data['signal'] = np.where(data['price'] < target_price, 'buy', 'hold')

print(data)
