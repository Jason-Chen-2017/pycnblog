###### 5.2.1 支付网关实现

```python
class PaymentGateway:
    def __init__(self):
        self.routes = {}

    def add_route(self, source, destination, cost):
        self.routes[(source, destination)] = cost

    def find_optimal_route(self, start, end):
        min_cost = float('inf')
        best_route = None
        for route in self.routes:
            if route[0] == start and route[1] == end:
                if self.routes[route] < min_cost:
                    min_cost = self.routes[route]
                    best_route = route
        return best_route
```

def calculate_total_cost(routes, path):
    total = 0
    for i in range(len(path) - 1):
        total += routes[(path[i], path[i+1])]
    return total
