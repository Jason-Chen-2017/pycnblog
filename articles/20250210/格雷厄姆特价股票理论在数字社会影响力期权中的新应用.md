```python
def black_scholes(option_type, S, K, r, T, sigma):
    # 计算d1和d2
    d1 = (ln(S/K) + (r + 0.5*sigma**2)*T) / (sigma*sqrt(T))
    d2 = d1 - sigma*sqrt(T)
    
    # 计算N(d1)和N(d2)
    N_d1 = cumulative_distribution(d1)
    N_d2 = cumulative_distribution(d2)
    
    if option_type == 'call':
        price = S * e^(-rT) * N_d1 - K * e^(-rT) * N_d2
    else:
        price = K * e^(-rT) * N_d2 - S * e^(-rT) * N_d1
        
    return price
```
