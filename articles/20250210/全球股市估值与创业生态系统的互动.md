```python
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression

# 数据加载
data = pd.read_csv('data.csv')

# 模型训练
model = LinearRegression()
model.fit(data[['创业生态系统指数', '宏观经济指标']], data['股市估值'])

# 结果展示
plt.scatter(data['创业生态系统指数'], data['股市估值'])
plt.plot(data['创业生态系统指数'], model.predict(data[['创业生态系统指数', '宏观经济指标']]), color='red')
plt.xlabel('创业生态系统指数')
plt.ylabel('股市估值')
plt.show()
```
