```python
import numpy as np
import pandas as pd

def calculate_intrinsic_value(cash_flow, discount_rate):
    # 计算净现值
    npv = np.npv(discount_rate, cash_flow)
    return npv

# 示例数据
cash_flow = [-100, 50, 60, 70]
discount_rate = 0.1

# 计算内在价值
intrinsic_value = calculate_intrinsic_value(cash_flow, discount_rate)
print("内在价值:", intrinsic_value)
```
