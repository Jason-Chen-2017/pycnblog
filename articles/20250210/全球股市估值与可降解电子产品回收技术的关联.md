### 10.2 Python源代码实现

```python
import pandas as pd
import statsmodels.api as sm

# 数据加载
data = pd.read_csv('data.csv')

# 变量选择
X = data[['环保政策', '技术创新']]
X = sm.add_constant(X)
Y = data['企业估值']

# 回归分析
model = sm.OLS(Y, X)
results = model.fit()

# 输出结果
print(results.summary())
```
