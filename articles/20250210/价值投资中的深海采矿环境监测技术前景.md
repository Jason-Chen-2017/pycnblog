import numpy as np
import pandas as pd

def data_cleaning(data):
    # 删除缺失值
    data = data.dropna()
    # 标准化处理
    data = (data - data.mean()) / data.std()
    return data

# 示例数据
data = pd.DataFrame({
    '水质指标': [10, 20, 30, np.nan, 50],
    '沉积物指标': [5, 15, 25, 35, np.nan],
    '生物指标': [1, 2, 3, 4, 5]
})

# 数据清洗
cleaned_data = data_cleaning(data)
print(cleaned_data)
