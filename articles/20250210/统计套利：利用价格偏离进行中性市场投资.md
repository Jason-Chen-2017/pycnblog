import pandas as pd

# 示例代码：计算价差和均值回归信号
def calculate_price_diff(close_prices):
    return close_prices - close_prices.rolling(2).mean()

def generate_mean_reversion_signal(price_diff, threshold=1):
    return np.where(price_diff > threshold, 'buy', np.where(price_diff < -threshold, 'sell', 'hold'))

# 示例数据
data = pd.DataFrame({
    'close_prices': [100, 102, 98, 105, 101]
})
price_diff = calculate_price_diff(data['close_prices'])
signal = generate_mean_reversion_signal(price_diff)
