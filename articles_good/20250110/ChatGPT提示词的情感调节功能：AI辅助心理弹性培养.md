import openai
import json

def emotional_adjustment(prompt, target_emotion):
    response = openai.Completion.create(
        engine="text-davinci-002",
        prompt=prompt,
        max_tokens=100,
        n=1,
        stop=None,
        temperature=0.5,
    )
    
    if target_emotion == "positive":
        adjustment_prompt = "更积极地描述这段话："
    elif target_emotion == "negative":
        adjustment_prompt = "更消极地描述这段话："
    else:
        adjustment_prompt = "保持原样："
    
    adjusted_response = openai.Completion.create(
        engine="text-davinci-002",
        prompt=f"{adjustment_prompt}{response.choices[0].text}",
        max_tokens=100,
        n=1,
        stop=None,
        temperature=0.5,
    )
    
    return adjusted_response.choices[0].text.strip()

# 示例
prompt = "我喜欢这个周末的天气。"
adjusted_prompt = emotional_adjustment(prompt, "positive")
print(adjusted_prompt)
