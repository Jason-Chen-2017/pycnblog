from tensorflow.keras.layers import Input, Dense, Reshape, Flatten, Conv2D, BatchNormalization, LeakyReLU

# 生成器模型
input_z = Input(shape=(100,))
x = Dense(128)(input_z)
x = BatchNormalization()(x)
x = LeakyReLU()(x)
x = Dense(256)(x)
x = BatchNormalization()(x)
x = LeakyReLU()(x)
x = Dense(512)(x)
x = BatchNormalization()(x)
x = LeakyReLU()(x)
x = Dense(1024)(x)

generator = Model(input_z, output)

# 判别器模型
input_x = Input(shape=(28, 28))
x = Flatten()(input_x)
x = Dense(512)(x)
x = BatchNormalization()(x)
x = LeakyReLU()(x)
x = Dense(256)(x)
x = BatchNormalization()(x)
x = LeakyReLU()(x)
x = Dense(128)(x)
x = BatchNormalization()(x)
x = LeakyReLU()(x)
output = Dense(1, activation='sigmoid')(x)

discriminator = Model(input_x, output)

# GAN模型
gan_output = discriminator(generator(input_z))
gan = Model(input_z, gan_output)

# 模型编译
