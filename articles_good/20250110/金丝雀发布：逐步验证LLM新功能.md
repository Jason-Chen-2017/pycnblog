import random
import string

def generate_random_string(length=10):
    """生成指定长度的随机字符串"""
    letters = string.ascii_letters
    return ''.join(random.choice(letters) for i in range(length))

def simulate_user_interaction(content_model):
    """模拟用户交互"""
    for _ in range(100):
        random_string = generate_random_string()
        content_model.generate_response(random_string)
        print(f"User input: {random_string}")
        print(f"Model response: {content_model.get_last_response()}")

class ContentModel:
    """内容模型类"""
    
    def __init__(self):
        self.responses = []

    def generate_response(self, input_text):
        """生成响应"""
        self.responses.append(input_text)

    def get_last_response(self):
        """获取最后响应"""
        return self.responses[-1]
