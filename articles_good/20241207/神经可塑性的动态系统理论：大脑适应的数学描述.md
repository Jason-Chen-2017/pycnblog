import numpy as np

# 初始化参数
learning_rate = 0.1
input_signal = np.array([1, 0, 1])
target_signal = np.array([0, 1, 0])
weights = np.random.rand(3, 3)

# 获取输入信号
def get_input_signal():
    return input_signal

# 调整突触权重
def adjust_weights(signal, weights):
    delta_weights = learning_rate * signal * weights
    return weights + delta_weights

# 更新神经网络状态
def update_neural_network(weights):
    output_signal = np.dot(weights.T, input_signal)
    return output_signal

# 计算输出信号与目标信号误差
def calculate_error(output_signal, target_signal):
    error = np.square(output_signal - target_signal)
    return error

# 主循环
while True:
    signal = get_input_signal()
    weights = adjust_weights(signal, weights)
    output_signal = update_neural_network(weights)
    error = calculate_error(output_signal, target_signal)
    
    if np.linalg.norm(error) < 1e-6:
        break

print("最终权重：", weights)
print("输出信号：", output_signal)
print("误差：", error)
```

#### 4.2 推导过程

**推导突触权重调整公式**：

神经可塑性的核心在于通过调整突触权重来实现神经网络的学习和适应。假设我们有一个神经网络，其中每个神经元 $i$ 都与其它神经元 $j$ 通过突触相连。突触权重 $w_{ij}$ 用于衡量神经元 $i$ 和神经元 $j$ 之间的连接强度。

当神经元 $i$ 收到输入信号 $s_i$ 时，其状态将更新为：

$$
s_i = \sum_{j=1}^{n} w_{ij} \cdot x_j
$$

其中，$x_j$ 是神经元 $j$ 的输入信号。

如果目标信号为 $t_i$，则误差 $e_i$ 可以表示为：

$$
e_i = (s_i - t_i)^2
$$

为了最小化误差，我们需要调整突触权重 $w_{ij}$。根据误差反馈，我们可以得到突触权重调整公式：

$$
\Delta w_{ij} = \eta \cdot (s_i - t_i) \cdot x_j
$$

其中，$\eta$ 是学习率，用于控制权重的调整速度。

**推导神经网络状态更新公式**：

神经网络状态更新是通过调整突触权重来实现的。假设神经网络中有 $n$ 个神经元，每个神经元的状态更新公式为：

$$
s_i(\text{new}) = \sum_{j=1}^{n} w_{ij} \cdot x_j
$$

其中，$s_i(\text{new})$ 是神经元 $i$ 的新状态。

**推导误差计算公式**：

误差计算用于评估神经网络的状态与目标状态之间的差异。假设神经网络中有 $n$ 个神经元，每个神经元的误差计算公式为：

$$
e_i = (s_i(\text{new}) - t_i)^2
$$

总误差 $E$ 可以表示为所有神经元误差的加和：

$$
E = \sum_{i=1}^{n} e_i

        return x

# 实例化神经网络

def train(model, criterion, optimizer, epochs, train_loader):
    model.train()
    for epoch in range(epochs):
        for inputs, targets in train_loader:
            optimizer.zero_grad()
            outputs = model(inputs)
            loss = criterion(outputs, targets)
            loss.backward()
            optimizer.step()
            if (epoch + 1) % 10 == 0:
                print(f'Epoch [{epoch + 1}/{epochs}], Loss: {loss.item()}')

# 生成训练数据
train_data = np.random.rand(100, 3)
train_targets = np.array([[0, 1, 0], [1, 0, 0], [0, 0, 1]])

# 训练模型
train(model, criterion, optimizer, epochs=50, train_loader=train_loader)

# 测试神经网络
def test(model, criterion, test_loader):
    model.eval()
    with torch.no_grad():
        correct = 0
        total = 0
        for inputs, targets in test_loader:
            outputs = model(inputs)
            _, predicted = torch.max(outputs.data, 1)
            total += targets.size(0)
            correct += (predicted == targets).sum().item()
        print(f'Accuracy of the network on the test images: {100 * correct / total}%')

# 生成测试数据

test_loader = torch.utils.data.DataLoader(torch.tensor(test_data), batch_size=10)

# 测试模型
test(model, criterion, test_loader)
```
