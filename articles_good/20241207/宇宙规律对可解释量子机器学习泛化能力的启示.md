qc.h(2)
qc.h(3)

qc.cx(0, 1)
qc.cx(1, 2)
qc.cx(2, 3)

result = execute(qc, backend).result()

# 获取量子态结果

# 转换量子态为特征

# 使用经典计算对特征进行分类
