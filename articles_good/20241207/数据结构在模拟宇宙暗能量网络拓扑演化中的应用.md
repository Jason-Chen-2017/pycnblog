def initialize_stars(num_stars, pos_range=(-1, 1)):
    stars = nx.Graph()
    for i in range(num_stars):
        x, y = np.random.uniform(*pos_range), np.random.uniform(*pos_range)
        stars.add_node(i, position=(x, y))
    return stars

# 更新星系位置
def update_positions(stars, time_step=1):
    for node in stars.nodes():
        x, y = stars.nodes[node]['position']
        stars.nodes[node]['position'] = (x + np.random.normal(0, time_step), y + np.random.normal(0, time_step))

# 计算引力相互作用
def calculate_gravity(stars, G=6.67430e-11, m=1.989e30):
    for node in stars.nodes():
        node_force = np.zeros(2)
        for neighbor in stars.neighbors(node):
            dx = stars.nodes[neighbor]['position'][0] - stars.nodes[node]['position'][0]
            dy = stars.nodes[neighbor]['position'][1] - stars.nodes[node]['position'][1]
            distance = np.sqrt(dx**2 + dy**2)
            force = G * m**2 / distance**2
            node_force += (force * dx, force * dy)
        stars.nodes[node]['force'] = node_force

# 更新星系速度
def update_velocities(stars, time_step=1):
    for node in stars.nodes():
        force = stars.nodes[node]['force']
        stars.nodes[node]['velocity'] = (force[0] / stars.nodes[node]['mass'], force[1] / stars.nodes[node]['mass'])

# 演化模拟
def simulate(stars, steps=100):
    for step in range(steps):
        update_positions(stars, time_step=1)
        calculate_gravity(stars)
        update_velocities(stars, time_step=1)
        print(f"Step {step}: Position updated and forces calculated.")

# 绘制网络拓扑
def draw_stars(stars):
    pos = nx.get_node_attributes(stars, 'position')
    nx.draw(stars, pos, with_labels=True)
    plt.show()

# 测试
if __name__ == "__main__":
    stars = initialize_stars(100)
    simulate(stars)
    draw_stars(stars)
