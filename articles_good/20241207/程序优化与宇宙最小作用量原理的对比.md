宇宙最小作用量原理（Principle of Least Action）是物理学中描述自然现象的基本原理之一。它指出，一个物体在从初始状态到最终状态的演化过程中，其作用量的积分值最小。

```python
from django.db import models

class Warehouse(models.Model):

class Destination(models.Model):
    name = models.CharField(max_length=100)

class Path(models.Model):

    destination = models.ForeignKey(Destination, on_delete=models.CASCADE)
    distance = models.FloatField()
```

#### 视图函数

```python
# views.py
from django.http import JsonResponse
from .models import Path
from geopy.distance import geodesic

def calculate_path(request):
    start = request.GET.get('start', '')
    end = request.GET.get('end', '')

    warehouse = Warehouse.objects.get(name=start)
    destination = Destination.objects.get(name=end)

    distance = geodesic(warehouse.location, destination.location).kilometers

    path = Path.objects.create(
        warehouse=warehouse,
        destination=destination,
        distance=distance
    )

    return JsonResponse({'distance': distance})
```
