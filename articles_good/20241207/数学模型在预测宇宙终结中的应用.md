```python
# 导入必要的库
import numpy as np
import scipy.integrate as integrate
import matplotlib.pyplot as plt
from mermaid import Mermaid

# 弗里德曼方程的参数
G = 6.674 * 10**-11  # 引力常数
k = -4 * np.pi       # 宇宙曲率参数（假设为平坦宇宙）

# 初始条件
a0 = 1               # 初始尺度因子
H0 = 100             # 初始哈勃参数

# 迭代次数
n_iterations = 1000

# 迭代计算弗里德曼方程
a_history = [a0]
H_history = [H0]

for _ in range(n_iterations):
    a = a_history[-1]
    H = H0 * np.sqrt((8 * np.pi * G) / (3 * a) - (k / a ** 2))
    a_history.append(a)
    H_history.append(H)

# 绘制结果
plt.plot(a_history, H_history)
plt.xlabel('Scale Factor (a)')
plt.ylabel('Hubble Parameter (H)')
plt.title('Friedmann Equation Solution')
plt.show()

# 熵增模型的微分方程
def entropy_model(S, t, rho0, H0, Q0):
    dSdt = (1/3) * (rho0 * H0 - Q0)
    return dSdt

# 初始条件
S0 = 0               # 初始熵

# 时间步长和总时间
timesteps = 100
t = np.linspace(0, 10, timesteps)

# 数值积分求解熵增模型
S_history = integrate.odeint(entropy_model, S0, t, args=(rho0, H0, Q0))

# 绘制结果
plt.plot(t, S_history)
plt.xlabel('Time (t)')
plt.ylabel('Entropy (S)')
plt.title('Entropy Increase Model')
plt.show()

# Mermaid流程图
mermaid = Mermaid()
mermaid.add(
    "graph TD\n"
    "A[初始化参数] --> B[计算尺度因子]\n"
    "B --> C[计算哈勃参数]\n"
    "C --> D[更新结果]\n"
    "D --> E[迭代计算]\n"
    "E --> B\n"
)

print(mermaid.render())
```
