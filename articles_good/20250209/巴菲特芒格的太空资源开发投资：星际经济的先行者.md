import numpy as np
import pandas as pd

# 假设我们有多个资产的收益率数据
returns = pd.DataFrame({
    'Asset1': [0.1, 0.05, 0.02],
    'Asset2': [0.08, 0.04, 0.01],
    'Asset3': [0.12, 0.06, 0.03]
})

# 计算协方差矩阵
cov_matrix = returns.cov()

# 计算各资产的期望收益
expected_returns = returns.mean()

# 使用马科维茨模型优化投资组合
from pypfopt import EfficientFrontier

ef = EfficientFrontier(n_assets=len(expected_returns), mu=expected_returns, sigma=cov_matrix)
weights = ef.max_sharpe()
