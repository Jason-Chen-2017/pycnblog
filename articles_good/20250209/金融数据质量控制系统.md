def data_cleaning(data, rules):
    cleaned_data = data.copy()
    for rule in rules:
        if rule['type'] == 'drop_duplicates':
            cleaned_data.drop_duplicates(inplace=True)
        elif rule['type'] == 'fillna':
            cleaned_data = cleaned_data.fillna(rule['value'])
    return cleaned_data
```

import pandas as pd

def detect_outliers(data, column, threshold=3):
    z_scores = (data[column] - data[column].mean()) / data[column].std()
    filtered_data = data[abs(z_scores) < threshold]
    return filtered_data

def isolation_forest_cleaning(X, contamination=0.05):
    model = IsolationForest(contamination=contamination)
    model.fit(X)
    outliers = model.predict(X) == -1
    cleaned_data = X[~outliers]
    return cleaned_data

import pandas as pd

def calculate_quality_metrics(data):
    metrics = {}
    # 数据完整性
    completeness = data.dropna().shape[0] / data.shape[0]
    metrics['completeness'] = completeness
    # 数据准确性
    accuracy = 1.0  # 假设数据是正确的
    metrics['accuracy'] = accuracy
    # 数据一致性
    consistency = data.duplicated().sum() / data.shape[0]
    metrics['consistency'] = consistency
    # 数据及时性
    timeliness = 1.0  # 假设数据是及时的
    metrics['timeliness'] = timeliness
    return metrics
```
