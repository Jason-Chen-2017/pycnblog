##### 1.1.1 当前国际贸易摩擦的主要表现形式

```mermaid
}
```

```python
# 预测结果
new_data = pd.DataFrame({
    '市场潜力': [0.7],
    '政策风险': [0.3]
})
prediction = model.predict(new_data[['市场潜力', '政策风险']])
print("预测企业资源评分：", prediction[0])
```
