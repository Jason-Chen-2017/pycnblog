```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract SupplyChainFinance {
    struct LoanApplication {
        address borrower;
        uint amount;
        bool status;
    }

    LoanApplication[] public applications;

    function applyLoan(address _borrower, uint _amount) external {
        applications.push(LoanApplication({borrower: _borrower, amount: _amount, status: false}));
    }

    function approveLoan(uint _index) external {
        applications[_index].status = true;
    }

    function fundLoan(uint _index) external payable {
        require(applications[_index].status == true, "Loan not approved");
        require(msg.value == applications[_index].amount, "Amount mismatch");
        payable(applications[_index].borrower).transfer(applications[_index].amount);
    }
}
```
