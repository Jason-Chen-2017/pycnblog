```python
import numpy as np
import tensorflow as tf

# 定义DQN网络
class DQNAgent:
    def __init__(self, state_space, action_space, learning_rate):
        self.state_space = state_space
        self.action_space = action_space
        self.learning_rate = learning_rate
        # 状态空间输入层
        self.inputs = tf.keras.Input(shape=(state_space,))
        # 隐藏层
        self.hidden1 = tf.keras.layers.Dense(24, activation='relu')(self.inputs)
        self.hidden2 = tf.keras.layers.Dense(24, activation='relu')(self.hidden1)
        # 输出层
        self.outputs = tf.keras.layers.Dense(action_space, activation='linear')(self.hidden2)
        # 编译模型
        self.model = tf.keras.Model(inputs=self.inputs, outputs=self.outputs)
        self.model.compile(optimizer=tf.keras.optimizers.Adam(learning_rate), loss='mse')

    # 动作选择
    def act(self, state):
        state = np.array(state)
        prediction = self.model.predict(state[np.newaxis])
        return np.argmax(prediction[0])

    # 模型训练
    def train(self, states, targets):
        self.model.fit(states, targets, epochs=1, verbose=0)
```
