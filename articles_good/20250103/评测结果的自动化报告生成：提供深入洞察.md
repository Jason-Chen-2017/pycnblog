from sklearn.ensemble import RandomForestRegressor

from flask import Flask, request, jsonify

app = Flask(__name__)

    df.to_csv(f"{employee_id}.csv", index=False)
    
    return jsonify({"message": "Data collected successfully."})

from flask import Flask, request, jsonify

app = Flask(__name__)

# 数据采集接口
@app.route('/api/collect_sales_data', methods=['POST'])
def collect_sales_data():
    data = request.get_json()
    sales_data = data['salesData']
    df = pd.DataFrame(sales_data)
    df.to_csv('sales_data.csv', index=False)
    return jsonify({'message': 'Sales data collected successfully.'})

# 数据分析接口
@app.route('/api/analyze_sales_data', methods=['GET'])
def analyze_sales_data():
    df = pd.read_csv('sales_data.csv')
    metrics = {
        'total_sales': df['sales'].sum(),
        'average_sales': df['sales'].mean(),
        'max_sales': df['sales'].max(),
        'min_sales': df['sales'].min()
    }
    return jsonify(metrics)

# 报告生成接口

def generate_performance_report():
