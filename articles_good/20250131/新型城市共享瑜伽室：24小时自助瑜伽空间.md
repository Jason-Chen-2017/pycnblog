```python
features = data[['login_time', 'use_frequency', 'preference_area', 'device_use_frequency']]
labels = data['next_use_time']

# 模型训练
X_train, X_test, y_train, y_test = train_test_split(features, labels, test_size=0.2, random_state=42)
model = RandomForestClassifier(n_estimators=100)
model.fit(X_train, y_train)

# 模型评估
y_pred = model.predict(X_test)
accuracy = accuracy_score(y_test, y_pred)
print(f"Model accuracy: {accuracy:.2f}")

# 预测应用
new_user_data = [[10, 5, 1, 3]]  # 示例数据
new_user_prediction = model.predict(new_user_data)
print(f"New user predicted next use time: {new_user_prediction[0]}")
```
