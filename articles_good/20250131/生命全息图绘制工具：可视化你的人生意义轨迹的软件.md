```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import pymysql

# 数据采集
def collect_data():
    # 连接数据库
    connection = pymysql.connect(host='localhost', user='root', password='password', database='life_hologram')
    # 查询数据
    cursor = connection.cursor()
    cursor.execute('SELECT * FROM phys_data')
    data = cursor.fetchall()
    cursor.close()
    connection.close()
    # 转换为Pandas DataFrame
    data = pd.DataFrame(data, columns=['id', 'user_id', 'data_type', 'value'])
    return data

# 数据处理
def process_data(data):
    # 数据清洗
    cleaned_data = data[data['data_type'] == 'age']
    # 数据分析
    stats = cleaned_data.describe()
    # 可视化展示
    plt.figure(figsize=(10, 6))
    plt.plot(cleaned_data['id'], cleaned_data['value'], 'o-')
    plt.xlabel('ID')
    plt.ylabel('Age')
    plt.title('Age Distribution')
    plt.show()

# 主程序
if __name__ == '__main__':
    data = collect_data()
    process_data(data)
```
