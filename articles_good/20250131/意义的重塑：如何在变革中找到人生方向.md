from flask import Flask, request, jsonify

app = Flask(__name__)

@app.route('/sense_change', methods=['POST'])
def sense_change():
    data = request.get_json()
    change_type = data['change_type']
    if change_type == 'positive':
        return jsonify({"result": "积极变革"}), 200
    elif change_type == 'negative':
        return jsonify({"result": "消极变革"}), 200
    else:
        return jsonify({"error": "未知变革类型"}), 400

if __name__ == '__main__':
    app.run(debug=True)
