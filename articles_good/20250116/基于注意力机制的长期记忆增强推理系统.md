```python
import tensorflow as tf
from tensorflow.keras.models import Model
from tensorflow.keras.layers import Input, LSTM, Dense, Embedding, TimeDistributed

def create_long_memory_model(input_shape, embedding_dim, hidden_units, output_size):
    # 输入层
    input_seq = Input(shape=input_shape)

    # 嵌入层
    embedded_seq = Embedding(input_dim=input_size, output_dim=embedding_dim)(input_seq)

    # LSTM层
    lstm_output, state_h, state_c = LSTM(hidden_units, return_sequences=True, return_state=True)(embedded_seq)

    # 注意力机制层
    attention_weights = Dense(1, activation='tanh')(lstm_output)
    attention_weights = tf.nn.softmax(attention_weights, axis=1)
    context_vector = tf.reduce_sum(attention_weights * lstm_output, axis=1)

    # 全连接层
    output = Dense(output_size, activation='softmax')(context_vector)

    # 构建模型
    model = Model(inputs=input_seq, outputs=output)

    # 编译模型
    model.compile(optimizer='adam', loss='categorical_crossentropy', metrics=['accuracy'])

    return model
```
