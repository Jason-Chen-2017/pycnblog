```python
def self_cognition_score(name, age):
    # 根据名字和年龄计算自我认知得分
    return len(name) * age

def self_acceptance_score(name, age):
    # 根据名字和年龄计算自我接纳得分
    return len(name) * (100 - age)

def emotion_management_score(name, age):
    # 根据名字和年龄计算情感管理得分
    return 100 - (len(name) * age)

# 测试算法
name = "张三"
age = 30
print("自我认知得分：", self_cognition_score(name, age))
print("自我接纳得分：", self_acceptance_score(name, age))
print("情感管理得分：", emotion_management_score(name, age))
```
