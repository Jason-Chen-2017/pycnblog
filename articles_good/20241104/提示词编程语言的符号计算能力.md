def parse_linear_equation(input_str):
    # 使用正则表达式解析输入的线性方程
    match = re.match(r'([-+]?\d+)x([-+]?(\d+))y=([-+]?\d+)', input_str)
    if match:
        a, b, c = match.groups()
        return sympy.sympify(a), sympy.sympify(b), sympy.sympify(c)
    else:
        raise ValueError("输入的方程格式不正确")

def solve_linear(equations):

logic_input = pb.input(prompt_logic).validate(r'^[A-Z] ∧ [A-Z] → [A-Z]$')

# 解析逻辑推理命题
def parse_logic(input_str):
    # 使用正则表达式解析输入的逻辑命题
    match = re.match(r'([A-Z]) ∧ ([A-Z]) → ([A-Z])', input_str)
    if match:
        p, q, r = match.groups()
        return sympy.Symbol(p), sympy.Symbol(q), sympy.Symbol(r)
    else:
        raise ValueError("输入的命题格式不正确")

# 进行逻辑推理
def logic_inference(p, q, r):
    return sympy.imply(p & q, r)

# 提示3：进行符号替换
prompt_sub = "请输入一个包含变量的表达式，格式如下：\nP(x) = x^2 + 2x + 1"

    match = re.match(r'([A-Z]\((\S+)\)) = (.+)$', input_str)
    if match:
        var, expr = match.groups()
        return sympy.Symbol(var), sympy.sympify(expr)
    else:
        raise ValueError("输入的表达式格式不正确")

# 进行符号替换
def substitute_variable(var, expr, value):
    return expr.subs(sympy.Symbol(var), sympy.sympify(value))
