# 定义参考文献格式化的函数
def format_references(reference_list, citation_style):
    formatted_references = []
    for reference in reference_list:
        if citation_style == "APA":
            formatted_references.append(f"{reference['author']}. ({reference['year']}). {reference['title']}. {reference['source']}.")
        elif citation_style == "MLA":
            formatted_references.append(f"{reference['author'].title()}. \"{reference['title']}.\" {reference['source'], reference['volume']}, no. {reference['issue'], reference['page_range'], {reference['year']}.")
        else:
            formatted_references.append(f"{reference['author'].title()}, \"{reference['title']}\", {reference['source']}, {reference['year']}.")
    return formatted_references

# 获取参考文献格式的输入
reference_list = [
    {"author": "Smith, John", "year": 2020, "title": "Effective Communication in the Workplace", "source": "Journal of Business Studies"},
    {"author": "Doe, Jane", "year": 2019, "title": "The Importance of Team Collaboration", "source": "International Journal of Management"},
]

# 调用参考文献格式化函数
formatted_references_result = format_references(reference_list, "APA")

# 输出参考文献格式化结果
for reference in formatted_references_result:
