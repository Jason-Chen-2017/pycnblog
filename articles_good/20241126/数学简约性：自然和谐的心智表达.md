#### 3.1 数学简约性的基本原则

数学简约性的基本原则包括以下几方面：

1. **简化性**：通过简化和抽象的方法，将复杂的数学表达式和算法转化为更简洁的形式。
2. **抽象性**：从具体问题中提取出普遍规律和共性，使问题更加一般化和可扩展。
3. **直观性**：使数学问题和算法更加直观和易于理解。

def compute_cost(X, y, theta):
    m = len(y)
    h = X.dot(theta)
    J = (1 / (2 * m)) * ((h - y)**2).sum()
    return J

# 计算梯度
def compute_gradient(X, y, theta):
    m = len(y)
    h = X.dot(theta)
    gradients = (1 / m) * (X.T.dot(h - y))
    return gradients

# 梯度下降法
def gradient_descent(X, y, theta, alpha, iters):
    m = len(y)
    for i in range(iters):
        h = X.dot(theta)
        theta = theta - alpha * (X.T.dot(h - y))
        J = compute_cost(X, y, theta)
        if i % 100 == 0:
            print(f"迭代次数：{i}, 代价函数值：{J}")
    return theta

# 超参数设置

iters = 1000
