```python
import numpy as np
import matplotlib.pyplot as plt

# 初始化细胞自动机网格
def initialize_grid(width, height, alive_prob):
    grid = np.random.rand(width, height)
    grid = (grid < alive_prob) * 1
    return grid

# 细胞自动机规则
def apply_rules(grid):
    new_grid = grid.copy()
    for i in range(grid.shape[0]):
        for j in range(grid.shape[1]):
            alive_neighbors = np.sum(grid[i-1:i+2, j-1:j+2]) - grid[i, j]
            if grid[i, j] == 1 and (alive_neighbors < 2 or alive_neighbors > 3):
                new_grid[i, j] = 0
            elif grid[i, j] == 0 and alive_neighbors == 3:
                new_grid[i, j] = 1
    return new_grid

# 运行细胞自动机
def run_game_of_life(grid, generations):
    for _ in range(generations):
        grid = apply_rules(grid)
    return grid

# 绘制细胞自动机结果
def plot_grid(grid):
    plt.imshow(grid, cmap='gray')
    plt.show()

# 主程序
if __name__ == "__main__":
    width, height = 100, 100
    alive_prob = 0.5
    grid = initialize_grid(width, height, alive_prob)
    generations = 100
    new_grid = run_game_of_life(grid, generations)
    plot_grid(new_grid)
```

- **生态系统模型**：这类模型关注更大尺度上的生物群落和生态过程，如食物网模型、景观模型和气候变化模型。
