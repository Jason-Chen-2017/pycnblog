自组织性（Self-Organization）是指在无外部控制的情况下，系统内部通过相互作用自发形成有序结构和功能的过程。在宇宙中，自组织性表现为从简单的初始条件出发，逐渐演化出复杂的结构，如星系、行星和生命。这个过程不仅发生在宏观宇宙层面，还体现在微观世界中，如生物体的自组织现象。

def koch_curve(level, x, y):
    if level == 0:
        return [(x, y)]
    delta = 60 / 3**level
    angles = [0, delta, 2*delta, 3*delta]
    points = []
    for angle in angles:
        dx = 3**level * np.cos(np.deg2rad(angle))
        dy = 3**level * np.sin(np.deg2rad(angle))
        points.extend(koch_curve(level-1, x+dx, y+dy))
    return points

    plt.axis('equal')
    plt.title('Koch Curve')

import matplotlib.pyplot as plt

    r = np.sqrt(r2)
    V = -G * M / r
    return V

def gravitational_force(x, y, M):

    r = np.sqrt(r2)
    f = -G * M * np.array([x, y]) / r2
    return f

import matplotlib.pyplot as plt

# 定义细胞自动机规则
def cell_automaton(grid, rule):
    new_grid = grid.copy()
    for i in range(len(grid)):
        for j in range(len(grid[0])):
            neighbors = np.sum(grid[i-1:i+2, j-1:j+2])
            new_grid[i, j] = rule[neighbors]
    return new_grid

# 初始化网格和规则
grid_size = 100
initial_grid = np.random.choice([0, 1], size=(grid_size, grid_size))
rule = [0, 1, 1, 0, 1, 0, 0, 1]

# 模拟细胞自动机
steps = 100
for step in range(steps):
    initial_grid = cell_automaton(initial_grid, rule)

# 绘制细胞自动机
plt.imshow(initial_grid, cmap='gray')
plt.title(f'Step {step}')
plt.xlabel('X')
plt.ylabel('Y')
plt.show()
