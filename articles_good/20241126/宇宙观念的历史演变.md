```python
# Python源代码：模拟牛顿的宇宙观念
import math

def gravitational_force(m1, m2, r):
    G = 6.674 * 10**-11
    F = G * (m1 * m2) / r**2
    return F

# 创建实例

mass_moon = 7.348 * 10**22
distance_earth_moon = 3.844 * 10**8

# 计算地球和月球之间的引力
```
