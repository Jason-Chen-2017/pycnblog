2. 行星与太阳的连线在相等时间内扫过相等的面积。
3. 行星绕太阳公转周期的平方与其轨道半长轴的立方成正比。

```python
import math

def simulate_planet_orbit(G, M, r):
    t = 0
    dt = 0.01
    v = math.sqrt(G * M / r)
    theta = 0

    while theta < 2 * math.pi:
        print(f"Time: {t}, Theta: {theta}, r: {r}")
        r = r * math.cos(theta) + M / (2 * G * math.sin(theta))
        theta += v * dt
        t += dt

    return r
