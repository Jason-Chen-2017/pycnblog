    """
    使用数学直觉来计算矩形的面积。

    :return: 矩形的面积
    """

    使用数学直觉来预测三角形的面积。
    :param side1: 三角形的第一条边

    模拟二项分布实验。
    :param n: 试验次数
    :param p: 单次试验成功的概率
    :return: 成功的次数
    """
    successes = sum(random.random() < p for _ in range(n))
    return successes

n = 100
p = 0.5
successes = simulate_probability_experiment(n, p)
print(f"实验中成功的次数：{successes}")

```python
import pandas as pd
import matplotlib.pyplot as plt

# 读取数据
data = pd.read_csv('员工薪资数据.csv')

# 数据预处理
data['薪资'] = data['薪资'].astype(float)
data.dropna(inplace=True)

# 数据分析
# 计算平均薪资
average_salary = data['薪资'].mean()
print(f"平均薪资：{average_salary}")

# 计算薪资中位数
median_salary = data['薪资'].median()
print(f"薪资中位数：{median_salary}")

# 计算薪资标准差
std_salary = data['薪资'].std()
print(f"薪资标准差：{std_salary}")

# 可视化薪资分布
plt.hist(data['薪资'], bins=20, alpha=0.5)
plt.xlabel('薪资')
plt.ylabel('频数')
plt.title('薪资分布')
plt.show()

# 绘制薪资箱线图
plt.boxplot(data['薪资'])
plt.xlabel('薪资')
plt.title('薪资箱线图')
plt.show()
```
