    # 分词和词性标注
    tokens = word_tokenize(text)
    tagged_tokens = nltk.pos_tag(tokens)
    
    # 情感分析
    sentiment = analyze_sentiment(tagged_tokens)
    
    return sentiment

def analyze_sentiment(tokens):
    analyzer = SentimentIntensityAnalyzer()
    sentiment = analyzer.polarity_scores(' '.join([token for token, pos in tokens]))
    return sentiment

def generate_image_sentiment(sentiment):
    if sentiment > 0:
        image = cv2.imread('happy.jpg')
    elif sentiment < 0:
        image = cv2.imread('sad.jpg')
    else:
        image = cv2.imread('neutral.jpg')
    return image

def preprocess_image(image):
    image = cv2.GaussianBlur(image, (5, 5), 0)
    image = cv2.equalizeHist(image)
    return image

        "interaction": "drag"
    }
