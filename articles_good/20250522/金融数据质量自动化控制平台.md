```python
import numpy as np

def data_validation(data):
    # 数据校验规则
    rules = {
        'age': (lambda x: x >= 0),
        'income': (lambda x: x >= 0)
    }
    # 应用规则
    for col, rule in rules.items():
        data[col] = data[col].apply(rule)
    return data
```
