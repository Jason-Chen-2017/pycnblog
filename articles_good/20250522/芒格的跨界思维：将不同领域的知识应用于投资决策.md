##### 3.1.2 概率算法的Python实现代码
```python
import numpy as np

def calculate_probability(data):
    probabilities = {}
    for outcome in data:
        if outcome in probabilities:
            probabilities[outcome] += 1
        else:
            probabilities[outcome] = 1
    total = len(data)
    for outcome in probabilities:
        probabilities[outcome] /= total
    return probabilities

# 示例数据
data = [1, 2, 3, 2, 2, 3, 1]
probabilities = calculate_probability(data)
print(probabilities)
```

##### 3.2.2 投资组合优化的Python实现代码
```python
import numpy as np

def optimize_portfolio(returns, weights):
    # 计算预期收益
    expected_return = np.sum(returns * weights)
    # 计算风险
    covariance_matrix = np.cov(returns)
    risk = np.sqrt(np.dot(weights.T, covariance_matrix * weights))
    return expected_return, risk

# 示例数据
returns = np.array([0.1, 0.2, 0.15])
weights = np.array([0.3, 0.5, 0.2])
expected_return, risk = optimize_portfolio(returns, weights)
print(expected_return, risk)
```
