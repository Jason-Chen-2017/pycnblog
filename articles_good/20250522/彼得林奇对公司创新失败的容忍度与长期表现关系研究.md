```python
import numpy as np
import pandas as pd

class InnovationToleranceAnalyzer:
    def calculate_tolerance(self, companies):
        # 创新失败次数
        failure_count = len([c for c in companies if c['innovation_failure_tolerance'] > 0])
        # 总创新尝试次数
        total_attempts = len(companies)
        # 创新失败容忍度
        tolerance = (failure_count / total_attempts) * 100
        return tolerance

# 示例数据

]

# 计算创新失败容忍度
```
