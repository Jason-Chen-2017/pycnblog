#### 3.1 算法原理

$$ \text{投资价值} = \text{企业盈利能力} \times \text{行业地位} \times \text{管理层能力} $$

#### 3.3 Python源代码实现  
```python
def evaluate_company(finance, management, industry):
    return finance * management * industry

def long_term_investment(stocks):
    selected_stocks = []
    for stock in stocks:
        score = evaluate_company(
            stock['finance'], 
            stock['management'], 
            stock['industry']
        )
        if score > 0.8:
            selected_stocks.append(stock)
    return selected_stocks

# 示例用法
stocks = [
    {'finance': 0.9, 'management': 0.8, 'industry': 0.8},
    {'finance': 0.7, 'management': 0.6, 'industry': 0.9}
]
selected = long_term_investment(stocks)
print(selected)
```

```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

def calculate_rogue_behavior(stocks):
    return np.mean(stocks['performance']) * stocks['governance']

# 示例数据
data = {
    'finance': [0.8, 0.9, 0.7],
    'management': [0.8, 0.7, 0.9],
    'industry': [0.8, 0.9, 0.7]
}
df = pd.DataFrame(data)

selected_stocks = []
for i in range(len(df)):
    score = calculate_rogue_behavior(df.iloc[i])
    if score > 0.8:
        selected_stocks.append(df.iloc[i])

print(selected_stocks)
```
