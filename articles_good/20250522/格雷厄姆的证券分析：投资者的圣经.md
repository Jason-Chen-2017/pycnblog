```python
import pandas as pd
import numpy as np

# 示例数据
data = {
    '公司名称': ['公司A', '公司B', '公司C'],
    '净利润': [100, 150, 200],
    '股价': [50, 60, 70]
}

df = pd.DataFrame(data)

# 计算内在价值（简化模型）
def calculate_intrinsic_value(net_income, price):
    return net_income / price

df['内在价值'] = df.apply(lambda x: calculate_intrinsic_value(x['净利润'], x['股价']), axis=1)

print(df)
```
