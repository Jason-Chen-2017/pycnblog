### 5.2 核心代码实现
```python
import pandas as pd
import numpy as np
from sklearn.linear_model import LinearRegression
import matplotlib.pyplot as plt

# 数据清洗
data = pd.read_csv('stock_csr.csv')
data.dropna(inplace=True)
data['CSR_Score'] = data['CSR_Score'].astype(float)

# 模型训练
X = data[['CSR_Score', 'Financial_Performance']]
y = data['Stock_Valuation']
model = LinearRegression()
model.fit(X, y)

# 模型预测
predicted_valuation = model.predict(X)
plt.scatter(y, predicted_valuation)
plt.xlabel('Actual Valuation')
plt.ylabel('Predicted Valuation')
plt.title('Actual vs Predicted Stock Valuation')
plt.show()
```
