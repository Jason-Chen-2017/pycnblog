#### 4.2.2 Python代码实现  
```python
import numpy as np
import gym

class MultiAgentAlgorithm:
    def __init__(self, env):
        self.env = env
        self.num_agents = env.n_agents

    def step(self, action):
        observation, reward, done, info = self.env.step(action)
        return observation, reward, done, info

# 示例环境
env = gym.make('MultiAgentEnvironment-v0')
algorithm = MultiAgentAlgorithm(env)
observation, reward, done, info = algorithm.step([1, 0])
```
