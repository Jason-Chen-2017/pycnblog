```python
import random

def evaluate(individual):
    return sum(individual),  # 目标函数

def mutate(individual):
    idx = random.randint(0, len(individual)-1)
    individual[idx] = 1 - individual[idx]
    return individual

def crossover(parent1, parent2):
    idx = random.randint(1, len(parent1)-1)
    return parent1[:idx] + parent2[idx:], parent2[:idx] + parent1[idx:]

# 初始化种群
population = [[random.randint(0,1) for _ in range(n)] for _ in range(population_size)]
# 迭代优化
for _ in range(max_iterations):
    population = [mutate(individual) for individual in population]
    population = [crossover(parent1, parent2) for parent1, parent2 in zip(population[::2], population[1::2])]
    # 选择优秀个体
    population = sorted(population, key=lambda x: evaluate(x), reverse=True)[:population_size//2]
```
