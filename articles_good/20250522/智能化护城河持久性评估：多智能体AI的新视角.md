## 5.2 系统核心实现

### 5.2.1 Q-learning算法实现
```python
import numpy as np
import gym

class QLearning:
    def __init__(self, state_size, action_size):
        self.state_size = state_size
        self.action_size = action_size
        self.Q = np.zeros((state_size, action_size))

    def choose_action(self, state, epsilon=0.1):
        if np.random.random() < epsilon:
            return np.random.randint(self.action_size)
        return np.argmax(self.Q[state])

    def update_Q(self, state, action, reward, next_state, gamma=0.99):
        self.Q[state, action] = reward + gamma * np.max(self.Q[next_state])
```

env = gym.make('CustomEnv-v0')
agent = QLearning(env.observation_space.shape[0], env.action_space.n)
for episode in range(1000):
    state = env.reset()
    for step in range(100):
        action = agent.choose_action(state)
        next_state, reward, done, _ = env.step(action)
        agent.update_Q(state, action, reward, next_state)
        if done:
            break
