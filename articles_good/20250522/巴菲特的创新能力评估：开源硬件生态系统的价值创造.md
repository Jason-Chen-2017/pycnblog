# 假设我们已经计算出项目的各项指标得分
def calculate创新能力评分(stars, forks, issues, contributors):
    # 定义各项指标的权重
    weights = {
        'Stars': 0.4,
        'Forks': 0.3,
        'Issues': 0.2,
        'Contributors': 0.1
    }
    
    # 计算创新能力评分
    创新能力评分 = (stars * weights['Stars'] +
                     forks * weights['Forks'] +
                     issues * weights['Issues'] +
                     contributors * weights['Contributors'])
    return 创新能力评分

# 使用示例数据进行计算
arduino_stars = 40500
arduino_forks = 12000

arduino_contributors = 50

arduino创新能力评分 = calculate创新能力评分(arduino_stars, arduino_forks, arduino_issues, arduino_contributors)
print(f"Arduino创新能力评分: {arduino创新能力评分}")
