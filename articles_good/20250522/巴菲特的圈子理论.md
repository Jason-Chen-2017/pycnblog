import networkx as nx
from sklearn.metrics.pairwise import cosine_similarity

def build_network(users, relationships):
    graph = nx.Graph()
    for user in users:
        graph.add_node(user)
    for rel in relationships:
        graph.add_edge(rel['source'], rel['target'], weight=rel['strength'])
    return graph
