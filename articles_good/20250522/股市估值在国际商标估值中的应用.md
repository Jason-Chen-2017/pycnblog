```python
def stock_valuation_method(price, earnings, market_value):
    # 市盈率法
    pe_ratio = price / earnings
    # 市净率法
    pb_ratio = market_value / (price * earnings)
    return pe_ratio, pb_ratio

def trademark_valuation_method(income, market_position, brand_age):
    # 直接评估法
    direct_value = income * market_position / 100
    # 市场比较法
    market_comparable = direct_value * (1 - brand_age / 10)
    return direct_value, market_comparable
```

### 3.5 数学公式与模型
- 市盈率公式：$$PE = \frac{\text{股价}}{\text{每股收益}}$$
- 市净率公式：$$PB = \frac{\text{市值}}{\text{每股净资产}}$$
- 现金流折现模型：$$DCF = \sum_{t=1}^{n} \frac{\text{现金流}}{(1 + r)^t}$$

```python
import pandas as pd
import numpy as np

def calculate_stock_valuation(stock_price, earnings_per_share):
    pe_ratio = stock_price / earnings_per_share
    market_value = stock_price * 100  # 示例市场价值计算
    pb_ratio = market_value / (stock_price * earnings_per_share)
    return pe_ratio, pb_ratio

def calculate_trademark_valuation(income, market_position, brand_age):
    direct_value = income * (market_position / 100)
    adjustment_factor = 1 - (brand_age / 10)
    adjusted_value = direct_value * adjustment_factor
    return direct_value, adjusted_value

# 示例数据
stock_price = 50
earnings_per_share = 5
income = 1000000
market_position = 80
brand_age = 5

pe_ratio, pb_ratio = calculate_stock_valuation(stock_price, earnings_per_share)
direct_value, adjusted_value = calculate_trademark_valuation(income, market_position, brand_age)

print(f"市盈率: {pe_ratio:.2f}")
print(f"市净率: {pb_ratio:.2f}")
print(f"直接价值: {direct_value:.2f}")
print(f"调整后价值: {adjusted_value:.2f}")
```
