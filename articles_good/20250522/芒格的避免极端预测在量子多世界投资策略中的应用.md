import numpy as np

# 风险评估
def calculate_risk(returns):
    variance = np.var(returns)
    std_dev = np.std(returns)
    return variance, std_dev

# 投资策略优化
def optimize_portfolio(returns, weights):
    portfolio_return = np.dot(returns, weights)
    portfolio_risk = np.sqrt(np.dot(weights.T, np.dot(returns.cov(), weights)))
    return portfolio_return, portfolio_risk

returns = np.array([0.1, 0.2, 0.05])
weights = np.array([0.3, 0.5, 0.2])
risk = calculate_risk(returns)
portfolio_return, portfolio_risk = optimize_portfolio(returns, weights)
print("Risk:", risk)
print("Portfolio Return:", portfolio_return)
print("Portfolio Risk:", portfolio_risk)
