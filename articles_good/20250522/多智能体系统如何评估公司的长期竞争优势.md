```python
import concurrent.futures

def compute(agent_id, data):
    return f"Agent {agent_id} computed {data}"

data = [1, 2, 3]
with concurrent.futures.ThreadPoolExecutor() as executor:
    futures = [executor.submit(compute, i, data[i]) for i in range(len(data))]
    for future in concurrent.futures.as_completed(futures):
        print(future.result())
```
