#### 3.2 算法实现代码

```python
import numpy as np
from scipy.stats import norm

def calculate_margin(returns, confidence_level=0.95):
    # 计算期望值
    expected_return = np.mean(returns)
    # 计算标准差
    std_dev = np.std(returns)
    # 计算安全边际
    margin = norm.ppf(confidence_level) * std_dev
    return margin

# 示例数据
returns = [0.1, 0.2, 0.15, 0.25, 0.2, 0.1, 0.15]
confidence_level = 0.95
margin = calculate_margin(returns, confidence_level)
print(f"安全边际：{margin}")
```
