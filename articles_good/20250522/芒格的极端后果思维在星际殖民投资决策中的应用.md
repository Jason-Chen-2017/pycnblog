| 资金链断裂       | 高                 | 资源分配问题     |
| 政治变动         | 中                 | 合作与竞争      |
| 自然灾害         | 中                 | 环境适应性       |

# 风险评估函数
def risk_assessment(factor):
    if factor < 0.5:
        return "低风险"
    elif 0.5 <= factor < 0.8:
        return "中风险"
    else:
        return "高风险"

# 资源分配函数
def resource_allocation(budget):
    if budget < 100:
        return "技术开发"
    elif 100 <= budget < 500:
        return "资金分配"
    else:
        return "人才招聘"

# 决策优化函数
def decision_optimization(objective):
    if objective == "生存":
        return "优先技术开发"
    elif objective == "资源":
        return "优先资金分配"
    else:
        return "优先人才招聘"
