```python
def calculate_investment_strategy(finance_data, technology_data):
    """
    基于芒格思维复杂性的投资策略计算函数
    :param finance_data: 财务数据
    :param technology_data: 技术数据
    :return: 投资策略
    """
    # 1. 财务数据分析
    financial_health = calculate_financial_health(finance_data)
    
    # 2. 技术数据分析
    technological_health = calculate_technological_health(technology_data)
    
    # 3. 综合评估
    overall_health = (financial_health + technological_health) / 2
    
    # 4. 投资策略制定
    if overall_health > 0.8:
        return "Strong Buy"
    elif 0.6 < overall_health <= 0.8:
        return "Buy"
    else:
        return "Sell"

# 示例数据

    "profit": 200000,
    "debt": 500000

    "patents": 50,
    "R&D_expense": 100000,
    "market_share": 0.3
}

# 调用函数
```

### 代码示例3: 芒格思维复杂性在投资中的系统架构设计

```python
# 代码实现：芒格思维复杂性在投资中的系统架构设计

class InvestmentStrategy:
    def __init__(self, finance_data, technology_data):
        self.finance_data = finance_data
        self.technology_data = technology_data
        
    def calculate_financial_health(self):
        """
        计算财务健康状况
        """
        revenue = self.finance_data['revenue']
        profit = self.finance_data['profit']
        debt = self.finance_data['debt']
        
        financial_health = (revenue + profit) / debt
        return financial_health
    
    def calculate_technological_health(self):
        """
        计算技术健康状况
        """
        patents = self.technology_data['patents']
        R&D_expense = self.technology_data['R&D_expense']
        market_share = self.technology_data['market_share']
        
        technological_health = (patents + R&D_expense) * market_share
        return technological_health
    
    def determine_investment_strategy(self):
        """
        确定投资策略
```
