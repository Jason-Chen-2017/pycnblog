def fetch_data(url):

    data = []
    for item in soup.find_all('div', class_='item'):
        data.append({

            'risk_factor': float(item.find('span', class_='risk').text)
        })
    return data

def evaluate_risk(data):
    X = [[item['risk_factor']] for item in data]
    y = [item['country'] for item in data]
    model = GaussianNB()
    model.fit(X, y)
    return model.predict(X)
