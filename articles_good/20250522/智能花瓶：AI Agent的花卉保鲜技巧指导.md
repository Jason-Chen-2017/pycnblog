- **花卉状态监测代码**
  ```python
  import numpy as np
  import pandas as pd
  from sklearn.linear_model import LinearRegression

  # 训练温度预测模型
  def train_temp_model(X, y):
      model = LinearRegression()
      model.fit(X, y)
      return model

  # 预测温度
  def predict_temp(model, humidity):
      return model.predict([[humidity]])

  # 示例数据
  X = np.array([20, 25, 30, 35, 40]).reshape(-1, 1)
  y = np.array([18, 22, 26, 30, 34])
  model = train_temp_model(X, y)
  print(predict_temp(model, 30))
  ```
