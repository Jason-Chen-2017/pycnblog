3. **无限细节**：分形结构具有无限细节，即无论放大到何种程度，都能发现新的细节。

```python
import numpy as np
import matplotlib.pyplot as plt

def box_count(data, max_box_size, num_bins):
    """
    使用盒计数法计算分形维数。
    """
    # 初始化计数数组
    counts, _ = np.histogram(data, bins=num_bins, range=(0, max_box_size))
    
    # 计算盒计数
    L = np.arange(1, num_bins+1)
    N = np.cumsum(np插值（counts, 1))
    
    # 计算豪斯道夫维数
    D = (np.log(N) - np.log(L)) / np.log(max_box_size)
    
    return D

# 读取CMB数据
cmb_data = np.load('cmb_data.npy')

# 计算分形维数
max_box_size = 100
num_bins = 1000
fractal_dim = box_count(cmb_data, max_box_size, num_bins)

print(f"分形维数: {fractal_dim}")

# 可视化
plt.hist(cmb_data, bins=num_bins, range=(0, max_box_size), alpha=0.5)
plt.xlabel('温度（K）')
plt.ylabel('计数')
plt.title('CMB温度分布')
plt.show()
```
