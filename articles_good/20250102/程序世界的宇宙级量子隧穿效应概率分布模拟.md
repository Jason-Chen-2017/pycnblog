def quantum_tunneling_simulation(num_particles, barrier_height, barrier_position, potential_energy):
    # 初始化概率分布数组
    probability_distribution = np.zeros(barrier_position.shape)

        # 更新概率分布

from flask import Flask, request, jsonify
import numpy as np

app = Flask(__name__)

def quantum_tunneling_simulation(num_particles, barrier_height, barrier_position, potential_energy):
    probability_distribution = np.zeros(barrier_position.shape)
    
    for i in range(num_particles):
        particle_position = np.random.uniform(0, barrier_position)
        particle_energy = np.random.uniform(0, potential_energy)
        
        probability = np.exp(-particle_energy / barrier_height)
        
        if particle_energy < barrier_height:
            probability_distribution[particle_position] += probability
    
    return probability_distribution

@app.route('/simulate', methods=['POST'])
def simulate():
    data = request.json
    num_particles = data['num_particles']
    barrier_height = data['barrier_height']
    barrier_position = data['barrier_position']
    potential_energy = data['potential_energy']
    
    probability_distribution = quantum_tunneling_simulation(num_particles, barrier_height, barrier_position, potential_energy)
    
    return jsonify(probability_distribution.tolist())

if __name__ == '__main__':
    app.run(debug=True)

def quantum_tunneling_simulation(num_particles, barrier_height, barrier_position, potential_energy):
    probability_distribution = np.zeros(barrier_position.shape)

import numpy as np

app = Flask(__name__)

def quantum_tunneling_simulation(num_particles, barrier_height, barrier_position, potential_energy):
    probability_distribution = np.zeros(barrier_position.shape)
    
    for i in range(num_particles):
        particle_position = np.random.uniform(0, barrier_position)
        particle_energy = np.random.uniform(0, potential_energy)

def simulate():
    data = request.json

    potential_energy = data['potential_energy']
    
    probability_distribution = quantum_tunneling_simulation(num_particles, barrier_height, barrier_position, potential_energy)
    
    return jsonify(probability_distribution.tolist())

if __name__ == '__main__':
