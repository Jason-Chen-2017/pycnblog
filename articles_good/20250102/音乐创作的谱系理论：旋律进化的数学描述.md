import numpy as np
import mido
import matplotlib.pyplot as plt

def generate_melody(duration, steps=12):
    times = np.linspace(0, duration, steps)
    frequencies = np.linspace(261.63, 329.63, steps)
    notes = [mido.Note(frequency=f, duration=duration/len(frequencies)) for f in frequencies]
    return notes

def play_melody(melody):
    mid = mido.MidiFile()
    for note in melody:
        mid.tracks.append(note)
    mid.save("melody.mid")
    mid.play()

melody = generate_melody(5)
play_melody(melody)
```
