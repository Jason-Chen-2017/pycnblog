        self.doctor_experience = doctor_experience

        health_status_set = mfc.FuzzySet(['健康状态'], {0: 0, 1: 0.5, 2: 1})
        doctor_experience_set = mfc.FuzzySet(['医生经验'], {0: 0, 1: 0.5, 2: 1})
        equipment_status_set = mfc.FuzzySet(['设备状态'], {0: 0, 1: 0.5, 2: 1})

        health_status_set.update({0: 1 if self.health_status == '健康' else 0})
        doctor_experience_set.update({0: 1 if self.doctor_experience == '丰富' else 0})
        equipment_status_set.update({0: 1 if self.equipment_status == '良好' else 0})

        rules = [
            ("健康状态良好", "医生经验丰富", "进行心脏复苏术"),
            ("健康状态一般", "医生经验丰富", "考虑心脏复苏术"),
            ("健康状态较差", "医生经验丰富", "不进行心脏复苏术"),
        ]

        result = mfc.FuzzyLogicEngine([health_status_set, doctor_experience_set, equipment_status_set], rules).fuzzy_retrieve()
