import numpy as np
from scipy.linalg import eig

def oscillation_probability(energy, mass_matrix):
    """
    计算中微子振荡概率的函数。
    
    参数:
    energy: 中微子的能量（单位：eV）
    mass_matrix: 中微子质量矩阵
    
    返回:
    oscillation_probability: 中微子振荡概率
    """
    eigenvalues, eigenvectors = eig(mass_matrix)
    omega = 1 / np.sqrt(eigenvalues)
    theta = np.arcsin(np.sqrt(1 - omega[2]**2))
    delta = np.arcsin(omega[2] * np.cos(theta))
    
    probability = 0.5 * (1 - np.cos(2 * theta) - np.sin(2 * theta) * np.sin(2 * delta))
    
    return probability

# 示例：计算一个中微子从μ子类型转变为τ子类型的概率
energy = 5e6  # 能量5MeV
mass_matrix = np.array([[7.42e-5, 2.45e-5, 1.04e-5],
                        [2.45e-5, 7.42e-5, 3.55e-5],
                        [1.04e-5, 3.55e-5, 7.42e-5]])

probability = oscillation_probability(energy, mass_matrix)
print("振荡概率:", probability)
