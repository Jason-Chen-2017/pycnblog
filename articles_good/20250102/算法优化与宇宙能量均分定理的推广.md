import numpy as np

def optimize_energy(E, N):
    """
    优化能量分布
    :param E: 总能量
    :param N: 部分数量
    :return: 最优能量分布
    """
    E = np.array(E)
    E = E / np.sum(E)
    return E

def update_state(E, N):
    """
    更新状态
    :param E: 能量分布
    :param N: 部分数量
    :return: 更新后的能量分布
    """
    E = optimize_energy(E, N)
    return E

def main():
    E = [10, 5, 3, 2, 1]
    N = 5
    print("初始状态：", E)
    for i in range(5):
        E = update_state(E, N)
        print(f"第{i+1}次优化后状态：", E)

if __name__ == "__main__":
    main()
```

import numpy as np

def optimize_energy(E, N):
    """
    优化能量分布
    :param E: 总能量
    :param N: 部分数量
    :return: 最优能量分布
    """
    E = np.array(E)
    E = E / np.sum(E)
    return E

def update_state(E, N):
    """
    更新状态
    :param E: 能量分布
    :param N: 部分数量
    :return: 更新后的能量分布
    """
    E = optimize_energy(E, N)
    return E

def main():

    N = 5
    print("初始状态：", E)
    for i in range(5):
        E = update_state(E, N)
        print(f"第{i+1}次优化后状态：", E)

if __name__ == "__main__":
    main()

    :param E: 总能量
