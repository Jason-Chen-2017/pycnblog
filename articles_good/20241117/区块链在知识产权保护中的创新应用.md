pragma solidity ^0.8.0;

contract CopyrightTrade {
    address public owner;
    mapping(uint256 => Copyright) public copyrights;

    struct Copyright {
        uint256 id;
        address creator;
        string title;
        bool isSold;
        address buyer;
    }

    event CopyrightSold(uint256 id, address buyer, uint256 price);

    constructor() {
        owner = msg.sender;
    }

    function registerCopyright(uint256 id, address creator, string memory title) public {
        require(!isSold(id), "版权已被注册或出售");
        copyrights[id] = Copyright(id, creator, title, false, address(0));
    }

    function buyCopyright(uint256 id, uint256 price) public payable {
        require(isSold(id) == false, "版权已被出售");
        require(msg.value >= price, "支付金额不足");

        copyrights[id].isSold = true;
        copyrights[id].buyer = msg.sender;

        payable(copyrights[id].creator).transfer(msg.value);

        emit CopyrightSold(id, msg.sender, price);
    }

    function isSold(uint256 id) public view returns (bool) {
        return copyrights[id].isSold;
    }
}
