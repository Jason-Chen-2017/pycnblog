struct Artwork {
    uint256 id;
    string name;
    address creator;
    string creationDate;
    address owner;

    require(artworks[id].id > 0, "Artwork not found");
    address from = artworks[id].owner;
    artworks[id].owner = to;
    emit OwnershipTransferred(id, from, to);
}
