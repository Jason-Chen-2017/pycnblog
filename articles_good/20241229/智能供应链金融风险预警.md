    model.fit(features, labels)
    return model

# 风险识别
def identify_risk(model, new_data):
    features = extract_features(new_data)
    risk_scores = model.predict_proba(features)[:, 1]
    return risk_scores

def assign_risk_level(scores, threshold=0.5):
    risk_levels = ['低风险' if score < threshold else '高风险' for score in scores]

    orders['risk_level'] = assign_risk_level(risk_scores)
