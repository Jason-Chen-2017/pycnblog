```python
import numpy as np

def facial_recognition(image_path, known_faces):
    # 加载预训练的人脸识别模型
    model = cv2.face.EigenFaceRecognizer_create()

    # 训练模型
    model.train(np.array(known_faces))

    # 加载测试图像
    image = cv2.imread(image_path)

    # 调整图像大小
    resized_image = cv2.resize(image, (160, 160))

    # 将图像转换为灰度图像
    gray_image = cv2.cvtColor(resized_image, cv2.COLOR_BGR2GRAY)

    # 进行人脸识别
    label, confidence = model.predict(gray_image)

    return label, confidence
```

```python
def make_decision(identification_results, known_users):
    user_id = None
    max_confidence = 0

    for result in identification_results:
        user = known_users.get(result['label'], None)
        if user and result['confidence'] > max_confidence:
            user_id = user['id']
            max_confidence = result['confidence']

    return user_id
```

### 最佳实践
