       },
       methods: {

               if (response.status === 201) {

                   this.duration = '';

               }
           },
           async getRecords() {

               if (response.status === 200) {

               }
           },
           async deleteRecord(id) {
               const response = await axios.delete('/api/meditation', { data: { id } });
               if (response.status === 200) {
                   this.records = this.records.filter(record => record.id !== id);
               }
           }
       },
       created() {
           this.getRecords();
       }
   };
