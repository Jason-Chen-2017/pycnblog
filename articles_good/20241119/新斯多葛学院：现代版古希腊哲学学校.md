```python
# app.py
from flask import Flask, request, jsonify
from models import Mood, db

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///mood.db'
db.init_app(app)

@app.route('/updateMood', methods=['POST'])
def update_mood():
    data = request.json
    mood = Mood(mood=data['mood'])
    db.session.add(mood)
    db.session.commit()
    return jsonify({'status': 'success'})

if __name__ == '__main__':
    app.run(debug=True)
```

```python
# models.py
from flask_sqlalchemy import SQLAlchemy

db = SQLAlchemy()

    mood = db.Column(db.String(50))
```
