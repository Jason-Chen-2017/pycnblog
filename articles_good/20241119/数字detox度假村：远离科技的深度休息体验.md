                    )''')

    if cursor.fetchone() is None:
        cursor.execute('''INSERT INTO sign_in (activity_id, user_id, sign_in_time) VALUES (?, ?, ?)''', (activity_id, user_id, datetime.now()))
        conn.commit()
        return "签到成功"
    else:
        return "已签到"

# 解签
def sign_out(activity_id, user_id):
    cursor.execute('''SELECT * FROM sign_in WHERE activity_id = ? AND user_id = ?''', (activity_id, user_id))
    if cursor.fetchone() is not None:
        cursor.execute('''DELETE FROM sign_in WHERE activity_id = ? AND user_id = ?''', (activity_id, user_id))
        conn.commit()
        return "解签成功"
    else:
        return "未签到"

# 关闭数据库连接
