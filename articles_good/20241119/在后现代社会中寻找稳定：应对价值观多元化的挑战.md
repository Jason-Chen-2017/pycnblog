// 社区对话计划后端API示例
const express = require('express');
const app = express();
app.use(express.json());

// 对话列表
const dialogues = [];

// 新增对话
app.post('/dialogues', (req, res) => {
    const newDialogue = req.body;
    dialogues.push(newDialogue);
    res.status(201).send(newDialogue);
});

// 获取对话列表
app.get('/dialogues', (req, res) => {
    res.status(200).send(dialogues);
});

// 更新对话
app.put('/dialogues/:id', (req, res) => {
    const id = req.params.id;
    const index = dialogues.findIndex(d => d.id === id);
    if (index !== -1) {
        dialogues[index] = req.body;
        res.status(200).send(dialogues[index]);
    } else {
        res.status(404).send('Dialogue not found');
    }
});

// 删除对话
app.delete('/dialogues/:id', (req, res) => {
    const id = req.params.id;
    const index = dialogues.findIndex(d => d.id === id);
    if (index !== -1) {
        dialogues.splice(index, 1);
        res.status(204).send();
    } else {
        res.status(404).send('Dialogue not found');
    }
});

// 启动服务器
const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
    console.log(`Server running on port ${PORT}`);
});
