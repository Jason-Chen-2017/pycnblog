```javascript
import React, { useState } from 'react';
import axios from 'axios';

function App() {
  const [behaviors, setBehaviors] = useState([]);
  const [newBehavior, setNewBehavior] = useState('');

  const addBehavior = async () => {
    if (newBehavior) {
      const response = await axios.post('/api/behaviors', { behavior: newBehavior });
      setBehaviors([...behaviors, response.data]);
      setNewBehavior('');
    }
  };

  return (
    <div>
      <h1>数字解毒助手</h1>
      <ul>
        {behaviors.map((behavior, index) => (
          <li key={index}>{behavior.behavior}</li>
        ))}
      </ul>
      <input
        type="text"
        value={newBehavior}
        onChange={(e) => setNewBehavior(e.target.value)}
      />
      <button onClick={addBehavior}>添加行为</button>
    </div>
  );
}

export default App;
```
