import numpy as np

def思维链_model(data, learning_rate, epochs):  
    model = np.zeros((data.shape[1], 1))  
    for _ in range(epochs):  
        predictions = np.dot(data, model)  
        errors = data - predictions  
        model += learning_rate * np.dot(data.T, errors)  
    return model

data = np.array([[1, 2], [3, 4], [5, 6]])  
learning_rate = 0.1  
epochs = 100

model =思维链_model(data, learning_rate, epochs)  
print("思维链模型参数:", model)  
```

def思维链_model(data, learning_rate, epochs):

        predictions = np.dot(data, model)  
        errors = data - predictions  
        model += learning_rate * np.dot(data.T, errors)  
    return model

learning_rate = 0.1  
epochs = 100
