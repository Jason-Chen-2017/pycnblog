```python
import time

def original_function(n):
    # 原始函数实现
    result = 0
    for i in range(n):
        result += i

    # 优化后的函数实现
    return (n * (n - 1)) // 2

# 测试原始函数

optimized_result = optimized_function(10000)
```

```python
# app.py
from flask import Flask, request, jsonify
from models import Order, db

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'mysql+pymongo://username:password@localhost/dbname'
db.init_app(app)

@app.route('/orders', methods=['GET'])
def get_orders():
    order_id = request.args.get('id')
    order = Order.query.filter_by(id=order_id).first()
    if order:
        return jsonify(order.to_dict())
    else:
        return jsonify({'error': 'Order not found'})

if __name__ == '__main__':
    app.run(debug=True)
```
