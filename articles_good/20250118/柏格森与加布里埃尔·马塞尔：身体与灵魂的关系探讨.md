```python
# body_soul_research.py

class BodyData:
    def __init__(self, name, age, gender):
        self.name = name
        self.age = age
        self.gender = gender

    def get_body_data(self):
        return f"Name: {self.name}, Age: {self.age}, Gender: {self.gender}"

class SoulData:
    def __init__(self, name, emotion, stress_level):
        self.name = name
        self.emotion = emotion
        self.stress_level = stress_level

    def get_soul_data(self):
        return f"Name: {self.name}, Emotion: {self.emotion}, Stress Level: {self.stress_level}"

def collect_data():
    body_data = BodyData("John", 30, "Male")
    soul_data = SoulData("John", "Happy", 5)
    return body_data, soul_data

def analyze_data(body_data, soul_data):
    analysis_result = f"Body Data: {body_data.get_body_data()}, Soul Data: {soul_data.get_soul_data()}"
    return analysis_result

if __name__ == "__main__":
    body_data, soul_data = collect_data()
    analysis_result = analyze_data(body_data, soul_data)
    print(analysis_result)
```
