```python
class Lexer:
    def __init__(self, text):
        self.text = text
        self.index = 0
        self.tokens = []

    def next_token(self):
        while self.index < len(self.text):
            char = self.text[self.index]

            if char.isalpha():
                self.read_identifier()
            elif char.isdigit():
                self.read_number()
            elif char in ['+', '-', '*', '/']:
                self.tokens.append(Token('OPERATOR', char))
                self.index += 1
            else:
                self.index += 1

        return self.tokens

    def read_identifier(self):
        identifier = ''
        while self.index < len(self.text) and (self.text[self.index].isalpha() or self.text[self.index].isdigit()):
            identifier += self.text[self.index]
            self.index += 1

        self.tokens.append(Token('IDENTIFIER', identifier))

    def read_number(self):
        number = ''
        while self.index < len(self.text) and (self.text[self.index].isdigit() or self.text[self.index] == '.'):
            number += self.text[self.index]
            self.index += 1

        self.tokens.append(Token('NUMBER', float(number) if '.' in number else int(number)))

class Token:
    def __init__(self, type, value):
        self.type = type
        self.value = value
```
