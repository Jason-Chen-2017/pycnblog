```python
import re

# 语法分析
def is_valid_syntax(prompt):
    pattern = r"^[a-zA-Z_][a-zA-Z0-9_]*=\s*[0-9]+$"
    return re.match(pattern, prompt) is not None

# 构建环境模型
def build_environment():
    environment = {}
    return environment

# 构建抽象语义模型
def build_abstract_semantics():
    abstract_semantics = {
        "assignment": lambda var, value: value
    }
    return abstract_semantics

# 执行程序
def execute(environment, abstract_semantics):
    if is_valid_syntax(prompt):
        var, _, value = prompt.partition("=")
        environment[var] = abstract_semantics["assignment"](var, value)
        return environment[var]
    else:
        return "语法错误"

# 测试代码
prompt = "a=5"
result = execute(build_environment(), build_abstract_semantics())
print(result)  # 输出：5
```
