## 3.3 算法实现代码
```python
import pandas as pd
from sklearn.ensemble import RandomForestClassifier

def investor_behavior_analysis(data):
    # 数据预处理
    data.dropna(inplace=True)
    # 特征提取
    features = data[['age', 'risk_tolerance', 'market_sentiment']]
    target = data['investment_strategy']
    # 训练模型
    model = RandomForestClassifier()
    model.fit(features, target)
    # 预测
    predictions = model.predict(features)
    return predictions

# 示例数据
data = {
    'age': [30, 45, 50],
    'risk_tolerance': [3, 5, 2],
    'market_sentiment': [2, 4, 3],
    'investment_strategy': ['growth', 'income', 'conservative']
}

investor_behavior_analysis(pd.DataFrame(data))
```

## 5.2 核心代码实现

```python
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score

# 数据加载
data = pd.read_csv('investor_data.csv')

# 数据预处理
data = data.dropna()

# 特征与目标分离
features = data[['age', 'income', 'risk_tolerance']]
target = data['strategy']

# 训练模型
model = RandomForestClassifier()
model.fit(features, target)

# 预测与评估
predictions = model.predict(features)
print("Accuracy:", accuracy_score(target, predictions))
```
