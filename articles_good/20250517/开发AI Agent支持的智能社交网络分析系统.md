import tweepy

auth = tweepy.OAuthHandler(consumer_key, consumer_secret)
auth.set_access_token(access_token, access_token_secret)
api = tweepy.API(auth)

def get_tweets(query, max_num):
    tweets = []
    for tweet in tweepy.Cursor(api.search, q=query).items(max_num):
        tweets.append(tweet.text)
    return tweets
