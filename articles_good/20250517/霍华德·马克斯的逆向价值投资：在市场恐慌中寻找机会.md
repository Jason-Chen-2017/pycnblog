#### 5.2 Python代码实现
```python
def calculate_margin_of_safety(value, price):
    return (value - price) / price

def is_in_panic_market(market_panic_index):
    return market_panic_index > 0.8

def main():
    assets = [{'value': 100, 'price': 50}, {'value': 80, 'price': 40}]
    panic_assets = []
    for asset in assets:
        mos = calculate_margin_of_safety(asset['value'], asset['price'])
        if is_in_panic_market(mos):
            panic_assets.append(asset)
    print("恐慌市场中可以投资的资产：", panic_assets)

if __name__ == "__main__":
    main()
```

#### 5.3 算法的优缺点分析
- 优点：能够有效识别低估资产，捕捉市场恐慌中的机会

#### 8.1 核心代码实现
```python
import pandas as pd

def calculate_margin_of_safety(df):
    df['安全边际'] = (df['内在价值'] - df['价格']) / df['价格']
    return df

def identify_panic_assets(df, threshold=0.8):
    return df[df['安全边际'] > threshold]

def main():
    data = {
        '资产': ['资产1', '资产2', '资产3'],
        '内在价值': [100, 80, 120],
        '价格': [50, 40, 60]
    }
    df = pd.DataFrame(data)
    df = calculate_margin_of_safety(df)
    panic_assets = identify_panic_assets(df)
    print("恐慌市场中可以投资的资产：", panic_assets)

if __name__ == "__main__":
    main()
```
