  import numpy as np
  from scipy.signal import butterworth

  def acquire_signal(sampling_rate=1000, duration=5):
      # 模拟信号采集
      t = np.linspace(0, duration, int(sampling_rate * duration))
      # 添加噪声
      noise = np.random.normal(0, 0.1, len(t))
      signal = np.sin(2 * np.pi * 50 * t) + noise
      return signal
  ```

- 5.2.2 数据处理模块实现

  def process_signal(signal, sampling_rate=1000):
      # 滤波
      from scipy.signal import butterworth
      # 设计滤波器
      nyq = 0.5 * sampling_rate
      low = 30
      high = 70
      order = 2
      # 计算系数
      b, a = butterworth(order, [low/nyq, high/nyq], btype='band')
      # 应用滤波器
      filtered_signal = butterworth.apply(b, a, signal)
      return filtered_signal
  ```
