def fisher_innovation_score(technology, market, management):
    # 权重分配
    weight_technology = 0.4
    weight_market = 0.3
    weight_management = 0.3

    # 计算综合评分
    score = (technology * weight_technology) + (market * weight_market) + (management * weight_management)
    return score

# 示例数据
technology = 80  # 技术创新评分
market = 70      # 市场需求评分
management = 85   # 管理团队评分

# 计算综合评分
score = fisher_innovation_score(technology, market, management)
print("综合创新评分：", score)
```

import pandas as pd
from sklearn.linear_model import LinearRegression

def load_data(file_path):
    return pd.read_csv(file_path)

def preprocess_data(data):
    # 数据预处理
    data = data.dropna()
    return data

def train_model(data, target):
    model = LinearRegression()
    model.fit(data, target)
    return model

def predict_innovation_score(model, data):
    return model.predict(data)

# 示例运行
data = load_data("innovation.csv")
data_processed = preprocess_data(data)
target = data_processed['score']
features = data_processed[['technology', 'market', 'management']]

model = train_model(features, target)
predictions = predict_innovation_score(model, features)
print("预测结果：", predictions)
```
