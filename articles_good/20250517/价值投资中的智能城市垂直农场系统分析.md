### 4.3 系统优化算法实现

#### 4.3.1 群智能算法代码实现

```python
class Particle:
    def __init__(self, position, velocity, best_position, best_fitness):
        self.position = position
        self.velocity = velocity
        self.best_position = best_position
        self.best_fitness = best_fitness

def update_particle(particle, best_particle, w, c1, c2):
    # 更新速度
    particle.velocity = w * particle.velocity + c1 * (particle.best_position - particle.position) + c2 * (best_particle.position - particle.position)
    # 更新位置
    particle.position += particle.velocity

def particle_swarm_optimization(pop_size, dim, bounds, fitness_func, max_iter=100):
    # 初始化粒子群
    particles = [Particle(
        position=[bounds[i][0] + (bounds[i][1]-bounds[i][0])*random.random() for i in range(dim)],
        velocity=[0]*dim,
        best_position=[0]*dim,
        best_fitness=0
    ) for _ in range(pop_size)]
    
    # 计算初始适应度
    for p in particles:
        p.best_fitness = fitness_func(p.position)
    
    for _ in range(max_iter):
        best = max(particles, key=lambda x: x.best_fitness)
        for p in particles:
            update_particle(p, best, w=0.8, c1=2, c2=2)
            # 更新个体最优
            if fitness_func(p.position) > p.best_fitness:
                p.best_position = p.position.copy()
                p.best_fitness = fitness_func(p.position)
    return particles
```
