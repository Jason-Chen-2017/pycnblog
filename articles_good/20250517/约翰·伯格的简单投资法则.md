#### 6.1.3 实战代码示例
```python
import numpy as np
import pandas as pd

# 假设我们有多个资产的收益率数据
returns = pd.DataFrame({
    '股票': [0.1, 0.05, -0.02, 0.08, 0.03],
    '债券': [0.02, 0.03, 0.04, 0.02, 0.05],
    '现金': [0, 0, 0, 0, 0]
})

# 计算投资组合的期望收益率和风险
def portfolio_return(weights, returns):
    return (weights * returns).sum()

def portfolio_risk(weights, returns):
    return np.sqrt(np.dot(weights.T, returns.cov() * weights))

# 示例投资组合：60%股票，30%债券，10%现金
weights = np.array([0.6, 0.3, 0.1])
return_ = portfolio_return(weights, returns)
risk = portfolio_risk(weights, returns)

print(f"期望收益率: {return_:.2%}")
print(f"风险: {risk:.2%}")
```
