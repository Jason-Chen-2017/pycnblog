import numpy as np
import pandas as pd
from scipy.optimize import minimize

# 定义优化目标函数
def portfolio_return(weights, returns):
    return np.sum(weights * returns)

# 定义风险函数
def portfolio_variance(weights, covariance_matrix):
    return np.dot(weights.T, np.dot(covariance_matrix, weights))

# 定义优化约束
def constraint(weights):
    return np.sum(weights) - 1

# 使用scipy.optimize.minimize进行优化
def optimize_portfolio(returns, covariance_matrix):
    n = len(weights)
    # 目标是最小化风险
    objective = {'type': 'min', 'fun': portfolio_variance, 'args': (covariance_matrix,)}
    # 约束条件：权重之和为1
    constraint = {'type': 'eq', 'fun': constraint}
    # 初始权重
    initial_weights = np.array([1.0/n]*n)
    # 进行优化
    result = minimize(objective, initial_weights, method='SLSQP', constraints=constraint)
    return result.x

# 示例数据
returns = pd.DataFrame({
    '房地产': [0.1, 0.05, 0.07],
    '私募股权': [0.08, 0.12, 0.06],
    '对冲基金': [0.15, 0.1, 0.08]
})

covariance_matrix = returns.cov()

# 优化投资组合
weights = optimize_portfolio(returns, covariance_matrix)
print("优化后的权重:", weights)
