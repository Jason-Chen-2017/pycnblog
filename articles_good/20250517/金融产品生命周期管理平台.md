```python
import numpy as np

def collaborative_filtering(X, user_id):
    # 计算用户相似度
    user_mean = np.mean(X, axis=1)
    X_centered = X - user_mean[:, np.newaxis]
    # 计算相似度矩阵
    similarity = np.dot(X_centered.T, X_centered)
    np.fill_diagonal(similarity, 0)
    # 找出最相似的用户
    similar_user = np.argmax(similarity[user_id, :])
    return similar_user
```
