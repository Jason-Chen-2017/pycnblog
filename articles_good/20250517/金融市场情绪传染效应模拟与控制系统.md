2. **传播模拟**：模拟情绪在网络中的传播过程。

### 5.1.2 情绪传播模拟的代码实现

以下是一个情绪传播模拟的Python代码示例：

```python
import networkx as nx
import numpy as np
import matplotlib.pyplot as plt

def simulate_emotion_spread(network, beta, gamma):
    G = nx.Graph(network)
    n = G.nodes()
    S = {node: 1 for node in n}
    I = {node: 0 for node in n}
    R = {node: 0 for node in n}
    
    for node in n:
        if node == 'source':
            I[node] = 1
            S[node] -= 1
            break
    
    for _ in range(100):
        new_I = I.copy()
        for node in n:
            if S[node] > 0:
                neighbors = list(G.neighbors(node))
                infected_neighbors = sum([I[neighbor] for neighbor in neighbors])
                probability = beta * infected_neighbors / len(neighbors) if len(neighbors) > 0 else 0
                if np.random.rand() < probability:
                    new_I[node] = 1
        for node in n:
            if new_I[node] != I[node]:
                if new_I[node] == 1:
                    S[node] -= 1
                    I[node] = 1
                else:
                    R[node] += 1
                    I[node] = 0
        I = new_I
    return S, I, R

network = {
    'source': ['node1', 'node2'],
    'node1': ['source', 'node3'],
    'node2': ['source', 'node3'],
    'node3': ['node1', 'node2']
}

beta = 0.5
gamma = 0.1

S, I, R = simulate_emotion_spread(network, beta, gamma)

print("Susceptible:", S)
print("Infected:", I)
print("Recovered:", R)

nx.draw(network, with_labels=True)
plt.show()
```
