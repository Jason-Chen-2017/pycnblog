## 3.5 协同过滤算法的Python实现示例

```python
import numpy as np
from sklearn.metrics.pairwise import cosine_similarity

# 示例数据：客户-产品评分矩阵
customers = ['C1', 'C2', 'C3', 'C4']
products = ['P1', 'P2', 'P3', 'P4']
ratings = {
    'C1': {'P1': 5, 'P2': 4, 'P3': 3, 'P4': 2},
    'C2': {'P1': 3, 'P2': 5, 'P3': 4, 'P4': 1},
    'C3': {'P1': 4, 'P2': 3, 'P3': 5, 'P4': 4},
    'C4': {'P1': 2, 'P2': 4, 'P3': 3, 'P4': 5}
}

# 创建评分矩阵
rating_matrix = np.array([[5,4,3,2],
                           [3,5,4,1],
                           [4,3,5,4],
                           [2,4,3,5]])

# 计算余弦相似度
similarity_matrix = cosine_similarity(rating_matrix)

# 示例：为C1推荐产品
def recommend_customer(customer_ratings, similarity_matrix, threshold=0.5):
    # 找出与目标客户相似度高于阈值的客户
    similar_indices = [i for i in range(len(similarity_matrix)) if similarity_matrix[i][0] > threshold]
    # 计算推荐评分
    recommendation_scores = np.zeros(len(customer_ratings[0]))
    for i in similar_indices:
        recommendation_scores += rating_matrix[i]
    # 推荐评分从高到低排序
    sorted_indices = np.argsort(-recommendation_scores)
    return [(products[i], recommendation_scores[i]) for i in sorted_indices]

# 为C1推荐产品
recommendations = recommend_customer(rating_matrix, similarity_matrix)
print("推荐结果：", recommendations)
```

#### 5.2.3 推荐系统实现

```python
def recommend_system(customer_id):
    # 示例代码，实际应用中需要根据具体数据进行调整
    customer_ratings = df[df['客户'] == customer_id].drop('客户', axis=1).values
    recommendations = {}
    for i in range(len(rating_matrix)):
        if i == customer_id:
            continue
        similarity = similarity_matrix[i][customer_id]
        recommendations[i] = similarity
    recommendations = sorted(recommendations.items(), key=lambda x: -x[1])
    return recommendations

recommendations = recommend_system(0)
print("推荐结果：", recommendations)
```
