```python
import pygad

# 遗传算法优化
def fitness_func(solution, solution_idx):
    return np.sum(solution)

ga = pygad.GA(
    num_generations=10,
    num_parents_mating=2,
    population_size=10,
    fitness_function=fitness_func,
    crossover_type='single_point',
    mutation_type='random',
    mutation_rate=0.1
)

ga.run()
```
