import numpy as np

def fitness(x):
    return x[0]**2 + x[1]**2

def genetic_algorithm(iterations=100):
    population = np.random.rand(100, 2)
    for _ in range(iterations):
        fitness_scores = np.apply_along_axis(fitness, 1, population)
        selected = population[np.argsort(fitness_scores)][:50]
        selected = selected.astype(float)
        selected[:, 0] += 0.1 * np.random.randn(50)
        selected[:, 1] += 0.1 * np.random.randn(50)
        population = np.vstack((selected, np.random.rand(50, 2)))
    return population

result = genetic_algorithm()
print(result)
```

import numpy as np
import matplotlib.pyplot as plt

def simulate_climate_control():
    # 初始化数据
    temperatures = np.random.normal(25, 2, 100)
    target = 25
    # 调节过程
    adjusted_temperatures = []
    for temp in temperatures:
        if temp < target:
            adjusted = temp + 1
        elif temp > target:
            adjusted = temp - 1
        else:
            adjusted = temp
        adjusted_temperatures.append(adjusted)
    # 可视化
    plt.plot(temperatures, label='原始温度')
    plt.plot(adjusted_temperatures, label='调整后温度')
    plt.legend()
    plt.show()

simulate_climate_control()
```
