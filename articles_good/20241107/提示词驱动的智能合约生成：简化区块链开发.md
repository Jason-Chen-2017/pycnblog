// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract SimpleCrowdfunding {
    struct Campaign {
        string name;
        uint256 target;
        uint256 deadline;
        uint256 amountRaised;
        mapping(address => uint256) contributions;
        bool campaignSuccessful;
    }

    Campaign[] public campaigns;

    function createCampaign(
        string memory name,
        uint256 target,
        uint256 deadline
    ) public {
        Campaign memory campaign = Campaign({
            name: name,
            target: target,
            deadline: deadline,
            amountRaised: 0
        });
        campaigns.push(campaign);
    }

    function contributeToCampaign(uint256 campaignId) public payable {
        require(
            msg.value > 0,
            "You need to contribute a positive amount!"
        );
        Campaign storage campaign = campaigns[campaignId];
        require(
            block.timestamp < campaign.deadline,
            "The campaign has already ended!"
        );
        campaign.contributions[msg.sender] += msg.value;
        campaign.amountRaised += msg.value;
    }

    function campaignSuccessful(uint256 campaignId) public {
        Campaign storage campaign = campaigns[campaignId];
        require(
            block.timestamp >= campaign.deadline,
            "The campaign is still ongoing!"
        );
        require(
            campaign.amountRaised >= campaign.target,
            "The campaign has not reached its target!"
        );
        campaign.campaignSuccessful = true;
    }

    function releaseFunds(uint256 campaignId) public {
        Campaign storage campaign = campaigns[campaignId];
        require(
            msg.sender == owner,
            "Only the owner can release funds!"
        );
        require(
            campaign.campaignSuccessful,
            "The campaign has not been successful!"
        );
        payable(msg.sender).transfer(campaign.amountRaised);
    }
}

    struct Order {
        address owner;
        bytes32 assetA;
        bytes32 assetB;
        uint256 amountA;
        uint256 amountB;
        bool filled;
    }

    mapping(bytes32 => mapping(bytes32 => Order[])) public orders;

    function createOrder(
        bytes32 assetA,
        bytes32 assetB,
        uint256 amountA,
        uint256 amountB
    ) public {
        require(amountA > 0 && amountB > 0, "Invalid amount!");
        require(amountA <= balanceOf(assetA)[msg.sender], "Insufficient balance!");
        require(amountB <= balanceOf(assetB)[msg.sender], "Insufficient balance!");

        bytes32 orderKey = keccak256(abi.encodePacked(assetA, assetB));
        orders[orderKey].push(Order({
            owner: msg.sender,
            assetA: assetA,
            assetB: assetB,
            amountA: amountA,
            amountB: amountB,
            filled: false
        }));
    }

    function fillOrder(
        bytes32 assetA,
        bytes32 assetB,
        uint256 index
    ) public {
        require(index < orders[keccak256(abi.encodePacked(assetA, assetB))].length, "Invalid index!");
        Order storage order = orders[keccak256(abi.encodePacked(assetA, assetB))][index];
        require(!order.filled, "Order already filled!");
        require(order.owner != msg.sender, "Cannot fill your own order!");

        );
        require(sent, "Failed to send Ether!");

            abi.encodeWithSignature("deposit(address)", assetB)
        );
        require(sent, "Failed to send Ether!");

        order.filled = true;
