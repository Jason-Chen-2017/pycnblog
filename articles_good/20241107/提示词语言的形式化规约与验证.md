```
    }

    }

        trigger = START
        action = MOVE robot forward
    }
    
    Transition T1_S2 {
        trigger = TURN robot right
        action = TURN robot right
    }
    
    Transition T2_S0 {
        trigger = TURN robot left

    }

    FinalState S2
}
```
