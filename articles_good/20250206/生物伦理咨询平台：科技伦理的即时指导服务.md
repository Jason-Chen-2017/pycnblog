from transformers import pipeline

app = Flask(__name__)
nlp = pipeline("question-generation")

@app.route('/api/question', methods=['POST'])
def handle_question():
    data = request.json
    question = data['query']
    # 使用NLP模型生成回答
    answers = nlp(question)
    return jsonify({'result': answers})
