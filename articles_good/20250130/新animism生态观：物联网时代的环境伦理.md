import requests
import pandas as pd

def fetch_energy_data(device_id):
    url = f"http://example.com/energy_data?device_id={device_id}"
    response = requests.get(url)
    if response.status_code == 200:
        data = response.json()
        return data
    else:
        return None

def clean_data(data):
    # 数据清洗逻辑
    df = pd.DataFrame(data)
    df.dropna(inplace=True)
    return df

def calculate_energy_consumption(df):
    # 计算能耗逻辑
    energy_consumption = df['power'].sum() * df['time']
    return energy_consumption

def monitor_energy_consumption(device_id):
    data = fetch_energy_data(device_id)
    if data:
        df = clean_data(data)
        consumption = calculate_energy_consumption(df)
        print(f"Device {device_id} consumed {consumption} kWh.")
    else:
        print("Failed to fetch data.")

# 示例
