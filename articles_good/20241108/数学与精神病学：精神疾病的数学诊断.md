def min_max_scaling(data):

    between_class_scatter = np.zeros((n_components, n_components))
    within_class_scatter = np.zeros((n_components, n_components))
    for class_label in np.unique(labels):
        class_data = data[labels == class_label]
        mean_class_data = np.mean(class_data, axis=0)
        between_class_scatter += (mean_class_data - np.mean(data)) * (mean_class_data - np.mean(data))
