```python
import numpy as np
import matplotlib.pyplot as plt

# 生成模拟引力波信号
def generate_gravity_wave(length, amplitude):
    t = np.linspace(0, length, num=length)
    signal = amplitude * np.sin(2 * np.pi * t)
    return signal

# 可视化引力波信号
def visualize_gravity_wave(signal):
    plt.plot(signal)
    plt.xlabel('Time (s)')
    plt.ylabel('Amplitude')
    plt.title('Simulated Gravity Wave Signal')
    plt.show()

# 测试代码
signal = generate_gravity_wave(100, 1)
visualize_gravity_wave(signal)
```
