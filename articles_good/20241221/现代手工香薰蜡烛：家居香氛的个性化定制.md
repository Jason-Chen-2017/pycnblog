from django.shortcuts import render
from .models import Material, Scent, Mold, CustomCandle
from .forms import CustomCandleForm

def customize_candle(request):
    if request.method == 'POST':
        form = CustomCandleForm(request.POST)
        if form.is_valid():
            material_id = form.cleaned_data['material']
            scent_id = form.cleaned_data['scent']
            mold_id = form.cleaned_data['mold']
            
            # Save custom candle to database
            custom_candle = CustomCandle(
                material_id=material_id,
                scent_id=scent_id,
                mold_id=mold_id
            )
            custom_candle.save()
            
            return render(request, 'confirmation.html')
    else:
        form = CustomCandleForm()

    return render(request, 'customize_candle.html', {'form': form})
```

from django.shortcuts import render
from .models import Material, Scent, Mold, CustomCandle
from .forms import CustomCandleForm

def customize_candle(request):
    if request.method == 'POST':
        form = CustomCandleForm(request.POST)
        if form.is_valid():
            material_id = form.cleaned_data['material']
            scent_id = form.cleaned_data['scent']
            mold_id = form.cleaned_data['mold']
            
            # Save custom candle to database
            custom_candle = CustomCandle(
                material_id=material_id,
                scent_id=scent_id,
                mold_id=mold_id
            )
            custom_candle.save()
            
            return render(request, 'confirmation.html')
    else:
        form = CustomCandleForm()

    return render(request, 'customize_candle.html', {'form': form})
```
