import numpy as np

def lidar_dataProcessing(data):
    # 假设data是一个包含位置信息的numpy数组
    distances = np.linalg.norm(data, axis=1)
    # 计算距离
    angles = np.arctan2(data[:, 1], data[:, 0])
    # 计算角度
    return distances, angles

# 示例数据

import socket

def send_data_to_5G(data, server_ip, server_port):
    # 创建TCP套接字
    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    # 连接服务器
    sock.connect((server_ip, server_port))
    # 发送数据
    sock.sendall(data)
    # 接收服务器响应
    response = sock.recv(1024)
    # 关闭套接字
    sock.close()
    return response
