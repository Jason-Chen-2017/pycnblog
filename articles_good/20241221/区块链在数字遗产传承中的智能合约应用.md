        owner = msg.sender;
        for (uint256 i = 0; i < _heirs.length; i++) {

        }

        distributionDeadline = _distributionDeadline;

    function distributeLegacy() external {
        require(msg.sender == owner || isHeir[msg.sender], "Only owner or heirs can call this function");

        uint256 shareValue = address(this).balance / totalShares;

        require(_heir != address(0), "Heir address cannot be the zero address");
        isHeir[_heir] = true;
    }

    receive() external payable {
        totalShares++;
    }
}
