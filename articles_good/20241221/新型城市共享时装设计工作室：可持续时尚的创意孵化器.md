# 资源管理模块示例代码
class ResourceManagement:
    def __init__(self):
        self.resources = {'design_software': [], '3d_printer': [], 'fabric_library': [], 'mold_library': []}

    def add_resource(self, resource_type, resource_details):
        if resource_type in self.resources:
            self.resources[resource_type].append(resource_details)
        else:
            print(f"Invalid resource type: {resource_type}")

    def remove_resource(self, resource_type, resource_id):
        if resource_type in self.resources:
            self.resources[resource_type].remove(resource_id)
        else:
            print(f"Invalid resource type: {resource_type}")

    def list_resources(self):
        for resource_type, resources in self.resources.items():
            print(f"{resource_type}: {resources}")
```

# 在线协作平台模块示例代码
class CollaborationPlatform:
    def __init__(self):
        self.users = {}
        self.tasks = {}

    def register_user(self, username, role):
        self.users[username] = role

    def create_task(self, task_id, task_details):
        self.tasks[task_id] = task_details

    def assign_task(self, username, task_id):
        if username in self.users and task_id in self.tasks:
            self.tasks[task_id]['assigned_to'] = username
        else:
            print("Invalid username or task ID")

    def list_tasks(self):
        for task_id, task_details in self.tasks.items():
            print(f"Task ID: {task_id}, Details: {task_details}")
```
