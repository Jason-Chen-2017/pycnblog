  # 输入职位描述
  input_text = "我们需要一名Python开发工程师，要求3年以上经验，熟悉Django框架。"
  inputs = tokenizer(input_text, return_tensors="np", padding=True, truncation=True)
  outputs = model(**inputs)
  job_vector = outputs.last_hidden_state[:, 0, :].numpy()
