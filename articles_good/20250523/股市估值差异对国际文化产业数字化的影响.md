  ```python
  import pandas as pd

  def calculate_valuation_difference(df):
      df['Valuation Difference'] = df['Market Value'] - df['Intrinsic Value']
      return df

  # 示例数据
  data = {
      'Market Value': [100, 80, 90],
      'Intrinsic Value': [80, 70, 85]
  }
  df = pd.DataFrame(data)
  result = calculate_valuation_difference(df)
  print(result)
  ```

- 5.2.2 数字化影响分析代码

  ```python
  import matplotlib.pyplot as plt

  def plot_impact_on_digitalization(df):

      plt.xlabel('Valuation Difference')
      plt.ylabel('Digital Investment')
      plt.legend()
      plt.show()

  # 示例数据
  data = {
      'Valuation Difference': [20, 10, 5],
  ```
