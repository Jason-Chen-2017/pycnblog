def execute_strategy(data, signal):

    for i in range(len(signal)):
        if i == 0:
            positions.append(signal[i])
        else:
            if signal[i] != positions[-1]:
                positions.append(signal[i])
            else:
                positions.append(positions[-1])
    return positions
