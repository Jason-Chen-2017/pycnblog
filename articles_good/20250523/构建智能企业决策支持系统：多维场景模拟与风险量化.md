import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestRegressor
from sklearn.metrics import mean_squared_error

# 加载数据
data = pd.read_csv('decision_data.csv')

# 特征选择
features = data.drop('target', axis=1)
target = data['target']

# 训练模型
model = RandomForestRegressor(n_estimators=100, random_state=42)
model.fit(features, target)

# 模拟运行
def simulate_scenario(params):
    predictions = model.predict(params)
    return predictions

# 分析案例
test_params = features.iloc[0].values.reshape(1, -1)
predicted_value = simulate_scenario(test_params)
print(f"预测值：{predicted_value}")
