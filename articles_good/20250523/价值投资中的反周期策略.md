```python
import pandas as pd

# 示例：计算股价与行业指数的偏离度
def calculate偏离度(stock_data, industry_data):
    stock_return = stock_data['Adj Close'].pct_change()
    industry_return = industry_data['Adj Close'].pct_change()
    deviation = stock_return - industry_return
    return deviation.abs().mean()
```
