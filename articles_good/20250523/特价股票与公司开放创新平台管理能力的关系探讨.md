import numpy as np
from sklearn.decomposition import PCA

def assess_management_capacity(data):
    pca = PCA(n_components=2)
    principal_components = pca.fit_transform(data[['factor1', 'factor2', 'factor3']])
    return principal_components

data = pd.DataFrame({
    'factor1': [1, 2, 3],
    'factor2': [4, 5, 6],
    'factor3': [7, 8, 9]
})
print(assess_management_capacity(data))
