### 6.2.1 数据处理代码
```python
import pandas as pd
import numpy as np

def data_cleaning(data):
    data.dropna(inplace=True)
    data['income'] = data['income'].astype(float)
    return data

def data_transformation(data):
    data['debt_to_income_ratio'] = data['debt'] / data['income']
    return data

# 示例数据
data = {
    'name': ['A', 'B', 'C'],
    'income': [5000, 6000, 4500],
    '支出': [3000, 3500, 2800],
    '资产': [100000, 80000, 90000],
    '负债': [50000, 40000, 45000]
}
df = pd.DataFrame(data)
df = data_cleaning(df)
df = data_transformation(df)
print(df)
```

### 6.2.2 模型计算代码
```python
from sklearn.tree import DecisionTreeClassifier
from sklearn.metrics import accuracy_score

def risk_assessment(data):
    model = DecisionTreeClassifier()
    model.fit(data[['income', '支出', '资产', '负债']], data['风险等级'])
    return model

# 示例数据
data = {
    'income': [5000, 6000, 4500],
    '支出': [3000, 3500, 2800],
    '资产': [100000, 80000, 90000],
    '负债': [50000, 40000, 45000],
    '风险等级': [2, 3, 1]
}
df = pd.DataFrame(data)
model = risk_assessment(df)
print("模型训练完成，准确率为：", accuracy_score(df['风险等级'], model.predict(df[['income', '支出', '资产', '负债']])))

```
