import numpy as np

# 粒子群优化（PSO）算法
def particle_swarm_optimization(func, bounds, num_particles=50, max_iter=100, w=0.5, c1=1, c2=2):
    # 初始化粒子
    particles = np.random.rand(num_particles, len(bounds))
    velocities = np.zeros_like(particles)
    best_particles = particles.copy()
    best_scores = func(best_particles)
    best_global = best_particles[np.argmin(best_scores)]

    for _ in range(max_iter):
        for i in range(num_particles):
            # 更新速度和位置
            r1 = np.random.rand()
            r2 = np.random.rand()
            velocities[i] = (
                w * velocities[i]
                + c1 * r1 * (best_particles[i] - particles[i])
                + c2 * r2 * (best_global - particles[i])
            )
            particles[i] += velocities[i]

            # 约束处理
            particles[i] = np.clip(particles[i], bounds[:, 0], bounds[:, 1])

            # 计算适应度
            score = func(particles[i])
            if score < best_scores[i]:
                best_scores[i] = score
                best_particles[i] = particles[i]

                # 更新个体最优
                if score < np.min(best_scores):
                    best_global = particles[i]
                    best_scores = np.array([np.inf] * num_particles)
                    best_scores[i] = score

    return best_global, best_particles

# 目标函数
