# 初始化量子位

psi = psi / np.linalg.norm(psi, axis=1)[:, np.newaxis]

# 初始化适应度函数

# 迭代过程
for _ in range(num_generations):
    # 计算适应度
    fitness = np.apply_along_axis(fitness_func, 1, psi)

    # 计算选择概率
    p = fitness / np.sum(fitness)

    # 计算交叉概率
    cross_prob = np.random.rand(num_individuals, num_bits) < p

    # 交叉操作

    # 计算变异概率
    mut_prob = np.random.rand(num_individuals, num_bits) < 0.1

    # 变异操作
    psi[mut_prob] = 1 - psi[mut_prob]

    # 输出当前代最优适应度
    print(f"Generation {_}: Best Fitness = {np.max(fitness)}")
