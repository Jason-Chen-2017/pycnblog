**1.1 多尺度推理的概念**

多尺度推理（Multi-scale Inference）是指在处理复杂问题时，考虑问题的不同尺度，通过在不同尺度上进行推理和决策，从而得到更全面和准确的结论。多尺度推理的核心在于将问题分解为不同的层次，每个层次都专注于特定尺度的信息，然后将这些信息整合起来，形成对整个问题的理解。

import tensorflow as tf
import tensorflow.keras.layers as layers

# 定义层次图注意力模型
class HierarchicalGraphAttentionModel(tf.keras.Model):
    def __init__(self, num_layers, d_model, num_heads, dff, input_shape):
        super(HierarchicalGraphAttentionModel, self).__init__()
        self.num_layers = num_layers
        self.d_model = d_model
        self.num_heads = num_heads
        self.dff = dff
        self.input_shape = input_shape

        # 定义模型层
        self.layers = [
            layers.Dense(d_model, activation='relu')
            for _ in range(num_layers)
        ]
        self.attention_layers = [
            layers.MultiHeadAttention(num_heads=num_heads, key_dim=d_model // num_heads)
            for _ in range(num_layers)
        ]
        self.output_layer = layers.Dense(input_shape[-1])

    def call(self, inputs, training=False):
        # 输入数据
        x = inputs

        # 模型层处理
        for i in range(self.num_layers):
            x = self.layers[i](x)

        # 注意力层处理
        for i in range(self.num_layers):
            x = self.attention_layers[i](x, x)

        # 输出层处理
        x = self.output_layer(x)

        return x

# 实例化模型
