def train_model(model, optimizer, epochs):

            loss = criterion(outputs, labels)
            optimizer.zero_grad()
            loss.backward()
            optimizer.step()
    return model

            predictions = torch.argmax(outputs, dim=1)
            accuracy = (predictions == labels).mean()
