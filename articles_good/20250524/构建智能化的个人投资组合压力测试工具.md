##### 3.4 Python实现代码示例

```python
import numpy as np

def monte_carlo_simulation(returns, num_simulations=1000):
    simulations = []
    for _ in range(num_simulations):
        # 生成随机情景
        scenario = np.random.multivariate_normal(
            mean=[0, 0], 
            cov=np.array([[1, 0.5], [0.5, 1]]), 
            size=1
        )
        simulations.append(scenario)
    return simulations

# 示例数据
returns = np.array([[0.1, 0.2], [0.3, 0.1]])

simulations = monte_carlo_simulation(returns)
print(simulations)
```

  import numpy as np
  import matplotlib.pyplot as plt

  def calculate VaR(simulated_returns, confidence_level=0.95):
      sorted_returns = np.sort(simulated_returns)
      VaR = sorted_returns[int(len(simulated_returns) * (1 - confidence_level))]
      return VaR

  # 示例数据
  simulated_returns = np.random.normal(0, 1, 1000)
  VaR_95 = calculate VaR(simulated_returns, 0.95)
  print(f"VaR at 95% confidence level: {VaR_95}")
