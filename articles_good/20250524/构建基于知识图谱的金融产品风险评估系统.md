```python
def fetch_data(url):

    soup = BeautifulSoup(response.text, 'html.parser')
    data = []
    for item in soup.find_all('div', class_='product-item'):
        product_name = item.find('h3').text
        product_info = item.find('p').text
        data.append({'name': product_name, 'info': product_info})
    return data
```

```python
from neo4j.exceptions import Neo4jError

# 图数据库操作
class GraphDBManager:
    def __init__(self, uri, user, password):
        self.driver = GraphDatabase.driver(uri, auth=(user, password))
    
    def create_node(self, label, properties):
        with self.driver.session() as session:
            session.write_transaction(
                lambda tx: tx.create_node(label, properties)
            )
    
    def create_relationship(self, start_label, start_id, rel_type, end_label, end_id):
```

```python
import numpy as np
from sklearn.metrics.pairwise import cosine_similarity

# 风险评估计算
def calculate_risk(entity_embeddings, threshold=0.7):
    similarities = cosine_similarity(entity_embeddings)
    risk_score = np.mean(similarities > threshold)
    return risk_score
```
