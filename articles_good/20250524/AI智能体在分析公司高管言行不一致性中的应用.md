#### 3.1 文本相似度计算模型

- **3.1.1 余弦相似度公式**  
  $$ \text{余弦相似度} = \frac{\vec{A} \cdot \vec{B}}{|\vec{A}| \cdot |\vec{B}|} $$  
  其中，$\vec{A}$和$\vec{B}$分别表示两个文本的向量表示。

- **3.1.2 编辑距离公式**  
  $$ \text{编辑距离} = \text{min}(d_{\text{删除}}, d_{\text{插入}}, d_{\text{替换}}) $$  
  用于衡量两个文本的编辑成本。

import numpy as np
from sklearn.metrics.pairwise import cosine_similarity
from sklearn.svm import SVC
from sklearn.feature_extraction.text import TfidfVectorizer
import spacy
from transformers import pipeline

# 初始化模型
nlp = spacy.load("en_core_web_sm")
sentiment_analyzer = pipeline("sentiment-analysis")

def compute_cosine_similarity(text1, text2):
    # 文本向量化
    vectorizer = TfidfVectorizer()
    vectors = vectorizer.fit_transform([text1, text2])
    return cosine_similarity(vectors[0], vectors[1])

def analyze_high_management_behavior(behavior_data):
    # 文本清洗
    cleaned_data = [text.strip() for text in behavior_data]
    # 特征提取
    tfidf_matrix = TfidfVectorizer().fit_transform(cleaned_data)
    # 模型训练
    model = SVC()
    model.fit(tfidf_matrix, [0, 1])
    # 预测结果
    return model.predict(tfidf_matrix)

# 示例数据
behavior_data = [
    "我们将继续推进全球化战略，扩大市场份额。",
    "今年我们将收缩海外业务，专注国内市场。"
]

# 文本相似度计算
similarity = compute_cosine_similarity(behavior_data[0], behavior_data[1])
print(f"文本相似度：{similarity[0][0]}")

# 情感分析
result = sentiment_analyzer(behavior_data[0])
