  import numpy as np
  from scipy.optimize import minimize

  def portfolio_optimization(weights, returns, risk_free_rate):
      n = len(weights)
      return minimize(lambda x: -x.T @ returns @ x, 
                      np.ones(n)/n, 
                      bounds= [(0,1) for _ in range(n)], 
                      method='SLSQP')
  
  # 示例调用
