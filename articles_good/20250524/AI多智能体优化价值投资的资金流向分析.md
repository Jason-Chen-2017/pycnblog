import numpy as np

# 数据采集
data = pd.read_csv('market_data.csv')

# 资金流向计算
def calculate_inflows(data):
    inflows = []
    for i in range(len(data)):
        if i == 0:
            inflows.append(0)
        else:
            inflows.append(data['close'][i] - data['close'][i-1])
    return inflows

inflows = calculate_inflows(data)
print(inflows)
