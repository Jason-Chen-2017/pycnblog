```python
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns

# 读取CMB观测数据
data = np.load('cmb_data.npy')

# 数据预处理
def preprocess_data(data):
    # 去除异常值
    clean_data = np.delete(data, np.where(np.isnan(data)))
    # 标准化数据
    normalized_data = (clean_data - np.mean(clean_data)) / np.std(clean_data)
    return normalized_data

preprocessed_data = preprocess_data(data)

# 数据可视化
def plot_data(data):
    plt.figure(figsize=(10, 6))
    sns.scatterplot(x=data['RA'], y=data['DEC'], hue=data['TEMP'], palette='viridis')
    plt.title('CMB Observation Data')
    plt.xlabel('Right Ascension')
    plt.ylabel('Declination')
    plt.show()

plot_data(preprocessed_data)
```
