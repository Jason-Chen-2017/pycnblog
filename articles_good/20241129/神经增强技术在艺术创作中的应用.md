```python
# 反向传播算法示例

import numpy as np

# 定义激活函数
def sigmoid(x):
    return 1 / (1 + np.exp(-x))

# 定义反向传播算法
def backpropagation(x, y, weights):
    output = sigmoid(np.dot(x, weights))
    error = y - output
    d_output = output * (1 - output)
    d_weights = np.dot(x.T, d_output * error)
    return d_weights

# 示例数据
x = np.array([1, 0])
y = np.array([0])
weights = np.random.rand(2, 1)

# 运行反向传播算法
d_weights = backpropagation(x, y, weights)

print("权重更新：", d_weights)
```
