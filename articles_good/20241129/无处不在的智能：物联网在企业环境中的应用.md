```python
# IoT Data Collection and Processing Project

import paho.mqtt.client as mqtt
import pymysql
import time
import numpy as np

# MQTT 服务器配置
MQTT_SERVER = "localhost"
MQTT_PORT = 1883
MQTT_TOPIC = "sensor_data"

# MySQL 数据库配置
DB_HOST = "localhost"
DB_PORT = 3306
DB_USER = "root"
DB_PASSWORD = "password"
DB_NAME = "iot_db"

# MQTT 客户端连接
def on_connect(client, userdata, flags, rc):
    print("Connected with result code "+str(rc))
    client.subscribe(MQTT_TOPIC)

# MQTT 消息处理
def on_message(client, userdata, msg):
    print(f"Received message '{msg.payload.decode()} on topic '{msg.topic}' with QoS {msg.qos} ")
    data = eval(msg.payload.decode())
    process_data(data)

# 数据处理函数
def process_data(data):
    temperature, pressure, speed = data
    preprocessed_data = preprocess_data(data)
    store_data(preprocessed_data)

# 数据预处理函数
def preprocess_data(data):
    temperature, pressure, speed = data
    temperature = np.mean([temperature] * 10)  # 取平均值
    pressure = np.mean([pressure] * 10)  # 取平均值
    speed = np.mean([speed] * 10)  # 取平均值
    return temperature, pressure, speed

# 数据存储函数
def store_data(data):
    temperature, pressure, speed = data
    connection = pymysql.connect(host=DB_HOST, port=DB_PORT, user=DB_USER, password=DB_PASSWORD, database=DB_NAME)
    try:
        with connection.cursor() as cursor:
            sql = "INSERT INTO sensor_data (temperature, pressure, speed) VALUES (%s, %s, %s)"
            cursor.execute(sql, (temperature, pressure, speed))
        connection.commit()
    except pymysql.Error as e:
        print(f"Error: {e}")
    finally:
        connection.close()

# MQTT 客户端初始化
client = mqtt.Client()
client.on_connect = on_connect
client.on_message = on_message
client.connect(MQTT_SERVER, MQTT_PORT, 60)

# 持续监听 MQTT 消息
client.loop_forever()
```
