```python
from googletrans import Translator

# 初始化翻译器
translator = Translator()

# 翻译文本
def translate_text(text, dest_language):
    translation = translator.translate(text, dest=dest_language)
    return translation.text

# 示例翻译
text_to_translate = "你好，我需要帮助。"
translated_text = translate_text(text_to_translate, "en")
print("翻译文本：", translated_text)

# 多语言输入处理
def process_multilingual_input(input_text):
    # 假设输入文本为中文和英文的混合
    translations = []
    for sentence in input_text.split('.'):
        if "中文" in sentence:
            translations.append(sentence)
        elif "English" in sentence:
            translation = translate_text(sentence, "en")
            translations.append(translation)
    return " ".join(translations)

# 示例多语言输入处理
input_text = "你好，我需要帮助。Hello, I need help."
processed_text = process_multilingual_input(input_text)
print("处理后的文本：", processed_text)
```
