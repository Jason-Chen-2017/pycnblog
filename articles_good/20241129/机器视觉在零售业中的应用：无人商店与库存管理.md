#### 1.2 图像处理基础

**核心算法原理讲解**

图像处理是机器视觉的基础，其中涉及多种核心算法，如图像滤波和边缘检测。

- **图像滤波**

图像滤波是一种用于减少图像噪声的算法。常见的滤波方法包括均值滤波、高斯滤波和中值滤波。

以下是一个简单的Python代码示例，用于实现均值滤波：

```python
import numpy as np

def mean_filter(image, filter_size):
    # 初始化滤波结果
    filtered_image = np.zeros_like(image)

    # 遍历每个像素
    for i in range(image.shape[0]):
        for j in range(image.shape[1]):
            # 获取邻域像素值
            neighbors = image[i-filter_size[0]:i+filter_size[0]+1,
                             j-filter_size[1]:j+filter_size[1]+1]

            # 计算均值
            filtered_image[i, j] = np.mean(neighbors)

    return filtered_image
```

- **边缘检测**

边缘检测是图像处理中的另一关键步骤，用于识别图像中的显著边缘。常见的方法包括Sobel算子、Canny算子和Laplacian算子。

以下是一个简单的Python代码示例，用于实现Sobel边缘检测：

```python
import cv2

def sobel_edge_detection(image):
    # 使用Sobel算子进行边缘检测
    sobelx = cv2.Sobel(image, cv2.CV_64F, 1, 0, ksize=3)
    sobely = cv2.Sobel(image, cv2.CV_64F, 0, 1, ksize=3)

    # 计算边缘强度
    edge_strength = np.sqrt(sobelx**2 + sobely**2)

    # 转换为8位图像
    edge_image = cv2.convertScaleAbs(edge_strength)

    return edge_image
```

#### 1.3 特征提取与匹配

**核心算法原理讲解**

特征提取是图像处理中用于识别图像关键点的过程。常见的特征提取方法包括SIFT（尺度不变特征变换）和SURF（加速稳健特征）。特征匹配则是用于将不同图像中的关键点进行匹配，从而实现图像的匹配与对齐。

以下是一个简单的Python代码示例，用于实现SIFT特征提取和匹配：

```python
import cv2

def sift_feature_extraction_and_matching(image1, image2):
    # 使用SIFT算法进行特征提取
    sift = cv2.xfeatures2d.SIFT_create()
    keypoints1, descriptors1 = sift.detectAndCompute(image1, None)
    keypoints2, descriptors2 = sift.detectAndCompute(image2, None)

    # 使用FLANN算法进行特征匹配
    FLANN_INDEX_KDTREE = 1
    index_params = dict(algorithm=FLANN_INDEX_KDTREE, trees=5)
    search_params = dict(checks=50)

    flann = cv2.FlannBasedMatcher(index_params, search_params)
    matches = flann.knnMatch(descriptors1, descriptors2, k=2)

    # 提取高质量匹配点
    good_matches = []
    for m, n in matches:
        if m.distance < 0.7 * n.distance:
            good_matches.append(m)

    return good_matches, keypoints1, keypoints2
```

import cv2
import numpy as np
import tensorflow as tf

# 顾客进店检测
def detect_customer(frame):
    # 转换为灰度图像
    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
    
    # 使用皮肤颜色模型检测顾客
    lower = np.array([0, 48, 80], dtype=np.uint8)
    upper = np.array([20, 255, 255], dtype=np.uint8)
    mask = cv2.inRange(gray, lower, upper)
    
    # 检测顾客轮廓
    contours, _ = cv2.findContours(mask, cv2.RETR_TREE, cv2.CHAIN_APPROX_SIMPLE)
    customers = []
    for contour in contours:
        if cv2.contourArea(contour) > 500:
            x, y, w, h = cv2.boundingRect(contour)
            customers.append((x, y, w, h))
    
    return customers

# 商品识别与分类
def recognize_goods(frame, model):
    # 转换为灰度图像
    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
    
    # 使用商品特征进行识别
    template = cv2.imread('template.jpg', cv2.IMREAD_GRAYSCALE)
    w, h = template.shape[:2]
    
    # 进行模板匹配
    result = cv2.matchTemplate(gray, template, cv2.TM_CCOEFF_NORMED)
    threshold = 0.8
    loc = np.where(result >= threshold)
    
    goods = []
    for pt in zip(*loc[::-1]):
        cv2.rectangle(frame, pt, (pt[0] + w, pt[1] + h), (0, 255, 0), 2)
        
        # 预测商品类别
        image = cv2.resize(frame[pt[1]:pt[1]+h, pt[0]:pt[0]+w], (28, 28))
        image = image / 255.0
        prediction = model.predict(np.expand_dims(image, axis=0))
        predicted_class = np.argmax(prediction)
        
        goods.append(predicted_class)
    
    return goods

def recognize_goods(frame):
    # 转换为灰度图像
    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
    
    # 使用商品特征进行识别
    template = cv2.imread('template.jpg', cv2.IMREAD_GRAYSCALE)
    w, h = template.shape[:2]
    
    # 进行模板匹配
    result = cv2.matchTemplate(gray, template, cv2.TM_CCOEFF_NORMED)
    threshold = 0.8
    loc = np.where(result >= threshold)
    
    goods = []
    for pt in zip(*loc[::-1]):
        cv2.rectangle(frame, pt, (pt[0] + w, pt[1] + h), (0, 255, 0), 2)
        
        # 获取商品ID
        template_id = cv2.matchTemplate(result, template, cv2.TM_CCOEFF_NORMED)
        best_match = np.argmax(template_id)
        goods.append(best_match)
    
    return goods

# 库存盘点
def inventory_check(frame, conn):
    # 识别商品
    goods = recognize_goods(frame)
    
    # 更新库存
    for good in goods:
        cursor.execute("SELECT quantity FROM products WHERE product_id=?", (good,))
        result = cursor.fetchone()
        if result:
            quantity = result[0]
            cursor.execute("UPDATE products SET quantity=? WHERE product_id=?", (quantity - 1, good))
            print(f"商品{good}库存更新成功：{quantity - 1}")
        else:
            print(f"商品{good}未在数据库中找到")
    
    # 提交更改
    conn.commit()

# 运行库存管理系统
while True:
    # 读取摄像头帧
    ret, frame = cap.read()
    
    # 库存盘点
    inventory_check(frame, conn)
    
    # 显示摄像头帧
    cv2.imshow('frame', frame)
    
    # 按下'q'键退出
    if cv2.waitKey(1) & 0xFF == ord('q'):
        break

# 释放摄像头资源
cap.release()
cv2.destroyAllWindows()

# 关闭数据库连接
conn.close()

# 需求预测
def demand_prediction(conn):
    cursor = conn.cursor()
    cursor.execute("SELECT product_id, SUM(demand) as total_demand FROM demand_prediction GROUP BY product_id")
    demand_data = cursor.fetchall()
    
    df = pd.DataFrame(demand_data, columns=["product_id", "total_demand"])
    df["forecast"] = pd.to_datetime(df["forecast"])
    
    # 使用移动平均法进行需求预测
    df["moving_average"] = df["total_demand"].rolling(window=3).mean()
    df["predicted_demand"] = df["moving_average"].shift(1)
    
    return df

# 库存优化
def inventory_optimization(df):
    # 定义优化模型
    product_ids = df["product_id"].unique()
    n = len(product_ids)
    
    # 初始化变量
    x = cp.Variable(n, non_negative=True)
    y = cp.Variable(n, non_negative=True)
    
    # 目标函数
    objective = cp.Minimize(cp.sum(cp.abs(x - y)))
    
    # 约束条件
    for i in range(n):
        product_id = product_ids[i]
        current_inventory = df.loc[df["product_id"] == product_id, "current_inventory"].iloc[-1]
        predicted_demand = df.loc[df["product_id"] == product_id, "predicted_demand"].iloc[-1]
        
        objective += cp.abs(x[i] - (current_inventory + predicted_demand))
    
    # 解优化模型
    prob = cp.Problem(objective, [])
    prob.solve()
    
    # 获取最优解
    optimal_inventory = [x[i].value for i in range(n)]
    
    return optimal_inventory

# 运输优化
def transport_optimization(df, optimal_inventory):
    # 定义运输模型
    product_ids = df["product_id"].unique()
    n = len(product_ids)
    m = df.shape[0]
    
    # 初始化变量
    z = cp.Variable((n, m), non_negative=True)
    
    # 目标函数
    objective = cp.Minimize(cp.sum(z))
    
    # 约束条件
    for i in range(n):
        product_id = product_ids[i]
        for j in range(m):
            date = df["forecast"][j]
            current_inventory = df.loc[(df["product_id"] == product_id) & (df["forecast"] == date), "current_inventory"].iloc[-1]
            predicted_demand = df.loc[(df["product_id"] == product_id) & (df["forecast"] == date), "predicted_demand"].iloc[-1]
            
            objective += z[i, j] * (optimal_inventory[i] - current_inventory + predicted_demand)
    
    # 解优化模型
    prob = cp.Problem(objective, [])
    prob.solve()
    
    # 获取最优解
    optimal_transport = [z[i, j].value for i in range(n) for j in range(m)]
    
    return optimal_transport

# 运行供应链优化系统
def run_supply_chain_optimization():

    df = demand_prediction(conn)
    print("需求预测结果：")

    # 库存优化
    optimal_inventory = inventory_optimization(df)
    print("库存优化结果：")
    print(optimal_inventory)
    
    # 运输优化
    optimal_transport = transport_optimization(df, optimal_inventory)
    print("运输优化结果：")
    print(optimal_transport)
    
    # 更新数据库
    cursor.execute('''CREATE TABLE IF NOT EXISTS transport_optimization
                      (product_id INTEGER,
                       date DATE,
                       transport_quantity INTEGER)''')
    conn.commit()
    
    for i, product_id in enumerate(optimal_inventory):
        for j, date in enumerate(df["forecast"]):
            cursor.execute("INSERT INTO transport_optimization (product_id, date, transport_quantity) VALUES (?, ?, ?)", (product_id, date, optimal_transport[i * len(df["forecast"]) + j]))
        conn.commit()

run_supply_chain_optimization()

conn.close()
