    # 清洗文本
    cleaned_prompt = TextBlob(prompt).correct()
    # 分词
    tokens = cleaned_prompt.words
    return tokens

# 模型模块：生成输出
def generate_output(prompt):
    response = openai.Completion.create(
        engine="text-davinci-002",
        prompt=prompt,
        max_tokens=100
    )
    return response.choices[0].text.strip()

# 调整模块：调整输出
def adjust_output(prompt, output):
    adjusted_output = f"{prompt}\n{output}"
    return adjusted_output

# 输出模块：展示结果
def show_result(adjusted_output):
    print("生成的文本：")
    print(adjusted_output)

# 主程序
def main():
    while True:
        prompt = get_user_input()
        tokens = preprocess_data(prompt)
        output = generate_output(prompt)
        adjusted_output = adjust_output(prompt, output)
        show_result(adjusted_output)
        
        # 是否继续
        continue_QMARK_ = input("是否继续？（y/n）：")
        if continue_QMARK_ != 'y':
            break

if __name__ == "__main__":
    main()

- **输出模块：** `show_result()` 函数负责将最终输出展示给用户。
