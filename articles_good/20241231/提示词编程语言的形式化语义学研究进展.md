### 3.2 Python源代码讲解

以下是一个简单的Python源代码示例，用于演示如何从用户输入的提示词中提取语义信息，并生成抽象语法树（AST）。

```python
from antlr4 import *
from parserLexer import parserLexer
from parserParser import parserParser
from ast import Node

def parse_input(input_str):
    # 初始化解析器
    stream = InputStream(input_str)
    lexer = parserLexer(stream)
    stream.setLexer(lexer)
    parser = parserParser(stream)
    tree = parser.program()

    # 生成AST
    visitor = AstVisitor()
    ast = visitor.visit(tree)
    return ast

class AstVisitor(Visitor):

    def visitProgram(self, ctx: parserParser.ProgramContext):
        # 处理程序语句
        for stmt in ctx.statement():
            self.visit(stmt)

    def visitStatement(self, ctx: parserParser.StatementContext):
        # 处理语句
        if ctx.expression():
            self.visit(ctx.expression())

    def visitExpression(self, ctx: parserParser.ExpressionContext):
        # 处理表达式
        if ctx.INT():
            return Node(ctx.INT().getText())
        elif ctx.ID():
            return Node(ctx.ID().getText())
        else:
            return Node("Unknown")

# 测试
input_str = "print('Hello, world!')"
ast = parse_input(input_str)
print(ast)
```

$$
\phi = \forall x (P(x) \rightarrow \exists y (R(x, y) \land S(y)))
$$

这个公式表示，对于所有元素 x，如果 P(x) 成立，那么一定存在一个元素 y，使得 R(x, y) 和 S(y) 同时成立。
