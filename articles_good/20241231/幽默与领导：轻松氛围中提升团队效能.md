# app.py

from flask import Flask, request, jsonify
from flask_sqlalchemy import SQLAlchemy
import pika

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'mysql+pymysql://username:password@localhost/db_name'
db = SQLAlchemy(app)

class Message(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    sender_id = db.Column(db.Integer)
    receiver_id = db.Column(db.Integer)
    content = db.Column(db.Text)
    timestamp = db.Column(db.DateTime, default=db.func.current_timestamp())

@app.route('/send_message', methods=['POST'])
def send_message():
    data = request.get_json()
    sender_id = data['sender_id']
    receiver_id = data['receiver_id']
    content = data['content']
    
    new_message = Message(sender_id=sender_id, receiver_id=receiver_id, content=content)
    db.session.add(new_message)
    db.session.commit()

    # 发送消息到消息队列
    connection = pika.BlockingConnection(pika.ConnectionParameters('localhost'))
    channel = connection.channel()
    channel.queue_declare(queue='message_queue')
    channel.basic_publish(exchange='',
                          routing_key='message_queue',
                          body=f"{sender_id} -> {receiver_id}: {content}")
    connection.close()

    return jsonify({'status': 'success'})

if __name__ == '__main__':
    db.create_all()
    app.run(debug=True)
```

@app.route('/recommend_users', methods=['GET'])
def recommend_users():
    user_id = request.args.get('user_id')
    user = User.query.get(user_id)
    
    # 基于用户兴趣推荐其他用户
    df = pd.read_sql_query('SELECT * FROM user;', db.session.connection())
    similar_users = df[df['interests'].apply(lambda x: x in user.interests)].drop(user.id).sample(n=3)
    
    recommended_users = [{'id': user.id, 'name': user.name} for user in similar_users]
    
    return jsonify({'recommended_users': recommended_users})

if __name__ == '__main__':
    db.create_all()
    app.run(debug=True)
```

```python
# reminder.py

from flask import Flask, request, jsonify
from flask_sqlalchemy import SQLAlchemy
from datetime import datetime, timedelta

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'mysql+pymysql://username:password@localhost/db_name'
db = SQLAlchemy(app)

class Task(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    user_id = db.Column(db.Integer)
    title = db.Column(db.String(100))
    deadline = db.Column(db.DateTime)

@app.route('/add_task', methods=['POST'])
def add_task():
    data = request.get_json()
    user_id = data['user_id']
    title = data['title']
    deadline = datetime.strptime(data['deadline'], '%Y-%m-%d %H:%M:%S')

    new_task = Task(user_id=user_id, title=title, deadline=deadline)
    db.session.add(new_task)
    db.session.commit()

    # 计算提醒时间
    reminder_time = deadline - timedelta(minutes=30)
    send_reminder(reminder_time, new_task)

    return jsonify({'status': 'success'})

def send_reminder(reminder_time, task):
    # 发送提醒消息到消息队列
    connection = pika.BlockingConnection(pika.ConnectionParameters('localhost'))
    channel = connection.channel()
    channel.queue_declare(queue='reminder_queue')
    channel.basic_publish(exchange='',
                          routing_key='reminder_queue',
                          body=f"Task reminder for {task.title}: {reminder_time.strftime('%Y-%m-%d %H:%M:%S')}")
    connection.close()

if __name__ == '__main__':
    db.create_all()
    app.run(debug=True)
```
