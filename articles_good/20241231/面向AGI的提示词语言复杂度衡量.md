2. **复杂度评估模块**：该模块使用信息熵和条件熵模型对生成的提示词进行复杂度评估。代码示例如下：

   ```python
   import math
   
   def calculate_entropy(words):
       probabilities = [1 / len(words)] * len(words)
       entropy = -sum(p * math.log2(p) for p in probabilities)
       return entropy
   
   def calculate_conditional_entropy(words, context):
       context_words = context.split()
       word_probabilities = {word: 0 for word in set(words + context_words)}
       for word in words:
           word_probabilities[word] += 1
       for word in context_words:
           word_probabilities[word] += 1
       total_probability = sum(word_probabilities.values())
       conditional_entropy = -sum((word_probabilities[word] / total_probability) * math.log2(word_probabilities[word] / total_probability) for word in words)
       return conditional_entropy
   
   prompt = "今天的天气非常好，适合户外运动。"
   words = prompt.split()
   context = "今天的天气非常好。"
   entropy = calculate_entropy(words)
   conditional_entropy = calculate_conditional_entropy(words, context)
   print(f"信息熵：{entropy}")
   print(f"条件熵：{conditional_entropy}")
   ```

   def optimize_prompt(prompt, max_entropy=2.5, max_conditional_entropy=1.5):
