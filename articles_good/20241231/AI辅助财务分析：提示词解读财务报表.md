```python
# 财务报表解读算法示例代码
import jieba

# 解析文本中的财务数据
def parse_finance_data(text):
    # 关键词列表
    keywords = ["营业收入", "净利润", "现金流量"]
    
    # 分词
    words = jieba.cut(text)
    
    # 解析财务数据
    finance_data = {}
    for word in words:
        if word in keywords:
            finance_data[word] = text.split(word)[1].split("。")[0]
    
    return finance_data

# 示例文本
text = "2022年营业收入为5000万元，净利润为800万元，现金流量为1000万元。"

# 解析财务数据
finance_data = parse_finance_data(text)
print(finance_data)
```

```python
import jieba

# 关键词列表
keywords = ['营业收入', '净利润', '现金流量']

# 提取关键词
def extract_keywords(text):
    words = jieba.cut(text)
    return [word for word in words if word in keywords]

# 解析财务数据
def parse_finance_data(text):
    keywords = extract_keywords(text)
    finance_data = {}
    for keyword in keywords:
        value = text.split(keyword)[1].split('。')[0].strip()
        finance_data[keyword] = value
    return finance_data

# 示例文本
text = "2022年营业收入为5000万元，净利润为800万元，现金流量为1000万元。"

# 解析财务数据
finance_data = parse_finance_data(text)
print(finance_data)
```
