class Community:
    def __init__(self):
        self.posts = []

    def post(self, user, content):
        post = {'user': user, 'content': content}
        self.posts.append(post)
        print("您的帖子已发布。")

    def reply(self, user, post_index, content):
        post = self.posts[post_index]
        reply = {'user': user, 'content': content}
        post['replies'].append(reply)
        print("您的回复已发布。")

    def display_posts(self):
        for post in self.posts:
            print("用户：", post['user'])
            print("内容：", post['content'])
            for reply in post['replies']:
                print("回复：", reply['user'], "-", reply['content'])
            print()

community = Community()
community.post("用户1", "今天我尝试了深呼吸练习。")
community.reply("用户2", 0, "深呼吸真的很有效，我也试试！")
community.display_posts()
```

class Community:
    def __init__(self):
        self.posts = []

    def post(self, user, content):
        post = {'user': user, 'content': content}
        self.posts.append(post)
        print("您的帖子已发布。")

    def reply(self, user, post_index, content):
        post = self.posts[post_index]
        reply = {'user': user, 'content': content}
        post['replies'].append(reply)
        print("您的回复已发布。")

    def display_posts(self):
        for post in self.posts:
            print("用户：", post['user'])
            print("内容：", post['content'])
            for reply in post['replies']:
                print("回复：", reply['user'], "-", reply['content'])
            print()

community = Community()
community.post("用户1", "今天我尝试了深呼吸练习。")
community.reply("用户2", 0, "深呼吸真的很有效，我也试试！")
