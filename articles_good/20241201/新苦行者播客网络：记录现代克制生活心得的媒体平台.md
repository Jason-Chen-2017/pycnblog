# 计算每月可支配收入
def calculate_monthly_income(income, expenses):

    return disposable_income

def calculate_monthly_savings(disposable_income, savings_rate):
    savings = disposable_income * savings_rate
    return savings

# 计算年度储蓄金额
def calculate_annual_savings(monthly_savings):
    annual_savings = monthly_savings * 12
    return annual_savings

monthly_expenses = 3000
savings_rate = 0.2

# 计算结果

monthly_savings = calculate_monthly_savings(monthly_disposable_income, savings_rate)
annual_savings = calculate_annual_savings(monthly_savings)

print("每月可支配收入：", monthly_disposable_income)
print("每月储蓄金额：", monthly_savings)
print("年度储蓄金额：", annual_savings)
