def time_allocation(total_time, max_time):
    if total_time <= max_time:
        return total_time
    else:
        return max_time

# 心理恢复算法
def mental_recovery(mental_score):
    if mental_score >= 7:
        return "良好"
    elif mental_score >= 4:
        return "一般"
    else:
        return "较差"

# 身体恢复算法
def physical_recovery(physical_score):
    if physical_score >= 8:
        return "良好"
    elif physical_score >= 6:
        return "一般"
    else:
        return "较差"

# 健康状态评估模型
def health_status(physical_score, mental_score):
    return (physical_score * 0.5) + (mental_score * 0.5)

# 工作效率评估模型
def work_efficiency(digital_device_time, k):
    return 1 / (1 + exp(-k * digital_device_time))

# 社交能力评估模型
def social_ability(social_activity_time, m):
    return 1 / (1 + exp(-m * social_activity_time))

# 实例化参数

k = 2
