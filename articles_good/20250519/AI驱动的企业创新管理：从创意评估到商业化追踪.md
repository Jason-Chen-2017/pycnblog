- **9.3.1 创意评估模块代码**
  ```python
  import numpy as np
  import pandas as pd
  from sklearn.feature_extraction.text import TfidfVectorizer
  from sklearn.svm import SVC
  from sklearn.metrics import accuracy_score

  # 文本预处理
  def preprocess(text):
      return text.lower().strip()

  # 创意文本分类
  def creative_text_classification(X_train, y_train, X_test):
      vectorizer = TfidfVectorizer()
      X_train_vec = vectorizer.fit_transform(X_train)
      model = SVC()
      model.fit(X_train_vec, y_train)
      X_test_vec = vectorizer.transform(X_test)
      y_pred = model.predict(X_test_vec)
      return accuracy_score(y_test, y_pred)

  # 示例数据
  X_train = ["创新点显著", "技术可行", "市场需求大"]
  y_train = [1, 1, 1]
  X_test = ["创新点不足", "技术不可行", "市场需求小"]
  y_test = [0, 0, 0]

  accuracy = creative_text_classification(X_train, y_train, X_test)
  print("分类准确率:", accuracy)
  ```

  import pandas as pd
  from sklearn.metrics import mean_squared_error
  from statsmodels.tsa.arima_model import ARIMA

  # 时间序列预测
  def forecast_sales(train_data, n_periods=6):
      model = ARIMA(train_data, order=(1, 1, 1))
      model_fit = model.fit(disp=0)
      forecast = model_fit.forecast(n_periods)
      return forecast

  # 示例数据
  train_data = [100, 200, 300, 400, 500]
  forecast = forecast_sales(train_data)
  print("预测值:", forecast)
