pragma solidity ^0.8.0;

contract DigitalArtAuction {
    struct ArtPiece {
        uint256 tokenId;
        address owner;
        uint256 reservePrice;
        uint256 startTime;
        uint256 endTime;
    }

    ArtPiece[] public auctionList;

    function createAuction(uint256 tokenId, uint256 reservePrice, uint256 duration) public {
        ArtPiece memory newAuction = ArtPiece({
            tokenId: tokenId,
            owner: msg.sender,
            reservePrice: reservePrice,
            startTime: block.timestamp,
            endTime: block.timestamp + duration
        });
        auctionList.push(newAuction);
    }

    function bid(uint256 auctionIndex, uint256 amount) public payable {
        ArtPiece memory auction = auctionList[auctionIndex];
        require(auction.endTime >= block.timestamp, "Auction has ended");
        require(amount >= auction.reservePrice, "Bid amount is too low");
        // 转账逻辑
        (bool sent, ) = auction.owner.call{value: amount}(new bytes(0));
        require(sent, "Failed to send funds");
        auctionList[auctionIndex].owner = msg.sender;
    }
}
