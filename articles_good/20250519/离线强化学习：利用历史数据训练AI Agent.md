  class OfflineRLAgent:
      def __init__(self, state_space, action_space):
          self.q_network = QNetwork(state_space, action_space)
          self.target_network = QNetwork(state_space, action_space)
          self.memory = ReplayBuffer(buffer_size=10000)
          self.batch_size = 32

      def train(self, dataset):
          for episode in dataset:
              for step in episode:
                  state, action, reward, next_state, done = step
                  self.memory.store(state, action, reward, next_state, done)

                  if self.memory.size() >= self.batch_size:
                      minibatch = self.memory.sample(self.batch_size)
                      states, actions, rewards, next_states, dones = minibatch
                      q_values = self.q_network.predict(states)
                      next_q_values = self.target_network.predict(next_states)
                      targets = rewards + (1 - dones) * gamma * np.max(next_q_values, axis=1)
                      self.q_network.fit(states, targets)

                      # 更新目标网络
                      self.target_network.soft_update(self.q_network)
