### 3.3 情感生成算法的代码实现
```python
import torch
import torch.nn as nn
import torch.optim as optim

class EmotionGenerator(nn.Module):
    def __init__(self, input_dim, output_dim):
        super(EmotionGenerator, self).__init__()
        self.embedding = nn.Embedding(input_dim, 50)
        self.lstm = nn.LSTM(50, 50, batch_first=True)
        self.fc = nn.Linear(50, output_dim)

    def forward(self, x):
        embed = self.embedding(x)
        output, _ = self.lstm(embed)
        output = self.fc(output[:, -1, :])
        return output

# 初始化模型和优化器
model = EmotionGenerator(input_dim, output_dim)
optimizer = optim.Adam(model.parameters(), lr=0.001)
criterion = nn.CrossEntropyLoss()
```
