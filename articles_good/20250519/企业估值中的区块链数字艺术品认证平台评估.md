contract Artwork {

        uint256 creationTime;
        string description;
        bytes32 hash;
    }

    mapping(uint256 => Artwork) public artworks;

    function uploadArtwork(uint256 id, string memory artist, uint256 creationTime, string memory description, bytes32 hash) public {

            hash: hash
        });
        artworks[id] = artwork;
    }

    function verifyArtwork(uint256 id) public view returns (bool) {
        Artwork memory artwork = artworks[id];
        return artwork.owner == msg.sender;
    }
}
