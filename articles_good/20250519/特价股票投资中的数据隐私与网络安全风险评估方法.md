###### 3.1.2 数据脱敏算法

###### 3.2.3 认证授权算法

def aes_encrypt(data, key):
    cipher = AES.new(key, AES.MODE_ECB)

def data_anonymization(data):
    anonymized = []
    for field in data:
        anonymized.append(field.permute())
    return anonymized

def rbac_authorization(user, action, resource):
    if user.role.permits(action, resource):
        return True
    else:
        return False
