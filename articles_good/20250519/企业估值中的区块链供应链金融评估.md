import numpy as np

def calculate_enterprise_value(cash_flows, discount_rate, periods):
    enterprise_value = 0
    for t in range(periods):
        if t >= len(cash_flows):
            cash_flow = 0
        else:
            cash_flow = cash_flows[t]
        enterprise_value += cash_flow / (1 + discount_rate) ** t
    return enterprise_value
