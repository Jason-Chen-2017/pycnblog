```python
import numpy as np
import matplotlib.pyplot as plt

# 模拟参数
num_particles = 1000
G = 6.6743e-11  # 万有引力常数
time_step = 1e-3  # 秒

# 初始位置和速度
positions = np.random.rand(num_particles, 3) * 1e22  # m

# 模拟宇宙演化
def simulate_universe(positions, velocities, G, time_step, num_steps):
    for _ in range(num_steps):
        # 计算引力
        forces = compute_gravity(positions, G)
        # 更新速度和位置
        velocities += forces / num_particles * time_step
        positions += velocities * time_step
        
    return positions

# 计算引力
def compute_gravity(positions, G):
    num_particles = len(positions)
    forces = np.zeros((num_particles, 3))
    for i in range(num_particles):
        for j in range(i+1, num_particles):
            delta_position = positions[i] - positions[j]
            distance = np.linalg.norm(delta_position)
            force = G * num_particles * num_particles / distance**2
            direction = delta_position / distance
            forces[i] -= force * direction
            forces[j] += force * direction
    return forces

# 运行模拟
positions = simulate_universe(positions, velocities, G, time_step, 1000)

# 可视化结果
plt.scatter(positions[:, 0], positions[:, 1])
plt.xlabel('Position X (m)')
plt.ylabel('Position Y (m)')
plt.title('Cosmic Evolution Simulation')
plt.show()
```
