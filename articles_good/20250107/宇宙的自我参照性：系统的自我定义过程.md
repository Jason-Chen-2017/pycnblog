```python
import numpy as np
import matplotlib.pyplot as plt

class SelfReferentialAlgorithm:
    def __init__(self, initial_state, reference_value, learning_rate):
        """
        初始化自我参照性算法的参数。
        :param initial_state: 初始状态
        :param reference_value: 参照值
        :param learning_rate: 学习率
        """
        self.state = initial_state  # 初始状态
        self.reference_value = reference_value  # 参照值
        self.learning_rate = learning_rate  # 学习率

    def update_state(self):
        """
        更新状态，使其趋近于参照值。
        :return: 更新后的状态
        """
        error = self.reference_value - self.state  # 计算误差
        self.state -= self.learning_rate * error  # 根据误差调整状态
        return self.state

    def run(self, iterations):
        """
        运行迭代过程。
        :param iterations: 迭代次数
        """
        states = [self.state]  # 保存状态历史记录
        for _ in range(iterations):
            self.state = self.update_state()  # 更新状态
            states.append(self.state)  # 记录当前状态

        return states

# 参数设置
initial_state = 50  # 初始状态
reference_value = 60  # 参照值
learning_rate = 0.1  # 学习率
iterations = 100  # 迭代次数

# 创建自我参照性算法实例并运行
algorithm = SelfReferentialAlgorithm(initial_state, reference_value, learning_rate)
states = algorithm.run(iterations)

# 绘制状态变化曲线
plt.plot(states)
plt.xlabel('Iterations')
plt.ylabel('State')
plt.title('State Change over Time')
plt.show()
```

3. **运行迭代**：在`run`方法中，进行多次迭代，每次迭代后更新状态并记录状态历史。
