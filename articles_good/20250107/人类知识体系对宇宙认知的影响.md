### 人类知识体系概述

from sklearn.cluster import KMeans
import numpy as np

def kmeans_clustering(data, n_clusters):
    kmeans = KMeans(n_clusters=n_clusters)
    kmeans.fit(data)
    return kmeans.labels_

# 示例数据
data = np.random.rand(100, 2)

# 进行K-均值聚类

# 绘制结果
