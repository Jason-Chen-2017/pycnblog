```python
import numpy as np

def calculate_gravity(m1, m2, r):
    """
    计算两个物体之间的引力。
    """
    G = 6.674 * 10**-11  # 引力常数
    force = G * m1 * m2 / r**2
    return force

def simulate_universe(masses, distances):
    """
    模拟宇宙中的引力相互作用。
    """
    num_objects = len(masses)
    positions = np.zeros((num_objects, 3))
    velocities = np.zeros((num_objects, 3))
    
    for i in range(num_objects):
        for j in range(i + 1, num_objects):
            force = calculate_gravity(masses[i], masses[j], distances[i, j])
            direction = (positions[j] - positions[i]) / distances[i, j]
            velocities[i] += force * direction
            velocities[j] -= force * direction
    
    return positions, velocities

# 初始参数
masses = np.array([1.989 * 10**30, 9.547 * 10**27, 1.99 * 10**30])  # 地球、月球、太阳的质量
distances = np.array([[0, 384400000], [384400000, 0], [0, 149597870700]])  # 地球-月球、月球-太阳、地球-太阳的距离

# 模拟宇宙中的引力相互作用
positions, velocities = simulate_universe(masses, distances)

# 输出结果
print("物体位置：", positions)
print("物体速度：", velocities)
```
