### 5.2 系统核心实现源代码

以下是一个简单的 Python 脚本，用于处理简约主义婚礼摄影的后期处理。这个脚本将使用 OpenCV 和 Matplotlib 库来处理图像，并使用 LaTeX 格式打印出数学公式。

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt
from IPython.display import LaTeX

def simple_post_processing(image_path, output_path):
    # 读取图像
    image = cv2.imread(image_path)

    # 转换为灰度图像
    gray_image = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)

    # 使用中值滤波去除噪声
    filtered_image = cv2.medianBlur(gray_image, 5)

    # 使用二值化处理图像
    _, binary_image = cv2.threshold(filtered_image, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)

    # 调整亮度和对比度
    alpha = 1.5
    beta = 50
    adjusted_image = cv2.convertScaleAbs(filtered_image, alpha=alpha, beta=beta)

    # 保存处理后的图像
    cv2.imwrite(output_path, adjusted_image)

    # 显示图像
    plt.imshow(adjusted_image, cmap='gray')
    plt.title(r'$\text{Processed Image}$')
    plt.xlabel(r'$\text{Brightness: } {:.2f}$'.format(alpha))
    plt.ylabel(r'$\text{Contrast: } {:.2f}$'.format(beta))
    plt.show()

# 测试代码
simple_post_processing('input.jpg', 'output.jpg')
```

#### 5.2 系统核心实现源代码

以下是一个简单的 Python 脚本，用于处理简约主义婚礼摄影的后期处理。这个脚本将使用 OpenCV 和 Matplotlib 库来处理图像，并使用 LaTeX 格式打印出数学公式。

```python
import cv2
import numpy as np
import matplotlib.pyplot as plt
from IPython.display import LaTeX

def simple_post_processing(image_path, output_path):
    # 读取图像
    image = cv2.imread(image_path)

    # 转换为灰度图像
    gray_image = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)

    # 使用中值滤波去除噪声
    filtered_image = cv2.medianBlur(gray_image, 5)

    # 使用二值化处理图像
    _, binary_image = cv2.threshold(filtered_image, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)

    # 调整亮度和对比度
    alpha = 1.5
    beta = 50
    adjusted_image = cv2.convertScaleAbs(filtered_image, alpha=alpha, beta=beta)

    # 保存处理后的图像
    cv2.imwrite(output_path, adjusted_image)

    # 显示图像
    plt.imshow(adjusted_image, cmap='gray')
    plt.title(r'$\text{Processed Image}$')
    plt.xlabel(r'$\text{Brightness: } {:.2f}$'.format(alpha))
    plt.ylabel(r'$\text{Contrast: } {:.2f}$'.format(beta))
    plt.show()

# 测试代码
simple_post_processing('input.jpg', 'output.jpg')
```
