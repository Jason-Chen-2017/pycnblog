def generate_hash(input_data):
    """生成数字指纹（哈希值）"""

def sign_data(data, private_key):

```solidity
pragma solidity ^0.8.0;

contract DigitalArtVerifier {
    struct Artwork {
        string hash;
        address owner;
        uint timestamp;
    }

    mapping(uint => Artwork) public artworks;

    event ArtworkRegistered(uint id, string hash, address owner, uint timestamp);
    event ArtworkSold(uint id, address buyer, uint price, uint timestamp);

    function registerArtwork(string memory hash) public {
        artworks[msg.sender].hash = hash;
        artworks[msg.sender].owner = msg.sender;
        artworks[msg.sender].timestamp = block.timestamp;
        emit ArtworkRegistered(msg.sender, hash, msg.sender, block.timestamp);
    }

    function sellArtwork(uint id, uint price) public {
        require(artworks[id].owner == msg.sender, "Not the owner");
        artworks[id].price = price;
        emit ArtworkSold(id, msg.sender, price, block.timestamp);
    }

    function buyArtwork(uint id) public payable {
        require(artworks[id].price > 0, "Artwork not for sale");
        require(msg.value == artworks[id].price, "Insufficient payment");
        address owner = artworks[id].owner;
        artworks[id].owner = msg.sender;
        artworks[id].price = 0;
        payable(owner).transfer(msg.value);
    }
}
```

    contract_instance.sellArtwork(id, price, transact={'from': w3.eth.defaultAccount})

def buy():
    id = int(request.form['id'])
    contract_instance.buyArtwork(id, transact={'value': w3.toWei('1', 'ether'), 'from': w3.eth.defaultAccount})
    return redirect(url_for('index'))
