class UserController:
    @staticmethod
    def register(username, password):
        # 注册用户
        hashed_password = bcrypt.hashpw(password.encode('utf-8'), bcrypt.gensalt())
        cursor.execute("INSERT INTO users (username, password) VALUES (%s, %s)", (username, hashed_password))
        connection.commit()
        return "注册成功"

    @staticmethod
    def login(username, password):
        # 用户登录
        cursor.execute("SELECT * FROM users WHERE username = %s", (username,))

class VehicleController:
    @staticmethod
    def rent_vehicle(vehicle_id):
        # 租赁车辆
        cursor.execute("SELECT * FROM vehicles WHERE id = %s AND status = 'available'", (vehicle_id,))
        vehicle = cursor.fetchone()
        if vehicle:
            cursor.execute("UPDATE vehicles SET status = 'rented' WHERE id = %s", (vehicle_id,))
            connection.commit()
            return "租赁成功"
        else:
            return "车辆不可用"

    @staticmethod
    def return_vehicle(vehicle_id):
        # 归还车辆
        cursor.execute("SELECT * FROM vehicles WHERE id = %s AND status = 'rented'", (vehicle_id,))
        vehicle = cursor.fetchone()
        if vehicle:
            cursor.execute("UPDATE vehicles SET status = 'available' WHERE id = %s", (vehicle_id,))
            connection.commit()
            return "归还成功"
        else:
            return "车辆不存在"

    @staticmethod
    def charge_vehicle(vehicle_id, charging_station_id):
        # 给车辆充电
        cursor.execute("SELECT * FROM charging_stations WHERE id = %s AND status = 'available'", (charging_station_id,))
        charging_station = cursor.fetchone()
        if charging_station:
            cursor.execute("UPDATE charging_stations SET status = 'charging' WHERE id = %s", (charging_station_id,))
            cursor.execute("UPDATE vehicles SET status = 'charging' WHERE id = %s", (vehicle_id,))
            connection.commit()
            return "充电成功"
        else:
            return "充电桩不可用"

### 6.1.1 绿色能源原理

绿色能源是指可再生能源，如太阳能、风能、水能等。这些能源具有清洁、环保、可再生等特点，能够减少对传统化石燃料的依赖，降低温室气体排放，有助于应对气候变化和能源危机。在新型城市共享电动车充电网络中，绿色能源的原理体现在以下几个方面：

from flask import Flask, request, jsonify

app = Flask(__name__)

db = SQLAlchemy(app)

# 用户模型
class User(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    username = db.Column(db.String(80), unique=True, nullable=False)
    password = db.Column(db.String(120), nullable=False)

# 用户注册
@app.route('/register', methods=['POST'])
def register():
    username = request.json['username']
    password = request.json['password']
    hashed_password = bcrypt.hashpw(password.encode('utf-8'), bcrypt.gensalt())
    new_user = User(username=username, password=hashed_password)
    db.session.add(new_user)
    db.session.commit()
    return jsonify({'message': 'User registered successfully.'})

# 用户登录
@app.route('/login', methods=['POST'])
def login():
    username = request.json['username']
    password = request.json['password']
    user = User.query.filter_by(username=username).first()
    if user and bcrypt.checkpw(password.encode('utf-8'), user.password.encode('utf-8')):
        return jsonify({'message': 'Login successful.'})
    else:
        return jsonify({'message': 'Invalid credentials.'})

if __name__ == '__main__':
    db.create_all()
    app.run(debug=True)
