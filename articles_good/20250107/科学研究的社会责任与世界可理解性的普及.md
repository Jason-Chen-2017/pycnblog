```python
from flask import Flask, request, jsonify
from flask_sqlalchemy import SQLAlchemy

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'mysql+pymysql://username:password@localhost/db_name'
db = SQLAlchemy(app)

class Responsibility(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    description = db.Column(db.String(255))

@app.route('/api/responsibilities', methods=['POST'])
def add_responsibility():
    data = request.json
    new_responsibility = Responsibility(description=data['description'])
    db.session.add(new_responsibility)
    db.session.commit()
    return jsonify({'id': new_responsibility.id})

@app.route('/api/responsibilities', methods=['GET'])
def get_responsibilities():
    responsibilities = Responsibility.query.all()
    return jsonify({'responsibilities': [r.description for r in responsibilities]})

if __name__ == '__main__':
    db.create_all()
    app.run(debug=True)
```
