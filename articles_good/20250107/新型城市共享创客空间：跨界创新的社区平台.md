class User:
    def __init__(self, username, password):
        self.username = username
        self.password = password
        self.status = "未登录"

    def login(self, username, password):
        if username == self.username and password == self.password:
            self.status = "已登录"
            return "登录成功"
        else:
            return "登录失败"

    def verify_permission(self):
        if self.status == "已登录":
            return True
        else:
            return False

class Member(User):
    def __init__(self, username, password, role):

        self.role = role

    def join_space(self, space_name):

            print(f"{self.username} 加入 {space_name} 共享创客空间")
        else:
            print("无权限加入共享创客空间")

class Space:
