```python
import requests
import time

def collect_data():
    # 收集数据
    url = "http://ai-service.example.com/monitor"
    response = requests.get(url)
    data = response.json()
    return data

def process_data(data):
    # 数据处理
    cpu_usage = data["cpu_usage"]
    memory_usage = data["memory_usage"]
    if cpu_usage > 90 or memory_usage > 90:
        trigger_alert()

def trigger_alert():
    # 发送告警
    print("Alert: CPU/Memory usage is too high!")

if __name__ == "__main__":
    while True:
        data = collect_data()
        process_data(data)
        time.sleep(60)
```

```python
def clean_data(data):
    # 数据清洗
    data["cpu_usage"] = max(0, min(100, data["cpu_usage"]))
    data["memory_usage"] = max(0, min(100, data["memory_usage"]))
    return data

def transform_data(data):
    # 数据转换
    transformed_data = {
        "service_id": data["service_id"],
        "timestamp": time.time(),
        "cpu_usage": data["cpu_usage"],
        "memory_usage": data["memory_usage"]
    }
    return transformed_data

def store_data(data):
    # 数据存储
    print("Storing data:", data)

if __name__ == "__main__":
    raw_data = {"service_id": "1", "cpu_usage": 95, "memory_usage": 90}
    cleaned_data = clean_data(raw_data)
    transformed_data = transform_data(cleaned_data)
    store_data(transformed_data)
```
