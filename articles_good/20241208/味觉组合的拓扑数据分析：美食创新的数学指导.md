from sklearn.cluster import KMeans

# 假设我们收集了以下味觉数据
tast_data = np.array([[1, 2], [3, 4], [5, 6], [7, 8]])

tast_data_preprocessed = tast_data / np.linalg.norm(tast_data, axis=1)[:, np.newaxis]

labels = kmeans.predict(tast_data_preprocessed)
