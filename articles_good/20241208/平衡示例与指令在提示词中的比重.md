# 示例和指令权重计算
def calculate_weights(examples, instructions):
    # 计算示例和指令的权重
    example_weight = len(examples) / (len(examples) + len(instructions))
    instruction_weight = len(instructions) / (len(examples) + len(instructions))
    return example_weight, instruction_weight

# 示例和指令比重计算
def calculate_ratio(examples, instructions):
    example_weight, instruction_weight = calculate_weights(examples, instructions)
    return example_weight / (example_weight + instruction_weight), instruction_weight / (example_weight + instruction_weight)
