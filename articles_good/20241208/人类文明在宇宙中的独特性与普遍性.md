```python
import numpy as np
import matplotlib.pyplot as plt

def chemical_evolution(t, base_pools, time_constant):
    """
    模拟生命起源过程中的化学进化。
    
    参数：
    t : float，当前时间
    base_pools : list of float，基础池的初始浓度

# 计算化学进化的过程
times = np.linspace(0, time_constant, 100)
for t in times:
    base_pools = chemical_evolution(t, base_pools, time_constant)

# 绘制化学进化过程
plt.plot(times, base_pools)
plt.xlabel('Time (time_constant units)')
plt.ylabel('Base Pool Concentrations')
plt.title('Chemical Evolution: Origin of Life')
plt.show()
```

**示例3：多宇宙理论的模拟**

```python
import numpy as np
import matplotlib.pyplot as plt
import random

def multi_universe_simulation(num_universes, survival_rate, time_step):
    """
    模拟多个宇宙中文明的出现和消亡。
    
    参数：
    num_universes : int，宇宙的数量
    survival_rate : float，文明的生存率
    time_step : int，时间步长
    """
    populations = [random.uniform(0, 100) for _ in range(num_universes)]
    
    for _ in range(time_step):
        new_populations = []
        for population in populations:
            if random.random() < survival_rate:
                new_population = population * 1.1  # 文明增长
            else:
                new_population = population * 0.9  # 文明消亡
            new_populations.append(new_population)
        
        populations = new_populations
    
    return populations

# 设置参数
num_universes = 10
survival_rate = 0.5
time_step = 50

# 进行模拟
populations = multi_universe_simulation(num_universes, survival_rate, time_step)

# 绘制结果
plt.plot([i for i in range(num_universes)], populations)
plt.xlabel('Universe')
plt.ylabel('Civilization Population')
plt.title('Multi-Verse Simulation: Civilization Survival')
plt.show()
```
