- **Gamma校正**  
  Gamma校正是通过幂函数对图像进行调整，公式如下：

  $$ I_{out} = I_{in}^{\gamma} $$

import numpy as np
import matplotlib.pyplot as plt

def rgb_to_cie_xyz(rgb):
    x = rgb[:, 0] / 255
    y = rgb[:, 1] / 255
    z = rgb[:, 2] / 255
    X = 0.4324 * x + 0.3423 * y + 0.1323 * z
    Y = 0.2698 * x + 0.6712 * y + 0.0582 * z
    Z = 0.0793 * x + 0.1281 * y + 0.7826 * z
    return np.stack([X, Y, Z], axis=-1)

rgb = np.array([[255, 0, 0], [0, 255, 0], [0, 0, 255]])
cie_xyz = rgb_to_cie_xyz(rgb)
print(cie_xyz)

import numpy as np

def gamma_correction(image, gamma=2.2):
    image = image / 255.0
    corrected = np.power(image, gamma)
    corrected = corrected * 255.0
    return corrected.astype(np.uint8)

image = np.array([[128, 192, 255], [255, 128, 64]])
