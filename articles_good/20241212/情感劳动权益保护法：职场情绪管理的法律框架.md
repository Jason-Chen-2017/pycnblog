def calculate_mental_health(employee_info):
    # 读取员工信息
    social_support = employee_info['social_support']
    work_pressure = employee_info['work_pressure']
    personal_mental_quality = employee_info['personal_mental_quality']

    # 计算心理健康状态
    mental_health = (social_support + personal_mental_quality - work_pressure) / 3

    # 输出结果
    return mental_health

# 示例
employee_info = {
    'social_support': 0.8,  # 社会支持（0-1分）
    'work_pressure': 0.6,    # 工作压力（0-1分）
    'personal_mental_quality': 0.7  # 个人心理素质（0-1分）
}

mental_health = calculate_mental_health(employee_info)
print(f'心理健康状态：{mental_health}')
