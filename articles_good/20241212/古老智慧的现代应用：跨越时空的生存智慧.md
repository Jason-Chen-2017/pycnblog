   ```python
   from flask_sqlalchemy import SQLAlchemy
   
   db = SQLAlchemy()

   class Experience(db.Model):
       id = db.Column(db.Integer, primary_key=True)
       title = db.Column(db.String(120), nullable=False)
       content = db.Column(db.Text, nullable=False)
   
   class Philosophy(db.Model):
       id = db.Column(db.Integer, primary_key=True)
       title = db.Column(db.String(120), nullable=False)
       content = db.Column(db.Text, nullable=False)
   
   class Technology(db.Model):
       id = db.Column(db.Integer, primary_key=True)
       title = db.Column(db.String(120), nullable=False)
       content = db.Column(db.Text, nullable=False)
   
   class Leadership(db.Model):
       id = db.Column(db.Integer, primary_key=True)
       title = db.Column(db.String(120), nullable=False)
       content = db.Column(db.Text, nullable=False)
   
   class Social(db.Model):
       id = db.Column(db.Integer, primary_key=True)
       title = db.Column(db.String(120), nullable=False)
       content = db.Column(db.Text, nullable=False)
   ```

2. **API接口**：

   ```python
   from flask import Flask, jsonify, request
   
   app = Flask(__name__)
   app.config.from_object('config')
   db.init_app(app)
   
   @app.route('/experience', methods=['GET'])
   def get_experience():
       experiences = Experience.query.all()
       return jsonify([{'id': exp.id, 'title': exp.title, 'content': exp.content} for exp in experiences])
   
   @app.route('/philosophy', methods=['GET'])
   def get_philosophy():
       philosophies = Philosophy.query.all()
       return jsonify([{'id': phil.id, 'title': phil.title, 'content': phil.content} for phil in philosophies])
   ```
