class User:
    def __init__(self, name):
        self.name = name

class Kitchen:
    def __init__(self, id, status="空闲"):
        self.id = id
        self.status = status

class Reservation:
    def __init__(self, user, kitchen, time):
        self.user = user
        self.kitchen = kitchen
        self.time = time
        self.status = "待确认"

    def confirm(self):

    def cancel(self):
        self.kitchen.status = "空闲"
        self.status = "已取消"

def reserve_kitchen(user, kitchen, time):
    reservation = Reservation(user, kitchen, time)
    if kitchen.status == "空闲":
        kitchen.status = "占用"
        reservation.confirm()
        print(f"{user.name}成功预约厨房{kitchen.id}，时间为{time}")
    else:
        print(f"厨房{kitchen.id}已被占用，无法预约")

def main():
    user = User("张三")
    kitchen1 = Kitchen(1)
    kitchen2 = Kitchen(2, "维修中")

    reserve_kitchen(user, kitchen1, "2023-11-05 18:00")
    reserve_kitchen(user, kitchen2, "2023-11-05 18:00")

if __name__ == "__main__":
    main()
