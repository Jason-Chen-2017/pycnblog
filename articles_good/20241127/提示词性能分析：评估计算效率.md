**A.1 提示词生成算法**

- **基于TF-IDF的提示词生成算法**

  ```python
  def generate_prompt_tf_idf(document, corpus, num_words=10):
      """
      生成基于TF-IDF的提示词。
      
      :param document: 待分析的单个文档。
      :param corpus: 整个文档集合。
      :param num_words: 提示词的数量。
      :return: 提示词列表。
      """
      # 计算每个单词的TF-IDF值
      tf_idf_scores = compute_tf_idf(document, corpus)
      
      # 选取最高的num_words个单词作为提示词
      prompt_words = [word for word, score in tf_idf_scores.items() if score > threshold]
      
      return prompt_words[:num_words]

  def compute_tf_idf(document, corpus):
      """
      计算单个文档中每个单词的TF-IDF值。
      
      :param document: 待分析的文档。
      :param corpus: 整个文档集合。
      :return: 单词及其TF-IDF值的字典。
      """
      tf = compute_term_frequency(document)
      idf = compute_inverse_document_frequency(document, corpus)
      tf_idf = {word: tf[word] * idf[word] for word in tf}
      
      return tf_idf

  def compute_term_frequency(document):
      """
      计算文档中每个单词的频率。
      
      :param document: 文档。
      :return: 单词及其频率的字典。
      """
      words = document.lower().split()
      word_count = Counter(words)
      total_words = len(words)
      tf = {word: count / total_words for word, count in word_count.items()}
      
      return tf

  def compute_inverse_document_frequency(document, corpus):
      """
      计算文档集合中每个单词的逆文档频率。
      
      :param document: 文档。
      :param corpus: 文档集合。
      :return: 单词及其逆文档频率的字典。
      """
      doc_frequency = Counter([word for doc in corpus for word in doc])
      total_documents = len(corpus)
      idf = {word: np.log(total_documents / (1 + doc_frequency[word])) for word in doc_frequency}
      
      return idf
  ```

- **基于Word2Vec的提示词生成算法**

  ```python
  from gensim.models import Word2Vec

  def generate_prompt_word2vec(document, model, num_words=10):
      """
      生成基于Word2Vec的提示词。
      
      :param document: 待分析的单个文档。
      :param model: Word2Vec模型。
      :param num_words: 提示词的数量。
      :return: 提示词列表。
      """
      # 将文档中的每个单词映射到Word2Vec向量
      word_vectors = [model[word] for word in document]
      
      # 计算所有单词向量的平均向量
      avg_vector = np.mean(word_vectors, axis=0)
      
      # 选取与平均向量最相似的num_words个单词作为提示词
      top_words = model.wv.most_similar(positive=[avg_vector], topn=num_words)
      
      return [word for word, _ in top_words]

  def train_word2vec_model(corpus, size=100, window=5, min_count=1, workers=4):
      """
      训练Word2Vec模型。
      
      :param corpus: 文档集合。
      :param size: 向量大小。
      :param window: 窗口大小。
      :param min_count: 最小计数。
      :param workers: 工作线程数。
      :return: Word2Vec模型。
      """
      model = Word2Vec(corpus, size=size, window=window, min_count=min_count, workers=workers)
      model.save('word2vec.model')
      return model
  ```
