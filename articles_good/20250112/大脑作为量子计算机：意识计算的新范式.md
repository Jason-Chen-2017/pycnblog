# 诊断建议生成

    if patient["symptoms"].count("fever") > 0:
        diagnosis_suggestions.append(f"{patient['name']} may have a fever-related illness.")
    if patient["symptoms"].count("headache") > 0:
        diagnosis_suggestions.append(f"{patient['name']} may have a headache-related illness.")

# 医生决策
for suggestion in diagnosis_suggestions:
    print(f"Doctor decision: {suggestion}")
