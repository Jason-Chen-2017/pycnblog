    status = db.Column(db.String(80), nullable=False)

def checkout():
    data = request.get_json()
    user_id = data.get('user_id')
    shopping_cart_ids = data.get('shopping_cart_ids')

    if not user_id or not shopping_cart_ids:
        return jsonify({'error': 'Missing required data.'}), 400

    shopping_carts = ShoppingCart.query.filter(ShoppingCart.id.in_(shopping_cart_ids), ShoppingCart.user_id == user_id).all()
    if len(shopping_carts) != len(shopping_cart_ids):

    new_order = Order(user_id=user_id, status='pending')
    db.session.add(new_order)
    db.session.commit()

    for shopping_cart in shopping_carts:
        shopping_cart.order_id = new_order.id
        db.session.delete(shopping_cart)

    db.session.commit()

    return jsonify({'message': 'Checkout completed successfully.', 'total_price': total_price}), 201

    coupon_code = data.get('coupon_code')

    total_price -= discount_amount

```python
from flask import Flask, request, jsonify

app = Flask(__name__)

# 中间件函数
def require_authorization():
    def wrapper(f):
        @wraps(f)
        def decorated_function(*args, **kwargs):
            token = request.headers.get('Authorization')
            if not token:
                return jsonify({"error": "Authorization token is required"}), 401
            # 在这里进行token验证逻辑
            return f(*args, **kwargs)
        return decorated_function
    return wrapper

# 使用中间件保护路由
@app.route('/protected', methods=['GET'])
@require_authorization()
def protected():
    return jsonify({"message": "You have accessed the protected resource"}), 200

if __name__ == '__main__':
    app.run(debug=True)
```
