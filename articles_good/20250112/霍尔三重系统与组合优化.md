# 初始化粒子群
def initialize_particles(n_particles, search_space):
    particles = np.random.uniform(low=search_space[0], high=search_space[1], size=(n_particles, dim))
    velocities = np.random.uniform(low=-1, high=1, size=(n_particles, dim))
    return particles, velocities

# 更新粒子速度和位置
def update_particles(particles, velocities, best_individual, global_best):
    for i in range(n_particles):
        r1, r2 = np.random.rand(2)
        cognitive_component = r1 * w * (best_individual - particles[i])
        social_component = r2 * (global_best - particles[i])
        velocities[i] = velocities[i] + cognitive_component + social_component
        particles[i] = particles[i] + velocities[i]
        particles[i] = np.clip(particles[i], search_space[0], search_space[1])
    return particles

# 粒子群优化算法
def particle_swarm_optimization(particles, velocities, best_individual, global_best, max_iterations, dim):
    for _ in range(max_iterations):
        best_individual = np.min(particles, axis=0)
        global_best = np.min(particles, axis=1)
        particles = update_particles(particles, velocities, best_individual, global_best)
    return global_best

# 参数设置
n_particles = 50
dim = 10
w = 0.5
max_iterations = 100

# 初始化粒子群
search_space = [[0, 10]] * dim
particles, velocities = initialize_particles(n_particles, search_space)

# 运行粒子群优化算法
global_best = particle_swarm_optimization(particles, velocities, best_individual, global_best, max_iterations, dim)

# 输出最优解

**案例背景：**某物流公司需要在春节期间安排一辆货车从上海运送一批货物到北京。货车容量为50立方米，运输时间为3天。物流公司希望找到最优的运输路线和装载方案，以最大化运输效率和降低成本。

**需求分析：**  
1. **优化目标**：在满足货车容量和运输时间的前提下，最大化运输效率（例如，最小化运输距离或最小化运输成本）。  
2. **约束条件**：货车容量不超过50立方米，运输时间为3天，路线需合法合规。  
3. **数据输入**：起点和终点坐标、各条路线的距离、运输成本、货物种类和体积。

**模型构建：**  
根据上述需求，我们可以构建一个基于霍尔三重系统的组合优化模型：

1. **定义变量**：设 \(x_1, x_2, \ldots, x_n\) 为从上海到北京的各条路线的权重（包括距离和成本）。

2. **设定目标函数**：最小化总运输成本，即 \(\min \sum_{i=1}^{n} w_i x_i\)，其中 \(w_i\) 为路线 \(i\) 的权重。

3. **确定约束条件**：  
   - 货车容量约束：\(\sum_{i=1}^{n} v_i x_i \leq 50\)，其中 \(v_i\) 为路线 \(i\) 上货物的体积。  
   - 运输时间约束：\(\sum_{i=1}^{n} t_i x_i = 3\)，其中 \(t_i\) 为路线 \(i\) 的运输时间。
