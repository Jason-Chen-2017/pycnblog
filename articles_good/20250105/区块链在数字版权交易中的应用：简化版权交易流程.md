contract CopyrightAuthorization {
    address public owner;
    mapping(string => bool) public authorizations;

    constructor() {
        owner = msg.sender;
    }

    function authorize(string memory workId) public {
        require(msg.sender == owner, "Only the owner can authorize");
        authorizations[workId] = true;

    }
}
