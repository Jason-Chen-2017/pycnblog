   - **RSA 同态加密**：
     $$ C = RSA_{pk}(M^e \mod n) $$

     其中，$M$为解密结果，$d$为解密指数。

   - **全同态加密**：
     $$ C = FFHE_{sk}(M \oplus R) $$
     其中，$C$为加密结果，$M$为明文，$R$为随机数，$sk$为密钥。

     $$ M = FFHE_{pk}(C \oplus R) $$
     其中，$M$为解密结果，$pk$为公钥。

### 6.2.2 权限管理

```solidity
contract PermissionManagement {
    struct Role {
        string name;
        mapping(string => bool) permissions;
    }

    mapping(string => Role) private roles;
    mapping(string => string[]) private userRoles;

    function createRole(string memory _name, string[] memory _permissions) public {
        require(roles[_name].name == "", "Role already exists");

        roles[_name] = Role(_name, _permissions);
    }

    function assignRoleToUser(string memory _username, string memory _roleName) public {
        require(roles[_roleName].name != "", "Role does not exist");
        require(!hasRole(_username, _roleName), "User already has this role");

        userRoles[_username].push(_roleName);
    }

    function hasRole(string memory _username, string memory _roleName) public view returns (bool) {
        for (uint i = 0; i < userRoles[_username].length; i++) {
            if (keccak256(abi.encodePacked(userRoles[_username][i])) == keccak256(abi.encodePacked(_roleName))) {
                return true;
            }
        }
        return false;
    }
}
```

contract DataManagement {
    struct Data {
        string key;
        string value;
    }

    mapping(string => mapping(string => Data[])) private userData;

    function storeData(string memory _username, string memory _key, string memory _value) public {
        require(hasRole(_username, "Admin"), "User does not have permission to store data");

        userData[_username][_key].push(Data(_key, _value));
    }

    function retrieveData(string memory _username, string memory _key) public view returns (string memory) {
        require(hasRole(_username, "User"), "User does not have permission to retrieve data");

        for (uint i = 0; i < userData[_username][_key].length; i++) {
            if (keccak256(abi.encodePacked(userData[_username][_key][i].key)) == keccak256(abi.encodePacked(_key))) {
                return userData[_username][_key][i].value;
            }
        }
        return "";
    }
}
