from transformers import AutoTokenizer, AutoModelForCausalLM

# 加载预训练的AI模型
tokenizer = AutoTokenizer.from_pretrained('t5-base')
model = AutoModelForCausalLM.from_pretrained('t5-base')

# 文本预处理
def preprocess_text(text):
    # 清洗文本
    text = text.lower()
    text = re.sub('[^a-zA-Z]', ' ', text)
    text = text.strip()
    # 分词
    tokens = tokenizer.tokenize(text)
    # 添加特殊 tokens
    tokens = ['']
    return tokens

# 提示词生成
def generate_prompt(tokens):

    prompt = ' '.join(tokens)
    return prompt

# 模型响应
def model_response(prompt):
    # 使用训练好的AI模型生成响应文本
    inputs = tokenizer(prompt, return_tensors='pt')

    return tokenizer.decode(response[:, inputs.attention_mask[:, -1] == 1])

# 系统交互
def interact():
    while True:
        # 接收用户输入

        tokens = preprocess_text(user_input)
        # 生成提示词
        prompt = generate_prompt(tokens)
        # 生成响应文本
        response = model_response(prompt)
        # 输出响应文本
        print('系统生成内容：', response)
        # 结束交互

interact()
