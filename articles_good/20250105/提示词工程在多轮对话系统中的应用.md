# 用户输入

    return [word for word, tag in tags if tag.startswith('NN')]

# 提示词生成
def generate_prompt(context):
    vectorizer = CountVectorizer()
    X = vectorizer.fit_transform([' '.join(context)])

    return classifier.predict(X)[0]

def use_prompt(prompt, context):
    return prompt + " " + " ".join(context)

    if intent == "booking":
        response = "你想预订明天的机票吗？"

#### 5.2.1 数据预处理

```python
import nltk
from nltk.tokenize import word_tokenize
from nltk.corpus import stopwords

# 加载停用词表
stop_words = set(stopwords.words('english'))

# 数据预处理函数
def preprocess_text(text):
    # 分词
    tokens = word_tokenize(text)
    # 去除停用词
    filtered_tokens = [token for token in tokens if token not in stop_words]
    # 小写化
    return ' '.join(filtered_tokens).lower()
```

#### 5.2.2 意图识别

```python
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.naive_bayes import MultinomialNB
from sklearn.pipeline import make_pipeline

# 意图识别模型
def train_intent_recognizer(data):
    vectorizer = TfidfVectorizer()
    classifier = MultinomialNB()
    model = make_pipeline(vectorizer, classifier)
    model.fit(data['text'], data['intent'])
    return model
```

#### 5.2.3 上下文提取

```python
# 上下文提取函数
def extract_context(data, window_size=3):
    context = []
    for index, row in data.iterrows():
        tokens = word_tokenize(row['text'])
        start = max(0, index - window_size)
        end = min(len(tokens), index + window_size)
        context.append(' '.join(tokens[start:end]))
    return context
```

#### 5.2.4 提示词生成

```python
from gensim.models import Word2Vec

# 提示词生成函数
def generate_prompt(context, model):
    context_vector = sum([model.wv[word] for word in context if word in model.wv]) / len(context)
    nearest_words = model.wv.most_similar(positive=[context_vector], topn=5)
    return ' '.join([word for word, _ in nearest_words])
```

#### 5.2.5 系统核心流程

```python
import numpy as np

# 系统核心流程
def chatbot(input_text, model_intent, context_model, word2vec_model):
    # 数据预处理
    processed_text = preprocess_text(input_text)
    # 意图识别
    intent = model_intent.predict([processed_text])[0]
    # 上下文提取
    context = extract_context(context_model, input_text, window_size=3)
    # 提示词生成
    prompt = generate_prompt(context, word2vec_model)
    # 回答生成
    response = f"您的问题是：'{input_text}'。我理解您是想询问关于'{intent}'的相关信息。{prompt}"
    return response
```
