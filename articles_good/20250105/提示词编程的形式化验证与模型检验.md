```python
def process_token(token):
    if token == '+':
        return add
    elif token == '-':
        return subtract
    elif token == '=':
        return assign
    else:
        return None

def add(a, b):
    return a + b

def subtract(a, b):
    return a - b

def assign(value):
    return value

def main():
    tokens = ["+", "2", "3", "=", "4"]
    result = 0
    for token in tokens:
        op = process_token(token)
        if op:
            if op == add:
                result = add(result, next_token())
            elif op == subtract:
                result = subtract(result, next_token())
            elif op == assign:
                result = assign(next_token())
    print("Result:", result)

if __name__ == "__main__":
    main()
```
