```python
import jieba
import pandas as pd

def preprocess_text(text):
    # 分词
    seg_list = jieba.cut(text)
    # 去除停用词
    stop_words = set(['的', '了', '在', '是', '和'])
    filtered_words = [word for word in seg_list if word not in stop_words]
    # 词性标注
    pos_tags = jieba.posseg.cut(filtered_words)
    # 提取名词
    noun_words = [word for word, flag in pos_tags if 'N' in flag]
    return noun_words
```

#### 提示词生成模块

```python
from sklearn.feature_extraction.text import TfidfVectorizer

def generate_keywords(texts, top_n=5):
    # 创建TF-IDF向量器
    vectorizer = TfidfVectorizer()
    # 训练向量器
    vectorizer.fit(texts)
    # 转换文本为向量
    text_vectors = vectorizer.transform(texts)
    # 计算每个词的TF-IDF值
    tfidf_matrix = text_vectors.toarray()
    # 计算词频和逆文档频率
    word_freq = vectorizer.get_feature_names_out()
    word_idf = vectorizer.idf_
    # 计算每个词的TF-IDF值
    word_tfidf = tfidf_matrix.mean(axis=0)
    # 排序并选取前top_n个关键词
    top_keywords = sorted(zip(word_tfidf, word_freq), reverse=True)[:top_n]
    return top_keywords
```
