```python
class TypeChecker:
    def __init__(self):
        self.type_table = {}

    def type_check(self, expression):
        if isinstance(expression, int):
            return 'int'
        elif isinstance(expression, float):
            return 'float'
        elif isinstance(expression, str):
            return 'str'
        elif isinstance(expression, list):
            return 'list'
        else:
            return 'unknown'

    def infer_type(self, expression):
        if isinstance(expression, (int, float, str)):
            return expression.__class__.__name__
        elif isinstance(expression, list):
            element_type = self.infer_type(expression[0])
            return f'list of {element_type}s'
        else:
            return 'unknown'
```

```python
class TypeChecker:
    def __init__(self):
        self.type_table = {}

    def type_check(self, expression):
        if isinstance(expression, int):
            return 'int'
        elif isinstance(expression, float):
            return 'float'
        elif isinstance(expression, str):
            return 'str'
        elif isinstance(expression, list):
            return 'list'
        else:
            return 'unknown'

    def infer_type(self, expression):
        if isinstance(expression, (int, float, str)):
            return expression.__class__.__name__
        elif isinstance(expression, list):
            element_type = self.infer_type(expression[0])
            return f'list of {element_type}s'
        else:
            return 'unknown'

# 测试代码

print(tc.type_check([1, 2, 3]))  # 输出：list
```
