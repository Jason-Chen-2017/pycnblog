import numpy as np

def initialize_state_transition_matrix(n):
    # 初始化状态转移矩阵
    P = np.zeros((n, n))
    for i in range(n):
        P[i] = np.random.rand(n)
        P[i] /= np.sum(P[i])
    return P

def update_state_transition_matrix(P, k):
    # 更新状态转移矩阵
    for _ in range(k):
        Pt = np.linalg.matrix_power(P, _)
        P = P * Pt
    return P

def main():
    # 主函数
    n = 2  # 状态数量
    k = 3  # 迭代次数
    P0 = initialize_state_transition_matrix(n)
    Pk = update_state_transition_matrix(P0, k)
    print("初始状态转移矩阵：")
    print(P0)
    print("最终状态转移矩阵：")
    print(Pk)

if __name__ == "__main__":
    main()
```
