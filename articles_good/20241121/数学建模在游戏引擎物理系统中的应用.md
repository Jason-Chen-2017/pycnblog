using UnityEngine;

public class FluidSimulation : MonoBehaviour
{
    public Material fluidMaterial;
    public Texture2D fluidTexture;

    private void Update()
    {
        // 调用 MATLAB 流体动力学计算脚本
        string script = "fluid_simulation.m";
        MATLABEngine matlab = new MATLABEngine();
        matlab.ExecuteScript(script);

        // 获取计算结果
        float[,] fluidData = matlab.ExecuteFunction<float[,]>("get_fluid_data");

        // 创建纹理
        Texture2D newTexture = new Texture2D(fluidData.GetLength(0), fluidData.GetLength(1));

        // 设置纹理颜色
        for (int i = 0; i < fluidData.GetLength(0); i++)
        {
            for (int j = 0; j < fluidData.GetLength(1); j++)
            {
                Color color = new Color(fluidData[i, j], fluidData[i, j], fluidData[i, j]);
                newTexture.SetPixel(i, j, color);
            }
        }

        // 应用纹理
        newTexture.Apply();
        fluidMaterial.mainTexture = newTexture;

using UnityEngine;

public class SoftBodySimulation : MonoBehaviour
{
    public Material softBodyMaterial;
    public MeshFilter meshFilter;

    private void Update()
    {
        // 调用 MATLAB 物理计算脚本
        string script = "soft_body_simulation.m";
        MATLABEngine matlab = new MATLABEngine();
        matlab.ExecuteScript(script);

        // 获取计算结果
        float[,] vertices = matlab.ExecuteFunction<float[,]>("get_vertices");

        // 创建网格
        Mesh newMesh = new Mesh();
        newMesh.vertices = new Vector3[vertices.GetLength(0) * vertices.GetLength(1)];
        for (int i = 0; i < vertices.GetLength(0); i++)
        {
            for (int j = 0; j < vertices.GetLength(1); j++)
            {
                newMesh.vertices[i * vertices.GetLength(1) + j] = new Vector3(vertices[i, j], vertices[i, j], vertices[i, j]);
            }
        }

        // 设置网格三角形
        int[] tris = new int[newMesh.vertices.Length * 6];
        int index = 0;
        for (int i = 0; i < newMesh.vertices.Length - 1; i += 2)
        {
            tris[index++] = i;
            tris[index++] = i + 1;
            tris[index++] = i + 2;
            tris[index++] = i;
            tris[index++] = i + 2;
            tris[index++] = i + 3;
        }
        newMesh.triangles = tris;

        // 应用网格
        meshFilter.mesh = newMesh;
    }
}
