  ```arduino
  #include <HTTPClient.h>

  const char* ssid = "your_SSID";
  const char* password = "your_PASSWORD";

  void setup() {
    Serial.begin(9600);
    WiFi.begin(ssid, password);
    while (WiFi.status() != WL_CONNECTED) {
      delay(500);
      Serial.print(".");
    }
    Serial.println("");
    Serial.print("WiFi connected: ");
    Serial.println(WiFi.localIP());
  }

  void loop() {
    if (WiFi.status() == WL_CONNECTED) {
      float temperature = readTemperature();
      float humidity = readHumidity();
      sendToServer(temperature, humidity);
    }
    delay(60000); // Wait for 1 minute
  }

  float readTemperature() {
    // 读取温度传感器数据
    return 25.0;
  }

  float readHumidity() {
    // 读取湿度传感器数据
    return 50.0;
  }

  void sendToServer(float temperature, float humidity) {
    HTTPClient http;
    String url = "http://your_server.com/data";
    String payload = "temperature=" + String(temperature) + "&humidity=" + String(humidity);
    http.begin(url, payload);
    int httpResponseCode = http.POST(payload);
    if (httpResponseCode > 0) {
      String response = http.getString();
      Serial.println(response);
    } else {
      Serial.print("Error on HTTP request: ");
      Serial.println(httpResponseCode);
    }
    http.end();
  }
  ```

### 第3章：物联网通信协议

#### 3.1 常见通信协议介绍

**物联网通信协议：**

- **Z-Wave：** 一种短距离无线通信协议，适用于家庭自动化领域。
- **Zigbee：** 一种低功耗无线通信协议，适用于智能家居、工业自动化等领域。
- **WiFi：** 一种无线局域网通信协议，适用于高速数据传输。
- **LoRa：** 一种长距离低功耗无线通信协议，适用于远程监控、物联网应用。
- **NB-IoT：** 一种窄带物联网通信协议，适用于物联网应用场景。
