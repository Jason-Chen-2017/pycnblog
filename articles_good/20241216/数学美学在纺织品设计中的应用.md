import numpy as np

def draw_symmetric_pattern(n, axis=None, center=None):
    """绘制对称图案"""
    if axis:
        # 轴对称
        height = n
        width = n
        matrix = np.zeros((height, width), dtype=int)
        for i in range(height):
            for j in range(width):
                matrix[i, j] = 1 if (i + j) % 2 == 0 else 0
    elif center:
        # 中心对称
        height = n
        width = n
        matrix = np.zeros((height, width), dtype=int)
        for i in range(height):
            for j in range(width):
                matrix[i, j] = 1 if (i - j) % 2 == 0 else 0
    else:
        raise ValueError("需要提供对称轴或对称中心")

    # 打印图案
    for row in matrix:
        print(' '.join(['█' if cell else ' ' for cell in row]))

# 轴对称
draw_symmetric_pattern(10, axis=True)

# 中心对称
draw_symmetric_pattern(10, center=True)
