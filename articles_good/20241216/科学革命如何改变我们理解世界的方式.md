#### 3.1.2 算法B：Python源代码与算法原理

```python
# 算法B：基于基因算法的优化问题求解
import numpy as np

def gene_algorithmObjectiveFunction(x):
    # 目标函数
    return x**2

def gene_algorithm(x_min, x_max, population_size, generations, crossover_rate, mutation_rate):
    # 初始化种群
    population = np.random.uniform(x_min, x_max, population_size)
    # 适应度评估
    fitness = gene_algorithmObjectiveFunction(population)
    # 迭代过程
    for _ in range(generations):
        # 交叉
        parents = np.random.choice(population, size=2, replace=False)
        child = (parents[0] + parents[1]) / 2
        # 变异
        if np.random.rand() < mutation_rate:
            child += np.random.normal(0, 1)
        # 更新种群
        population = np.concatenate((population[:len(population)//2], [child]))
        # 适应度评估
        fitness = gene_algorithmObjectiveFunction(population)
    # 返回最优解
    best_fitness = np.min(fitness)
    best_solution = population[fitness.argmin()]
    return best_solution, best_fitness

# 实例化
best_solution, best_fitness = gene_algorithm(-10, 10, 100, 1000, 0.8, 0.05)
```
