3. **调查法：** 学生通过问卷调查、访谈等方式收集信息。

def assess_scientific_aptitude(knowledge, skills, attitude):
    """
    评估个体的科学素养水平。
    
    参数：
    knowledge：知识分数（0-100分）
    skills：技能分数（0-100分）
    attitude：态度分数（0-100分）
    
    返回值：
    科学素养总分
    """
    total_score = (knowledge + skills + attitude) / 3
