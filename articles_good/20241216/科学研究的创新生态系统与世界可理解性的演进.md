# 伪代码：科学研究创新生态系统核心实现

class ResearchInstitution:
    def __init__(self):
        self.knowledge_base = KnowledgeBase()
        self.enterprise = Enterprise()
        self.government = Government()
        self.investor = Investor()
        self.social_public = SocialPublic()

    def search_knowledge(self, keyword):
        return self.knowledge_base.search(keyword)

    def share_knowledge(self, knowledge):
        self.knowledge_base.insert(knowledge)
        self.enterprise.share_knowledge(knowledge)
        self.social_public.share_knowledge(knowledge)

    def request_funding(self, project):
        self.investor.request_funding(project)

    def report_progress(self, progress):
        self.government.report_progress(progress)
        self.social_public.report_progress(progress)

class KnowledgeBase:
    def __init__(self):
        self.knowledge = []

    def search(self, keyword):
        return [k for k in self.knowledge if keyword in k]

    def insert(self, knowledge):
        self.knowledge.append(knowledge)

class Enterprise:
    def __init__(self):
        self.technology_base = TechnologyBase()

    def share_knowledge(self, knowledge):
        self.technology_base.insert(knowledge)

class Government:
    def __init__(self):
        self.policy_base = PolicyBase()

class SocialPublic:

        self.info_base = InfoBase()

    def share_knowledge(self, knowledge):
        self.info_base.insert(knowledge)

    def report_progress(self, progress):
        self.info_base.insert(progress)

class TechnologyBase:
    def __init__(self):

class PolicyBase:

class FundingBase:

class InfoBase:
    def __init__(self):
        self.info = []

    def insert(self, info):
