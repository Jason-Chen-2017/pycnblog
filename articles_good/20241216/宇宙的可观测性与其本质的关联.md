```python
import numpy as np

def preprocess_data(spectra, luminosity):
    """
    对光谱和亮度数据进行预处理，包括去噪和标准化。
    """
    # 去除异常值
    cleaned_spectra = np.where(np.isfinite(spectra), spectra, np.nan)
    cleaned_luminosity = np.where(np.isfinite(luminosity), luminosity, np.nan)
    
    # 标准化数据
    mean_spectra = np.mean(cleaned_spectra)
    std_spectra = np.std(cleaned_spectra)
    mean_luminosity = np.mean(cleaned_luminosity)
    std_luminosity = np.std(cleaned_luminosity)
    
    normalized_spectra = (cleaned_spectra - mean_spectra) / std_spectra
    normalized_luminosity = (cleaned_luminosity - mean_luminosity) / std_luminosity
    
    return normalized_spectra, normalized_luminosity
```

```python
def preprocess_data(spectra, luminosity):
    # 去除异常值
    cleaned_spectra = np.where(np.isfinite(spectra), spectra, np.nan)
    cleaned_luminosity = np.where(np.isfinite(luminosity), luminosity, np.nan)
    
    # 标准化数据
    mean_spectra = np.mean(cleaned_spectra)
    std_spectra = np.std(cleaned_spectra)
    mean_luminosity = np.mean(cleaned_luminosity)
    std_luminosity = np.std(cleaned_luminosity)
    
    normalized_spectra = (cleaned_spectra - mean_spectra) / std_spectra
    normalized_luminosity = (cleaned_luminosity - mean_luminosity) / std_luminosity
    
    return normalized_spectra, normalized_luminosity
```

- **去除异常值**：`np.where` 函数用于检查光谱和亮度数据中的异常值。如果数据值是有限的（非NaN或无穷大），则保留该值；否则，将其替换为NaN。
- **标准化数据**：通过计算均值和标准差，我们将原始数据减去均值并除以标准差，实现数据的标准化。这一步骤有助于消除不同数据量级的影响，使得数据更适合后续分析。

```python
def calculate_expansion_rate(spectra, luminosity):

    mean_spectra = np.mean(spectra)
```
