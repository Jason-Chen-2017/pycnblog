```python
import cv2
def apply_makeup(image, makeup_type):
    # 转换为HSV颜色空间
    hsv = cv2.cvtColor(image, cv2.COLOR_BGR2HSV)
    # 根据化妆类型调整色调和饱和度
    if makeup_type == 'lipstick':
        # 唇膏效果
        hsv[:, :, 0] = (hsv[:, :, 0] * 0.5).astype(np.uint8)
        hsv[:, :, 1] = (hsv[:, :, 1] * 0.5).astype(np.uint8)
    elif makeup_type == 'eyeshadow':
        # 眼影效果
        hsv[:, :, 0] = (hsv[:, :, 0] * 0.6).astype(np.uint8)
        hsv[:, :, 1] = (hsv[:, :, 1] * 0.6).astype(np.uint8)
    # 转换回BGR颜色空间
    result = cv2.cvtColor(hsv, cv2.COLOR_HSV2BGR)
    return result
```
