### 第2章 区块链技术基础

#### 2.1 核心概念

区块链技术是一种分布式数据库技术，其核心概念包括区块、链、分布式账本、共识算法等。

- **区块**：区块是区块链的基本组成单元，每个区块包含一定数量的交易记录。区块通过加密算法和时间戳等手段，确保其数据的完整性和不可篡改性。
- **链**：区块链是由多个区块按照一定规则链接而成的数据结构。区块链的链式结构确保了数据的有序性和可追溯性。
- **分布式账本**：分布式账本是指区块链上的所有参与者都拥有一份完整的账本记录，从而实现了去中心化的数据管理。
- **共识算法**：共识算法是区块链网络中达成共识的机制。常见的共识算法包括工作量证明（PoW）、权益证明（PoS）等，它们保证了区块链网络的安全性和数据的正确性。

#### 2.2 区块链结构

区块链结构通常包括以下几个部分：

1. **区块**：每个区块包含一个或多个交易记录，以及一个由前一个区块的哈希值生成的随机数。区块通过哈希函数与上一个区块链接，形成链式结构。
2. **链**：区块链是由多个区块按照时间顺序链接而成的。每个区块都包含了前一个区块的哈希值，从而确保了区块链的完整性。
3. **分布式账本**：区块链上的每个参与者都拥有一个完整的账本记录，确保了数据的去中心化存储。
4. **网络**：区块链网络是由多个节点组成的。每个节点都参与区块的生成、验证和传播，确保了区块链的运行。

#### 2.3 区块链的工作原理

区块链的工作原理可以概括为以下步骤：

1. **交易记录生成**：交易发起者将交易记录发送到区块链网络中。
2. **交易验证**：网络中的节点对交易记录进行验证，确保交易的有效性。
3. **区块生成**：当一定数量的交易记录被验证通过后，节点将它们打包成一个区块。
4. **区块链接**：新生成的区块通过哈希函数与上一个区块链接，形成区块链。
5. **共识达成**：区块链网络中的节点通过共识算法达成共识，确保区块链的完整性。
6. **区块传播**：新生成的区块被传播到区块链网络中的所有节点。

// Blockchain node implementation in Go
package main

import (
    "crypto/sha256"
    "encoding/hex"
    "time"
)

type Block struct {
    Index     int
    Timestamp string
    Data      string
    Hash      string
    PrevHash  string
}

func calculateHash(block *Block) string {
    blockString := structToString(block)
    hash := sha256.Sum256([]byte(blockString))
    return hex.EncodeToString(hash[:])
}

func structToString(block *Block) string {
    return fmt.Sprintf("%d\t%s\t%s\t%s", block.Index, block.Timestamp, block.Data, block.PrevHash)
}

func generateBlock(lastBlock *Block, data string) *Block {
    index := lastBlock.Index + 1
    timestamp := time.Now().Format(time.RFC3339)
    prevHash := lastBlock.Hash
    hash := calculateHash(&Block{index, timestamp, data, "", prevHash})
    return &Block{index, timestamp, data, hash, prevHash}
}

func main() {
    genesisBlock := &Block{0, "2023-04-01T12:00:00Z", "Genesis Block", "", ""}
    blockchain := []*Block{genesisBlock}

    for {
        block := generateBlock(lastBlock(blockchain), "New Block")
        blockchain = append(blockchain, block)
        fmt.Printf("Added block %d to the blockchain\n", block.Index)
    }
}

pragma solidity ^0.8.0;

contract DigitalArt {
    struct Artwork {
        id          uint256;
        creator     address;
        certificationStatus bool;
        owner       address;
    }

    mapping(uint256 => Artwork) public artworks;
    uint256 public artworkCount;

    event ArtworkRegistered(uint256 id, address creator);
    event ArtworkCertified(uint256 id, bool certificationStatus);
    event ArtworkTransferred(uint256 id, address from, address to);

    function registerArtwork(uint256 id, string memory artist) public {
        artworks[id] = Artwork(id, msg.sender, false, msg.sender);
        artworkCount++;
        emit ArtworkRegistered(id, msg.sender);
    }

    function certifyArtwork(uint256 id, bool status) public {
        require(artworks[id].creator == msg.sender, "Only the creator can certify the artwork.");
        artworks[id].certificationStatus = status;
        emit ArtworkCertified(id, status);
    }

    function transferArtwork(uint256 id, address to) public {
        require(artworks[id].owner == msg.sender, "Only the owner can transfer the artwork.");
        artworks[id].owner = to;
        emit ArtworkTransferred(id, msg.sender, to);
    }
}
