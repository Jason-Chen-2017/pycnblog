#### 代码9.1 领导力在项目管理的代码实现

以下是使用Python编写的项目进度跟踪系统代码示例：

```python
class Project:
    def __init__(self, name, tasks):
        self.name = name
        self.tasks = tasks
    
    def add_task(self, task):
        self.tasks.append(task)
    
    def display_tasks(self):
        for task in self.tasks:
            print(f"任务名称：{task.name}")
            print(f"开始时间：{task.start}")
            print(f"持续时间：{task.duration}")
            print()

class Task:
    def __init__(self, name, start, duration):
        self.name = name
        self.start = start
        self.duration = duration

# 创建项目实例
project = Project("项目A", [])

# 添加任务
project.add_task(Task("任务1", 0, 5))
project.add_task(Task("任务2", 5, 3))
project.add_task(Task("任务3", 8, 2))

# 显示任务列表
project.display_tasks()
```

class Team:
    def __init__(self, name, members):
        self.name = name
        self.members = members
    
    def add_member(self, member):
        self.members.append(member)
    
    def display_members(self):
        for member in self.members:
            print(f"成员姓名：{member.name}")
            print(f"角色：{member.role}")
            print(f"参与会议：{member.attended_meetings}")
            print()

class Member:
    def __init__(self, name, role, attended_meetings):
        self.name = name
        self.role = role
        self.attended_meetings = attended_meetings

# 创建团队实例
team = Team("团队A", [])

# 添加成员
team.add_member(Member("张三", "开发者", 5))
team.add_member(Member("李四", "设计师", 4))
