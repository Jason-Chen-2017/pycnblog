```python
import tensorflow as tf
from tensorflow.keras.models import Model
from tensorflow.keras.layers import Input, LSTM, Dense

# 定义序列到序列模型
def build_seq2seq_model(input_vocab_size, target_vocab_size, embedding_dim, units):
    input_seq = Input(shape=(None, input_vocab_size))
    encoder = LSTM(units, return_state=True)
    encoder_output, state_h, state_c = encoder(input_seq)

    decoder = LSTM(units, return_sequences=True, return_state=True)
    decoder_output, _, _ = decoder(encoder_output, initial_state=[state_h, state_c])

    decoder_dense = Dense(target_vocab_size, activation='softmax')
    decoder_output = decoder_dense(decoder_output)

    model = Model([input_seq], [decoder_output])
    return model

# 编译模型
model = build_seq2seq_model(input_vocab_size, target_vocab_size, embedding_dim, units)
model.compile(optimizer='rmsprop', loss='categorical_crossentropy')

# 训练模型
model.fit([input_seq], [target_seq], batch_size=64, epochs=10)
```
