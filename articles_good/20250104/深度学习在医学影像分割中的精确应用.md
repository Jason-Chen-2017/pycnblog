import torch.optim as optim

from torchvision import datasets, transforms

# 定义U-Net网络架构
class UNet(nn.Module):
    def __init__(self):
        super(UNet, self).__init__()
        # 下采样路径
        self.conv1 = nn.Sequential(nn.Conv2d(1, 64, 3, padding=1),
                                  nn.ReLU(inplace=True),
                                  nn.Conv2d(64, 64, 3, padding=1),
                                  nn.ReLU(inplace=True),
                                  nn.MaxPool2d(2, 2))
        self.conv2 = nn.Sequential(nn.Conv2d(64, 128, 3, padding=1),
                                  nn.ReLU(inplace=True),
                                  nn.Conv2d(128, 128, 3, padding=1),
                                  nn.ReLU(inplace=True),
                                  nn.MaxPool2d(2, 2))
        self.conv3 = nn.Sequential(nn.Conv2d(128, 256, 3, padding=1),
                                  nn.ReLU(inplace=True),
                                  nn.Conv2d(256, 256, 3, padding=1),
                                  nn.ReLU(inplace=True),
                                  nn.MaxPool2d(2, 2))
        self.conv4 = nn.Sequential(nn.Conv2d(256, 512, 3, padding=1),
                                  nn.ReLU(inplace=True),
                                  nn.Conv2d(512, 512, 3, padding=1),
                                  nn.ReLU(inplace=True),
                                  nn.MaxPool2d(2, 2))
        # 上采样路径
        self.up4 = nn.Sequential(nn.ConvTranspose2d(512, 256, 2, 2),
                                 nn.ReLU(inplace=True))
        self.conv4d = nn.Sequential(nn.Conv2d(512, 256, 3, padding=1),
                                    nn.ReLU(inplace=True),
                                    nn.Conv2d(256, 256, 3, padding=1),
                                    nn.ReLU(inplace=True))
        self.up3 = nn.Sequential(nn.ConvTranspose2d(256, 128, 2, 2),
                                 nn.ReLU(inplace=True))
        self.conv3d = nn.Sequential(nn.Conv2d(256, 128, 3, padding=1),
                                    nn.ReLU(inplace=True),
                                    nn.Conv2d(128, 128, 3, padding=1),
                                    nn.ReLU(inplace=True))
        self.up2 = nn.Sequential(nn.ConvTranspose2d(128, 64, 2, 2),
                                 nn.ReLU(inplace=True))
        self.conv2d = nn.Sequential(nn.Conv2d(128, 64, 3, padding=1),
                                    nn.ReLU(inplace=True),
                                    nn.Conv2d(64, 64, 3, padding=1),
                                    nn.ReLU(inplace=True))
        self.up1 = nn.Sequential(nn.ConvTranspose2d(64, 1, 2, 2),
                                 nn.Sigmoid())

    def forward(self, x):
        c1 = self.conv1(x)
        c2 = self.conv2(c1)
        c3 = self.conv3(c2)
        c4 = self.conv4(c3)
        d4 = self.up4(c4)
        d4 = torch.cat((c3, d4), 1)
        d4 = self.conv4d(d4)
        d3 = self.up3(d4)
        d3 = torch.cat((c2, d3), 1)
        d3 = self.conv3d(d3)
        d2 = self.up2(d3)
        d2 = torch.cat((c1, d2), 1)
        d2 = self.conv2d(d2)
        outs = self.up1(d2)
        return outs

# 训练模型

for epoch in range(num_epochs):

        images = images.float()

        labels = torch.reshape(labels, (labels.size(0), 1, 28, 28))
        images = torch.reshape(images, (images.size(0), 1, 28, 28))

        optimizer.zero_grad()

        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()

        if (i+1) % 100 == 0:

# 测试模型

    correct = 0
    total = 0

        labels = labels.float()
        labels = torch.reshape(labels, (labels.size(0), 1, 28, 28))

        predicted = outputs.round()

#### 1. 网络架构解析

上述代码定义了一个基于U-Net架构的医学影像分割模型。U-Net是一种深度卷积神经网络（CNN）架构，专门用于医学影像分割任务。其主要特点在于其对称的收缩（下采样）和扩张（上采样）路径，这使得模型能够在保持高分辨率的同时提取深层次的特征。

##### （1）收缩路径

收缩路径包括四个卷积层和两个最大池化层。每个卷积层后跟一个ReLU激活函数。具体实现如下：

#### 2. 数学模型详解

##### （1）卷积层

卷积层的主要操作是卷积运算，其数学公式如下：

$$
\text{output}(i, j) = \sum_{k} \text{filter}(k, i, j) \cdot \text{input}(i - k, j - k)
$$

其中，`filter` 表示卷积核，`input` 表示输入图像，`output` 表示输出特征图。卷积层通过滑动卷积核在输入图像上进行操作，以提取特征。

##### （2）ReLU激活函数

ReLU（Rectified Linear Unit）激活函数是一种常用的非线性激活函数，其数学公式如下：

$$
\text{ReLU}(x) = \max(0, x)
$$

ReLU函数将输入值设置为非负值，从而引入非线性特性，有助于加速模型训练。

##### （3）最大池化层

最大池化层用于降低图像分辨率，其数学公式如下：

$$
\text{output}(i, j) = \max_{k, l} \text{input}(i - k, j - l)
$$

最大池化层选择每个邻域内的最大值作为输出，以保留图像的主要特征。

##### （4）转置卷积层

转置卷积层是一种特殊类型的卷积层，其目的是将高分辨率特征图恢复到原始分辨率。其数学公式与卷积层类似，但输入和输出的维度进行了调整：

$$
\text{output}(i, j) = \sum_{k} \text{filter}(k, i, j) \cdot \text{input}(i + k, j + l)
$$

通过转置卷积层，U-Net能够将上采样的特征图与下采样的特征图进行融合，从而提高分割的精度。
