### 3. Python源代码演示

以下是一个简单的Python代码示例，用于演示提示词优化算法的基本原理：

```python
import jieba
from nltk.corpus import stopwords

def optimize_prompt(text):
    # 分词
    words = jieba.cut(text)
    # 去停用词
    stop_words = set(stopwords.words('english'))
    filtered_words = [word for word in words if word not in stop_words]
    # 关键词筛选
    keywords = jieba.analyse.extract_tags(filtered_words, topK=5)
    # 语义增强和上下文扩充
    enhanced_keywords = [word + " is important" for word in keywords]
    # 生成提示词
    prompt = " ".join(enhanced_keywords)
    return prompt
```

def optimize_prompt(text):
    # 分词
    words = jieba.cut(text)
    # 去停用词
    filtered_words = [word for word in words if word not in stop_words]

    keywords = jieba.analyse.extract_tags(filtered_words, topK=5)

    # 生成提示词
    prompt = " ".join(enhanced_keywords)
    return prompt
