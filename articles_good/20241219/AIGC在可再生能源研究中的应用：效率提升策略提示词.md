```python
import tensorflow as tf
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Dense, LSTM
import numpy as np

# 数据预处理
def preprocess_data(data):
    # 对数据进行归一化处理
    normalized_data = (data - np.mean(data)) / np.std(data)
    return normalized_data

# 模型训练
def train_model(data, labels):
    model = Sequential([
        LSTM(50, activation='relu', input_shape=(data.shape[1], 1)),
        Dense(1)
    ])

    model.compile(optimizer='adam', loss='mse')
    model.fit(data, labels, epochs=100, batch_size=32)
    return model

# 实时分析
def real_time_analysis(model, data):
    processed_data = preprocess_data(data)
    prediction = model.predict(processed_data)
    return prediction

# 故障预警
def alarm WARNING_THRESHOLD):
    if prediction < WARNING_THRESHOLD:
        print("预警：设备可能存在故障！")
    else:
        print("正常：设备运行正常。")

# 主函数
def main():
    # 读取数据
    data = np.loadtxt("data.txt")
    labels = np.loadtxt("labels.txt")

    # 训练模型
    model = train_model(data, labels)

    # 实时分析
    real_time_data = np.random.rand(100, 1)
    prediction = real_time_analysis(model, real_time_data)

    # 故障预警
    alarm(WARNING_THRESHOLD=0.5)

if __name__ == "__main__":
    main()
```
