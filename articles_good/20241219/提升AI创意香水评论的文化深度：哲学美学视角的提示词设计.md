from nltk.tokenize import word_tokenize

# 加载哲学美学提示词

# 情感分析
sentiment = sia.polarity_scores(description)

# 选择评论模板
if sentiment['compound'] > 0.5:
    template = "这款香水散发着{positive}的香气，让人感到{positive}，非常适合{occasion}。"
else:
    template = "这款香水散发着{negative}的香气，让人感到{negative}，适合{occasion}。"

# 填充评论模板
if random.random() > 0.5:
    comment = template.format(positive=random.choice(positive_words), negative=random.choice(negative_words), occasion="春季")
else:
    comment = template.format(positive=random.choice(positive_words), negative=random.choice(negative_words), occasion="秋季")

# 输出评论
print(comment)
