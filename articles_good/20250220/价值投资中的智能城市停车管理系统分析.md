    # 图像预处理
    blurred = cv2.GaussianBlur(gray, (3, 3), 0)
    # 边缘检测
    edges = cv2.Canny(blurred, 50, 150)
    # 查找车牌区域

    max_area = 0
    license_plate = None
    for contour in contours:
        area = cv2.contourArea(contour)

            max_area = area
            license_plate = contour
