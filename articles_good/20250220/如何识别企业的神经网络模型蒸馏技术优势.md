##### 3.1 神经网络模型蒸馏技术的数学模型  
- 3.1.1 知识蒸馏的损失函数  
  $$L = \alpha L_{cls} + (1-\alpha)L_{dist}$$  
  其中，$L_{cls}$ 是分类损失，$L_{dist}$ 是蒸馏损失，$\alpha$ 是平衡系数。

import torch
import torch.nn as nn
import torch.optim as optim

class TeacherModel(nn.Module):
    def __init__(self):

        x = x.view(-1, 784)

        for inputs, labels in dataloaders:
            teachers.eval()
            with torch.no_grad():
                teacher_outputs = teachers(inputs)
            students.train()
            student_outputs = students(inputs)
            
            # 计算蒸馏损失
            distillation_loss = criterion(student_outputs, teacher_outputs)
