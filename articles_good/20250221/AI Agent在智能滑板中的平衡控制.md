- **3.3.1 PID控制的数学推导**

### 3.4 算法实现的代码示例
以下是PID控制算法的Python代码示例：

```python
class PIDController:
    def __init__(self, Kp, Ki, Kd):
        self.Kp = Kp
        self.Ki = Ki
        self.Kd = Kd
        self.error_integral = 0
        self.last_error = 0

    def compute_output(self, current_error, dt):
        self.error_integral += current_error * dt
        derivative_error = (current_error - self.last_error) / dt
        output = self.Kp * current_error + self.Ki * self.error_integral + self.Kd * derivative_error
        self.last_error = current_error
        return output
```

import time

class MPU6050:
    def __init__(self, address=0x68):

import time

class SkateboardBalancer:
    def __init__(self):
        self.mpu = MPU6050()
        self.controller = PIDController(2, 0.5, 0.1)
        self.last_time = time.time()
    
    def balance(self):
        while True:
            current_time = time.time()
            dt = current_time - self.last_time
            self.last_time = current_time

            x, y, z = self.mpu.get_acceleration()
            error = y  # 假设y轴是倾斜角度

            voltage = self.controller.compute_output(error, dt)
            # 控制马达输出电压
            print(f"Voltage: {voltage}")
            time.sleep(0.01)

if __name__ == "__main__":
    balancer = SkateboardBalancer()
    balancer.balance()
