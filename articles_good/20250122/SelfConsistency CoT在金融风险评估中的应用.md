```python
def consistency_detection(model, X, y, batch_size=32):
    n_batches = len(X) // batch_size + 1
    errors = []

    for i in range(n_batches):
        start = i * batch_size
        end = (i + 1) * batch_size
        predictions = model.predict(X[start:end])
        errors.append(np.mean(np.abs(predictions - y[start:end])))

    return consistency_score
