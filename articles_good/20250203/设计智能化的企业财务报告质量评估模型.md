\begin{aligned}
S &= \sum_{i=1}^{n} \frac{1}{|D_i|} \sum_{j \in D_i} \frac{1}{|R_j|} \sum_{k \in R_j} \frac{1}{|A_{jk}|} \\
R_j &= \{r \in R | r \in S_j\} \\
A_{jk} &= \{a \in A | a \in S_j \land a \in R_j\}
\end{aligned}
$$
