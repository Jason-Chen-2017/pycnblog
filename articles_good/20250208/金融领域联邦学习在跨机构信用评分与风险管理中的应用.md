## 7.2 核心代码实现

### 7.2.1 联邦平均算法实现
```python
import numpy as np
from fedlearning import FedAvg

class CreditScoreModel:
    def __init__(self, input_dim):
        self.theta = np.zeros(input_dim)

    def forward(self, X):
        return np.dot(X, self.theta)

    def backward(self, X, y):
        return 2 * np.dot(X.T, (y - np.dot(X, self.theta)))

def main():
    # 初始化模型
    model = CreditScoreModel(input_dim=10)
    # 初始化FedAvg
    fed_avg = FedAvg(parties=5, model=model)
    # 开始训练
    fed_avg.train(epochs=10)
    # 输出结果
    print(fed_avg.get_global_model())

if __name__ == "__main__":
    main()
```
