### 3.2.3 遗传算法的Python代码实现
```python
import random

def fitness(individual):
    # 计算适应度函数
    return sum(w * x for w, x in zip(weights, individual))

def select(individuals, fitness_values):
    # 选择操作
    total = sum(fitness_values)
    probabilities = [fv / total for fv in fitness_values]
    selected = random.choices(individuals, weights=probabilities, k=1)
    return selected[0]

def crossover(parent1, parent2):
    # 单点交叉
    point = random.randint(1, len(parent1)-1)
    child1 = parent1[:point] + parent2[point:]
    child2 = parent2[:point] + parent1[point:]
    return child1, child2

def mutate(individual):
    # 变异操作
    point = random.randint(0, len(individual)-1)
    individual[point] = 1 - individual[point]
    return individual

# 初始化种群
population = [[0, 1, 0, 1, 1] for _ in range(10)]
weights = [0.5, 0.3, 0.2, 0.1, 0.8]

# 迭代优化
for _ in range(100):
    fitness_values = [fitness(individual) for individual in population]
    selected = select(population, fitness_values)
    parent1 = selected
    parent2 = select(population, fitness_values)
    child1, child2 = crossover(parent1, parent2)
    mutated1 = mutate(child1)
    mutated2 = mutate(child2)
    population = [mutated1, mutated2] + [individual for individual in population if individual not in (parent1, parent2)]
```
