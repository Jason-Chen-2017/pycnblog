```python
import torch
import torch.nn as nn

class FocusAnalyzer(nn.Module):
    def __init__(self):
        super(FocusAnalyzer, self).__init__()
        self.lstm = nn.LSTM(input_size=10, hidden_size=128, num_layers=2)
        self.fc = nn.Linear(128, 1)
    
    def forward(self, x):
        out, _ = self.lstm(x)
        out = self.fc(out)
        return out
```
