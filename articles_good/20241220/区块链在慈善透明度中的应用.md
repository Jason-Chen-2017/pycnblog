pragma solidity ^0.8.0;

contract Charity {
    struct Donation {
        address donor;
        uint256 amount;
        uint256 timestamp;
    }

    mapping(uint256 => Donation[]) public donations;
    mapping(uint256 => bool) public projectCompleted;

    event DonationReceived(uint256 projectId, address donor, uint256 amount);

    function donate(uint256 projectId) public payable {
        require(projectId > 0, "Invalid project ID");
        donations[projectId].push(Donation(msg.sender, msg.value, block.timestamp));
        emit DonationReceived(projectId, msg.sender, msg.value);
    }

    function getDonations(uint256 projectId) public view returns (Donation[] memory) {
        return donations[projectId];
    }

    function markProjectCompleted(uint256 projectId) public {
        require(projectId > 0, "Invalid project ID");
        require(!projectCompleted[projectId], "Project already completed");
        projectCompleted[projectId] = true;
    }
}
