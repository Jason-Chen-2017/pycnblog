import numpy as np

# 基于用户的最近邻推荐算法
def user_based_recommendation(user_id, user_history, items, k=5):
    user_vector = [user_history[item] for item in items[user_id]]
    user_vector = np.array(user_vector).reshape(1, -1)
    neighbors = nearest_neighbors.kneighbors(user_vector, n_neighbors=k)
    neighbor_user_ids = neighbors[1][0]
    recommended_items = []

    for neighbor_id in neighbor_user_ids:
        recommended_items.extend(items[neighbor_id])

    return list(set(recommended_items) - set(items[user_id]))
