   $$\Delta w_i' = \Delta w_i / \Delta w_{total}$$

```python
import pandas as pd
from sklearn.linear_model import LinearRegression

# 读取投资组合数据
portfolio_data = pd.read_csv('portfolio_data.csv')

# 计算当前投资组合的资产比例
current_weights = portfolio_data['weight'].values

# 定义再平衡触发条件（资产偏离度阈值）
threshold = 0.05

# 计算资产偏离度
deviation = current_weights - target_weights

# 判断是否触发再平衡
if abs(deviation).sum() > threshold:
    # 生成再平衡计划
    rebalance_plan = generate_rebalance_plan(current_weights, target_weights)
    # 执行再平衡操作
    execute_rebalance_plan(rebalance_plan)
else:
    print("当前投资组合无再平衡需求。")

# 再平衡计划的生成函数
def generate_rebalance_plan(current_weights, target_weights):
    rebalance_plan = {}
    for asset in current_weights.index:
        rebalance_plan[asset] = {'action': '', 'amount': 0}
        if current_weights[asset] > target_weights[asset]:
            rebalance_plan[asset]['action'] = 'sell'
            rebalance_plan[asset]['amount'] = current_weights[asset] - target_weights[asset]
        elif current_weights[asset] < target_weights[asset]:
            rebalance_plan[asset]['action'] = 'buy'
            rebalance_plan[asset]['amount'] = target_weights[asset] - current_weights[asset]
    return rebalance_plan

# 再平衡操作的执行函数
def execute_rebalance_plan(rebalance_plan):
    print("执行再平衡操作：")
    for asset, action in rebalance_plan.items():
        if action['action'] == 'sell':
            print(f"卖出{asset}，数量：{action['amount']}。")
        elif action['action'] == 'buy':
            print(f"买入{asset}，数量：{action['amount']}。")
```
