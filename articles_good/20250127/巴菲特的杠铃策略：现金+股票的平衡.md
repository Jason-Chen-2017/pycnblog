# 投资策略服务
class InvestmentStrategy:
    def __init__(self, cash_ratio, stock_ratio):
        self.cash_ratio = cash_ratio
        self.stock_ratio = stock_ratio

    def adjust_strategy(self, market_conditions):
        if market_conditions == "high_uncertainty":
            self.cash_ratio += 0.1
            self.stock_ratio -= 0.1
        elif market_conditions == "low_uncertainty":
            self.cash_ratio -= 0.1
            self.stock_ratio += 0.1
        else:
            pass

# 投资组合管理服务
class Portfolio:
    def __init__(self, total_capital):
        self.total_capital = total_capital
        self.investment_strategy = InvestmentStrategy(0.5, 0.5)

    def calculate_investment(self):
        cash_investment = self.total_capital * self.investment_strategy.cash_ratio
        stock_investment = self.total_capital * self.investment_strategy.stock_ratio
        return cash_investment, stock_investment

    def update_strategy(self, market_conditions):
        self.investment_strategy.adjust_strategy(market_conditions)

# 用户界面
class UserInterface:
    def __init__(self):
        self.portfolio = Portfolio(100000)

    def display_portfolio(self):
        cash_investment, stock_investment = self.portfolio.calculate_investment()
        print(f"Current portfolio: Cash = ${cash_investment:.2f}, Stocks = ${stock_investment:.2f}")

    def submit_investment_request(self, market_conditions):
        self.portfolio.update_strategy(market_conditions)
        self.display_portfolio()

# 主函数
if __name__ == "__main__":
    ui = UserInterface()
    ui.submit_investment_request("high_uncertainty")
    ui.submit_investment_request("low_uncertainty")
