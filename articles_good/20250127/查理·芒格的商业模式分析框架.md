from django.shortcuts import render
from .models import User
from django.contrib.auth import login

def register(request):
    if request.method == 'POST':
        username = request.POST['username']
        email = request.POST['email']
        password = request.POST['password']
        user = User.objects.create_user(username=username, email=email, password=password)
        login(request, user)
        return render(request, 'base.html')
    return render(request, 'register.html')

# app/courses/models.py

本文由AI天才研究院（AI Genius Institute）与《禅与计算机程序设计艺术》（Zen And The Art of Computer Programming）的作者合作撰写，旨在为读者提供深入、实用的商业模式分析框架，以助力企业的长期成功。AI天才研究院专注于前沿人工智能技术的研发和应用，致力于推动人工智能在各领域的创新与发展。《禅与计算机程序设计艺术》则是一部经典的计算机科学著作，强调了编程中的思维艺术和哲学思考。两相结合，本文力求为读者带来全新的商业模式分析视角和实用技巧。
