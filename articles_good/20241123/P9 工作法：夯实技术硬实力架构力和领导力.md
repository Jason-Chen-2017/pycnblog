# Python代码实现电商购物车

class ShoppingCart:
    def __init__(self):
        self.items = []

    def add_item(self, item):
        self.items.append(item)

    def remove_item(self, item):
        self.items.remove(item)

    def total(self):
        total = 0
        for item in self.items:
            total += item.price
        return total

# 测试代码

class ShoppingCart:
