    for char in plaintext:
        ciphertext += key ^ char
    return ciphertext

# 解密算法
def decrypt(ciphertext, key):
    plaintext = ""
    for char in ciphertext:
        plaintext += key ^ char
    return plaintext

# 认证算法
def authenticate(plaintext, key, hash_value):
    calculated_hash = hash(plaintext)
    return calculated_hash == hash_value
```
