```python
    """
    领导者决策模型，根据问题、选项和权重计算最优决策。

    :param options: 选项列表
```

```python
class DecisionNode:
    def __init__(self, name, decision_rule=None, left=None, right=None, probability=None, reward=None):
        self.name = name
        self.decision_rule = decision_rule
        self.left = left
        self.right = right
        self.probability = probability
        self.reward = reward

def select_best_strategy(decision_tree, current_context):
    """
    基于决策树选择最佳战略
    :param decision_tree: 决策树
    :param current_context: 当前情境
    :return: 最佳战略
    """
    if decision_tree.decision_rule.is_satisfied(current_context):
        return decision_tree.reward
    else:
        if decision_tree.left:
            left_reward = select_best_strategy(decision_tree.left, current_context)
        else:
            left_reward = 0
        if decision_tree.right:
            right_reward = select_best_strategy(decision_tree.right, current_context)
        else:
            right_reward = 0
        return max(left_reward, right_reward)
```

```python
    """
    贪心算法求解问题
    :param problem: 问题
    :param state: 初始状态
    :return: 解决方案
```

```python
import numpy as np

def decision_tree(decision_points, probabilities, rewards):
    """
    基于决策树的决策模型
    :param decision_points: 决策点
    :param probabilities: 概率
    :param rewards: 奖励
    :return: 最佳决策结果
    """
    expected_rewards = np.zeros(len(decision_points))
    for i, decision_point in enumerate(decision_points):
        if probabilities[i] > 0:
            expected_reward = probabilities[i] * rewards[i]
            expected_rewards[i] = expected_reward
    best_index = np.argmax(expected_rewards)
    return decision_points[best_index]

# 示例数据
decision_points = ['方案A', '方案B', '方案C']
probabilities = [0.3, 0.5, 0.2]
rewards = [10, 15, 8]

# 决策
best_decision = decision_tree(decision_points, probabilities, rewards)
print(f"最佳决策：{best_decision}")
```
