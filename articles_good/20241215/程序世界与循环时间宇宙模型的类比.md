```python
    if len(arr) <= 1:
        return arr
    
    pivot_index = random.randint(0, len(arr) - 1)  # 随机选择分区元素
    pivot = arr[pivot_index]

    i = 0
    j = len(arr) - 1
    
    # 分区操作
    for k in range(len(arr) - 1):
        if arr[k] < pivot:
            arr[i], arr[k] = arr[k], arr[i]
            i += 1
        elif arr[k] == pivot:
            arr[j], arr[k] = arr[k], arr[j]
            j -= 1
    
    arr[i:], arr[j+1:] = arr[j+1:i+1], []  # 将分区后的数组重新组合
    
    # 递归调用
    left = quicksort(arr[:i])
    right = quicksort(arr[i+1:])
    
    return left + [pivot] + right
```
