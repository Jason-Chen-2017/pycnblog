# 导入必要的库
import numpy as np
import pandas as pd
from scipy.sparse import csr_matrix
from mermaid import Mermaid

# 函数：求解最大子序列和
def max_subsequence_sum(sequence):
    """
    使用数学简约性算法求解最大子序列和。
