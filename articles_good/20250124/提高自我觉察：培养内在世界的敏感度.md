```python
# 数据安全防护系统核心实现
import json
import hashlib
import base64

# 用户登录验证
def user_login(username, password):
    # 读取用户信息
    with open('user_data.json', 'r') as file:
        user_data = json.load(file)

    # 验证用户名和密码
    if username in user_data and user_data[username]['password'] == hashlib.sha256(password.encode()).hexdigest():
        return True
    else:
        return False

# 数据加密
def encrypt_data(data, key):
    return base64.b64encode(hashlib.sha256(data.encode() + key.encode()).digest()).decode()

# 数据备份
def backup_data(data):
    with open('data_backup.json', 'w') as file:
        json.dump(data, file)

# 主程序
if __name__ == '__main__':
    # 用户登录
    username = input("请输入用户名：")
    password = input("请输入密码：")
    if user_login(username, password):
        print("登录成功！")
        # 上传数据
        data = input("请上传数据：")
        # 加密数据
        key = input("请输入加密密钥：")
        encrypted_data = encrypt_data(data, key)
        print(f"加密数据：{encrypted_data}")
        # 备份数据
        backup_data({'data': encrypted_data})
        print("数据备份完成。")
    else:
        print("登录失败，请重新输入！")
```

**代码应用解读与分析**：
- 用户登录验证：读取用户数据，通过SHA256加密算法验证用户名和密码。
- 数据加密：使用base64编码和SHA256加密算法对数据进行加密。
- 数据备份：将加密后的数据写入备份文件。

**实际案例分析和详细讲解剖析**：
- 案例一：用户登录并上传敏感数据，系统自动进行加密和备份，确保数据安全。
- 案例二：用户密码泄露，但由于使用了SHA256加密算法，即使密码泄露也无法破解。
