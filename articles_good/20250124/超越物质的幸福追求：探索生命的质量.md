#### 核心概念原理

1. **幸福感（Happiness）**：幸福感是指个体对生活总体满意度和情感愉悦度的主观感受。它不仅仅是对物质生活的满足，还包括心理、社会、情感和灵性等多方面的体验。幸福感的核心在于个体的内心体验，即对生活的积极感受和满足感。

2. **心理健康（Mental Health）**：心理健康是指个体在心理上保持良好的状态，能够有效应对生活中的各种挑战和压力，保持积极、乐观的心态。心理健康不仅包括情绪的稳定性，还涉及认知功能、人际关系和自我调节能力等方面。

3. **自我认知（Self-awareness）**：自我认知是指个体对自己内心感受、情感状态和思维模式的了解和认识。通过自我认知，个体能够更好地理解自己的需求和动机，从而做出更明智的决策和选择，提升自我价值和幸福感。

4. **人生价值观（Life Values）**：人生价值观是指个体对生活中最重要的事物的信念和评价。它包括对家庭、工作、友谊、健康、精神成长等方面的重视程度。人生价值观决定了个体在追求幸福的过程中所采取的行动和态度。

```python
# app.py

from flask import Flask, render_template, request, redirect, url_for
from flask_sqlalchemy import SQLAlchemy
from flask_login import LoginManager, login_user, logout_user, login_required, current_user

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///happiness.db'
app.config['SECRET_KEY'] = 'your_secret_key'

db = SQLAlchemy(app)
login_manager = LoginManager(app)

@login_manager.user_loader
def load_user(user_id):
    return User.query.get(int(user_id))

class User(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    username = db.Column(db.String(100), unique=True, nullable=False)
    password = db.Column(db.String(100), nullable=False)
    happiness_score = db.Column(db.Integer, default=5)
    mental_health_score = db.Column(db.Integer, default=5)
    life_satisfaction_score = db.Column(db.Integer, default=5)
    social_support_score = db.Column(db.Integer, default=5)

@app.route('/')
@login_required
def index():
    return render_template('index.html', user=current_user)

@app.route('/register', methods=['GET', 'POST'])
def register():
    if request.method == 'POST':
        username = request.form['username']
        password = request.form['password']
        new_user = User(username=username, password=password)
        db.session.add(new_user)
        db.session.commit()
        return redirect(url_for('login'))
    return render_template('register.html')

@app.route('/login', methods=['GET', 'POST'])
def login():
    if request.method == 'POST':
        username = request.form['username']
        password = request.form['password']
        user = User.query.filter_by(username=username).first()
        if user and user.password == password:
            login_user(user)
            return redirect(url_for('index'))
        return 'Invalid username or password'
    return render_template('login.html')

@app.route('/logout')
@login_required
def logout():
    logout_user()
    return redirect(url_for('login'))

@app.route('/评估', methods=['GET', 'POST'])
@login_required
def assess():
    if request.method == 'POST':
        mental_health_score = request.form['mental_health']
        life_satisfaction_score = request.form['life_satisfaction']
        social_support_score = request.form['social_support']
        current_user.mental_health_score = mental_health_score
        current_user.life_satisfaction_score = life_satisfaction_score
        current_user.social_support_score = social_support_score
        db.session.commit()
        happiness_score = (0.4 * mental_health_score) + (0.3 * life_satisfaction_score) + (0.3 * social_support_score)
        current_user.happiness_score = happiness_score
        db.session.commit()
        return redirect(url_for('index'))
    return render_template('assess.html', user=current_user)

if __name__ == '__main__':
    db.create_all()
    app.run(debug=True)
```
