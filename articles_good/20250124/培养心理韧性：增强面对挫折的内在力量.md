```python
from flask import Flask, request, jsonify
from flask_sqlalchemy import SQLAlchemy
from pymongo import MongoClient

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///psychology.db'
db = SQLAlchemy(app)
client = MongoClient('localhost', 27017)
db_client = client.psychology

class Employee(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(100))
    questionnaire_data = db.Column(db.String(1000))

@app.route('/submit_questionnaire', methods=['POST'])
def submit_questionnaire():
    data = request.json
    employee = Employee(
        name=data['name'],
        questionnaire_data=data['questionnaire_data']
    )
    db.session.add(employee)
    db.session.commit()
    return jsonify({"status": "success", "message": "Questionnaire submitted."})

@app.route('/get_questionnaire_data', methods=['GET'])
def get_questionnaire_data():
    employee_id = request.args.get('id')
    employee = Employee.query.get(employee_id)
    if employee:
        return jsonify({"status": "success", "data": employee.questionnaire_data})
    else:
        return jsonify({"status": "error", "message": "Employee not found."})

if __name__ == '__main__':
    db.create_all()
    app.run(debug=True)
```
