from flask import Flask, render_template, request, redirect, url_for

class User(db.Model):

    username = db.Column(db.String(80), unique=True, nullable=False)
    password = db.Column(db.String(120), nullable=False)
    dependency_level = db.Column(db.Integer, nullable=False)

class Holiday(db.Model):
    id = db.Column(db.Integer, primary_key=True)

class DependencyAssessment(db.Model):

class EnvironmentSetup(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    holiday_id = db.Column(db.Integer, db.ForeignKey('holiday.id'), nullable=False)
    setup_instructions = db.Column(db.Text, nullable=False)

class HolidayExecution(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    holiday_id = db.Column(db.Integer, db.ForeignKey('holiday.id'), nullable=False)
    execution_status = db.Column(db.Boolean, nullable=False)
    execution_date = db.Column(db.Date, nullable=False)

class EffectEvaluation(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    holiday_id = db.Column(db.Integer, db.ForeignKey('holiday.id'), nullable=False)
    effect = db.Column(db.Integer, nullable=False)

@app.route('/')
def index():
    return render_template('index.html')

@app.route('/register', methods=['GET', 'POST'])
def register():
    if request.method == 'POST':
        username = request.form['username']
        password = request.form['password']
        user = User(username=username, password=password)
        db.session.add(user)
        db.session.commit()

    return render_template('register.html')

@app.route('/login', methods=['GET', 'POST'])
def login():
    if request.method == 'POST':
        username = request.form['username']
        password = request.form['password']
        user = User.query.filter_by(username=username, password=password).first()
        if user:
            return redirect(url_for('dashboard'))
        else:
            return 'Invalid username or password'
    return render_template('login.html')

@app.route('/dashboard')
def dashboard():
    return render_template('dashboard.html')

if __name__ == '__main__':
    db.create_all()
    app.run(debug=True)
