    def __init__(self):
        self.users = []

    def add_user(self, user):
        self.users.append(user)

    def get_user(self, username):
        for user in self.users:
            if user.username == username:
                return user
        return None

# 关系管理模块
class RelationshipManager:
    def __init__(self):
        self.relationships = []

    def add_relationship(self, user1, user2, relationship):
        self.relationships.append((user1, user2, relationship))

    def get_relationship(self, user1, user2):
        for user1, user2, relationship in self.relationships:
            if user1 == user1 and user2 == user2:
                return relationship
        return None
