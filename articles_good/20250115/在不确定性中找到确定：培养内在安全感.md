import pandas as pd
from sklearn.cluster import KMeans
import numpy as np

# 假设我们已经收集到了情绪数据、行为数据和目标数据
emotion_data = pd.DataFrame({'emotion': [0.3, 0.7, 0.5, 0.8]})

#### 算法原理的数学模型和公式

ISEA的核心数学模型是基于聚类分析的，其中使用了K-means算法来识别个体在应对不确定性时的主要特征。以下是一个简单的数学模型，用于描述K-means算法：

$$
\text{K-means} \text{算法模型} \\
\text{给定} \ \mathbf{X} = \{\mathbf{x}_i\}_{i=1}^n \\
\text{目标是最小化} \\
J = \sum_{i=1}^n \sum_{j=1}^k ||\mathbf{x}_i - \mathbf{c}_j||^2
$$

其中，$\mathbf{X}$ 是数据集，$\mathbf{x}_i$ 是数据集中的第 $i$ 个样本，$k$ 是聚类数目，$\mathbf{c}_j$ 是聚类中心。目标是最小化所有样本与聚类中心的欧几里得距离平方和。

#### 举例说明

假设我们有以下数据集：

$$
\mathbf{X} = \{ \mathbf{x}_1 = [1, 2], \mathbf{x}_2 = [2, 1], \mathbf{x}_3 = [1, 1], \mathbf{x}_4 = [2, 2] \}
$$

我们使用K-means算法将其分为两个聚类。首先，随机选择两个聚类中心：

$$
\mathbf{c}_1 = [1.5, 1.5], \ \mathbf{c}_2 = [2.5, 2.5]
$$

然后，计算每个样本与聚类中心的距离，并重新分配样本到最近的聚类中心。迭代进行直到聚类中心不再发生变化。通过这个简单的例子，我们可以看到K-means算法的基本运行过程。

### 数学模型与公式讲解

在理解内在安全感增强算法（ISEA）时，数学模型和公式起到了关键作用。以下我们将详细讲解ISEA中的主要数学模型和公式，并使用LaTeX格式进行展示。

#### 数学模型的基本概念

ISEA的核心数学模型是基于聚类分析的。聚类分析是一种无监督学习方法，用于将数据集划分为若干个簇，使得同一簇内的数据点彼此之间距离较近，而不同簇的数据点之间距离较远。K-means算法是其中一种常用的聚类方法，其目标是找到一组聚类中心，使得所有数据点到聚类中心的距离平方和最小。

#### 公式详解

以下是K-means算法的主要公式：

1. **初始化聚类中心**：

   $$
   \text{初始化} \ \mathbf{c}_j^{(0)} = \arg\min_{\mathbf{c}_j} \sum_{i=1}^n ||\mathbf{x}_i - \mathbf{c}_j||^2
   $$

   其中，$\mathbf{c}_j^{(0)}$ 是初始聚类中心，$\mathbf{x}_i$ 是第 $i$ 个数据点。

2. **更新聚类中心**：

   $$
   \mathbf{c}_j^{(t+1)} = \arg\min_{\mathbf{c}_j} \sum_{i=1}^n ||\mathbf{x}_i - \mathbf{c}_j||^2
   $$

   其中，$\mathbf{c}_j^{(t+1)}$ 是在 $t+1$ 次迭代后的聚类中心。

3. **计算聚类标签**：

   $$
   \text{聚类标签} \ \mathbf{y}_i^{(t)} = \arg\min_{j} ||\mathbf{x}_i - \mathbf{c}_j^{(t)}||^2
   $$

   其中，$\mathbf{y}_i^{(t)}$ 是第 $i$ 个数据点在 $t$ 次迭代后的聚类标签。

4. **终止条件**：

   $$
   \text{如果} \ \sum_{i=1}^n ||\mathbf{x}_i - \mathbf{c}_j^{(t+1)}||^2 - ||\mathbf{x}_i - \mathbf{c}_j^{(t)}||^2 < \epsilon, \text{则终止迭代}
   $$

   其中，$\epsilon$ 是预设的收敛阈值。

def analyze_data():
    conn = sqlite3.connect('isea.db')

    c.execute("SELECT * FROM users")
    data = c.fetchall()
    conn.close()

# K-means聚类分析
def kmeans_analysis(data, n_clusters=3):
    data_normalized = (np.array(data) - np.mean(data, axis=0)) / np.std(data, axis=0)
    kmeans = KMeans(n_clusters=n_clusters)
    kmeans.fit(data_normalized)
    return kmeans.labels_

# 建议生成模块
def generate_suggestions(labels):
    suggestions = []
    for label in labels:
        if label == 0:
            suggestions.append("调整情绪管理策略，提高积极情绪比例。")
        elif label == 1:
            suggestions.append("优化行为模式，提高效率。")
        elif label == 2:
            suggestions.append("设定更明确的目标，提升目标达成率。")
    return suggestions

# 反馈模块
def save_suggestion(user_id, suggestion):
    conn = sqlite3.connect('isea.db')
    c = conn.cursor()
    c.execute("INSERT INTO suggestions (user_id, suggestion) VALUES (?, ?)", (user_id, suggestion))
    conn.commit()
    conn.close()

# 主程序
if __name__ == "__main__":
    # 收集用户数据

    collect_data("user2", 0.4, 0.6, 0.8)

    # 分析用户数据
    data = analyze_data()
    labels = kmeans_analysis(data)

    # 生成个性化建议
    suggestions = generate_suggestions(labels)

    # 存储建议
    for suggestion in suggestions:
        save_suggestion(1, suggestion)
