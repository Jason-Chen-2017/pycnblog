    # 计算初始速度
    if action_type == 'swing':
        initial_velocity = np.sqrt(2 * g * rope_length)
        initial_angle = np.pi / 4  # 45度摆动角
    elif action_type == 'twist':
        initial_velocity = weight * g / rope_length
        initial_angle = np.pi / 2  # 90度旋转
    
    # 计算运动轨迹
    trajectory = {
        'initial_velocity': initial_velocity,
        'initial_angle': initial_angle,
        'time_to_max_height': 2 * initial_velocity * np.sin(initial_angle) / g,
        'max_height': initial_velocity * np.sin(initial_angle) ** 2 / (2 * g),
    }
    
    return trajectory
