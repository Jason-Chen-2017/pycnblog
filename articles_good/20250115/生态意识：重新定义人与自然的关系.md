from flask import Flask, request, jsonify
from flask_mongoengine import MongoEngine

app = Flask(__name__)
app.config['MONGODB_SETTINGS'] = {
    'db': 'eco_education_platform'
}
db = MongoEngine(app)

class User(db.Document):
    username = db.StringField(required=True, unique=True)
    password = db.StringField(required=True)
    role = db.StringField(default='student')

class Course(db.Document):
    title = db.StringField(required=True)
    description = db.StringField()
    chapters = db.ListField(db.ReferenceField('Chapter'))

class Chapter(db.Document):
    course = db.ReferenceField(Course)
    title = db.StringField(required=True)
    content = db.StringField()

@app.route('/api/users', methods=['POST'])
def create_user():
    data = request.json
    user = User(**data).save()
    return jsonify({'status': 'success', 'user': user.to_dict()})

@app.route('/api/courses', methods=['GET'])
def get_courses():
    courses = Course.objects().all()
    return jsonify({'status': 'success', 'courses': [course.to_dict() for course in courses]})

if __name__ == '__main__':
    app.run(debug=True)
```

以上代码展示了如何使用Python和Flask框架构建一个简单的生态意识教育平台后端。我们定义了`User`、`Course`和`Chapter`三个MongoDB文档模型，分别表示用户、课程和章节。`create_user`函数用于创建新用户，`get_courses`函数用于获取所有课程列表。这些函数通过REST API暴露出来，供前端调用。
