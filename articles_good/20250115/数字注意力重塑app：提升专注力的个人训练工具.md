import pandas as pd
import numpy as np
from sklearn.preprocessing import StandardScaler

# 假设已经收集到了用户的数据，存储在一个DataFrame中
data = pd.DataFrame({
    'attention_time': [120, 90, 150, 80],
    'distract_count': [3, 5, 2, 4],
    'task_time': [30, 40, 45, 35]
})

# 数据预处理
scaler = StandardScaler()
data_scaled = scaler.fit_transform(data)

# 数据分析
mean_attention_time = np.mean(data_scaled[:, 0])
max_distract_count = np.max(data_scaled[:, 1])
mean_task_time = np.mean(data_scaled[:, 2])

print(f"平均专注时间：{mean_attention_time:.2f}分钟")
print(f"最大分心次数：{max_distract_count}")
