```python
import numpy as np
import matplotlib.pyplot as plt
from scipy.signal import filtfilt

def preprocess_signal(signal, cutoff_freq, fs):
    # 低通滤波去除高频噪声
    b, a = signal.butter(cutoff_freq, fs)
    filtered_signal = filtfilt(b, a, signal)
    return filtered_signal

def analyze_signal(filtered_signal, fs):
    # 傅里叶变换进行频率分析
    freqs, signal_fft = signal.fft(filtered_signal, nperseg=fs)
    return freqs, signal_fft

def visualize_result(freqs, signal_fft):
    # 生成频率-幅度图
    plt.plot(freqs, np.abs(signal_fft))
    plt.xlabel('Frequency (Hz)')
    plt.ylabel('Amplitude')
    plt.show()

# 示例数据
signal = np.random.normal(0, 1, 1000)
fs = 100  # 采样频率
cutoff_freq = 10  # 低通滤波截止频率

# 信号预处理
preprocessed_signal = preprocess_signal(signal, cutoff_freq, fs)

# 信号分析
freqs, signal_fft = analyze_signal(preprocessed_signal, fs)

# 结果可视化
visualize_result(freqs, signal_fft)
```
