##### 2.1.1 区块链的定义

区块链是一个由多个按时间顺序排列的“区块”所组成的链式数据结构。每一个区块都包含了特定时间段内发生的一系列交易记录，这些区块通过加密技术连接起来，形成一个不可篡改的数据链条。

1. **数据存储**：区块链将交易数据存储在区块中，每个区块包含一定数量的交易记录。这些区块按时间顺序链接在一起，形成区块链。

2. **加密技术**：区块链利用密码学技术确保数据的安全性和完整性。每个区块都包含一个时间戳和一个哈希值，哈希值用于验证数据的完整性和链接其他区块。

3. **分布式节点**：区块链由多个分布式节点共同维护，每个节点都保存完整的区块链副本。节点通过共识算法（如工作量证明Proof of Work）达成一致，确保数据的一致性和安全性。

4. **去中心化**：区块链的去中心化特性使其不受单一中心控制，增强了系统的容错性和抗攻击能力。

import requests
from bs4 import BeautifulSoup
from nltk.tokenize import word_tokenize
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.model_selection import train_test_split
from sklearn.naive_bayes import MultinomialNB
from sklearn.pipeline import make_pipeline

# 网络爬虫
def crawl_website(url):
    try:
        response = requests.get(url)
        response.raise_for_status()
        return response.text
    except requests.RequestException as e:
        print(e)
        return None

# 侵权行为识别
def detect_infringement(content, model):
    # 提取关键词
    tokens = word_tokenize(content)
    keywords = set(tokens)
    
    # 计算关键词的TF-IDF值
    vectorizer = TfidfVectorizer()
    features = vectorizer.transform([' '.join(keywords)])
    
    # 预测侵权行为
    prediction = model.predict(features)
    return prediction[0] == 1

# 侵权行为通知
def notify_infringement(copyright_id, infringement_content):
    # 这里可以使用邮件、短信或其他通知方式
    print(f"Infringement detected for copyright ID {copyright_id}: {infringement_content}")

# 训练侵权行为识别模型
def train_model(infringing_texts, non_infringing_texts):
    # 划分训练集和测试集
    X_train, X_test, y_train, y_test = train_test_split(infringing_texts + non_infringing_texts, [1] * len(infringing_texts) + [0] * len(non_infringing_texts), test_size=0.2, random_state=42)
    
    # 创建TF-IDF向量器和朴素贝叶斯分类器
    model = make_pipeline(TfidfVectorizer(), MultinomialNB())
    
    # 训练模型
    model.fit(X_train, y_train)
    
    # 评估模型
    print("Model accuracy on test set:", model.score(X_test, y_test))
    
    return model

# 示例数据
infringing_texts = ["This is a copy of your original work.", "Your work has been stolen and used without permission."]
non_infringing_texts = ["This is a different work.", "Your work is safe and not being copied."]

# 训练模型
infringement_model = train_model(infringing_texts, non_infringing_texts)

# 检测侵权行为
url = "http://example.com/content"
content = crawl_website(url)
if content:
    is_infringed = detect_infringing(content, infringement_model)
    if is_infringed:
        notify_infringement("12345", content)
