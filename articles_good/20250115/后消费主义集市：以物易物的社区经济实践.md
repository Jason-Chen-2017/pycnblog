```python
        self.type = type
        self.item = item
        self.service = service

def match_demand(demand, items, services):
    matched_items = []
    matched_services = []

    if demand.type == "物品":
        for item in items:
            if item.value == demand.item.value:
                matched_items.append(item)
    elif demand.type == "服务":
        for service in services:
            if service.value == demand.service.value:
                matched_services.append(service)

    return matched_items if demand.type == "物品" else matched_services

# 初始化需求
```

```python
# 以物易物系统核心实现

from flask import Flask, request, jsonify
from flask_pymongo import PyMongo
from bson.json_util import dumps

app = Flask(__name__)
app.config["MONGO_URI"] = "mongodb://localhost:27017/swap社区"
mongo = PyMongo(app)

# 用户管理
@app.route("/register", methods=["POST"])
def register():
    data = request.json
    username = data["username"]
    password = data["password"]
    user = {"username": username, "password": password}
    result = mongo.db.users.insert_one(user)
    return jsonify({"status": "success", "userId": str(result.inserted_id)})

@app.route("/login", methods=["POST"])
def login():
    data = request.json
    username = data["username"]
    password = data["password"]
    user = mongo.db.users.find_one({"username": username, "password": password})
    if user:
        return jsonify({"status": "success", "userId": str(user["_id"])})
    else:
        return jsonify({"status": "error", "message": "用户名或密码错误"})

# 物品和服务管理
@app.route("/items", methods=["POST"])
def add_item():
    data = request.json
    item = {
        "userId": data["userId"],
        "name": data["name"],
        "description": data["description"],
        "value": data["value"],
    }
    result = mongo.db.items.insert_one(item)
    return jsonify({"status": "success", "itemId": str(result.inserted_id)})

@app.route("/items", methods=["GET"])
def get_items():
    items = list(mongo.db.items.find())
    return jsonify(dumps(items))

@app.route("/services", methods=["POST"])
def add_service():
    data = request.json
    service = {
        "userId": data["userId"],
        "name": data["name"],
        "description": data["description"],
        "value": data["value"],
    }
    result = mongo.db.services.insert_one(service)
    return jsonify({"status": "success", "serviceId": str(result.inserted_id)})

@app.route("/services", methods=["GET"])
def get_services():
    services = list(mongo.db.services.find())
    return jsonify(dumps(services))

# 交换管理
@app.route("/exchanges", methods=["POST"])
def create_exchange():
    data = request.json
    exchange = {
        "itemId": data["itemId"],
        "serviceId": data["serviceId"],
        "userId": data["userId"],
        "status": "待确认",
    }
    result = mongo.db.exchanges.insert_one(exchange)
    return jsonify({"status": "success", "exchangeId": str(result.inserted_id)})

@app.route("/exchanges", methods=["GET"])
def get_exchanges():
    exchanges = list(mongo.db.exchanges.find())
    return jsonify(dumps(exchanges))

if __name__ == "__main__":
    app.run(debug=True)
```

   ```python
   }
   ```


3. **系统匹配交换**：
   ```python
   # 匹配交换
   exchange = {
       "itemId": "itemA_id",
       "serviceId": "serviceB_id",
       "userId": "userA_id",
       "status": "待确认"
   ```

   ```python
   # 用户A确认交换
   ```

   ```python
   # 用户B确认交换
   exchange["status"] = "已完成"
   mongo.db.exchanges.update_one({"_id": exchange["_id"]}, {"$set": exchange})
   ```
