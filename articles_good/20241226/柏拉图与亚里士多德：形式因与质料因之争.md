from flask import Flask, request, jsonify
from flask_sqlalchemy import SQLAlchemy
import mermaid

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'mysql+pymysql://username:password@localhost/db_name'
db = SQLAlchemy(app)

class Form(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    shape = db.Column(db.String(255))
    structure = db.Column(db.String(255))

class Material(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    substance = db.Column(db.String(255))
    properties = db.Column(db.String(255))

class Entity(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    form_id = db.Column(db.Integer, db.ForeignKey('form.id'))
    material_id = db.Column(db.Integer, db.ForeignKey('material.id'))

@app.route('/api/forms', methods=['GET'])
def get_forms():
    forms = Form.query.all()
    return jsonify([{'id': form.id, 'shape': form.shape, 'structure': form.structure} for form in forms])

@app.route('/api/materials', methods=['GET'])
def get_materials():
    materials = Material.query.all()
    return jsonify([{'id': material.id, 'substance': material.substance, 'properties': material.properties} for material in materials])

@app.route('/api/entities', methods=['POST'])
def create_entity():
    form_id = request.json['form_id']
    material_id = request.json['material_id']
    form = Form.query.get(form_id)
    material = Material.query.get(material_id)
    entity = Entity(form=form, material=material)
    db.session.add(entity)
    db.session.commit()
    return jsonify({'id': entity.id})

if __name__ == '__main__':
    db.create_all()
    app.run(debug=True)

```python
from flask import Flask, request, jsonify
from flask_sqlalchemy import SQLAlchemy
import mermaid

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'mysql+pymysql://username:password@localhost/db_name'
db = SQLAlchemy(app)

class Form(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    shape = db.Column(db.String(255))
    structure = db.Column(db.String(255))

class Material(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    substance = db.Column(db.String(255))
    properties = db.Column(db.String(255))

class Entity(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    form_id = db.Column(db.Integer, db.ForeignKey('form.id'))
    material_id = db.Column(db.Integer, db.ForeignKey('material.id'))

@app.route('/api/forms', methods=['GET'])
def get_forms():
    forms = Form.query.all()
    return jsonify([{'id': form.id, 'shape': form.shape, 'structure': form.structure} for form in forms])

@app.route('/api/materials', methods=['GET'])
def get_materials():
    materials = Material.query.all()
    return jsonify([{'id': material.id, 'substance': material.substance, 'properties': material.properties} for material in materials])

@app.route('/api/entities', methods=['POST'])
def create_entity():
    form_id = request.json['form_id']
    material_id = request.json['material_id']
    form = Form.query.get(form_id)
    material = Material.query.get(material_id)
    entity = Entity(form=form, material=material)
    db.session.add(entity)
    db.session.commit()
    return jsonify({'id': entity.id})

if __name__ == '__main__':
    db.create_all()
    app.run(debug=True)
```

### 8.6.3 Python代码示例

以下是一个简单的Python代码示例，用于演示如何创建形式因和质料因，并构建实体：

```python
# 形式因定义
class Form:
    def __init__(self, shape, structure):
        self.shape = shape
        self.structure = structure

# 质料因定义
class Material:
    def __init__(self, substance, properties):
        self.substance = substance
        self.properties = properties

# 实体定义
class Entity:
    def __init__(self, form, material):
        self.form = form
        self.material = material
    
    def display(self):
        print(f"实体形式：{self.form.shape}, 结构：{self.form.structure}")
        print(f"实体质料：{self.material.substance}, 属性：{self.material.properties}")

# 创建形式因
form = Form("圆形", "规则")

# 创建质料因
material = Material("金属", "坚硬")

# 创建实体
entity = Entity(form, material)

# 显示实体信息
entity.display()
```
