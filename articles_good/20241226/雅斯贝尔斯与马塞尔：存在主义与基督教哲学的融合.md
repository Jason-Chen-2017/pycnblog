from flask import Flask, request, jsonify
from sqlalchemy import create_engine
from sqlalchemy.orm import sessionmaker
from models import User, Question, Answer

app = Flask(__name__)

# 创建数据库连接
engine = create_engine('sqlite:///philosophy_db.db')
Session = sessionmaker(bind=engine)
session = Session()

# 哲学思想融合推理函数
def integrate_philosophy(question_content):
    # 存在主义推理
    existentialism_answer = "存在主义认为，人类存在的本质在于自由和责任。"
    
    # 基督教哲学推理
    christianity_answer = "基督教哲学认为，人类存在的意义在于与上帝建立关系，寻求救赎。"
    
    # 融合哲学推理
    integration_answer = f"结合存在主义和基督教哲学，我们认为，人类存在的本质在于自由地选择并承担责任，同时与上帝建立关系，寻求救赎。具体来说，{question_content}。"
    
    return integration_answer

# 用户注册接口
@app.route('/register', methods=['POST'])
def register():
    name = request.form['name']
    email = request.form['email']
    password = request.form['password']
    user = User(name=name, email=email, password=password)
    session.add(user)
    session.commit()
    return jsonify({'status': 'success', 'message': '注册成功'})

# 提问接口
@app.route('/ask', methods=['POST'])
def ask():
    content = request.form['content']
    user_id = request.form['user_id']
    question = Question(content=content, user_id=user_id)
    session.add(question)
    session.commit()
    return jsonify({'status': 'success', 'message': '提问成功'})

# 回答接口
@app.route('/answer', methods=['POST'])
def answer():
    question_id = request.form['question_id']
    content = request.form['content']
    answer = Answer(question_id=question_id, content=content)
    session.add(answer)
    session.commit()
    return jsonify({'status': 'success', 'message': '回答成功'})

# 获取回答接口
@app.route('/get_answer', methods=['GET'])
def get_answer():
    question_id = request.args.get('question_id')
    answer = session.query(Answer).filter_by(question_id=question_id).first()
    return jsonify({'status': 'success', 'answer': answer.content})

if __name__ == '__main__':
    app.run(debug=True)
