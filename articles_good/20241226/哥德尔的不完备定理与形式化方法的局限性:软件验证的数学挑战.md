class TradeExecutionEngine:

        self.order_manager = order_manager
        self.market_data_service = market_data_service
        self.exchange_interface = exchange_interface

    def execute_order(self, order_id):
        try:
            order = self.order_manager.get_order(order_id)
            if order.status != 'pending':
                return 'Order is not in pending state.'
            
            market_data = self.market_data_service.get_market_data(order.symbol)
            if market_data.price < order.price:
                return 'Market price is below order price.'
            
            self.exchange_interface.submit_trade(order)
            self.order_manager.update_order_status(order_id, 'executed')
            return 'Trade executed successfully.'
        except Exception as e:
            self.order_manager.update_order_status(order_id, 'error', error_message=str(e))
            return f'Error executing trade: {e}'

    def cancel_order(self, order_id):
        order = self.order_manager.get_order(order_id)
        if order.status in ['pending', 'executed']:
            self.order_manager.update_order_status(order_id, 'cancelled')
            return 'Order cancelled successfully.'
        else:
            return 'Order is not in a cancellable state.'

    def partial_execute_order(self, order_id, quantity):
        order = self.order_manager.get_order(order_id)
        if order.status != 'pending':
            return 'Order is not in pending state.'
        
        available_quantity = self.exchange_interface.get_available_quantity(order.symbol)
        if quantity > available_quantity:
            return 'Insufficient available quantity.'
        
        self.exchange_interface.submit_trade(order, quantity)
        self.order_manager.update_order_status(order_id, 'partially_executed', quantity_executed=quantity)
        return 'Order partially executed successfully.'
