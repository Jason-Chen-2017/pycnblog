def send_data(data):

# 高清视频远程会诊示例
def remote_consultation(video_stream):
    while True:
        frame = video_stream.read()
        if frame is None:
            break
        data = {'frame': frame.tobytes()}
        response = send_data(data)
        if response['status'] == 'error':
            print("数据传输失败：", response['message'])
            break
        print("数据传输成功：", response['message'])
    print("远程会诊结束")

# 主程序

    video_stream = cv2.VideoCapture('path/to/video_stream')
    remote_consultation(video_stream)
    video_stream.release()
