##### 2.3 算法基础

算法是解决问题的步骤集合。它通过数据结构和控制结构来处理数据，以实现特定功能。算法的效率直接影响到程序的运行速度和性能。

以下是一个简单的冒泡排序算法示例：

```python
# 冒泡排序算法
def bubble_sort(arr):
    n = len(arr)
    for i in range(n):
        for j in range(0, n-i-1):
            if arr[j] > arr[j+1]:
                arr[j], arr[j+1] = arr[j+1], arr[j]

# 测试冒泡排序算法
arr = [64, 34, 25, 12, 22, 11, 90]
bubble_sort(arr)
print("Sorted array:", arr)
```

##### 第5.2 数据库操作

```python
from sqlalchemy import create_engine, Column, Integer, String
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker

# 创建数据库引擎
engine = create_engine('sqlite:///mydatabase.db')

# 创建基类
Base = declarative_base()

# 定义学生模型
class Student(Base):
    __tablename__ = 'students'
    id = Column(Integer, primary_key=True)
    name = Column(String(50))
    age = Column(Integer)

# 创建表
Base.metadata.create_all(engine)

# 创建会话
Session = sessionmaker(bind=engine)
session = Session()

# 插入数据
alice = Student(name='Alice', age=30)
bob = Student(name='Bob', age=25)
session.add(alice)
session.add(bob)
session.commit()

# 查询数据
students = session.query(Student).all()
for student in students:
    print(student.name, student.age)

# 更新数据
alice.age = 31
session.commit()

# 删除数据
session.delete(alice)
session.commit()

# 关闭会话
session.close()
```

##### 第9章：区块链技术

区块链技术是一种分布式账本技术，具有去中心化、安全性高和不可篡改的特点。

##### 第9.1 区块链基础

区块链基础包括区块链结构、共识机制和智能合约。以下是一个简单的区块链结构示例：

```python
import hashlib

# 创建区块
class Block:
    def __init__(self, index, transactions, timestamp, previous_hash):
        self.index = index
        self.transactions = transactions
        self.timestamp = timestamp
        self.previous_hash = previous_hash
        self.hash = self.compute_hash()

    def compute_hash(self):
        block_string = f"{self.index}{self.transactions}{self.timestamp}{self.previous_hash}"
        return hashlib.sha256(block_string.encode()).hexdigest()

# 创建区块链
class Blockchain:
    def __init__(self):
        self.unconfirmed_transactions = []
        self.chain = []
        self.create_genesis_block()

    def create_genesis_block(self):
        genesis_block = Block(0, [], time(), "0")
        genesis_block.hash = genesis_block.compute_hash()
        self.chain.append(genesis_block)

    def add_new_transaction(self, transaction):
        self.unconfirmed_transactions.append(transaction)

    def mine(self):
        if not self.unconfirmed_transactions:
            return False
        last_block = self.chain[-1]
        new_block = Block(index=last_block.index + 1,
                          transactions=self.unconfirmed_transactions,
                          timestamp=time(),
                          previous_hash=last_block.hash)
        new_block.hash = new_block.compute_hash()
        self.chain.append(new_block)
        self.unconfirmed_transactions = []
        return new_block.index

# 创建区块链实例
blockchain = Blockchain()

# 添加交易
blockchain.add_new_transaction("Alice to Bob - 10 BTC")
blockchain.add_new_transaction("Alice to Carol - 5 BTC")

# 挖掘区块
blockchain.mine()

# 打印区块链
for block in blockchain.chain:
    print(block)
```

    uint256 public balance;

    constructor() {
        owner = msg.sender;
        balance = 0;
    }

        balance += msg.value;

##### 第10.2 密码学基础

密码学基础包括加密和解密算法。以下是一个简单的加密示例：

```python
from Crypto.Cipher import AES
from Crypto.Util.Padding import pad, unpad
from Crypto.Random import get_random_bytes

# 定义加密和解密函数
def encrypt(plaintext, key):
    cipher = AES.new(key, AES.MODE_CBC)
    ct_bytes = cipher.encrypt(pad(plaintext.encode(), AES.block_size))
    iv = cipher.iv
    return iv + ct_bytes

def decrypt(ct, key, iv):
    ct = ct[iv len():]
    cipher = AES.new(key, AES.MODE_CBC, iv)
    pt = unpad(cipher.decrypt(ct), AES.block_size)
    return pt.decode()

# 创建密钥和初始化向量
key = get_random_bytes(16)
iv = get_random_bytes(16)

# 加密明文
plaintext = "Hello, World!"
ciphertext = encrypt(plaintext, key)

# 解密密文
decrypted_text = decrypt(ciphertext, key, iv)
print(decrypted_text)
```
