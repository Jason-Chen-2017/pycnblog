```python
from flask import Flask, request, jsonify

db = SQLAlchemy(app)

# 定义模型
class TechTarget(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(100), nullable=False)
    description = db.Column(db.Text, nullable=True)
    status = db.Column(db.String(50), nullable=False)

# 初始化数据库
db.create_all()

# API接口
@app.route('/targets', methods=['GET', 'POST'])
def targets():
    if request.method == 'GET':
        targets = TechTarget.query.all()
        return jsonify([target.to_dict() for target in targets])
    elif request.method == 'POST':
        data = request.json
        new_target = TechTarget(name=data['name'], description=data['description'], status=data['status'])
        db.session.add(new_target)
        db.session.commit()
        return jsonify(new_target.to_dict()), 201

# 定义模型转换方法
def to_dict(self):
    return {
        'id': self.id,
        'name': self.name,
        'description': self.description,
        'status': self.status
    }

if __name__ == '__main__':
    app.run(debug=True)
```
