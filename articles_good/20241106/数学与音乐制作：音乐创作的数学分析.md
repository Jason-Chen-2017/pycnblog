```python
import numpy as np
import matplotlib.pyplot as plt

def calculate_rhythm(bpm):
    """
    计算每个拍子之间的时间间隔。
    """
    return 60 / bpm

def analyze_rhythm(rhythm_data):
    """
    分析节奏数据。
    """
    rhythm_values = np.array(rhythm_data)
    mean_rhythm = np.mean(rhythm_values)
    std_rhythm = np.std(rhythm_values)
    return mean_rhythm, std_rhythm

def plot_rhythm(rhythm_data):
    """
    可视化节奏数据。
    """
    plt.plot(rhythm_data)
    plt.xlabel('拍子编号')
    plt.ylabel('时间间隔（秒）')
    plt.title('节奏数据')
    plt.show()

# 示例数据
rhythm_data = [0.0, 0.125, 0.25, 0.375, 0.5, 0.625, 0.75, 0.875, 1.0]

# 计算每个拍子之间的时间间隔
bpm = 120
rhythm_interval = calculate_rhythm(bpm)

# 可视化节奏数据
plot_rhythm(rhythm_data)

print("平均节奏：", mean_rhythm)
print("标准差：", std_rhythm)
```
