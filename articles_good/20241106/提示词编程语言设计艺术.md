```python
# 文本编辑器源代码

class TextEditor:
    def __init__(self):
        self.text = ""

    def insert(self, position, text):
        self.text = self.text[:position] + text + self.text[position:]

    def delete(self, position, length):
        self.text = self.text[:position] + self.text[position + length:]

    def find(self, text):
        return self.text.find(text)

    def replace(self, old_text, new_text):
        self.text = self.text.replace(old_text, new_text)

    def display(self):
        print(self.text)

# 测试代码
editor = TextEditor()
editor.insert(0, "Hello, World!")
editor.display()  # 输出：Hello, World!
editor.delete(7, 7)
editor.display()  # 输出：World!
editor.replace("World", "Hello")
editor.display()  # 输出：Hello
```

- `delete`方法：删除指定位置和长度的文本。
- `find`方法：查找指定文本的位置。
- `replace`方法：替换文本中的指定字符串。
- `display`方法：显示当前文本。
