#### 系统核心实现

以下是系统的核心实现，包括用户管理、练习管理和反馈评估：

```python
# 用户管理
from flask_sqlalchemy import SQLAlchemy

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///users.db'
db = SQLAlchemy(app)

class User(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    username = db.Column(db.String(80), unique=True, nullable=False)
    password = db.Column(db.String(120), nullable=False)

@app.route('/register', methods=['POST'])
def register():
    username = request.form['username']
    password = request.form['password']
    
    new_user = User(username=username, password=password)
    db.session.add(new_user)
    db.session.commit()
    
    return jsonify({'status': 'success'})

# 练习管理
class Exercise(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(120), nullable=False)
    description = db.Column(db.Text, nullable=True)

@app.route('/exercises', methods=['GET'])
def get_exercises():
    exercises = Exercise.query.all()
    return jsonify([{'id': exercise.id, 'name': exercise.name, 'description': exercise.description} for exercise in exercises])

# 反馈评估
class Feedback(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    user_id = db.Column(db.Integer, db.ForeignKey('user.id'), nullable=False)
    exercise_id = db.Column(db.Integer, db.ForeignKey('exercise.id'), nullable=False)
    feedback = db.Column(db.Text, nullable=True)

@app.route('/feedback', methods=['POST'])
def submit_feedback():
    user_id = request.form['user_id']
    exercise_id = request.form['exercise_id']
    feedback = request.form['feedback']
    
    new_feedback = Feedback(user_id=user_id, exercise_id=exercise_id, feedback=feedback)
    db.session.add(new_feedback)
    db.session.commit()
    
    return jsonify({'status': 'success'})

if __name__ == '__main__':
    db.create_all()
    app.run(debug=True)
```
