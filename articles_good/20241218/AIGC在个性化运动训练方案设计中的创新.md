# Python代码示例：生成跑步训练方案
def generate_training_plan(heart_rate, speed, frequency):

    plan = {
        'week_1': {'heart_rate': heart_rate * 0.8, 'speed': speed * 0.9, 'frequency': 3},

# Python代码示例：训练方案优化
def optimize_training_plan(plan, current_heart_rate, current_speed):

    plan['week_1']['speed'] = current_speed * 1.2
