$$

def calculate_co_t(model, test_loader):
    model.eval()
    true_positives = 0
    false_negatives = 0
    with torch.no_grad():
        for inputs, labels in test_loader:
            outputs = model(inputs)
            predicted_labels = torch.round(torch.sigmoid(outputs))
            true_positives += (predicted_labels == labels).sum().item()
            false_negatives += ((predicted_labels != labels) & (labels == 1)).sum().item()
    
    co_t = true_positives / (true_positives + false_negatives)
    return co_t
