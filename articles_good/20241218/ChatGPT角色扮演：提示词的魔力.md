import requests

class ChatGPTClient:

    def ask_question(self, question):
        response = self.chatgpt.answer(question)
        return response

    def evaluate_answer(self, answer, rating):
        evaluation = self.chatgpt.evaluate_answer(answer, rating)
        return evaluation
