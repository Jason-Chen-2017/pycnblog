```python
from flask import Flask, request, jsonify
from model import train_model, generate_decision
from database import db, EthicsRule

app = Flask(__name__)

@app.route('/train', methods=['POST'])
def train():
    data = request.get_json()
    model = train_model(data['prompts'], data['data'])
    return jsonify({'status': 'success', 'model': model})

@app.route('/generate_decision', methods=['POST'])
def generate_decision():
    data = request.get_json()
    decision = generate_decision(data['model'], data['scenario'])
    return jsonify({'status': 'success', 'decision': decision})

if __name__ == '__main__':
    app.run(debug=True)
```

**model.py**：

```python
import tensorflow as tf
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Dense, LSTM

def train_model(prompts, data):
    # 创建模型
    model = Sequential([
        LSTM(50, activation='relu', input_shape=(None, data.shape[1])),
        Dense(1, activation='sigmoid')
    ])

    # 编译模型
    model.compile(optimizer='adam', loss='binary_crossentropy', metrics=['accuracy'])

    # 训练模型
    model.fit(data, prompts, epochs=10, batch_size=32)

    return model

def generate_decision(model, scenario):
    # 使用模型生成决策
    decision = model.predict(scenario)
    return decision
```

**database.py**：

```python
from flask_sqlalchemy import SQLAlchemy

db = SQLAlchemy()

def init_app(app):
    db.init_app(app)

class EthicsRule(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    prompt = db.Column(db.String(255))
    decision = db.Column(db.String(255))

    def __repr__(self):
        return f"EthicsRule('{self.prompt}', '{self.decision}')"
```
