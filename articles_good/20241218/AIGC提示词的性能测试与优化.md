### 系统核心实现源代码

以下是一个简单的AIGC提示词生成系统实现，包括用户输入处理、AIGC提示词生成、提示词筛选与排序等模块。

```python
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader
from transformers import GPT2Tokenizer, GPT2LMHeadModel

# 加载预训练模型
tokenizer = GPT2Tokenizer.from_pretrained('gpt2')
model = GPT2LMHeadModel.from_pretrained('gpt2')

# 设定参数
device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
model.to(device)
model.eval()

# 用户输入处理
def user_input_process(user_input):
    return tokenizer.encode(user_input, return_tensors='pt')

# AIGC提示词生成
def generate_prompt_word(input_ids):
    input_ids = input_ids.to(device)
    output = model.generate(input_ids, max_length=20, num_return_sequences=5)
    return tokenizer.decode(output[0], skip_special_tokens=True)

# 提示词筛选与排序
def filter_and_sort(prompt_words):
    # 根据语义和语法质量筛选和排序
    return sorted(prompt_words, key=lambda x: x['score'], reverse=True)

# 主函数
def main():
    # 读取用户输入
    user_input = input("请输入您的问题：")

    # 处理用户输入
    input_ids = user_input_process(user_input)

    # 生成AIGC提示词
    prompt_words = generate_prompt_word(input_ids)

    # 筛选与排序
    filtered_prompt_words = filter_and_sort(prompt_words)

    # 输出最终提示词
    print("生成的AIGC提示词：")
    for word in filtered_prompt_words:
        print(word)

# 运行主函数
if __name__ == '__main__':
    main()
```
