```python
from scipy.optimize import minimize

# 假设输入数据为哈勃常数H0和宇宙背景辐射强度TB
H0 = 70
TB = 2.725

# 模型构建函数
def model(params, H0, TB):
    h, Omega_m = params
    Omega_Lambda = 1 - Omega_m
    H = H0 * np.sqrt(Omega_m * np.exp(-3 * (1 - Omega_Lambda) * (TB / 2.725)**2)
    return H

# 参数优化函数
def optimize_model(H0, TB):
    initial_params = [0.7, 0.3]  # 初始参数
    result = minimize(lambda params: (model(params, H0, TB) - H0)**2, initial_params)
    return result.x

# 预测计算函数
def predict_critical_time(H0, TB):
    params = optimize_model(H0, TB)
    H = model(params, H0, TB)
    T_critical = -1 / (H * np.sqrt(Omega_m))
    return T_critical

# 输出结果
print(predict_critical_time(H0, TB))
```
