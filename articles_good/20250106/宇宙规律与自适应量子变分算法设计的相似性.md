```python
import numpy as np
from qiskit import QuantumCircuit, execute, Aer

# 初始化参数
theta = np.random.rand(4)

# 创建量子电路
qc = QuantumCircuit(4)

# 设置初始态
qc.h(0)
qc.cx(0, 1)
qc.cx(0, 2)
qc.cx(0, 3)

# 应用变分过程
for i in range(4):
    qc.rx(theta[i], i)

# 应用测量
qc.measure_all()

# 执行电路
backend = Aer.get_backend("qasm_simulator")
result = execute(qc, backend).result()

# 解析结果
counts = result.get_counts()
print(counts)
```
