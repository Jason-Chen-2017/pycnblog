   ```cpp
   #include <cuda_runtime.h>

   __global__ void vector_add(const float *A, const float *B, float *C, int n) {
       int index = threadIdx.x + blockIdx.x * blockDim.x;
       int stride = blockDim.x * gridDim.x;
       for (int i = index; i < n; i += stride)
           C[i] = A[i] + B[i];
   }

   int main() {
       float *h_A, *h_B, *h_C;
       float *d_A, *d_B, *d_C;
       int n = 1000000;

       // 分配内存
       h_A = (float *)malloc(n * sizeof(float));
       h_B = (float *)malloc(n * sizeof(float));
       h_C = (float *)malloc(n * sizeof(float));

       // 初始化数据
       for (int i = 0; i < n; i++) {
           h_A[i] = 1.0f;
           h_B[i] = 2.0f;
       }

       // 分配设备内存
       cudaMalloc(&d_A, n * sizeof(float));
       cudaMalloc(&d_B, n * sizeof(float));
       cudaMalloc(&d_C, n * sizeof(float));

       // 将数据从主机复制到设备
       cudaMemcpy(d_A, h_A, n * sizeof(float), cudaMemcpyHostToDevice);
       cudaMemcpy(d_B, h_B, n * sizeof(float), cudaMemcpyHostToDevice);

       // 设置并行执行的线程块和线程数
       int blockSize = 1024;
       int gridSize = (n + blockSize - 1) / blockSize;

       // 执行并行计算
       vector_add<<<gridSize, blockSize>>>(d_A, d_B, d_C, n);

       // 将计算结果从设备复制回主机
       cudaMemcpy(h_C, d_C, n * sizeof(float), cudaMemcpyDeviceToHost);

       // 清理内存
       free(h_A);
       free(h_B);
       free(h_C);
       cudaFree(d_A);
       cudaFree(d_B);
       cudaFree(d_C);

       // 输出结果
       printf("C[0] = %f\n", h_C[0]);

       return 0;
   }
   ```

#### 6.2 实例数学模型建立

##### 伊辛模型的基本原理

伊辛模型是一个离散的统计物理模型，描述了粒子在二维晶格中的相互作用。模型中的每个粒子可以处于两种状态之一（通常表示为+1和-1），并且粒子之间的相互作用能由以下公式给出：

\[ E = -J \sum_{\langle i,j \rangle} S_i S_j \]

其中，\( J \) 是相互作用强度，\( \langle i,j \rangle \) 表示相邻粒子对，\( S_i \) 是第 \( i \) 个粒子的状态。

##### 数学模型的建立

在这个实例中，我们使用以下数学模型来描述伊辛模型：

1. **能量函数**：

\[ E = -J \sum_{\langle i,j \rangle} S_i S_j \]

2. **配分函数**：

\[ Z = \sum_{所有配置} e^{-\beta E} \]

其中，\( \beta = 1/k_B T \)，\( k_B \) 是玻尔兹曼常数，\( T \) 是温度。

3. **临界指数**：

临界指数可以通过以下公式计算：

\[ \nu = \left( \frac{\partial \ln Z}{\partial \beta} \right)_{T} \]

##### 计算步骤

1. **初始化**：生成一个二维晶格，每个粒子随机初始化为+1或-1。

2. **采样**：从初始配置开始，进行一系列随机采样，包括：
   - 随机选择一个粒子，并将其状态进行翻转。
   - 计算状态翻转后的能量变化。
   - 根据能量变化和温度，使用玻尔兹曼分布决定是否接受状态翻转。

3. **统计**：对多次采样结果进行统计，计算系统的宏观性质，如能量、磁化强度等。

4. **收敛性判断**：通过计算结果的变化趋势，判断模拟是否达到稳定状态。
