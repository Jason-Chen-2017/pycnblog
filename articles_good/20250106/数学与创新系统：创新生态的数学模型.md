import numpy as np
from scipy.optimize import linprog

# 初始化参数
c = [-1, -1]  # 目标函数系数
A = [[1, 1], [2, 1]]  # 约束条件系数矩阵
b = [5, 7]  # 约束条件向量
x0 = [0, 0]  # 初始解

# 求解线性规划模型
result = linprog(c, A_ub=A, b_ub=b, x0=x0, method='highs')

# 输出优化结果
if result.success:
    print("Optimal solution:", result.x)
    print("Objective value:", -result.fun)
else:
    print("No optimal solution found")
