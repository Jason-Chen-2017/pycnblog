##### Python 源代码：

```python
import nltk
from nltk.tokenize import word_tokenize
from nltk.corpus import stopwords
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.model_selection import train_test_split
from sklearn.naive_bayes import MultinomialNB

# 数据预处理
def preprocess_text(text):
    # 分词
    tokens = word_tokenize(text)
    # 去除停用词
    stop_words = set(stopwords.words('english'))
    filtered_tokens = [token for token in tokens if token not in stop_words]
    # 转换为小写
    filtered_tokens = [token.lower() for token in filtered_tokens]
    return ' '.join(filtered_tokens)

# 提取特征
def extract_features(text):
    vectorizer = TfidfVectorizer()
    return vectorizer.fit_transform([text])

# 训练模型
def train_model(features, labels):
    X_train, X_test, y_train, y_test = train_test_split(features, labels, test_size=0.2, random_state=42)
    classifier = MultinomialNB()
    classifier.fit(X_train, y_train)
    return classifier, X_test, y_test

# 预测和评估
def evaluate_model(classifier, X_test, y_test):
    predictions = classifier.predict(X_test)
    accuracy = (predictions == y_test).mean()
    return accuracy

# 示例
text = "这是关于欧拉公式的诗歌，表达了数学的和谐之美。"
preprocessed_text = preprocess_text(text)
features = extract_features(preprocessed_text)
model, X_test, y_test = train_model(features, [1])  # 假设所有文本都是关于数学美学的
accuracy = evaluate_model(model, X_test, y_test)
print("模型准确率：", accuracy)
```

在这个示例中，我们首先对输入文本进行预处理，然后使用TF-IDF向量器提取特征，并使用朴素贝叶斯分类器进行训练。最后，我们评估模型的准确率，以确定数学美学元素在文学作品中的提取和匹配效果。

```python
# 文本预处理
def preprocess_text(text):
    # 分词
    tokens = nltk.word_tokenize(text)
    # 去停用词
    stop_words = set(nltk.corpus.stopwords.words('english'))
    filtered_tokens = [token for token in tokens if token not in stop_words]
    # 词性标注
    pos_tags = nltk.pos_tag(filtered_tokens)
    return pos_tags

# 数学美学元素提取
def extract_math_aesthetics(pos_tags):
    math_aesthetics = []
    for token, pos in pos_tags:
        if pos.startswith('NN') or pos.startswith('VB'):
            math_aesthetics.append(token)
    return math_aesthetics

# 呈现方式分析
def analyze_presentation(math_aesthetics, text):
    # 假设呈现方式为文本中出现的数学术语
    presentation_style = set([token for token in text.split() if token.isdigit() or token.startswith('[')])
    matched_elements = [element for element in math_aesthetics if element in presentation_style]
    return matched_elements

# 主函数
def main():
    text = "欧拉公式是数学中的经典，$e^{i\pi} + 1 = 0$ 令人赞叹。"
    pos_tags = preprocess_text(text)
    math_aesthetics = extract_math_aesthetics(pos_tags)
    matched_elements = analyze_presentation(math_aesthetics, text)
    print("数学美学元素：", math_aesthetics)
    print("匹配的呈现方式：", matched_elements)

if __name__ == "__main__":
    main()
```

最后，我们定义了一个呈现方式分析函数`analyze_presentation`，该函数根据文本中出现的数学术语和数字，确定数学美学的呈现方式。这里我们假设呈现方式为文本中出现的数学术语和数字，但实际上，呈现方式可以更加复杂和多样化。
