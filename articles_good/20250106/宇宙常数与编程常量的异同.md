```python
HUBBLE_CONSTANT = 70

# 模拟宇宙膨胀
def expand_universe(time):
    # 使用哈勃定律计算膨胀距离
    distance = HUBBLE_CONSTANT * time
    return distance

# 计算膨胀速度
def expansion_speed(time):
    distance = expand_universe(time)
    speed = distance / time
    return speed

# 示例：模拟宇宙膨胀

distance = expand_universe(time)
speed = expansion_speed(time)

print(f"宇宙膨胀速度: {speed} 光年/年")
```
