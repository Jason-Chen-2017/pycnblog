class ZeroShotCoTModel(nn.Module):
    def __init__(self, embed_size, num_classes):
        super(ZeroShotCoTModel, self).__init__()

        self.fc = nn.Linear(embed_size, num_classes)
    
    def forward(self, x):

        x = self.fc(x)
        return x
