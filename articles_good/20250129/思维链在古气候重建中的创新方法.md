```python
import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestRegressor
from sklearn.model_selection import train_test_split
from sklearn.metrics import mean_squared_error

def integrate_data(data1, data2):
    """
    数据整合函数，将两个数据源进行整合。
    """
    # 数据清洗
    data1.dropna(inplace=True)
    data2.dropna(inplace=True)
    
    # 数据拼接
    integrated_data = pd.merge(data1, data2, on='月份')
    
    return integrated_data

def extract_features(data):
    """
    特征提取函数，从数据中提取与气温变化相关的特征。
    """
    # 特征工程
    data['历史气温变化率'] = data['历史气温'].diff().fillna(0)
    data['现代气温变化率'] = data['现代气温'].diff().fillna(0)
    data['地质特征'] = data['地质数据']
    
    # 选择特征
    features = data[['年份', '历史气温变化率', '现代气温变化率', '地质特征']]
    
    return features

def train_model(features, labels):
    """
    模型训练函数，使用提取的特征训练预测模型。
    """
    # 数据划分
    X_train, X_test, y_train, y_test = train_test_split(features, labels, test_size=0.2, random_state=42)
    
    # 模型训练
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    
    return model

def predict(model, features):
    """
    预测函数，使用训练好的模型进行预测。
    """
    predictions = model.predict(features)
    
    return predictions

# 数据整合
data1 = pd.read_csv('历史气温数据.csv')
data2 = pd.read_csv('现代气温数据.csv')
integrated_data = integrate_data(data1, data2)

# 特征提取
features = extract_features(integrated_data)

# 数据预处理
labels = features['预测气温']
features.drop('预测气温', axis=1, inplace=True)

# 模型训练
model = train_model(features, labels)

# 预测结果
predictions = predict(model, features)

# 评估模型
mse = mean_squared_error(labels, predictions)
print("均方误差：", mse)
```
