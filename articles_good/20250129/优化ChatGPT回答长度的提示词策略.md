import openai

# 预处理函数
def preprocess(text):
    # 去除停用词
    stop_words = set(["a", "an", "the", "and", "or", "but", "if", "then", "else"])
    words = text.split()
    filtered_words = [word for word in words if word.lower() not in stop_words]
    return " ".join(filtered_words)

# 预测模型函数
def predict_length(text):
    # 这里使用简单的统计方法预测文本长度
    return len(text.split())

def adjust_parameters(length, max_length):
    # 如果长度超过最大长度，降低模型温度以生成更短的回答
    if length > max_length:
        return 0.5  # 降低温度
    else:
        return 1.0  # 保持原温度

# 生成回答函数
def generate_response(prompt, max_length=100):
    prompt = preprocess(prompt)
    length = predict_length(prompt)
    temperature = adjust_parameters(length, max_length)
    
    response = openai.Completion.create(
        engine="text-davinci-002",
        prompt=prompt,
        temperature=temperature,
        max_tokens=max_length
    )
    return response.choices[0].text.strip()

prompt = "请告诉我关于人工智能的历史。"
response = generate_response(prompt)
