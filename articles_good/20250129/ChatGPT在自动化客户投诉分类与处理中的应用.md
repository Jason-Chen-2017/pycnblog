```python
import torch
from transformers import GPT2Tokenizer, GPT2LMHeadModel
from transformers import pipeline

# 初始化模型
tokenizer = GPT2Tokenizer.from_pretrained("gpt2")
model = GPT2LMHeadModel.from_pretrained("gpt2")

def classify_complaint(complaint):
    # 对投诉文本进行预处理
    inputs = tokenizer.encode(complaint, return_tensors='pt')
    # 使用模型进行分类
    outputs = model(inputs)
    # 解析分类结果
    predicted_labels = torch.argmax(outputs.logits, dim=-1)
    labels = tokenizer.decode(predicted_labels, skip_special_tokens=True)
    return labels

# 生成回复
def generate_response(complaint):
    # 对投诉文本进行预处理
    inputs = tokenizer.encode(complaint, return_tensors='pt')
    # 使用模型生成回复
    response = generator(inputs, max_length=50, num_return_sequences=1)
    # 解析回复文本
    response_text = response[0]["generated_text"]
    return response_text

# 处理投诉
def process_complaint(complaint):
    # 分类投诉
    category = classify_complaint(complaint)
    # 生成回复
    response = generate_response(complaint)
    return category, response

# 示例
complaint = "我昨天在你们平台上购买了一部手机，但是今天收到手机后，发现手机屏幕有裂痕，无法正常使用。"

print("投诉类别：", category)
print("回复文本：", response)
```
