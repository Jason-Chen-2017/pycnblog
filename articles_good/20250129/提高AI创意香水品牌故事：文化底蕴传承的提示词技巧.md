**提示词生成模块**

```python
import nltk
from nltk.tokenize import word_tokenize
from nltk.corpus import stopwords

# 加载停用词表
nltk.download('stopwords')
stop_words = stopwords.words('english')

def generate_prompts(data):
    # 数据预处理
    tokens = word_tokenize(data)
    tokens = [token.lower() for token in tokens if token.isalnum()]
    tokens = [token for token in tokens if token not in stop_words]

    # 提取关键词
    keywords = []
    for token in tokens:
        if token not in keywords:
            keywords.append(token)

    # 生成提示词
    prompts = []
    for keyword in keywords:
        prompt = f"{keyword} in the context of the brand story"
        prompts.append(prompt)

    return prompts
```

**故事结构设计模块**

```python
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.cluster import KMeans

def design_story_structure(data):
    # 文本数据预处理
    texts = data.split('.')
    texts = [text.strip() for text in texts if text.strip()]

    # 提取特征
    vectorizer = TfidfVectorizer()
    X = vectorizer.fit_transform(texts)

    # 聚类分析
    kmeans = KMeans(n_clusters=3)
    kmeans.fit(X)

    # 分配故事结构
    story_structures = {}
    for i, label in enumerate(kmeans.labels_):
        if label not in story_structures:
            story_structures[label] = []
        story_structures[label].append(texts[i])
```

```python
from nltk.tag import pos_tag

def design_characters(data):
    # 数据预处理
    sentences = nltk.sent_tokenize(data)
    tagged_sentences = [pos_tag(sentence) for sentence in sentences]

    # 提取角色特征
    character_features = {}
    for sentence in tagged_sentences:
        for word, tag in sentence:
            if tag.startswith('NN'):  # 名词
                if word not in character_features:
                    character_features[word] = []
                character_features[word].append(sentence)

    # 生成角色设定
    characters = []
    for word, sentences in character_features.items():
        character = {
            'name': word,
            'description': ' '.join([sentence for sentence in sentences])
        }
        characters.append(character)

    return characters
```
