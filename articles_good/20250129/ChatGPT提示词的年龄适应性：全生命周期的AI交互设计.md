#### 5.2 系统核心实现

接下来，我们将展示系统核心实现的源代码。以下是一个简单的示例，用于演示如何使用ChatGPT进行交互，并根据用户年龄调整交互方式：

```python
import openai
import sys

# 设置OpenAI API密钥
openai.api_key = sys.getenv("OPENAI_API_KEY")

def generate_response(question, age):
    # 根据用户年龄调整ChatGPT的回答复杂度
    if age < 12:
        temperature = 0.5  # 简单回答
    elif age < 20:
        temperature = 0.7  # 中等复杂度
    else:
        temperature = 1.0  # 复杂回答

    # 使用ChatGPT生成回答
    response = openai.Completion.create(
        engine="text-davinci-002",
        prompt=question,
        max_tokens=50,
        n=1,
        stop=None,
        temperature=temperature
    )
    return response.choices[0].text.strip()

# 测试代码
if __name__ == "__main__":
    question = "什么是人工智能？"
    age = int(input("请输入您的年龄："))
    answer = generate_response(question, age)
    print(f"您的回答是：{answer}")
```
