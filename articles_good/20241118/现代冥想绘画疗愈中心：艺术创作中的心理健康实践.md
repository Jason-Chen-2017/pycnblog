```jsx
// 组件：MeditationPainting.js
import React, { useState } from 'react';
import Easel from 'easel.js';

function MeditationPainting() {
  const [easel, setEasel] = useState(null);

  const initEasel = () => {
    const canvas = document.getElementById('paintingCanvas');
    const easel = new Easel(canvas);
    setEasel(easel);
    easel.stage.clear();
  };

  const handleDraw = (event) => {
    if (easel) {
      easel.stage.draw();
      easel.registerMouseDrawEvent(event);
    }
  };

  React.useEffect(() => {
    initEasel();
  }, []);

  return (
    <div>
      <canvas id="paintingCanvas" width="800" height="600" onMouseDown={handleDraw} />
    </div>
  );
}

export default MeditationPainting;
```
