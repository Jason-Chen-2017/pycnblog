# 模拟柏格森的直觉主义时间序列模型
def bergson_intuition_sequence(start, end, step):
    times = np.arange(start, end, step)
    perceptions = np.random.rand(len(times))  # 随机生成感知值
    intuitions = perceptions  # 假设直觉与感知相同
    
    plt.plot(times, perceptions, label='Perception')
    plt.plot(times, intuitions, label='Intuition')
    plt.xlabel('Time')
    plt.ylabel('Value')
    plt.legend()
    plt.show()

# 柏格森的连续性与离散性模型
def bergson_continuity_discreteness(start, end, step):
    times = np.arange(start, end, step)
    flows = np.sin(times)  # 使用正弦函数模拟流
    changes = np.diff(flows)  # 计算瞬时变化率
    
    plt.plot(times[1:], changes)
    plt.xlabel('Time')
    plt.ylabel('Change Rate')
    plt.title('Bergson\'s Continuity and Discreteness')
    plt.show()

# 柏格森流与变化模型
def bergson_flow_change(start, end, step):
    times = np.arange(start, end, step)
    flows = np.sin(times)
    changes = np.gradient(flows, times)
    
    plt.plot(times, flows, label='Flow')
    plt.plot(times, changes, label='Change')
    plt.xlabel('Time')
    plt.ylabel('Value')
    plt.legend()
    plt.show()

# 柏格森实现
