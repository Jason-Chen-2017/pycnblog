- 匹配算法：利用欧氏距离、余弦相似度等算法进行特征匹配。

```python
import cv2
import face_recognition

def verify_user(face_image, database):
    # 采集人脸图像
    face_locations = face_recognition.face_locations(face_image)
    
    # 预处理人脸图像
    face_image = cv2.cvtColor(face_image, cv2.COLOR_BGR2GRAY)
    face_image = cv2.threshold(face_image, 0, 255, cv2.THRESH_BINARY)[1]
    
    # 特征提取
    face_encoding = face_recognition.face_encodings(face_image)[0]
    
    # 特征匹配
    matches = face_recognition.compare_faces(database, face_encoding)
    
    # 身份验证
    if True in matches:
        return "验证通过"
    else:
        return "验证失败"
```
