```python
        return guess_hash[:4] == '0000'

        self.unconfirmed_transactions.append(transaction)

        if not self.unconfirmed_transactions:
            return False

        last_block = self.chain[-1]

        proof = self.proof_of_work(last_block_hash)

            'previous_hash': last_block_hash,
```

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

import "@openzeppelin/contracts/token/ERC721/ERC721.sol";
import "@openzeppelin/contracts/token/ERC721/extensions/ERC721URIStorage.sol";

contract DigitalContent is ERC721, ERC721URIStorage {
    uint256 public contentCounter;

    constructor() ERC721("Digital Content", "DCT") {
        contentCounter = 0;
    }

    function mintContent(string memory tokenURI) public {
        uint256 tokenId = contentCounter;
        _mint(msg.sender, tokenId);
        _setTokenURI(tokenId, tokenURI);
        contentCounter++;
    }

    function transferContentOwnership(uint256 tokenId, address to) public {
        require(_isApprovedOrOwner(msg.sender, tokenId), "Not owner");
        _transfer(msg.sender, to, tokenId);
    }
}
```

**后端API（Express.js）**：

```javascript
        res.status(500).json({ message: "Error minting content", error: error.message });
    }
});

app.post("/transfer", async (req, res) => {
    const { tokenId, to } = req.body;
    try {
        const tx = await contract.transferContentOwnership(tokenId, to);
        await tx.wait();
        res.status(200).json({ message: "Content ownership transferred successfully", tx: tx.hash });
    } catch (error) {
        res.status(500).json({ message: "Error transferring content ownership", error: error.message });
    }
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
    console.log(`Server listening on port ${PORT}`);
});
```
