```python
import time
import random

def tomato_timer(work_time, break_time):
    print(f"开始工作，时间：{work_time}分钟")
    while work_time > 0:
        print(f"剩余工作时间：{work_time}分钟")
        time.sleep(60)  # 模拟工作
        work_time -= 1
    
    print(f"休息时间，时间：{break_time}分钟")
    while break_time > 0:
        print(f"剩余休息时间：{break_time}分钟")
        time.sleep(60)  # 模拟休息
        break_time -= 1
    
    print("训练完成！")

def main():
    work_time = 25  # 工作25分钟
    break_time = 5  # 休息5分钟
    tomato_timer(work_time, break_time)

if __name__ == "__main__":
    main()
```

- **函数`main`**：这是程序的主函数，它设置了工作时间和休息时间，并调用`tomato_timer`函数开始训练。
