### 附录A：相关性分析代码示例

```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error

# 示例数据加载
data = pd.DataFrame({
    'Price': [10, 12, 15, 14, 16],
    'Innovation': [0.8, 0.9, 0.7, 0.6, 0.9]
})

# 数据可视化
plt.scatter(data['Innovation'], data['Price'])
plt.xlabel('Innovation')
plt.ylabel('Price')
plt.title('Scatter Plot of Innovation vs Price')
plt.show()

# 回归分析
model = LinearRegression()
model.fit(data[['Innovation']], data['Price'])
predictions = model.predict(data[['Innovation']])
mse = mean_squared_error(data['Price'], predictions)
r_squared = model.score(data[['Innovation']], data['Price'])

print(f'回归系数：{model.coef_}')
print(f'截距：{model.intercept_}')
print(f'R²值：{r_squared}')
print(f'MSE：{mse}')
```
