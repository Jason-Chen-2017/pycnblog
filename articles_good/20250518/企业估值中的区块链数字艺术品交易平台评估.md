          require(index == tokenId, "Invalid token ID");
          Token storage token = tokens[index];
          require(msg.value >= token.price, "Insufficient funds");
          payable(token.owner).transfer(msg.value);
          token.owner = msg.sender;
          tokenIndex[msg.sender] = index;
      }
  }
