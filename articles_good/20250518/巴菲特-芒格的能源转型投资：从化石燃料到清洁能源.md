  # 数据清洗
  df.dropna(inplace=True)
  df = df[ df['Year'] > 2010 ]

  # 数据分析
  plt.figure(figsize=(10,6))
  plt.plot(df['Year'], df['Energy_Price'], label='Energy Price')
  plt.plot(df['Year'], df['Renewable_Portfolio'], label='Renewable Portfolio')
  plt.xlabel('Year')
  plt.ylabel('Value')
  plt.title('Energy Price and Renewable Portfolio')
  plt.legend()
  plt.show()
