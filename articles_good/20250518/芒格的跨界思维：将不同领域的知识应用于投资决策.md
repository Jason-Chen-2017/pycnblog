import matplotlib.pyplot as plt

data['日期'] = pd.to_datetime(data['日期'])
data.set_index('日期', inplace=True)

# 财务数据分析
financial_data = data['收入'].rolling(window=4).mean()
plt.plot(data.index, data['收入'], label='收入')
plt.plot(financial_data.index, financial_data, label='移动平均收入')
plt.legend()
plt.show()
