### 核心代码实现

```python
import pandas as pd

def calculate_risk_index(data):
    # 示例：计算风险指数
    alpha = 0.4
    beta = 0.3
    gamma = 0.3
    risk_index = alpha * data['市盈率'] + beta * data['市净率'] + gamma * data['财务异常指标']
    return risk_index

def trigger_alarm(risk_index, threshold=0.8):
    # 示例：预警触发条件
    if risk_index > threshold:
        return "触发预警"
    else:
        return "未触发预警"

# 示例运行
data = pd.DataFrame({
    '市盈率': [15, 20, 25, 30],
    '市净率': [1.5, 2.0, 2.5, 3.0],
    '财务异常指标': [0.2, 0.3, 0.4, 0.5]
})
risk_index = calculate_risk_index(data)
data['风险指数'] = risk_index
data['预警结果'] = data['风险指数'].apply(lambda x: trigger_alarm(x))

print(data)
```
