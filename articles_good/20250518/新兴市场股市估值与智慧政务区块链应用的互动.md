# MACD算法实现
def calculate_macd(close_prices, short_window=12, long_window=26):
    ema_short = []
    ema_long = []
    macd = []
    for price in close_prices:
        if len(ema_short) == 0:
            ema_short.append(price)
        else:
            ema_short.append(ema_short[-1] * (short_window - 1)/short_window + price * 1/short_window)
        if len(ema_long) == 0:
            ema_long.append(price)
        else:
            ema_long.append(ema_long[-1] * (long_window - 1)/long_window + price * 1/long_window)
        macd.append(ema_short[-1] - ema_long[-1])
    return macd
