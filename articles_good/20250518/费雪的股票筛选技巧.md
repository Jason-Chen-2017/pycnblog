import pandas as pd

def fisher_stock_screening(data, thresholds):
    # 计算综合评分
    weights = [0.4, 0.3, 0.2, 0.1]
    scores = []
    for index, row in data.iterrows():
        score = sum(weights[i] * row[指标i] for i in range(len(weights)))
        scores.append(score)
    
    # 条件筛选
    selected = data[(data['ROE'] >= thresholds['roe']) &
                    (data['毛利率'] >= thresholds['margin']) &
                    (data['ROA'] >= thresholds['roa'])]
    return selected

# 示例数据
