```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract Asset {
    mapping(address => uint) public balances;

    function transfer(address recipient, uint amount) external {
        require(balances[msg.sender] >= amount, "Insufficient balance");
        balances[recipient] += amount;
        balances[msg.sender] -= amount;
    }
}
```

```python
def register_user(address, name):
    tx = asset_contract.functions.register(address, name).transact()
    return w3.eth.waitForTransactionReceipt(tx)

def login_user(address, name):
    is_registered = asset_contract.functions.isRegistered(address).call()
    if not is_registered:
        raise ValueError('User not registered')
    return True

# 数字资产交易
```
