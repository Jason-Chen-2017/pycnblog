def value_investment_strategy(stocks):
    # 计算市盈率和市净率
    stocks['市盈率'] = stocks['股价'] / stocks['净利润']
    stocks['市净率'] = stocks['股价'] / stocks['净资产']
    # 选择市盈率和市净率低于行业平均水平的股票
    selected = stocks[(stocks['市盈率'] < stocks['市盈率'].mean()) & 
                     (stocks['市净率'] < stocks['市净率'].mean())]
    return selected

selected_stocks = value_investment_strategy(stocks)
print(selected_stocks)
