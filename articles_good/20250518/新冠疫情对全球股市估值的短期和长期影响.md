# 数据预处理
import pandas as pd
import numpy as np

# 模型训练
from sklearn.preprocessing import MinMaxScaler
from keras.models import Sequential
from keras.layers import LSTM, Dense

# 结果展示
import matplotlib.pyplot as plt

# 加载数据
data = pd.read_csv('stock_price.csv')
data = data[['Close']]
data = data.values

# 数据标准化
scaler = MinMaxScaler()
data_scaled = scaler.fit_transform(data)

# 划分训练集和测试集
train_size = int(len(data_scaled) * 0.8)
test_size = len(data_scaled) - train_size
train_data = data_scaled[:train_size]
test_data = data_scaled[train_size:]

# 构建数据集
def create_dataset(data, look_back=1):
    X, Y = [], []
    for i in range(len(data) - look_back):
        X.append(data[i:i+look_back])
        Y.append(data[i+look_back])
    return np.array(X), np.array(Y)

X_train, Y_train = create_dataset(train_data, look_back=10)
X_test, Y_test = create_dataset(test_data, look_back=10)

# 构建LSTM模型
model = Sequential()
model.add(LSTM(50, input_shape=(10,1)))
model.add(Dense(1))
model.compile(loss='mean_squared_error', optimizer='adam')

# 模型训练
model.fit(X_train, Y_train, epochs=50, batch_size=64, verbose=1)

# 模型预测
Y_train_pred = model.predict(X_train)
Y_test_pred = model.predict(X_test)

# 结果展示
plt.plot(Y_train, label='Train Real')
plt.plot(Y_train_pred, label='Train Predict')
plt.legend()
plt.show()

plt.plot(Y_test, label='Test Real')
plt.plot(Y_test_pred, label='Test Predict')
plt.legend()
plt.show()
```

### 6.3 代码应用解读与分析
- **数据预处理**
  - 使用pandas读取数据并进行标准化处理，确保数据的可比性。
  - 将数据划分为训练集和测试集，训练集用于模型训练，测试集用于模型验证。

- **模型训练**
  - 使用LSTM模型进行时间序列预测，LSTM能够捕捉数据的时序特征。
  - 模型经过50次迭代，每次迭代处理64个样本，逐步优化模型参数。

- **结果展示**
  - 使用matplotlib绘制实际值和预测值的对比图，直观展示模型的预测效果。
  - 通过对比图可以发现，模型在训练集和测试集上的预测效果较好，验证了模型的有效性。
