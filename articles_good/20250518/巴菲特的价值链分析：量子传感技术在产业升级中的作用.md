```python
import numpy as np

def quantum_state_measurement(rho, basis):
    # rho是量子态的密度矩阵，basis是测量基
    # 测量结果
    result = np.zeros_like(rho)
    for i in range(len(basis)):
        # 计算测量算符
        P = np.outer(basis[i], basis[i])
        # 计算测量结果
        result += np.trace(np.dot(P, rho)) * P
    return result
```

```python
import numpy as np

def measure_quantum_state(rho):
    # rho是量子态的密度矩阵
    eigenvalues, eigenvectors = np.linalg.eigh(rho)
    # 计算测量结果
    result = np.zeros_like(rho)
    for i in range(len(eigenvalues)):
        result += eigenvalues[i] * np.outer(eigenvectors[:,i], eigenvectors[:,i])
    return result
```
