```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract StockTrader {
    address payable public owner;
    uint256 public targetPrice;
    uint256 public currentPrice;

    constructor(uint256 _targetPrice) {
        owner = payable(msg.sender);
        targetPrice = _targetPrice;
    }

    function buyStock(uint256 _currentPrice) public {
        if (_currentPrice < targetPrice) {
            owner.transfer(msg.value);
        }
    }

    function sellStock(uint256 _currentPrice) public {
        if (_currentPrice > targetPrice) {
            owner.transfer(msg.value);
        }
    }
}
```

### 6.2.2 特价股票策略的实现代码

```python
import requests

def get_stock_price(symbol):
    response = requests.get(f"https://api.example.com/stock/{symbol}")
    return response.json()['price']

def execute_trade(symbol, price, target_price):
    if price < target_price:
        buy_stock(symbol, target_price)
    elif price > target_price * 1.1:
        sell_stock(symbol, target_price)

def main():
    symbol = "AAPL"
    target_price = 100
    price = get_stock_price(symbol)
    execute_trade(symbol, price, target_price)

if __name__ == "__main__":
    main()
```
