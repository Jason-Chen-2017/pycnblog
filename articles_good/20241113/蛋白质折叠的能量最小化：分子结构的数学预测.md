import numpy as np
import matplotlib.pyplot as plt

def calculate_energy(structure):
    # 假设能量函数为结构坐标的平方和
    energy = np.sum(np.square(structure))
    return energy

def energy_minimization(protein_sequence, max_iterations, initial_temp, cooling_rate):
    current_temp = initial_temp
    current_structure = np.random.rand(len(protein_sequence)) * 10
    best_structure = current_structure.copy()
    best_energy = calculate_energy(current_structure)

    for _ in range(max_iterations):
        new_structure = np.random.rand(len(protein_sequence)) * 10
        new_energy = calculate_energy(new_structure)

        if new_energy < best_energy:
            best_structure = new_structure.copy()
            best_energy = new_energy

        if np.random.rand() < np.exp((best_energy - new_energy) / current_temp):
            current_structure = new_structure.copy()

        current_temp *= cooling_rate

    return best_structure, best_energy

protein_sequence = [1, 2, 3, 4, 5]
max_iterations = 1000
initial_temp = 1000.0
cooling_rate = 0.99

best_structure, best_energy = energy_minimization(protein_sequence, max_iterations, initial_temp, cooling_rate)
print("Best Structure:", best_structure)
print("Best Energy:", best_energy)

plt.plot(range(max_iterations), best_energy)
plt.xlabel("Iterations")
plt.ylabel("Energy")
plt.title("Energy Minimization")
plt.show()
