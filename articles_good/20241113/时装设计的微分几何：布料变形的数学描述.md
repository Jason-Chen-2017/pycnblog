import numpy as np
import scipy.sparse
from scipy.sparse.linalg import spsolve
import matplotlib.pyplot as plt

# 定义参数
E = 200e9  # 弹性模量（Pa）
h = 0.01   # 厚度（m）
nu = 0.3   # 泊松比
k1 = E * h / (12 * (1 - nu**2))
k2 = 12 * E * h * (1 - nu**2) / (h**3)

# 定义网格
nodes = np.array([[0, 0], [1, 0], [1, 1], [0, 1]])
elements = np.array([[0, 1], [1, 2], [2, 3], [3, 0]])

# 构建稀疏矩阵
A = scipy.sparse.lil_matrix((4, 4), dtype=np.float64)
A[0, 0] = 2 * k1 + k2
A[0, 1] = k1
A[1, 0] = k1
A[1, 1] = 2 * k1 + k2
A[2, 2] = k2
A[3, 3] = k2

# 初始位移
b = np.array([0, 0, 0, 0])

# 求解方程
solution = spsolve(A.tocsc(), b)

# 可视化结果
plt.plot(nodes[:, 0], nodes[:, 1], 'ro-')
plt.plot(nodes[:, 0], nodes[:, 1] + solution, 'b--')
plt.axis('equal')
plt.show()
