```python
import numpy as np

def predict_trajectory(initial_state, time_step, duration):
    state = initial_state
    trajectory = []

    for _ in range(duration / time_step):
        state = apply_force(state, time_step)
        trajectory.append(state)

    return trajectory

def apply_force(state, time_step):
    acceleration = 9.8  # 重力加速度
    velocity = state[2]
    position = state[0]

    new_state = [
        position + velocity * time_step,
        velocity,
        acceleration
    ]

    return new_state
```
