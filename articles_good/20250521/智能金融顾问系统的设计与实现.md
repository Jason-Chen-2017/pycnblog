### 4.6.2 投资组合优化代码

```python
import numpy as np
from scipy.optimize import minimize

# 定义目标函数
def portfolio_variance(weights, covariance_matrix):
    return np.dot(weights.T, np.dot(covariance_matrix, weights))

# 定义约束条件
def constraint(weights):
    return np.sum(weights) - 1

# 定义权重非负约束
def non_negative(weights):
    return weights >= 0

# 优化问题
covariance_matrix = np.array([[2, 0.5], [0.5, 3]])
initial_guess = np.array([0.5, 0.5])

# 使用 scipy.optimize.minimize 进行优化
result = minimize(portfolio_variance, initial_guess, args=(covariance_matrix, ),
                 method='SLSQP', constraints={'type': 'eq', 'fun': constraint},
                 bounds=[(0, None), (0, None)])

print(result)
```
