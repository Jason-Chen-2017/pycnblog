```python
import openai

def zero_shot_cot(query):
    # 初始化API客户端
    client = openai.Client()
    # 调用零射击CoT API
    response = client.chat.completions.create(
        model="gpt-4",
        messages=[{
            "role": "user",
            "content": query
        }],
        temperature=0.7,
        max_tokens=1000
    )
    return response.choices[0].message.content

# 示例调用
query = "分析人工智能在医学领域的创新潜力"
result = zero_shot_cot(query)
print(result)
```
