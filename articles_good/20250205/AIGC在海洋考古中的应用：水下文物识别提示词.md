```python
def generate_prompt(image_data):
    # 提示词生成函数
    prompt = f"这是一个水下文物图像，包含{image_data['object']}的特征，" \
              f"请生成一个详细的提示词用于文物识别。"
    return prompt

def aigc_algorithm(image_path):
    import openai
    # 初始化OpenAI客户端
    client = openai.Client(api_key="your_api_key")
    # 生成提示词
    prompt = generate_prompt(image_data)
    response = client.chat.completions.create(
        model="gpt-4",
        messages=[{"role": "user", "content": prompt}]
    )
    return response.choices[0].message.content

# 示例调用
image_data = {"object": "陶罐"}
result = aigc_algorithm(image_data)
print(result)
```
