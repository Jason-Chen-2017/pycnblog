def train_step(images):
    noise = tf.random.normal([batch_size, 100])
    generated_images = generator(noise, training=True)
    real_output = discriminator(images, training=True)
    generated_output = discriminator(generated_images, training=True)

    # 生成器损失
    gen_loss = cross_entropy(tf.ones_like(generated_output), generated_output)
    # 判别器损失
    disc_loss = cross_entropy(tf.ones_like(real_output), real_output) + cross_entropy(tf.zeros_like(generated_output), generated_output)
    
    # 更新生成器参数
    gradients_of_generator = generator_optimizer.compute_gradients(gen_loss, generator.trainable_weights)
    generator_optimizer.apply_gradients(gradients_of_generator)
    
    # 更新判别器参数
    gradients_of_discriminator = discriminator_optimizer.compute_gradients(disc_loss, discriminator.trainable_weights)
    discriminator_optimizer.apply_gradients(gradients_of_discriminator)
