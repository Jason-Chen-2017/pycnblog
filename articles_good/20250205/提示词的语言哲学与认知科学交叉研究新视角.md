def generate_prompt(demand):
    # 选择提示词模板
    prompt_templates = {
        '语言哲学': '请描述{demand}在语言哲学中的意义。',
        '认知科学': '请从认知科学的角度分析{demand}。'

    # 生成提示词
    prompt = prompt_templates.get(demand, '请解释{demand}的概念。')

print(generate_prompt('提示词'))
