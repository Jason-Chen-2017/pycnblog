  import networkx as nx

  def compute_node_weights(G):
      weights = {}
      for node in G.nodes():
          weight = 0
          for neighbor in G.neighbors(node):
              weight += 1 / nx.shortest_path_length(G, node, neighbor)
          weights[node] = weight
      return weights

  def compute_relation_strength(G, weights):
      relation_strength = {}
      for edge in G.edges():
          i, j = edge
          common_neighbors = list(nx.common_neighbors(G, i, j))
          max_common = len(common_neighbors)
          if max_common == 0:
              strength = 0
          else:
              strength = len(common_neighbors) / max_common
          relation_strength[edge] = strength
      return relation_strength

  # 示例历史数据

  G.add_edge("person1", "event1")
  G.add_edge("person2", "event2")
  G.add_edge("event1", "event2")

  node_weights = compute_node_weights(G)
  relation_strength = compute_relation_strength(G, node_weights)

#### **3.2 数学模型与公式**
- **3.2.1 节点权重计算**
  $$
  w_i = \sum_{j \in N} \frac{1}{\text{distance}(i, j)}
  $$
  其中，$w_i$代表节点$i$的权重，$N$代表节点集合，$\text{distance}(i, j)$代表节点$i$和节点$j$之间的距离。
- **3.2.2 关系强度计算**
  $$
  r_{ij} = \frac{\text{common\_attributes}(i, j)}{\text{max\_common\_attributes}(i, j)}
  $$
  其中，$r_{ij}$代表节点$i$和节点$j$之间的关系强度，$\text{common\_attributes}(i, j)$代表节点$i$和$j$的共同属性数量，$\text{max\_common\_attributes}(i, j)$代表节点$i$和$j$的最大可能共同属性数量。
