```python
from transformers import GPT2Tokenizer, GPT2ModelWithLMHead
import numpy as np

def preprocess_data(data):
    tokenizer = GPT2Tokenizer.from_pretrained('gpt2')
    inputs = tokenizer(data, return_tensors='np')
    return inputs

def generate_treatment_plan(prompt):
    model = GPT2ModelWithLMHead.from_pretrained('gpt2')
    inputs = preprocess_data(prompt)
    outputs = model.generate(**inputs, max_length=100)
    return tokenizer.decode(outputs[0], skip_special_tokens=True)

if __name__ == "__main__":
    raw_prompt = "基于基因数据，生成个性化治疗方案。"
    treatment_plan = generate_treatment_plan(raw_prompt)
    print(treatment_plan)
```
