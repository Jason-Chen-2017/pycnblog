class WasteCollectionSystem:
    def __init__(self):
        self.waste = []

    def collect_waste(self, waste):
        self.waste.append(waste)

    def preprocess_waste(self):
        # 预处理操作，例如筛除金属和塑料
        self.waste = [w for w in self.waste if not self.is_non_degradable(w)]

    def is_non_degradable(self, waste):
        # 判断垃圾是否可降解
        return 'metal' in waste or 'plastic' in waste

    def compost_waste(self):
        # 堆肥化操作
        self.waste = self.waste + ['compost']

    def get_compost_product(self):
        return self.waste[-1]

# 使用示例

system.collect_waste('metal can')
system.preprocess_waste()
system.compost_waste()

**详细讲解**

### 数学模型和数学公式

我们使用一个简单的数学模型来描述厨余堆肥的完成度，该模型采用了一个Sigmoid函数：

$$C = \frac{1}{1 + e^{-\beta \cdot (x - \theta)}}$$

其中：

- $C$：堆肥化完成度，取值范围在0到1之间，表示堆肥的成熟程度。
- $\beta$：学习率，用于调整堆肥化过程的敏感度，其值越大，堆肥化过程的响应就越快。
- $x$：垃圾的特性值，可以是有机物含量、水分含量等与堆肥化过程相关的参数。
- $\theta$：阈值，用于判断垃圾是否适合堆肥化，当$x > \theta$时，垃圾适合堆肥化。

### 详细讲解

这个数学模型中的Sigmoid函数是一个常见的非线性激活函数，它在机器学习和神经网络中广泛应用。在厨余堆肥的上下文中，$C$表示堆肥化过程的完成度，$\beta$是调节堆肥化速度的参数，$x$是垃圾的特性值，反映了厨余垃圾的适宜度，而$\theta$是一个预先设定的阈值。

- 当$x$接近$\theta$时，$C$的值接近0.5，表示堆肥化过程尚未完全进行。
- 当$x$远大于$\theta$时，$C$趋近于1，表示堆肥化过程已经完成。
- 当$x$远小于$\theta$时，$C$趋近于0，表示该垃圾不适合进行堆肥化。

### 举例说明

假设我们有一个厨余垃圾样本，其有机物含量$x$为0.8，阈值$\theta$为0.5，学习率$\beta$为0.1。我们可以将这些值代入上述公式，计算堆肥化完成度$C$：

$$C = \frac{1}{1 + e^{-0.1 \cdot (0.8 - 0.5)}} = \frac{1}{1 + e^{-0.1 \cdot 0.3}} \approx 0.9$$

import numpy as np
import pandas as pd

class WasteCollectionSystem:
    def __init__(self):
        self.waste_data = pd.DataFrame(columns=['waste_type', 'waste_quantity'])

    def collect_waste(self, waste_type, waste_quantity):
        new_data = pd.DataFrame([[waste_type, waste_quantity]], columns=['waste_type', 'waste_quantity'])
        self.waste_data = pd.concat([self.waste_data, new_data])

    def preprocess_waste(self):
        # 去除不可降解的垃圾
        self.waste_data = self.waste_data[self.waste_data['waste_type'].str.contains('degradable')]

    def compost_waste(self):
        # 堆肥化处理
        self.waste_data['compost_quantity'] = self.waste_data['waste_quantity']

    def report_compost(self):
        # 输出堆肥数据
        print(self.waste_data[['waste_type', 'waste_quantity', 'compost_quantity']])

# 实例化厨余堆肥系统
system = WasteCollectionSystem()

system.collect_waste('metal can', 5)
system.preprocess_waste()
system.compost_waste()
system.report_compost()
