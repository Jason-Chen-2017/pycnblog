```python
from . import models as user_models
from . import models as workout_models

    workout = models.ForeignKey(workout_models.Workout, on_delete=models.CASCADE)

    fatigue_level = models.FloatField()
    duration = models.IntegerField()
```

```python
    username='alice',
    password='password',
    name='Alice Johnson',

)

# 创建运动计划
workout = Workout.objects.create(
    user=alice,
    start_date=datetime.date.today(),
    end_date=datetime.date.today() + datetime.timedelta(days=30)
)

# 添加运动记录
for day in range(30):
    DataRecord.objects.create(
        user=alice,
        workout=workout,
        date=datetime.date.today() + datetime.timedelta(days=day),
        fatigue_level=10,
        duration=60
    )
```

- **运动记录**：在运动计划期间，每天创建了一个新的运动记录，记录了疲劳度和运动时间。
