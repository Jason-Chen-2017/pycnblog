import librosa
import numpy as np

def generate_meditation_music(duration=300):
    """
    生成冥想音乐，参数duration为音乐时长（秒）
    """
    # 采样频率设置为44100Hz，与音频CD相同
    sr = 44100
    # 生成一个频率为440Hz的正弦波
    t = np.linspace(0, duration, sr * duration)
    freq = 440
    sine_wave = np.sin(2 * np.pi * freq * t)
    
    # 添加噪声
    noise = np.random.normal(0, 0.05, size=sine_wave.shape)
    meditation_music = sine_wave + noise
    
    # 保存音乐文件
    librosa.output.write_wav('meditation_music.wav', meditation_music, sr)
    
    return meditation_music

# 生成3分钟冥想音乐
generate_meditation_music(180)
```

def sound_healing_algorithm(audio_data, healing_freq=440):
    """
    应用声音疗愈算法，参数healing_freq为疗愈频率（Hz）
    """
    # 提取声音特征
    freq_spectrum = librosa.stft(audio_data)
    magnitude_spectrum = np.abs(freq_spectrum)
    
    # 应用滤波器，保留特定频率范围
    f = np.fft.rfftfreq(len(audio_data), 1/sr)
    filtering_mask = (f >= healing_freq - 5) & (f <= healing_freq + 5)
    magnitude_spectrum[filtering_mask] = 0
    
    # 反变换
    audio_data_healed = np.fft.irfft(magnitude_spectrum)
    
    return audio_data_healed

# 加载音频文件
audio_data = librosa.load('meditation_music.wav')[0]

# 应用声音疗愈算法
healed_audio_data = sound_healing_algorithm(audio_data, healing_freq=440)

# 保存疗愈后的音乐文件
librosa.output.write_wav('healed_meditation_music.wav', healed_audio_data, sr)
```
