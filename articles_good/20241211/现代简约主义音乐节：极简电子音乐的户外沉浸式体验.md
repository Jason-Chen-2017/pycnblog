```python
import pygame

def render_scene(scene_model_path, rendered_image_path):
    pygame.init()
    screen = pygame.display.set_mode((800, 600))
    screen.fill((255, 255, 255))

    # 加载场景模型
    with open(scene_model_path, "r") as f:
        scene_model = f.read()

    # 渲染场景
    for line in scene_model.split("\n"):
        if line.startswith("line"):
            x1, y1, x2, y2 = map(int, line.split()[1:])
            pygame.draw.line(screen, (0, 0, 0), (x1, y1), (x2, y2), 1)

    pygame.image.save(screen, rendered_image_path)
    pygame.quit()

render_scene("scene_model.txt", "rendered_image.png")
```
