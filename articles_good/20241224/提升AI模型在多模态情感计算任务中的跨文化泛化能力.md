import torch
import torch.nn as nn
import torch.optim as optim

# 定义模型
class MultiModalCNNRNN(nn.Module):
    def __init__(self, input_dim, hidden_dim, output_dim):
        super(MultiModalCNNRNN, self).__init__()
        self.cnn = nn.Sequential(
            nn.Conv2d(input_dim[2], hidden_dim[2], kernel_size=3, padding=1),
            nn.ReLU(),
            nn.MaxPool2d(kernel_size=2, stride=2)
        )
        self.rnn = nn.Sequential(
            nn.LSTM(hidden_dim[2] * hidden_dim[1], hidden_dim[3], num_layers=2, batch_first=True),
            nn.Linear(hidden_dim[3], output_dim)
        )
    
    def forward(self, x):
        x = self.cnn(x)
        x = x.view(x.size(0), -1)
        x = self.rnn(x)
