1. **自洽性指标（Consistency Index）**：

### 4.2 系统核心实现源代码

以下是 Self-Consistency CoT 系统的核心实现代码，包括数据预处理、模型训练、自洽性检验和动态调整。

```python
import numpy as np
import pandas as pd
from sklearn.linear_model import LinearRegression
from sklearn.model_selection import train_test_split
from sklearn.metrics import mean_squared_error

# 数据预处理
def preprocess_data(data):
    # 清洗数据
    data = data[data['price'].notnull()]
    # 归一化数据
    mean = np.mean(data['price'])
    std = np.std(data['price'])
    data['price_normalized'] = (data['price'] - mean) / std
    return data

# 模型训练
def train_model(data):
    # 划分训练集和测试集
    X_train, X_test, y_train, y_test = train_test_split(data[['price_normalized']], data['price'], test_size=0.2, random_state=42)
    # 创建线性回归模型
    model = LinearRegression()
    # 训练模型
    model.fit(X_train, y_train)
    # 评估模型
    y_pred = model.predict(X_test)
    mse = mean_squared_error(y_test, y_pred)
    return model, mse

# 自洽性检验
def evaluate_model(model, data):
    y_pred = model.predict(data[['price_normalized']])
    CI = mean_squared_error(data['price'], y_pred)
    return CI

# 动态调整
def adjust_model(model, data, alpha):
    CI = evaluate_model(model, data)
    delta_theta = alpha * (CI - np.mean(CI_history))
    CI_history.append(CI)
    model.coef_ += delta_theta
    return model

# 示例数据
data = pd.DataFrame({
    'timestamp': pd.date_range(start='2023-01-01', end='2023-01-10', freq='H'),
    'price': np.random.rand(10) * 100
})

# 数据预处理
data_processed = preprocess_data(data)

# 模型训练
model, mse = train_model(data_processed)
print(f"Initial MSE: {mse}")

# 自洽性检验与动态调整
alpha = 0.01
CI_history = []
for i in range(5):
    CI = evaluate_model(model, data_processed)
    model = adjust_model(model, data_processed, alpha)
    print(f"Step {i}: CI = {CI}")

# 输出调整后的模型参数
print(model.coef_)
```
