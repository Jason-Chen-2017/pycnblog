```python
from sklearn.metrics import accuracy_score, precision_score, recall_score

def evaluate_prompt(prompt, actual_answers, model_predictions):
    """
    自动评估提示词的有效性
    :param prompt: 提示词
    :param actual_answers: 实际答案
    :param model_predictions: 模型预测结果
    :return: 评估结果（准确率、精确度、召回率）
    """
    correct_predictions = [answer == pred for answer, pred in zip(actual_answers, model_predictions)]
    accuracy = accuracy_score(correct_predictions)
    precision = precision_score(correct_predictions)
    recall = recall_score(correct_predictions)
    return accuracy, precision, recall
```
