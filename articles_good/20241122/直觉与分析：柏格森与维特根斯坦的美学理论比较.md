import PIL
from PIL import Image
from nltk.corpus import stopwords
from nltk.tokenize import word_tokenize

def analyze_intuition(image_path):
    # 加载图像
    image = PIL.Image.open(image_path)
    
    # 转换为灰度图像
    gray_image = image.convert('L')
    
    # 分割图像
    pixels = list(gray_image.getdata())
    
    # 提取直觉元素
    intuition_elements = [pixel for pixel in pixels if pixel > 128]
    
    # 计算直觉元素的比例
    intuition_ratio = len(intuition_elements) / len(pixels)

    text = [word for word in text if word not in stopwords.words('english')]
    
    # 分析文字中的直觉元素
    intuition_text_ratio = len([word for word in text if 'intuition' in word]) / len(text)
    
    # 返回分析结果
    return intuition_ratio, intuition_text_ratio

    intuition_text_ratio = len([word for word in text if 'intuition' in word]) / len(text)
