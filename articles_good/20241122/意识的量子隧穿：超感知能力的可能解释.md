    # 创建数据结构来存储实验结果
    results = {"samples": [], "conditions": [], "repeats": []}
    # 遍历受试者
    for sample in range(n_samples):
        # 遍历实验条件

                # 记录实验结果
                results["samples"].append(sample)
                results["conditions"].append(condition)
                results["repeats"].append(repeat)
    return results
