import string

# 生成短信验证码
def generate_sms_code():
    return ''.join(random.choices(string.digits, k=4))

# 用户登录验证
def login_verification(username, password, sms_code):
    # 假设用户数据库中存储的用户名和密码为'alice'和'1234'，短信验证码正确为'5678'
    stored_username = 'alice'
    stored_password = '1234'
    correct_sms_code = '5678'
    
    if username == stored_username and password == stored_password:
        if sms_code == correct_sms_code:
            return "登录成功！"
        else:
            return "短信验证码错误！"
    else:
        return "用户名或密码错误！"

# 用户输入登录信息
