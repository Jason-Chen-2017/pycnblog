V = 0.1 \cdot 10000 + 0.2 \cdot 50000 + 0.3 \cdot 20000 = 11000 + 10000 + 6000 = 27000

from flask import Flask, request, jsonify
from flask_mysqldb import MySQL
import config

app = Flask(__name__)
app.config['MYSQL_HOST'] = config.MYSQL_HOST
app.config['MYSQL_USER'] = config.MYSQL_USER
app.config['MYSQL_PASSWORD'] = config.MYSQL_PASSWORD
app.config['MYSQL_DB'] = config.MYSQL_DB

mysql = MySQL(app)

@app.route('/register', methods=['POST'])
def register():
    cursor = mysql.connection.cursor()
    user_data = request.get_json()
    query = "INSERT INTO users (username, password) VALUES (%s, %s)"
    cursor.execute(query, (user_data['username'], user_data['password']))
    mysql.connection.commit()
    return jsonify({'message': 'User registered successfully'})

@app.route('/assets', methods=['POST'])
def add_asset():
    cursor = mysql.connection.cursor()
    asset_data = request.get_json()
    query = "INSERT INTO assets (user_id, name, value, type) VALUES (%s, %s, %s, %s)"
    cursor.execute(query, (asset_data['user_id'], asset_data['name'], asset_data['value'], asset_data['type']))
    mysql.connection.commit()
    return jsonify({'message': 'Asset added successfully'})

@app.route('/assets/evaluate', methods=['GET'])
def evaluate_assets():
    cursor = mysql.connection.cursor()
    query = "SELECT * FROM assets WHERE user_id = %s"
    cursor.execute(query, (request.args.get('user_id'),))
    assets = cursor.fetchall()
    total_value = sum(asset['value'] for asset in assets)
    return jsonify({'total_value': total_value})

if __name__ == '__main__':
    app.run(debug=True)
