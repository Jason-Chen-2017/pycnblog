```python
import pandas as pd

# 情绪日记记录
def record_emotion(entry):
    emotion_journal = pd.DataFrame({'Date': [entry['Date']], 'Emotion': [entry['Emotion']], 'Description': [entry['Description']]})
    return emotion_journal

# 情绪日记评估
def assess_emotion_journal(journal):
    total_entries = len(journal)
    positive_entries = len(journal[journal['Emotion'] == 'Positive'])
    negative_entries = len(journal[journal['Emotion'] == 'Negative'])
    
    emotion_score = (positive_entries / total_entries) * 100
    
    return emotion_score

# 示例数据
entry1 = {'Date': '2023-10-01', 'Emotion': 'Positive', 'Description': 'Had a great day at work.'}
entry2 = {'Date': '2023-10-02', 'Emotion': 'Negative', 'Description': 'Had a stressful day due to project deadlines.'}

# 记录情绪日记
journal = record_emotion(entry1)
journal = journal.append(record_emotion(entry2))

# 评估情绪日记
emotion_score = assess_emotion_journal(journal)
print(f"Emotion Score: {emotion_score:.2f}%")
```
