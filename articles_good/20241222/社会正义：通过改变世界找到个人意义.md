```python
# 初始化节点属性
node_features = {
    'fairness': 0.5,
    'equity': 0.5,
    'moral_responsibility': 0.5,
    'legal_protection': 0.5

# 更新节点属性
def update_node_features(node_features, neighbors, alpha_values):
    for feature, alpha in zip(node_features.keys(), alpha_values):
        for neighbor in neighbors:
            node_features[feature] += alpha * (neighbor[feature] - node_features[feature])

# 示例邻居节点
neighbors = [

]

# 更新权重
alpha_values = [0.1, 0.1, 0.1, 0.1]

# 更新节点属性
update_node_features(node_features, neighbors, alpha_values)

print(node_features)
```

**后端Flask应用**：

```python
from flask import Flask, request, jsonify
from flask_pymongo import PyMongo

app = Flask(__name__)
app.config["MONGO_URI"] = "mongodb://localhost:27017/justice-community"
mongo = PyMongo(app)

@app.route('/api/posts', methods=['GET', 'POST'])
def handle_posts():
    if request.method == 'POST':
        post_data = request.json
        mongo.db.posts.insert_one(post_data)
        return jsonify({"status": "success", "message": "Post created."}), 201

    posts = list(mongo.db.posts.find())
    return jsonify(posts)

if __name__ == '__main__':
    app.run(debug=True)
```
