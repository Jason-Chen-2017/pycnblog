def collaborative_filtering(users, ratings, user_id):
    # 计算用户user_id与其他用户的相似度
    similarity_scores = []
    for other_user_id in users:
        if other_user_id != user_id:
            similarity = cosine_similarity(users[user_id], users[other_user_id])
            similarity_scores.append((other_user_id, similarity))
    
    # 根据相似度计算用户评分预测
    predicted_ratings = {}
    for other_user_id, similarity in similarity_scores:
        predicted_rating = similarity * (ratings[other_user_id] - users[other_user_id][user_id])
        predicted_ratings[other_user_id] = predicted_rating
    
    # 返回推荐结果
