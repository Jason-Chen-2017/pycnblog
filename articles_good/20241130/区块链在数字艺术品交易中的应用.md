contract DigitalArt {
    struct Art {
        string name;
        string imageHash;
        address owner;
        uint256 price;
    }

    mapping(uint256 => Art) public artworks;
    uint256 public artCounter;

    function createArt(string memory _name, string memory _imageHash, uint256 _price) public {
        artworks[artCounter] = Art(_name, _imageHash, msg.sender, _price);
        artCounter++;

        require(msg.value >= art.price, "Insufficient payment");
        require(art.owner != address(0), "Art does not exist");

        artworks[_artId] = art;
        payable(art.owner).transfer(msg.value);
    }
