  pragma solidity ^0.8.0;

   contract IPIntellectualProperty {
       struct License {
           address licensee;
           uint256 licenseId;
           string licenseType;
           uint256 expirationDate;
       }

       mapping(uint256 => License[]) private licenses;

       function createLicense(
           uint256 intellectualPropertyId,
           address licensee,
           string memory licenseType,
           uint256 expirationDate
       ) public {
           licenses[intellectualPropertyId].push(
               License({
                   licensee: licensee,
                   licenseId: licenses[intellectualPropertyId].length,
                   licenseType: licenseType,
                   expirationDate: expirationDate
               })
           );
       }

       function getLicense(
           uint256 intellectualPropertyId,
           uint256 licenseId
       ) public view returns (License memory) {
           return licenses[intellectualPropertyId][licenseId - 1];
       }

       function transferLicense(
           uint256 intellectualPropertyId,
           uint256 licenseId,
           address newLicensee
       ) public {
           License memory license = getLicense(intellectualPropertyId, licenseId);
           require(msg.sender == license.licensee, "Only the licensee can transfer the license");
           licenses[intellectualPropertyId][licenseId - 1].licensee = newLicensee;
       }
   }
