import matplotlib.pyplot as plt
import numpy as np

# 定义陶器尺寸
ceramic_toys = [
    {'name': '小碗', 'diameter': 5, 'height': 3},
    {'name': '大碗', 'diameter': 10, 'height': 6}
]

# 计算稳定性
def calculate_stability(toy):
    weight = toy['diameter'] * toy['height']
    base_area = np.pi * (toy['diameter'] / 2) ** 2
    return weight / base_area

# 绘制稳定性图
def plot_stabilities(toys):
    stabilities = [calculate_stability(toy) for toy in toys]
    plt.bar([toy['name'] for toy in toys], stabilities)
    plt.xlabel('Toy Name')
    plt.ylabel('Stability')
    plt.title('Stability of Ceramic Toys')
    plt.show()

plot_stabilities(ceramic_toys)
```

import matplotlib.pyplot as plt
import numpy as np

# 定义机器人尺寸
robot_toys = [
    {'name': '小型机器人', 'width': 5, 'height': 10},
    {'name': '中型机器人', 'width': 10, 'height': 20},
    {'name': '大型机器人', 'width': 15, 'height': 30}
]

# 计算稳定性
def calculate_stability(toy):
    weight = toy['width'] * toy['height']
    base_area = np.pi * (toy['width'] / 2) ** 2
    return weight / base_area

# 绘制稳定性图
def plot_stabilities(toys):
    stabilities = [calculate_stability(toy) for toy in toys]
    plt.bar([toy['name'] for toy in toys], stabilities)
    plt.xlabel('Toy Name')
    plt.ylabel('Stability')
    plt.title('Stability of Robot Toys')
    plt.show()

plot_stabilities(robot_toys)
```

from tensorflow.keras.models import Sequential
