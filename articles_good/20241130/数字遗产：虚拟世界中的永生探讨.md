import numpy as np

# 定义一个简单的神经网络
class SimpleNeuralNetwork:
    def __init__(self):
        # 初始化权重和偏置
        self.weights = np.random.randn(2, 1)
        self.bias = np.random.randn(1)

    def forward(self, x):
        # 前向传播
        return np.dot(x, self.weights) + self.bias

    def backward(self, x, y, output):
        # 反向传播
        output_error = y - output
        dweights = np.dot(x.T, output_error)
        dbias = np.sum(output_error)
        return dweights, dbias

# 创建一个神经网络实例
nn = SimpleNeuralNetwork()

# 生成一些随机输入数据
x = np.array([[1], [2], [3], [4]])
y = np.array([[0], [1], [0], [1]])

# 前向传播
output = nn.forward(x)

# 计算误差
error = y - output

# 反向传播
dweights, dbias = nn.backward(x, y, output)

print("输出结果：", output)
print("误差：", error)
print("权重更新：", dweights)
print("偏置更新：", dbias)
