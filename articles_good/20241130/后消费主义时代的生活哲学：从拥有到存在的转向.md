    inner_experience = values[1]

    social_participation = values[2]

    total_value = 0.5 * material_pursuit + 0.3 * inner_experience + 0.2 * social_participation

    punishment = 0
    if inner_experience < social_participation:
        punishment += (social_participation - inner_experience) * 0.1
    if material_pursuit > inner_experience:
        punishment += (material_pursuit - inner_experience) * 0.1
    
    total_value -= punishment
    
    return total_value
```
