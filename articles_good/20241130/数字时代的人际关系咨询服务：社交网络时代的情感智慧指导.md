```python
import nltk
from nltk.sentiment import SentimentIntensityAnalyzer

# 初始化情感分析器
sia = SentimentIntensityAnalyzer()

def analyze_sentiment(text):
    """

    """
    scores = sia.polarity_scores(text)
    return scores
```

### 源代码实现

```python
# 导入必需的库
import nltk
from nltk.corpus import movie_reviews
from nltk.classify import NaiveBayesClassifier

def extract_features(words):
    return dict([(word, True) for word in words])

def build_model():
    documents = [(list(movie_reviews.words(fileid)), category) 
                 for category in movie_reviews.categories() 
                 for fileid in movie_reviews.fileids(category)]
    
    featuresets = [(extract_features(words), category) for (words, category) in documents]
    
    # 划分训练集和测试集
    training_set = featuresets[:1900]
    testing_set = featuresets[1900:]
    
    # 训练朴素贝叶斯分类器
    classifier = NaiveBayesClassifier.train(training_set)
    
    # 测试分类器
    print("Accuracy:", nltk.classify.accuracy(classifier, testing_set))
    
    # 显示分类器预测
    print("\nExample predictions:")
    for i in range(5):
        words = testing_set[i][0]
        print(f"{i+1}: {'Positive' if classifier.classify(extract_features(words)) == 'pos' else 'Negative'} - {' '.join(words)}")

# 运行模型
build_model()
```

#### 实际案例：社交网络情感分析应用

为了更好地理解情感智慧在人际关系咨询中的应用，我们可以通过一个实际案例来分析社交网络上的情感表达。

案例：某社交媒体平台上的一项调查显示，用户在评论时使用的积极情感词汇和消极情感词汇的比例。

```python
import pandas as pd
from textblob import TextBlob

# 读取评论数据
data = pd.read_csv('social_media_comments.csv')

# 计算积极情感和消极情感的比例
def calculate_sentiment(scores):
    pos_scores = scores['pos']
    neg_scores = scores['neg']
    if pos_scores > neg_scores:
        return 'Positive'
    elif pos_scores < neg_scores:
        return 'Negative'
    else:
        return 'Neutral'

data['sentiment'] = data.apply(lambda row: calculate_sentiment(row['sentiment_scores']), axis=1)

# 分析情感分布
sentiment_counts = data['sentiment'].value_counts()
print(sentiment_counts)
```

该案例使用TextBlob库对评论进行情感分析，并根据积极情感和消极情感得分计算情感比例。通过分析结果，咨询师可以了解用户的情感倾向，从而为用户提供更个性化的建议。
