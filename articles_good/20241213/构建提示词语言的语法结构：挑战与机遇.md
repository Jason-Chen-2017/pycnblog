import spacy

nlp = spacy.load("en_core_web_sm")

def build_grammar_structure(prompt):
    doc = nlp(prompt)
    grammar_rules = []

    for token in doc:
        grammar_rules.append(token.text)

    return grammar_rules
```

from random import choice

def generate_text(grammar_rules):
    sentence = ""
    while not sentence.endswith("."):
        word = choice(grammar_rules)
        sentence += word + " "

    return sentence.strip()
```
