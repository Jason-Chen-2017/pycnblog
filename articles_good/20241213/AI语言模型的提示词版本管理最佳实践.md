```python
import json
import git
import datetime

class VersionManager:
    def __init__(self, version_file='version_records.json'):
        self.version_file = version_file
        self.load_version_records()

    def load_version_records(self):
        try:
            with open(self.version_file, 'r') as f:
                self.version_records = json.load(f)
        except FileNotFoundError:
            self.version_records = {
                'version_id': 1,
                'create_time': datetime.datetime.now(),
                'current_version': [],
                'history_versions': []
            }
            self.save_version_records()

    def save_version_records(self):
        with open(self.version_file, 'w') as f:
            json.dump(self.version_records, f, indent=4)

    def compare_versions(self, current_version, new_version):
        added = list(set(new_version) - set(current_version))
        removed = list(set(current_version) - set(new_version))
        modified = [item for item in new_version if item in current_version and current_version[item] != new_version[item]]
        return added, removed, modified

    def update_version(self, new_version):
        added, removed, modified = self.compare_versions(self.version_records['current_version'], new_version)
        self.version_records['current_version'] = new_version
        self.version_records['history_versions'].append({
            'version_id': self.version_records['version_id'] + 1,
            'create_time': datetime.datetime.now(),
            'content': new_version
        })
        self.save_version_records()

    def rollback_version(self, target_version_id):
        for version in self.version_records['history_versions']:
            if version['version_id'] == target_version_id:
                self.version_records['current_version'] = version['content']
                self.save_version_records()
                return True
        return False

# 实例化版本管理器
version_manager = VersionManager()

# 更新版本
new_version = ['提示词1', '提示词2', '新提示词4', '新提示词5']
version_manager.update_version(new_version)

# 回滚版本
version_manager.rollback_version(1)

# 打印当前版本
print(version_manager.version_records)
```
