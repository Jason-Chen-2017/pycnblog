2. **数据预处理模块**：对关键词数据进行清洗、去重和标准化处理。
3. **关键词分类模块**：对关键词进行分类。

```python
import pandas as pd
from sklearn.feature_extraction.text import CountVectorizer
from sklearn.preprocessing import LabelEncoder

# 数据采集模块
def collect_data(file_path):
    data = pd.read_csv(file_path)
    return data['query']

# 数据预处理模块
def preprocess_data(data):
    # 去除停用词
    stop_words = set(['the', 'and', 'is', 'in', 'to'])
    data = [word for word in data if word not in stop_words]
    # 去重
    data = list(set(data))
    # 标准化
    data = [word.lower() for word in data]
    return data

# 关键词分类模块
def classify_keywords(data):
    # 建立分类体系
    categories = ['技术', '娱乐', '生活', '体育', '财经']
    # 标签化
    label_encoder = LabelEncoder()
    labels = label_encoder.fit_transform(data)
    return categories, labels

# 标签化模块
def tag_keywords(data, categories):
    tags = [categories[i] for i in data]
    return tags

# 权重分配模块
def assign_weights(data):
    vectorizer = CountVectorizer()
    X = vectorizer.fit_transform(data)
    weight = X.toarray().sum(axis=0)
    return weight

# 用户行为分析模块
def analyze_user_behavior(data, user_data):
    # 根据用户兴趣调整权重
    user_interest = user_data['interest']
    weight = assign_weights(data)
    for i, interest in enumerate(user_interest):
        weight[i] *= 1.5
    return weight

# 主函数
def main():
    # 数据采集
    data = collect_data('search_log.csv')
    # 数据预处理
    data = preprocess_data(data)
    # 关键词分类
    categories, labels = classify_keywords(data)
    # 标签化
    tags = tag_keywords(data, categories)
    # 权重分配
    weight = assign_weights(data)
    # 用户行为分析
    user_data = pd.read_csv('user_interest.csv')
    weight = analyze_user_behavior(data, user_data)
    # 更新提示词模板库
    template库 = {'keyword': data, 'label': labels, 'tag': tags, 'weight': weight}
    pd.DataFrame(template库).to_csv('template库.csv', index=False)

if __name__ == '__main__':
    main()
```
