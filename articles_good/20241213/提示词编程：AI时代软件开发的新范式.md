def preprocess_image(image_path):

        rescale=1./255,
        shear_range=0.2,
        zoom_range=0.2,
        horizontal_flip=True
    )
    return datagen.flow_from_directory(
        image_path,
        target_size=(224, 224),
        batch_size=32,
        class_mode='categorical'
    )

def extract_features(model, data):
    return model.predict(data)

# 分类器训练
def train_classifier(model, features, labels):
    model.compile(optimizer=Adam(learning_rate=0.001), loss='categorical_crossentropy', metrics=['accuracy'])
    model.fit(features, labels, epochs=10, batch_size=32)
    return model

# 主函数
def main(image_path):
    train_data = preprocess_image(image_path)
    features = extract_features(ResNet50(weights='imagenet'), train_data)
    labels = train_data.labels
    classifier = train_classifier(Model(inputs=features, outputs=ResNet50(weights='imagenet').output), features, labels)
    print(classifier.metrics_names)
