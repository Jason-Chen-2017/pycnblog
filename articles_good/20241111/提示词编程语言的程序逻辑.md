```python
import os
import json

# 存储待办事项的文件路径
TODO_FILE = "todos.json"

# 添加待办事项
def add_todo(todo):
    todos = load_todos()
    todos.append(todo)
    save_todos(todos)

# 删除待办事项
def delete_todo(todo_id):
    todos = load_todos()
    todos = [todo for todo in todos if todo['id'] != todo_id]
    save_todos(todos)

# 查看待办事项
def view_todos():
    todos = load_todos()
    for todo in todos:
        print(todo['id'], todo['title'])

# 加载待办事项
def load_todos():
    if os.path.exists(TODO_FILE):
        with open(TODO_FILE, 'r') as f:
            return json.load(f)
    else:
        return []

# 保存待办事项
def save_todos(todos):
    with open(TODO_FILE, 'w') as f:
        json.dump(todos, f)

# 主程序
def main():
    while True:
        print("1. 添加待办事项")
        print("2. 删除待办事项")
        print("3. 查看待办事项")
        print("4. 退出")
        choice = input("请选择操作：")

        if choice == "1":
            title = input("请输入待办事项标题：")
            add_todo({'id': str(len(load_todos()) + 1), 'title': title})
            print("待办事项添加成功！")

        elif choice == "2":
            id = input("请输入待办事项ID：")
            delete_todo(id)
            print("待办事项删除成功！")

        elif choice == "3":
            view_todos()

        elif choice == "4":
            break

        else:
            print("无效选择，请重新输入！")

if __name__ == "__main__":
    main()
```

```python
from django.db import models
from django.contrib.auth.models import AbstractUser
from django.contrib.auth.hashers import make_password

class CustomUser(AbstractUser):
    # 自定义用户表，继承AbstractUser类
    pass

class RegisterView(View):
    def get(self, request):
        # 处理GET请求，返回注册页面
        return render(request, 'register.html')

    def post(self, request):
        # 处理POST请求，处理注册逻辑
        username = request.POST.get('username')
        password = request.POST.get('password')
        # 验证用户名和密码的有效性
        if not username or not password:
            return HttpResponse('用户名或密码不能为空！')
        # 验证用户名的唯一性
        if CustomUser.objects.filter(username=username).exists():
            return HttpResponse('用户名已存在！')
        # 将密码加密存储
        encrypted_password = make_password(password)
        # 创建新用户
        user = CustomUser.objects.create_user(username=username, password=encrypted_password)
        # 登录用户
        authenticate(username=username, password=password)
        # 重定向到登录成功页面
        return HttpResponseRedirect('/login/success/')
```

```python
    def post(self, request):
        # 处理POST请求，处理登录逻辑
```

#### 源代码实现

以下是天气查询应用的源代码实现：

```python
import json
import requests

# 查询天气信息
def query_weather(city):
    url = f"http://api.openweathermap.org/data/2.5/weather?q={city}&appid=your_api_key"
    response = requests.get(url)
    if response.status_code == 200:
        weather_data = response.json()
        main_data = weather_data['main']
        temp = main_data['temp'] - 273.15  # 将开尔文温度转换为摄氏度
        humidity = main_data['humidity']
        weather = weather_data['weather'][0]['description']
        return f"城市：{city}\n温度：{temp}℃\n湿度：{humidity}%\n天气：{weather}"
    else:
        return "查询失败，请检查网络连接或城市名称。"

# 主程序
def main():
    while True:
        city = input("请输入城市名称（输入'exit'退出）：")
        if city.lower() == 'exit':
            break
        weather_info = query_weather(city)
        print(weather_info)

if __name__ == "__main__":
    main()
```
