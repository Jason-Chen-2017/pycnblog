from nltk.corpus import stopwords

# 准备数据
questions = ["你好，有什么问题我可以帮忙吗？",
             "我想要购买一件红色的衬衫。",
             "请问这个商品有没有折扣？"]
answers = ["你好！请问有什么可以帮助您的？",
            "您好，我们有一件红色的衬衫，您想了解更多信息吗？",
            "当然有，这个商品现在正在打折，您想要购买吗？"]

# 去除停用词
stop_words = set(stopwords.words('english'))
filtered_questions = [question.lower().translate(str.maketrans('', '', string.punctuation)) for question in questions]
filtered_answers = [answer.lower().translate(str.maketrans('', '', string.punctuation)) for answer in answers]

# 构建TF-IDF向量模型
vectorizer = TfidfVectorizer(stop_words=stop_words)
question_vectors = vectorizer.fit_transform(filtered_questions)
answer_vectors = vectorizer.transform(filtered_answers)

# 计算相似度
similarity_scores = []
for question_vector in question_vectors:
    scores = cosine_similarity(question_vector, answer_vectors)
    similarity_scores.append(scores.max())

# 根据相似度分数匹配答案
matched_answers = [answers[i] for i, score in enumerate(similarity_scores) if score == max(similarity_scores)]

# 回答用户问题
def answer_question(question):
    filtered_question = question.lower().translate(str.maketrans('', '', string.punctuation))
    question_vector = vectorizer.transform([filtered_question])
    scores = cosine_similarity(question_vector, answer_vectors)
    matched_answer = [answers[i] for i, score in enumerate(scores) if score == max(scores)]
    return matched_answer[0]
