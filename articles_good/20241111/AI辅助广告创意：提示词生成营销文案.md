import openai

# 设置API密钥
openai.api_key = "your-api-key"

# 生成广告文案
def generate_advertisement_copy(prompt):
    response = openai.Completion.create(
        engine="text-davinci-002",
        prompt=prompt,
        max_tokens=50,
        n=1,
        stop=None,
        temperature=0.5
    )
    return response.choices[0].text.strip()

# 测试
prompt = "为双十一活动创作一句吸引人的广告语。"

print(copy)
```
