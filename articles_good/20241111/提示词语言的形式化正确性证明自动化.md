coq = Coq()

# 定义提示词语言的规则和模式
def rule1(item):
    return item == "apple" or item == "banana"

def rule2(item):
    return item == "orange" or item == "grape"

# 定义验证条件
def verify(item):
    assert rule1(item) or rule2(item)

# 自动化证明
coq.push()
coq.write("Set Import coq.all.")
coq.write("Induction rule1.")
coq.write("Induction rule2.")
coq.write("Definition verify := ", verify, ".")
coq.write("Theorem verify_right : forall item, verify item.")
coq.prove("forall item, verify item.", tactic="axiom")

# 验证结果分析
print(coq.get_last_proof())
```

module TipLang where

-- 定义提示词语言的规则和模式
data Rule = Rule1 [String] String | Rule2 [String] String deriving (Show)

-- 定义验证条件
isValid :: [String] -> Bool
isValid items = 
    let 
        rules = [Rule1 ["apple", "banana"] "fruit", Rule2 ["orange", "grape"] "fruit"]
    in
    any (\rule -> applyRule rule items) rules

-- 应用规则
applyRule :: Rule -> [String] -> Bool
applyRule (Rule1 conditions result) items = 
    all (\condition -> elem condition items) conditions && elem result items

applyRule (Rule2 conditions result) items = 
    all (\condition -> elem condition items) conditions && elem result items

-- 主函数
main :: IO ()
main = do
    putStrLn "Enter a list of items:"
    items <- getLine
    let itemsList = words items
    putStrLn $ "Is the list valid? " ++ show (isValid itemsList)
```
