import numpy as np

class Agent:
    def __init__(self, id):

        self.state = None
        self.action = None

    def receive(self, message):
        self.state = message

    def decide(self):
        # 假设action是一个简单决策

    def send(self):
        return f'Agent {self.id}: {self.action}'

# 创建多个智能体
agents = [Agent(i) for i in range(5)]

# 模拟通信过程
for agent in agents:
    message = f'Current market state: {np.random.random() > 0.5}'
    agent.receive(message)
    agent.decide()
    print(agent.send())

##### 5.2.2 資產配置模型的实现
资产配置模型的实现可以通过以下代码实现：

```python
import numpy as np
from scipy.optimize import minimize

# 定义收益和风险函数
def portfolio_return(weights, returns):
    return np.dot(weights.T, returns)

def portfolio_risk(weights, cov_matrix):
    return np.sqrt(np.dot(weights.T, np.dot(cov_matrix, weights)))

# 定义优化目标
def optimization_function(weights, returns, cov_matrix):
    return portfolio_risk(weights, cov_matrix)

# 定义约束条件
def constraint(weights):
    return np.sum(weights) - 1

# 初始化权重
n_assets = 4
initial_weights = np.array([0.25] * n_assets)

# 优化求解
result = minimize(optimization_function, initial_weights, 
                  args=(returns, cov_matrix), method='SLSQP',
                  constraints=({'type': 'eq', 'fun': constraint}))

# 输出结果
print(result)
```
