#### 3.3.2 核心代码
```python
from transformers import AutoTokenizer, AutoModelForMaskedLM
import networkx as nx
import matplotlib.pyplot as plt

# 加载预训练模型
model_name = "bert-large-uncased"
tokenizer = AutoTokenizer.from_pretrained(model_name)
model = AutoModelForMaskedLM.from_pretrained(model_name)

def extract_financial_terms(text):
    # 使用模型进行词汇抽取
    inputs = tokenizer.encode_plus(text, return_tensors="pt", truncation=True, padding=True)
    with torch.no_grad():
        outputs = model(**inputs)
    # 提取词汇
    terms = tokenizer.decode(outputs.logits.argmax(dim=-1).squeeze().tolist())
    return terms

def build_network(terms):
    # 构建词汇网络
    G = nx.Graph()
    G.add_nodes_from(terms)
    # 添加语义关系
    # 这里简化为随机连接，实际应根据语义分析
    for i in range(len(terms)):
        for j in range(i+1, len(terms)):
            if random.random() < 0.3:
                G.add_edge(terms[i], terms[j])
    return G

def visualize_network(G):
    # 可视化网络
    plt.figure(figsize=(10, 10))
    nx.draw(G, node_size=1000, alpha=0.8, edge_color="b", edge_alpha=0.6)
    plt.title("Financial Terms Network")
    plt.show()

# 示例
text = "在金融市场中，股票、债券、基金等都是重要的投资工具。风险管理、资产配置是金融分析的核心内容。"
terms = extract_financial_terms(text)
G = build_network(terms)
visualize_network(G)
```

### 3.4 算法数学模型与公式

#### 3.4.2 语义关系计算的相似度公式
$$ \text{Similarity}(a, b) = \frac{\sum_{i=1}^{n} w_i a_i b_i}{\sqrt{\sum_{i=1}^{n} w_i a_i^2} \cdot \sqrt{\sum_{i=1}^{n} w_i b_i^2}} } $$
