```python
# order_processing_system.py

import logging
from order_validator import OrderValidator
from inventory_manager import InventoryManager
from payment_processor import PaymentProcessor

class OrderProcessingSystem:
    def __init__(self):
        self.validator = OrderValidator()
        self.inventory_manager = InventoryManager()
        self.payment_processor = PaymentProcessor()
        self.logger = logging.getLogger("OrderProcessingSystem")

    def process_order(self, order_data):
        try:
            # 验证订单数据
            if not self.validator.validate(order_data):
                raise ValueError("Invalid order data")

            # 检查库存
            inventory_status = self.inventory_manager.check_inventory(order_data)
            if inventory_status == "Insufficient inventory":
                raise ValueError("Insufficient inventory")

            # 处理支付请求
            payment_result = self.payment_processor.process_payment(order_data)
            if payment_result != "Payment successful":
                raise ValueError("Payment failed")

            return "Order processed successfully"

        except ValueError as e:
            self.logger.error(f"Error processing order: {str(e)}")
            return "Error processing order"

# order_validator.py

class OrderValidator:
    def validate(self, order_data):
        # 验证订单数据格式
        if not isinstance(order_data, dict):
            return False
        if "order_id" not in order_data or "customer_id" not in order_data or "items" not in order_data:
            return False
        if not isinstance(order_data["order_id"], str) or not isinstance(order_data["customer_id"], str) or not isinstance(order_data["items"], list):
            return False
        return True

# inventory_manager.py

class InventoryManager:
    def check_inventory(self, order_data):
        # 检查库存
        for item in order_data["items"]:
            if item["quantity"] > item["available_quantity"]:
                return "Insufficient inventory"
        return "Sufficient inventory"

# payment_processor.py

class PaymentProcessor:
    def process_payment(self, order_data):
        # 处理支付请求
        # 这里简化为直接返回"Payment successful"
        return "Payment successful"
```
