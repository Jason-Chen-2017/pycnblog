```python
def build_models(vocab_size, embedding_dim, sequence_length):
    # 输入层
    input_sequence = Input(shape=(sequence_length,))
    
    # 嵌入层
    embedded_sequence = Embedding(vocab_size, embedding_dim)(input_sequence)
    
    # LSTM层
    lstm_output = LSTM(128)(embedded_sequence)
    
    # 判别器模型
    discriminator = Model(inputs=input_sequence, outputs=lstm_output)
    
    # 生成器模型
    generator = Model(inputs=input_sequence, outputs=lstm_output)
    
    return generator, discriminator

            real_sequence = batch
            fake_sequence = generator.predict(real_sequence)
            
            # 训练判别器
            discriminator.train_on_batch(real_sequence, real_sequence)
            discriminator.train_on_batch(fake_sequence, fake_sequence)
            
            # 训练生成器
            generated_sequences = generator.predict(real_sequence)
            generator.train_on_batch(real_sequence, generated_sequences)
```
