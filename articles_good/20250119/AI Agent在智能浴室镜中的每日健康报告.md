```python
import numpy as np

def calculate_heart_rate(data):
    # 计算心率
    heart_rate = np.mean(data)
    return heart_rate

def calculate_temperature(data):
    # 计算体温
    temperature = np.mean(data)
    return temperature

def calculate_respiration_rate(data):
    # 计算呼吸频率
    respiration_rate = np.mean(data)
    return respiration_rate

# 示例数据
heart_rate_data = [75, 76, 74, 77, 75]
temperature_data = [36.5, 36.4, 36.6, 36.5, 36.6]
respiration_rate_data = [20, 19, 21, 20, 20]

# 计算健康数据
heart_rate = calculate_heart_rate(heart_rate_data)
temperature = calculate_temperature(temperature_data)
respiration_rate = calculate_respiration_rate(respiration_rate_data)

print("心率：", heart_rate)
print("体温：", temperature)
print("呼吸频率：", respiration_rate)
```
