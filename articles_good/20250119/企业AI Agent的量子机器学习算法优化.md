import qiskit
from qiskit import QuantumCircuit, Aer, execute
from qiskit.circuit import QuantumRegister, ClassicalRegister
from qiskit.visualization import plot_bloch_vector

# 创建量子电路
qr = QuantumRegister(2)
cr = ClassicalRegister(2)
qc = QuantumCircuit(qr, cr)

# Hadamard门初始化量子态
qc.h(qr[0])
qc.h(qr[1])

# 控制非门实现量子比特间的交互
qc.cx(qr[0], qr[1])

# 测量量子比特
qc.measure(qr, cr)

# 运行量子电路
backend = Aer.get_backend("qasm_simulator")
result = execute(qc, backend, shots=1024).result()
counts = result.get_counts(qc)

# 输出结果
print("输出结果：", counts)
print("概率分布：", counts.values())

# 可视化量子态
state_vector = qc.initialize()
backend = Aer.get_backend("statevector_simulator")
result = execute(qc, backend, statevector=True).result()
state = result.get_statevector()
plot_bloch_vector(state, title="量子态可视化")
```

qc.cx(qr[0], qr[1])

backend = Aer.get_backend("qasm_simulator")

qc.measure(qr, cr)

backend = Aer.get_backend("qasm_simulator")
result = execute(qc, backend, shots=1024).result()

# 输出结果
print("输出结果：", counts)
print("概率分布：", counts.values())

# 可视化量子态
state_vector = qc.initialize()
backend = Aer.get_backend("statevector_simulator")

state_vector = qc.initialize()

import qiskit
from qiskit import QuantumCircuit, Aer, execute
from qiskit.circuit import QuantumRegister, ClassicalRegister
from qiskit.visualization import plot_bloch_vector

# 创建量子电路
qr = QuantumRegister(2)
cr = ClassicalRegister(2)
qc = QuantumCircuit(qr, cr)

# Hadamard门初始化量子态
qc.h(qr[0])
qc.h(qr[1])

# 控制非门实现量子比特间的交互
qc.cx(qr[0], qr[1])

# 测量量子比特
qc.measure(qr, cr)

# 运行量子电路
backend = Aer.get_backend("qasm_simulator")
result = execute(qc, backend, shots=1024).result()
counts = result.get_counts(qc)

# 输出结果
print("输出结果：", counts)
print("概率分布：", counts.values())

# 可视化量子态
state_vector = qc.initialize()
backend = Aer.get_backend("statevector_simulator")
result = execute(qc, backend, statevector=True).result()
state = result.get_statevector()
plot_bloch_vector(state, title="量子态可视化")
```
