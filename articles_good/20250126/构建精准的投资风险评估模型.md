以下是一个简单的核心代码实现，用于数据收集、预处理和风险评估：

```python
import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestRegressor

# 数据收集
def collect_data():
    # 这里使用pandas读取数据，具体实现根据数据来源进行调整
    data = pd.read_csv("investment_data.csv")
    return data

# 数据预处理
def preprocess_data(data):
    # 数据清洗、转换和归一化等操作
    # 这里仅做简单的数据清洗和分割
    data.dropna(inplace=True)
    train_data, test_data = np.split(data, [int(len(data) * 0.8)])
    return train_data, test_data

# 风险评估
def assess_risk(train_data, test_data):
    # 使用随机森林回归模型进行风险评估
    model = RandomForestRegressor()
    model.fit(train_data.iloc[:, :-1], train_data.iloc[:, -1])
    risk_scores = model.predict(test_data.iloc[:, :-1])
    return risk_scores

# 主函数
def main():
    data = collect_data()
    train_data, test_data = preprocess_data(data)
    risk_scores = assess_risk(train_data, test_data)
    print(risk_scores)

if __name__ == "__main__":
    main()
```
