# 前向传播
def forward_pass(x, weights):
    z = np.dot(x, weights)
    return z

# 反向传播
def backward_pass(x, weights, z, y):
    delta = z - y
    weights -= delta * x
    return weights

# 训练模型
def train_model(x, y, weights, epochs):
    for epoch in range(epochs):
        z = forward_pass(x, weights)
        weights = backward_pass(x, weights, z, y)
    return weights
