#### 3.2 提示词分层的数学模型与公式

提示词分层的数学模型可以表示为：

$$
T(n) = \sum_{i=1}^{k} T_i(n)
$$

其中，$T(n)$表示整个任务的处理时间，$T_i(n)$表示第$i$个层次的任务处理时间，$k$表示层次的个数。

# 分层处理数据
def hierarchical_processing(data, layers):
    for layer in layers:
        # 在每个层次上进行数据处理
        if layer == "特征提取":
            # 特征提取逻辑
            data = pd.get_dummies(data)
        elif layer == "降维":
            # 降维逻辑
            data = pd.DataFrame(np.random.rand(data.shape[0], data.shape[1] // 2))
        elif layer == "分类":
            # 分类逻辑
            from sklearn.linear_model import LogisticRegression
            model = LogisticRegression()
            model.fit(data, y)
            y_pred = model.predict(data)
            data['prediction'] = y_pred
        else:
            raise ValueError(f"未知层次：{layer}")
    return data

# 定义层次列表
layers = ["特征提取", "降维", "分类"]

# 执行分层处理
data_processed = hierarchical_processing(X, layers)

# 可视化结果
plt.scatter(data_processed.iloc[:, 0], data_processed.iloc[:, 1], c=data_processed['prediction'])
plt.xlabel('Feature 1')
plt.ylabel('Feature 2')
plt.title('Data after Hierarchical Processing')
plt.show()
