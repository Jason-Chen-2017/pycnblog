```python
import numpy as np
import matplotlib.pyplot as plt

# 定义噪声谱
def noise_spectrum(f, N):
    return np.sum([N[i] * np.exp(-1j * 2 * np.pi * f * i) for i in range(len(N))])

# 定义滤波器传递函数
def filter_transfer_function(f, f_c):
    return 1 / (1 + (f / f_c)**2)

# 生成噪声
N = [1, 2, 3]
f = np.linspace(0, 10, 1000)

# 计算噪声谱
noise = noise_spectrum(f, N)

# 计算滤波器传递函数
f_c = 5
H_f = filter_transfer_function(f, f_c)

# 过滤噪声
filtered_noise = noise * H_f

# 绘制噪声谱和滤波后的噪声谱
plt.plot(f, noise, label='Noise Spectrum')
plt.plot(f, filtered_noise, label='Filtered Noise')
plt.xlabel('Frequency (Hz)')
plt.ylabel('Noise Intensity')
plt.legend()
plt.show()
```
