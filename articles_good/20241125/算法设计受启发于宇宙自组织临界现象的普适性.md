```python
import torch.nn as nn
from torch.optim import Adam
from torchtext.datasets import IMDB
from torchtext.data import Field, BatchIterator

# 定义循环神经网络（RNN）模型
class RNNModel(nn.Module):
    def __init__(self, input_dim, embedding_dim, hidden_dim, output_dim, num_layers, dropout=0.5):
        super().__init__()
        self.embedding = nn.Embedding(input_dim, embedding_dim)
        self.rnn = nn.RNN(embedding_dim, hidden_dim, num_layers, dropout=dropout, batch_first=True)
        self.fc = nn.Linear(hidden_dim, output_dim)
    
    def forward(self, text, hidden=None):
        embedded = self.embedding(text)
        output, hidden = self.rnn(embedded, hidden)
        hidden = hidden[-1, :, :]
        return self.fc(hidden.view(hidden.size(0), -1))

# 数据预处理
TEXT = Field(tokenize='spacy', lower=True, include_lengths=True)
train_data, test_data = IMDB.splits(TEXT, TEXT)
TEXT.build_vocab(train_data, max_size=25000, vectors="glove.6B.100d")
train_iterator, test_iterator = BatchIterator(train_data, test_data, batch_size=64, shuffle=True)

# 建立并训练模型
model = RNNModel(len(TEXT.vocab), 100, 256, 1, 2, dropout=0.5)
optimizer = Adam(model.parameters(), lr=0.001)
criterion = nn.BCEWithLogitsLoss()

for epoch in range(10):
    model.train()
    for batch in train_iterator:
        optimizer.zero_grad()
        predictions = model(batch.text, batch.hidden).squeeze(1)
        loss = criterion(predictions, batch.label)
        loss.backward()
        optimizer.step()

# 测试模型性能
model.eval()
with torch.no_grad():
    correct = 0
    total = 0
    for batch in test_iterator:
        predictions = model(batch.text, batch.hidden).squeeze(1)
        predicted = (predictions > 0).float()
        total += batch.label.size(0)
        correct += (predicted.round() == batch.label).sum().item()

print(f'\nTest Accuracy: {100 * correct / total:.2f}%')
```

```python
from torchtext.data import Field, BucketIterator

# 数据预处理

train_data, test_data = IMDB.splits(TEXT, TEXT)

# 模型设计
class RNNModel(nn.Module):

        super().__init__()
        self.embedding = nn.Embedding(input_dim, embedding_dim)
        self.rnn = nn.RNN(embedding_dim, hidden_dim, num_layers, dropout=dropout, batch_first=True)
        self.fc = nn.Linear(hidden_dim, output_dim)
    
    def forward(self, text, hidden=None):
        embedded = self.embedding(text)
        output, hidden = self.rnn(embedded, hidden)
        hidden = hidden[-1, :, :]
        return self.fc(hidden.view(hidden.size(0), -1))

# 训练模型

    for batch in train_iterator:
        optimizer.zero_grad()
        predictions = model(batch.text, batch.hidden).squeeze(1)

        loss.backward()
```
