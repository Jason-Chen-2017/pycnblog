### 6.2 模拟宇宙的实际案例

以下是一个模拟宇宙的实际案例，我们将使用 Python 编程语言和 matplotlib 库来实现。

#### 案例介绍

我们模拟一个简单的宇宙模型，其中包含若干个行星，每个行星都有质量、位置和速度。行星之间的相互作用由牛顿引力定律来描述。我们将使用欧拉法来模拟行星的运动。

#### 源代码实现

```python
import numpy as np
import matplotlib.pyplot as plt

# 参数设置
G = 6.67430e-11  # 引力常数
dt = 0.01  # 时间步长
t_end = 1000  # 模拟时间

# 初始条件
planets = [
    {"mass": 5.972e24, "pos": np.array([0, 0]), "vel": np.array([0, 0])},
    {"mass": 7.348e22, "pos": np.array([1, 0]), "vel": np.array([-0.1, 0])},
    {"mass": 3.35e23, "pos": np.array([-1, 0]), "vel": np.array([0.1, 0])}
]

# 欧拉法
for _ in range(int(t_end / dt)):
    # 计算引力
    forces = [np.zeros(2) for _ in range(len(planets))]
    for i, planet_i in enumerate(planets):
        for j, planet_j in enumerate(planets):
            if i != j:
                dx = planet_j["pos"] - planet_i["pos"]
                distance = np.linalg.norm(dx)
                force = G * planet_i["mass"] * planet_j["mass"] / distance**2
                force_dir = dx / distance
                forces[i] += force_dir * force

    # 更新速度和位置
    for i, planet in enumerate(planets):
        planet["vel"] += forces[i] / planet["mass"] * dt
        planet["pos"] += planet["vel"] * dt

# 可视化
plt.figure(figsize=(8, 6))
for planet in planets:
    plt.plot(planet["pos"][0], planet["pos"][1], 'o')
plt.xlabel("X")
plt.ylabel("Y")
plt.title("Orbits of planets")
plt.show()
```

#### 代码解读

- 我们首先导入所需的库，包括 numpy 和 matplotlib.pyplot。
- 设置参数，如引力常数 G、时间步长 dt 和模拟时间 t_end。
- 定义初始条件，包括行星的质量、位置和速度。
- 使用欧拉法计算行星之间的引力，并更新速度和位置。
- 最后，使用 matplotlib 库将行星的位置可视化。

import numpy as np
import matplotlib.pyplot as plt

# 参数设置
G = 6.67430e-11  # 引力常数
dt = 0.01  # 时间步长
t_end = 1000  # 模拟时间

# 初始条件
planets = [
    {"mass": 5.972e24, "pos": np.array([0, 0]), "vel": np.array([0, 0])},
    {"mass": 7.348e22, "pos": np.array([1, 0]), "vel": np.array([-0.1, 0])},
    {"mass": 3.35e23, "pos": np.array([-1, 0]), "vel": np.array([0.1, 0])}
]

# 欧拉法
for _ in range(int(t_end / dt)):
    # 计算引力
    forces = [np.zeros(2) for _ in range(len(planets))]
    for i, planet_i in enumerate(planets):
        for j, planet_j in enumerate(planets):
            if i != j:
                dx = planet_j["pos"] - planet_i["pos"]
                distance = np.linalg.norm(dx)
                force = G * planet_i["mass"] * planet_j["mass"] / distance**2
                force_dir = dx / distance
                forces[i] += force_dir * force

    # 更新速度和位置
    for i, planet in enumerate(planets):
        planet["vel"] += forces[i] / planet["mass"] * dt
        planet["pos"] += planet["vel"] * dt

# 可视化
plt.figure(figsize=(8, 6))
for planet in planets:
    plt.plot(planet["pos"][0], planet["pos"][1], 'o')
plt.xlabel("X")
plt.ylabel("Y")
plt.title("Orbits of planets")
plt.show()

import numpy as np
import matplotlib.pyplot as plt

# 参数设置
G = 6.67430e-11  # 引力常数
dt = 0.01  # 时间步长
t_end = 1000  # 模拟时间

# 初始条件
planets = [
    {"mass": 5.972e24, "pos": np.array([0, 0]), "vel": np.array([0, 0])},
    {"mass": 7.348e22, "pos": np.array([1, 0]), "vel": np.array([-0.1, 0])},
    {"mass": 3.35e23, "pos": np.array([-1, 0]), "vel": np.array([0.1, 0])}
]

# 欧拉法
for _ in range(int(t_end / dt)):
    # 计算引力
    forces = [np.zeros(2) for _ in range(len(planets))]
    for i, planet_i in enumerate(planets):
        for j, planet_j in enumerate(planets):
            if i != j:
                dx = planet_j["pos"] - planet_i["pos"]
                distance = np.linalg.norm(dx)
                force = G * planet_i["mass"] * planet_j["mass"] / distance**2
                force_dir = dx / distance
                forces[i] += force_dir * force

    # 更新速度和位置
    for i, planet in enumerate(planets):
        planet["vel"] += forces[i] / planet["mass"] * dt
        planet["pos"] += planet["vel"] * dt

# 可视化
plt.figure(figsize=(8, 6))
for planet in planets:
    plt.plot(planet["pos"][0], planet["pos"][1], 'o')
plt.xlabel("X")
plt.ylabel("Y")
plt.title("Orbits of planets")
plt.show()
```

### 6.2 模拟宇宙的实际案例

以下是一个模拟宇宙的实际案例，我们将使用 Python 编程语言和 matplotlib 库来实现。

#### 案例介绍

我们模拟一个简单的宇宙模型，其中包含若干个行星，每个行星都有质量、位置和速度。行星之间的相互作用由牛顿引力定律来描述。我们将使用欧拉法来模拟行星的运动。

#### 源代码实现

```python
import numpy as np
import matplotlib.pyplot as plt

# 参数设置
G = 6.67430e-11  # 引力常数
dt = 0.01  # 时间步长
t_end = 1000  # 模拟时间

# 初始条件
planets = [
    {"mass": 5.972e24, "pos": np.array([0, 0]), "vel": np.array([0, 0])},
    {"mass": 7.348e22, "pos": np.array([1, 0]), "vel": np.array([-0.1, 0])},
    {"mass": 3.35e23, "pos": np.array([-1, 0]), "vel": np.array([0.1, 0])}
]

# 欧拉法
for _ in range(int(t_end / dt)):
    # 计算引力
    forces = [np.zeros(2) for _ in range(len(planets))]
    for i, planet_i in enumerate(planets):
        for j, planet_j in enumerate(planets):
            if i != j:
                dx = planet_j["pos"] - planet_i["pos"]
                distance = np.linalg.norm(dx)
                force = G * planet_i["mass"] * planet_j["mass"] / distance**2
                force_dir = dx / distance
                forces[i] += force_dir * force

    # 更新速度和位置
    for i, planet in enumerate(planets):
        planet["vel"] += forces[i] / planet["mass"] * dt
        planet["pos"] += planet["vel"] * dt

# 可视化
plt.figure(figsize=(8, 6))
for planet in planets:
    plt.plot(planet["pos"][0], planet["pos"][1], 'o')
plt.xlabel("X")
plt.ylabel("Y")
plt.title("Orbits of planets")
plt.show()
```

- 我们首先导入所需的库，包括 numpy 和 matplotlib.pyplot。
- 设置参数，如引力常数 G、时间步长 dt 和模拟时间 t_end。
- 定义初始条件，包括行星的质量、位置和速度。
- 使用欧拉法计算行星之间的引力，并更新速度和位置。
- 最后，使用 matplotlib 库将行星的位置可视化。
