```python
import numpy as np

def compute_cost(X, y, w):
    m = len(y)
    h = X.dot(w)
    J = (1 / (2 * m)) * np.sum((h - y)**2)
    return J

def gradient_descent(X, y, w, alpha, num_iters):
    m = len(y)
    J_history = []

    for i in range(num_iters):
        h = X.dot(w)
        dw = (1 / m) * X.T.dot(h - y)
        w = w - alpha * dw
        J_history.append(compute_cost(X, y, w))

    return w, J_history

X = np.array([[1], [2], [3], [4]])
y = np.array([1, 2, 3, 4])
w = np.zeros((4, 1))
alpha = 0.01
num_iters = 1000

w, J_history = gradient_descent(X, y, w, alpha, num_iters)
print("w:", w)
print("J_history:", J_history)
```
