```python
# 区块链溯源算法示例

# 导入所需的库
from hashlib import sha256
from time import time

# 初始化区块链
blockchain = [
    {
        'index': 0,
        'timestamp': time(),
        'data': 'Genesis Block',
        'previous_hash': '0',
        'hash': '1',
    }
]

def calculate_hash(index, timestamp, data, previous_hash):
    # 计算哈希值
    value = f'{index}{timestamp}{data}{previous_hash}'.encode('utf-8')
    return sha256(value).hexdigest()

def create_block(data):
    # 创建区块
    last_block = blockchain[-1]
    index = last_block['index'] + 1
    timestamp = time()
    hash = calculate_hash(index, timestamp, data, last_block['hash'])
    new_block = {
        'index': index,
        'timestamp': timestamp,
        'data': data,
        'previous_hash': last_block['hash'],
        'hash': hash,
    }
    return new_block

def add_block(new_block):
    # 添加区块到区块链
    blockchain.append(new_block)

def is_chain_valid(blockchain):
    # 验证区块链的有效性
    for i in range(1, len(blockchain)):
        current = blockchain[i]
        previous = blockchain[i-1]
        if current['hash'] != calculate_hash(current['index'], current['timestamp'], current['data'], previous['hash']):
            return False
    return True

# 测试区块链溯源算法
new_block = create_block('DigitalArtwork A')
add_block(new_block)
new_block = create_block('DigitalArtwork B')
add_block(new_block)

print(blockchain)
print("Blockchain Valid:", is_chain_valid(blockchain))
```

#### 算法原理与数学模型

区块链的算法原理主要包括以下步骤：

1. **初始化区块链**：创建一个初始区块，作为区块链的基础。
2. **创建区块**：根据当前的数据和前一个区块的哈希值，创建一个新的区块。
3. **计算哈希值**：使用SHA-256算法计算区块的哈希值。
4. **添加区块**：将新生成的区块添加到区块链中。
5. **验证区块链**：检查区块链中的每个区块是否按照上述规则正确生成。

区块链的数学模型可以表示为：

$$ Hash(i) = \text{SHA-256}(i) $$

其中，\( i \) 表示区块的索引、时间戳、数据和前一个区块的哈希值。

假设区块链中有三个区块，分别是区块1、区块2和区块3。

```python
# blockchain.py

import hashlib
import json
from time import time

class Block:
    def __init__(self, index, timestamp, data, previous_hash):
        self.index = index
        self.timestamp = timestamp
        self.data = data
        self.previous_hash = previous_hash
        self.hash = self.calculate_hash()

    def calculate_hash(self):
        block_string = json.dumps(self.__dict__, sort_keys=True)
        return hashlib.sha256(block_string.encode()).hexdigest()

class Blockchain:
    def __init__(self):
        self.unconfirmed_transactions = []
        self.chain = []
        self.create_genesis_block()

    def create_genesis_block(self):
        genesis_block = Block(0, time(), "Genesis Block", "0")
        genesis_block.hash = genesis_block.calculate_hash()
        self.chain.append(genesis_block)

    def add_new_transaction(self, transaction):
        self.unconfirmed_transactions.append(transaction)

    def mine_pending_transactions(self, difficulty=4):
        last_block = self.chain[-1]
        last_hash = last_block.hash
        nonce = 0
        while self.calculate_hash(last_hash, nonce, difficulty)[:-difficulty] != '0' * difficulty:
            nonce += 1
        print(f"Block mined with nonce: {nonce}")
        last_hash = self.calculate_hash(last_hash, nonce, difficulty)
        block = Block(len(self.chain), time(), self.unconfirmed_transactions, last_hash)
        block.hash = block.calculate_hash()
        self.chain.append(block)
        self.unconfirmed_transactions = []

    @staticmethod
    def calculate_hash(last_hash, nonce, difficulty):
        block_string = f'{last_hash}{nonce}'.encode()
        return hashlib.sha256(block_string).hexdigest()

    def is_chain_valid(self):
        for i in range(1, len(self.chain)):
            current = self.chain[i]
            previous = self.chain[i - 1]
            if current.hash != current.calculate_hash():
                return False
            if current.previous_hash != previous.hash:
                return False
        return True

def main():
    blockchain = Blockchain()
    blockchain.add_new_transaction('Transaction 1')
    blockchain.add_new_transaction('Transaction 2')
    blockchain.mine_pending_transactions()

    print(f"Blockchain validity: {blockchain.is_chain_valid()}")
    print(f"Blockchain: {blockchain.chain}")

if __name__ == '__main__':
    main()
```

1. **创建创世区块（genesis_block）**：初始化区块链时，创建一个包含初始数据的创世区块。
2. **添加新交易（add_new_transaction）**：将新的交易记录添加到未确认交易列表中。
3. **挖掘未确认交易（mine_pending_transactions）**：通过计算区块链的哈希值，找到一个满足难度要求的随机数（nonce），并将其添加到区块链中。
4. **验证区块链（is_chain_valid）**：检查区块链中每个区块的哈希值和链接是否正确。
