    xss_patterns = ["<script>", "</script>"]

            return True
    return False

    sql_injection = detect_sql_injection(input_data)
    xss_attack = detect_xss_attack(input_data)
    
    result = {
        'sql_injection': sql_injection,
        'xss_attack': xss_attack
    }
    return jsonify(result)

if __name__ == '__main__':
    app.run(debug=True)
