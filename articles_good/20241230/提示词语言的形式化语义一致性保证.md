**3.2 Python源代码解释**

```python
# 引入依赖库
import spacy

# 加载NLP模型
nlp = spacy.load("en_core_web_sm")

# 解析提示词
def parse_prompt(prompt):
    doc = nlp(prompt)
    entities = [(ent.text, ent.label_) for ent in doc.ents]
    return entities

# 检查一致性
def check_consistency(prompt, rules):
    entities = parse_prompt(prompt)
    for rule in rules:
        # 假设rule是一个包含条件和结论的字典
        if all(entity in entities for entity in rule['conditions']):
            if rule['conclusion'] not in entities:
                return False
    return True

# 主函数
def main():
    prompt = "计算两个数的和"
    rules = [
        {
            'conditions': [('两个数', 'NUM'), ('和', 'ADP')],
            'conclusion': [('和', 'NOUN')]
        }
    ]
    is_consistent = check_consistency(prompt, rules)
    print(f"提示词的语义一致性：{is_consistent}")

# 运行主函数
if __name__ == "__main__":
    main()
```
