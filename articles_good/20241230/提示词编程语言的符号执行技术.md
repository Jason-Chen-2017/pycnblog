```python
# 符号执行器实现
class SymbolicExecutor:
    def __init__(self, program):
        self.program = program
        self.symbolic_vars = {}
        self.symbolic_paths = []
        self.symbolic_constraints = []

    def execute(self, input_values):
        self.symbolic_vars = {var: self.symbolic_value(input_values[var]) for var in input_values}
        self.symbolic_paths = [self.execute_path([], [])]
        self.symbolic_constraints = self.solve_constraints()

    def execute_path(self, path, constraints):
        # 执行一条路径
        if not self.program:
            return path
        instruction = self.program.pop(0)
        if instruction[0] == "assign":
            var, value = instruction[1], instruction[2]
            self.symbolic_vars[var] = self.symbolic_value(value)
            constraints.append(f"{var} = {self.symbolic_vars[var]}")
        elif instruction[0] == "if":
            condition = instruction[1]
            if self.eval_constraint(condition, self.symbolic_vars):
                path.append("true_path")
                constraints.append(f"{condition} = true")
                path = self.execute_path(self.program, constraints)
            else:
                path.append("false_path")
                constraints.append(f"{condition} = false")
                path = self.execute_path(self.program, constraints)
            self.program = instruction[2]
        return path

    def solve_constraints(self):
        # 求解约束
        constraints = self.symbolic_constraints
        for i in range(len(constraints) - 1, -1, -1):
            constraint = constraints[i]
            if constraint.endswith(" = true"):
                constraints.pop(i)
                continue
            if constraint.endswith(" = false"):
                constraints.pop(i)
                for j in range(i + 1, len(constraints)):
                    constraints[j] = constraints[j].replace(" = true", " = false")
                    constraints[j] = constraints[j].replace(" = false", " = true")
                continue
            if " < " in constraint and " > " in constraint:
                var, left, right = constraint.split(" ")
                left = left.strip()
                right = right.strip()
                if left == var and right.isdigit():
                    constraints[i] = f"{var} <= {right}"
                    continue
                if right == var and left.isdigit():
                    constraints[i] = f"{var} >= {left}"
                    continue
        return constraints

    def symbolic_value(self, value):
        # 获取符号值
        if isinstance(value, str) and value.isdigit():
            return value
        if value in self.symbolic_vars:
            return self.symbolic_vars[value]
        return f"_{value}_"

    def eval_constraint(self, constraint, symbolic_vars):
        # 评估约束
        for var, value in symbolic_vars.items():
            constraint = constraint.replace(var, value)
        if " < " in constraint and " > " in constraint:
            return False
        if "=" in constraint:
            return True
        return eval(constraint)

# 示例程序
program = [
    ["assign", "x", "5"],
    ["assign", "y", "10"],
    ["if", "x < y", [["assign", "z", "x + y"]], [["assign", "z", "x * y"]]]
]

executor = SymbolicExecutor(program)
executor.execute({"x": "x", "y": "y"})

for path in executor.symbolic_paths:
    print(path)
```
