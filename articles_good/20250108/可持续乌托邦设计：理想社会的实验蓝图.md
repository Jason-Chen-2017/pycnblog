```python
        self.waste_output += waste

    def get_usage(self):
        return {
            'energy_usage': self.energy_usage,
            'water_usage': self.water_usage,
            'waste_output': self.waste_output
        }

# 2. 资源分配模块
class ResourceAllocator:
    def __init__(self, monitor):
        self.monitor = monitor

    def allocate_resources(self, residents):
        total_resources = self.monitor.get_usage()
        allocated_resources = {}

        for resident in residents:
            resident_share = {
                'energy': total_resources['energy_usage'] / len(residents),
                'water': total_resources['water_usage'] / len(residents),
                'waste': total_resources['waste_output'] / len(residents)
            }
            allocated_resources[resident] = resident_share

        return allocated_resources
```
