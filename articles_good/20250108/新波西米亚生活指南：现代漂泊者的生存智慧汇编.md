from flask_pymongo import PyMongo

app = Flask(__name__)
app.config["MONGO_URI"] = "mongodb://localhost:27017/myDatabase"
mongo = PyMongo(app)

@app.route("/users", methods=["POST"])
def create_user():
    user_data = request.get_json()
    user = mongo.db.users.insert_one(user_data)
    return jsonify({"_id": str(user.inserted_id)})

@app.route("/users/<id>", methods=["GET"])
def get_user(id):
    user = mongo.db.users.find_one({"_id": id})
    return jsonify(user)

@app.route("/tasks", methods=["POST"])
def create_task():
    task_data = request.get_json()
    task = mongo.db.tasks.insert_one(task_data)
    return jsonify({"_id": str(task.inserted_id)})

@app.route("/tasks/<id>", methods=["GET"])
def get_task(id):
    task = mongo.db.tasks.find_one({"_id": id})
    return jsonify(task)

if __name__ == "__main__":
