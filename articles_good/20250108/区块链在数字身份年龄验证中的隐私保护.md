# 构建Merkle树
from hashlib import sha256

def build_merkle_tree(data_list):
    if len(data_list) == 1:
        return data_list[0]
    else:
        tree = []
        for i in range(0, len(data_list), 2):
            left_hash = sha256(data_list[i].encode()).hexdigest()
            right_hash = sha256(data_list[i+1].encode()).hexdigest()
            tree.append(sha256((left_hash + right_hash).encode()).hexdigest())
        return build_merkle_tree(tree)
