用户管理模块主要包括用户注册、登录和查询用户信息等功能。

```python
# user.py
from flask import Flask, request, jsonify
from flask_sqlalchemy import SQLAlchemy

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'mysql+pymysql://library_admin:password@localhost/shared_instrument_library'
db = SQLAlchemy(app)

class User(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(50), nullable=False)
    email = db.Column(db.String(100), nullable=False)
    password = db.Column(db.String(100), nullable=False)

# 用户注册
@app.route('/api/register', methods=['POST'])
def register():
    data = request.get_json()
    new_user = User(name=data['name'], email=data['email'], password=data['password'])
    db.session.add(new_user)
    db.session.commit()
    return jsonify({"message": "User registered successfully."})

# 用户登录
@app.route('/api/login', methods=['POST'])
def login():
    data = request.get_json()
    user = User.query.filter_by(email=data['email'], password=data['password']).first()
    if user:
        return jsonify({"message": "Login successful.", "user": user.id})
    else:
        return jsonify({"message": "Invalid credentials."})

# 查询用户信息
@app.route('/api/users/<int:user_id>', methods=['GET'])
def get_user(user_id):
    user = User.query.get(user_id)
    return jsonify(user)

if __name__ == '__main__':
    app.run(debug=True)
```

# instrument.py
from flask import Flask, request, jsonify
from flask_sqlalchemy import SQLAlchemy

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'mysql+pymysql://library_admin:password@localhost/shared_instrument_library'
db = SQLAlchemy(app)

    name = db.Column(db.String(50), nullable=False)
    type = db.Column(db.String(50), nullable=False)

    new_instrument = Instrument(name=data['name'], type=data['type'], status=data['status'])
    db.session.add(new_instrument)
    db.session.commit()

# 删除乐器
@app.route('/api/instruments/<int:instrument_id>', methods=['DELETE'])
