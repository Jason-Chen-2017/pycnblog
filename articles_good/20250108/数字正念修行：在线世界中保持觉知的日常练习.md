from flask import Flask, request, jsonify

app = Flask(__name__)

def connect_db():
    return mysql.connector.connect(
        host="localhost",
        user="root",
        password="password",
        database="digital_mindfulness"
    )

# 登录接口
@app.route('/login', methods=['POST'])
def login():
    # 获取用户名和密码
    username = request.form['username']
    password = request.form['password']
    
    # 连接数据库
    db = connect_db()
    cursor = db.cursor()
    
    # 查询用户信息
    cursor.execute("SELECT * FROM users WHERE username = %s AND password = %s", (username, password))
    user = cursor.fetchone()
    
    # 关闭数据库连接
    cursor.close()
    db.close()
    
    # 判断用户信息是否正确
    if user:
        return jsonify({"status": "success", "message": "登录成功"}), 200
    else:
        return jsonify({"status": "error", "message": "用户名或密码错误"}), 401

# 开始数字正念修行接口
@app.route('/start_practice', methods=['POST'])
def start_practice():
    # 获取用户ID和修行时间
    user_id = request.form['user_id']
    practice_time = request.form['practice_time']
    
    # 连接数据库
    db = connect_db()
    cursor = db.cursor()
    
    # 更新用户修行状态
    cursor.execute("UPDATE users SET practice_time = %s WHERE id = %s", (practice_time, user_id))
    db.commit()
    
    # 关闭数据库连接

    db.close()
    
    return jsonify({"status": "success", "message": "开始数字正念修行"}), 200
