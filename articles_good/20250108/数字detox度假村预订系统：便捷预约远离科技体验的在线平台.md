public ResponseEntity<?> createReservation(@RequestBody ReservationRequest reservationRequest) {
    Optional<User> userOptional = userRepository.findById(reservationRequest.getUserId());
    if (!userOptional.isPresent()) {
        return ResponseEntity.badRequest().body("Error: User not found!");
    }

    Room room = roomRepository.findById(reservationRequest.getRoomId())
        .orElseThrow(() -> new ResourceNotFoundException("Room not found with id: " + reservationRequest.getRoomId()));

    Reservation reservation = new Reservation();
    reservation.setUser(userOptional.get());
    reservation.setRoom(room);
    reservation.setReservationTime(reservationRequest.getReservationTime());
    reservation.setCheckoutTime(reservationRequest.getCheckoutTime());
    reservation.setNumberOfPeople(reservationRequest.getNumberOfPeople());
    reservation.setStatus(ReservationStatus.PENDING);

    reservationRepository.save(reservation);

    return ResponseEntity.ok("Reservation created successfully!");
}
