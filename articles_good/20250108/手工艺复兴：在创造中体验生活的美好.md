#### 6.2 系统核心实现源代码

以下是一个简单的手工艺复兴管理系统的核心实现源代码，包括数据库模型、路由定义和API接口：

```python
# app.py

from flask import Flask, request, jsonify
from flask_sqlalchemy import SQLAlchemy
from flask_restful import Api, Resource

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'mysql+pymysql://username:password@localhost/handicraft_renaissance'
db = SQLAlchemy(app)
api = Api(app)

# 数据库模型
class Craftsmanship(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(100), nullable=False)
    works = db.relationship('Work', backref='craftsman', lazy=True)

class Work(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    title = db.Column(db.String(100), nullable=False)
    craftsman_id = db.Column(db.Integer, db.ForeignKey('craftsmanship.id'), nullable=False)

# 路由定义
class CraftsmanshipResource(Resource):
    def get(self):
        craftsmen = Craftsmanship.query.all()
        return {'craftsmen': [{'id': c.id, 'name': c.name} for c in craftsmen]}

    def post(self):
        data = request.get_json()
        new_craftsman = Craftsmanship(name=data['name'])
        db.session.add(new_craftsman)
        db.session.commit()
        return {'id': new_craftsman.id}, 201

class WorkResource(Resource):
    def get(self, craftsmanship_id):
        craftsman = Craftsmanship.query.get(craftsmanship_id)
        if craftsman:
            works = Work.query.filter_by(craftsman_id=craftsmanship_id).all()
            return {'works': [{'id': w.id, 'title': w.title} for w in works]}
        return {'error': 'Craftsmanship not found'}, 404

    def post(self, craftsmanship_id):
        data = request.get_json()
        new_work = Work(title=data['title'], craftsman_id=craftsmanship_id)
        db.session.add(new_work)
        db.session.commit()
        return {'id': new_work.id}, 201

# API接口
api.add_resource(CraftsmanshipResource, '/craftsmen', '/craftsmen/<int:craftsmanship_id>')
api.add_resource(WorkResource, '/works', '/works/<int:craftsmanship_id>')

if __name__ == '__main__':
    db.create_all()
    app.run(debug=True)
```

在这个代码中，我们定义了两个数据库模型 `Craftsmanship`（手工艺人）和 `Work`（作品），并使用Flask-RESTful为这两个模型创建了相应的API接口。`CraftsmanshipResource` 负责处理手工艺人的创建和查询请求，`WorkResource` 负责处理作品的创建和查询请求。

  - `id`：主键，用于唯一标识手工艺人。
