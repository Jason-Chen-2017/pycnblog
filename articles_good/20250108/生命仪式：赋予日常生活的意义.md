   ```python
       return render_template('login.html', form=form)
   ```

```python
@app.route('/join_ceremony/<int:ritual_id>')
@login_required
def join_ceremony(ritual_id):
    ritual = Ritual.query.get(ritual_id)
    if ritual:
        # 检查用户是否已参与该婚礼
        if not Ceremony.query.filter_by(ritual_id=ritual_id, participant_id=current_user.id).first():
            ceremony = Ceremony(ritual_id=ritual_id, participant_id=current_user.id)
```
