# 引入必要的库
import random

# 灵感生成
def generate_insight():
    # 这里使用随机生成的方法来模拟灵感的产生
    return random.choice(["抽象", "表现", "超现实"])

# 创意组合
def combine_insights(insight1, insight2):
    # 这里使用简单的组合方法来模拟创意的产生
    return insight1 + "与" + insight2

# 创意评估
def evaluate_creativity(creativity):
    # 这里使用简单的评估方法来模拟创意的评估
    if "抽象" in creativity or "超现实" in creativity:
        return "具有创意"
    else:
        return "普通创意"

# 主函数
def main():
    insight1 = generate_insight()
    insight2 = generate_insight()
    creativity = combine_insights(insight1, insight2)
    evaluation = evaluate_creativity(creativity)
    print(f"创意生成：{creativity}")
    print(f"创意评估：{evaluation}")

# 执行主函数
if __name__ == "__main__":
    main()
```

在创意表达算法中，我们可以使用以下数学模型和公式：

1. 灵感生成概率模型：
   $$ P(\text{insight}) = \frac{\text{创意素材库中与当前情景匹配的素材数量}}{\text{创意素材库中所有素材的总数量}} $$

2. 创意组合模型：
   $$ C(\text{insight1}, \text{insight2}) = \text{insight1} + \text{insight2} $$

3. 创意评估模型：
   $$ E(\text{creativity}) = \begin{cases} 
   \text{"具有创意"} & \text{if } \text{抽象} \text{ 或 } \text{超现实} \text{ 出现在创意中} \\
   \text{"普通创意"} & \text{otherwise} 
   \end{cases} $$
