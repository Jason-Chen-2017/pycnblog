```python
import numpy as np
import matplotlib.pyplot as plt

class Node:
    def __init__(self, state, parent=None):
        self.state = state
        self.parent = parent
        self.children = []
        self.visits = 0
        self.reward = 0

    def select_child(self, exploration_factor=0.5):
        # 选择具有最大访问次数和最小均值的子节点
        return max(self.children, key=lambda child: (child.visits, -child.reward.mean()))

    def expand(self):
        # 根据当前状态生成子节点
        next_state = self.state.next_state()
        new_node = Node(next_state, self)
        self.children.append(new_node)
        return new_node

    def backpropagate(self, reward):
        # 更新节点的访问次数和奖励
        self.visits += 1
        self.reward += reward

def MCTS_search(node, num_simulations):
    for _ in range(num_simulations):
        current = node
        path = [current]
        
        while current.children:
            current = current.select_child()
            path.append(current)
        
        # 执行模拟并返回奖励
        reward = current.execute_simulation()
        path = path[::-1]
        
        for current in path:
            current.backpropagate(reward)
