### 4.1 家庭价值评估模型

家庭价值评估模型可以表示为：

$$
V_f = w_1 \cdot E_1 + w_2 \cdot E_2 + w_3 \cdot E_3
$$

其中，$E_1$ 为情感纽带得分，$E_2$ 为社会环境适应度，$E_3$ 为亲密关系得分，$w_1$、$w_2$、$w_3$ 为权重。

### 4.2 亲密关系评估模型

亲密关系评估模型可以表示为：

$$
R_i = \frac{1}{N} \sum_{i=1}^{N} I_i
$$

其中，$N$ 为互动次数，$I_i$ 为第 $i$ 次互动的亲密程度得分。

```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# 家庭价值评估模型
def evaluate_family_value(family_members):
    # 情感纽带得分
    E1 = np.mean(family_members['emotional_bond'])

    # 社会环境适应度
    E2 = np.mean(family_members['social_environment_adaptation'])

    # 亲密关系得分
    E3 = np.mean(family_members['emotional_intimacy'])

    # 权重
    w1, w2, w3 = 0.3, 0.3, 0.4

    # 家庭价值评估
    Vf = w1 * E1 + w2 * E2 + w3 * E3

    return Vf

# 亲密关系评估模型
def evaluate_intimacy_relationship(family_members):
    # 互动次数
    N = len(family_members)

    # 亲密程度得分
    Ii = np.mean(family_members['intimacy_level'])

    # 亲密关系评估
    Ri = Ii / N

    return Ri

# 数据处理
def process_data(family_members):
    # 情感纽带得分
    family_members['emotional_bond'] = family_members.apply(lambda x: 1 if x['communication_frequency'] > 5 else 0, axis=1)

    # 社会环境适应度
    family_members['social_environment_adaptation'] = family_members.apply(lambda x: 1 if x['economic_status'] == 'stable' else 0, axis=1)

    # 亲密关系得分
    family_members['emotional_intimacy'] = family_members.apply(lambda x: 1 if x['conflict_resolution'] == 'satisfied' else 0, axis=1)

    # 亲密程度得分
    family_members['intimacy_level'] = family_members.apply(lambda x: 1 if x['shared_goals'] == 'agreed' else 0, axis=1)

    return family_members

# 实例化DataFrame
family_members = pd.DataFrame({
    'communication_frequency': [3, 7, 4, 6],
    'economic_status': ['unstable', 'stable', 'stable', 'unstable'],
    'conflict_resolution': ['dissatisfied', 'satisfied', 'satisfied', 'dissatisfied'],
    'shared_goals': ['disagreed', 'agreed', 'disagreed', 'agreed']
})

# 数据处理
family_members_processed = process_data(family_members)

# 家庭价值评估
Vf = evaluate_family_value(family_members_processed)

# 亲密关系评估
Ri = evaluate_intimacy_relationship(family_members_processed)

# 输出结果
print("家庭价值评估得分：", Vf)
print("亲密关系评估得分：", Ri)
```
