import matplotlib.pyplot as plt
from scipy.integrate import solve_ivp

# 定义路径规划模型
def path_planning_model(t, y, params):
    x, y, v, theta = y
    alpha = params['alpha']
    k = params['k']
    x_des = params['x_des']
    y_des = params['y_des']
    
    # 目标点位置
    x_dot = v * np.cos(theta)
    y_dot = v * np.sin(theta)
    theta_dot = alpha
    
    # 控制输入
    u = k * (x_des - x)
    
    return [x_dot, y_dot, theta_dot, u]

# 参数设置
params = {
    'alpha': 0.1,
    'k': 1.0,
    'x_des': 10.0,
    'y_des': 0.0
}

# 初始条件
y0 = [0.0, 0.0, 0.0, 0.0]

# 时间范围
t_span = [0, 20]
t_eval = np.linspace(t_span[0], t_span[1], 100)

# 求解模型
solution = solve_ivp(path_planning_model, t_span, y0, args=(params,), t_eval=t_eval)

# 绘制结果
plt.plot(solution.t, solution.y[0], label='x(t)')
plt.plot(solution.t, solution.y[1], label='y(t)')
plt.legend()
plt.show()
