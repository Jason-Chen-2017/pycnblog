    if no empty cells:
        return True
    else:
        (row, col) = find_empty_cell(board)
        for num in range(1, 10):
            if is_valid(board, row, col, num):
                board[row][col] = num

                    return True

        return False

# 数独游戏伪代码
function solve_sudoku(board):
    if no empty cells:
        return True
    else:
        (row, col) = find_empty_cell(board)
        for num in range(1, 10):
            if is_valid(board, row, col, num):
                board[row][col] = num
                if solve_sudoku(board):
                    return True
                board[row][col] = 0
