#### 5.2.1 用户注册与登录

```python
from flask import Flask, request, jsonify
from werkzeug.security import generate_password_hash, check_password_hash

app = Flask(__name__)

# 用户注册
@app.route('/register', methods=['POST'])
def register():
    data = request.get_json()
    hashed_password = generate_password_hash(data['password'], method='sha256')
    # 存储用户信息到数据库（此处为简化，未实现具体数据库操作）
    users.append({
        'username': data['username'],
        'password': hashed_password
    })
    return jsonify({'message': '注册成功'})

# 用户登录
@app.route('/login', methods=['POST'])
def login():
    data = request.get_json()
    user = next((u for u in users if u['username'] == data['username']), None)
    if user and check_password_hash(user['password'], data['password']):
        return jsonify({'message': '登录成功'})
    else:
        return jsonify({'message': '用户名或密码错误'})

if __name__ == '__main__':
    app.run(debug=True)
```

#### 5.2.2 社区活动参与

```python
# 社区活动创建
@app.route('/activity/create', methods=['POST'])
def create_activity():
    data = request.get_json()
    activities.append({
        'title': data['title'],
        'description': data['description'],
        'participants': []
    })
    return jsonify({'message': '活动创建成功'})

# 社区活动参与
@app.route('/activity/join', methods=['POST'])
def join_activity():
    data = request.get_json()
    activity = next((a for a in activities if a['title'] == data['title']), None)
    if activity:
        activity['participants'].append(data['username'])
        return jsonify({'message': '参与活动成功'})
    else:
        return jsonify({'message': '活动不存在'})

if __name__ == '__main__':
    app.run(debug=True)
```
