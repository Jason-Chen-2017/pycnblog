    username = request.data.get('username')
    password = request.data.get('password')
    # 验证用户名和密码的有效性
    if not username or not password:
        return Response({'message': '用户名或密码不能为空'}, status=status.HTTP_400_BAD_REQUEST)
    
    # 创建用户
    user = User.objects.create_user(username=username, password=password)
    token = jwt_token_generator.get_token(user)
    
    return Response({'token': token, 'message': '用户注册成功'})

# 用户登录接口
@PostMapping("/login")
def login_user(request):
    username = request.data.get('username')
    password = request.data.get('password')
    
    user = authenticate(username=username, password=password)
    
    if not user:
        return Response({'message': '用户名或密码错误'}, status=status.HTTP_401_UNAUTHORIZED)
    
    token = jwt_token_generator.get_token(user)
    
    return Response({'token': token, 'message': '登录成功'})

@token_required
def analyze_emotion(request, user):
    user = get_user(request)
    start_date = request.data.get('start_date')
    end_date = request.data.get('end_date')
    
    if not start_date or not end_date:
        return Response({'message': '日期信息不完整'}, status.HTTP_400_BAD_REQUEST)
    
    emotions = Diary.objects.filter(user=user, created__range=(start_date, end_date))
    emotion_counts = emotions.values('emotion').annotate(count=Count('emotion'))
    
    return Response({'emotion_counts': emotion_counts})
