#### 源代码详细实现和代码解读

```python
from Cryptodome.PublicKey import ECC
from Cryptodome.Random import get_random_bytes
from Cryptodome.Util.number import long_to_bytes
from Cryptodome.Hash import SHA256

# 生成密钥对
def generate_key_pair():
    key = ECC.generate(curve=ECC.NIST256p())
    private_key = key.export_key(format='DER', use_pkcs8=True, preserve私钥=True)
    public_key = key.public_key().export_key(format='DER')
    return private_key, public_key

# 生成地址
def generate_address(public_key):
    sha256_hash = SHA256.new(public_key)
    hash_bytes = sha256_hash.digest()
    address = long_to_bytes(int.from_bytes(hash_bytes, 'big'), 20).hex()
    return address

# 签名
def sign_message(message, private_key):
    message_hash = SHA256.new(message.encode('utf-8'))
    message_hash_bytes = message_hash.digest()
    key = ECC.import_key(private_key)
    signature = key.sign(message_hash_bytes)
    return signature

# 验证签名
def verify_signature(message, signature, public_key):
    message_hash = SHA256.new(message.encode('utf-8'))
    message_hash_bytes = message_hash.digest()
    key = ECC.import_key(public_key)
    return key.verify(message_hash_bytes, signature)

# 主函数
if __name__ == "__main__":
    # 生成密钥对
    private_key, public_key = generate_key_pair()

    # 生成地址
    address = generate_address(public_key)
    print(f"生成的地址: {address}")

    # 签名
    message = "This is a test message."
    signature = sign_message(message, private_key)
    print(f"签名结果: {signature.hex()}")

    # 验证签名
    is_valid = verify_signature(message, signature, public_key)
    print(f"签名验证结果: {'有效' if is_valid else '无效'}")
```

2. **生成地址**：将公钥通过SHA256哈希后，转换为20字节的地址。
3. **签名**：使用私钥和SHA256哈希值进行签名。
4. **验证签名**：使用公钥和SHA256哈希值验证签名是否有效。
