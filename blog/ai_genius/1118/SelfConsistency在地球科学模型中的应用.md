def self_consistency_model(atm_temp, ocean_temp, geo_heat_flux, num_iterations):
    for i in range(num_iterations):
        # 计算大气层与海洋的热力学相互作用
        atm_to_ocean_heat = atm_temp - ocean_temp
        ocean_to_atm_heat = ocean_temp - atm_temp

        # 更新温度
        atm_temp += atm_to_ocean_heat / 2.0
        ocean_temp += ocean_to_atm_heat / 2.0

        # 考虑地质结构的热流动影响
        atm_temp += geo_heat_flux / 2.0
        ocean_temp -= geo_heat_flux / 2.0

        # 打印迭代信息
        print(f"Iteration {i+1}: ATM Temp = {atm_temp}K, Ocean Temp = {ocean_temp}K")

    return atm_temp, ocean_temp

# 运行模型
