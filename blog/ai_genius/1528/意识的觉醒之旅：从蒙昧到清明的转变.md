class AwakeningPractice:
    def __init__(self):
        self.meditation_sessions = []

    def add_meditation_session(self, duration, insights):
        self.meditation_sessions.append({
            'duration': duration,
            'insights': insights
        })

    def total_meditation_time(self):
        return sum(session['duration'] for session in self.meditation_sessions)

    def total_insights(self):
        return len(self.meditation_sessions)

# 使用示例
