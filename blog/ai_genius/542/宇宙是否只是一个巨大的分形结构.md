```python
import matplotlib.pyplot as plt
import numpy as np

def mandelbrot(c, max_iter):
    z = 0
    n = 0
    while abs(z) <= 2 and n < max_iter:
        z = z*z + c
        n += 1
    return n

def plot_mandelbrot(xmin, xmax, ymin, ymax, width, height, max_iter):
    real_min, real_max = xmin, xmax
    imag_min, imag_max = ymin, ymax
    dx = (real_max - real_min) / width
    dy = (imag_max - imag_min) / height

    x = np.arange(real_min, real_max + dx, dx)
    y = np.arange(imag_min, imag_max + dy, dy)
    X, Y = np.meshgrid(x, y)
    C = X + 1j * Y

    Z = np.zeros(C.shape, dtype=int)
    for i in range(height):
        for j in range(width):
            Z[i, j] = mandelbrot(C[i, j], max_iter)

    plt.imshow(Z, extent=(real_min, real_max, imag_min, imag_max))
    plt.colorbar()
    plt.title('Mandelbrot Set')
    plt.xlabel('Real axis')
    plt.ylabel('Imaginary axis')
    plt.show()

if __name__ == '__main__':
    plot_mandelbrot(-2.0, 1.0, -1.5, 1.5, 800, 600, 100)
```

```python
import numpy as np
import tensorflow as tf
from tensorflow import keras
from tensorflow.keras import layers

def generate_fractal_data samples, noise_level=0.1):
    X = np.random.uniform(-1, 1, (samples, 2))
    Y = np.zeros((samples, 1))
    for i, x in enumerate(X):
        z = x
        for _ in range(20):
            z = z * z + x
        Y[i] = np.linalg.norm(z)
    Y += noise_level * np.random.normal(size=samples)
    return X, Y

def build_fractal_model(input_shape):
    model = keras.Sequential([
        layers.Dense(64, activation='relu', input_shape=input_shape),
        layers.Dense(64, activation='relu'),
        layers.Dense(1)
    ])
    model.compile(optimizer='adam', loss='mean_squared_error')
    return model

if __name__ == '__main__':
    # 生成分形数据集
    X_train, Y_train = generate_fractal_data(1000)
    X_test, Y_test = generate_fractal_data(100)

    # 构建深度学习模型
    model = build_fractal_model(input_shape=(2,))

    # 训练模型
    model.fit(X_train, Y_train, epochs=50, batch_size=32, validation_data=(X_test, Y_test))

    # 预测
    predictions = model.predict(X_test)

    # 可视化
    plt.scatter(X_test[:, 0], X_test[:, 1], c=predictions[:, 0], cmap=plt.cm.Blues)
    plt.colorbar()
    plt.title('Fractal Data with Predictions')
    plt.xlabel('Real axis')
    plt.ylabel('Imaginary axis')
    plt.show()
```
