```python
import jinja2
import nltk
from nltk.tokenize import word_tokenize
from nltk.tag import pos_tag

# 1. 加载自然语言处理库
nltk.download('punkt')
nltk.download('averaged_perceptron_tagger')

# 2. 定义自然语言描述文本
description = "创建一个用户接口，用于添加、删除和查询用户信息。"

# 3. 分词和词性标注
tokens = word_tokenize(description)
tagged_tokens = pos_tag(tokens)

# 4. 语义分析（简单示例）
functions = []
data_structures = []
interfaces = []

for word, tag in tagged_tokens:
    if tag == 'VB':  # 动词
        functions.append(word)
    elif tag == 'NN':  # 名词
        data_structures.append(word)
    elif tag == 'NNP':  # 名词（专有名词）
        interfaces.append(word)

# 5. 定义模板
template = """
import {{ interface|titlecase }}
from data_structure import User

class {{ interface }}({{ interface|titlecase }}Interface):
    def add_user(self, user: User) -> None:
        # 添加用户逻辑
        pass

    def delete_user(self, user_id: int) -> None:
        # 删除用户逻辑
        pass

    def query_user(self, user_id: int) -> User:
        # 查询用户逻辑
        return User()
"""

# 6. 渲染模板
jinja_env = jinja2.Environment(trim_blocks=True)
template = jinja_env.from_string(template)

result = template.render(interface=interfaces[0], data_structure=data_structures[0])

# 7. 输出生成的代码
with open('generated_code.py', 'w') as f:
    f.write(result)

print("生成的代码已保存到generated_code.py")
```
