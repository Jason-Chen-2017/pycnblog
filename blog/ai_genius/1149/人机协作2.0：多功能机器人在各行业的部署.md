import rospy
from std_msgs.msg import String
from sensor_msgs.msg import Image
import cv2
import numpy as np

# 初始化节点
rospy.init_node('human_robot_collaboration')

# 初始化话题发布者和订阅者
cmd_vel_pub = rospy.Publisher('/cmd_vel', String, queue_size=10)
robot_state_sub = rospy.Subscriber('/robot_state', String, callback)

# 设置机器人速度的函数
def set_speed(linear_speed, angular_speed):
    velocity_message = String()
    velocity_message.data = f"linear_speed={linear_speed},angular_speed={angular_speed}"
    cmd_vel_pub.publish(velocity_message)

# 机器人状态回调函数
def callback(data):
    # 假设机器人状态包含图像数据
    image_data = data.data
    
    # 使用OpenCV处理图像
    image_array = np.frombuffer(image_data, dtype=np.uint8)
    image = cv2.imdecode(image_array, cv2.IMREAD_COLOR)
    
    # 简单的图像处理示例
    gray_image = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    _, thresh = cv2.threshold(gray_image, 128, 255, cv2.THRESH_BINARY_INV + cv2.THRESH_OTSU)
    
    # 显示处理后的图像
    cv2.imshow('Processed Image', thresh)
    cv2.waitKey(1)

# 主函数
def main():
    rate = rospy.Rate(10)  # 10 Hz
    
    while not rospy.is_shutdown():
        # 假设我们根据图像处理结果设置机器人的速度
        set_speed(linear_speed=0.5, angular_speed=-0.5)
        rate.sleep()

if __name__ == '__main__':
    main()
```

1. **初始化节点**：使用`rospy.init_node('human_robot_collaboration')`初始化ROS节点。
2. **发布和订阅话题**：创建一个话题发布者`cmd_vel_pub`和一个话题订阅者`robot_state_sub`，用于发送速度命令和接收机器人状态。
3. **设置机器人速度**：定义`set_speed`函数，通过发布速度消息控制机器人的线性速度和角速度。
4. **机器人状态回调函数**：定义`callback`函数，处理接收到的机器人状态数据，这里使用OpenCV库处理图像数据。
5. **主函数**：在`main`函数中，以10Hz的频率循环调用`set_speed`函数，实现简单的机器人控制。

import rospy
from std_msgs.msg import String
from sensor_msgs.msg import Image
import cv2
import numpy as np

# 初始化节点
rospy.init_node('human_robot_collaboration')

# 初始化话题发布者和订阅者
cmd_vel_pub = rospy.Publisher('/cmd_vel', String, queue_size=10)
robot_state_sub = rospy.Subscriber('/robot_state', String, callback)

# 设置机器人速度的函数
def set_speed(linear_speed, angular_speed):
    velocity_message = String()
    velocity_message.data = f"linear_speed={linear_speed},angular_speed={angular_speed}"
    cmd_vel_pub.publish(velocity_message)

# 机器人状态回调函数
def callback(data):
    # 假设机器人状态包含图像数据
    image_data = data.data
    
    # 使用OpenCV处理图像
    image_array = np.frombuffer(image_data, dtype=np.uint8)
    image = cv2.imdecode(image_array, cv2.IMREAD_COLOR)
    
    # 简单的图像处理示例
    gray_image = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    _, thresh = cv2.threshold(gray_image, 128, 255, cv2.THRESH_BINARY_INV + cv2.THRESH_OTSU)
    
    # 显示处理后的图像
    cv2.imshow('Processed Image', thresh)
    cv2.waitKey(1)

# 主函数
def main():
    rate = rospy.Rate(10)  # 10 Hz
    
    while not rospy.is_shutdown():
        # 假设我们根据图像处理结果设置机器人的速度
        set_speed(linear_speed=0.5, angular_speed=-0.5)
        rate.sleep()

if __name__ == '__main__':
    main()
```

3. **设置机器人速度**：定义`set_speed`函数，通过发布速度消息控制机器人的线性速度和角速度。
4. **机器人状态回调函数**：定义`callback`函数，处理接收到的机器人状态数据，这里使用OpenCV库处理图像数据。
5. **主函数**：在`main`函数中，以10Hz的频率循环调用`set_speed`函数，实现简单的机器人控制。
