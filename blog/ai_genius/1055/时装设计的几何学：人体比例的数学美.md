graph TB

C --> F[腿长: {}cm]
C --> G[手臂: {}cm]
""".format(head_height, shoulder_width, waist_circumference, hip_width, leg_length, arm_length)

mermaid.plot(mermaid_code)

# 使用LaTeX格式展示人体比例公式
latex_formulas = r"""
$$
\begin{aligned}
L_{head} &= \frac{h}{7.5}, \\
L_{shoulders} &= \frac{h}{4}, \\

L_{arms} &= \frac{h}{4.6}, \\
L_{legs} &= h - L_{head} - L_{shoulders} - L_{waist} - L_{hips}.
\end{aligned}
$$
"""

print(latex_formulas)

# 绘制人体比例图
fig, ax = plt.subplots()
ax.bar(['头部', '肩膀', '腰部', '臀部', '手臂', '腿长'], [head_height, shoulder_width, waist_circumference, hip_width, arm_length, leg_length])
ax.set_ylabel('长度（厘米）')
ax.set_title('人体比例图')
plt.show()
