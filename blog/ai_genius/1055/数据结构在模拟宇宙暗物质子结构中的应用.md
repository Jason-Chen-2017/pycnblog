##### 1.1 数据结构的概念

数据结构是指用于存储数据的组织方式。根据数据的存储方式，数据结构可以分为以下几种类型：

常见的图形数据结构包括：

- **图**：图是一种由节点和边组成的数据结构，用于表示复杂的关系。

```python
import numpy as np
import networkx as nx
import matplotlib.pyplot as plt

def create_particles(num_particles, radius):
    positions = np.random.uniform(-radius, radius, (num_particles, 2))
    masses = np.random.uniform(1, 10, num_particles)
    velocities = np.random.uniform(-1, 1, (num_particles, 2))
    return np.hstack((positions, masses[:, np.newaxis], velocities))

def simulate_particles(particles, G, dt, steps):
    for _ in range(steps):
        force = compute_gravity_force(particles, G)
        for i in range(len(particles)):
            particles[i][0] += particles[i][3] * dt
            particles[i][1] += particles[i][4] * dt
            particles[i][3] += force[i][0] / particles[i][2] * dt
            particles[i][4] += force[i][1] / particles[i][2] * dt
    return particles

def plot_particles(particles):
    x = particles[:, 0]
    y = particles[:, 1]
    plt.scatter(x, y)
    plt.xlabel('X')
    plt.ylabel('Y')
    plt.show()

# 初始化参数
num_particles = 100
radius = 100
G = 6.67430e-11  # 万有引力常数
dt = 1e-5        # 时间步长
steps = 1000      # 模拟步数

# 创建粒子
particles = create_particles(num_particles, radius)

# 模拟宇宙
particles = simulate_particles(particles, G, dt, steps)

# 分析结果
plot_particles(particles)
```
