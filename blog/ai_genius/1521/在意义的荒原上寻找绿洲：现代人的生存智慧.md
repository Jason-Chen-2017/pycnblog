3. 意义评估：根据分析结果，评估个人的生活状态，判断是否存在意义危机。

```python
# 数据收集
def collect_data():
    # 收集用户生活事件数据
    life_events = []
    event_name = input("请输入生活事件名称：")
    event_date = input("请输入生活事件发生日期：")
    event_type = input("请输入生活事件类型：")
    life_events.append({'name': event_name, 'date': event_date, 'type': event_type})
    return life_events

# 数据分析
def analyze_data(life_events):
    # 分析用户生活事件数据
    analysis_results = {}
    for event in life_events:
        # 计算生活事件权重
        weight = calculate_weight(event)
        # 计算生活事件满意度
        satisfaction = calculate_satisfaction(event)
        # 存储分析结果
        analysis_results[event['name']] = {'weight': weight, 'satisfaction': satisfaction}
    return analysis_results

# 意义评估
def evaluate_meaning(analysis_results):
    # 评估用户意义评分
    total_weight = sum([result['weight'] for result in analysis_results.values()])
    total_satisfaction = sum([result['satisfaction'] for result in analysis_results.values()])
    meaning_score = total_weight / total_satisfaction
    return meaning_score

# 结果呈现
def present_result(meaning_score):
    # 呈现评估结果
    print("您的意义评分：", meaning_score)
    if meaning_score < 0.5:
        print("您的生命状态较差，建议寻求专业帮助。")
    elif meaning_score >= 0.5 and meaning_score < 0.7:
        print("您的生命状态一般，可以尝试调整生活状态。")
    else:
        print("您的生命状态良好，请继续保持。")

# 主程序
def main():
    # 收集用户生活事件数据
    life_events = collect_data()
    # 分析用户生活事件数据
    analysis_results = analyze_data(life_events)
    # 评估用户意义评分
    meaning_score = evaluate_meaning(analysis_results)
    # 呈现评估结果
    present_result(meaning_score)

if __name__ == "__main__":
    main()
```
