```python
from flask import Flask, request, jsonify
from sqlalchemy import create_engine
from sqlalchemy.orm import sessionmaker

app = Flask(__name__)

# 数据库连接配置
DATABASE_URI = '数据库连接字符串'
engine = create_engine(DATABASE_URI)
Session = sessionmaker(bind=engine)

# 配方模型
class CocktailModel:
    def __init__(self, user_id, base_liquid, flavoring, ice_cubes):
        self.user_id = user_id
        self.base_liquid = base_liquid
        self.flavoring = flavoring
        self.ice_cubes = ice_cubes

    def save_to_database(self, session):
        session.add(self)
        session.commit()

# API接口
@app.route('/create_cocktail', methods=['POST'])
def create_cocktail():
    data = request.json
    user_id = data['user_id']
    base_liquid = data['base_liquid']
    flavoring = data['flavoring']
    ice_cubes = data['ice_cubes']

    session = Session()
    cocktail = CocktailModel(user_id, base_liquid, flavoring, ice_cubes)
    cocktail.save_to_database(session)

    return jsonify({'message': 'Cocktail created successfully.'})

@app.route('/get_cocktail', methods=['GET'])
def get_cocktail():
    user_id = request.args.get('user_id')
    session = Session()
    cocktails = session.query(CocktailModel).filter_by(user_id=user_id).all()

    return jsonify([{'id': cocktail.id, 'base_liquid': cocktail.base_liquid, 'flavoring': cocktail.flavoring, 'ice_cubes': cocktail.ice_cubes} for cocktail in cocktails])

if __name__ == '__main__':
    app.run(debug=True)
```
