import numpy as np
import matplotlib.pyplot as plt

# 评估团队凝聚力
def assess_team_cohesion(surveys):
    scores = [survey['cohesion_score'] for survey in surveys]
    average_score = np.mean(scores)
    return average_score

# 可视化团队凝聚力
def visualize_team_cohesion(surveys):
    scores = [survey['cohesion_score'] for survey in surveys]
    plt.scatter(range(len(surveys)), scores)
    plt.xlabel('Team Member')
    plt.ylabel('Cohesion Score')
    plt.title('Team Cohesion Assessment')
    plt.show()

# 测试代码
surveys = [
    {'team_member': 'Alice', 'cohesion_score': 8.5},
    {'team_member': 'Bob', 'cohesion_score': 7.0},
    {'team_member': 'Charlie', 'cohesion_score': 9.0},
]

average_score = assess_team_cohesion(surveys)
print(f'Average Team Cohesion Score: {average_score:.2f}')

visualize_team_cohesion(surveys)
