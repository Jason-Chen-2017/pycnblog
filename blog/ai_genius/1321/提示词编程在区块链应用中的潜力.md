            'index': len(self.chain) + 1,
            'timestamp': time(),
            'transactions': self.current_transactions,

            'previous_hash': previous_hash or self.hash(self.chain[-1]),

            proof += 1
        return proof

### 代码实现

#### 1. 智能合约（Solidity）

以下是一个简单的智能合约示例，实现数字货币交易功能：

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract DeFiPlatform {
    mapping(address => uint256) public balances;

    function deposit() external payable {
        balances[msg.sender()] += msg.value;
    }

    function withdraw(uint256 amount) external {
        require(balances[msg.sender()] >= amount, "余额不足");
        balances[msg.sender()] -= amount;
        payable(msg.sender()).transfer(amount);
    }

    function getBalance() external view returns (uint256) {
        return balances[msg.sender()];
    }
}
```

#### 2. 前端应用（HTML/JavaScript）

以下是一个简单的HTML界面，用于与智能合约进行交互：

```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>DeFi Platform</title>
</head>
<body>
    <h1>DeFi Platform</h1>
    <div>
        <p>您的余额：<span id="balance">0</span> ETH</p>
        <button onclick="deposit()">存款</button>
        <button onclick="withdraw()">取款</button>
    </div>
    <script src="https://cdnjs.cloudflare.com/ajax/libs/web3/1.2.8/web3.min.js"></script>
    <script>
        // 初始化Web3
        const web3 = new Web3(window.ethereum);
        // 连接到以太坊网络
        window.ethereum.request({ method: 'eth_requestAccounts' })
            .then(accounts => {
                web3.eth.defaultAccount = accounts[0];
                // 加载智能合约
                const contractAddress = '0x...';
                const contractAbi = [
                    {
                        "inputs": [{"internalType": "address", "name": "user", "type": "address"}],
                        "stateMutability": "nonpayable",
                        "type": "constructor"
                    },
                    {
                        "anonymous": false,
                        "inputs": [{"indexed": true, "internalType": "address", "name": "user", "type": "address"},
                                    {"indexed": false, "internalType": "uint256", "name": "amount", "type": "uint256"}],
                        "name": "Deposit",
                        "type": "event"
                    },
                    ...
                ];
                const contract = new web3.eth.Contract(contractAbi, contractAddress);
                // 显示余额
                contract.methods.getBalance().call({ from: web3.eth.defaultAccount })
                    .then(balance => {
                        document.getElementById('balance').textContent = web3.utils.fromWei(balance, 'ether');
                    });
                // 存款
                function deposit() {
                    contract.methods.deposit().send({ from: web3.eth.defaultAccount, value: web3.utils.toWei('1', 'ether') })
                        .on('transactionHash', (hash) => {
                            console.log(`Deposit transaction hash: ${hash}`);
                        })
                        .on('confirmation', (confirmationNumber, receipt) => {
                            console.log(`Deposit transaction confirmed: ${confirmationNumber}`);
                        })
                        .on('error', (error) => {
                            console.error(`Deposit transaction error: ${error.message}`);
                        });
                }
                // 取款
                function withdraw() {
                    contract.methods.withdraw(web3.utils.toWei('0.5', 'ether')).send({ from: web3.eth.defaultAccount })
                        .on('transactionHash', (hash) => {
                            console.log(`Withdraw transaction hash: ${hash}`);
                        })
                        .on('confirmation', (confirmationNumber, receipt) => {
                            console.log(`Withdraw transaction confirmed: ${confirmationNumber}`);
                        })
                        .on('error', (error) => {
                            console.error(`Withdraw transaction error: ${error.message}`);
                        });
                }
            });
    </script>
</body>
</html>
```
