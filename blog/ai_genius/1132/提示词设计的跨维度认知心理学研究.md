import tensorflow as tf
from tensorflow.keras.layers import Input, Dense
from tensorflow.keras.models import Model

# 注意力模型
input_data = Input(shape=(10,))
attention_weights = Dense(units=1, activation='sigmoid')(input_data)

memory_weights = Dense(units=1, activation='sigmoid')(input_data)
memory_model = Model(inputs=input_data, outputs=memory_weights)

thinking_weights = Dense(units=1, activation='sigmoid')(input_data)
thinking_model = Model(inputs=input_data, outputs=thinking_weights)

attention_model.compile(optimizer='adam', loss='binary_crossentropy')
