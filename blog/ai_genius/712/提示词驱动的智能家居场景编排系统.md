```python
def recognize_intent(input_text):
    # 初始化意图词典
    intents = {
        'turn_on_light': ['turn on the light', 'turn the light on'],
        'turn_off_light': ['turn off the light', 'turn the light off'],
        'adjust_temperature': ['adjust the temperature', 'set the temperature'],
        # ... 其他意图
    }
    
    # 将输入文本转换为小写
    input_text = input_text.lower()
    
    # 遍历意图词典，查找匹配的意图
    for intent, phrases in intents.items():
        for phrase in phrases:
            if phrase in input_text:
                return intent
    
    # 如果没有找到匹配的意图，返回None
    return None
```
