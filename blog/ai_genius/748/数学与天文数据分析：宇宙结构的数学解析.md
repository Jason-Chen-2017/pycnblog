import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import curve_fit

# 数据读取
data = np.loadtxt('star_data.txt')

# 数据预处理
positions = data[:, 0]
velocity = data[:, 1]
distance = np.sqrt(np.sum(positions**2, axis=1))

# 曲线拟合
def rotational_curve_model(distance, G, M):
    return np.sqrt(G * M / distance)

params, cov = curve_fit(rotational_curve_model, distance, velocity)

# 结果分析
fitted_curve = rotational_curve_model(distance, *params)
plt.plot(distance, velocity, 'o', distance, fitted_curve, '-')
plt.xlabel('Distance')
plt.ylabel('Velocity')
plt.title('Rotational Curve Fitting')
plt.show()
```
