```python
# 示例：使用Python编写一个简单的时间管理脚本

import datetime

def plan_daily_schedule(start_time, end_time, tasks):
    schedule = {}
    current_time = datetime.datetime.strptime(start_time, "%H:%M")
    end_time = datetime.datetime.strptime(end_time, "%H:%M")

    for task in tasks:
        start, end = task
        start_time = datetime.datetime.strptime(start, "%H:%M")
        end_time = datetime.datetime.strptime(end, "%H:%M")
        schedule[start_time.strftime("%H:%M")] = (start_time, end_time)

        while start_time < end_time:
            current_time += datetime.timedelta(minutes=30)
            if current_time in schedule:
                start_time = schedule[current_time][0]
            else:
                schedule[current_time.strftime("%H:%M")] = (current_time, end_time)
                break

    return schedule

start_time = "08:00"
end_time = "20:00"
tasks = [("09:00", "11:00"), ("14:00", "17:00"), ("18:00", "20:00")]

schedule = plan_daily_schedule(start_time, end_time, tasks)
print("每日时间安排：")
for time, (start, end) in schedule.items():
    print(f"{time} - {end.strftime('%H:%M')}")
```
