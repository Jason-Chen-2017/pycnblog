以下是一个简单的 Q-学习模型构建的 Python 代码示例：

```python
import pandas as pd
import numpy as np
from collections import defaultdict

# 加载数据
data = pd.read_csv('exercise_data.csv')

# 分割特征和目标变量
X = data[['duration', 'intensity']]
y = data['health_impact']

# 初始化Q表
Q = defaultdict(lambda: np.zeros(100))

# 定义Q-学习模型
def q_learning(X, y, alpha=0.1, gamma=0.9, epsilon=0.1, episodes=1000):
    for episode in range(episodes):
        state = np.random.randint(0, 100)
        action = np.argmax(Q[state])
        next_state = np.random.randint(0, 100)
        reward = y[next_state] - y[state]
        Q[state][action] += alpha * (reward + gamma * np.max(Q[next_state]) - Q[state][action])
        state = next_state

# 训练模型
q_learning(X, y)

# 输出Q表
for state, action_values in Q.items():
    print(f'State {state}: Actions {action_values}')
```
