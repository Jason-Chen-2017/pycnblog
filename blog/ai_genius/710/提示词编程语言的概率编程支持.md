```python
import random
import matplotlib.pyplot as plt

# 随机变量定义
def random_walk(steps):
    x = 0
    y = 0
    for _ in range(steps):
        x += random.choice([-1, 1])
        y += random.choice([-1, 1])
    return x, y

# 主程序

    plt.show()

if __name__ == "__main__":
    main()
```
