import math

def barber_shop_problem(people, time):

    while people:
        time_elapsed += time
        if people[0] <= math.sqrt(time_elapsed):
            people.pop(0)
        else:
            people.append(people.pop(0)**2)
    return time_elapsed

def decrypt(ciphertext, private_key):
    cipher = PKCS1_OAEP.new(private_key)
