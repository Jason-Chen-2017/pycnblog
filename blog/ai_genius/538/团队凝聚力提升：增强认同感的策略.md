#### 算法原理：

    """
    评估团队凝聚力。
    
    参数:
    - network: 社会网络图，表示团队成员之间的交互关系。
    
    返回:
    - cohesion_score: 团队凝聚力得分。
    """
    
    # 计算平均边密度
    avg_edge_density = calculate_average_edge_density(network)
    
    # 计算群体平均最小生成树宽度
    avg_min_spanning_tree_width = calculate_average_mst_width(network)
    
    # 计算凝聚力得分
    cohesion_score = avg_edge_density / avg_min_spanning_tree_width
    
    return cohesion_score
```

    total_edges = len(network.edges)
    total_nodes = len(network.nodes)

    average_mst_width = sum(mst_widths) / len(mst_widths)

    cohesion_score = avg_edge_density / avg_min_spanning_tree_width
    
    return cohesion_score
```

```python
def calculate_average_edge_density(network):
    total_edges = len(network.edges)
    total_nodes = len(network.nodes)
    average_edge_density = total_edges / total_nodes
    return average_edge_density

def calculate_average_mst_width(network):
    mst_widths = []
    for node in network.nodes:
        mst_width = calculate_mst_width_for_node(network, node)
        mst_widths.append(mst_width)
    average_mst_width = sum(mst_widths) / len(mst_widths)
    return average_mst_width

def calculate_mst_width_for_node(network, node):
    # 使用Prim算法计算以node为根的最小生成树宽度
    mst = prim_algorithm(network, node)
    mst_width = sum(edge.weight for edge in mst.edges)
    return mst_width
```
