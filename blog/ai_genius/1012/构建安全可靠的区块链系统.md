// 简单的区块链实现（Go语言）

package main

import (
    "crypto/sha256"
    "encoding/hex"
    "fmt"
)

type Block struct {
    Index     int
    Timestamp string
    Data      string
    Hash      string
    PrevHash  string
}

func CreateGenesisBlock() *Block {
    return &Block{
        Index:     0,
        Timestamp: "2023-10-01T00:00:00Z",
        Data:      "Genesis Block",
        Hash:      calculateHash("0"),
        PrevHash:  "",
    }
}

func calculateHash(block *Block) string {
    blockString := fmt.Sprintf("%d%d%d%d",
        block.Index, block.Timestamp, block.Data, block.PrevHash)
    hash := sha256.Sum256([]byte(blockString))
    return hex.EncodeToString(hash[:])
}

func main() {
    blockchain := []*Block{}
    blockchain = append(blockchain, CreateGenesisBlock())

    // 添加区块到区块链
    blockchain = append(blockchain, &Block{
        Index:     1,
        Timestamp: "2023-10-02T00:00:00Z",
        Data:      "Second Block",
        Hash:      calculateHash(blockchain[len(blockchain)-1]),
        PrevHash:  blockchain[len(blockchain)-1].Hash,
    })

    // 打印区块链
    for _, block := range blockchain {
        fmt.Println(*block)
    }
}
```

package main

import (
    "crypto/sha256"
    "encoding/hex"
    "fmt"
)

type Block struct {
    Index     int
    Timestamp string
    Data      string
    Hash      string
    PrevHash  string
}

func CreateGenesisBlock() *Block {
    return &Block{
        Index:     0,
        Timestamp: "2023-10-01T00:00:00Z",
        Data:      "Genesis Block",
        Hash:      calculateHash("0"),
        PrevHash:  "",
    }
}

func calculateHash(block *Block) string {
    blockString := fmt.Sprintf("%d%d%d%d",
        block.Index, block.Timestamp, block.Data, block.PrevHash)
    hash := sha256.Sum256([]byte(blockString))
    return hex.EncodeToString(hash[:])
}

func main() {
    blockchain := []*Block{}
    blockchain = append(blockchain, CreateGenesisBlock())

    // 添加区块到区块链
    blockchain = append(blockchain, &Block{
        Index:     1,
        Timestamp: "2023-10-02T00:00:00Z",
        Data:      "Second Block",
        Hash:      calculateHash(blockchain[len(blockchain)-1]),
        PrevHash:  blockchain[len(blockchain)-1].Hash,
    })

    // 打印区块链
    for _, block := range blockchain {
        fmt.Println(*block)
    }
}
