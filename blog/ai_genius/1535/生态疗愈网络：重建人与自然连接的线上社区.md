```python
import numpy as np
from sklearn.metrics.pairwise import cosine_similarity

def user_recommendation(user_vector, activity_vectors, similarity_threshold=0.5):
    """
    用户推荐算法实现
    :param user_vector: 用户特征向量
    :param activity_vectors: 活动特征向量列表
    :param similarity_threshold: 相似度阈值
    :return: 推荐活动列表
    """
    # 计算用户与活动之间的相似度矩阵
    similarity_matrix = cosine_similarity([user_vector], activity_vectors)
    
    # 筛选出相似度高于阈值的推荐活动
    recommendations = []
    for i, similarity in enumerate(similarity_matrix[0]):
        if similarity > similarity_threshold:
            recommendations.append(i)
    
    return recommendations
```
