def perceptron(x, w, b):
    z = np.dot(x, w) + b
    return 1 if z > 0 else 0

# 线性回归模型
def linear_regression(x, y, w, b):
    z = np.dot(x, w) + b
    return z

# 逻辑回归模型
def logistic_regression(x, y, w, b):
    z = linear_regression(x, y, w, b)
    return 1 / (1 + np.exp(-z))

def train_perceptron(x, y, epochs, learning_rate):
    w = np.zeros(x.shape[1])
    b = 0
    for _ in range(epochs):
        for xi, yi in zip(x, y):
            prediction = perceptron(xi, w, b)
            if prediction != yi:
                w += learning_rate * (yi - prediction) * xi
                b += learning_rate * (yi - prediction)
    return w, b

# 线性回归训练
def train_linear_regression(x, y, epochs, learning_rate):
    w = np.zeros(x.shape[1])
    b = np.zeros(x.shape[0])
    for _ in range(epochs):
        z = linear_regression(x, y, w, b)
        error = z - y
        w -= learning_rate * np.dot(x.T, error)
        b -= learning_rate * np.sum(error)
    return w, b

# 逻辑回归训练
def train_logistic_regression(x, y, epochs, learning_rate):

    b = np.zeros(x.shape[0])
    for _ in range(epochs):
        z = logistic_regression(x, y, w, b)
        error = z - y
        w -= learning_rate * np.dot(x.T, error * (1 - z))
        b -= learning_rate * np.sum(error * (1 - z))
    return w, b
