#### 3.2 算法实践

在本节中，我们将介绍一种常见的优化算法——遗传算法（Genetic Algorithm，GA），并使用Python源代码进行实现。

**遗传算法原理**：

遗传算法是一种基于生物进化的优化算法，通过模拟自然选择和遗传机制，搜索最优解。遗传算法的基本流程如下：

1. **初始化种群**：随机生成一组初始解。
2. **适应度评估**：对种群中的每个解进行适应度评估。
3. **选择**：根据适应度高低选择优胜个体。
4. **交叉**：对选中个体进行交叉操作，生成新个体。
5. **变异**：对个体进行变异操作，增加种群多样性。
6. **迭代**：重复执行选择、交叉、变异操作，直至满足停止条件。

**Python代码实现**：

```python
import random

def fitness_function(chromosome):
    # 计算适应度
    return sum(chromosome)

def selection(population, fitness_values, num_parents):
    # 选择操作
    selected_parents = []
    for _ in range(num_parents):
        max_fitness = max(fitness_values)
        selected_parents.append(population[fitness_values.index(max_fitness)])
        fitness_values[fitness_values.index(max_fitness)] = -1
    return selected_parents

def crossover(parent1, parent2):
    # 交叉操作
    crossover_point = random.randint(1, len(parent1) - 1)
    child1 = parent1[:crossover_point] + parent2[crossover_point:]
    child2 = parent2[:crossover_point] + parent1[crossover_point:]
    return child1, child2

def mutate(chromosome, mutation_rate):
    # 变异操作
    for i in range(len(chromosome)):
        if random.random() < mutation_rate:
            chromosome[i] = 1 - chromosome[i]
    return chromosome

def genetic_algorithm(population_size, chromosome_length, mutation_rate, num_generations):
    # 遗传算法主函数
    population = [[random.randint(0, 1) for _ in range(chromosome_length)] for _ in range(population_size)]
    for _ in range(num_generations):
        fitness_values = [fitness_function(chromosome) for chromosome in population]
        selected_parents = selection(population, fitness_values, population_size // 2)
        next_generation = []
        for _ in range(population_size // 2):
            parent1, parent2 = random.sample(selected_parents, 2)
            child1, child2 = crossover(parent1, parent2)
            next_generation.append(mutate(child1, mutation_rate))
            next_generation.append(mutate(child2, mutation_rate))
        population = next_generation
    return population

# 示例
population_size = 100
chromosome_length = 10
mutation_rate = 0.01
num_generations = 100

best_solution = genetic_algorithm(population_size, chromosome_length, mutation_rate, num_generations)
print("最优解：", best_solution)
print("最优解适应度：", fitness_function(best_solution))
```

#### Python源代码

以下是使用Python实现的遗传算法（GA）示例代码：

```python
import random

def fitness_function(chromosome):
    # 计算适应度
    return sum(chromosome)

def selection(population, fitness_values, num_parents):
    # 选择操作
    selected_parents = []
    for _ in range(num_parents):
        max_fitness = max(fitness_values)
        selected_parents.append(population[fitness_values.index(max_fitness)])
        fitness_values[fitness_values.index(max_fitness)] = -1
    return selected_parents

def crossover(parent1, parent2):
    # 交叉操作
    crossover_point = random.randint(1, len(parent1) - 1)
    child1 = parent1[:crossover_point] + parent2[crossover_point:]
    child2 = parent2[:crossover_point] + parent1[crossover_point:]
    return child1, child2

def mutate(chromosome, mutation_rate):
    # 变异操作
    for i in range(len(chromosome)):
        if random.random() < mutation_rate:
            chromosome[i] = 1 - chromosome[i]
    return chromosome

def genetic_algorithm(population_size, chromosome_length, mutation_rate, num_generations):
    # 遗传算法主函数
    population = [[random.randint(0, 1) for _ in range(chromosome_length)] for _ in range(population_size)]
    for _ in range(num_generations):
        fitness_values = [fitness_function(chromosome) for chromosome in population]
        selected_parents = selection(population, fitness_values, population_size // 2)
        next_generation = []
        for _ in range(population_size // 2):
            parent1, parent2 = random.sample(selected_parents, 2)
            child1, child2 = crossover(parent1, parent2)
            next_generation.append(mutate(child1, mutation_rate))
            next_generation.append(mutate(child2, mutation_rate))
        population = next_generation
    return population

# 示例
population_size = 100
chromosome_length = 10
mutation_rate = 0.01
num_generations = 100

best_solution = genetic_algorithm(population_size, chromosome_length, mutation_rate, num_generations)
print("最优解：", best_solution)
print("最优解适应度：", fitness_function(best_solution))
```
