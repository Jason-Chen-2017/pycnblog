```python
import numpy as np
import matplotlib.pyplot as plt
from scipy.integrate import solve_ivp

# 初始条件
N = 1000
L = 100
M = 1e12
G = 6.6743e-11

# 计算引力
def compute_gravity(positions, masses):
    forces = np.zeros_like(positions)
    for i, pos_i in enumerate(positions):
        for j, pos_j in enumerate(positions):
            if i != j:
                distance = np.linalg.norm(pos_i - pos_j)
                force = G * masses[i] * masses[j] / distance**2
                force_vector = (pos_j - pos_i) * force / distance
                forces[i] += force_vector
    return forces

# 模拟宇宙演化
def simulate_universe(positions, masses, t):
    forces = compute_gravity(positions, masses)
    velocities = positions / t
    accelerations = forces / masses
    return velocities + accelerations

# 求解微分方程
tspan = (0, 100)
t_eval = np.linspace(0, 100, 1000)
positions = np.random.uniform(size=(N, 3))
masses = np.random.uniform(size=N) * M

solution = solve_ivp(simulate_universe, tspan, [positions, masses], t_eval=t_eval, rtol=1e-6, atol=1e-9)

# 可视化展示
plt.scatter(solution.y[0], solution.y[1], s=10, c=solution.t, cmap='viridis')
plt.xlabel('X')
plt.ylabel('Y')
plt.colorbar(label='Time')
plt.show()
```
