#### 4.3 源代码实现

```python
import numpy as np
import random

# 初始化参数
N = 10  # 生产线数量
M = 5   # 生产任务数量
CROSSOVER_RATE = 0.8  # 交叉率
MUTATION_RATE = 0.05  # 变异率

# 目标函数
def objective_function(solution):
    total_efficiency = 0
    for task in solution:
        total_efficiency += task['efficiency']
    return total_efficiency

# 约束条件
def constraints(solution):
    for task in solution:
        if task['time'] > 24:
            return False
    return True

# 遗传算法
def genetic_algorithm(population):
    while not constraints(population):
        new_population = []
        for _ in range(len(population)):
            parent1 = random.choice(population)
            parent2 = random.choice(population)
            child = crossover(parent1, parent2)
            child = mutation(child)
            new_population.append(child)
        population = new_population
    return population

# 交叉操作
def crossover(parent1, parent2):
    index = random.randint(0, len(parent1) - 1)
    child = parent1[:index] + parent2[index:]
    return child

# 变异操作
def mutation(solution):
    for i in range(len(solution)):
        if random.random() < MUTATION_RATE:
            solution[i] = random.randint(0, 1)
    return solution

# 主函数
def main():
    population = [[random.randint(0, 1) for _ in range(N)] for _ in range(M)]
    best_solution = genetic_algorithm(population)
    best_efficiency = objective_function(best_solution)
    print("最优生产计划：", best_solution)
    print("最优效率：", best_efficiency)

if __name__ == "__main__":
    main()
```
