#### 4.1 最小二乘法

最小二乘法是一种常用的参数估计方法，通过最小化拟合误差的平方和来找到最佳参数配置。以下是一个使用 Python 实现的最小二乘法示例。

```python
import numpy as np

def least_squares(x, y, a, b):
    n = len(x)
    y_fit = a * x + b
    residual = y - y_fit
    error = np.sum(residual**2)
    return error

# 示例数据
x = np.array([1, 2, 3, 4, 5])
y = np.array([2, 4, 5, 4, 5])

# 最小二乘法求解参数 a 和 b
a = 1.0
b = 0.0
error = least_squares(x, y, a, b)
print("Error:", error)

# 迭代求解
for i in range(100):
    y_fit = a * x + b
    residual = y - y_fit
    delta_a = -2 * np.sum(x * residual) / np.sum(x**2)
    delta_b = -2 * np.sum(residual) / n
    a -= delta_a
    b -= delta_b
    error = least_squares(x, y, a, b)
    print("Iteration", i+1, ": Error", error)

print("Final parameters:", a, b)
```

最大后验估计（Maximum a Posteriori, MAP）是一种基于概率统计的参数估计方法，通过最大化后验概率来找到最佳参数配置。以下是一个使用 Python 实现的最大后验估计示例。

```python
import numpy as np

def log_likelihood(x, y, a, b):
    y_fit = a * x + b
    log_likelihood = -0.5 * np.sum((y - y_fit)**2)
    return log_likelihood

def log_prior(a, b):
    log_prior = -0.5 * (a**2 + b**2)
    return log_prior

def log_posterior(a, b, x, y):
    log_posterior = log_likelihood(x, y, a, b) + log_prior(a, b)
    return log_posterior

def map_estimation(x, y, a0, b0, learning_rate):
    a = a0
    b = b0
    for i in range(100):
        y_fit = a * x + b
        log_likelihood = log_likelihood(x, y, a, b)
        log_prior = log_prior(a, b)
        log_posterior = log_posterior(a, b, x, y)
        gradient_a = -(x * (y - y_fit) + a) / (2 * a)
        gradient_b = -(y - y_fit + b) / (2 * b)
        a -= learning_rate * gradient_a
        b -= learning_rate * gradient_b
    return a, b

# 示例数据

b0 = 0.0

```

#### 4.3 贝叶斯估计

贝叶斯估计是一种基于贝叶斯统计学的参数估计方法，通过计算后验概率分布来找到最佳参数估计。以下是一个使用 Python 实现的贝叶斯估计示例。

```python
import numpy as np
from scipy.stats import norm

def likelihood(x, y, a, b):
    y_fit = a * x + b
    likelihood = norm.pdf(y, y_fit, 1).mean()
    return likelihood

def prior(a, b):
    prior = norm.pdf(a, 0, 1).mean() * norm.pdf(b, 0, 1).mean()
    return prior

def posterior(a, b, x, y):
    likelihood = likelihood(x, y, a, b)
    prior = prior(a, b)
    posterior = likelihood * prior
    return posterior

def bayesian_estimation(x, y, a0, b0, alpha, beta):
    a = a0
    b = b0
    for i in range(100):
        likelihood = likelihood(x, y, a, b)
        prior = prior(a, b)
        posterior = posterior(a, b, x, y)
        gradient_a = -(x * likelihood * prior - alpha * prior) / (2 * alpha)
        gradient_b = -(likelihood * prior - beta * prior) / (2 * beta)
        a -= alpha * gradient_a
        b -= beta * gradient_b
    return a, b

# 示例数据
x = np.array([1, 2, 3, 4, 5])
y = np.array([2, 4, 5, 4, 5])
a0 = 1.0
b0 = 0.0
alpha = 1.0
beta = 1.0

a, b = bayesian_estimation(x, y, a0, b0, alpha, beta)
print("Bayesian estimation:", a, b)
```
