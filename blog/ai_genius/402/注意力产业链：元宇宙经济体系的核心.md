$$
A_i = \frac{w_i \cdot s_i}{\sum_{j=1}^{N} w_j \cdot s_j}

class VirtualAssetTransaction:
    def __init__(self, buyer, seller, asset):
        self.buyer = buyer
        self.seller = seller
        self.asset = asset
        
    def execute_transaction(self):
        # 检查交易是否合法
        if self.is_valid_transaction():
            # 执行交易
            self.transfer_asset()
            # 记录交易日志
            self.log_transaction()
            return "Transaction successful"
        else:
            return "Invalid transaction"
        
    def is_valid_transaction(self):
        # 检查买家是否有足够的虚拟资产
        if self.buyer.balance >= self.asset.price:
            return True
        else:
            return False
        
    def transfer_asset(self):
        # 转移虚拟资产
        self.buyer.balance -= self.asset.price
        self.seller.balance += self.asset.price
        
    def log_transaction(self):
        # 记录交易日志
        transaction_log = f"Buyer: {self.buyer}, Seller: {self.seller}, Asset: {self.asset}, Time: {datetime.now()}"
        print(transaction_log)
