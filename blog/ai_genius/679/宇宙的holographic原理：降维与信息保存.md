import numpy as np

from mpl_toolkits.mplot3d import Axes3D

# 步骤1：生成三维数据
# 定义一个三维坐标系统，生成一些数据点
x = np.linspace(-5, 5, 100)
y = np.linspace(-5, 5, 100)
x, y = np.meshgrid(x, y)

# 步骤2：计算全息图
# 对三维数据进行傅里叶变换，得到全息图
f = np.fft.fftshift(np.fft.fft2(z))

plt.show()
