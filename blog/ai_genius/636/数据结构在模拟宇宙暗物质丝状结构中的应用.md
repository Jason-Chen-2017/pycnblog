#### 数据结构的基本概念

数据结构是计算机科学中的一个核心概念，它涉及如何有效地存储、组织和管理数据。数据结构不仅决定了程序的性能，还影响了算法的效率和数据的可访问性。常见的数据结构包括数组、链表、栈、队列、树和图等。每种数据结构都有其独特的特点和适用场景。

数组是一种线性数据结构，它通过索引来访问元素，提供了快速的随机访问能力。链表则是通过节点之间的链接来存储数据，它更适合动态数据集合，但随机访问速度较慢。栈和队列是特殊的线性结构，分别遵循后进先出（LIFO）和先进先出（FIFO）的原则。树是一种层次结构，常用于表示具有父子关系的数据。图则是更一般化的结构，它可以表示任意复杂的关系网络。

引力势能公式用于计算两个质点之间的引力势能。以下是引力势能的公式：

$$

```cpp
#include <iostream>
#include <vector>
#include <cmath>
#include <cstdlib>

// 定义粒子类
class Particle {
public:
    double x, y, z; // 粒子的位置
    double vx, vy, vz; // 粒子的速度
    double mass; // 粒子的质量

    Particle(double x, double y, double z, double vx, double vy, double vz, double mass) {
        this->x = x;
        this->y = y;
        this->z = z;
        this->vx = vx;
        this->vy = vy;
        this->vz = vz;
        this->mass = mass;
    }
};

// 初始化粒子
std::vector<Particle> InitializeParticles(int N) {
    std::vector<Particle> particles;
    for (int i = 0; i < N; ++i) {
        double x = (rand() / double(RAND_MAX)) * 100;
        double y = (rand() / double(RAND_MAX)) * 100;
        double z = (rand() / double(RAND_MAX)) * 100;
        double vx = (rand() / double(RAND_MAX)) * 10;
        double vy = (rand() / double(RAND_MAX)) * 10;
        double vz = (rand() / double(RAND_MAX)) * 10;
        double mass = 1.0;
        particles.push_back(Particle(x, y, z, vx, vy, vz, mass));
    }
    return particles;
}

// 更新粒子位置
void UpdateParticlePositions(std::vector<Particle>& particles, double t) {
    for (auto& p : particles) {
        p.x += p.vx * t;
        p.y += p.vy * t;
        p.z += p.vz * t;
    }
}

// 主函数
int main() {
    int N = 1000; // 粒子数量
    double t = 1.0; // 模拟时间

    // 初始化粒子
    std::vector<Particle> particles = InitializeParticles(N);

    // 模拟粒子运动
    UpdateParticlePositions(particles, t);

    // 输出结果
    for (const auto& p : particles) {
        std::cout << "Particle " << p.x << ", " << p.y << ", " << p.z << std::endl;
    }

    return 0;
}
```
