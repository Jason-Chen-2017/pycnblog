   ```python
   velocities = box_size * np.random.randn(num_particles, 3)
   ```

2. **引力场计算**：
   ```python
   # 计算引力场
   G = 6.67430e-11  # 万有引力常数
   
   def calculate_gravity(positions):
       gravity_force = np.zeros((num_particles, 3))
       for i in range(num_particles):
           for j in range(num_particles):
               if i != j:
                   distance = positions[i] - positions[j]
                   distance = np.linalg.norm(distance)
                   force = G * particle_mass * particle_mass / distance**2
                   direction = distance / distance
                   gravity_force[i] += force * direction
       return gravity_force
   ```

3. **天体运动模拟**：
   ```python
   # 更新天体位置和速度
   def update_particles(velocities, positions, dt=0.01):
       gravity_force = calculate_gravity(positions)
       new_velocities = velocities + gravity_force / particle_mass * dt
       new_positions = positions + new_velocities * dt
       return new_velocities, new_positions
   
   # 模拟天体运动
   for _ in range(1000):
       velocities, positions = update_particles(velocities, positions)
   ```
