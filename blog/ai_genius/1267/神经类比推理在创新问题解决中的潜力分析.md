    # 输入两个问题

    similarity_score = cosine_similarity([input_1, input_2])
    return similarity_score

def neural_solution(problem):
    # 输入一个问题
    # 输出解决方案
    similarity_scores = []
    for known_problem in known_problems:
        similarity_score = neural_similarity(problem, known_problem)
        similarity_scores.append(similarity_score)
    max_similarity_index = np.argmax(similarity_scores)
    solution = known_solutions[max_similarity_index]
    return solution

problem = ['问题描述1', '问题描述2', '问题描述3']
