# 计算梯度
def compute_gradient(input_data, output_data, weights, bias, learning_rate):
    layer_1 = sigmoid(np.dot(input_data, weights) + bias)
    error = output_data - layer_1
    dweights = np.dot(input_data.T, error * layer_1 * (1 - layer_1))
    dbias = np.sum(error * layer_1 * (1 - layer_1))
    weights -= learning_rate * dweights
    bias -= learning_rate * dbias
    return weights, bias

# 运行梯度下降

def qnn_decision_algorithm(inputs, targets, weights, bias, learning_rate, epochs):
    for epoch in range(epochs):
        # 前向传播
        outputs = qnn_decision_model(inputs, weights, bias)
        
        # 计算误差
        errors = targets - outputs
        
        # 反向传播
        dweights = np.dot(inputs.T, errors * outputs * (1 - outputs))
        dbias = np.sum(errors * outputs * (1 - outputs))

        bias -= learning_rate * dbias
    
    return weights, bias
