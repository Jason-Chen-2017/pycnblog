    ```python
        if depth_limit == 0 or node.is_terminal:
            return node.payoff
        best_payoff = -float('inf')
        for child in node.children:
            child_payoff = search_game_tree(child, strategy, depth_limit - 1)
            best_payoff = max(best_payoff, child_payoff)
        return best_payoff
    ```
