import numpy as np

def circular_evolution(r0, t):
    return r0 + t

a0 = 1

plt.xlabel('Time')
plt.ylabel('Value')
plt.show()

# 绘制指数函数时间演化图
a = 1
k = 0.5
t = np.linspace(0, 10, 100)
function_values = exponential_function(a, k, t)
plt.plot(t, function_values)
plt.title('Exponential Function Evolution')
plt.xlabel('Time')
plt.ylabel('Value')
plt.show()

# 绘制圆的时间演化图
r0 = 1
t = np.linspace(0, 10, 100)

fig.show()
