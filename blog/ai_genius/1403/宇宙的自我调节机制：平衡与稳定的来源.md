```python
# 恒星和行星的初始参数
initial_conditions = {
    '恒星质量': 1.989e30,
    '行星质量': 5.972e24,
    '恒星初始位置': np.array([0.0, 0.0]),
    '行星初始位置': np.array([1.0, 0.0]),
    '恒星初始速度': np.array([0.0, 0.0]),
    '行星初始速度': np.array([0.0, 1.0]),
}

# 牛顿引力定律
def gravitational_force(m1, m2, r1, r2):
    distance = np.linalg.norm(r1 - r2)
    force_magnitude = G * m1 * m2 / distance**2
    force_vector = (r2 - r1) * force_magnitude / distance
    return force_vector

# 系统的微分方程
def system_of_equations(t, y):
    G = 6.67430e-11
    m1, m2, r1, r2, v1, v2 = y
    force_magnitude = gravitational_force(m1, m2, r1, r2)
    dv1dt = force_magnitude / m1
    dv2dt = -force_magnitude / m2
    dr1dt = v1
    dr2dt = v2
    return [dv1dt, dv2dt, dr1dt, dr2dt]

# 解微分方程
solution = solve_ivp(system_of_equations, [0, 100], [initial_conditions['恒星质量'], initial_conditions['行星质量'], *initial_conditions['恒星初始位置'], *initial_conditions['行星初始位置'], *initial_conditions['恒星初始速度'], *initial_conditions['行星初始速度']], t_eval=np.linspace(0, 100, 1000))

# 绘制结果
plt.plot(solution.t, solution.y[2], label='恒星轨迹')
plt.plot(solution.t, solution.y[5], label='行星轨迹')
plt.legend()
plt.xlabel('时间')
plt.ylabel('位置')
plt.title('恒星与行星的轨道演化')
plt.show()
```
