```python
import pandas as pd
from sklearn.model_selection import train_test_split
from tensorflow.keras.preprocessing.text import Tokenizer
from tensorflow.keras.preprocessing.sequence import pad_sequences

# 读取数据
data = pd.read_csv('data.csv')
texts = data['text']
labels = data['label']

# 分词与去停用词
tokenizer = Tokenizer(num_words=10000)
tokenizer.fit_on_texts(texts)
sequences = tokenizer.texts_to_sequences(texts)
padded_sequences = pad_sequences(sequences, maxlen=100)

# 切分训练集和测试集
X_train, X_test, y_train, y_test = train_test_split(padded_sequences, labels, test_size=0.2, random_state=42)
```

#### 7.2.2 模型构建

在构建神经符号系统模型时，我们将结合卷积神经网络（CNN）和循环神经网络（RNN）的优势。以下是Python代码示例：

```python
import tensorflow as tf
from tensorflow.keras.models import Model
from tensorflow.keras.layers import Embedding, Conv1D, MaxPooling1D, LSTM, Dense, Input

# 输入层
input_layer = Input(shape=(100,))

# 嵌入层
embedding_layer = Embedding(10000, 64)(input_layer)

# 卷积层
conv_layer = Conv1D(128, 5, activation='relu')(embedding_layer)
pooling_layer = MaxPooling1D(5)(conv_layer)

# 循环层
lstm_layer = LSTM(128)(pooling_layer)

# 输出层
output_layer = Dense(1, activation='sigmoid')(lstm_layer)

# 构建模型
model = Model(inputs=input_layer, outputs=output_layer)

# 编译模型
model.compile(optimizer='adam', loss='binary_crossentropy', metrics=['accuracy'])

# 模型结构
model.summary()
```
