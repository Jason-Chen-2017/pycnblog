def attention(Q, K, V):
    # 计算注意力得分
    scores = dot(Q, K.T) / sqrt(len(K))
    
    # 计算注意力权重
    weights = softmax(scores)
    
    # 计算加权输出
    output = dot(weights, V)
    
    return output
