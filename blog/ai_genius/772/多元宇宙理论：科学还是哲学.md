```python
import numpy as np

def dirac_equation(ψ, m):
    """
    计算狄拉克方程。
    
    参数：
    ψ：量子场的场量。
    m：质量。
    """
    gamma = np.array([[0, 1], [1, 0]])
    momentum = np.matmul(gamma, ψ)
    return momentum - m * ψ

# 创建一个 2x2 的量子场矩阵
ψ = np.array([[1, 0], [0, 1]])

# 计算狄拉克方程的结果
result = dirac_equation(ψ, 1.0)

print(result)
```
