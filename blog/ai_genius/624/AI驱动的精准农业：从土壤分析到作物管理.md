2. **航空和卫星遥感**：利用航空和卫星遥感技术，可以从空中获取大范围的土壤数据。这些数据可以包括土壤反射率、光谱特性等，为土壤分析提供宏观视角。

import numpy as np
import matplotlib.pyplot as plt
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Dense
from tensorflow.keras.optimizers import Adam

# 1. 数据预处理

soil_humidity_data = np.array([[20], [30], [40], [50], [60], [70], [80], [90], [100]])

soil_humidity_normalized = (soil_humidity_data - soil_humidity_data.min()) / (soil_humidity_data.max() - soil_humidity_data.min())

model = Sequential()
model.add(Dense(units=64, activation='relu', input_shape=(1,)))
model.add(Dense(units=32, activation='relu'))
model.add(Dense(units=1))

# 3. 模型编译
model.compile(optimizer=Adam(learning_rate=0.001), loss='mean_squared_error')

# 4. 模型训练
model.fit(soil_humidity_normalized, soil_humidity_normalized, epochs=100, batch_size=32)

# 5. 模型预测
predicted_humidity = model.predict(np.array([[60]]))

# 6. 数据可视化
plt.plot(soil_humidity_normalized, label='Actual Soil Humidity')
