class User:
    def __init__(self, role: str):
        self.role = role

    def has_permission(self, permission: str) -> bool:
        return self.role == "admin" or permission in self.allowed_permissions

class Admin(User):
    def __init__(self):
        super().__init__("admin")
        self.allowed_permissions = ["delete", "modify"]

class RegularUser(User):
    def __init__(self):
        super().__init__("regular")
        self.allowed_permissions = ["read"]

admin = Admin()
print(admin.has_permission("delete"))  # 输出：True
regular_user = RegularUser()
print(regular_user.has_permission("delete"))  # 输出：False
