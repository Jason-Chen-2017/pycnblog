        text = re.sub('[\s]+', ' ', text)
        return text
    
    def segment_paragraphs(text):

        return paragraphs
    
    def extract_clauses(paragraphs):
        clauses = []
        for paragraph in paragraphs:
            if "条款" in paragraph:
                clauses.append(paragraph.split("条款")[1].strip())
        return clauses
    
    def audit_contract(contract_text):
        preprocessed_text = preprocess_text(contract_text)
        paragraphs = segment_paragraphs(preprocessed_text)
        clauses = extract_clauses(paragraphs)
        risks = identify_risks(clauses)
        return risks

        risk_levels = {'unfair_clause': '高', 'missing_clause': '中'}
        risk_assessments = {risk: risk_levels[risk] for risk in risks}
        return risk_assessments
