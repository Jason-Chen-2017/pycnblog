```python
from tokenize import tokenize
from ast import Node
from ontology import Ontology

def build_ast(tokens):
    ast = Node()
    for token in tokens:
        if token.is_terminal:
            ast.add_child(TokenNode(token))
        else:
            ast.add_child(build_ast(token.children))
    return ast

class TokenNode(Node):
    def __init__(self, token):
        self.token = token

def generate_prompt(ast):
    prompt = ""
    for node in ast.children:
        if isinstance(node, TokenNode):
            prompt += node.token.value
        else:
            prompt += generate_prompt(node)
    return prompt

def define_ontology(prompt):
    ontology = Ontology()
    for term in prompt.terms:
        ontology.add_definition(term, term.meaning)
    return ontology
```
