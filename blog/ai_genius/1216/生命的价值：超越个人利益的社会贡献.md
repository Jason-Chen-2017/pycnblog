class SocialContributionIncentive:
    def __init__(self, base_reward, bonus_rate):
        self.base_reward = base_reward
        self.bonus_rate = bonus_rate
        self.total_contributions = 0

    def calculate_reward(self, contribution):
        total_reward = self.base_reward + (contribution * self.bonus_rate)
        self.total_contributions += contribution
        return total_reward

# 实例化激励机制

reward = incentive.calculate_reward(50)
