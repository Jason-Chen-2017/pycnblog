  }

import (
    "encoding/json"
    "math/rand"
    "time"

// 时空碎片结构体
type TemporalFragment struct {
    StartTime int64  `json:"start_time"`
    EndTime   int64  `json:"end_time"`
    Data      []int `json:"data"`
}

// 生成随机时空碎片
func generateTemporalFragment() TemporalFragment {
    startTime := time.Now().UnixNano()
    endTime := startTime + 60*1000000000 // 1分钟
    dataSize := rand.Intn(100) + 1       // 随机数据大小

    data := make([]int, dataSize)
    for i := 0; i < dataSize; i++ {
        data[i] = rand.Intn(100)
    }

    return TemporalFragment{StartTime: startTime, EndTime: endTime, Data: data}
}

// 存储时空碎片到文件
func storeTemporalFragment(fragment TemporalFragment) error {
    jsonData, err := json.Marshal(fragment)
    if err != nil {
        return err
    }

    fileName := "fragment_" + time.Now().Format("20060102_150405") + ".json"
    err = ioutil.WriteFile(fileName, jsonData, 0644)
    if err != nil {
        return err
    }

    return nil
}

func main() {
    fragment := generateTemporalFragment()
    err := storeTemporalFragment(fragment)
    if err != nil {
        log.Fatal(err)
    }
}
