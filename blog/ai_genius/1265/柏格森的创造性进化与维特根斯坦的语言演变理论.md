  ```python
  import numpy as np

  # 初始化神经网络模型
  def initialize_network(num_inputs, num_outputs):
      weights = np.random.rand(num_inputs, num_outputs)
      return weights

  # 选择操作
  def selection(population, fitness_func):
      selected_individuals = []
      for individual in population:
          fitness = fitness_func(individual)
          if fitness > np.mean(fitness):
              selected_individuals.append(individual)
      return selected_individuals

  # 交叉操作
  def crossover(parent1, parent2):
      child = (parent1 + parent2) / 2
      return child

  # 变异操作
  def mutation(individual):
      mutated_individual = individual + np.random.normal(0, 1)
      return mutated_individual

  # 创造性进化算法
  def creative_evolution(population, fitness_func):
      next_generation = []
      for individual in population:
          selected_individual = selection([individual], fitness_func)
          crossed_individual = crossover(individual, selected_individual[0])
          mutated_individual = mutation(crossed_individual)
          next_generation.append(mutated_individual)
      return next_generation

  # 主函数
  def main():
      num_inputs = 10
      num_outputs = 5
      population_size = 100
      generations = 10

      population = [initialize_network(num_inputs, num_outputs) for _ in range(population_size)]

      for _ in range(generations):
          population = creative_evolution(population, fitness_func)

      best_individual = max(population, key=lambda x: fitness_func(x))
      print("Best individual:", best_individual)

  if __name__ == "__main__":
      main()
  ```
