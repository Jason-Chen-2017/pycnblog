##### 数学公式

$$
T(n) = \sum_{i=1}^{n} A_i
$$

其中，\( T(n) \) 表示第 \( n \) 次迭代的值，\( A_i \) 表示第 \( i \) 个元素。

##### Python代码

```python
def iterate_sequence(a, n):
    """
    对一个序列进行迭代
    :param a: 序列
    :param n: 迭代次数
    :return: 迭代后的序列
    """
    result = []
    for i in range(n):
        result.append(sum(a))
        a = result[-1]
    return result

sequence = [1, 2, 3]
iterations = 3

print(iterate_sequence(sequence, iterations))

import numpy as np
import matplotlib.pyplot as plt

def lorenz_attractor(x0, y0, z0, sigma, rho, beta, t_max):
    """
    模拟洛伦兹吸引子
    :param x0: 初始x值
    :param y0: 初始y值
    :param z0: 初始z值
    :param sigma: 参数
    :param rho: 参数
    :param beta: 参数
    :param t_max: 模拟时间
    :return: 洛伦兹吸引子的轨迹
    """
    t = np.linspace(0, t_max, 10000)
    x = x0
    y = y0
    z = z0

    trajectory = np.zeros((len(t), 3))

    for i in range(len(t)):
        trajectory[i] = [x, y, z]
        dx_dt = sigma * (y - x)
        dy_dt = x * (rho - z) - y
        dz_dt = x * y - beta * z
        x += dx_dt * dt
        y += dy_dt * dt
        z += dz_dt * dt

    return trajectory

def plot_lorenz_trajectory(trajectory):
    """
    绘制洛伦兹吸引子的轨迹
    :param trajectory: 洛伦兹吸引子的轨迹
    """
    x, y, z = trajectory.T
    fig = plt.figure()
    ax = fig.add_subplot(111, projection='3d')
    ax.plot(x, y, z)
    ax.set_xlabel('X axis')
    ax.set_ylabel('Y axis')
    ax.set_zlabel('Z axis')
    plt.show()

# 模拟洛伦兹吸引子
trajectory = lorenz_attractor(1, 1, 1, 10, 28, 8.0/3, 100)
# 绘制轨迹
plot_lorenz_trajectory(trajectory)
```
