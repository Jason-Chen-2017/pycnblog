**Python 源代码讲解：** 区块链技术的基础原理可以通过Python实现。以下是一个简单的区块链节点的创建和添加块的示例。

```python
import hashlib
import json
from time import time

class Block:
    def __init__(self, index, transactions, timestamp, previous_hash):
        self.index = index
        self.transactions = transactions
        self.timestamp = timestamp
        self.previous_hash = previous_hash
        self.hash = self.compute_hash()

    def compute_hash(self):
        block_string = json.dumps(self.__dict__, sort_keys=True)
        return hashlib.sha256(block_string.encode()).hexdigest()


class Blockchain:
    def __init__(self):
        self.unconfirmed_transactions = []
        self.chain = []
        self.create_genesis_block()

    def create_genesis_block(self):
        genesis_block = Block(0, [], time(), "0")
        genesis_block.hash = genesis_block.compute_hash()
        self.chain.append(genesis_block)

    def add_new_transaction(self, transaction):
        self.unconfirmed_transactions.append(transaction)

    def mine(self):
        if not self.unconfirmed_transactions:
            return False
        last_block = self.chain[-1]
        new_block = Block(index=last_block.index + 1,
                          transactions=self.unconfirmed_transactions,
                          timestamp=time(),
                          previous_hash=last_block.hash)
        new_block.hash = new_block.compute_hash()
        self.chain.append(new_block)
        self.unconfirmed_transactions = []
        return new_block.index

# 创建区块链实例
blockchain = Blockchain()

# 添加交易
blockchain.add_new_transaction("Transaction 1")
blockchain.add_new_transaction("Transaction 2")

# 挖掘新块
blockchain.mine()

# 查看区块链
print(blockchain.chain)
```

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract DigitalContentCopyright {
    mapping(address => mapping(uint => bool)) public copyrights;
    mapping(address => uint) public total_copies;
    address public owner;

    event CopyrightRegistered(address indexed owner, uint id);
    event CopyrightTransferred(address indexed from, address indexed to, uint id);

        owner = msg.sender;
    }

    function registerCopyright(uint id) public {
        require(!copyrights[msg.sender][id], "Copyright already registered");
        copyrights[msg.sender][id] = true;
        total_copies[msg.sender]++;

    }

        require(copyrights[msg.sender][id], "You don't own this copyright");
        require(to != address(0), "Invalid address");
        copyrights[msg.sender][id] = false;
        copyrights[to][id] = true;
        total_copies[msg.sender]--;
        total_copies[to]++;
        emit CopyrightTransferred(msg.sender, to, id);
    }
}
