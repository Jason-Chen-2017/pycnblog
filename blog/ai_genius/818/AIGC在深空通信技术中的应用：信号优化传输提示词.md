    time = np.linspace(0, duration, int(duration * 1000))
    signal = amplitude * np.sin(2 * np.pi * frequency * time)
    return signal

# 信道模型模拟
def simulate_channel(signal, noise_power, frequency):
    noise = np.random.normal(0, np.sqrt(noise_power), signal.shape)
    channel_signal = signal + noise
    channel_signal = channel_signal * np.exp(1j * 2 * np.pi * frequency * time)
    return channel_signal

# 信号处理
def process_signal(signal, gain):
    processed_signal = signal * gain
    return processed_signal

# 信道编码
def encode_signal(signal):
    # 简单的BPSK编码
    encoded_signal = np.array([1 if s > 0 else -1 for s in signal])
    return encoded_signal

def control_power(signal, max_power):
    power = np.mean(signal ** 2)
    gain = max_power / np.sqrt(power)
    return gain

# 调制与解调
def modulate_signal(signal, frequency):
    modulated_signal = signal * np.exp(1j * 2 * np.pi * frequency * time)
    return modulated_signal

def demodulate_signal(modulated_signal, frequency):
    demodulated_signal = modulated_signal * np.exp(-1j * 2 * np.pi * frequency * time)
    return demodulated_signal

# 主函数
def main():

    duration = 1  # 信号持续时间（秒）

    encoded_signal = encode_signal(processed_signal)

    modulated_signal = modulate_signal(encoded_signal, frequency)

    # 解调
    demodulated_signal = demodulate_signal(modulated_signal, frequency)
