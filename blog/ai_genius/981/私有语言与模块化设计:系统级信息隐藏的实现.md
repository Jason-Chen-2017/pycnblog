模块化设计是将系统划分为多个模块，每个模块负责不同的功能。模块化设计的基本原则包括：

def tokenize(expression):
    # 根据语法规则分解表达式
    tokens = []
    current_token = ""
    for char in expression:
        if char.isdigit():
            current_token += char
        elif char.isalpha():
            current_token += char
            if current_token.isalnum():
                continue
            else:
                tokens.append(current_token)
                current_token = ""
        elif char in "+-*/":
            tokens.append(char)
        else:
            raise SyntaxError("Invalid character")
    if current_token:
        tokens.append(current_token)
    return tokens

    # 构建抽象语法树
    ast = []
    for i in range(0, len(tokens), 2):
        operator = tokens[i]
        operand1 = tokens[i+1]
        operand2 = None
        if i+2 < len(tokens) and tokens[i+2] in "+-*/":
            operand2 = tokens[i+3]
        ast.append((operator, operand1, operand2))
    return ast

# 伪代码
def evaluate_ast(ast):
    # 计算抽象语法树表达式的值
    if ast[0] == '+':
        return ast[1] + ast[2]
    elif ast[0] == '-':
        return ast[1] - ast[2]
    elif ast[0] == '*':
        return ast[1] * ast[2]
    elif ast[0] == '/':
        return ast[1] / ast[2]
