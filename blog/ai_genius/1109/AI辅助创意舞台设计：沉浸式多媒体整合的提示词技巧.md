```python
print(f'Model accuracy: {accuracy:.2f}')
```

```python
import cv2

# 读取视频文件
video = cv2.VideoCapture('input_video.mp4')

# 创建输出视频文件
output = cv2.VideoWriter('output_video.mp4', cv2.VideoWriter_fourcc('mp4v', 'DIVX'), 30, (640, 480))

# 视频处理循环
while video.isOpened():
    ret, frame = video.read()
    if not ret:
        break
    
    # 应用特效（例如，添加滤镜）
    filtered_frame = cv2.cvtColor(frame, cv2.COLOR_BGR2HSV)
    filtered_frame = cv2.applyColorMap(filtered_frame, cv2.COLORMAP_JET)
    
    # 写入输出视频
    output.write(filtered_frame)

# 释放资源
video.release()
output.release()
```

#### 核心算法原理讲解
