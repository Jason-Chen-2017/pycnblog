### 源代码实现

```python
import numpy as np
import cv2

# 生成全息图
def generate_hologram(image):
    # 将图像转换为灰度图像
    gray_image = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    
    # 计算图像的傅里叶变换
    Fourier_image = cv2.dft(np.float32(gray_image), flags=cv2.DFT_COMPLEX_OUTPUT)
    
    # 计算全息图
    hologram = Fourier_image.copy()
    hologram[:, :, 0] = 0
    hologram[:, :, 1] = Fourier_image[:, :, 1]
    
    return hologram

# 重建三维图像
def reconstruct_3d_image(hologram, light_angle):
    # 生成参考光波
    ref_light = np.zeros(hologram.shape, dtype=np.float32)
    ref_light[:, :, 0] = np.cos(light_angle)
    ref_light[:, :, 1] = np.sin(light_angle)
    
    # 重建图像
    reconstructed_image = cv2.mulSpectrums(hologram, ref_light, 0)
    reconstructed_image = cv2.idft(np.float32(reconstructed_image), flags=cv2.DFT_COMPLEX_OUTPUT)
    
    # 转换为灰度图像
    reconstructed_image = cv2.magnitude(reconstructed_image[:, :, 0], reconstructed_image[:, :, 1])
    
    return reconstructed_image

# 测试图像
image = cv2.imread('example_image.jpg')

# 生成全息图
hologram = generate_hologram(image)

# 设置光波角度
light_angle = np.pi / 4

# 重建三维图像
reconstructed_image = reconstruct_3d_image(hologram, light_angle)

# 显示结果
cv2.imshow('Original Image', image)
cv2.imshow('Reconstructed Image', reconstructed_image)
cv2.waitKey(0)
cv2.destroyAllWindows()
```

### 代码解读

上述代码分为两个主要部分：生成全息图和重建三维图像。

- **生成全息图**：首先将输入图像转换为灰度图像，然后计算其傅里叶变换。接着，创建一个全息图，其中实部为0，虚部为傅里叶变换的虚部。
- **重建三维图像**：根据全息图和参考光波，使用乘积傅里叶变换（MulSpectrums）和逆傅里叶变换（idft）来重建三维图像。最后，将复数图像转换为灰度图像，以获得重建的三维图像。
