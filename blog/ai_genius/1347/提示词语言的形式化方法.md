    sentence_structure = {'please': 'vp', 'help': 'v', 'book': 'v', 'flight': 'n', 'tomorrow': 'n', 'morning': 'n', 'ticket': 'n'}

    semantics = {'please': 'request', 'help': 'action', 'book': 'action', 'flight': 'object', 'tomorrow': 'time', 'morning': 'time', 'ticket': 'object'}
