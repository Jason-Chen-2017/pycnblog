```python
import numpy as np
from scipy.signal import match_filter

# 生成模拟的引力波信号
def generate_gravitational_wave(signal_length, sampling_rate):
    t = np.linspace(0, signal_length, int(sampling_rate * signal_length))
    f = 1 / (2 * t.max())  # 假设信号频率为1/(2*t.max())
    signal = np.sin(2 * np.pi * f * t)
    return signal, t

# 匹配滤波器算法
def match_filter_algorithm(signal, template, sampling_rate):
    template_length = len(template)
    overlap = int(template_length * 0.5)  # 设定重叠部分为模板长度的一半
    
    # 生成匹配滤波器
    filtered_signal = match_filter(signal, template, flank=overlap, threshold='log-amplitude')
    
    # 信号重建
    reconstructed_signal = filtered_signal * template
    
    return reconstructed_signal

# 测试

sampling_rate = 1000  # 单位：Hz/s

template = np.sin(2 * np.pi * 0.5 * t)  # 模板信号，频率为0.5Hz

reconstructed_signal = match_filter_algorithm(signal, template, sampling_rate)
```
