def self_cognition(questions):
    """
    自我认知模块，输入问卷问题，输出自我认知结果。
    
    参数：
    - questions：问卷问题，为一个包含问题的列表。

        cognition[f"Q{i+1}"] = int(answer)
    
    return cognition

def social_interaction(network, opportunities):
    """
    社会互动模块，输入社交网络和互动机会，输出互动结果。
    
    参数：
    - network：社交网络，为一个包含好友信息的字典。
