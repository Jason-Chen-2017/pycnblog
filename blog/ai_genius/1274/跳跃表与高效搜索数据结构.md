import random

class Node:
    def __init__(self, value):
        self.value = value
        self.next = []

class SkipList:
    def __init__(self, max_level, p):
        self.head = Node(None)
        self.max_level = max_level
        self.p = p
        self.level = 0

    def random_level(self):
        level = 0
        while random.random() < self.p and level < self.max_level:
            level += 1
        return level

    def search(self, value):
        current = self.head
        for i in range(self.level, -1, -1):
            while current.next[i] and current.next[i].value < value:
                current = current.next[i]
        current = current.next[0]
        if current and current.value == value:
            return current
        return None

    def insert(self, value):
        update = [None] * (self.max_level + 1)
        current = self.head
        for i in range(self.level, -1, -1):
            while current.next[i] and current.next[i].value < value:
                current = current.next[i]
            update[i] = current
        current = current.next[0]
        if not current or current.value != value:
            level = self.random_level()
            if level > self.level:
                for i in range(self.level + 1, level + 1):
                    self.head.next.append(None)
                self.level = level
            node = Node(value)
            for i in range(level + 1):
                node.next.append(update[i].next[i])
                update[i].next[i] = node

    def delete(self, value):
        update = [None] * (self.max_level + 1)
        current = self.head
        for i in range(self.level, -1, -1):
            while current.next[i] and current.next[i].value < value:
                current = current.next[i]
            update[i] = current
        current = current.next[0]
        if current and current.value == value:
            for i in range(self.level + 1):
                if update[i].next[i] != current:
                    break
                update[i].next[i] = current.next[i]
            while self.level > 0 and not self.head.next[self.level]:
                self.level -= 1

### 5.2 跳跃表的并发控制

在多线程环境下，跳跃表的并发操作可能会导致数据不一致和竞争条件。为了确保并发控制，可以采用以下几种机制：

#### 乐观锁

乐观锁通过在操作前检查数据版本，确保在操作过程中不会出现数据冲突。具体实现可以通过在节点中添加版本号，每次修改数据时，检查版本号是否匹配。

#### 排他锁

排他锁（X锁）确保在同一时刻，只有一个线程可以修改数据。通过在节点上添加排他锁，可以避免多个线程同时修改数据，保证数据的一致性。

#### 共享锁

共享锁（S锁）允许多个线程同时读取数据，但禁止同时修改数据。通过在节点上添加共享锁，可以提高多线程环境下的并发性能。

#### Read-Write Lock

读-写锁结合了共享锁和排他锁的特点，允许多个线程同时读取数据，但在修改数据时，确保只有一个线程进行操作。这样可以最大化利用多线程环境下的并发性能。

```python
import random

class Node:
    def __init__(self, value):
        self.value = value
        self.next = []

class SkipList:
    def __init__(self, max_level, p):
        self.head = Node(None)
        self.max_level = max_level
        self.p = p
        self.level = 0

    def random_level(self):
        level = 0
        while random.random() < self.p and level < self.max_level:
            level += 1
        return level

    def search(self, value):
        current = self.head
        for i in range(self.level, -1, -1):
            while current.next[i] and current.next[i].value < value:
                current = current.next[i]
        current = current.next[0]
        if current and current.value == value:
            return current
        return None

    def insert(self, value):
        update = [None] * (self.max_level + 1)
        current = self.head
        for i in range(self.level, -1, -1):
            while current.next[i] and current.next[i].value < value:
                current = current.next[i]
            update[i] = current
        current = current.next[0]
        if not current or current.value != value:
            level = self.random_level()
            if level > self.level:
                for i in range(self.level + 1, level + 1):
                    self.head.next.append(None)
                self.level = level
            node = Node(value)
            for i in range(level + 1):
                node.next.append(update[i].next[i])
                update[i].next[i] = node

    def delete(self, value):
        update = [None] * (self.max_level + 1)
        current = self.head
        for i in range(self.level, -1, -1):
            while current.next[i] and current.next[i].value < value:
                current = current.next[i]
            update[i] = current
        current = current.next[0]
        if current and current.value == value:
            for i in range(self.level + 1):
                if update[i].next[i] != current:
                    break
                update[i].next[i] = current.next[i]
            while self.level > 0 and not self.head.next[self.level]:
                self.level -= 1
```
