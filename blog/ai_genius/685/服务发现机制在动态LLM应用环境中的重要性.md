```java
@PostConstruct
public void registerService() {
    String serviceName = "smart-customer-service";
    String serviceAddress = host.getAddress().getHostAddress();
    int servicePort = serverPort;
    consulClient.agent().services().registerService(serviceName, serviceAddress, servicePort);
}

// 服务查询
public List<ServiceInstance> discoverServices(String serviceName) {
    List<ServiceInstance> serviceInstances = consulClient.discovery().instances(serviceName);
    return serviceInstances;
}

// 服务选择
public ServiceInstance selectServiceInstance(List<ServiceInstance> instances) {
    int index = instances.size() == 1 ? 0 : new Random().nextInt(instances.size());
    return instances.get(index);
}

// 服务调用
public String callServiceInstance(ServiceInstance instance) {
    String url = "http://" + instance.getHost() + ":" + instance.getPort() + "/api/v1/message";
    String requestBody = "{\"message\": \"Hello, how may I help you?\"}";
    HttpClient httpClient = HttpClient.newHttpClient();
    HttpRequest request = HttpRequest.newBuilder()
            .uri(URI.create(url))
            .POST(HttpRequest.BodyPublishers.ofString(requestBody))
            .build();
    HttpResponse<String> response = httpClient.send(request, HttpResponse.BodyHandlers.ofString());
    return response.body();
}
```
