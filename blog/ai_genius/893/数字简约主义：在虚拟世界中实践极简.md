```python
def add_schedule(schedule_name, date):
    schedules.append((schedule_name, date))
    print(f"Schedule '{schedule_name}' added for {date}.")

# 删除日程
def delete_schedule(schedule_name):
    for i, (name, _) in enumerate(schedules):
        if name == schedule_name:
            schedules.pop(i)
            print(f"Schedule '{schedule_name}' deleted.")
            break
        else:
            print(f"Schedule '{schedule_name}' not found.")

# 查看日程列表
def list_schedules():
    print("Schedules:")
    for schedule in schedules:
        name, date = schedule
        print(f"- {name} on {date}")
```
