import numpy as np
import matplotlib.pyplot as plt

# 初始化宇宙参数
t = np.arange(0, 138e9, 1e9)  # 时间

    temp[i] /= 10
    density[i] /= 10

K = 1e9  # 环境容纳量

for i in range(len(t)):

    N[i + 1] = N[i] + dNdt

# 绘制结果
plt.plot(t / 1e9, temp / 1e12, label='Temperature')
plt.plot(t / 1e9, density / 1e12, label='Density')
plt.plot(t / 1e9, N, label='Population')
plt.xlabel('Time (years)')
plt.ylabel('Value')
plt.legend()
plt.show()
