```python
# app.py
from flask import Flask, request, jsonify
from models import Participant, Workshops, HealthIndicators
from database import db

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///database.db'
db.init_app(app)

@app.route('/register', methods=['POST'])
def register():
    name = request.form['name']
    email = request.form['email']
    password = request.form['password']
    participant = Participant(name=name, email=email, password=password)
    db.session.add(participant)
    db.session.commit()
    return jsonify({'message': '注册成功'})

@app.route('/login', methods=['POST'])
def login():
    email = request.form['email']
    password = request.form['password']
    participant = Participant.query.filter_by(email=email, password=password).first()
    if participant:
        return jsonify({'message': '登录成功'})
    else:
        return jsonify({'message': '登录失败'})

@app.route('/join_workshop', methods=['POST'])
def join_workshop():
    participant_id = request.form['participant_id']
    workshop_id = request.form['workshop_id']
    participant = Participant.query.get(participant_id)
    workshop = Workshops.query.get(workshop_id)
    participant.workshops.append(workshop)
    db.session.commit()
    return jsonify({'message': '报名成功'})

@app.route('/record_health_indicator', methods=['POST'])
def record_health_indicator():
    participant_id = request.form['participant_id']
    emotion = request.form['emotion']
    cognition = request.form['cognition']
    physical_state = request.form['physical_state']
    social_connection = request.form['social_connection']
    participant = Participant.query.get(participant_id)
    health_indicator = HealthIndicators(
        participant_id=participant_id,
        emotion=emotion,
        cognition=cognition,
        physical_state=physical_state,
        social_connection=social_connection
    )
    participant.health_indicators.append(health_indicator)
    db.session.commit()
    return jsonify({'message': '记录成功'})

if __name__ == '__main__':
    app.run(debug=True)
```
