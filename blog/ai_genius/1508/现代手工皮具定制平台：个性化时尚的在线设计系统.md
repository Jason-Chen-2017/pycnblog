```jsx
import React, { useState } from "react";

function DesignTool() {
  const [design, setDesign] = useState({ color: "black", material: "leather" });

  const handleColorChange = (e) => {
    setDesign({ ...design, color: e.target.value });
  };

  const handleMaterialChange = (e) => {
    setDesign({ ...design, material: e.target.value });
  };

  return (
    <div>
      <h2>Design Tool</h2>
      <label>
        Color:
        <select value={design.color} onChange={handleColorChange}>
          <option value="black">Black</option>
          <option value="brown">Brown</option>
          <option value="red">Red</option>
        </select>
      </label>
      <label>
        Material:
        <select value={design.material} onChange={handleMaterialChange}>
          <option value="leather">Leather</option>
          <option value="suede">Suede</option>
          <option value="vinyl">Vinyl</option>
        </select>
      </label>
      <h3>Preview</h3>
      <div style={{ backgroundColor: design.color, width: "100px", height: "100px" }} />
    </div>
  );
}

export default DesignTool;
```

#### 5.2.2 后端核心代码

以下是后端代码示例：

```javascript
const express = require("express");
const MongoClient = require("mongodb").MongoClient;
const ObjectId = require("mongodb").ObjectId;

const app = express();
app.use(express.json());

const uri = "mongodb://localhost:27017/";
const client = new MongoClient(uri, { useNewUrlParser: true, useUnifiedTopology: true });

app.post("/designs", async (req, res) => {
  try {
    await client.connect();
    const design = req.body;
    const database = client.db("leathercraft");
    const collection = database.collection("designs");
    await collection.insertOne(design);
    res.status(201).send({ message: "Design saved", id: design._id });
  } catch (error) {
    res.status(500).send({ message: "Error saving design", error: error.message });
  } finally {
    await client.close();
  }
});

app.get("/designs/:id", async (req, res) => {
  try {
    await client.connect();
    const id = req.params.id;
    const database = client.db("leathercraft");
    const collection = database.collection("designs");
    const design = await collection.findOne({ _id: ObjectId(id) });
    if (design) {
      res.status(200).send(design);
    } else {
      res.status(404).send({ message: "Design not found" });
    }
  } catch (error) {
    res.status(500).send({ message: "Error retrieving design", error: error.message });
  } finally {
    await client.close();
  }
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`Server running on port ${PORT}`);
});
```

## 第五部分：最佳实践与总结

### 6.1 最佳实践

#### 6.1.1 前端优化

1. **懒加载**：对于大型图片和视频，可以使用懒加载技术，减少页面加载时间。
2. **代码分割**：将代码分割为不同的块，按需加载，提高页面性能。
3. **性能监控**：使用性能监控工具，如 Lighthouse，定期检查页面性能，并及时优化。

#### 6.1.2 后端优化

1. **缓存**：使用 Redis 等缓存技术，减少数据库查询次数，提高系统性能。
2. **分库分表**：对于大数据量，可以采用分库分表策略，提高数据库查询性能。
3. **异步处理**：使用异步处理，提高系统并发处理能力。

#### 6.1.3 安全性优化

1. **数据加密**：对敏感数据进行加密存储，确保数据安全。
2. **身份验证**：使用 HTTPS 和 JWT 等技术，确保用户身份验证的安全性。
3. **输入验证**：对用户输入进行严格验证，防止 SQL 注入等安全漏洞。
