```python
import numpy as np
from scipy.optimize import linear_sum_assignment

# 假设我们有5个会议室和3个会议请求
meetings = np.array([[1, 1, 3], [2, 2, 2], [3, 3, 4], [4, 4, 1], [5, 5, 2]])
rooms = np.array([[1, 1, 1], [1, 2, 2], [1, 3, 3]])

# 调度算法
assignment = linear_sum_assignment(meetings - rooms)
assigned_meetings = assignment[:, 1]

# 输出调度结果
for i, room in enumerate(assigned_meetings):
    print(f'Meeting {i+1} assigned to Room {room+1}')
```

在上面的代码中，我们假设有5个会议室和3个会议请求，每个会议都有开始时间和持续时间。我们使用线性规划中的匈牙利算法对会议和会议室进行调度，以最大化会议室的利用率。

```python
import numpy as np
from scipy.optimize import linear_sum_assignment

def schedule_meetings(meetings, rooms):
    # 创建调度矩阵
    schedule_matrix = np.zeros((len(meetings), len(rooms)), dtype=int)
    for i, meeting in enumerate(meetings):
        for j, room in enumerate(rooms):
            if room['available']:
                schedule_matrix[i][j] = 1
    
    # 使用匈牙利算法进行调度
    assignment = linear_sum_assignment(schedule_matrix)
    assigned_meetings = [meetings[i] for i in assignment[:, 1]]

    return assigned_meetings

# 示例数据
meetings = [
    {'start_time': '09:00', 'duration': 1},
    {'start_time': '10:00', 'duration': 1},
    {'start_time': '11:00', 'duration': 1}
]
rooms = [
    {'room_id': 1, 'available': True},
    {'room_id': 2, 'available': False},
    {'room_id': 3, 'available': True}
]

# 调度会议室
assigned_meetings = schedule_meetings(meetings, rooms)
print(assigned_meetings)
```
