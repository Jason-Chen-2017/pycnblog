   import numpy as np
   import soundfile as sf

   def generate_breathing_sound(frequency, duration):
       t = np.linspace(0, duration, int(frequency * duration), endpoint=False)
       sound = 0.5 * (1 + np.cos(2 * np.pi * frequency * t))
