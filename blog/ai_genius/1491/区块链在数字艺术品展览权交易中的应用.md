pragma solidity ^0.8.0;

// 定义艺术品结构体
struct Artwork {
    string id; // 艺术品ID
    string name; // 艺术品名称
    string creator; // 艺术家姓名
    string hash; // 艺术品哈希值
    address owner; // 艺术品所有者地址
    uint price; // 艺术品价格
    bool isSold; // 艺术品是否已售出
}

// 定义艺术品合约
contract ArtworkContract {
    mapping(string => Artwork) public artworks; // 艺术品映射表

    // 创建艺术品
    function createArtwork(string memory _id, string memory _name, string memory _creator, string memory _hash, uint _price) public {
        require(!artworks[_id].isSold, "Artwork is already sold.");
        artworks[_id] = Artwork(_id, _name, _creator, _hash, msg.sender, _price, false);
    }

    // 转让艺术品
    function transferArtwork(string memory _id, address _buyer) public payable {
        require(artworks[_id].owner == msg.sender, "Only the owner can transfer artwork.");
        require(msg.value == artworks[_id].price, "Payment must match the price.");
        
        artworks[_id].owner = _buyer;
        artworks[_id].isSold = true;
        payable(msg.sender).transfer(msg.value);
    }

    // 查询艺术品信息
    function getArtworkInfo(string memory _id) public view returns (Artwork memory) {
        return artworks[_id];
    }
}
