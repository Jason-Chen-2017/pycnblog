#### 2.3 智能合约代码解析

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract AdPlatform {
    struct AdRequest {
        address advertiser;
        string adType;
        string targetAudience;
        uint256 budget;
        bool isAwarded;
    }

    mapping(uint256 => AdRequest) public adRequests;
    mapping(uint256 => mapping(address => uint256)) public bids;

    event AdCreated(uint256 requestId);
    event BidPlaced(uint256 requestId, address bidder, uint256 amount);
    event AdAwarded(uint256 requestId, address platform);

    function createAdRequest(
        string memory _adType,
        string memory _targetAudience,
        uint256 _budget
    ) public {
        uint256 requestId = adRequests.length;
        adRequests[requestId] = AdRequest(
            msg.sender,
            _adType,
            _targetAudience,
            _budget,
            false
        );
        emit AdCreated(requestId);
    }

    function placeBid(uint256 requestId, uint256 amount) public {
        require(adRequests[requestId].isAwarded == false, "Ad already awarded");
        bids[requestId][msg.sender] = amount;
        emit BidPlaced(requestId, msg.sender, amount);
    }

    function awardAd(uint256 requestId, address platform) public {
        require(adRequests[requestId].isAwarded == false, "Ad already awarded");
        require(msg.sender == adRequests[requestId].advertiser, "Only advertiser can award");
        adRequests[requestId].isAwarded = true;
        emit AdAwarded(requestId, platform);
    }
}
```

在这个示例中，`AdPlatform` 智能合约包含了广告需求的创建、竞价和奖励等功能。广告主可以通过 `createAdRequest` 函数创建广告需求，广告投放平台可以通过 `placeBid` 函数竞价，广告主可以通过 `awardAd` 函数选择最佳投放平台。
