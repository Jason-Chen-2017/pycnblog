pragma solidity ^0.8.0;

contract DigitalArtLease {
    address public landlord; // 出租方地址
    address public renter; // 租赁方地址
    uint256 public leaseDuration; // 租赁期限
    uint256 public rentalPrice; // 租金

    // 租赁开始时间
    uint256 public leaseStartTime;

    // 租赁状态
    enum LeaseState {
        NotStarted,
        Ongoing,
        Completed
    }

    LeaseState public leaseState;

    constructor(uint256 _leaseDuration, uint256 _rentalPrice) {
        landlord = msg.sender;
        leaseDuration = _leaseDuration;
        rentalPrice = _rentalPrice;
        leaseState = LeaseState.NotStarted;
    }

    // 租赁开始
    function startLease() public {
        require(leaseState == LeaseState.NotStarted, "Lease has already started");
        require(msg.sender == landlord, "Only the landlord can start the lease");
        renter = msg.sender;
        leaseStartTime = block.timestamp;
        leaseState = LeaseState.Ongoing;
    }

    // 租赁结束
    function endLease() public {
        require(leaseState == LeaseState.Ongoing, "Lease is not ongoing");
        require(msg.sender == landlord, "Only the landlord can end the lease");
        leaseState = LeaseState.Completed;
    }

    // 支付租金
    function payRental() public payable {
        require(leaseState == LeaseState.Ongoing, "Lease is not ongoing");
        require(msg.sender == renter, "Only the renter can pay the rental");
        require(msg.value >= rentalPrice, "Insufficient payment");
        payable(landlord).transfer(msg.value);
    }

    // 查询租赁状态
    function checkLeaseState() public view returns (LeaseState) {
        if (block.timestamp - leaseStartTime >= leaseDuration) {
            return LeaseState.Completed;
        } else {
            return LeaseState.Ongoing;
        }
    }
}
