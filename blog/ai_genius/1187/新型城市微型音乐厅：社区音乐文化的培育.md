import numpy as np
import matplotlib.pyplot as plt
from scipy.integrate import solve_ode

# 声学模拟函数
def 声学模拟(x0, t):
    # 声学模型参数
    c = 343  # 声速
    p0 = 0  # 初始声压
    phi0 = 0  # 初始声势
    
    # 声学模拟方程
    def dvar_dtime(y, x):
        p, phi = y
        dpdt = -c * np.gradient(phi, x) / np.gradient(p, x)
        dphidt = -p / c
    
        return [dpdt, dphidt]

    # 求解声学模拟方程
    result = solve_ode(dvar_dtime, (p0, phi0), (0, t), x0)
    
    # 返回声压分布
    return result.y[0, :]

# 声学模拟示例

x = np.linspace(-5, 5, 100)
