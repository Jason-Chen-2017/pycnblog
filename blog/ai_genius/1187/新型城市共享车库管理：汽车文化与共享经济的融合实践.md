```python
import random
import time

class CarPark:
    def __init__(self, name, capacity):
        self.name = name
        self.capacity = capacity
        self.parking_spots = [True] * capacity  # 初始化停车位状态

    def park_car(self, car_id):
        if any(self.parking_spots):
            # 找到第一个空的停车位
            for i in range(len(self.parking_spots)):
                if self.parking_spots[i]:
                    self.parking_spots[i] = False
                    print(f"Car {car_id} parked at spot {i}.")
                    return True
            print("No available parking spots.")
            return False
        else:
            print("CarPark is full.")
            return False

    def leave_car(self, car_id):
        for i in range(len(self.parking_spots)):
            if not self.parking_spots[i]:
                self.parking_spots[i] = True
                print(f"Car {car_id} left spot {i}.")
                return True
        print("No matching car found.")
        return False

# 模拟共享车库管理系统
car_park = CarPark("Central CarPark", 100)

# 模拟停车
cars = [random.randint(1000, 9999) for _ in range(50)]
for car in cars:
    car_park.park_car(car)

# 模拟离车
cars = random.sample(cars, 20)
for car in cars:
    car_park.leave_car(car)
```

```python
            # 找到第一个空的停车位

                if self.parking_spots[i]:

        for i in range(len(self.parking_spots)):
            if not self.parking_spots[i]:
                self.parking_spots[i] = True
                print(f"Car {car_id} left spot {i}.")
                return True

        return False
```

- **交易完成**：在供需匹配完成后，通过平台完成交易，实现资源的使用和收益。

## 共享经济的原理与实践

共享经济是一种基于资源共享的商业模式，通过平台连接供给者和需求者，实现闲置资源的最大化利用。共享经济在多个领域得到了广泛应用，如住宿、交通、餐饮等。

### 1. 共享经济的原理

共享经济的基本原理如下：

- **供需匹配**：通过平台，将需求者的需求与供给者的资源进行匹配，实现资源的最大化利用。
- **价值创造**：通过共享模式，为供需双方创造价值。供给者通过共享闲置资源获得收益，需求者通过共享资源获得便利。
- **交易完成**：在供需匹配完成后，通过平台完成交易，实现资源的使用和收益。

### 2. 共享经济的实践

共享经济在多个领域都有成功的实践，以下以住宿和交通为例：

- **住宿**：共享住宿平台（如Airbnb）将个人闲置的房间或房屋共享给游客，实现了闲置资源的充分利用。

### 附录A：代码实现示例

以下是一个简单的Python代码示例，用于实现共享车库管理系统的一部分功能：

```python
class CarPark:
    def __init__(self, name, capacity):
        self.name = name
        self.capacity = capacity
        self.parking_spots = [True] * capacity  # 初始化停车位状态

    def park_car(self, car_id):
        if any(self.parking_spots):
            # 找到第一个空的停车位
            for i in range(len(self.parking_spots)):
                if self.parking_spots[i]:
                    self.parking_spots[i] = False
                    print(f"Car {car_id} parked at spot {i}.")
                    return True
            print("No available parking spots.")
            return False
        else:
            print("CarPark is full.")
            return False

    def leave_car(self, car_id):
        for i in range(len(self.parking_spots)):
            if not self.parking_spots[i]:
                self.parking_spots[i] = True
                print(f"Car {car_id} left spot {i}.")
                return True
        print("No matching car found.")
        return False

# 模拟共享车库管理系统
car_park = CarPark("Central CarPark", 100)

# 模拟停车
cars = [random.randint(1000, 9999) for _ in range(50)]
for car in cars:
    car_park.park_car(car)

# 模拟离车
cars = random.sample(cars, 20)
for car in cars:
    car_park.leave_car(car)
```
