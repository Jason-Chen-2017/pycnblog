```python
# leadership_system.py

import json
from collections import defaultdict

# 评估模型
class AssessmentModel:
    def __init__(self):
        self.assessments = defaultdict(list)

    def add_assessment(self, leader_id, assessment_data):
        self.assessments[leader_id].append(assessment_data)

    def get_average_score(self, leader_id):
        scores = [item['score'] for item in self.assessments[leader_id]]
        return sum(scores) / len(scores) if scores else 0

# 提升计划模型
class ImprovementPlanModel:
    def __init__(self):
        self.plans = defaultdict(dict)

    def add_plan(self, leader_id, plan_data):
        self.plans[leader_id] = plan_data

    def get_plan(self, leader_id):
        return self.plans[leader_id]

# 领导力管理系统
class LeadershipSystem:
    def __init__(self):
        self.assessment_model = AssessmentModel()
        self.improvement_plan_model = ImprovementPlanModel()

    def add_leader_assessment(self, leader_id, assessment_data):
        self.assessment_model.add_assessment(leader_id, assessment_data)

    def generate_improvement_plan(self, leader_id):
        average_score = self.assessment_model.get_average_score(leader_id)
        plan_data = {
            'average_score': average_score,
            'suggestions': self._generate_suggestions(average_score)
        }
        self.improvement_plan_model.add_plan(leader_id, plan_data)
        return plan_data

    def _generate_suggestions(self, average_score):
        suggestions = []
        if average_score < 5:
            suggestions.append("加强沟通技能培训")
        if average_score < 7:
            suggestions.append("提高决策能力")
        if average_score < 8:
            suggestions.append("加强团队建设")
        return suggestions

# 测试
if __name__ == "__main__":
    system = LeadershipSystem()
    system.add_leader_assessment('leader1', {'communication_skills': 4, 'decision_making_ability': 5, 'team_building': 6})
    plan = system.generate_improvement_plan('leader1')
    print(json.dumps(plan, indent=2))
```

**代码应用解读与分析：**

1. **评估模型**：该模型用于存储和管理领导者的评估数据。`add_assessment` 方法用于添加新的评估数据，`get_average_score` 方法用于计算领导者的平均评分。

2. **提升计划模型**：该模型用于生成和管理提升计划。`add_plan` 方法用于添加新的提升计划，`get_plan` 方法用于获取特定领导者的提升计划。

3. **领导力管理系统**：该系统是整个领导力管理流程的核心。`add_leader_assessment` 方法用于添加领导者的评估数据，`generate_improvement_plan` 方法用于根据评估结果生成提升计划。
