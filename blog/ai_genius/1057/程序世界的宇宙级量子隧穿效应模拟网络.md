import matplotlib.pyplot as plt
from qiskit import QuantumCircuit
from qiskit.visualization import plot_bloch_vector

# 定义量子隧穿模拟参数
V0 = 5.0  # 势垒高度
L = 10.0  # 势垒宽度
E = 2.5   # 粒子能量
dx = 0.1  # 空间离散化步长
N = int(L / dx)  # 空间离散化点数
psi = np.zeros(N)  # 初始化波函数

# 计算入射波、反射波和透射波的系数
k = np.sqrt(2 * E / (m * hbar**2))  # 波数
A = np.sqrt(2 * m * (E - V0) / hbar**2) * np.exp(-k**2 * x0**2 / 2)  # 入射波系数
B = -k * np.exp(-k**2 * x0**2 / 2)  # 反射波系数
psi[0:N//2] = A * np.exp(k * x[:N//2])
psi[N//2:] = B * np.exp(-k * x[N//2:])

# 应用时间演化
t_final = 20.0  # 时间演化结束点
dt = 0.01       # 时间步长
for t in np.arange(0, t_final, dt):
    for i in range(1, N - 1):
        psi[i] = np.exp(-1j * (E - V0) * dt / hbar) * (psi[i - 1] + psi[i + 1] - 2 * psi[i])

# 可视化波函数
x = np.linspace(0, L, N)
plt.plot(x, np.abs(psi)**2)
plt.xlabel('Position')
plt.ylabel('Probability Density')
plt.title('Quantum Tunnelling Effect Simulation')
plt.show()
