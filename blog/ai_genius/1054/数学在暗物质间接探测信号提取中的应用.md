import numpy as np

def preprocess_signal(signal, filter_type='lowpass'):
    if filter_type == 'lowpass':
        # 低通滤波
        cutoff_frequency = 10
        b, a = signal.filter_design.lowpass_filter(cutoff_frequency)
        signal = signal.lfilter(b, a)
    elif filter_type == 'highpass':
        # 高通滤波
        cutoff_frequency = 10
        b, a = signal.filter_design.highpass_filter(cutoff_frequency)

## Python源代码实例

以下是一个Python源代码实例，用于展示如何使用NumPy库进行信号预处理。这段代码实现了低通滤波和高通滤波两种常见预处理方法。

```python
import numpy as np
from scipy.signal import butter, filtfilt

# 生成模拟信号
signal = np.random.normal(size=1000)

# 低通滤波
def lowpass_filter(signal, cutoff_frequency):
    b, a = butter(4, cutoff_frequency)
    filtered_signal = filtfilt(b, a, signal)
    return filtered_signal

# 高通滤波
def highpass_filter(signal, cutoff_frequency):
    b, a = butter(4, cutoff_frequency, btype='high')
    filtered_signal = filtfilt(b, a, signal)
    return filtered_signal

# 应用低通滤波
lowpass_signal = lowpass_filter(signal, cutoff_frequency=5)

# 应用高通滤波
highpass_signal = highpass_filter(signal, cutoff_frequency=5)

# 绘制原始信号和滤波后信号
import matplotlib.pyplot as plt

plt.figure(figsize=(10, 5))
plt.plot(signal, label='原始信号')
plt.plot(lowpass_signal, label='低通滤波后信号')
plt.plot(highpass_signal, label='高通滤波后信号')
plt.legend()
plt.show()
```

这段代码首先生成了一个模拟信号，然后定义了低通滤波和高通滤波的两个函数。通过调用这些函数，可以分别对原始信号进行低通滤波和高通滤波。最后，使用matplotlib库绘制了原始信号和滤波后信号，以便观察滤波效果。
