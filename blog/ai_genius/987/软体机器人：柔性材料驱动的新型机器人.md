### 3. 源代码详细实现

以下是软体机器人家庭清洁项目的源代码实现：

```python
#!/usr/bin/env python
import rospy
import numpy as np
from geometry_msgs.msg import Twist
from sensor_msgs.msg import LaserScan
from std_msgs.msg import String

class SoftRobotCleaner:
    def __init__(self):
        rospy.init_node('soft_robot_cleaner', anonymous=True)
        self.cmd_vel_pub = rospy.Publisher('/cleaner_robot/cmd_vel', Twist, queue_size=10)
        self.laser_sub = rospy.Subscriber('/cleaner_robot/laser', LaserScan, self.laser_callback)
        self.obstacle_detected = False

    def move_forward(self, speed):
        twist = Twist()
        twist.linear.x = speed
        self.cmd_vel_pub.publish(twist)

    def move_backward(self, speed):
        twist = Twist()
        twist.linear.x = -speed
        self.cmd_vel_pub.publish(twist)

    def turn_left(self, angle):
        twist = Twist()
        twist.angular.z = angle
        self.cmd_vel_pub.publish(twist)

    def turn_right(self, angle):
        twist = Twist()
        twist.angular.z = -angle
        self.cmd_vel_pub.publish(twist)

    def laser_callback(self, data):
        ranges = data.ranges
        min_range = min(ranges)
        if min_range < 0.5:
            self.obstacle_detected = True

    def clean_room(self):
        while not rospy.is_shutdown():
            if self.obstacle_detected:
                self.turn_left(0.5)
            else:
                self.move_forward(0.5)
            rospy.sleep(0.1)

if __name__ == '__main__':
    try:
        cleaner = SoftRobotCleaner()
        cleaner.clean_room()
    except rospy.ROSInterruptException:
        pass
```

- `turn_right`方法：实现机器人的右转功能，根据给定的角度参数发送角度命令。

- `clean_room`方法：实现机器人的清洁功能，根据激光雷达数据控制机器人的运动。
