def select_prompt_words(user_input, context):
    # 1. 初始化提示词列表
    prompt_list = []

    # 2. 遍历用户输入中的每个单词
    for word in user_input.split():
        # 3. 如果单词与情感相关，添加到提示词列表
        if is_emotion_related(word):
            prompt_list.append(word)

    # 4. 根据上下文中的词频对提示词列表进行排序
    prompt_list = sorted(prompt_list, key=lambda w: context.count(w), reverse=True)

    # 5. 返回前5个提示词
    return prompt_list[:5]

# 辅助函数：判断单词是否与情感相关
def is_emotion_related(word):
    emotion_keywords = ["happy", "sad", "angry", "excited"]
    return word.lower() in emotion_keywords
