```python
import matplotlib.pyplot as plt

# 参数设置

N = 1000  # 天体数量
dt = 1e-3  # 时间步长
t_max = 1e5  # 总时间

# 初始化天体位置和速度

velocities = np.random.rand(N, 3) * 100  # 速度在[-100, 100]范围内

# N-body模拟主循环
for t in range(int(t_max / dt)):
    # 计算引力势能
    forces = np.zeros((N, 3))
    for i in range(N):
        for j in range(N):
            if i != j:
                distance = positions[i] - positions[j]
                r = np.linalg.norm(distance)
                force = G * (positions[i] * positions[j]) / r**2
                forces[i] += force
                forces[j] -= force
    
    # 更新位置和速度
    positions += velocities * dt
    velocities += forces / np.sum(forces) * dt

# 绘制结果

plt.title('N-body Simulation: Star Formation')
plt.xlabel('Position X')
plt.ylabel('Position Y')
plt.show()
```

通过这些公式，我们可以计算每个天体的加速度，并据此更新位置和速度。

```python
# N-body模拟主循环
for t in range(int(t_max / dt)):
    # 计算引力
    forces = np.zeros((N, 3))
    for i in range(N):
        for j in range(N):
            if i != j:
                distance = positions[i] - positions[j]
                r = np.linalg.norm(distance)
                force = G * (positions[i] * positions[j]) / r**2
                forces[i] += force
                forces[j] -= force
    
    # 更新位置和速度
    positions += velocities * dt
    velocities += forces / np.sum(forces) * dt
```
