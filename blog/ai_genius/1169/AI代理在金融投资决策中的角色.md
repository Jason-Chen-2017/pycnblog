  ```python
  y_train = np.array([1, 2, 2.5, 3, 3.5])

  # 创建线性回归模型
  model = LinearRegression()

  # 训练模型
  model.fit(X_train, y_train)

  # 预测
  predictions = model.predict(X_train)

  # 打印预测结果
  print(predictions)
  ```

```python
gamma = 0.95

# 创建Q表
Q = np.zeros((state_size, action_size))

# 强化学习训练
for episode in range(1000):
    state = scaled_data[episode]
    done = False
    total_reward = 0

    while not done:
        action = np.argmax(Q[state])
        next_state, reward, done = environment.step(state, action)
        Q[state, action] = Q[state, action] + learning_rate * (reward + gamma * np.max(Q[next_state]) - Q[state, action])
        state = next_state
        total_reward += reward

# 打印Q表
```

```python
# 投资组合优化实例：遗传算法

import numpy as np
import random
from sklearn.metrics import mean_squared_error

# 假设已有资产预期收益和风险数据
expected_returns = np.array([0.05, 0.06, 0.04])
risks = np.array([0.1, 0.15, 0.12])

# 初始化参数
n_assets = len(expected_returns)
population_size = 100
generations = 100
crossover_rate = 0.8
mutation_rate = 0.1

# 生成初始种群
population = np.random.rand(population_size, n_assets)

# 适应度函数
def fitness_function(population, expected_returns, risks):
    portfolios = []
    for i in range(population_size):
        portfolio = population[i]
        portfolio_return = np.dot(portfolio, expected_returns)
        portfolio_risk = np.sqrt(np.dot(portfolio.T, np.dot(risks, portfolio)))
        portfolios.append((portfolio, portfolio_return, portfolio_risk))
    portfolios.sort(key=lambda x: x[2])
    return [1 / x[2] for x in portfolios]

# 遗传算法过程
for generation in range(generations):
    fitness_scores = fitness_function(population, expected_returns, risks)
    new_population = []

    for _ in range(population_size):
        parent1, parent2 = random.sample(population, 2)
        crossover_point = random.randint(1, n_assets - 1)

        child = parent1[:crossover_point] + parent2[crossover_point:]
        if random.random() < mutation_rate:
            child[random.randint(0, n_assets - 1)] = 1 - child[random.randint(0, n_assets - 1)]

        new_population.append(child)

    population = new_population

# 找到最优投资组合
best_fitness = max(fitness_scores)
best_portfolio = population[fitness_scores.index(best_fitness)]

# 打印最优投资组合
print("最优投资组合：", best_portfolio)
```

在这个实例中，我们使用遗传算法优化投资组合，目标是降低投资组合的风险。通过多次迭代，找到最优的投资组合。
