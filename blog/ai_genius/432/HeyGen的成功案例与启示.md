```python
# 伪代码：基于协同过滤的推荐算法
def collaborative_filtering(user_similarity_matrix, user_preferences, item_scores):
    # 计算用户相似度
    user_similarity = calculate_similarity(user_similarity_matrix)
    
    # 计算推荐分数
    recommendation_scores = []
    for user, preferences in user_preferences.items():
        scores = []
        for item, preference in preferences.items():
            score = calculate_score(user_similarity[user], item, preference, item_scores)
            scores.append(score)
        recommendation_scores.append((user, sum(scores) / len(scores)))
    
    # 排序和返回推荐列表
    sorted_recommendations = sorted(recommendation_scores, key=lambda x: x[1], reverse=True)
    return sorted_recommendations
```

\end{aligned}
$$
