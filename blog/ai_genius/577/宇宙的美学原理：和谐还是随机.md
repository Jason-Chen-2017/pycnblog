import numpy as np

def simulate_cosmic_background_radiation(n_samples=1000, temperature=2.7):
    """
    模拟宇宙背景辐射的温度分布。
    
    参数：
    n_samples: int，模拟的温度数据点的数量。
    temperature: float，宇宙背景辐射的平均温度。
    
    返回：

plt.ylabel('频数')

plt.legend()
plt.show()
