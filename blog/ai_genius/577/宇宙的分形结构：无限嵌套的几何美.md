分形的主要性质包括：

1. **自相似性**：分形在不同尺度上保持相似的结构。
2. **无限性**：分形具有无限细节，无论我们如何放大或缩小，都无法达到其结构的尽头。
3. **非整数维数**：分形的维数通常不是整数，而是介于整数之间。这种非整数维数反映了分形在空间中的复杂性和多样性。

import numpy as np
import matplotlib.pyplot as plt

def mandelbrot(c, max_iter):
    z = 0
    n = 0
    while abs(z) <= 2 and n < max_iter:
        z = z*z + c
        n += 1
    return n

def generate_mandelbrot(xmin, xmax, ymin, ymax, width, height, max_iter):
    real_part = np.linspace(xmin, xmax, width)
    imag_part = np.linspace(ymin, ymax, height)
    return mandelbrot(np.array([real_part, imag_part]).T, max_iter)

def plot_mandelbrot(mandelbrot_image):
    plt.imshow(mandelbrot_image, cmap='hot', extent=[xmin, xmax, ymin, ymax])
    plt.colorbar()
    plt.xlabel('Real')
    plt.ylabel('Imaginary')
    plt.title('Mandelbrot Set')
    plt.show()

if __name__ == "__main__":
