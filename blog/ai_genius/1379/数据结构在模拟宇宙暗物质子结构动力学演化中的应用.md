import numpy as np

def initialize_particles(num_particles, size):
    # 初始化粒子的位置和速度
    positions = np.random.rand(num_particles, 3) * size
    velocities = np.random.rand(num_particles, 3) * size
    masses = np.random.rand(num_particles) * 1e11
    return positions, velocities, masses

def compute_interactions(positions, masses):
    # 计算粒子之间的相互作用力
    forces = np.zeros((len(positions), 3))
    for i in range(len(positions)):
        for j in range(i+1, len(positions)):
            distance = positions[i] - positions[j]
            distance = np.linalg.norm(distance)
            force = -G * masses[i] * masses[j] / distance**2
            force = force * distance / (distance + 1e-10)
            forces[i] += force
            forces[j] -= force
    return forces

def update_particles(positions, velocities, masses, dt):
    # 更新粒子的位置和速度
    forces = compute_interactions(positions, masses)
    accelerations = forces / masses
    velocities += accelerations * dt
    positions += velocities * dt
    return positions, velocities

def simulate_evolution(num_particles, size, dt, total_time):
    # 模拟暗物质子结构的演化
    positions, velocities, masses = initialize_particles(num_particles, size)
    while True:
        positions, velocities = update_particles(positions, velocities, masses, dt)
        if np.linalg.norm(positions[-1] - positions[-2]) < 1e-5:
            break
    return positions

#### 4.3 数学模型

在算法中，我们使用了牛顿引力定律来计算粒子之间的相互作用力。牛顿引力定律可以表示为：

$$
F = G \frac{m_1 m_2}{r^2}
$$

其中，$F$ 是相互作用力，$G$ 是万有引力常数，$m_1$ 和 $m_2$ 是两个粒子的质量，$r$ 是它们之间的距离。

为了在计算机上实现这个模型，我们需要对距离和力的计算进行数值近似。在上述算法中，我们使用了欧几里得距离来计算粒子之间的距离：

$$
r = \sqrt{(x_2 - x_1)^2 + (y_2 - y_1)^2 + (z_2 - z_1)^2}
$$

其中，$(x_1, y_1, z_1)$ 和 $(x_2, y_2, z_2)$ 分别是两个粒子的位置。
