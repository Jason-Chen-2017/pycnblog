```python
def rgb_to_hsv(r, g, b):
    r, g, b = r / 255.0, g / 255.0, b / 255.0
    max_color = max(r, g, b)
    min_color = min(r, g, b)
    delta = max_color - min_color

    if delta == 0:
        h = 0
    elif max_color == r:
        h = (60 * ((g - b) / delta) + 360) % 360
    elif max_color == g:
        h = (60 * ((b - r) / delta) + 120) % 360
    elif max_color == b:
        h = (60 * ((r - g) / delta) + 240) % 360

    if max_color == 0:
        s = 0
    else:
        s = delta / max_color

    v = max_color

    return h, s, v
```

def match_colors(color1, color2):
