HuffmanCoding(data):
    # 计算每个字符出现的频率
    frequency = [0] * 256
    for char in data:
        frequency[ord(char)] += 1

    # 构建霍夫曼树
    heap = MinHeap()
    for i in range(256):
        if frequency[i] > 0:
            node = Node(char=chr(i), frequency=frequency[i])
            heap.insert(node)

    while heap.size() > 1:
        left = heap.extract_min()
        right = heap.extract_min()
        merged = Node(left=frequency[left], right=frequency[right])
        heap.insert(merged)

    # 生成编码表
    encoding = {}
    def build_encoding(node, path):
        if node.left is None and node.right is None:
            encoding[node.char] = path
        else:
            build_encoding(node.left, path + "0")
            build_encoding(node.right, path + "1")

    build_encoding(heap.root, "")

    # 编码数据
    encoded_data = ""
    for char in data:
        encoded_data += encoding[char]

    return encoded_data

##### 源代码实现

```python
import heapq

class Node:
    def __init__(self, char=None, freq=None, left=None, right=None):
        self.char = char
        self.freq = freq
        self.left = left
        self.right = right

    def is_leaf(self):
        return self.left is None and self.right is None

def build_huffman_tree(data):
    frequency = [0] * 256
    for char in data:
        frequency[ord(char)] += 1

    heap = [Node(freq=frequency[i]) for i in range(256) if frequency[i] > 0]
    heapq.heapify(heap)

    while len(heap) > 1:
        left = heapq.heappop(heap)
        right = heapq.heappop(heap)
        merged = Node(left=left, right=right, freq=left.freq + right.freq)
        heapq.heappush(heap, merged)

    return heap[0]

def build_encoding(node, path="", encoding={}):
    if node.is_leaf():
        encoding[node.char] = path
        return encoding

    build_encoding(node.left, path + "0", encoding)
    build_encoding(node.right, path + "1", encoding)

    return encoding

def huffman_encoding(data):
    tree = build_huffman_tree(data)
    encoding = build_encoding(tree)
    encoded_data = ""

    for char in data:
        encoded_data += encoding[char]

    return encoded_data

def huffman_decoding(encoded_data, encoding):
    decoded_data = ""
    current_char = ""
    for bit in encoded_data:
        current_char += bit
        if current_char in encoding.values():
            decoded_data += encoding.inverse[current_char]
            current_char = ""

    return decoded_data

data = "AAAAABBBB"
encoded_data = huffman_encoding(data)
print(f"Encoded data: {encoded_data}")

decoded_data = huffman_decoding(encoded_data, encoding)
print(f"Decoded data: {decoded_data}")
```

- `Node`类：表示霍夫曼树的节点，包含字符、频率、左子节点和右子节点。
- `build_huffman_tree`函数：构建霍夫曼树，首先计算每个字符的频率，然后使用最小堆构建霍夫曼树。
- `build_encoding`函数：生成编码表，从根节点开始，递归构建编码表。
- `huffman_encoding`函数：编码数据，遍历数据，根据编码表生成编码后的数据。
- `huffman_decoding`函数：解码数据，遍历编码后的数据，根据编码表还原原始数据。
