const users = [];
const contents = [];

app.post('/api/auth/register', async (req, res) => {
  const { username, password } = req.body;
  const hashedPassword = await bcrypt.hash(password, 10);
  users.push({ username, password: hashedPassword });
  res.status(200).json({ message: '注册成功！' });
});

app.post('/api/auth/login', async (req, res) => {
  const { username, password } = req.body;
  const user = users.find((u) => u.username === username);
  if (user && await bcrypt.compare(password, user.password)) {
    const token = jwt.sign({ username }, 'secretKey');
    res.status(200).json({ token });
  } else {
    res.status(401).json({ message: '登录失败！' });
  }
});

app.post('/api/content', async (req, res) => {
  const { content } = req.body;
  contents.push({ content });
  res.status(200).json({ message: '发布成功！' });
});

app.get('/api/content', (req, res) => {
  res.status(200).json({ contents });
});

const PORT = process.env.PORT || 3000;

from flask import Flask, render_template, request, jsonify
import json
import psycopg2

app = Flask(__name__)

# 数据库连接
def get_db_connection():
    conn = psycopg2.connect(
        host="localhost",
        database="slow_life_health",
        user="postgres",
        password="password"
    )
    return conn

@app.route('/')
def index():
    return render_template('index.html')

@app.route('/evaluate', methods=['POST'])
def evaluate():
    data = request.json
    result = evaluate_health(data)
    return jsonify(result)

def evaluate_health(data):
    # 数据处理和分析
    stress = data['stress']
    anxiety = data['anxiety']
    depression = data['depression']
    mental_health = data['mental_health']

    # 数据存储
    conn = get_db_connection()
    cursor = conn.cursor()
    cursor.execute("INSERT INTO health_records (stress, anxiety, depression, mental_health) VALUES (%s, %s, %s, %s)", (stress, anxiety, depression, mental_health))
    conn.commit()
    cursor.close()
    conn.close()

    # 结果分析
    result = analyze_health(stress, anxiety, depression, mental_health)

    return result

def analyze_health(stress, anxiety, depression, mental_health):
    # 分析逻辑
    if stress > 5 or anxiety > 5 or depression > 5:
        return "您的心理健康状态较为严峻，建议寻求专业心理帮助。"
    elif mental_health < 3:
        return "您的心理健康状态一般，请注意调整生活节奏和自我管理能力。"
    else:
        return "您的心理健康状态良好，请继续保持并关注心理健康。"

if __name__ == '__main__':
    app.run(debug=True)
