import numpy as np

def is_safe(queen_position, new_position):
    for i in range(len(queen_position)):
        if queen_position[i] == new_position[i] or \
           queen_position[i] - new_position[i] == i - j or \
           queen_position[i] - new_position[i] == -i + j:
            return False
    return True

def solve_n_queens(queen_position, j):
    if j == len(queen_position):
        return True
    for i in range(len(queen_position)):
        if is_safe(queen_position, [i, j]):
            queen_position[j] = i
            if solve_n_queens(queen_position, j+1):
                return True
            queen_position[j] = -1
    return False

def print_solution(queen_position):
    board = [[0 for x in range(len(queen_position))] for y in range(len(queen_position))]
    for i in range(len(queen_position)):
        board[queen_position[i]][i] = 1
    for row in board:
        print(' '.join(str(x) for x in row))

def main():
    queen_position = [-1] * len(queen_position)
    if not solve_n_queens(queen_position, 0):
        print("No solution exists")
    else:
        print_solution(queen_position)

if __name__ == "__main__":
    main()
