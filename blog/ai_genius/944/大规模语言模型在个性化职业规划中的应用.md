```python
# 分词和词嵌入
tokens = sent_tokenize(user_input)
embeddings = model.layers[0].call(tokens)

# 模型推理
predictions = model.predict(embeddings)

# 解码预测结果
predicted_job = decode_predictions(predictions)

# 输出职业建议
print("根据您的需求，以下职业可能适合您：")
for job in predicted_job:
    print(job)
```
