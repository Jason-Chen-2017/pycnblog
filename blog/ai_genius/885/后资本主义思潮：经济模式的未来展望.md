import numpy as np
import pandas as pd

def allocate_resources(resource_pool, user_demand):
    """
    分配资源的函数
    :param resource_pool: 可用资源池
    :param user_demand: 用户需求
    :return: 分配后的资源列表
    """
    # 初始化分配的资源列表
    allocated_resources = []

    # 对每个用户进行资源分配
    for user, demand in user_demand.items():
        # 分配用户资源
        allocated_resource = min(resource_pool[user], demand)
        allocated_resources.append((user, allocated_resource))

        # 更新资源池
        resource_pool[user] -= allocated_resource

    return allocated_resources

def calculate_efficiency(total_value, total_cost):
    """
    计算经济效率
    :param total_value: 总价值
    :param total_cost: 总成本
    :return: 经济效率
    """
    efficiency = total_value / total_cost
    return efficiency

if __name__ == "__main__":
