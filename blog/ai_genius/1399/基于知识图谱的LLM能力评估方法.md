```python
import spacy
from nltk.tokenize import word_tokenize

def preprocess_text(text):
    # 使用Spacy进行文本清洗和分词
    nlp = spacy.load('en_core_web_sm')
    doc = nlp(text)
    tokens = [token.text.lower() for token in doc if not token.is_punct]
    
    # 使用NLTK进行实体识别
    entities = []
    for ent in doc.ents:
        entities.append({
            'entity': ent.text,
            'type': ent.label_
        })
    
    return tokens, entities
```
