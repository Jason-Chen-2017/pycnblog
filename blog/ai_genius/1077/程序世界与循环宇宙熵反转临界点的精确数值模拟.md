```python
import numpy as np
import matplotlib.pyplot as plt

def lorenz_system(x, y, z, sigma=10, rho=28, beta=8/3):
    dx = sigma * (y - x)
    dy = x * (rho - z) - y
    dz = x * y - beta * z
    return dx, dy, dz

def simulate_lorenz(x0, y0, z0, t_max=100, dt=0.01):
    t = 0
    x, y, z = x0, y0, z0
    trajectory = [(x, y, z)]
    while t < t_max:
        dx, dy, dz = lorenz_system(x, y, z)
        x += dx * dt
        y += dy * dt
        z += dz * dt
        t += dt
        trajectory.append((x, y, z))
    return trajectory

trajectory = simulate_lorenz(1, 1, 1)
x, y, z = zip(*trajectory)

plt.plot(x, y)
plt.xlabel("X")
plt.ylabel("Y")
plt.title("Lorenz Attractor")
plt.show()
```
