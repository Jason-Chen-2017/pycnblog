```python

    if n == 0:
        return 1
    else:
        return n * factorial(n - 1)

print(factorial(5))  # 输出120
```

2. **概率算法**：概率算法用于计算事件发生的概率。以下是一个简单的概率算法示例，用于计算骰子投掷得到特定数字的概率：

```python
import random

def roll_dice():
    return random.randint(1, 6)

def probability_of_specific_number(number, trials=1000):
    count = 0
    for _ in range(trials):
        if roll_dice() == number:
            count += 1
    return count / trials

print(probability_of_specific_number(6))  # 输出接近1/6的值
```

```python
import numpy as np
import matplotlib.pyplot as plt
import random

# 递归算法：计算阶乘
def factorial(n):
    if n == 0:
        return 1
    else:
        return n * factorial(n - 1)

# 概率算法：计算特定数字的概率
def roll_dice():
    return random.randint(1, 6)

def probability_of_specific_number(number, trials=1000):
    count = 0
    for _ in range(trials):
        if roll_dice() == number:
            count += 1
    return count / trials

# 主函数
def main():
    # 递归算法实验
    n = 5
    print("递归算法结果：", factorial(n))
    
    # 概率算法实验
    number = 6
    trials = 1000
    print("概率算法结果：", probability_of_specific_number(number, trials))
    
    # 可视化结果
    probabilities = [probability_of_specific_number(i, trials) for i in range(1, 7)]
    plt.bar(range(1, 7), probabilities)
    plt.xlabel("数字")
    plt.ylabel("概率")
    plt.title("概率算法结果")
    plt.show()

if __name__ == "__main__":
    main()
```
