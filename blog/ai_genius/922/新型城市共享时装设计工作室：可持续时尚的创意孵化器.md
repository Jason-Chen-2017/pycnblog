$$
E = \sum_{i=1}^{n} w_i \times e_i

其中，\( E \) 是材料总环境影响，\( w_i \) 是第 \( i \) 种材料的使用量，\( e_i \) 是第 \( i \) 种材料的环境影响系数。

```python
import random
import math

# 遗传算法参数设置
population_size = 100
chromosome_length = 10
max_generations = 100
crossover_rate = 0.8
mutation_rate = 0.05

# 初始化种群
def initialize_population():
    population = []
    for _ in range(population_size):
        chromosome = [random.randint(0, 1) for _ in range(chromosome_length)]
        population.append(chromosome)
    return population

# 计算适应度
def evaluate_fitness(population):
    fitness = []
    for chromosome in population:
        fitness_value = calculate_fitness(chromosome)
        fitness.append(fitness_value)
    return fitness

# 计算适应度值
def calculate_fitness(chromosome):
    # 这里可以定义具体的适应度计算方法
    fitness_value = sum(chromosome)
    return fitness_value

# 选择操作
def selection(population, fitness):
    selected = []
    for _ in range(len(population)):
        # 罗马诺夫选择法
        total_fitness = sum(fitness)
        random_fitness = random.uniform(0, total_fitness)
        current_fitness = 0
        for i in range(len(population)):
            current_fitness += fitness[i]
            if current_fitness >= random_fitness:
                selected.append(population[i])
                break
    return selected

# 交叉操作
def crossover(selected):
    offspring = []
    for _ in range(len(selected) // 2):
        parent1 = selected[random.randint(0, len(selected) - 1)]
        parent2 = selected[random.randint(0, len(selected) - 1)]
        crossover_point = random.randint(1, chromosome_length - 1)
        child1 = parent1[:crossover_point] + parent2[crossover_point:]
        child2 = parent2[:crossover_point] + parent1[crossover_point:]
        offspring.append(child1)
        offspring.append(child2)
    return offspring

# 变异操作
def mutate(offspring):
    mutated = []
    for chromosome in offspring:
        for i in range(chromosome_length):
            if random.random() < mutation_rate:
                chromosome[i] = 1 if chromosome[i] == 0 else 0
        mutated.append(chromosome)
    return mutated

# 主函数
def main():
    population = initialize_population()
    for generation in range(max_generations):
        fitness = evaluate_fitness(population)
        selected = selection(population, fitness)
        offspring = crossover(selected)
        mutated = mutate(offspring)
        population = mutated
        best_solution = get_best_solution(population)
        print(f"Generation {generation}: Best Solution = {best_solution}")

# 输出最优解
def get_best_solution(population):
    fitness = evaluate_fitness(population)
    best_fitness = max(fitness)
    best_solution_index = fitness.index(best_fitness)
    return population[best_solution_index]

# 运行主函数
main()
```
