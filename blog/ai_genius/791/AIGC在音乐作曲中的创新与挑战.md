def wave_net(inputs, reuse=None):
    with tf.variable_scope("wave_net", reuse=reuse):
        # 输入层
        x = inputs
        # 隐藏层
        for i in range(5):
            x = tf.layers.conv1d(x, filters=256, kernel_size=5, activation=tf.nn.relu)
        # 输出层
        outputs = tf.layers.conv1d(x, filters=1, kernel_size=5, activation=tf.nn.sigmoid)
    return outputs

# 定义训练过程

    targets = tf.placeholder(tf.float32, shape=[None, None])

    logits = wave_net(inputs)
    loss = tf.reduce_mean(tf.square(logits - targets))
    optimizer = tf.train.AdamOptimizer(learning_rate=learning_rate)
    train_op = optimizer.minimize(loss)

    # 训练模型
    with tf.Session() as sess:
