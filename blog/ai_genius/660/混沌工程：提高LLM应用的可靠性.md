#### 案例二：增强LLM的鲁棒性

```python
import numpy as np
import matplotlib.pyplot as plt

# Lorenz系统函数
def lorenz_system(x, y, z, sigma, rho, beta):
    dx = sigma * (y - x)
    dy = x * (rho - z) - y
    dz = x * y - beta * z
    return [dx, dy, dz]

# 生成Lorenz系统轨迹
def generate_lorenz_trajectory(x0, y0, z0, sigma, rho, beta, n):
    x, y, z = x0, y0, z0
    trajectory = [[x, y, z]]
    for _ in range(n):
        dx, dy, dz = lorenz_system(x, y, z, sigma, rho, beta)
        x += dx
        y += dy
        z += dz
        trajectory.append([x, y, z])
    return trajectory

# 参数设置
x0, y0, z0 = 1.0, 1.0, 1.0
sigma = 10.0
rho = 28.0
beta = 8.0 / 3.0
n = 1000

# 生成轨迹
trajectory = generate_lorenz_trajectory(x0, y0, z0, sigma, rho, beta, n)

# 绘制轨迹图
fig = plt.figure()
ax = fig.add_subplot(111, projection='3d')
ax.plot(trajectory[:, 0], trajectory[:, 1], trajectory[:, 2])
ax.set_xlabel('X axis')
ax.set_ylabel('Y axis')
ax.set_zlabel('Z axis')
plt.show()
```

    return sequence
