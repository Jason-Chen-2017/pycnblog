```python
# users/views.py

from django.shortcuts import render, redirect
from .forms import UserRegisterForm
from django.contrib import messages
from django.contrib.auth import authenticate, login

def register(request):
    if request.method == 'POST':
        form = UserRegisterForm(request.POST)
        if form.is_valid():
            user = form.save()
            authenticate_user = authenticate(username=user.username, password=form.cleaned_data['password1'])
            login(request, authenticate_user)
            messages.success(request, f'Account created! You are now logged in.')
            return redirect('home')
        else:
            messages.error(request, f'Error registering account: {form.errors}')
    else:
        form = UserRegisterForm()

    return render(request, 'users/register.html', {'form': form})
```
