import numpy as np

def fractal_dimension(data, threshold):
    unique_values, counts = np.unique(data, return_counts=True)
    cumulative_counts = np.cumsum(counts)
    cumulative_counts = cumulative_counts / cumulative_counts[-1]
    log_threshold = np.log(threshold)
    log_counts = np.log(cumulative_counts)
    return -np.sum((log_counts - log_threshold) * cumulative_counts) / np.sum(cumulative_counts)
