```python
import torch
import torch.nn as nn
import torch.optim as optim

# 初始化参数
vocab_size = 1000
embedding_size = 64
batch_size = 32
learning_rate = 0.001
num_epochs = 10

# 初始化数据集
dataset = ["我 爱 编程", "编程 很 有趣"]

# 初始化模型
model = nn.Embedding(vocab_size, embedding_size)

# 定义损失函数和优化器
criterion = nn.CrossEntropyLoss()
optimizer = optim.Adam(model.parameters(), lr=learning_rate)

# 训练模型
for epoch in range(num_epochs):
    for sentence in dataset:
        inputs = [model.weight[vocab[word]] for word in sentence]
        targets = [vocab[word] for word in sentence[1:]]
        
        # 前向传播
        outputs = model(inputs)
        loss = criterion(outputs, torch.tensor(targets))
        
        # 反向传播
        optimizer.zero_grad()
        loss.backward()
        optimizer.step()
        
        if epoch % 10 == 0:
            print(f"Epoch [{epoch + 1}/{num_epochs}], Loss: {loss.item()}")
```
