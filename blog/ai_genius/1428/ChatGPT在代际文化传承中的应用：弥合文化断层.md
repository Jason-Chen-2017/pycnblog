```python
from flask import Flask, request, jsonify
import openai

app = Flask(__name__)

@app.route('/query', methods=['POST'])
def query():
    question = request.form['question']
    response = openai.Completion.create(
      engine="text-davinci-002",
      prompt=f"问题：{question}\n回答：",
      max_tokens=100
    )

if __name__ == '__main__':
    app.run(debug=True)
```
