## 核心算法原理讲解及Python源代码

### 1. 贝塞尔曲线原理

贝塞尔曲线是一种广泛应用于数字艺术创作的数学曲线。其基本原理是通过控制点的位置和数量，生成一条光滑的曲线。贝塞尔曲线的方程为：

$$
y = ax^3 + bx^2 + cx + d
$$

其中，$a, b, c, d$ 为常数，$x$ 为变量。

下面是使用 Python 实现贝塞尔曲线的源代码：

```python
import matplotlib.pyplot as plt
import numpy as np

def bezier_curve(p0, p1, p2, p3, n=100):
    t = np.linspace(0, 1, n)
    curve = np.zeros((n, 2))

    for i in range(n):
        curve[i, 0] = (1 - t[i])**3 * p0[0] + 3 * (1 - t[i])**2 * t[i] * p1[0] + 3 * (1 - t[i]) * t[i]**2 * p2[0] + t[i]**3 * p3[0]
        curve[i, 1] = (1 - t[i])**3 * p0[1] + 3 * (1 - t[i])**2 * t[i] * p1[1] + 3 * (1 - t[i]) * t[i]**2 * p2[1] + t[i]**3 * p3[1]

    return curve

p0 = [0, 0]
p1 = [3, 0]
p2 = [3, 3]
p3 = [0, 3]

curve = bezier_curve(p0, p1, p2, p3)

plt.plot(curve[:, 0], curve[:, 1], label='贝塞尔曲线')
plt.scatter(p0[0], p0[1], label='控制点P0')
plt.scatter(p1[0], p1[1], label='控制点P1')
plt.scatter(p2[0], p2[1], label='控制点P2')
plt.scatter(p3[0], p3[1], label='控制点P3')
plt.legend()
plt.show()
```

### 2. 概率分布原理

概率分布是数学中的一个重要概念，描述了随机变量取值的概率分布情况。在数字艺术创作中，概率分布可以用来生成具有随机美的图案。

常用的概率分布有正态分布、均匀分布等。下面是使用 Python 实现正态分布的源代码：

```python
import matplotlib.pyplot as plt
import numpy as np

mu, sigma = 0, 1
x = np.random.normal(mu, sigma, 1000)

plt.hist(x, bins=30, alpha=0.5, label='正态分布')
plt.plot(x, (1/np.sqrt(2 * np.pi * sigma**2)) * np.exp(- (x - mu)**2 / (2 * sigma**2)), label='正态分布曲线')
plt.legend()
plt.show()
```

```python
import matplotlib.pyplot as plt
import numpy as np

def bezier_curve(p0, p1, p2, p3, n=100):
    t = np.linspace(0, 1, n)
    curve = np.zeros((n, 2))

    for i in range(n):
        curve[i, 0] = (1 - t[i])**3 * p0[0] + 3 * (1 - t[i])**2 * t[i] * p1[0] + 3 * (1 - t[i]) * t[i]**2 * p2[0] + t[i]**3 * p3[0]
        curve[i, 1] = (1 - t[i])**3 * p0[1] + 3 * (1 - t[i])**2 * t[i] * p1[1] + 3 * (1 - t[i]) * t[i]**2 * p2[1] + t[i]**3 * p3[1]

    return curve

def generate_pattern(n, m, x_range, y_range, color='blue'):
    x = np.random.uniform(x_range[0], x_range[1], n)
    y = np.random.uniform(y_range[0], y_range[1], m)
    pattern = np.zeros((n, m, 3))

    for i in range(n):
        for j in range(m):
            p0 = [x[i], y[j]]
            p1 = [x[i] + np.random.uniform(-x_range[1]/5, x_range[1]/5), y[j] + np.random.uniform(-y_range[1]/5, y_range[1]/5)]
            p2 = [x[i] + np.random.uniform(-x_range[1]/5, x_range[1]/5), y[j] + np.random.uniform(-y_range[1]/5, y_range[1]/5)]
            p3 = [x[i], y[j]]
            curve = bezier_curve(p0, p1, p2, p3, n=100)

            for k in range(100):
                x1, y1 = curve[k, 0], curve[k, 1]
                if x1 >= x_range[0] and x1 <= x_range[1] and y1 >= y_range[0] and y1 <= y_range[1]:
                    pattern[i, j, 0] = color[0]
                    pattern[i, j, 1] = color[1]
                    pattern[i, j, 2] = color[2]

    return pattern

n, m = 100, 100
x_range = [-10, 10]
y_range = [-10, 10]
pattern = generate_pattern(n, m, x_range, y_range, color='blue')

plt.imshow(pattern)
plt.axis('off')
plt.show()
```
