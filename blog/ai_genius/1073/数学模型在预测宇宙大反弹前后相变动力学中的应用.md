```python
import numpy as np
from scipy.integrate import odeint
import matplotlib.pyplot as plt

# 弗里德曼方程
def friedmann_equation(E, t, H0, Omega_m, Omega_Lambda):
    a = 1 + E[0]
    dot_a = E[1]
    H = dot_a / a
    ddot_a = -3 * H * (Omega_m * (1 - a**(-3)) + Omega_Lambda * a**(-3))
    return [dot_a, ddot_a]

# 初始条件
E0 = [0, 1]  # 初始尺度因子a = 1，初始速度为1
t = np.linspace(0, 10, 1000)  # 时间从0到10 Gyr
H0 = 70   # 哈勃常数，单位为 km/s/Mpc
Omega_m = 0.3  # 宇宙质量密度参数
Omega_Lambda = 0.7  # 宇宙真空能量密度参数

# 求解微分方程
result = odeint(friedmann_equation, E0, t, args=(H0, Omega_m, Omega_Lambda))

# 提取尺度因子a
a = 1 + result[:, 0]

# 计算宇宙膨胀率H
H = result[:, 1] / a

# 绘图
plt.figure(figsize=(10, 5))
plt.subplot(1, 2, 1)
plt.plot(t, a, label=r'$a(t)$')
plt.xlabel('Time (Gyr)')
plt.ylabel('Scale Factor')
plt.legend()

plt.subplot(1, 2, 2)
plt.plot(t, H, label=r'$H(t)$')
plt.xlabel('Time (Gyr)')
plt.ylabel('Hubble Parameter (km/s/Mpc)')
plt.legend()

plt.tight_layout()
plt.show()
```
