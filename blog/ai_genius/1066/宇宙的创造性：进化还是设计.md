```python
import numpy as np
import matplotlib.pyplot as plt

def genetic_algorithm():
    population = initial_population()
    while not convergence():
        new_population = []
        for individual in population:
            if is_fit(individual):
                new_population.append(individual)
            else:
                new_population.append(mutate(individual))
        population = new_population
    return best_individual(population)

def fitness_function(individual):
    # 计算个体适应度
    fitness = 0
    for gene in individual:
        fitness += gene
    return fitness

def mutate(individual):
    # 产生新的个体
    new_individual = individual.copy()
    for gene in new_individual:
        gene = np.random.randint(0, 2)
    return new_individual

def best_individual(population):
    # 返回最优个体
    best_fitness = -1
    best_individual = None
    for individual in population:
        if fitness_function(individual) > best_fitness:
            best_fitness = fitness_function(individual)
            best_individual = individual
    return best_individual

def initial_population():
    # 生成初始种群
    population = []
    for _ in range(100):
        individual = [np.random.randint(0, 2) for _ in range(10)]
        population.append(individual)
    return population

def is_fit(individual):
    # 判断个体是否适应
    return True if all(gene == 1 for gene in individual) else False

def main():
    population = initial_population()
    while not convergence():
        new_population = []
        for individual in population:
            if is_fit(individual):
                new_population.append(individual)
            else:
                new_population.append(mutate(individual))
        population = new_population
    best_individual = best_individual(population)
    print(f"最优个体：{best_individual}")

if __name__ == "__main__":
    main()
```
