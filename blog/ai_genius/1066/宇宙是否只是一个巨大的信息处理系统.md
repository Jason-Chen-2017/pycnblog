import random

# 定义个体
class Individual:
    def __init__(self, genes):
        self.genes = genes
        self.fitness = self.calculate_fitness()

    def calculate_fitness(self):
        # 计算个体的适应度
        fitness = sum(self.genes)
        return fitness

# 创建种群
def create_population(size, gene_range):
    population = []
    for _ in range(size):
        genes = [random.randint(gene_range[0], gene_range[1]) for _ in range(len(gene_range))]
        population.append(Individual(genes))
    return population

# 选择操作
def selection(population):
    # 根据适应度选择个体
    sorted_population = sorted(population, key=lambda x: x.fitness, reverse=True)
    selected_individuals = sorted_population[:len(population) // 2]
    return selected_individuals

# 交叉操作
def crossover(parent1, parent2):
    # 交叉产生子代
    crossover_point = random.randint(1, len(parent1.genes) - 1)
    child1_genes = parent1.genes[:crossover_point] + parent2.genes[crossover_point:]
    child2_genes = parent2.genes[:crossover_point] + parent1.genes[crossover_point:]
    return Individual(child1_genes), Individual(child2_genes)

# 变异操作
def mutate(individual):
    # 变异产生新的个体
    for i in range(len(individual.genes)):
        if random.random() < 0.1:
            individual.genes[i] = random.randint(-1, 1)
    individual.fitness = individual.calculate_fitness()
    return individual

# 自然选择算法
def natural_selection(population, generations):
    for _ in range(generations):
        selected_individuals = selection(population)
        next_generation = []
        for _ in range(len(population) // 2):
            parent1, parent2 = random.sample(selected_individuals, 2)
            child1, child2 = crossover(parent1, parent2)
            next_generation.append(mutate(child1))
            next_generation.append(mutate(child2))
        population = next_generation
    return population

# 测试自然选择算法
population_size = 100
gene_range = (-10, 10)
generations = 10

population = create_population(population_size, gene_range)
best_individual = max(population, key=lambda x: x.fitness)
print(f"Best individual after {generations} generations: {best_individual.genes}, Fitness: {best_individual.fitness}")

# 输出每代最佳个体的适应度
for generation in range(1, generations + 1):
    best_individual = max(population, key=lambda x: x.fitness)
    print(f"Generation {generation}: Best fitness = {best_individual.fitness}")
```

#### B. 演化算法示例代码

以下是一个简单的演化算法示例，使用Python实现：

```python
import random

# 定义个体
class Individual:
    def __init__(self, genes):
        self.genes = genes
        self.fitness = self.calculate_fitness()

    def calculate_fitness(self):
        # 计算个体的适应度
        fitness = sum(self.genes)
        return fitness

# 创建种群
def create_population(size, gene_range):
    population = []
    for _ in range(size):
        genes = [random.randint(gene_range[0], gene_range[1]) for _ in range(len(gene_range))]
        population.append(Individual(genes))
    return population

# 选择操作
def selection(population):
    # 根据适应度选择个体
    sorted_population = sorted(population, key=lambda x: x.fitness, reverse=True)
    selected_individuals = sorted_population[:len(population) // 2]
    return selected_individuals

# 交叉操作
def crossover(parent1, parent2):
    # 交叉产生子代
    crossover_point = random.randint(1, len(parent1.genes) - 1)
    child1_genes = parent1.genes[:crossover_point] + parent2.genes[crossover_point:]
    child2_genes = parent2.genes[:crossover_point] + parent1.genes[crossover_point:]
    return Individual(child1_genes), Individual(child2_genes)

# 变异操作
def mutate(individual):
    # 变异产生新的个体
    for i in range(len(individual.genes)):
        if random.random() < 0.1:
            individual.genes[i] = random.randint(-1, 1)
    individual.fitness = individual.calculate_fitness()
    return individual

# 自然选择算法
def natural_selection(population, generations):
    for _ in range(generations):
        selected_individuals = selection(population)
        next_generation = []
        for _ in range(len(population) // 2):
            parent1, parent2 = random.sample(selected_individuals, 2)
            child1, child2 = crossover(parent1, parent2)
            next_generation.append(mutate(child1))
            next_generation.append(mutate(child2))
        population = next_generation
    return population

# 测试自然选择算法
population_size = 100
gene_range = (-10, 10)
generations = 10

population = create_population(population_size, gene_range)
best_individual = max(population, key=lambda x: x.fitness)
print(f"Best individual after {generations} generations: {best_individual.genes}, Fitness: {best_individual.fitness}")

# 输出每代最佳个体的适应度
for generation in range(1, generations + 1):
    best_individual = max(population, key=lambda x: x.fitness)
    print(f"Generation {generation}: Best fitness = {best_individual.fitness}")
```

在这个示例中，我们定义了一个`Individual`类来表示个体，每个个体都有一些基因和一个适应度。`create_population`函数用于生成初始种群，`selection`函数用于选择适应度较高的个体，`crossover`函数用于进行交叉操作，`mutate`函数用于进行变异操作。`natural_selection`函数用于执行自然选择算法，它会在指定数量的代数内迭代优化种群。

#### B. 演化算法示例代码

以下是一个简单的演化算法示例，使用Python实现：

```python
import random

# 定义个体
class Individual:
    def __init__(self, genes):
        self.genes = genes
        self.fitness = self.calculate_fitness()

    def calculate_fitness(self):
        # 计算个体的适应度
        fitness = sum(self.genes)
        return fitness

# 创建种群
def create_population(size, gene_range):
    population = []
    for _ in range(size):
        genes = [random.randint(gene_range[0], gene_range[1]) for _ in range(len(gene_range))]
        population.append(Individual(genes))
    return population

# 选择操作
def selection(population):
    # 根据适应度选择个体
    sorted_population = sorted(population, key=lambda x: x.fitness, reverse=True)
    selected_individuals = sorted_population[:len(population) // 2]
    return selected_individuals

# 交叉操作
def crossover(parent1, parent2):
    # 交叉产生子代
    crossover_point = random.randint(1, len(parent1.genes) - 1)
    child1_genes = parent1.genes[:crossover_point] + parent2.genes[crossover_point:]
    child2_genes = parent2.genes[:crossover_point] + parent1.genes[crossover_point:]
    return Individual(child1_genes), Individual(child2_genes)

# 变异操作
def mutate(individual):
    # 变异产生新的个体
    for i in range(len(individual.genes)):
        if random.random() < 0.1:
            individual.genes[i] = random.randint(-1, 1)
    individual.fitness = individual.calculate_fitness()
    return individual

# 自然选择算法
def natural_selection(population, generations):
    for _ in range(generations):
        selected_individuals = selection(population)
        next_generation = []
        for _ in range(len(population) // 2):
            parent1, parent2 = random.sample(selected_individuals, 2)
            child1, child2 = crossover(parent1, parent2)
            next_generation.append(mutate(child1))
            next_generation.append(mutate(child2))
        population = next_generation
    return population

# 测试自然选择算法
population_size = 100
gene_range = (-10, 10)
generations = 10

population = create_population(population_size, gene_range)
best_individual = max(population, key=lambda x: x.fitness)
print(f"Best individual after {generations} generations: {best_individual.genes}, Fitness: {best_individual.fitness}")

# 输出每代最佳个体的适应度
for generation in range(1, generations + 1):
    best_individual = max(population, key=lambda x: x.fitness)
    print(f"Generation {generation}: Best fitness = {best_individual.fitness}")
```

在这个示例中，我们定义了一个`Individual`类来表示个体，每个个体都有一些基因和一个适应度。`create_population`函数用于生成初始种群，`selection`函数用于选择适应度较高的个体，`crossover`函数用于进行交叉操作，`mutate`函数用于进行变异操作。`natural_selection`函数用于执行自然选择算法，它会在指定数量的代数内迭代优化种群。
