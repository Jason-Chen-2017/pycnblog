```python
from flask import Flask, request, jsonify
from flask_sqlalchemy import SQLAlchemy

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///technology_manager.db'
db = SQLAlchemy(app)

class TechnologyManager(db.Model):
    id = db.Column(db.String(120), primary_key=True)
    name = db.Column(db.String(120), nullable=False)
    technical_depth = db.Column(db.Float, nullable=False)
    management_width = db.Column(db.Float, nullable=False)

class RoleEvaluation(db.Model):
    id = db.Column(db.String(120), primary_key=True)
    user_id = db.Column(db.String(120), nullable=False)
    evaluation_date = db.Column(db.Date, nullable=False)
    evaluation_results = db.Column(db.Float, nullable=False)

@app.route('/api/role-evaluation', methods=['POST'])
def evaluate_role():
    data = request.get_json()
    user_id = data['user_id']
    technical_depth = data['technical_depth']
    management_width = data['management_width']

    role_evaluation = RoleEvaluation(
        user_id=user_id,
        evaluation_date=date.today(),
        evaluation_results=technical_depth + management_width
    )
    db.session.add(role_evaluation)
    db.session.commit()

    return jsonify({"evaluation_id": role_evaluation.id, "evaluation_date": role_evaluation.evaluation_date, "evaluation_results": role_evaluation.evaluation_results})

if __name__ == '__main__':
    db.create_all()
    app.run(debug=True)
```
