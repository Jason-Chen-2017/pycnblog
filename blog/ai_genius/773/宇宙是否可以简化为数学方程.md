```python
# 定义宇宙时间步长和观测者位置
time_steps = np.linspace(0, 13.8, 1000)  # 时间从0到13.8亿年
distance_modulus = np.zeros_like(time_steps)

# 模拟宇宙膨胀
for i, t in enumerate(time_steps):
    distance_modulus[i] = cosmo.comoving_distance(t).value

# 计算红移
redshift = 1 / distance_modulus - 1

# 可视化宇宙膨胀曲线
plt.figure(figsize=(10, 5))
plt.plot(time_steps, redshift, label='Cosmological Expansion')
plt.xlabel('Time (Gyrs)')
plt.ylabel('Redshift')
plt.legend()
plt.title('Cosmological Expansion')
plt.show()
```
