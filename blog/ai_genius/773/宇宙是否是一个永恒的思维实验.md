def compute_expansion_factor(z):
    return 1 / (1 + z)

# 计算宇宙密度
def compute_density(z):
    return cosmo.rho_c(z)

# 计算温度
def compute_temperature(z):
    return cosmo.T_CMB / (1 + z)

    plt.plot(z, a)
    plt.xlabel('Redshift')
    plt.ylabel('Expansion Factor')
    plt.title('Cosmic Expansion Factor')
    plt.show()

# 绘制宇宙密度曲线
def plot_density(z_max, z_step):
    z = np.arange(z_step, z_max, z_step)
    rho = compute_density(z)
    plt.plot(z, rho)
    plt.xlabel('Redshift')
    plt.ylabel('Density')
    plt.title('Cosmic Density')
    plt.show()

# 绘制宇宙温度曲线
def plot_temperature(z_max, z_step):
