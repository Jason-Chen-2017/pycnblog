for i in range(N-2):

H = Hcx.compose(H, front=True)

# 定义演化操作
U = QuantumCircuit(N)
for t in range(T):

    for i in range(N-1):
        q.cx(i, i+1)
    q.rx(np.pi/2, i)

# 执行量子行走算法
q = U.compose(q, front=True)

# 测量量子态
result = execute(q, Aer.get_backend("qasm_simulator"), shots=1000).result()
