    def calculate_interactions(self):
        # 计算捕食者对被捕食者的影响
        predation_rate = self.params['predation_rate']
        for predator in self.predators:
            prey_to_eat = min(predator.hunger, self.prey)
            predator.hunger -= prey_to_eat
            self.prey -= predation_rate * prey_to_eat

    def update_population(self, dt):
        # 更新种群数量
        birth_rate = self.params['birth_rate']
        death_rate = self.params['death_rate']
        for organism in self.predators + [self.prey]:
            organism.population *= (1 + birth_rate * dt - death_rate * dt)
