# 初始化权重
weights = np.random.rand(3, 3)

# 激活函数
def sigmoid(x):
    return 1 / (1 + np.exp(-x))

# 训练数据
X = np.array([[0, 0], [0, 1], [1, 0], [1, 1]])
y = np.array([[0], [1], [1], [0]])

# 学习率
learning_rate = 0.1

# 训练过程
for _ in range(1000):
    for x, target in zip(X, y):
        # 前向传播
        output = sigmoid(np.dot(x, weights))
        # 计算误差
        error = target - output
        # 反向传播
        d_output = error * (output * (1 - output))
        d_weights = np.dot(x.T, d_output)

    # 更新权重

# 测试
print(sigmoid(np.dot([0, 1], weights)))
print(sigmoid(np.dot([1, 0], weights)))

### 自组织机制的理论基础

自组织机制的理论基础是理解宇宙中自组织现象的关键，它涉及多个学科领域的交叉研究。首先，我们需要理解什么是自组织。自组织是指系统在没有外部指令的情况下，通过内部相互作用自发形成有序结构的过程。这一概念最初在20世纪中叶由理论生物学家和物理学家提出，并在多个领域得到了广泛应用。

#### 反馈机制

反馈机制是自组织过程中的核心机制之一。反馈分为正反馈和负反馈，它们在自组织系统中起着不同的作用。正反馈是一种自我增强的过程，它使系统状态向某一方向迅速变化。例如，在生态系统中，物种数量的增加可能会引发更多的资源消耗，进而导致物种数量的进一步增加。这种正反馈机制可以迅速推动系统达到某一临界状态，从而引发自组织现象。

以下是一个简单的正反馈机制的Python代码示例：

```python
def positive_feedback(population, resource, growth_rate):
    new_population = population + growth_rate * (resource / population)
    return new_population

# 初始条件
population = 100
resource = 1000
growth_rate = 0.05

# 运行100次迭代
for _ in range(100):
    resource -= population
    population = positive_feedback(population, resource, growth_rate)
    print(f"Iteration { _:02d}: Population = {population:.2f}, Resource = {resource:.2f}")
```

在这个示例中，我们模拟了一个种群数量和资源之间的关系，展示了正反馈机制如何推动种群数量的增长。

另一方面，负反馈是一种自我调节的过程，它通过抑制系统状态的变化来保持系统的稳定性。例如，在生态系统中，种群数量的增加可能会导致资源消耗增加，进而引发种群数量的下降，从而维持生态平衡。负反馈机制在自组织过程中起到了维持系统稳定的作用。

以下是一个简单的负反馈机制的Python代码示例：

```python
def negative_feedback(population, resource, growth_rate, resource_threshold):
    resource_limit = resource_threshold * population
    new_population = population - growth_rate * (resource / resource_limit)
    return max(new_population, 0)

# 初始条件
population = 100
resource = 1000
growth_rate = 0.05
resource_threshold = 500

# 运行100次迭代
for _ in range(100):
    resource -= population
    population = negative_feedback(population, resource, growth_rate, resource_threshold)
    print(f"Iteration { _:02d}: Population = {population:.2f}, Resource = {resource:.2f}")
```

在这个示例中，我们引入了一个资源阈值，使得种群数量不会超过资源限制，从而展示了负反馈机制如何维持系统稳定。

#### 自相似结构

自相似结构是自组织系统中的另一个重要特征。自相似性指的是系统在不同尺度上表现出相似的结构和功能。自相似结构在自然界中广泛存在，例如，海岸线、雪花和河流分支系统都表现出自相似性。自相似结构使得系统在不同尺度上具有相似的行为，从而展示了自组织现象的层次性。

空间和时间在自组织过程中也扮演着关键角色。空间维度决定了系统内部元素之间的相互作用方式，而时间维度则影响了系统演化过程的速度和稳定性。在不同的空间和时间尺度上，自组织现象表现出不同的特征。

### 自组织过程的数学模型

自组织过程的数学模型是理解自组织现象的重要工具。这些模型通过数学表达式和算法，描述了系统如何在没有外部指令的情况下，通过内部相互作用自发形成有序结构。在这一部分，我们将详细讨论几种常用的数学模型，包括非线性动力系统、相变模型和复杂系统的计算模拟。

#### 非线性动力系统

非线性动力系统是研究动态系统中非线性相互作用的重要模型。在非线性动力系统中，系统的行为往往取决于初始条件和系统参数，因此表现出高度复杂的行为。一个经典的非线性动力系统模型是洛伦兹系统（Lorenz system），由气象学家洛伦兹（Ed Lorenz）在研究大气动力学时提出。以下是一个简化的洛伦兹系统方程：

$$\dot{x} = \sigma (y - x)$$
$$\dot{y} = x (\rho - z) - y$$
$$\dot{z} = xy - \beta z$$

其中，\(x\)、\(y\) 和 \(z\) 是系统的状态变量，\(\sigma\)、\(\rho\) 和 \(\beta\) 是系统参数。这个系统在参数空间中具有一个奇怪吸引子，展示了非线性动力系统如何通过复杂的行为形成有序结构。

以下是一个使用Python和matplotlib库来模拟洛伦兹系统的示例代码：

```python
import numpy as np
import matplotlib.pyplot as plt

# 参数设置
sigma = 10.0
rho = 28.0
beta = 8.0/3.0
dt = 0.01
t_max = 100

# 初始条件
x, y, z = 1.0, 1.0, 1.0

# 洛伦兹系统方程
def lorenz_system(x, y, z, sigma, rho, beta, dt):
    dxdt = sigma * (y - x)
    dydt = x * (rho - z) - y
    dzdt = x * y - beta * z
    return np.array([dxdt, dydt, dzdt])

# 模拟
t = np.arange(0, t_max, dt)
X, Y, Z = np.zeros((t_max,)), np.zeros((t_max,)), np.zeros((t_max,))

for i in range(1, t_max):
    dX, dY, dZ = lorenz_system(X[i-1], Y[i-1], Z[i-1], sigma, rho, beta, dt)
    X[i] = X[i-1] + dX * dt
    Y[i] = Y[i-1] + dY * dt
    Z[i] = Z[i-1] + dZ * dt

# 绘图
fig = plt.figure()
ax = fig.add_subplot(111, projection='3d')
ax.plot(X, Y, Z, lw=0.5)
ax.set_xlabel('X Axis')
ax.set_ylabel('Y Axis')
ax.set_zlabel('Z Axis')
plt.show()
```

这段代码模拟了洛伦兹系统的动态行为，通过绘制三维相空间中的轨迹，展示了非线性动力系统如何形成复杂有序结构。

#### 相变模型

相变模型描述了系统从一种相态转变为另一种相态的过程，例如从固态到液态的转变。在相变过程中，系统表现出特殊的临界行为，这些行为在临界点附近尤为显著。一个经典的相变模型是Ising模型，它用于描述磁性材料的磁化过程。以下是一个简化的Ising模型方程：

$$H = -J \sum_{<ij>} s_i s_j - h \sum_{i} s_i$$

其中，\(s_i\) 表示第 \(i\) 个磁性原子的磁化方向，\(J\) 是相邻原子之间的耦合强度，\(h\) 是外部磁场强度。当系统温度降低到临界温度以下时，系统会经历从无序到有序的相变。

以下是一个使用Python和NumPy库来模拟Ising模型的示例代码：

```python
import numpy as np
import matplotlib.pyplot as plt

# 参数设置
J = 1.0
h = 1.0
L = 100
T_c = 2.2674 / J

# 初始状态
 spins = 2 * np.random.randint(0, 2, size=(L, L)) - 1

# 能量函数
def energy(spins, J, h):
    return -J * np.sum([spins[i][j] * spins[i][k] for i in range(L) for j in range(L) if i != k]) - h * np.sum(spins)

# 更新规则
def update_spins(spins, J, h, T):
    for i in range(L):
        for j in range(L):
            dE = 2 * (h * spins[i][j] - J * np.sum([spins[i][k] * spins[i][l] for k in range(L) for l in range(L) if (k != i and l != j)]))
            if dE < 0 or np.random.rand() < np.exp(-dE / T):
                spins[i][j] *= -1

# 模拟
T = 1.5 * T_c
for i in range(1000):
    E = energy(spins, J, h)
    update_spins(spins, J, h, T)
    if i % 100 == 0:
        plt.imshow(spins, cmap='gray')
        plt.show()

# 绘图
plt.imshow(spins, cmap='gray')
plt.show()
```

这段代码模拟了Ising模型在温度 \(T\) 下降到临界温度以下时，从无序状态向有序状态的相变过程，通过绘制二维网格中的磁化方向，展示了相变模型如何描述自组织现象。
