import re
import math
from collections import Counter
import jieba

# 去除标点符号
def remove_punctuation(text):
    return re.sub(r'[^\w\s]', '', text)

# 停用词过滤
def filter_stop_words(text, stop_words):
    words = text.split()
    return [word for word in words if word not in stop_words]

# 计算概率分布
def calculate_probabilities(words):
    total = len(words)
    probabilities = {word: (words.count(word) / total) for word in set(words)}
    return probabilities

# 计算信息熵
def calculate_entropy(probabilities):
    entropy = -sum(p * math.log2(p) for p in probabilities.values())
    return entropy

# 计算幽默感
def calculate_humor(original_text, modified_text, stop_words):
    # 文本预处理
    original_text = remove_punctuation(original_text)
    modified_text = remove_punctuation(modified_text)

    # 去除停用词
    original_words = filter_stop_words(original_text, stop_words)
    modified_words = filter_stop_words(modified_text, stop_words)

    # 计算概率分布
    original_probabilities = calculate_probabilities(original_words)
    modified_probabilities = calculate_probabilities(modified_words)

    # 计算信息熵
    original_entropy = calculate_entropy(original_probabilities)
    modified_entropy = calculate_entropy(modified_probabilities)

    # 计算幽默感
    humor = modified_entropy - original_entropy
    return humor

# 示例

modified_text = "我今天早上吃了一个红苹果。"
stop_words = ["我", "今天", "上", "一", "个"]

import re
import nltk
from jieba import posseg
from sklearn.feature_extraction.text import CountVectorizer
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error, r2_score

# 去除标点符号
def remove_punctuation(text):
    return re.sub(r'[^\w\s]', '', text)

# 停用词过滤
def filter_stop_words(text, stop_words):
    words = text.split()
    return [word for word in words if word not in stop_words]

# 计算概率分布
def calculate_probabilities(words):
    total = len(words)
    probabilities = {word: (words.count(word) / total) for word in set(words)}
    return probabilities

# 计算信息熵
def calculate_entropy(probabilities):
    entropy = -sum(p * math.log2(p) for p in probabilities.values())
    return entropy

# 计算幽默感
def calculate_humor(original_text, modified_text, stop_words):
    # 文本预处理
    original_text = remove_punctuation(original_text)
    modified_text = remove_punctuation(modified_text)

    # 去除停用词
    original_words = filter_stop_words(original_text, stop_words)
    modified_words = filter_stop_words(modified_text, stop_words)

    # 计算概率分布
    original_probabilities = calculate_probabilities(original_words)
    modified_probabilities = calculate_probabilities(modified_words)

    # 计算信息熵
    original_entropy = calculate_entropy(original_probabilities)
    modified_entropy = calculate_entropy(modified_probabilities)

    # 计算幽默感
    humor = modified_entropy - original_entropy
    return humor

# 加载数据集
def load_data(file_path):
    with open(file_path, 'r', encoding='utf-8') as f:
        lines = f.readlines()
    texts = [line.strip() for line in lines]
    return texts

# 主函数
def main():
    # 加载数据集
    data_path = 'humor_data.txt'
    texts = load_data(data_path)

    # 停用词列表
    stop_words = set(nltk.corpus.stopwords.words('chinese'))

    # 训练数据与测试数据划分
    train_texts, test_texts = train_test_split(texts, test_size=0.2, random_state=42)

    # 特征提取
    vectorizer = CountVectorizer()
    X_train = vectorizer.fit_transform(train_texts)
    X_test = vectorizer.transform(test_texts)

    # 模型训练
    model = LinearRegression()
    model.fit(X_train, train_texts)

    # 模型评估
    y_pred = model.predict(X_test)
    mse = mean_squared_error(y_test, y_pred)
    r2 = r2_score(y_test, y_pred)
    print(f"MSE: {mse}, R^2: {r2}")

    # 预测应用
    new_text = "我今天早上吃了一个苹果。"
    new_text = remove_punctuation(new_text)
    new_words = filter_stop_words(new_text, stop_words)
    probabilities = calculate_probabilities(new_words)
    entropy = calculate_entropy(probabilities)
    print(f"新文本信息熵：{entropy}")

if __name__ == '__main__':
    main()
