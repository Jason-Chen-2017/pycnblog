#### 3. 编写源代码

在项目文件夹中创建一个名为 "watermarking.py" 的 Python 文件，并在其中编写以下代码：

```python
import cv2
import numpy as np
from skimage import transform

def PreprocessImage(image_path):
    # 读取图片
    image = cv2.imread(image_path, cv2.IMREAD_GRAYSCALE)
    
    # 对图片进行归一化处理
    image = image / 255.0
    
    return image

def EmbedWatermark(image, watermark, alpha):
    # 对水印进行归一化处理
    watermark = watermark / 255.0
    
    # 将水印缩放到与图像相同的大小
    watermark = transform.resize(watermark, image.shape[::-1], interpolation=transform.INTER_CUBIC)
    
    # 将水印嵌入到图像中
    watermarked_image = image * (1 - alpha) + watermark * alpha
    
    return watermarked_image

def ExtractWatermark(image, watermark, alpha):
    # 对水印进行归一化处理
    watermark = watermark / 255.0
    
    # 将水印缩放到与图像相同的大小
    watermark = transform.resize(watermark, image.shape[::-1], interpolation=transform.INTER_CUBIC)
    
    # 从图像中提取水印
    extracted_watermark = image * (1 - alpha) - watermark * alpha
    
    return extracted_watermark

if __name__ == "__main__":
    # 加载原始图像和水印
    image_path = "images/original.jpg"
    watermark_path = "images/watermark.png"
    image = PreprocessImage(image_path)
    watermark = cv2.imread(watermark_path, cv2.IMREAD_GRAYSCALE)
    
    # 设置嵌入系数
    alpha = 0.5
    
    # 嵌入水印
    watermarked_image = EmbedWatermark(image, watermark, alpha)
    
    # 保存嵌入水印后的图像
    cv2.imwrite("results/watermarked_image.jpg", watermarked_image * 255)
    
    # 提取水印
    extracted_watermark = ExtractWatermark(watermarked_image, watermark, alpha)
    
    # 保存提取的水印
    cv2.imwrite("results/extracted_watermark.jpg", extracted_watermark * 255)
```
