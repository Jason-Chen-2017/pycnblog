```javascript
const dbName = 'attentionTraining';

MongoClient.connect(url, { useUnifiedTopology: true }, (err, client) => {
  if (err) throw err;
  console.log('Connected to MongoDB');
  const db = client.db(dbName);
  app.use((req, res, next) => {
    req.db = db;
    next();
  });

  // 创建训练记录
  app.post('/train', (req, res) => {
    const data = req.body;
    db.collection('trainRecords').insertOne(data, (err, result) => {
      if (err) throw err;
      res.status(201).send('Training record created');
    });
  });

  // 获取训练记录
  app.get('/train', (req, res) => {
    db.collection('trainRecords').find({}).toArray((err, docs) => {
      if (err) throw err;
      res.status(200).send(docs);
    });
  });

  app.listen(3000, () => {
    console.log('Server running on port 3000');
  });
});
```

上述代码实现了两个主要功能：

1. **脑电图技术**：脑电图技术通过贴在头皮上的电极，捕捉大脑产生的脑电波信号。这些信号经过放大、滤波和数字化处理，最终转化为可以分析的电信号。

2. **脑电波分析**：脑电波分析包括频率分析（如α波、β波、θ波、δ波）和振幅分析。通过分析脑电波的特征，可以了解大脑在不同认知任务中的活动状态。

3. **反馈调节**：一些注意力训练硬件设备还具备反馈调节功能，即通过声、光、电等刺激，实时调整用户的注意力水平，帮助用户保持专注。
