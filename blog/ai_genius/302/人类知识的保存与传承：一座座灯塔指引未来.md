function classifyKnowledge(knowledgeList) {
  // 创建分类字典
  classificationDict = {}

  // 遍历知识列表，根据知识类型分类
  for each knowledge in knowledgeList {
    if knowledge not in classificationDict {
      classificationDict[knowledge] = [knowledge]
    } else {
      classificationDict[knowledge].append(knowledge)
    }
  }

  // 返回分类字典
  return classificationDict
}

```python
import sqlite3
import json

# 数据库连接
conn = sqlite3.connect('library.db')
cursor = conn.cursor()

# 创建数据库表
cursor.execute('''CREATE TABLE IF NOT EXISTS books (id INTEGER PRIMARY KEY, title TEXT, author TEXT, publication_date TEXT, digital_link TEXT)''')

# 插入书籍数据
book_data = [
  ('001', 'The Great Gatsby', 'F. Scott Fitzgerald', '1925-04-10', 'https://example.com/gatsby.pdf'),
  ('002', 'To Kill a Mockingbird', 'Harper Lee', '1960-07-11', 'https://example.com/mockingbird.pdf'),
  # 更多书籍数据...
]

cursor.executemany('INSERT INTO books (id, title, author, publication_date, digital_link) VALUES (?, ?, ?, ?, ?)', book_data)

# 提交事务
conn.commit()

# 查询书籍
def search_books(title):
  cursor.execute('SELECT * FROM books WHERE title LIKE ?', ('%' + title + '%',))
  books = cursor.fetchall()
  return books

# 示例：查询《The Great Gatsby》
books = search_books('The Great Gatsby')
for book in books:
  print(json.dumps(book, indent=2))

# 关闭数据库连接
conn.close()
```
