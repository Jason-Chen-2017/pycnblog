# 导入相关库
import numpy as np

# 定义层次分析模型
class AnalyticHierarchyProcess:
    def __init__(self, criteria):
        self.criteria = criteria
        self.weights = None

    def calculate_weights(self):
        # 计算各准则的权重
        criteria_matrix = np.eye(len(self.criteria))
        for i in range(len(self.criteria)):
            for j in range(len(self.criteria)):
                if i != j:
                    criteria_matrix[i][j] = float(input(f"请输入 {self.criteria[i]} 相对于 {self.criteria[j]} 的权重（1-9标度法）："))
        self.weights = np.linalg.inv(criteria_matrix).dot(np.ones(len(criteria_matrix)))
        self.weights = self.weights / np.sum(self.weights)

    def get_weights(self):
        if self.weights is not None:
            return self.weights
        else:
            return "请先计算权重。"

# 示例
criteria = ['仁', '义', '礼', '智', '信']
ahp = AnalyticHierarchyProcess(criteria)
ahp.calculate_weights()
print(ahp.get_weights())
