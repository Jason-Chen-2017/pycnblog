        return -1;

        return -1;

        render();
        glfwPollEvents();

    // 退出

    glfwTerminate();
    return 0;

    // 绘制三角形
    glBegin(GL_TRIANGLES);

int main() {
    // 初始化GLFW
    if (!glfwInit()) {
        return -1;
    }

    // 创建窗口
    GLFWwindow* window = glfwCreateWindow(800, 600, "Render Example", NULL, NULL);
    if (!window) {
        glfwTerminate();
        return -1;

    // 初始化GLAD
    if (!gladLoadGLLoader((GLADloadproc)glfwGetProcAddress)) {
        return -1;

    // 渲染循环
    while (!glfwWindowShouldClose(window)) {
        render();
        glfwPollEvents();
    }

    // 退出
    glfwDestroyWindow(window);
    glfwTerminate();
    return 0;
}
