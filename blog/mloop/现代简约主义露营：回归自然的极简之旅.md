    # 根据温度、地形和持续时间评估所需装备

        equipment_needed = ["warm tent", "warm sleeping bag", "windproof clothing"]
    elif terrain == "mountain":
        equipment_needed = ["mountain tent", "warm sleeping bag", "snow shoes"]
    else:
        equipment_needed = ["standard tent", "sleeping bag", "sandals"]

def calculate_costs(equipment_list, prices):
    # 计算装备总成本
    total_cost = 0

    total_cost = calculate_costs(equipment_needed, prices)
    return equipment_needed, total_cost
