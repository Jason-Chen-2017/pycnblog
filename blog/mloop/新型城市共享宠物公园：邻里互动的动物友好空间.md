### 3.3 Python源代码实现

以下是一个简单的空间优化算法的Python源代码实现示例：

```python
import numpy as np

def space_optimization(areas, requirements):
    """
    空间优化算法实现
    :param areas: 区域列表，每个区域为一个字典，包含名称、面积等信息
    :param requirements: 需求列表，每个需求为一个字典，包含名称、面积要求等信息
    :return: 优化后的区域分配结果
    """
    # 初始化分配结果
    allocation = {requirement['name']: None for requirement in requirements}

    # 计算各区域的可用面积
    available_areas = {area['name']: area['area'] for area in areas}

    # 按需求优先级进行分配
    for requirement in sorted(requirements, key=lambda x: x['priority']):
        for area in areas:
            if available_areas[area['name']] >= requirement['size']:
                allocation[requirement['name']] = area['name']
                available_areas[area['name']] -= requirement['size']
                break

    return allocation

# 示例数据
areas = [
    {'name': '宠物活动区', 'area': 500},
    {'name': '社区活动区', 'area': 300},
    {'name': '宠物医疗区', 'area': 200}
]

requirements = [
    {'name': '宠物活动需求', 'size': 400, 'priority': 1},
    {'name': '社区活动需求', 'size': 200, 'priority': 2},
    {'name': '宠物医疗需求', 'size': 150, 'priority': 3}
]

# 执行空间优化算法
allocation_result = space_optimization(areas, requirements)

# 输出结果
print(allocation_result)
```
