def connect_db():
    db = pymysql.connect(
        host="localhost",
        user="root",
        password="password",
        database="parenting_resource_center",
        charset="utf8"
    )

@app.route('/api/resource/reserve', methods=['POST'])
def reserve_resource():
    db = connect_db()
    cursor = db.cursor()

    resource_id = request.form['resource_id']
    reserve_date = request.form['reserve_date']

    sql = "INSERT INTO reservation (resource_id, reserve_date) VALUES (%s, %s)"
    try:
        cursor.execute(sql, (resource_id, reserve_date))
        db.commit()
        return jsonify({"status": "success", "message": "资源预约成功"})
    except Exception as e:
        db.rollback()
        return jsonify({"status": "error", "message": str(e)})

if __name__ == '__main__':
    app.run(debug=True)

def connect_db():

        host="localhost",
        user="root",

@app.route('/api/resource/reserve', methods=['POST'])
def reserve_resource():
    db = connect_db()
    cursor = db.cursor()

    resource_id = request.form['resource_id']
    reserve_date = request.form['reserve_date']

    sql = "INSERT INTO reservation (resource_id, reserve_date) VALUES (%s, %s)"
    try:
        cursor.execute(sql, (resource_id, reserve_date))
        db.commit()
        return jsonify({"status": "success", "message": "资源预约成功"})
    except Exception as e:
        db.rollback()
        return jsonify({"status": "error", "message": str(e)})

if __name__ == '__main__':
