        return f"计划：{self.计划}, 进度：{self.进度}, 完成：{self.完成}"

def main():
    # 创建一个小组，包含3名成员
    members = ["成员A", "成员B", "成员C"]
    group = Group(members)

    # 不断循环，模拟小组工作过程
    while not group.完成:
        需求 = "需求"
        if group.评估需求(需求):
            group.制定计划(需求)
            group.执行计划()
            if group.监控进度():
                group.评估结果()
                group.反馈与改进(group.评估结果())
            else:

        else:
            print("需求评估未通过，调整需求...")

    # 输出小组最终状态
    print(group.状态())

if __name__ == "__main__":
    main()
