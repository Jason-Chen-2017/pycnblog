### 3.2 Python源代码

```python
import random

def preprocess_prompt(prompt):
    # 对提示词进行预处理，去除前后空格并转为小写
    return prompt.strip().lower()

def is_valid_prompt(prompt):
    # 判断提示词是否有效，要求提示词长度至少为1且不为空
    return len(prompt) > 0

def generate_creative_candidates(prompt):
    # 假设使用一个简单的随机生成器生成创意候选集
    candidates = []
    for _ in range(5):
        candidate = f"创意候选 {_}: {random.choice(['广告', '设计', '内容', '产品'])}"
        candidates.append(candidate)
    return candidates

def evaluate_creatives(candidates):
    # 简单评估创意质量，返回得分最高的创意
    scores = [random.uniform(0.5, 1) for _ in range(len(candidates))]
    best_index = scores.index(max(scores))
    return candidates[best_index]

def main():
    prompt = input("请输入提示词：")
    processed_prompt = preprocess_prompt(prompt)
    if is_valid_prompt(processed_prompt):
        candidates = generate_creative_candidates(processed_prompt)
        best_creative = evaluate_creatives(candidates)
        print(f"最佳创意：{best_creative}")
    else:
        print("提示词无效，请重新输入。")

if __name__ == "__main__":
    main()
```
