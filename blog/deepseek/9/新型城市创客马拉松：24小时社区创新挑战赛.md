import numpy as np

def linear_regression(X, y, alpha, num_iterations):
    m = len(y)
    theta = np.zeros(X.shape[1])
    X = np.hstack([np.ones((m, 1)), X])  # 添加偏置项

    for i in range(num_iterations):
        h = X.dot(theta)
        loss = -(1 / m) * (y.dot(np.log(h)) + (1 - y).dot(np.log(1 - h)))
        theta = theta - (alpha / m) * X.T.dot(h - y)

    return theta

X = np.array([[1, 2], [2, 3], [3, 4]])
y = np.array([1, 0, 1])
theta = linear_regression(X, y, 0.01, 1000)

print(theta)

**数学公式**：

线性回归的损失函数为：

$$ \text{Loss} = -\frac{1}{m} \sum_{i=1}^{m} (y_i \cdot \log(h_i) + (1 - y_i) \cdot \log(1 - h_i)) $$

其中，$h_i = \sigma(\theta^T \cdot x_i)$，$\sigma(z) = \frac{1}{1 + e^{-z}}$。
