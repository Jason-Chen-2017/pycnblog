### 3.2 Python源代码

```python
import numpy as np
import tensorflow as tf

class GAN:
    def __init__(self, input_dim, hidden_dim, output_dim):
        self.input_dim = input_dim
        self.hidden_dim = hidden_dim
        self.output_dim = output_dim
        self.gan_model = self.build_model()

    def build_model(self):
        # 定义生成器和判别器
        generator = self.build_generator()
        discriminator = self.build_discriminator()
        return tf.keras.Model(inputs=[generator.input, discriminator.input], outputs=[generator.output, discriminator.output])

    def build_generator(self):
        model = tf.keras.Sequential()
        model.add(tf.keras.layers.Dense(self.hidden_dim, activation='relu', input_dim=self.input_dim))
        model.add(tf.keras.layers.Dense(self.output_dim, activation='sigmoid'))
        return model

    def build_discriminator(self):
        model = tf.keras.Sequential()
        model.add(tf.keras.layers.Dense(self.hidden_dim, activation='relu', input_dim=self.output_dim))
        model.add(tf.keras.layers.Dense(1, activation='sigmoid'))
        return model

    def train(self, X_train, y_train, epochs=100):
        self.gan_model.compile(loss='binary_crossentropy', optimizer='adam')
        for epoch in range(epochs):
            # 训练生成器和判别器
            noise = np.random.randn(len(X_train), self.input_dim)
            gen = self.gan_model.layers[0].predict(noise)
            d_loss = self.gan_model.layers[1].train_on_batch(gen, y_train)
            g_loss = self.gan_model.train_on_batch([noise, gen], np.zeros(len(gen)))
            print(f'Epoch {epoch}, D loss: {d_loss}, G loss: {g_loss}')
```

### 3.3 数学模型和公式

生成对抗网络（GAN）的目标是最小化生成器和判别器的损失函数：

$$ L_{D} = -\mathbb{E}_{x \sim p_{\text{data}}}[\log D(x)] - \mathbb{E}_{z \sim p_{z}}[\log (1 - D(G(z)))] $$
$$ L_{G} = -\mathbb{E}_{z \sim p_{z}}[\log D(G(z))] $$

其中，$D$ 是判别器，$G$ 是生成器，$z$ 是噪声向量。
