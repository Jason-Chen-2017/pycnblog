```python
from django.db import models

class User(models.Model):
    username = models.CharField(max_length=100)
    email = models.EmailField(unique=True)
    password = models.CharField(max_length=100)

class Schedule(models.Model):
    user = models.ForeignKey(User, on_delete=models.CASCADE)
    activity = models.CharField(max_length=100)
    start_time = models.DateTimeField()
    end_time = models.DateTimeField()

class Journal(models.Model):
    user = models.ForeignKey(User, on_delete=models.CASCADE)
    entry = models.TextField()
    created_at = models.DateTimeField(auto_now_add=True)
```

```python
        # 这里需要实现用户注册逻辑

        return render(request, 'register.html', {'message': '注册成功'})

def schedule(request):
    user = request.user
    schedules = Schedule.objects.filter(user=user)
    return render(request, 'schedule.html', {'schedules': schedules})

def journal(request):
    user = request.user
    journals = Journal.objects.filter(user=user)
    return render(request, 'journal.html', {'journals': journals})

def add_schedule(request):
    if request.method == 'POST':
        user = request.user
        activity = request.POST['activity']
        start_time = request.POST['start_time']
        end_time = request.POST['end_time']
        schedule = Schedule(user=user, activity=activity, start_time=start_time, end_time=end_time)
        schedule.save()
        return render(request, 'schedule.html', {'message': '日程安排添加成功'})
    return render(request, 'add_schedule.html')

def add_journal(request):
    if request.method == 'POST':
        user = request.user
        entry = request.POST['entry']
        journal = Journal(user=user, entry=entry)
        journal.save()
        return render(request, 'journal.html', {'message': '反思记录添加成功'})
    return render(request, 'add_journal.html')
```
