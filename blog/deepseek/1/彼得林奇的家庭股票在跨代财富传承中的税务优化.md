  ```python
  import numpy as np
  import pandas as pd

  def calculate_tax(family_wealth, tax_rates):
      inheritance_tax = family_wealth * tax_rates['inheritance_tax_rate']
      gift_tax = family_wealth * tax_rates['gift_tax_rate']
      return inheritance_tax + gift_tax

  def optimize_shares(family_wealth, tax_rates):
      optimized_shares = family_wealth * (1 - tax_rates['inheritance_tax_rate'])
      return optimized_shares

  def plan_succession(family_wealth, tax_rates):
      optimized_shares = optimize_shares(family_wealth, tax_rates)
      return {
          'optimized_shares': optimized_shares,
          'tax_saving': calculate_tax(family_wealth, tax_rates) - calculate_tax(optimized_shares, tax_rates)
      }
  ```
