import matplotlib.pyplot as plt

num_agents = 100
num_iterations = 100
investment_portfolios = []

# 生成智能体
agents = np.random.rand(num_agents, 2)  # 每个智能体由两个参数表示

# 智能体互动
for _ in range(num_iterations):
    # 计算每个智能体的投资组合

```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# 数据采集与预处理
def data_preprocessing(data_path):
    # 读取数据
    data = pd.read_csv(data_path)
    
    # 数据清洗
    data.dropna(inplace=True)
    
    # 数据转换
    data['date'] = pd.to_datetime(data['date'])
    data.set_index('date', inplace=True)
    
    # 数据整合
    data = data.resample('M').mean()
    
    return data

# 多智能体AI模型构建
def build_agent_model(data, num_agents, num_iterations):
    # 初始化智能体
    agents = np.random.rand(num_agents, data.shape[1])
    
    # 智能体互动
    for _ in range(num_iterations):
        # 计算每个智能体的投资组合
        portfolios = np.apply_along_axis(lambda x: np.mean(x), 1, agents)
        
        # 更新智能体
        agents = agents + np.random.randn(num_agents, data.shape[1])
    
    return agents, portfolios

# 投资决策支持
def investment_decision(portfolios):
    # 计算投资组合的收益率
    returns = np.diff(portfolios, axis=0)
    
    # 选择收益率最高的投资组合
    best_portfolio = portfolios[returns.argmax()]
    
    return best_portfolio

# 投资组合管理
def manage_portfolio(data, best_portfolio):
    # 计算投资组合的收益率
    portfolio_returns = np.diff(best_portfolio, axis=0)
    
    # 绘制收益率曲线
    plt.plot(data.index, portfolio_returns)
    plt.xlabel('Date')
    plt.ylabel('Return')
    plt.title('Portfolio Return')
    plt.show()

# 主函数
def main():
    # 加载数据
    data_path = 'stock_data.csv'
    data = data_preprocessing(data_path)
    
    # 构建智能体模型
    num_agents = 100
    num_iterations = 100
    agents, portfolios = build_agent_model(data, num_agents, num_iterations)
    
    # 投资决策
    best_portfolio = investment_decision(portfolios)
    
    # 投资组合管理
    manage_portfolio(data, best_portfolio)

# 运行主函数
if __name__ == '__main__':
    main()
```
