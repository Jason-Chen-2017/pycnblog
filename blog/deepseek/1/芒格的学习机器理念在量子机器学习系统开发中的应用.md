def quantum_feature_extraction(data):
    # 初始化量子电路
    n = len(data)
    qc = QuantumCircuit(n)
    # 应用量子叠加
    for i in range(n):
        qc.h(i)
    qc.measure_all()
    # 执行量子计算
    result = execute(qc, Aer.get_backend('qasm_simulator')).result()
    # 获取结果
    counts = result.get_counts()
    return counts
