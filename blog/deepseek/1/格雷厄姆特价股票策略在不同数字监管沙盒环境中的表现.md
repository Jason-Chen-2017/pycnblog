```python
import pandas as pd

def graham_strategy(stock_data):
    # 计算市盈率
    stock_data['pe'] = stock_data['price'] / stock_data['earnings']
    # 计算市净率
    stock_data['pb'] = stock_data['price'] / stock_data['book_value']
    # 计算股息率
    stock_data['dividend_yield'] = stock_data['dividend'] / stock_data['price']
    # 筛选低市盈率股票
    low_pe_stocks = stock_data[stock_data['pe'] < stock_data['pe'].mean()]
    # 筛选低市净率股票
    low_pb_stocks = stock_data[stock_data['pb'] < stock_data['pb'].mean()]
    # 筛选高股息率股票
    high_dividend_stocks = stock_data[stock_data['dividend_yield'] > stock_data['dividend_yield'].mean()]
    return low_pe_stocks, low_pb_stocks, high_dividend_stocks

# 示例数据
data = {
    'price': [100, 200, 300, 400, 500],
    'earnings': [20, 30, 40, 50, 60],
    'book_value': [50, 60, 70, 80, 90],
    'dividend': [5, 10, 15, 20, 25]
}
stock_data = pd.DataFrame(data)
low_pe_stocks, low_pb_stocks, high_dividend_stocks = graham_strategy(stock_data)
print("低市盈率股票:", low_pe_stocks)
print("低市净率股票:", low_pb_stocks)
print("高股息率股票:", high_dividend_stocks)
```
