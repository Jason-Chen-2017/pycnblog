problem = pulp.LpProblem("TaxOptimization", pulp.LpMinimize)

# 定义变量
x = pulp.LpVariable.dicts("x", range(1, 11), cat="Continuous")

# 定义目标函数
problem += pulp.lpSum([x[i] * tax_rate[i] for i in range(1, 11)])

# 定义约束条件

    problem += x[i] <= budget[i]

# 解决问题
problem.solve()

# 输出结果

print("最小化税收：", pulp.value(problem.objective))
