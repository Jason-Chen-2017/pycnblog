import numpy as np
from scipy.cluster.hierarchy import linkage, fcluster

# 示例代码：层次分析法（AHP）计算指标权重
def calculate_weights(matrix):
    # 计算一致性系数
    n = matrix.shape[0]
    consistency = 0
    for i in range(n):
        row = matrix[i]
        total = sum(row)
        consistency += total / (n-1)
    consistency /= n
    # 计算特征向量
    eigenvectors = np.linalg.eig(matrix)
    weights = eigenvectors[0] / sum(eigenvectors[0])
    return weights

# 示例代码：综合得分计算
def calculate_score(weights, data):
    score = np.dot(weights, data)
    return score
