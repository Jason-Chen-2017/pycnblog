### 3.1 多尺度分析算法

多尺度分析算法是金融时间序列多尺度分析平台的核心。它通过对时间序列数据进行不同时间尺度的分析，可以揭示金融市场的波动特征和内在规律。常见的多尺度分析算法包括：

波动性度量是衡量时间序列数据波动程度的重要方法。它可以帮助我们识别市场的波动性，为风险管理提供依据。常见的波动性度量方法包括：

- **标准差**：标准差是衡量时间序列数据波动程度的一种常用方法。它表示数据偏离均值的大小。
  \[ \sigma = \sqrt{\frac{1}{N-1} \sum_{i=1}^{N} (x_i - \mu)^2} \]
  其中，\( \sigma \) 是标准差，\( N \) 是数据点数量，\( x_i \) 是第 \( i \) 个数据点，\( \mu \) 是均值。

- **方差**：方差是标准差的平方，表示数据的离散程度。
  \[ \sigma^2 = \frac{1}{N-1} \sum_{i=1}^{N} (x_i - \mu)^2 \]

#### 3.1.2 周期性分析

周期性分析是识别时间序列数据中周期性特征的一种方法。它可以帮助我们理解市场的周期性波动，为投资策略提供指导。常见的周期性分析方法包括：

- **傅里叶变换**：傅里叶变换可以将时间序列数据从时域转换为频域，揭示数据的周期性特征。
  \[ X(f) = \int_{-\infty}^{\infty} x(t) e^{-j2\pi ft} dt \]
  其中，\( X(f) \) 是频域信号，\( x(t) \) 是时域信号，\( f \) 是频率。

- **小波变换**：小波变换是一种多尺度分析工具，可以在不同时间尺度上分析时间序列数据。
  \[ W(f, t) = \frac{1}{\sqrt{2\pi}} \int_{-\infty}^{\infty} x(t) \overline{\psi}(t - \tau) e^{-j2\pi f\tau} d\tau \]

#### 3.1.3 分形分析

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from sklearn.preprocessing import MinMaxScaler
from sklearn.ensemble import IsolationForest
from mermaid import Mermaid

# 6.2.1 数据预处理模块

def preprocess_data(data):
    """
    数据预处理模块，包括噪声过滤、异常值处理和时序平滑。
    """
    # 噪声过滤：使用移动平均法
    window_size = 3
    data['filtered'] = data['close'].rolling(window=window_size, center=True).mean()

    # 异常值处理：使用孤立森林法
    model = IsolationForest(contamination=0.01)
    outliers = model.fit_predict(data[['filtered']])
    data['outliers'] = outliers
    data = data[data['outliers'] == 1]

    # 时序平滑：使用指数平滑法
    alpha = 0.2
    data['smoothed'] = alpha * data['filtered'] + (1 - alpha) * data['smoothed'].shift(1)
    data['smoothed'].iloc[0] = data['filtered'].iloc[0]

    return data

# 6.2.2 多尺度分析模块

def multi_scale_analysis(data):
    """
    多尺度分析模块，包括波动性度量、周期性分析和分形分析。
    """
    # 波动性度量：计算标准差
    volatility = data['smoothed'].std()

    # 周期性分析：使用傅里叶变换
    data['fft'] = np.fft.fft(data['smoothed'])
    freq = np.fft.fftfreq(len(data['smoothed']))
    plt.figure()
    plt.plot(freq, np.abs(data['fft']))
    plt.title('Fourier Transform')
    plt.xlabel('Frequency')
    plt.ylabel('Magnitude')
    plt.show()

    # 分形分析：使用盒子计数法
    box_size = 2
    count = np.zeros((len(data['smoothed']), box_size))
    for i in range(box_size, len(data['smoothed']) - box_size + 1):
        count[i - box_size] = np.count_nonzero(np.abs(data['smoothed'][i - box_size:i + box_size + 1]) > 0.1)
    fractal_dimension = np.mean(np.log2(count))

    return volatility, fractal_dimension

# 6.2.3 结果展示模块

def show_results(data, volatility, fractal_dimension):
    """
    结果展示模块，展示分析结果。
    """
    # 展示波动性曲线
    plt.figure()
    plt.plot(data['smoothed'])
    plt.title('Volatility')
    plt.xlabel('Time')
    plt.ylabel('Price')
    plt.show()

    # 展示分形维数
    print(f'Fractal Dimension: {fractal_dimension:.2f}')

# 6.2.4 主函数

def main():
    # 加载数据
    data = pd.read_csv('stock_price.csv')
    data = data[['date', 'close']]

    # 数据预处理

    volatility, fractal_dimension = multi_scale_analysis(data)

    # 结果展示
    show_results(data, volatility, fractal_dimension)

if __name__ == '__main__':
