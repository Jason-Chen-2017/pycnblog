   ```python
   dataset = load_dataset("squad", "en", split="train")
   def preprocess(example):
       question = tokenizer.encode_plus(example["question"], add_special_tokens=True, max_length=40, pad_to_max_length=True, return_tensors="pt")
       context = tokenizer.encode_plus(example["context"], add_special_tokens=True, max_length=150, pad_to_max_length=True, return_tensors="pt")
       return {"question": question, "context": context}
   dataset = dataset.map(preprocess, batched=True)
   ```
