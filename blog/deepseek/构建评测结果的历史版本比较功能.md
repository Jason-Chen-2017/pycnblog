```python
# 导入依赖库
import mysql.connector
import pymongo
from datetime import datetime

# MySQL数据库配置
mysql_config = {
    'user': 'root',
    'password': 'password',
    'host': 'localhost',
    'database': 'evaluation'
}

# MongoDB数据库配置
mongo_config = {
    'username': 'admin',
    'password': 'password',
    'host': 'localhost',
    'database': 'evaluation'
}

# 连接MySQL数据库
mysql_connection = mysql.connector.connect(**mysql_config)
mysql_cursor = mysql_connection.cursor()

# 连接MongoDB数据库
mongo_client = pymongo.MongoClient(**mongo_config)
mongo_database = mongo_client['evaluation']

# 查询用户数据
def query_user(username):
    mysql_cursor.execute("SELECT * FROM user WHERE username = %s", (username,))
    user = mysql_cursor.fetchone()
    return user

# 查询评测结果数据
def query_evaluation_results(user_id, start_time, end_time):
    mysql_cursor.execute("SELECT * FROM evaluation_result WHERE user_id = %s AND timestamp BETWEEN %s AND %s", (user_id, start_time, end_time))
    results = mysql_cursor.fetchall()
    return results

# 比较评测结果
def compare_evaluation_results(results1, results2):
    result1_dict = {result['timestamp']: result['result'] for result in results1}
    result2_dict = {result['timestamp']: result['result'] for result in results2}
    differences = []
    for timestamp in set(result1_dict.keys()).union(result2_dict.keys()):
        if result1_dict.get(timestamp) != result2_dict.get(timestamp):
            differences.append(timestamp)
    return differences

# 测试数据查询与比较

print(user)
```
