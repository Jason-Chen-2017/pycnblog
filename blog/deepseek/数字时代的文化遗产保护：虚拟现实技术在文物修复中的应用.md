    kmeans.fit(point_cloud)
    labels = kmeans.labels_
    new_point_cloud = point_cloud[labels == 0]
    return new_point_cloud

# 纹理映射函数
def texture_mapping(point_cloud, texture_image):
    # 计算点云的法线向量
    tree = cKDTree(point_cloud)
    distances, indices = tree.query(point_cloud, k=2)
    normals = point_cloud[indices[:, 1]] - point_cloud[indices[:, 0]]
    normals = normals / np.linalg.norm(normals, axis=1)[:, np.newaxis]

    # 使用SVD分解纹理图像
    [U, S, V] = np.linalg.svd(texture_image)

    # 计算纹理映射
    texture_map = U[:point_cloud.shape[0], :] @ normals.T
    return texture_map
