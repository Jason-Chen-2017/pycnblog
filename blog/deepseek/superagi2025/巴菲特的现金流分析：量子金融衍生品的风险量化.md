```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

def calculate_cash_flow(cash_inflows, cash_outflows, discount_rate):
    net_cash_flows = [inflow - outflow for inflow, outflow in zip(cash_inflows, cash_outflows)]
    present_value = np.npv(discount_rate, net_cash_flows)
    return present_value

def main():
    cash_inflows = [100, 200, 300]
    cash_outflows = [50, 100, 70]
    discount_rate = 0.1
    pv = calculate_cash_flow(cash_inflows, cash_outflows, discount_rate)
    print(f"净现值为：{pv:.2f}")

if __name__ == "__main__":
    main()
```

```python
def calculate_risk(cash_flows, confidence_level):
    cash_flows_sorted = sorted(cash_flows)
    n = len(cash_flows_sorted)
    quantile = (n - 1) * confidence_level / 100
    risk_level = cash_flows_sorted[int(quantile)]
    return risk_level

def main():
    cash_flows = [100, 200, 300, 400, 500]
    confidence_level = 95
    risk = calculate_risk(cash_flows, confidence_level)
    print(f"风险水平为：{risk}")

if __name__ == "__main__":
    main()
```
