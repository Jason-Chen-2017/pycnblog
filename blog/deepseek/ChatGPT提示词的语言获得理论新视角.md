```python
import openai

def generate_text(prompt):
    # 初始化API客户端
    client = openai.Client()
    # 调用API生成文本
    response = client.chat.completions.create(
        model="gpt-3.5-turbo",
        messages=[{"role": "user", "content": prompt}]
    )
    return response.choices[0].message.content

# 示例
prompt = "写一篇关于人工智能的文章"
result = generate_text(prompt)
print(result)
```
