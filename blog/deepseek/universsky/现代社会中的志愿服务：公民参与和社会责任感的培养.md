   def register(request):
       if request.method == 'POST':
           username = request.POST['username']
           password = request.POST['password']
           user = User.objects.create_user(username, password)
           user.save()

   def login(request):

           user = authenticate(username=username, password=password)
           if user is not None:

       return render(request, 'login.html')

   from .models import Project
   from django.contrib.auth.models import User

   def add_project(request):
       if request.method == 'POST':
           title = request.POST['title']
           description = request.POST['description']
           project = Project.objects.create(title=title, description=description)
           project.save()
           return redirect('projects')
       return render(request, 'add_project.html')

   def apply_project(request, project_id):
       if request.method == 'POST':
           user = User.objects.get(id=request.user.id)
           project = Project.objects.get(id=project_id)
           project.volunteers.add(user)
           return redirect('projects')
       return render(request, 'apply_project.html')
   ```

3. **任务分配与跟踪**：

   from .models import Task
   from datetime import datetime

   def assign_task(request, project_id):
       if request.method == 'POST':
           task_title = request.POST['task_title']
           task_description = request.POST['task_description']
           due_date = request.POST['due_date']
           task = Task.objects.create(title=task_title, description=task_description, due_date=due_date, project_id=project_id)
           task.save()
           return redirect('tasks', project_id=project_id)
       return render(request, 'assign_task.html', {'project_id': project_id})

   def track_task(request, task_id):
       if request.method == 'POST':
           task = Task.objects.get(id=task_id)
           task.completed_at = datetime.now()
           task.save()
           return redirect('tasks', project_id=task.project_id)
       return render(request, 'track_task.html', {'task_id': task_id})
   ```
