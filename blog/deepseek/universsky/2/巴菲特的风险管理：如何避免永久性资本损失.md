```python
import numpy as np
import pandas as pd
from scipy import stats

def calculate_intrinsic_value(cash_flows, discount_rate):
    return np.sum(cash_flows / (1 + discount_rate) ** np.arange(len(cash_flows)))

def calculate_margin_of_safety(market_price, intrinsic_value):
    return (intrinsic_value - market_price) / intrinsic_value

# 示例数据
cash_flows = [100, 120, 150, 200]
discount_rate = 0.08
market_price = 100

intrinsic_value = calculate_intrinsic_value(cash_flows, discount_rate)
margin_of_safety = calculate_margin_of_safety(market_price, intrinsic_value)

print(f"内在价值：{intrinsic_value}")
print(f"安全边际：{margin_of_safety}")
```

```python
import numpy as np

def calculate_risk(assets, weights):
    covariance_matrix = np.cov(assets)
    risk = np.sqrt(np.dot(weights.T, covariance_matrix.dot(weights)))
    return risk

def optimize_portfolio(assets, returns, risk_tolerance):
    n = len(assets)
    return np.ones(n) / n

# 示例数据
assets = np.array([[0.1, 0.05], [0.08, 0.03]])
returns = [0.1, 0.08]
risk_tolerance = 0.05

weights = optimize_portfolio(assets, returns, risk_tolerance)
risk = calculate_risk(assets, weights)

print(f"权重：{weights}")
print(f"风险：{risk}")
```

```python
import numpy as np

def calculate_intrinsic_value(cash_flows, discount_rate):
    return np.sum(cash_flows / (1 + discount_rate) ** np.arange(len(cash_flows)))

def calculate_margin_of_safety(market_price, intrinsic_value):
    return (intrinsic_value - market_price) / intrinsic_value

# 示例数据：可口可乐公司
cash_flows = [100, 120, 150, 200]
discount_rate = 0.08
market_price = 100

intrinsic_value = calculate_intrinsic_value(cash_flows, discount_rate)
margin_of_safety = calculate_margin_of_safety(market_price, intrinsic_value)

print(f"内在价值：{intrinsic_value}")
print(f"安全边际：{margin_of_safety}")
```
