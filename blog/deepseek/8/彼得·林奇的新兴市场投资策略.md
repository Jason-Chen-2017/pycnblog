```python
import numpy as np

def calculate_financial_metrics(df):
    """
    计算财务指标
    """
    # 计算盈利能力指标
    df['pe_ratio'] = df['close'] / df['profit']
    df['roe'] = df['net_profit'] / df['total_assets']
    # 计算财务稳定性指标
    df['debt_ratio'] = df['liabilities'] / df['total_assets']
    df['current_ratio'] = df['current_assets'] / df['current_liabilities']
    return df
```
