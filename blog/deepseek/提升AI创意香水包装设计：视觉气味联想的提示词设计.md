#### 5.3.2 关键代码解读

1. **生成器和判别器模型**：生成器和判别器是GAN的核心组成部分。

   ```python
   import tensorflow as tf
   from tensorflow.keras.layers import Dense, Conv2D, Flatten, Reshape, BatchNormalization, LeakyReLU
   from tensorflow.keras.models import Sequential

   # 定义生成器模型
   def build_generator(z_dim):
       model = Sequential()
       model.add(Dense(128 * 8 * 8, input_dim=z_dim, activation='relu'))
       model.add(LeakyReLU(alpha=0.01))
       model.add(BatchNormalization(momentum=0.8))
       model.add(Reshape((8, 8, 128)))
       
       model.add(Conv2D(256, kernel_size=(5, 5), padding='same', activation='relu'))
       model.add(LeakyReLU(alpha=0.01))
       model.add(BatchNormalization(momentum=0.8))
       
       model.add(Conv2D(512, kernel_size=(5, 5), padding='same', activation='relu'))
       model.add(LeakyReLU(alpha=0.01))
       model.add(BatchNormalization(momentum=0.8))
       
       model.add(Reshape((32, 32, 512)))
       model.add(Conv2D(1, kernel_size=(5, 5), padding='same', activation='tanh'))
       model.add(Reshape((32, 32, 1)))
       
       return model

   # 定义判别器模型
   def build_discriminator(img_shape):
       model = Sequential()
       model.add(Conv2D(512, kernel_size=(5, 5), padding='same', input_shape=img_shape, activation='relu'))
       model.add(LeakyReLU(alpha=0.01))
       model.add(Dropout(0.3))
       
       model.add(Conv2D(512, kernel_size=(5, 5), padding='same', activation='relu'))
       model.add(LeakyReLU(alpha=0.01))
       model.add(Dropout(0.3))
       
       model.add(Flatten())
       model.add(Dense(1, activation='sigmoid'))
       
       return model
   ```

2. **遗传算法**：遗传算法通过种群初始化、交叉和变异操作、评估和选择过程，优化设计作品。

   ```python
   import numpy as np

   # 种群初始化
   def initialize_population(pop_size, num_genes):
       population = []
       for _ in range(pop_size):
           individual = np.random.uniform(0, 1, num_genes)
           population.append(individual)
       return population

   # 交叉操作
   def crossover(parent1, parent2, crossover_rate):
       if np.random.random() < crossover_rate:
           crossover_point = np.random.randint(1, num_genes - 1)
           child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
           child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
       else:
           child1 = parent1
           child2 = parent2
       return child1, child2

   # 变异操作
   def mutate(individual, mutation_rate):
       for i in range(num_genes):
           if np.random.random() < mutation_rate:
               individual[i] = np.random.uniform(0, 1)
       return individual

   # 评估和选择
   def evaluate_population(population, fitness_func):
       fitness_scores = [fitness_func(individual) for individual in population]
       sorted_population = [x for _, x in sorted(zip(fitness_scores, population), reverse=True)]
       return sorted_population[:2]
   ```

   遗传算法通过交叉、变异和评估操作，生成新一代种群，不断优化设计作品。
