##### 3.2 Python源代码详细阐述

```python
# 导入相关库
import pandas as pd
import numpy as np
from sklearn.cluster import KMeans
from sklearn.metrics import silhouette_score

# 数据预处理
def preprocess_data(data):
    # 数据清洗
    data = data.dropna()
    # 数据转换
    data['datetime'] = pd.to_datetime(data['datetime'])
    data.set_index('datetime', inplace=True)
    # 数据整合
    data = data.resample('H').mean()
    return data

# 污染源识别
def identify_source(data, n_clusters):
    kmeans = KMeans(n_clusters=n_clusters, random_state=0)
    kmeans.fit(data)
    labels = kmeans.predict(data)
    return labels

# 影响范围分析
def analyze_impact(data, labels):
    # 计算轮廓系数
    silhouette_avg = silhouette_score(data, labels)
    # 计算平均浓度
    avg_concentration = np.mean(data, axis=1)
    # 计算污染范围
    pollution_range = data[avg_concentration > np.mean(avg_concentration)]
    return pollution_range, silhouette_avg

# 责任归属确定
def assign责任人(data, pollution_range):
    # 计算责任人占比
    responsibility_ratio = np.mean(data[pollution_range], axis=1) / np.mean(data, axis=1)
    # 确定责任人
    responsible_parties = np.where(responsibility_ratio > 0.5)[0]
    return responsible_parties
```
