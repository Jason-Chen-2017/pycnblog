**后端服务代码（Flask实现）**：

```python
from flask import Flask, request, jsonify
from models import User, LearningArea, KnowledgeModule, LearningActivity

app = Flask(__name__)

# 用户注册接口
@app.route('/register', methods=['POST'])
def register():
    username = request.form['username']
    password = request.form['password']
    user = User(username, password)
    user.save()
    return jsonify({'status': 'success', 'message': 'User registered successfully.'})

# 用户登录接口
@app.route('/login', methods=['POST'])
def login():
    username = request.form['username']
    password = request.form['password']
    user = User.authenticate(username, password)
    if user:
        return jsonify({'status': 'success', 'message': 'Login successful.', 'user_id': user.id})
    else:
        return jsonify({'status': 'error', 'message': 'Invalid username or password.'})

# 学习领域管理接口
@app.route('/learning_areas', methods=['GET', 'POST'])
def learning_areas():
    if request.method == 'POST':
        name = request.form['name']
        description = request.form['description']
        learning_area = LearningArea(name, description)
        learning_area.save()
        return jsonify({'status': 'success', 'message': 'Learning area created successfully.'})
    else:
        learning_areas = LearningArea.query.all()
        return jsonify({'status': 'success', 'learning_areas': [la.to_dict() for la in learning_areas]})

# 知识模块管理接口
@app.route('/knowledge_modules', methods=['GET', 'POST'])
def knowledge_modules():
    if request.method == 'POST':
        name = request.form['name']
        description = request.form['description']
        knowledge_module = KnowledgeModule(name, description)
        knowledge_module.save()
        return jsonify({'status': 'success', 'message': 'Knowledge module created successfully.'})
    else:
        knowledge_modules = KnowledgeModule.query.all()
        return jsonify({'status': 'success', 'knowledge_modules': [km.to_dict() for km in knowledge_modules]})

# 学习活动管理接口
@app.route('/learning_activities', methods=['GET', 'POST'])
def learning_activities():
    if request.method == 'POST':
        name = request.form['name']
        description = request.form['description']
        start_time = request.form['start_time']
        end_time = request.form['end_time']
        learning_activity = LearningActivity(name, description, start_time, end_time)
        learning_activity.save()
        return jsonify({'status': 'success', 'message': 'Learning activity created successfully.'})
    else:
        learning_activities = LearningActivity.query.all()
        return jsonify({'status': 'success', 'learning_activities': [la.to_dict() for la in learning_activities]})

if __name__ == '__main__':
    app.run(debug=True)
```
