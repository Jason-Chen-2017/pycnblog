   # 加载用户评论数据

   def preprocess_data(data):
       # 清洗文本
       data['cleaned_text'] = data['text'].str.replace(r'[^\w\s]+', '')
       # 去掉停用词
       stop_words = set(nltk.corpus.stopwords.words('english'))
       data['cleaned_text'] = data['cleaned_text'].apply(lambda x: ' '.join([word for word in x.split() if word not in stop_words]))

       data['tokens'] = data['cleaned_text'].apply(lambda x: nltk.word_tokenize(x))

   nlp = spacy.load('en_core_web_sm')

   # 将文本映射到词向量
   def vectorize_text(text):
       doc = nlp(text)
       return [token.vector for token in doc]

   train_vectors = [vectorize_text(text) for text in train_data['cleaned_text']]

   # 定义图神经网络模型
   def build_model(num_nodes, num_features, num_classes):
       inputs = Input(shape=(num_features,))

       x = GraphConvolution(num_classes)(x)
       outputs = Activation('softmax')(x)

   model = build_model(num_nodes=len(train_vectors), num_features=300, num_classes=10)
