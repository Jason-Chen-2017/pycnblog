import pandas as pd
import numpy as np
from transformers import GPT2LMHeadModel, GPT2Tokenizer

def generate_simulation_data(num_samples, data_path):
    # 加载GPT-2模型和分词器
    model = GPT2LMHeadModel.from_pretrained('gpt2')
    tokenizer = GPT2Tokenizer.from_pretrained('gpt2')

    # 生成模拟数据
    data = []
    for _ in range(num_samples):
        inputs = tokenizer.encode("开始生成风能数据", return_tensors='pt')
        outputs = model.generate(inputs, max_length=100, num_return_sequences=1)
        text = tokenizer.decode(outputs[0], skip_special_tokens=True)
        data.append(text)

    # 存储数据
    df = pd.DataFrame(data, columns=['wind_data'])
    df.to_csv(data_path, index=False)

generate_simulation_data(1000, 'simulation_data.csv')

##### 5.2.2 模型训练

```python
import tensorflow as tf
from transformers import TFGPT2LMHeadModel

def train_model(data_path):
    # 加载数据
    dataset = pd.read_csv(data_path)
    inputs = dataset['wind_data'].values

    # 初始化模型
    model = TFGPT2LMHeadModel.from_pretrained('gpt2')

    # 定义优化器和损失函数
    optimizer = tf.keras.optimizers.Adam(learning_rate=5e-5)
    loss_fn = tf.keras.losses.SparseCategoricalCrossentropy(from_logits=True)

    # 训练模型
    for epoch in range(3):
        print(f"Epoch {epoch+1}")
        total_loss = 0
        for inputs in inputs:
            with tf.GradientTape() as tape:
                logits = model(inputs)
                loss = loss_fn(inputs, logits)
            grads = tape.gradient(loss, model.trainable_variables)
            optimizer.apply_gradients(zip(grads, model.trainable_variables))
            total_loss += loss

        print(f"Total Loss: {total_loss}")

    # 保存模型
    model.save_pretrained('wind_energy_model')

train_model('simulation_data.csv')
```
