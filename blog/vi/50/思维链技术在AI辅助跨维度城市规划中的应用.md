```python
import numpy as np

# 定义思维链模型
class MindChainModel:
    def __init__(self):
        # 初始化模型参数
        self.weights = np.random.rand(10, 1)
        self.bias = np.random.rand(1)

    def forward(self, x):
        # 前向传播
        self.x = x
        self.z = np.dot(self.weights, self.x) + self.bias
        self.a = 1 / (1 + np.exp(-self.z))
        return self.a

    def backward(self, d_output):
        # 反向传播
        d_z = d_output * (1 - self.a) * self.a
        d_weights = np.dot(self.x.T, d_z)
        d_bias = np.sum(d_z, axis=0, keepdims=True)
        return d_weights, d_bias

    def update(self, d_weights, d_bias, learning_rate):
        # 更新模型参数
        self.weights -= learning_rate * d_weights
        self.bias -= learning_rate * d_bias

# 创建思维链模型实例
model = MindChainModel()

# 前向传播
output = model.forward(np.array([0.5, 0.5]))

# 反向传播
d_output = np.array([0.1])
d_weights, d_bias = model.backward(d_output)

# 更新模型参数
model.update(d_weights, d_bias, learning_rate=0.1)
```
