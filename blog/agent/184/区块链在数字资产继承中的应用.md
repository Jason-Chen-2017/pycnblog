import "@openzeppelin/contracts/token/ERC20/IERC20.sol";

contract DigitalAssetInheritance {
    address public deceased;
    mapping(address => mapping(address => uint256)) public assetDistribution;

    constructor() public {
        deceased = msg.sender;
    }

    function depositAsset(address assetAddress, uint256 amount) public {
        require(msg.sender == deceased, "Only the deceased can deposit assets.");
        require(amount > 0, "Amount must be greater than zero.");

        IERC20 token = IERC20(assetAddress);
        token.transferFrom(msg.sender, address(this), amount);

        emit AssetDeposited(assetAddress, amount);
    }

    function distributeAssets(address[] memory heirs) public {
        require(msg.sender == deceased, "Only the deceased can distribute assets.");
        require(heirs.length > 0, "At least one heir is required.");

        for (uint256 i = 0; i < heirs.length; i++) {
            address heir = heirs[i];
            require(heir != address(0), "Invalid heir address.");
            require(assetDistribution[deceased][heir] > 0, "No assets available for distribution.");

            IERC20 token = IERC20(assetDistribution[deceased][heir]);
            token.transfer(heir, assetDistribution[deceased][heir]);

            emit AssetDistributed(heir, assetDistribution[deceased][heir]);
            delete assetDistribution[deceased][heir];
        }
    }

    function claimAsset(address heir, address assetAddress) public {
        require(msg.sender == heir, "Only the heir can claim assets.");
        require(assetDistribution[deceased][heir] > 0, "No assets available for claim.");

        IERC20 token = IERC20(assetAddress);
        token.transfer(heir, assetDistribution[deceased][heir]);

        emit AssetClaimed(heir, assetAddress, assetDistribution[deceased][heir]);
        delete assetDistribution[deceased][heir];
    }

    event AssetDeposited(address assetAddress, uint256 amount);
    event AssetDistributed(address heir, uint256 amount);
    event AssetClaimed(address heir, address assetAddress, uint256 amount);
}
