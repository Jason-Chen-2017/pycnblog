```python
# app.py
from flask import Flask, render_template, request, redirect, url_for
from flask_sqlalchemy import SQLAlchemy
from flask_login import LoginManager, login_user, logout_user, login_required, current_user

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'postgresql://username:password@localhost/postmaterialism_db'
app.config['SECRET_KEY'] = 'your_secret_key'

db = SQLAlchemy(app)
login_manager = LoginManager(app)
login_manager.login_view = 'login'

# 用户模型
class User(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    username = db.Column(db.String(100), unique=True, nullable=False)
    password = db.Column(db.String(100), nullable=False)

# 员工模型
class Employee(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(100), nullable=False)
    user_id = db.Column(db.Integer, db.ForeignKey('user.id'), nullable=False)

@login_manager.user_loader
def load_user(user_id):
    return User.query.get(int(user_id))

@app.route('/')
@login_required
def index():
    return render_template('index.html')

@app.route('/login', methods=['GET', 'POST'])
def login():
    if request.method == 'POST':
        username = request.form['username']
        password = request.form['password']
        user = User.query.filter_by(username=username).first()
        if user and user.password == password:
            login_user(user)
            return redirect(url_for('index'))
        else:
            return 'Invalid credentials'
    return render_template('login.html')

@app.route('/logout')
@login_required
def logout():
    logout_user()
    return redirect(url_for('login'))

if __name__ == '__main__':
    app.run(debug=True)
```

1. **用户模型**（`User`）：

   用户模型定义了用户的基本信息，包括用户ID、用户名和密码。用户名和密码是唯一且不可为空的，用于用户登录验证。

   ```python
   class User(db.Model):
       id = db.Column(db.Integer, primary_key=True)
       username = db.Column(db.String(100), unique=True, nullable=False)
       password = db.Column(db.String(100), nullable=False)
   ```

2. **员工模型**（`Employee`）：

   员工模型定义了员工的基本信息，包括员工ID、姓名和关联的用户ID。员工ID和用户ID通过外键关联，确保每个员工都有一个对应的用户账户。

   ```python
   class Employee(db.Model):
       id = db.Column(db.Integer, primary_key=True)
       name = db.Column(db.String(100), nullable=False)
       user_id = db.Column(db.Integer, db.ForeignKey('user.id'), nullable=False)
   ```

3. **登录管理**（`login`、`logout` 和 `login_required`）：

   登录管理部分负责用户的登录、登出和权限控制。用户通过输入用户名和密码进行登录，系统验证用户信息后，将用户登录状态设置为`True`。用户登出时，登录状态被重置。`login_required`装饰器用于保护需要登录才能访问的视图函数。

   ```python
   @app.route('/login', methods=['GET', 'POST'])
   def login():
       # ... 登录逻辑 ...

   @app.route('/logout')
   @login_required
   def logout():
       logout_user()
       return redirect(url_for('login'))

   def login_required(f):
       @wraps(f)
       def decorated_function(*args, **kwargs):
           if not current_user.is_authenticated:
               return redirect(url_for('login'))
           return f(*args, **kwargs)
       return decorated_function
   ```

4. **主视图函数**（`index`）：

   主视图函数用于展示系统的首页，用户登录后可以访问该页面。

   ```python
   @app.route('/')
   @login_required
   def index():
       return render_template('index.html')
   ```
