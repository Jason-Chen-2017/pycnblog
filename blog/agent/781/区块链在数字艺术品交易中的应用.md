pragma solidity ^0.8.0;

contract DigitalArt {

    // 数字艺术品结构
    struct Art {
        string name;
        string author;
        string fingerprint;
        address owner;
    }

    // 艺术品映射
    mapping(uint256 => Art) public artworks;

    // 艺术品ID计数器
    uint256 public artCounter;

    // 事件：艺术品注册
    event ArtRegistered(uint256 id, string name, string author, string fingerprint, address owner);

    // 事件：艺术品出售
    event ArtSold(uint256 id, address buyer, uint256 price);

    // 注册艺术品
    function registerArt(string memory name, string memory author, string memory fingerprint) public {
        require(artworks[artCounter].name == "", "Art already registered");
        artworks[artCounter] = Art(name, author, fingerprint, msg.sender);
        emit ArtRegistered(artCounter, name, author, fingerprint, msg.sender);
        artCounter++;
    }

    // 购买艺术品
    function buyArt(uint256 id, uint256 price) public payable {
        Art storage art = artworks[id];
        require(art.name != "", "Art not found");
        require(msg.value >= price, "Insufficient payment");
        require(art.owner != msg.sender, "You already own this art");
        address previousOwner = art.owner;
        previousOwner.transfer(price);
        art.owner = msg.sender;
        emit ArtSold(id, msg.sender, price);
    }

    // 验证艺术品指纹
    function verifyFingerprint(uint256 id, string memory fingerprint) public view returns (bool) {
        Art storage art = artworks[id];
        return keccak256(abi.encodePacked(art.fingerprint)) == keccak256(abi.encodePacked(fingerprint));
    }
}
