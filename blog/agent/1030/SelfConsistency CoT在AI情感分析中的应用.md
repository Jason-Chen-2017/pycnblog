### 5.2 系统核心实现源代码

以下是系统核心实现的源代码：

```python
import tensorflow as tf
import numpy as np
import spacy
from nltk.tokenize import word_tokenize
from nltk.corpus import stopwords

# 加载 spaCy 模型
nlp = spacy.load("en_core_web_sm")

# 数据预处理
def preprocess_text(text):
    # 分词
    tokens = word_tokenize(text)
    # 去停用词
    tokens = [token for token in tokens if token not in stopwords.words("english")]
    # 词性标注
    doc = nlp(" ".join(tokens))
    concepts = []
    for token in doc:
        if token.pos_ in ["NOUN", "VERB", "ADJ"]:
            concepts.append(token.text)
    return concepts

# 概念图生成
def generate_concept_map(tokens):
    concept_map = {}
    for token in tokens:
        concept_map[token] = []
    return concept_map

# 自我一致性约束
def self_consistency_constraint(concept_map):
    weights = {}
    angles = {}
    for token, neighbors in concept_map.items():
        weights[token] = len(neighbors)
        angles[token] = np.mean([np.arccos(np.dot(neighbors[i], neighbors[j])) for i in range(len(neighbors)) for j in range(len(neighbors))])
    return weights, angles

# 情感分类
def sentiment_classification(text):
    concepts = preprocess_text(text)
    concept_map = generate_concept_map(concepts)
    weights, angles = self_consistency_constraint(concept_map)
    # 生成文本特征向量
    text_vector = np.array([weights[token] * np.cos(angles[token]) for token in concepts])
    # 训练模型
    model = tf.keras.Sequential([
        tf.keras.layers.Dense(units=128, activation='relu', input_shape=[len(concepts)]),
        tf.keras.layers.Dense(units=64, activation='relu'),
        tf.keras.layers.Dense(units=1, activation='sigmoid')
    ])
    model.compile(optimizer='adam', loss='binary_crossentropy', metrics=['accuracy'])
    model.fit(text_vector, np.array([1] * len(concepts)), epochs=10, batch_size=32)
    # 预测情感类别
    predicted_sentiment = model.predict(text_vector)
    return "Positive" if predicted_sentiment[0][0] > 0.5 else "Negative"

# 测试
text = "I love this product!"
print(sentiment_classification(text))
```
