3. **分配**：在轮询过程中，如果当前用户有租赁请求且存在可用乐器，则将该乐器分配给该用户。

```python
def create_instrument():
    data = request.json
    instrument = Instrument(name=data['name'], quantity=data['quantity'])
    db.session.add(instrument)
    db.session.commit()
    return jsonify({'id': instrument.id})

@app.route('/rentals', methods=['POST'])
def create_rental():
    data = request.json
    rental = Rental(user_id=data['user_id'], instrument_id=data['instrument_id'], start_date=data['start_date'], end_date=data['end_date'])
    db.session.add(rental)
    db.session.commit()
    return jsonify({'id': rental.id})

if __name__ == '__main__':

```
