```python
import pandas as pd
import numpy as np
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

# 用户需求分析
def user_demand_analysis(data):
    # 读取用户行为数据
    user_data = pd.read_csv(data)
    
    # 计算用户点击率和购买率
    click_rate = user_data['click_count'] / len(user_data)
    purchase_rate = user_data['purchase_count'] / len(user_data)
    
    return click_rate, purchase_rate

# 营销目标确定
def marketing_objective_determination(click_rate, purchase_rate):
    # 确定营销目标
    if click_rate > 0.3 and purchase_rate > 0.2:
        objective = '提高品牌知名度'
    elif click_rate > 0.2 and purchase_rate > 0.3:
        objective = '增加销售额'
    else:
        objective = '提高用户参与度'
    
    return objective

# 提示词生成
def tip_word_generation(user_demand, marketing_objective):
    # 生成提示词
    if marketing_objective == '提高品牌知名度':
        tip_word = '欢迎了解我们的品牌，品质保障，值得信赖！'
    elif marketing_objective == '增加销售额':
        tip_word = '限时优惠，抢购进行中，错过不再有！'
    else:
        tip_word = '欢迎参与互动，更多惊喜等你来！'
    
    return tip_word

# 提示词评估
def tip_word_evaluation(data, tip_words):
    # 评估提示词效果
    user_data = pd.read_csv(data)
    user_evaluation = user_data['evaluation']
    correct_evaluation = user_evaluation[tip_words == user_evaluation]
    
    accuracy = correct_evaluation.shape[0] / user_evaluation.shape[0]
    
    return accuracy

# 主函数
def main():
    data = 'user_data.csv'
    click_rate, purchase_rate = user_demand_analysis(data)
    marketing_objective = marketing_objective_determination(click_rate, purchase_rate)
    tip_word = tip_word_generation(click_rate, purchase_rate)
    accuracy = tip_word_evaluation(data, tip_word)
    
    print(f"营销目标：{marketing_objective}")
    print(f"提示词：{tip_word}")
    print(f"评估结果：{accuracy:.2f}")

if __name__ == '__main__':
    main()
```
