import numpy as np

def normalize_vector(v):
    """归一化向量"""
    norm = np.linalg.norm(v)
    return v / norm

def is_rotational_symmetry(actions, angle):
    """检测旋转对称性"""
    for i in range(len(actions)):
        rotated_action = np.rot90(actions[i], k=angle)
        if np.array_equal(actions[i], rotated_action):
            return True
    return False

def is_mirrored_symmetry(actions):
    """检测镜像对称性"""
    for i in range(len(actions) // 2):
        mirrored_action = actions[-i-1::-1]
        if np.array_equal(actions[i], mirrored_action):
            return True
    return False

def detect_symmetry(actions):
    """检测对称性"""
    normalized_actions = [normalize_vector(action) for action in actions]
    if is_rotational_symmetry(normalized_actions, 1):
        return "旋转对称性"
    elif is_mirrored_symmetry(normalized_actions):
        return "镜像对称性"
    else:
        return "无对称性"

#### 1. 李群和李代数的介绍

李群的定义可以形式化如下：

$$
G = \{ (x, v) \mid x \in X, v \in V \}
$$

其中，$X$ 是一个有限维线性空间，$V$ 是 $X$ 的对偶空间，即 $V = \{ f \mid f: X \to \mathbb{R} \}$。$(x, v)$ 表示李群 $G$ 中的一个元素，其中 $x$ 是 $X$ 中的一个点，$v$ 是 $X$ 上的一个线性泛函。

李群的一个重要概念是对偶李群（Dual Group），对偶李群是李群的李代数（Lie Algebra）的对偶空间。李代数描述了李群的线性部分，是李群理论的核心概念之一。

李代数的定义可以形式化如下：

$$
\mathfrak{g} = \{ A \mid A: X \to V \}
$$

其中，$A$ 是 $X$ 上的一个线性映射。李代数的基本性质包括其内部的加法和标量乘法，并满足李括号性质。

```python
import numpy as np
import matplotlib.pyplot as plt

def generate_dance_action(angle):
    """生成一个具有旋转对称性的舞蹈动作"""
    num_steps = 16
    rotation_angle = angle / num_steps

    dance_action = np.zeros((num_steps, 2))
    for i in range(num_steps):
        dance_action[i] = [np.cos(rotation_angle * i), np.sin(rotation_angle * i)]

    return dance_action

def plot_dance_action(dance_action):
    """绘制舞蹈动作"""
    plt.figure(figsize=(8, 6))
    for action in dance_action:
        plt.scatter(*action, s=100, c='r', marker='o')

    plt.axis('equal')
    plt.xlabel('X')
    plt.ylabel('Y')
    plt.title('Dance Action')
    plt.show()

angle = np.pi / 4
dance_action = generate_dance_action(angle)
plot_dance_action(dance_action)
```

`plot_dance_action` 函数用于绘制舞蹈动作。该函数接受一个舞蹈动作序列作为输入，并使用Matplotlib库绘制坐标点。
