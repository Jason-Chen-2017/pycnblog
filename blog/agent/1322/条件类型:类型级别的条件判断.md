```python
# 购物车服务
from flask import Flask, request, jsonify
from sqlalchemy.orm import sessionmaker
from models import ShoppingCart, db

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///shoppingcart.db'
db.init_app(app)

@app.route('/shoppingcart', methods=['POST'])
def create_shopping_cart():
    data = request.json
    shopping_cart = ShoppingCart(id=data['id'], products=data['products'])
    db.session.add(shopping_cart)
    db.session.commit()
    return jsonify({'message': '购物车已创建'})

# 订单服务
from models import Order, db

@app.route('/order', methods=['POST'])
def create_order():
    data = request.json
    order = Order(id=data['id'], shopping_cart_id=data['shopping_cart_id'])
    db.session.add(order)
    db.session.commit()
    return jsonify({'message': '订单已创建'})

# 推荐引擎
from models import Recommendation, db
from recommendation_engine import generate_recommendation

@app.route('/recommendation', methods=['POST'])
def generate_recommendation_list():
    data = request.json
    recommendation = generate_recommendation(data['shopping_cart_id'])
    db.session.add(recommendation)
    db.session.commit()
    return jsonify({'message': '推荐列表已生成'})

if __name__ == '__main__':
    app.run(debug=True)
```
