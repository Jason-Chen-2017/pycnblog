#### 5.2 系统核心实现源代码

接下来，我们将实现系统核心功能，包括原材料管理、生产管理、订单管理和库存管理。以下是核心实现部分的Python代码示例。

```python
# 巧克力工坊系统核心实现源代码

# 导入所需模块
from flask import Flask, request, jsonify
from flask_sqlalchemy import SQLAlchemy

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'mysql+pymysql://root:password@localhost/chocolate_factory'
db = SQLAlchemy(app)

# 定义模型
class Material(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(80), nullable=False)
    quantity = db.Column(db.Float, nullable=False)

class Production(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    start_date = db.Column(db.Date, nullable=False)
    end_date = db.Column(db.Date, nullable=False)
    status = db.Column(db.String(20), nullable=False)

class Order(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    order_date = db.Column(db.Date, nullable=False)
    status = db.Column(db.String(20), nullable=False)
    total_amount = db.Column(db.Float, nullable=False)

# 实现接口
@app.route('/materials', methods=['GET', 'POST'])
def material_api():
    if request.method == 'GET':
        materials = Material.query.all()
        return jsonify([m.serialize() for m in materials])
    elif request.method == 'POST':
        material = Material(name=request.json['name'], quantity=request.json['quantity'])
        db.session.add(material)
        db.session.commit()
        return jsonify({'message': 'Material added successfully.'})

@app.route('/productions', methods=['GET', 'POST'])
def production_api():
    if request.method == 'GET':
        productions = Production.query.all()
        return jsonify([p.serialize() for p in productions])
    elif request.method == 'POST':
        production = Production(start_date=request.json['start_date'], end_date=request.json['end_date'], status=request.json['status'])
        db.session.add(production)
        db.session.commit()
        return jsonify({'message': 'Production added successfully.'})

@app.route('/orders', methods=['GET', 'POST'])
def order_api():
    if request.method == 'GET':
        orders = Order.query.all()
        return jsonify([o.serialize() for o in orders])
    elif request.method == 'POST':
        order = Order(order_date=request.json['order_date'], status=request.json['status'], total_amount=request.json['total_amount'])
        db.session.add(order)
        db.session.commit()
        return jsonify({'message': 'Order added successfully.'})

# 定义模型序列化方法
class MaterialSchema(serializers.ModelSerializer):
    class Meta:
        model = Material
        fields = '__all__'

class ProductionSchema(serializers.ModelSerializer):
    class Meta:
        model = Production
        fields = '__all__'

class OrderSchema(serializers.ModelSerializer):
    class Meta:
        model = Order
        fields = '__all__'

if __name__ == '__main__':
    app.run(debug=True)
```
