import qiskit
from qiskit import QuantumCircuit, Aer, execute
from qiskit.aqua.algorithms import QAOA
from qiskit.aqua.components.operators import WeightedPauliOperator

# 创建量子电路

qc.h(0)
qc.cx(0, 1)
qc.cx(1, 2)

# 定义哈密顿量
pauli_word = {'paulis': [{'pauli': 'I', 'coefficient': -1},

# 创建QAOA算法实例
qaoa = QAOA(hamiltonian, qubit_order=[0, 1, 2, 3])

# 运行算法
result = qaoa.run()

# 输出结果
print("最优解：", result["solution"])
print("最优解对应的能量：", result["energy"])
