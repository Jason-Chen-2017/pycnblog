```python
import numpy as np
import pandas as pd
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.linear_model import LinearRegression
import spacy

# 加载中文模型
nlp = spacy.load("zh_core_web_sm")

def preprocess_text(text):
    # 使用spaCy进行预处理
    doc = nlp(text)
    tokens = [token.text for token in doc if not token.is_stop]
    return " ".join(tokens)

def extract_features(texts):
    # 提取文本特征
    vectorizer = TfidfVectorizer()
    X = vectorizer.fit_transform(texts)
    return X.toarray()

def train_model(X, y):
    # 训练线性回归模型
    model = LinearRegression()
    model.fit(X, y)
    return model

def calculate_score(model, X):
    # 计算文本评分
    scores = model.predict(X)
    return scores

# 示例文本
texts = ["人工智能是未来发展的关键。", "机器学习能够提高生产效率。", "深度学习正在改变我们的生活方式。"]
labels = [1, 0.8, 0.6]

# 预处理文本
preprocessed_texts = [preprocess_text(text) for text in texts]

# 提取特征
X = extract_features(preprocessed_texts)

# 训练模型
model = train_model(X, labels)

# 测试文本
test_texts = ["强化学习是人工智能的重要组成部分。"]

# 预处理测试文本
test_preprocessed = preprocess_text(test_texts[0])

# 提取测试文本特征
test_X = extract_features([test_preprocessed])

# 计算测试文本评分
test_score = calculate_score(model, test_X)

print(f"测试文本的创新性评分：{test_score[0]}")
```
