```python
import pandas as pd
import numpy as np
import pymysql
import matplotlib.pyplot as plt
import seaborn as sns

# 数据采集
def collect_data():
    # 这里使用一个示例数据集
    data = pd.read_csv('cosmic_data.csv')
    return data

# 数据存储
def store_data(data):
    # 连接到MySQL数据库
    connection = pymysql.connect(host='localhost', user='your_username', password='your_password', database='cosmic_db')
    
    # 创建一个表格
    with connection.cursor() as cursor:
        cursor.execute('''CREATE TABLE IF NOT EXISTS cosmic_data (
            id INT PRIMARY KEY AUTO_INCREMENT,
            name VARCHAR(255),
            magnitude FLOAT,
            type VARCHAR(255)
        )''')
        connection.commit()
    
    # 插入数据
    data.to_sql('cosmic_data', connection, if_exists='append', index=False)

# 数据处理
def process_data(data):
    # 数据清洗和预处理
    data['magnitude'] = pd.to_numeric(data['magnitude'], errors='coerce')
    data.dropna(subset=['magnitude'], inplace=True)
    
    # 数据分组和聚合
    grouped_data = data.groupby('type').mean().reset_index()

    return grouped_data

# 数据分析
def analyze_data(grouped_data):
    # 数据可视化
    sns.barplot(x='name', y='magnitude', data=grouped_data)
    plt.title('Average Magnitude by Type')
    plt.xlabel('Type')
    plt.ylabel('Magnitude')
    plt.show()

# 系统主函数
def main():
    data = collect_data()
    store_data(data)
    processed_data = process_data(data)
    analyze_data(processed_data)

if __name__ == '__main__':
    main()
```
