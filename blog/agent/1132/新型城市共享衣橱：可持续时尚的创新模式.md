**5.1 库存管理算法**

共享衣橱的库存管理是确保服务质量的关键。以下是一个简单的库存管理算法：

```python
# Python实现简单库存管理算法
def manage_inventory(available_closets, desired_closets):
    # 初始化库存状态
    inventory_status = {'available': [], 'unavailable': []}

    # 对每个用户需求的衣橱进行检查
    for closet_id, closet in desired_closets.items():
        if closet_id in available_closets['available']:
            # 如果衣橱可用，将其从可用列表移动到不可用列表
            available_closets['available'].remove(closet_id)
            available_closets['unavailable'].append(closet_id)
            inventory_status['available'].append(closet_id)
        else:
            # 如果衣橱不可用，将其添加到不可用列表
            inventory_status['unavailable'].append(closet_id)

    return inventory_status

# 示例数据
available_closets = {'available': [1, 2, 3], 'unavailable': [4, 5]}
desired_closets = {1: 'User1', 4: 'User2'}

# 调用函数
inventory_status = manage_inventory(available_closets, desired_closets)
print(inventory_status)
```

```python
# Python实现用户行为预测算法
def predict_user_behavior(user_history, recent_activity):
    # 计算用户行为概率
    behavior_probabilities = {'rent': 0.5, 'return': 0.3, 'no_activity': 0.2}

    # 根据最近活动更新概率
    if recent_activity == 'rent':
        behavior_probabilities['rent'] += 0.1
    elif recent_activity == 'return':
        behavior_probabilities['return'] += 0.1

    # 归一化概率
    total_prob = sum(behavior_probabilities.values())
    for behavior in behavior_probabilities:
        behavior_probabilities[behavior] /= total_prob

    return behavior_probabilities

# 示例数据

recent_activity = 'rent'

# 调用函数
behavior_probabilities = predict_user_behavior(user_history, recent_activity)

```
