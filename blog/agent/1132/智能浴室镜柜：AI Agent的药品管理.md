3. **药物剂量控制**：AI Agent可以根据用户身高、体重等个人信息，自动计算出合适的药物剂量，并提醒用户按时服药。

import sqlite3

    drug_data = request.json

#### 5.2.2 剂量计算模块

```python
# 剂量计算模块
import numpy as np
from sklearn.linear_model import LinearRegression

# 训练剂量计算模型
def train_dosage_model():
    # 训练数据
    X = np.array([[1, 70], [1, 60], [1, 80]])  # 用户身高、体重
    y = np.array([100, 50, 150])  # 药物剂量

    # 创建线性回归模型
    model = LinearRegression()
    # 训练模型
    model.fit(X, y)

    return model

# 计算药物剂量
def calculate_dosage(height, weight):
    # 加载训练好的模型
    model = train_dosage_model()
    # 计算剂量
    dosage = model.predict([[1, height, weight]])[0]
    return dosage

# 示例
print(calculate_dosage(170, 60))  # 输出：约90
```

# 计算药物过期天数
def calculate_expire_days(expire_date):
    today = datetime.now().strftime('%Y-%m-%d')
    expire_date = datetime.strptime(expire_date, '%Y-%m-%d')
    days = (expire_date - datetime.strptime(today, '%Y-%m-%d')).days
    return days

# 提醒药物过期
def remind_expire_drugs():
    cursor.execute("SELECT * FROM drugs WHERE expire_date < (SELECT date('now') FROM drugs)")
    expire_drugs = cursor.fetchall()
    for drug in expire_drugs:
        print(f"Drug '{drug[1]}' expires in {calculate_expire_days(drug[5])} days.")
    return
