  # memory_system.py
  import tensorflow as tf

  def create_memory_system():
      # 创建模型
      model = tf.keras.Sequential([

          layers.Dense(64, activation='relu'),

      ])

      return model
  
  def train_memory_system(model, dataset, epochs=10, batch_size=32):
      # 训练模型
      history = model.fit(dataset['X_train'], dataset['y_train'],
                          epochs=epochs,
                          batch_size=batch_size,
                          validation_split=0.2)
      
      return history
  
  def evaluate_memory_system(model, dataset):
      # 评估模型
      test_loss, test_acc = model.evaluate(dataset['X_test'], dataset['y_test'])
      print(f"Test accuracy: {test_acc:.2f}")
  
  def save_memory_system(model, filename='memory_system.h5'):
      # 保存模型
      model.save(filename)
  
  def load_memory_system(filename='memory_system.h5'):
      # 加载模型
      model = tf.keras.models.load_model(filename)
