```python
from flask import Flask, request, jsonify
from sqlalchemy.orm import sessionmaker
from models import Behavior

app = Flask(__name__)

# 创建数据库会话

session = Session()

# API接口：提交行为记录

def submit_behavior():
    data = request.get_json()
    behavior = Behavior(user_id=data['user_id'], behavior=data['behavior'], score=data['score'])
    session.add(behavior)
    session.commit()
    return jsonify({'status': 'success', 'message': 'Behavior recorded.'})

# API接口：获取社会责任得分
@app.route('/api/scores', methods=['GET'])
def get_scores():
    user_id = request.args.get('user_id')
    behaviors = session.query(Behavior).filter_by(user_id=user_id).all()
    total_score = sum(behavior.score for behavior in behaviors)
    return jsonify({'status': 'success', 'total_score': total_score})

if __name__ == '__main__':
    app.run(debug=True)
```
