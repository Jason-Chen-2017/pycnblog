```python
# 定义注意力权重计算函数
def compute_attention_weights(inputs):
    # 计算相似度矩阵
    similarity_matrix = tf.keras.backend.dot(inputs, inputs, transpose_b=True)
    
    # 计算注意力权重
    attention_weights = tf.keras.backend.softmax(similarity_matrix, axis=1)
    
    # 计算注意力加权特征
    attention_weighted_features = tf.keras.backend.dot(attention_weights, inputs)
    
    return attention_weighted_features
```
