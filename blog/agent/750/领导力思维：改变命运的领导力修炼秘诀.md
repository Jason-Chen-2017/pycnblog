class Resource:

        self.type = type
        self.capacity = capacity

    def allocate(self, resource需求):
        if self.capacity >= resource需求:
            self.capacity -= resource需求
            return True
        else:
            return False

def manage_resources(resource_list, resource_demand):
    for resource in resource_list:
