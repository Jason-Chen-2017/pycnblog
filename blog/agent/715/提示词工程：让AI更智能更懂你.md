```python
def select_prompt(data_vectorized):
    similarity_matrix = cosine_similarity(data_vectorized)
    max_similarity = np.max(similarity_matrix)
    return data_vectorized.columns[similarity_matrix == max_similarity].iloc[0]

# 调整权重
def adjust_weights(data_vectorized, prompt):
    data_vectorized[prompt] *= 2
    return data_vectorized
```

```python
import pandas as pd

# 收集用户提问和客服回答的数据
data = pd.DataFrame({"question": ["如何使用产品A？", "产品B的保修期限是多久？", "有哪些产品推荐？"],
                     "answer": ["请参考产品A的使用手册。", "产品B的保修期限为一年。", "推荐产品C和产品D。"]})

# 数据预处理
def preprocess_data(data):
    from sklearn.feature_extraction.text import CountVectorizer
    vectorizer = CountVectorizer()
    X = vectorizer.fit_transform(data["answer"])
    return X

X = preprocess_data(data)
```

**2. 提示词生成**

```python
from sklearn.naive_bayes import MultinomialNB

# 模型训练
model = MultinomialNB()
model.fit(X, data["question"])

# 提示词生成
def generate_prompt(model, vectorizer, question):
    predicted_question = model.predict(vectorizer.transform([question]))
    return vectorizer.get_feature_names()[predicted_question[0]]

question = "产品B的保修期限是多久？"
prompt = generate_prompt(model, vectorizer, question)
print(prompt)
```

**3. 提示词优化**

```python
from sklearn.metrics.pairwise import cosine_similarity

# 提示词优化
def optimize_prompt(prompt, data_vectorized):
    similarity_matrix = cosine_similarity(data_vectorized)
    max_similarity = np.max(similarity_matrix)
    return data_vectorized.columns[similarity_matrix == max_similarity].iloc[0]

optimized_prompt = optimize_prompt(prompt, X)
print(optimized_prompt)
```
