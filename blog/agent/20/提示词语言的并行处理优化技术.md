```python
import multiprocessing
from transformers import pipeline

def parallel_text_generation(keywords):
    # 创建文本生成模型
    generator = pipeline("text-generation", model="gpt2")

    # 生成文本
    texts = [generator(keyword, max_length=50, num_return_sequences=1) for keyword in keywords]

    # 返回生成的文本
    return texts

if __name__ == "__main__":
    # 输入关键词
    keywords = ["智能手表", "蓝牙耳机", "智能音箱"]

    # 创建进程池
    pool = multiprocessing.Pool(processes=3)

    # 并行生成文本
    texts = pool.map(parallel_text_generation, keywords)

    # 打印生成的文本
    for text in texts:
        print(text)
```
