```python
import random
import string

def random_replace(sensitive_info, length):
    """
    随机替换敏感信息
    :param sensitive_info: 敏感信息
    :param length: 替换后的长度
    :return: 脱敏后的信息
    """
    result = ''.join(random.choices(string.ascii_letters + string.digits, k=length))
    return result

# 示例

print(f"脱敏后：{masked_info}")
```
