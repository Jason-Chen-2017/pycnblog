#### 7.1.1 磁场生成算法代码

以下是使用Python实现的磁场生成算法的代码示例：

```python
import numpy as np
import matplotlib.pyplot as plt

# 参数设置
N = 100  # 网格点数量
L = 10   # 网格大小
dt = 0.01  # 时间步长
mu0 = 4*np.pi*10**-7  # 真空磁导率

# 初始化磁场和流体参数
B = np.zeros((N, N))
v = np.zeros((N, N))
rho = np.zeros((N, N))

# 磁流体动力学方程
def mhd_eqns(B, v, rho, dt):
    # 连续方程
    div_v = np.sum(v, axis=1)
    d_rho_dt = -np.sum(v * rho, axis=1)
    
    # 动量方程
    E = -B + 1j * np.cross(v, B)
    F = -mu0 * np.cross(E, v)
    d_B_dt = F / (mu0 * rho) - np.cross(np.gradient(B), B)
    
    # 能量方程
    d_T_dt = -np.dot(v, np.gradient(B)) * np.linalg.norm(B)
    
    # 更新磁场和流体参数
    B_new = B + dt * d_B_dt
    v_new = v
    rho_new = rho + dt * d_rho_dt
    
    return B_new, v_new, rho_new

# 模拟时间步
t = 0
while t < 1:
    B, v, rho = mhd_eqns(B, v, rho, dt)
    t += dt

# 绘制磁场分布
plt.imshow(np.abs(B[:, N//2:].T), cmap='hot', interpolation='nearest')
plt.colorbar()
plt.show()
```
