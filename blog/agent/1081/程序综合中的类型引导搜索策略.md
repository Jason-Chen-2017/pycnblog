# 深度优先搜索
def depth_first_search(G, start_node, target_type):
    stack = [(start_node, [])]
    while stack:
        node, path = stack.pop()
        if node == target_type:
            return path + [node]
        for neighbor in G.neighbors(node):
            if neighbor not in path:
                stack.append((neighbor, path + [node]))
    return None
