# 初始化量子比特

qubit.compose(QuantumCircuit.from_instruction(input_state), qubits=[0])

# 量子神经网络计算

quantum_gate.rx(np.pi/4, 0)
qubit.compose(quantum_gate, qubits=[0])

# 输出解码
qubit.measure(0, 0)

# 执行量子电路
backend = Aer.get_backend("qasm_simulator")
job = execute(qubit, backend, shots=1000)
result = job.result()

# 测量结果
measures = result.get_counts(qubit)
print(measures)
