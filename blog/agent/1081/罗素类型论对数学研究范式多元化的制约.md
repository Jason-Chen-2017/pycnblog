```python
from flask_sqlalchemy import SQLAlchemy

db = SQLAlchemy()

class MathProblem(db.Model):

    description = db.Column(db.String(255))
    algorithm = db.Column(db.Text)
    model = db.Column(db.Text)
```

**服务层（后端）**：

```python
# 服务层实现
from flask import Flask, request, jsonify
from models import db, MathProblem

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///math-problems.db'
db.init_app(app)

@app.route('/api/math-problems', methods=['POST'])
def create_math_problem():
    data = request.get_json()
    problem = MathProblem(description=data['description'], algorithm=data['algorithm'], model=data['model'])
    db.session.add(problem)
    db.session.commit()
    return jsonify({'id': problem.id})

if __name__ == '__main__':
    app.run(debug=True)
```
