```python
import openai
from django.http import HttpResponse

# 设置OpenAI API密钥
openai.api_key = 'your-api-key'

def generate_suggestion(requirement):
    """
    使用ChatGPT生成职业建议
    """
    # 构建提示词
    prompt = f"请根据以下职业需求提供职业建议：{requirement}"
    
    # 调用ChatGPT API
    response = openai.Completion.create(
        engine="text-davinci-003",
        prompt=prompt,
        max_tokens=100
    )
    
    # 返回生成的职业建议
    return response.choices[0].text.strip()

def get_suggestion(request):
    """
    处理职业建议请求
    """
    # 获取用户输入的职业需求
    requirement = request.POST.get('requirement', '')

    # 生成职业建议
    suggestion = generate_suggestion(requirement)

    # 返回职业建议HTML页面
    return HttpResponse(f"<h1>职业建议：</h1><p>{suggestion}</p>")
```
