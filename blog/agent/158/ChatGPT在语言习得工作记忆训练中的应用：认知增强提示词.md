#### Python源代码

以下是一个简单的Python源代码示例，展示如何使用ChatGPT进行语言习得工作记忆训练：

```python
import openai

# 设置API密钥
openai.api_key = 'your_api_key'

# 定义输入文本
input_text = "请告诉我关于工作记忆训练的信息。"

# 预处理文本
def preprocess_text(text):
    # 对输入文本进行预处理，如去除无关字符、转换文本格式等
    processed_text = text.strip().lower()
    return processed_text

processed_text = preprocess_text(input_text)

# 生成提示词
def generate_prompt(processed_text):
    # 根据预处理后的文本生成提示词
    prompt = f"请解释什么是工作记忆训练，并给出几个例子。输入文本：{processed_text}"
    return prompt

prompt = generate_prompt(processed_text)

# 使用ChatGPT生成文本
response = openai.Completion.create(
    engine="text-davinci-002",
    prompt=prompt,
    max_tokens=50
)

# 打印生成文本
print(response.choices[0].text.strip())
```
