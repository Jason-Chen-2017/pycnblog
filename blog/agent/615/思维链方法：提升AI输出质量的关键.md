```python
import tensorflow as tf
from tensorflow.keras.preprocessing.sequence import pad_sequences

# 加载数据集
data = "人工智能改变了我们的生活。随着人工智能技术的发展，我们的生活发生了翻天覆地的变化。"
words = data.split()
word_index = {word: i for i, word in enumerate(words)}
sequences = [[word_index[word] for word in sentence] for sentence in data.split()]

# 切分训练集和测试集
train_sequences = sequences[:int(len(sequences) * 0.8)]
test_sequences = sequences[int(len(sequences) * 0.8):]

# 序列填充
max_sequence_length = max([len(seq) for seq in train_sequences])

# 构建神经网络
model = tf.keras.Sequential([
    tf.keras.layers.Embedding(input_dim=len(word_index) + 1, output_dim=32),
    tf.keras.layers.LSTM(128),
    tf.keras.layers.Dense(len(word_index) + 1, activation='softmax')
])

# 编译模型
model.compile(optimizer='adam', loss='categorical_crossentropy', metrics=['accuracy'])

# 训练模型
model.fit(train_padded, train_padded, epochs=10, batch_size=32)

# 文本生成
input_sequence = pad_sequences([[word_index[word] for word in words[:10]]], maxlen=max_sequence_length)
```
