import matplotlib.pyplot as plt

def simulate_traffic_flow(prompt):
    # 根据提示词生成交通流量模型
    if prompt == "fast_way":
        traffic_model = np.random.normal(5000, 1000, 100)
    elif prompt == "bus_system":
        traffic_model = np.random.normal(4000, 800, 100)
    elif prompt == "nonmotorized":
        traffic_model = np.random.normal(3000, 500, 100)
    else:
        traffic_model = np.random.normal(2000, 500, 100)
    
    # 执行模拟
    simulation_results = np.cumsum(traffic_model)
    
    # 结果分析
    if np.mean(simulation_results) > 4500:
        print("交通拥堵")
    else:
        print("交通畅通")
    
    # 结果展示
    plt.plot(simulation_results)
    plt.xlabel('时间')
    plt.ylabel('交通流量')
    plt.show()

simulate_traffic_flow("fast_way")

import pandas as pd
import numpy as np

def analyze_traffic_flow(data):
    # 数据预处理
    data['traffic'] = pd.cut(data['traffic'], bins=[0, 2000, 4000, 6000], labels=['低', '中', '高'])
    
    # 分析执行
    analysis_results = data.groupby('traffic').mean()
    
    # 结果评估
    if analysis_results['traffic'].mean() > 3:
        print("交通拥堵")
    else:
        print("交通畅通")
    
    # 结果展示
    analysis_results.plot(kind='bar')
    plt.xlabel('交通流量')
    plt.ylabel('平均值')
    plt.show()

data = pd.DataFrame({

    'traffic': np.random.normal(3500, 1000, 100)
})

analyze_traffic_flow(data)
