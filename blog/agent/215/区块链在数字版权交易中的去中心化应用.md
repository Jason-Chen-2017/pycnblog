## 第2章：区块链技术基础

### 2.1 区块链概述

#### 2.1.1 区块链的定义

区块链是一种分布式数据库技术，其核心特点是去中心化、不可篡改和透明性。区块链通过一系列加密算法，将数据分散存储在多个节点上，每个节点都保存一份完整的数据副本。这使得区块链上的数据无法被单一点破坏或篡改，从而保证了数据的安全性和完整性。

#### 2.1.2 区块链的核心特点

1. **去中心化**：区块链上的数据不是存储在单一的中央服务器上，而是分布在多个节点上。这意味着区块链上的数据不是由单一的机构或个人控制，而是由整个网络共同维护。
2. **不可篡改**：区块链上的数据一旦被写入，就几乎不可能被篡改。这是因为区块链采用了加密算法和时间戳技术，确保了数据的真实性和可靠性。
3. **透明性**：区块链上的数据对所有参与者都是可见的，这意味着交易过程是公开透明的。

#### 2.1.3 区块链的架构

区块链的架构主要包括三个部分：数据层、网络层、共识层。

1. **数据层**：数据层包括区块和链结构，是区块链最基本的部分。每个区块都包含一定数量的事务数据，并通过哈希函数与链中的其他区块相连。
2. **网络层**：网络层包括分布式节点和网络协议，负责数据的传输和同步。
3. **共识层**：共识层包括共识算法，负责在网络中达成数据的一致性。

   ```go
   package main

   import (
       "log"
       "github.com/gin-gonic/gin"
       "github.com/ethereum/go-ethereum/crypto"
   )

   type Copyright struct {
       ID         string `json:"id"`
       Owner      string `json:"owner"`
       Title      string `json:"title"`
       Description string `json:"description"`
       IsRegistered bool   `json:"is_registered"`
   }

   type RegisterRequest struct {
       ID             string `json:"id"`
       Owner          string `json:"owner"`
       Title          string `json:"title"`
       Description    string `json:"description"`
   }

   type TransferRequest struct {
       ID         string `json:"id"`
       NewOwner   string `json:"new_owner"`
   }

   var copyrights = make(map[string]Copyright)

   func registerCopyright(c *gin.Context) {
       var req RegisterRequest
       if err := c.ShouldBindJSON(&req); err != nil {
           c.JSON(400, gin.H{"error": err.Error()})
           return
       }

       id := req.ID
       if _, exists := copyrights[id]; exists {
           c.JSON(400, gin.H{"error": "Copyright already registered"})
           return
       }

       copyrights[id] = Copyright{
           ID:         id,
           Owner:      req.Owner,
           Title:      req.Title,
           Description: req.Description,
           IsRegistered: true,
       }

       c.JSON(200, gin.H{"message": "Copyright registered successfully"})
   }

   func transferCopyright(c *gin.Context) {
       var req TransferRequest
       if err := c.ShouldBindJSON(&req); err != nil {
           c.JSON(400, gin.H{"error": err.Error()})
           return
       }

       id := req.ID
       if _, exists := copyrights[id]; !exists {
           c.JSON(400, gin.H{"error": "Copyright not registered"})
           return
       }

       if msgSender := c.GetHeader("Authorization"); msgSender != copyrights[id].Owner {
           c.JSON(400, gin.H{"error": "Not the owner"})
           return
       }

       copyrights[id].Owner = req.NewOwner

       c.JSON(200, gin.H{"message": "Copyright transferred successfully"})
   }

   func getCopyright(c *gin.Context) {
       id := c.Param("id")
       if _, exists := copyrights[id]; !exists {
           c.JSON(400, gin.H{"error": "Copyright not registered"})
           return
       }

       c.JSON(200, copyrights[id])
   }

   func main() {
       router := gin.Default()

       router.POST("/register", registerCopyright)
       router.POST("/transfer", transferCopyright)
       router.GET("/copyright/:id", getCopyright)

       router.Run(":8080")
   }
   ```

   ```go
   }

   type RegisterRequest struct {
       ID             string `json:"id"`
       Owner          string `json:"owner"`
       Title          string `json:"title"`
       Description    string `json:"description"`
   }

   type TransferRequest struct {
       ID         string `json:"id"`

   var copyrights = make(map[string]Copyright)

   func registerCopyright(c *gin.Context) {
       var req RegisterRequest
       if err := c.ShouldBindJSON(&req); err != nil {

           return
       }

       id := req.ID
       if _, exists := copyrights[id]; exists {
           c.JSON(400, gin.H{"error": "Copyright already registered"})
           return
       }

       copyrights[id] = Copyright{
           ID:         id,
           Owner:      req.Owner,
           Title:      req.Title,
           Description: req.Description,
           IsRegistered: true,
       }

       c.JSON(200, gin.H{"message": "Copyright registered successfully"})
   }

   func transferCopyright(c *gin.Context) {
       var req TransferRequest
       if err := c.ShouldBindJSON(&req); err != nil {
           c.JSON(400, gin.H{"error": err.Error()})
           return
       }

       id := req.ID
       if _, exists := copyrights[id]; !exists {
           c.JSON(400, gin.H{"error": "Copyright not registered"})
           return
       }

       if msgSender := c.GetHeader("Authorization"); msgSender != copyrights[id].Owner {
           c.JSON(400, gin.H{"error": "Not the owner"})
           return
       }

       copyrights[id].Owner = req.NewOwner

       c.JSON(200, gin.H{"message": "Copyright transferred successfully"})
   }

   func getCopyright(c *gin.Context) {
       id := c.Param("id")
       if _, exists := copyrights[id]; !exists {
           c.JSON(400, gin.H{"error": "Copyright not registered"})
           return
       }

       c.JSON(200, copyrights[id])
   }
   ```
