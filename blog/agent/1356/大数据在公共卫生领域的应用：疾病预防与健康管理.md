```python
# 数据采集
import random

def collect_health_data():
    # 假设采集的身体健康数据包括体重、心率、血压等
    weight = random.uniform(50, 100)
    heart_rate = random.uniform(60, 100)
    blood_pressure = random.uniform(90, 140)
    return weight, heart_rate, blood_pressure

# 数据存储
import pymysql

def store_health_data(data):
    # 连接MySQL数据库
    connection = pymysql.connect(host='localhost', user='root', password='password', database='health_data')
    cursor = connection.cursor()
    
    # 存储健康数据到数据库
    sql = "INSERT INTO health_data (weight, heart_rate, blood_pressure) VALUES (%s, %s, %s)"
    cursor.execute(sql, data)
    
    # 提交事务
    connection.commit()
    
    # 关闭连接
    cursor.close()
    connection.close()

# 数据分析
from sklearn.linear_model import LinearRegression

def analyze_health_data():
    # 从数据库中查询健康数据
    connection = pymysql.connect(host='localhost', user='root', password='password', database='health_data')
    cursor = connection.cursor()
    sql = "SELECT * FROM health_data"
    cursor.execute(sql)
    results = cursor.fetchall()
    
    # 分析健康数据，建立线性回归模型
    X = [result[0] for result in results]
    y = [result[1] for result in results]
    model = LinearRegression()
    model.fit(X, y)
    
    # 输出模型参数
    print("模型参数：")
    print(model.coef_)
    print(model.intercept_)
    
    # 关闭连接
    cursor.close()
    connection.close()
```

2. **数据存储**：`store_health_data` 函数用于将采集到的健康数据存储到MySQL数据库中。通过连接MySQL数据库，执行插入操作，将数据存储到数据库表中。

3. **数据分析**：`analyze_health_data` 函数用于分析健康数据，建立线性回归模型。从数据库中查询健康数据，提取体重和心率等特征，使用线性回归模型进行拟合，并输出模型参数。
