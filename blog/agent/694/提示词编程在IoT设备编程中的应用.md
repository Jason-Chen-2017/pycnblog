4. **MQTT**：MQTT是一种轻量级的消息队列协议，适用于物联网设备之间的数据传输。MQTT协议的特点是低带宽消耗、高可靠性、支持异步消息传输，广泛应用于物联网设备和服务器之间的通信。

- **功耗**：对于电池供电的物联网设备，应选择低功耗的通信协议。
- **稳定性**：选择具有高稳定性的通信协议，以确保设备之间的可靠通信。
- **安全性**：选择支持数据加密、身份认证等安全功能的通信协议，保障设备数据的安全。

常见的提示词生成算法包括基于规则的方法、基于机器学习的方法和基于自然语言处理的方法。

   ```python
           return "打开灯"
       elif prediction == 1:
           return "关闭灯"
       else:
           return "未知指令"
   ```

   基于自然语言处理的方法利用自然语言处理技术，对输入文本进行分析和转换，生成合适的提示词。常见的技术包括词性标注、句法分析和语义角色标注等。

   ```python
       doc = nlp(text)
       for token in doc:
           if token.tag_ == "VB":
               return token.text
       return "未知指令"
   ```

```cpp
    } else if (payload == "yellow") {
        // 控制黄灯亮
    } else if (payload == "red") {
        // 控制红灯亮
    }
}

void setup() {
    Serial.begin(115200);
    WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
    while (WiFi.status() != WL_CONNECTED) {
        delay(500);
        Serial.print(".");
    }
    Serial.println("WiFi connected");

    client.setDebugging(false);

    while (!client.connect("arduinoClient", MQTT_USER, MQTT_PASSWORD)) {
        Serial.print(".");
        delay(500);

}
```
