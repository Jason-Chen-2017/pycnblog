        question = refine_question(question)
    return question

def is_ambiguous(question):
    # 判断问题是否明确
    return "不确定" in question

def refine_question(question):
    # 细化问题

    chatgpt = ChatGPT()
    response = chatgpt.answer(prompt)
    return response

def evaluate_efficiency(response):
    # 评估交互效率

```python
import openai
import nltk
from nltk.tokenize import word_tokenize
from mermaid import Mermaid

# 设置openai的API密钥
openai.api_key = '您的API密钥'

# 分析问题
def analyze_question(question):
    # 使用nltk进行分词，获取问题中的关键词
    tokens = word_tokenize(question)
    # 判断问题是否明确
    if "不确定" in tokens or "请" in tokens:
        return False
    return True

# 细化问题
def refine_question(question):
    return question + "，请具体说明。"

# 构建提示词
def build_prompt(question):
    return f"请回答关于{question}的问题。"

# 输入ChatGPT
def input_chatgpt(prompt):
    response = openai.Completion.create(
        engine="text-davinci-003",
        prompt=prompt,
        max_tokens=50
    )
    return response.choices[0].text.strip()

# 评估交互效率
def evaluate_efficiency(response):
    # 这里简单地通过回答长度来判断交互效率
    if len(response) > 10:
        return "高效"
    else:
        return "低效"

# 主函数
def main():
    question = "你有什么特点？"
    # 分析问题
    if not analyze_question(question):
        question = refine_question(question)
    prompt = build_prompt(question)
    # 输入ChatGPT
    response = input_chatgpt(prompt)
    # 评估交互效率
    efficiency = evaluate_efficiency(response)
    print(f"交互效率：{efficiency}")
    print(f"回答：{response}")

# 运行主函数
if __name__ == "__main__":
    main()
```

**代码解释**：

1. **分析问题**：`analyze_question`函数用于判断问题是否明确。这里简单通过检查问题中是否包含“不确定”或“请”等词语来判定。

2. **细化问题**：`refine_question`函数用于细化问题。例如，如果问题是“你有什么特点？”，则返回“你有什么特点，请具体说明？”

3. **构建提示词**：`build_prompt`函数用于构建提示词。根据分析结果，生成相应的提示词。

4. **输入ChatGPT**：`input_chatgpt`函数用于将提示词输入到ChatGPT中，并获取回答。这里使用了openai的Completion接口，可以生成50个词的答案。

5. **评估交互效率**：`evaluate_efficiency`函数用于评估交互效率。这里简单通过计算回答的长度来判断，如果回答长度大于10，则视为高效。

6. **主函数**：`main`函数是整个算法的主入口。首先输入问题，然后进行问题分析、构建提示词、输入ChatGPT和评估交互效率。
