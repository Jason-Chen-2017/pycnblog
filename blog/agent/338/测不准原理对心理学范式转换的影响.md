```python
# app.py

from flask import Flask, request, jsonify
from models import db, Experiment, Question, Answer

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///database.db'
db.init_app(app)

@app.route('/login', methods=['POST'])
def login():
    username = request.form['username']
    password = request.form['password']
    # 登录验证逻辑
    return jsonify({'status': 'success'})

@app.route('/start_experiment', methods=['POST'])
def start_experiment():
    title = request.form['title']
    # 创建实验逻辑
    experiment = Experiment(title=title)
    db.session.add(experiment)
    db.session.commit()
    return jsonify({'experiment_id': experiment.id})

@app.route('/get_questions', methods=['GET'])
def get_questions():
    experiment_id = request.args.get('experiment_id')
    # 获取问题列表逻辑
    questions = Question.query.filter_by(experiment_id=experiment_id).all()
    return jsonify({'questions': [q.to_dict() for q in questions]})

@app.route('/select_answer', methods=['POST'])
def select_answer():
    question_id = request.form['question_id']
    answer_id = request.form['answer_id']
    # 选择答案逻辑
    question = Question.query.get(question_id)
    answer = Answer.query.get(answer_id)
    question.select_answer(answer)
    db.session.commit()
    return jsonify({'status': 'success'})

@app.route('/get_results', methods=['GET'])
def get_results():
    experiment_id = request.args.get('experiment_id')
    # 获取结果逻辑
    experiment = Experiment.query.get(experiment_id)
    results = experiment.get_results()
    return jsonify({'results': results})

if __name__ == '__main__':
    app.run(debug=True)
```
