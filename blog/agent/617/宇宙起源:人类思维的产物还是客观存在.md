# 宇宙起源研究Python代码示例

import numpy as np

# 假设宇宙初始状态为点状，半径为r0
r0 = 1e-10
# 假设宇宙初始温度为T0
T0 = 1e9

# 计算宇宙膨胀率
def HubbleParameter(r):
    return np.sqrt(np.pi * r**2 / (3 * r0 * T0))

# 计算宇宙初期状态的密度
def Density(r):
    return np.exp(-r / r0)

# 计算宇宙初期状态的能量
def Energy(r):
    return Density(r) * r**2

# 计算宇宙初期状态的熵
def Entropy(r):
    return np.log(Density(r))

# 打印宇宙初期状态的相关参数
print("宇宙初期状态：")
print("半径：", r0)
print("温度：", T0)
print("膨胀率：", HubbleParameter(r0))
print("密度：", Density(r0))
print("能量：", Energy(r0))
print("熵：", Entropy(r0))
```
