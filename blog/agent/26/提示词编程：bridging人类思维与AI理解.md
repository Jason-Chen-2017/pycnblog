```python
import nltk
from nltk.tokenize import word_tokenize
from nltk.corpus import stopwords

# 提取关键词
def extract_key_words(input_text):
    tokens = word_tokenize(input_text)
    stop_words = set(stopwords.words('english'))
    filtered_words = [word for word in tokens if word not in stop_words]
    key_words = nltk.FreqDist(filtered_words).most_common(5)
    return key_words

# 生成候选提示词
def generate_candidate_prompts(user_demand):
    key_words = extract_key_words(user_demand)
    candidate_prompts = []
    for word, _ in key_words:
        candidate_prompts.append(f"Do you want to know about {word}?")
    return candidate_prompts

# 筛选有效提示词
def filter_valid_prompts(candidate_prompts, user_demand):
    valid_prompts = []
    for prompt in candidate_prompts:
        if prompt in user_demand:
            valid_prompts.append(prompt)
    return valid_prompts

# 评估提示词效果
def evaluate_prompt_effect(prompt, user_demand):
    # 简单的评估逻辑，实际应用中可以使用更复杂的评估方法
    return prompt in user_demand

# 主函数
def prompt_engineering(user_demand):
    candidate_prompts = generate_candidate_prompts(user_demand)
    valid_prompts = filter_valid_prompts(candidate_prompts, user_demand)
    best_prompt = max(valid_prompts, key=evaluate_prompt_effect)
    return best_prompt
```
