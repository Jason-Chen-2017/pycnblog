#### 1.2 系统核心实现源代码

以下是系统核心实现的Python源代码：

```python
import re
from typing import Optional

class PromptRecognizer:
    def __init__(self):
        self.prompt_pattern = re.compile(r"^\s*(def|class|if)\s+([a-zA-Z0-9_]+)\s*\(.*\)\s*:$")

    def is_valid_prompt(self, prompt: str) -> bool:
        return bool(self.prompt_pattern.match(prompt))

    def generate_code_from_prompt(self, prompt: str) -> Optional[str]:
        if not self.is_valid_prompt(prompt):
            return None

        match = self.prompt_pattern.match(prompt)
        statement_type = match.group(1)
        name = match.group(2)

        if statement_type == "def":
            return f"def {name}(args): return args"
        elif statement_type == "class":
            return f"class {name}(): pass"
        elif statement_type == "if":
            return f"if {name}: pass"

class CodeGenerator:
    def __init__(self, recognizer: PromptRecognizer):
        self.recognizer = recognizer

    def generate_code(self, prompt: str) -> Optional[str]:
        return self.recognizer.generate_code_from_prompt(prompt)

class CodeExecutor:
    def __init__(self, generator: CodeGenerator):
        self.generator = generator

    def execute_code(self, code: str) -> str:
        try:
            exec(code)
            return "Code executed successfully."
        except Exception as e:
            return f"Error executing code: {e}"

if __name__ == "__main__":
    recognizer = PromptRecognizer()
    generator = CodeGenerator(recognizer)
    executor = CodeExecutor(generator)

    prompt = input("Enter a prompt: ")
    if recognizer.is_valid_prompt(prompt):
        code = generator.generate_code(prompt)
        if code:
            result = executor.execute_code(code)
            print(result)
    else:
        print("Invalid prompt.")
```

这个源代码包括三个核心组件：`PromptRecognizer`、`CodeGenerator` 和 `CodeExecutor`。`PromptRecognizer` 负责识别和解析输入的提示词；`CodeGenerator` 负责根据提示词生成代码；`CodeExecutor` 负责执行生成的代码。

import operator

def prompt_calculator():

    ops = {
        "add": operator.add,
