probabilities = [float(counts['00']), float(counts['01']), float(counts['10']), float(counts['11'])]

print(weights)

| 资产 | 风险 | 返回 |
| ---- | ---- | ---- |
| A    | 0.1  | 0.3  |

| C    | 0.2  | 0.5  |

| E    | 0.5  | 0.7  |

我们的目标是找到一种投资组合，使总风险最小化，同时保证预期回报率。

from qiskit.optimization import OptimizationProblem, NumPyOptimizer

# 定义资产的预期回报和风险
returns = np.array([0.3, 0.4, 0.5, 0.6, 0.7])
risks = np.array([0.1, 0.3, 0.2, 0.4, 0.5])

# 创建一个优化问题
problem = OptimizationProblem()

# 添加目标函数，最小化总风险
problem.add_variable('x', 5, minimize=True)
problem.add_constraint('sum(x) == 1')  # 约束条件：投资比例总和为1
problem.add_constraint('x[i] >= 0', variables=[i for i in range(5)])  # 约束条件：投资比例非负

# 添加目标函数的系数，每个资产的风险
problem.add_objective('sum(risks * x)')

# 创建一个优化器，并解决优化问题
optimizer = NumPyOptimizer()
solution = optimizer.solve(problem)

# 输出投资组合权重
weights = solution.x
print(weights)

# 计算最小化风险和预期回报
min_risk = np.dot(risks, weights)
print(f"最小化风险: {min_risk}")
