    cursor.close()

    product_descriptions = [product[2] for product in products]
    product_descriptions = [desc.replace('\n', '') for desc in product_descriptions]

    vectorizer = TfidfVectorizer()
    product_matrix = vectorizer.fit_transform(product_descriptions)
    product_vector = product_matrix[product_id - 1]

    similarity_scores = cosine_similarity(product_vector, product_matrix)
    similarity_scores = similarity_scores.flatten()

    recommended_indices = similarity_scores.argsort()[::-1]
    recommended_indices = recommended_indices[1:6]  # 排除当前产品本身

    recommended_products = [products[i] for i in recommended_indices]
    return recommended_products
