```python
import ast

class LexicalAnalyzer:
    def __init__(self, input_text):
        self.input_text = input_text
        self.position = 0

    def next_token(self):
        while self.position < len(self.input_text):
            char = self.input_text[self.position]
            if char.isspace():
                self.position += 1
                continue
            if char.isalpha() or char == '_':
                self.position += 1
                while self.position < len(self.input_text) and (self.input_text[self.position].isalpha() or self.input_text[self.position] == '_'):
                    self.position += 1
                return ast.Name(self.input_text[self.position - 1])
            if char.isdigit():
                self.position += 1
                while self.position < len(self.input_text) and self.input_text[self.position].isdigit():
                    self.position += 1
                return ast.Num(int(self.input_text[:self.position]))
            if char == '+':
                self.position += 1
                return ast.BinOp(ast.Name('+'), ast.Name('+'), ast.Name('+'))
            if char == '-':
                self.position += 1
                return ast.BinOp(ast.Name('-'), ast.Name('-'), ast.Name('-'))
            if char == '*':
                self.position += 1
                return ast.BinOp(ast.Name('*'), ast.Name('*'), ast.Name('*'))
            if char == '/':
                self.position += 1
                return ast.BinOp(ast.Name('/'), ast.Name('/'), ast.Name('/'))
            raise ValueError(f"未知字符：{char}")

    def analyze(self):
        tokens = []
        while True:
            token = self.next_token()
            if isinstance(token, ast.Name):
                tokens.append(token)
            elif isinstance(token, ast.Num):
                tokens.append(token)
            elif isinstance(token, ast.BinOp):
                tokens.append(token)
            else:
                raise ValueError(f"未知Token类型：{token}")
            if self.position == len(self.input_text):
                break
        return tokens
```

    def __init__(self, tokens):
        self.tokens = tokens
        self.index = 0

    def next_token(self):
        if self.index < len(self.tokens):
            return self.tokens[self.index]
        return None

    def parse(self):
        ast = []
        while self.index < len(self.tokens):
            token = self.next_token()
            if isinstance(token, ast.Name):
                ast.append(ast.Name(token.id))
            elif isinstance(token, ast.Num):
                ast.append(ast.Num(token.n))
            elif isinstance(token, ast.BinOp):
                left = self.parse()
                right = self.parse()
                ast.append(ast.BinOp(left, token.op, right))
            else:
                raise ValueError(f"未知Token类型：{token}")
        return ast

    def check(self):
        for node in self.ast:
            if isinstance(node, ast.Name):
                type = self.context.get(node.id, None)
