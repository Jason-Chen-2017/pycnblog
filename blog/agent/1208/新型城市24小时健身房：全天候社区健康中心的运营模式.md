```python
popular_service = services.groupby('service_id').size().idxmax()
print(f"会员使用最频繁的服务：{popular_service}")

plt.figure(figsize=(10, 5))

plt.xlabel('日期')
plt.ylabel('会员数量')
plt.grid(True)
plt.show()

# 健身设备使用频率分布
plt.figure(figsize=(10, 5))
plt.bar(equipments['equipment_id'], equipments['usage_count'])

plt.ylabel('使用次数')
plt.xticks(rotation=45)
plt.grid(True)
plt.show()

# 服务使用频率分布
plt.figure(figsize=(10, 5))
plt.bar(services['service_id'], services['usage_count'])
plt.title('服务使用频率分布')
plt.xlabel('服务ID')
plt.ylabel('使用次数')
plt.xticks(rotation=45)
plt.grid(True)
plt.show()
```
