```python
    def __init__(self, storage):
        self.storage = storage

    def store(self, prompt):
        if not self.storage.exists(prompt):
            self.storage.insert(prompt)
            return "成功存储提示词"
        else:
            return "提示词已存在，无法存储"

    def retrieve(self, prompt):
        if self.storage.exists(prompt):
            return "成功检索提示词"
        else:
            return "提示词不存在，无法检索"

    def manage(self, prompt, management_strategy):
        if management_strategy(prompt):
            return "提示词被管理"
        else:
            return "提示词未被管理"
```
