```python
import random

def cognitive_reconstruction(current_cognitive):
    """
    认知重构算法：根据当前认知评估，重构负面认知为积极认知。
    :param current_cognitive: 当前认知（字符串）

    """
    if "负面" in current_cognitive:

        return positive_cognitive
    else:

    """
```

```python
# app.py

from flask import Flask, request, jsonify
from pymysql import connect, cursors
import bcrypt

app = Flask(__name__)

def connect_db():
    connection = connect(host="localhost", user="root", password="password", database="emotional_regulation")
    return connection

@app.route('/register', methods=['POST'])
def register():
    username = request.json['username']
    password = request.json['password']
    
    # 连接数据库
    connection = connect_db()
    cursor = connection.cursor()
    
    # 检查用户名是否已存在
    cursor.execute("SELECT * FROM users WHERE username = %s", (username,))
    user = cursor.fetchone()
    
    if user:
        return jsonify({'error': '用户名已存在'}), 409
    
    # 存储用户信息
    hashed_password = bcrypt.hashpw(password.encode('utf-8'), bcrypt.gensalt())
    cursor.execute("INSERT INTO users (username, password) VALUES (%s, %s)", (username, hashed_password))
    connection.commit()
    
    return jsonify({'message': '用户注册成功'})

if __name__ == '__main__':
    app.run(debug=True)
```

```python
    elif sentiment.polarity < 0:
```

```python
import random

    """

    :return: 重构后的认知（字符串）

        return random.choice(emotions)
```

```python
import random

def cognitive_reconstruction(current_cognitive):
    """
    认知重构算法：根据当前认知评估，重构负面认知为积极认知。
    :param current_cognitive: 当前认知（字符串）
    :return: 重构后的认知（字符串）
    """
    if "负面" in current_cognitive:
        # 重构负面认知
        positive_cognitive = current_cognitive.replace("负面", "积极")
        return positive_cognitive
    else:
        return current_cognitive

def emotional_regulation(positive_cognitive):
    """
    情感调节算法：根据重构后的积极认知，调节情感状态。
    :param positive_cognitive: 重构后的积极认知（字符串）
    :return: 调节后的情感状态（字符串）
    """
    emotions = ['焦虑', '平静', '自信', '沮丧']
    if "自信" in positive_cognitive:
        return random.choice(['自信', '平静'])
    else:
        return random.choice(emotions)

def action_planning(positive_cognitive, emotion):
    """
    行动规划算法：根据重构后的积极认知和情感状态，制定行动计划。
    :param positive_cognitive: 重构后的积极认知（字符串）
    :param emotion: 调节后的情感状态（字符串）
    :return: 行动计划（字符串）
    """
    actions = {
        '自信': '尝试新方法',
        '平静': '放松身体',
        '焦虑': '深呼吸',
        '沮丧': '找朋友聊天'
    }
    return actions[emotion]
```

#### 6.1.3 算法原理讲解

1. **初始化**：算法开始时，接收个体的当前认知评估。
2. **评估当前认知**：根据认知评估，判断是积极认知还是负面认知。
3. **识别正面认知**：如果当前认知是正面的，直接增强正面认知。
4. **识别负面认知**：如果当前认知是负面的，进行认知重构，将负面认知转换为积极认知。
5. **情感调节**：根据重构后的积极认知，调节情感状态，选择合适的情感调节策略。
6. **行动规划**：根据调节后的情感状态，制定具体的行动计划。
7. **执行计划**：执行行动计划，采取实际步骤来应对挑战。
8. **反馈与调整**：根据执行结果，进行反馈和调整，优化认知重构和情感调节策略。

通过这个算法，个体可以在面对负面认知时，通过认知重构和情感调节来改变自己的情绪和行为反应，从而提高心理韧性和生活质量。
