```python
import numpy as np

def cosmic_distance(z):
    """
    计算给定红移的宇宙学距离。
    """
    H0 = 70 * np.km / s / Mpc
    omegam = 0.3
    omegade = 0.7
    a = 1 / (1 + z)
    Hubble_param = H0 / np.sqrt(omegam * a**3 + omegade)
    distance = a * (1 + z) * Hubble_param**-1
    return distance

def bayesian_inference(data, model):
    """
    贝叶斯参数估计。
    """
    # 定义先验分布
    prior = np.exp(-0.5 * (model - data)**2)
    # 计算后验分布
    posterior = np.sum(prior)
    return posterior
```
