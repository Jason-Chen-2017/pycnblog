```python
import tkinter as tk
from tkinter import messagebox

# 计算黄金角
class GoldenRatio:
    def calculate(self, length):
        return (1 + (5 ** 0.5)) / 2

# 计算斐波那契数列
class FibonacciSequence:
    def calculate(self, n):
        sequence = [0, 1]
        for i in range(2, n):
            sequence.append(sequence[i-1] + sequence[i-2])
        return sequence

# 展示自然界应用例子
class NatureExample:
    def display(self, example):
        messagebox.showinfo("自然界应用例子", example)

# GUI界面
class GUI(tk.Tk):
    def __init__(self):
        super().__init__()
        self.title("黄金角与斐波那契数列")
        self.geometry("400x300")

        # 黄金角计算
        self golden_ratio = GoldenRatio()
        tk.Label(self, text="输入线段长度：").place(x=10, y=10)
        self.length = tk.Entry(self)
        self.length.place(x=120, y=10)
        tk.Button(self, text="计算黄金角", command=self.calculate_golden_ratio).place(x=10, y=40)

        # 斐波那契数列计算
        self.fibonacci_sequence = FibonacciSequence()
        tk.Label(self, text="输入项数：").place(x=10, y=70)
        self.n = tk.Entry(self)
        self.n.place(x=120, y=70)
        tk.Button(self, text="计算斐波那契数列", command=self.calculate_fibonacci_sequence).place(x=10, y=100)

        # 显示自然界应用例子
        tk.Button(self, text="显示自然界应用例子", command=self.display_nature_example).place(x=10, y=130)

    def calculate_golden_ratio(self):
        length = float(self.length.get())
        golden_ratio = self.golden_ratio.calculate(length)
        messagebox.showinfo("黄金角结果", f"黄金角：{golden_ratio}")

    def calculate_fibonacci_sequence(self):
        n = int(self.n.get())
        sequence = self.fibonacci_sequence.calculate(n)
        messagebox.showinfo("斐波那契数列结果", f"斐波那契数列：{sequence}")

    def display_nature_example(self):
        example = "向日葵的花盘上的花瓣按照黄金角的比例排列，形成了一个完美的螺旋。"
        self.nature_example = NatureExample()
        self.nature_example.display(example)
```
