import numpy as np
import pandas as pd
from sklearn.preprocessing import StandardScaler

# 数据采集
def data_collection(sensor_data):
    # 假设sensor_data是一个包含温度、气体浓度、火焰亮度的列表
    temp = sensor_data[0]
    gas_concentration = sensor_data[1]
    flame_brightness = sensor_data[2]
    return temp, gas_concentration, flame_brightness

# 数据预处理
def data_preprocessing(data):
    # 去噪、归一化等预处理操作
    scaler = StandardScaler()
    processed_data = scaler.fit_transform(data)
    return processed_data

# 特征提取
def feature_extraction(processed_data):
    # 提取温度、气体浓度、火焰亮度等特征
    temp_feature = processed_data[:, 0]
    gas_concentration_feature = processed_data[:, 1]
    flame_brightness_feature = processed_data[:, 2]
    return temp_feature, gas_concentration_feature, flame_brightness_feature

# 主函数
def main():
    # 假设sensor_data是从传感器采集到的原始数据
    sensor_data = np.array([[37, 0.1, 200], [38, 0.2, 210], [36, 0.15, 205]])
    
    # 数据采集
    temp, gas_concentration, flame_brightness = data_collection(sensor_data)
    
    # 数据预处理
    processed_data = data_preprocessing(np.array([temp, gas_concentration, flame_brightness]).T)
    
    # 特征提取
    temp_feature, gas_concentration_feature, flame_brightness_feature = feature_extraction(processed_data)
    
    print("Temperature Feature:", temp_feature)
    print("Gas Concentration Feature:", gas_concentration_feature)
    print("Flame Brightness Feature:", flame_brightness_feature)

# 执行主函数
if __name__ == "__main__":
    main()

### 3.4 算法原理与数学模型

下面，我们将对上述算法的原理进行详细解释，并给出相应的数学模型。

#### 3.4.1 数据采集与预处理算法

数据采集与预处理算法的核心在于从传感器网络获取数据，并对其进行去噪和归一化处理。

1. **去噪**：去噪的目的是去除数据中的随机噪声，使数据更加平滑。常用的去噪方法包括移动平均、中值滤波等。

2. **归一化**：归一化的目的是将不同特征的数据转换到同一尺度，便于后续分析和处理。常用的归一化方法包括最小-最大归一化和标准归一化。

数学模型如下：

$$
x_{\text{normalized}} = \frac{x - x_{\text{min}}}{x_{\text{max}} - x_{\text{min}}}
$$

其中，$x$为原始数据，$x_{\text{min}}$和$x_{\text{max}}$分别为数据的最小值和最大值。

#### 3.4.2 特征提取算法

特征提取算法的核心在于从预处理后的数据中提取出对烹饪安全有重要意义的特征。

1. **温度特征**：温度是衡量烹饪过程中安全性的重要指标。通常使用温度的均值、方差等统计量作为温度特征。

2. **气体浓度特征**：气体浓度是衡量燃气泄漏风险的指标。通常使用气体浓度的均值、标准差等统计量作为气体浓度特征。

3. **火焰亮度特征**：火焰亮度是衡量火焰强度的重要指标。通常使用火焰亮度的均值、最大值等统计量作为火焰亮度特征。

数学模型如下：

$$
\text{Mean}(x) = \frac{1}{n}\sum_{i=1}^{n}x_i
$$

$$
\text{Variance}(x) = \frac{1}{n-1}\sum_{i=1}^{n}(x_i - \text{Mean}(x))^2
$$

其中，$x_i$为数据中的第$i$个值，$n$为数据总数。

#### 3.4.3 监控与预警算法

监控与预警算法的核心在于实时监控烹饪过程中的潜在风险，并在检测到异常时发出预警信号。

1. **异常检测**：异常检测的目的是识别出数据中的异常值。常用的方法包括基于阈值的异常检测、基于机器学习的异常检测等。

2. **预警信号**：预警信号的目的是提醒用户采取相应措施。常用的预警信号包括警报、邮件通知等。

数学模型如下：

$$
d(x) = \begin{cases}
0, & \text{if } x \text{ is normal} \\
1, & \text{if } x \text{ is abnormal}
\end{cases}
$$

其中，$d(x)$为预警信号，$x$为数据。
