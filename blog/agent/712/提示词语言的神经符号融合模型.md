```python
from sympy import symbols, Eq

def create_symbolic_logic_tree(question):
    x, y = symbols('x y')
    if 'and' in question:
        # 处理包含 "and" 的情况
        left, right = question.split('and')
        equation = Eq(x, 1) & Eq(y, 1)
    elif 'or' in question:
        # 处理包含 "or" 的情况
        left, right = question.split('or')
        equation = Eq(x, 1) | Eq(y, 1)
    else:
        # 处理其他情况
        equation = Eq(x, 1)
    return equation
```
