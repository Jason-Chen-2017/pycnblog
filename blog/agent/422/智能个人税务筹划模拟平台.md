```python
import pandas as pd
from sklearn.linear_model import LinearRegression

# 数据收集
def collect_data():
    data = pd.read_csv('financial_data.csv')
    return data

# 数据处理
def process_data(data):
    # 数据清洗和整理
    processed_data = data[['income', 'expense', 'investment']]
    return processed_data

# 政策应用
def apply_policy(processed_data, tax_policy):
    # 应用税种、税率等政策
    # 这里只是一个简化的示例，实际中可能需要更复杂的计算
    tax_adjusted_data = processed_data * tax_policy['rate']
    return tax_adjusted_data

# 方案优化
def optimize_solution(tax_adjusted_data):
    # 使用线性回归进行优化
    model = LinearRegression()
    model.fit(tax_adjusted_data[['income']], tax_adjusted_data['expense'])
    optimized_solution = model.predict(tax_adjusted_data[['income']])
    return optimized_solution

# 结果展示
def display_results(optimized_solution):
    print("Optimized Tax Plan:")
    print(optimized_solution)

# 主函数
def main():
    data = collect_data()
    processed_data = process_data(data)
    tax_policy = {'rate': 0.3}  # 假设税率为30%
    tax_adjusted_data = apply_policy(processed_data, tax_policy)
    optimized_solution = optimize_solution(tax_adjusted_data)
    display_results(optimized_solution)

if __name__ == '__main__':
    main()
```

5. **线性回归模型**：

   \[ E = aI + bR + c \]

   \[ I_{税后} = 150,000 \times (1 - 0.2) = 120,000 \]

   \[ Z = (150,000 - 80,000 - 20,000) \times 0.2 = 18,400 \]

5. **使用线性回归模型预测支出**：

   假设我们得到以下回归模型参数：

   \( a = 0.7 \)，\( b = 0.15 \)，\( c = 0 \)

   \[ E = 0.7I + 0.15R + 0 \]

   \[ E = 0.7 \times 150,000 + 0.15 \times 20,000 + 0 = 103,000 \]
