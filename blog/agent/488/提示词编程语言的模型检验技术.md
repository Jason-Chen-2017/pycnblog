```python
# 导入必要的库
import re

# 提示词解析函数
def parse_prompt(prompt):
    if re.match(r'^\d+$', prompt):
        return '数字', int(prompt)
    elif re.match(r'^[a-zA-Z]+$', prompt):
        return '文本', prompt
    else:
        raise ValueError("Invalid prompt")

# 执行操作函数
def execute_action(action, value):
    if action == '数字':
        return value * 2
    elif action == '文本':
        return value.lower()
    else:
        raise ValueError("Invalid action")

# 主函数
def main():
    prompt = input("请输入提示词：")
    action, value = parse_prompt(prompt)
    result = execute_action(action, value)
    print("输出结果：", result)

if __name__ == "__main__":
    main()
```

### 6.2 系统核心实现

在本节中，我们将实现系统核心功能，包括用户接口、API服务、数字操作服务和文本操作服务。以下是核心实现的源代码：

```python
from flask import Flask, request, jsonify
import re

app = Flask(__name__)

# 数字操作服务
class NumberActionService:
    def perform_action(self, value):
        return value * 2

# 文本操作服务
class TextActionService:
    def perform_action(self, value):
        return value.lower()

# 用户接口
class UserInterface:
    def receive_prompt(self):
        prompt = input("请输入提示词：")
        return prompt

    def display_result(self, result):
        print("输出结果：", result)

# API服务
class ApiService:
    def parse_prompt(self, prompt):
        if re.match(r'^\d+$', prompt):
            return '数字', int(prompt)
        elif re.match(r'^[a-zA-Z]+$', prompt):
            return '文本', prompt
        else:
            raise ValueError("Invalid prompt")

    def execute_action(self, action, value):
        if action == '数字':
            return self.number_action_service.perform_action(value)
        elif action == '文本':
            return self.text_action_service.perform_action(value)
        else:
            raise ValueError("Invalid action")

# 初始化服务
number_action_service = NumberActionService()
text_action_service = TextActionService()

@app.route('/api/execute', methods=['POST'])
def execute_action():
    data = request.get_json()
    action, value = ApiService().parse_prompt(data['prompt'])
    result = ApiService().execute_action(action, value)
    return jsonify(result=result)

if __name__ == '__main__':
    app.run(debug=True)
```
