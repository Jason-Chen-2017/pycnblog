```python
from flask import Flask, request, jsonify
from flask_sqlalchemy import SQLAlchemy
from datetime import datetime

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'mysql://username:password@localhost/digital_diet'
db = SQLAlchemy(app)

class User(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(255), nullable=False)
    email = db.Column(db.String(255), unique=True, nullable=False)
    password = db.Column(db.String(255), nullable=False)

class DietPlan(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    user_id = db.Column(db.Integer, nullable=False)
    start_time = db.Column(db.DateTime, nullable=False)
    end_time = db.Column(db.DateTime, nullable=False)
    status = db.Column(db.Enum('pending', 'in_progress', 'completed'), default='pending')
    user = db.relationship('User', backref='diet_plans')

@app.route('/register', methods=['POST'])
def register():
    data = request.get_json()
    user = User(name=data['name'], email=data['email'], password=data['password'])
    db.session.add(user)
    db.session.commit()
    return jsonify({'message': 'User registered successfully.'})

@app.route('/login', methods=['POST'])
def login():
    data = request.get_json()
    user = User.query.filter_by(email=data['email'], password=data['password']).first()
    if user:
        return jsonify({'token': 'token_value'})
    else:
        return jsonify({'error': 'Invalid credentials.'})

@app.route('/create_plan', methods=['POST'])
def create_plan():
    data = request.get_json()
    plan = DietPlan(user_id=data['user_id'], start_time=datetime.now(), end_time=datetime.now() + data['duration'])
    db.session.add(plan)
    db.session.commit()
    return jsonify({'message': 'Diet plan created successfully.'})

@app.route('/get_plans', methods=['GET'])
def get_plans():
    user_id = request.args.get('user_id')
    plans = DietPlan.query.filter_by(user_id=user_id).all()
    return jsonify({'diet_plans': [{'id': plan.id, 'start_time': plan.start_time, 'end_time': plan.end_time, 'status': plan.status} for plan in plans]})

if __name__ == '__main__':
    db.create_all()
    app.run(debug=True)
```
