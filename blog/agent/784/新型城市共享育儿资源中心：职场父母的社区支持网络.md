// backend/app.js

const express = require('express');
const mysql = require('mysql2');
const bodyParser = require('body-parser');
const cors = require('cors');

const app = express();
app.use(cors());
app.use(bodyParser.json());

// MySQL配置
const db = mysql.createPool({
  host: 'localhost',
  user: 'root',
  password: 'password',
  database: 'parent_support_center'
});

// 用户注册接口
app.post('/register', (req, res) => {
  const { username, password } = req.body;
  const sql = `INSERT INTO users (username, password) VALUES (?, ?)`;
  db.execute(sql, [username, password], (err, results) => {
    if (err) {
      res.status(500).json({ error: '注册失败' });
    } else {
      res.status(200).json({ message: '注册成功' });
    }
  });
});

// 用户登录接口
app.post('/login', (req, res) => {
  const { username, password } = req.body;
  const sql = `SELECT * FROM users WHERE username = ? AND password = ?`;
  db.execute(sql, [username, password], (err, results) => {
    if (err) {
      res.status(500).json({ error: '登录失败' });
    } else if (results.length > 0) {
      res.status(200).json({ message: '登录成功', user: results[0] });
    } else {
      res.status(401).json({ error: '用户名或密码错误' });
    }
  });
});

// 资源查询接口
app.get('/resources', (req, res) => {
  const sql = `SELECT * FROM resources`;
  db.execute(sql, [], (err, results) => {
    if (err) {
      res.status(500).json({ error: '查询失败' });
    } else {
      res.status(200).json({ resources: results });
    }
  });
});

const PORT = 3000;
app.listen(PORT, () => {
  console.log(`Server running on port ${PORT}`);
});
```

// frontend/src/App.js

import React, { useState } from 'react';
import axios from 'axios';

const API_URL = 'http://localhost:3000';

const App = () => {
  const [username, setUsername] = useState('');
  const [password, setPassword] = useState('');
  const [resources, setResources] = useState([]);

  const handleRegister = async () => {
    try {
      const response = await axios.post(`${API_URL}/register`, { username, password });
      alert(response.data.message);
    } catch (error) {
      alert(error.response.data.error);
    }
  };

  const handleLogin = async () => {
    try {
      const response = await axios.post(`${API_URL}/login`, { username, password });
      alert(response.data.message);
    } catch (error) {
      alert(error.response.data.error);
    }
  };

  const handleResources = async () => {
    try {
      const response = await axios.get(`${API_URL}/resources`);
      setResources(response.data.resources);
    } catch (error) {
      alert(error.response.data.error);
    }
  };

  return (
    <div>
      <h1>新型城市共享育儿资源中心</h1>
      <div>
        <label>用户名：</label>
        <input type="text" value={username} onChange={(e) => setUsername(e.target.value)} />
      </div>
      <div>
        <label>密码：</label>
        <input type="password" value={password} onChange={(e) => setPassword(e.target.value)} />
      </div>
      <button onClick={handleRegister}>注册</button>
      <button onClick={handleLogin}>登录</button>
      <button onClick={handleResources}>查询资源</button>
      {resources.length > 0 && (
        <ul>
          {resources.map((resource) => (
            <li key={resource.id}>{resource.name}</li>
          ))}
        </ul>
      )}
    </div>
  );
};

export default App;
```

#### 7.5.2 数学模型

以下是一个简单的协同过滤推荐算法的数学模型：

1. **用户相似度计算**：
   $$ 
   sim(u_i, u_j) = \frac{q \cdot \sum_{r \in R} r_{i,j}}{\sqrt{\sum_{r \in R} r_{i,r}^2 \cdot \sum_{r \in R} r_{j,r}^2}} 
   $$
   其中，$u_i$ 和 $u_j$ 是两个用户，$r_{i,j}$ 是用户 $u_i$ 对资源 $r_j$ 的评分，$R$ 是资源集合。

2. **资源推荐计算**：
   $$ 
   r_j^{rec} = \sum_{u \in N_i} sim(u_i, u_j) \cdot r_{u,j} 
   $$
   其中，$N_i$ 是与用户 $u_i$ 相似的其他用户集合，$r_{u,j}$ 是用户 $u$ 对资源 $j$ 的评分。

N = 3  # 资源数量

# 计算用户相似度矩阵
sim = np.zeros((N, N))
for i in range(N):
    for j in range(N):
        if i != j:
            common_elements = np.count_nonzero(R[:, [i, j]] != 0)
            num = np.dot(R[i], R[j])
