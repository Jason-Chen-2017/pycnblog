2. **二分搜索**：将列表排序后，使用二分搜索算法代替线性搜索，时间复杂度降低到 $O(\log n)$。

$$

$$
其中，$x^\mu$ 是四维时空中的坐标，$\tau$ 是固有时间，$\Gamma^\mu_{\alpha\beta}$ 是黎曼曲率张量，描述了引力场的效应。

$$

$$
其中，$\lambda$ 是固有时间参数，$u^\mu$ 是四速度。

宇宙最小时间测地线可以通过以下变分原理来求解：
$$
\min_{\gamma} \int_{t_1}^{t_2} \sqrt{-g_{\mu\nu} u^\mu u^\nu} d\lambda

其中，$\gamma$ 是测地线，$t_1$ 和 $t_2$ 是测地线上的两个时间点。

    dlon = math.radians(lon2 - lon1)

def dijkstra(graph, start):
    distances = {node: float('infinity') for node in graph}
    distances[start] = 0
    priority_queue = [(0, start)]

    while priority_queue:
        current_distance, current_node = heapq.heappop(priority_queue)

        if current_distance > distances[current_node]:
            continue

        for neighbor, weight in graph[current_node].items():
            distance = current_distance + weight

            if distance < distances[neighbor]:
                distances[neighbor] = distance
                heapq.heappush(priority_queue, (distance, neighbor))

    return distances

    'B': {'A': 1, 'C': 1, 'D': 2},
    'C': {'A': 3, 'B': 1, 'D': 1},
    'D': {'B': 2, 'C': 1}
}

#### 3.1 宇宙最小时间测地线的数学模型

宇宙最小时间测地线的数学模型可以通过测地线方程来描述。在四维时空（三个空间维度和一个时间维度）中，测地线方程可以表示为：
$$
\frac{d^2 x^\mu}{d\tau^2} + \Gamma^\mu_{\alpha\beta} \frac{dx^\alpha}{d\tau} \frac{dx^\beta}{d\tau} = 0
$$
其中，$x^\mu$ 是四维时空中的坐标，$\tau$ 是固有时间，$\Gamma^\mu_{\alpha\beta}$ 是黎曼曲率张量，描述了引力场的效应。

$$
u^\mu = g^{\mu\nu} \frac{dx^\nu}{d\lambda}
$$
其中，$\lambda$ 是固有时间参数，$u^\mu$ 是四速度。

宇宙最小时间测地线可以通过以下变分原理来求解：
$$
\min_{\gamma} \int_{t_1}^{t_2} \sqrt{-g_{\mu\nu} u^\mu u^\nu} d\lambda

其中，$\gamma$ 是测地线，$t_1$ 和 $t_2$ 是测地线上的两个时间点。

    dlon = math.radians(lon2 - lon1)

import heapq

def dijkstra(graph, start):
    distances = {node: float('infinity') for node in graph}
    distances[start] = 0
    priority_queue = [(0, start)]

    while priority_queue:
        current_distance, current_node = heapq.heappop(priority_queue)

        if current_distance > distances[current_node]:
            continue

        for neighbor, weight in graph[current_node].items():
            distance = current_distance + weight

            if distance < distances[neighbor]:
                distances[neighbor] = distance
                heapq.heappush(priority_queue, (distance, neighbor))

    return distances

    'B': {'A': 1, 'C': 1, 'D': 2},
    'C': {'A': 3, 'B': 1, 'D': 1},
    'D': {'B': 2, 'C': 1}

$$

$$

$$

其中，$\lambda$ 是固有时间参数，$u^\mu$ 是四速度。

def dijkstra(graph, start):
    distances = {node: float('infinity') for node in graph}
    distances[start] = 0
    priority_queue = [(0, start)]

    while priority_queue:
        current_distance, current_node = heapq.heappop(priority_queue)

        if current_distance > distances[current_node]:
            continue

        for neighbor, weight in graph[current_node].items():

                heapq.heappush(priority_queue, (distance, neighbor))

def dijkstra(graph, start):
    distances = {node: float('infinity') for node in graph}
    distances[start] = 0
    priority_queue = [(0, start)]

    while priority_queue:
        current_distance, current_node = heapq.heappop(priority_queue)

        if current_distance > distances[current_node]:
            continue

        for neighbor, weight in graph[current_node].items():
            distance = current_distance + weight

            if distance < distances[neighbor]:
                distances[neighbor] = distance
                heapq.heappush(priority_queue, (distance, neighbor))

    return distances

    'B': {'A': 1, 'C': 1, 'D': 2},
    'C': {'A': 3, 'B': 1, 'D': 1},

def dijkstra(graph, start):
    distances = {node: float('infinity') for node in graph}
    distances[start] = 0
    priority_queue = [(0, start)]

    while priority_queue:
        current_distance, current_node = heapq.heappop(priority_queue)

        if current_distance > distances[current_node]:
            continue

        for neighbor, weight in graph[current_node].items():
            distance = current_distance + weight

                heapq.heappush(priority_queue, (distance, neighbor))

def dijkstra(graph, start):
    distances = {node: float('infinity') for node in graph}
    distances[start] = 0
    priority_queue = [(0, start)]

    while priority_queue:
        current_distance, current_node = heapq.heappop(priority_queue)

        if current_distance > distances[current_node]:
            continue

        for neighbor, weight in graph[current_node].items():
            distance = current_distance + weight

            if distance < distances[neighbor]:
                distances[neighbor] = distance
                heapq.heappush(priority_queue, (distance, neighbor))

    return distances
