import numpy as np
import matplotlib.pyplot as plt
import mermaid

# 定义势垒函数
def potential_barrier(x, barrier_height, barrier_width):
    return barrier_height * (1 - np.exp(-x**2 / barrier_width**2))

# 定义量子隧穿函数
def quantum_tunneling(x, barrier_height, barrier_width, particle_energy):
    wave_function_left = np.sqrt(2 / barrier_width) * np.exp(-x**2 / (2 * barrier_width**2))
    wave_function_right = np.sqrt(2 / barrier_width) * np.exp((-x - barrier_width)**2 / (2 * barrier_width**2))
    
    # 计算隧穿概率
    tunneling_probability = np.abs(wave_function_left**2 + wave_function_right**2) * np.exp(-particle_energy**2 / (2 * barrier_height**2))
    
    return tunneling_probability

# 定义信号传递函数
def signal_transmission(x, barrier_height, barrier_width, particle_energy):
    tunneling_probability = quantum_tunneling(x, barrier_height, barrier_width, particle_energy)
    signal_intensity = np.sum(tunneling_probability) * np.exp(-x**2 / (2 * barrier_width**2))
    
    return signal_intensity

# 设置参数
barrier_height = 1.0
barrier_width = 1.0
particle_energy = 0.5

# 生成x坐标
x = np.linspace(-3, 3, 1000)

# 计算势垒、隧穿概率和信号传递
potential = potential_barrier(x, barrier_height, barrier_width)
tunneling_probability = quantum_tunneling(x, barrier_height, barrier_width, particle_energy)
signal_intensity = signal_transmission(x, barrier_height, barrier_width, particle_energy)

# 绘制结果
plt.figure()
plt.plot(x, potential, label='Potential Barrier')
plt.plot(x, tunneling_probability, label='Tunneling Probability')
plt.plot(x, signal_intensity, label='Signal Intensity')
plt.xlabel('Position (x)')
plt.ylabel('Value')
plt.legend()
plt.show()

# 渲染Mermaid流程图
mermaid.render('quantum_tunneling.mmd', 'quantum_tunneling.png')
