import re

def check_format(paper_text):
    # 检测字体错误
    font_error = re.findall(r'[^A-Za-z0-9\s]', paper_text)
    if font_error:
        return {'error': 'Font error found!'}

    # 检测间距错误
    space_error = re.findall(r'\s{2,}', paper_text)
    if space_error:
        return {'error': 'Space error found!'}

    # 检测引用格式错误
    citation_error = re.findall(r'[^A-Za-z0-9\(\)\[\]]', paper_text)
    if citation_error:
        return {'error': 'Citation format error found!'}

    return {'message': 'Format check passed!'}

# 测试代码
