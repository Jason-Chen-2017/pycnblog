```python
# 导入必要的库
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.metrics import accuracy_score
from keras.models import Sequential
from keras.layers import Dense, Conv1D, MaxPooling1D, Flatten
from keras.utils import to_categorical

# 加载和预处理数据
# 假设我们有一个CSV文件，包含脑电图数据和一个标签列（0表示正常，1表示癫痫）
data = pd.read_csv('EEG_data.csv')
X = data.iloc[:, :-1].values  # 特征矩阵
y = data.iloc[:, -1].values    # 标签向量

# 归一化数据
X_normalized = (X - X.mean()) / X.std()

# 切分数据集
X_train, X_test, y_train, y_test = train_test_split(X_normalized, y, test_size=0.2, random_state=42)

# 转换标签为独热编码
y_train_encoded = to_categorical(y_train)
y_test_encoded = to_categorical(y_test)

# 构建神经网络模型
model = Sequential()
model.add(Conv1D filters=64, kernel_size=3, activation='relu', input_shape=(X_train.shape[1], X_train.shape[2]))
model.add(MaxPooling1D pool_size=2))
model.add(Flatten())
model.add(Dense units=50, activation='relu'))
model.add(Dense units=2, activation='softmax'))

# 编译模型
model.compile(optimizer='adam', loss='categorical_crossentropy', metrics=['accuracy'])

# 训练模型
model.fit(X_train, y_train_encoded, epochs=10, batch_size=32, validation_split=0.1)

# 评估模型
predictions = model.predict(X_test)
predicted_labels = np.argmax(predictions, axis=1)
accuracy = accuracy_score(y_test, predicted_labels)
print(f'Accuracy: {accuracy:.2f}')

# 保存模型
model.save('epilepsy_diagnosis_model.h5')
```
