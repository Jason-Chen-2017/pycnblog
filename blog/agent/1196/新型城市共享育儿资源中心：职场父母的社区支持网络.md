# 共享育儿资源中心算法原理示例
def resource_allocation(requests):
    """
    资源分配算法
    :param requests: 职场父母的育儿需求列表
    :return: 资源分配结果
    """
    resources = ["儿童游乐设施", "母婴室", "育儿知识库"]
    allocation = {resource: [] for resource in resources}
    
    for request in requests:
        for resource in resources:
            if request["resource"] == resource:
                allocation[resource].append(request["parent_id"])
                break
                
    return allocation

# 示例数据
requests = [
    {"parent_id": 1, "resource": "儿童游乐设施"},
    {"parent_id": 2, "resource": "母婴室"},
    {"parent_id": 3, "resource": "育儿知识库"}
]

# 执行资源分配算法
result = resource_allocation(requests)
print(result)
