def build_inverted_index(documents):
    inverted_index = {}
    for doc in documents:
        terms = tokenize(doc)
        for term in terms:
            if term in inverted_index:
                inverted_index[term].append(doc)
            else:
                inverted_index[term] = [doc]
    return inverted_index

# Python代码示例：查询处理
def search_inverted_index(inverted_index, query):
    query_results = set()
    for term in tokenize(query):
        if term in inverted_index:
            query_results.update(inverted_index[term])
    return query_results

# LaTeX公式：倒排索引定义

Inverted\ Index = \{(t, \{d_1, d_2, ..., d_n\}) | t \in Terms, d_i \in Documents\}
$$

#### 5.2 系统核心实现源代码

以下是系统核心实现的部分源代码，包括倒排索引的构建和查询处理：

```python
# 导入相关库
import numpy as np
from sklearn.feature_extraction.text import CountVectorizer
from sklearn.metrics.pairwise import cosine_similarity
import json

# 倒排索引构建函数
def build_inverted_index(documents):
    vectorizer = CountVectorizer()
    term_doc_matrix = vectorizer.fit_transform(documents)
    term_index = vectorizer.get_feature_names_out()
    inverted_index = {}
    for i, doc in enumerate(documents):
        term_ids = term_doc_matrix[i].nonzero()[1]
        inverted_index[doc] = [term_index[tid] for tid in term_ids]
    return inverted_index

# 查询处理函数
def search_inverted_index(inverted_index, query):
    query_vector = CountVectorizer().transform([query])
    query_terms = query_vector.toarray().flatten()
    results = []
    for doc, terms in inverted_index.items():
        doc_vector = CountVectorizer().transform([doc])
        doc_terms = doc_vector.toarray().flatten()
        similarity = cosine_similarity(doc_terms.reshape(1, -1), query_terms.reshape(1, -1))
        if similarity > 0.5:
            results.append((doc, similarity[0][0]))
    results.sort(key=lambda x: x[1], reverse=True)
    return results

# 测试代码
documents = ["我爱编程，编程使我快乐。", "编程是一种艺术，它能解决复杂问题。", "学习编程，从基础开始。"]
inverted_index = build_inverted_index(documents)
query = "编程艺术"
results = search_inverted_index(inverted_index, query)
print(results)
```

#### 5.3 代码应用解读与分析

上述代码首先定义了两个核心函数：`build_inverted_index`用于构建倒排索引，`search_inverted_index`用于处理查询并返回结果。

1. **倒排索引构建**：
   - 使用`CountVectorizer`将文档转换为倒排索引，该向量器能够将文本转换为词频矩阵。
   - 对于每个文档，提取其词频矩阵的非零元素，即对应的词汇，构建倒排索引。

2. **查询处理**：
   - 将查询文本转换为词频向量。
   - 对于每个文档，计算其与查询文本的余弦相似度，如果相似度超过某个阈值（例如0.5），则认为该文档与查询相关。
