def analyze_galaxy_data():
    connection = pymysql.connect(host='localhost', user='universe_user', password='password', database='universe_db')
    cursor = connection.cursor()
    
    cursor.execute("SELECT * FROM galaxy_stars")
    stars = cursor.fetchall()
    
    # 计算恒星间的距离和速度差
    distances = []
    velocity_diffs = []
    for i in range(len(stars)):
        for j in range(i + 1, len(stars)):
            distance = np.linalg.norm(np.array(stars[i]['position']) - np.array(stars[j]['position']))
            velocity_diff = np.linalg.norm(np.array(stars[i]['velocity']) - np.array(stars[j]['velocity']))
            distances.append(distance)
            velocity_diffs.append(velocity_diff)
    
    # 预测碰撞事件
    collision_threshold = 100  # 单位：光年
    collision_stars = []
    for i in range(len(distances)):
        if distances[i] < collision_threshold and velocity_diffs[i] < 0:
            collision_stars.append(stars[i])
    
    return collision_stars

collision_stars = analyze_galaxy_data()

    print("No predicted collision stars.")
