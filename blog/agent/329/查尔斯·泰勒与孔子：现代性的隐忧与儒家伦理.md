```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# 伦理分析模块
class EthicsAnalysis:
    def __init__(self, dilemma):
        self.dilemma = dilemma
    
    def analyze(self):
        # 在这里，我们可以使用自然语言处理技术来分析道德困境
        # 例如，使用词频分析、主题建模等方法
        # 以下是一个简化的示例
        words = self.dilemma.split()
        word_freq = pd.Series(words).value_counts()
        return word_freq

# 数据可视化模块
class VisualizationModule:
    def __init__(self, word_freq):
        self.word_freq = word_freq
    
    def generate_visualization(self):
        # 使用matplotlib绘制词频直方图
        plt.bar(word_freq.index, word_freq.values)
        plt.xlabel('Words')
        plt.ylabel('Frequency')
        plt.title('Word Frequency in Moral Dilemma')
        plt.show()

# 用户界面
def main():
    dilemma = input("请输入您的道德困境：")
    analysis = EthicsAnalysis(dilemma)
    word_freq = analysis.analyze()
    visualization = VisualizationModule(word_freq)
    visualization.generate_visualization()

if __name__ == "__main__":
    main()
```
