  f(\theta, x_i) = \sigma(\theta^T x_i)

  $$
  f(\theta, x_i) = \phi(\theta^T x_i)

#### 系统核心实现源代码

```python
import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split
from sklearn.metrics import accuracy_score
import cv2

# 数据采集
def collect_data(sensor_data, image_data):
    # 传感器数据
    data = pd.DataFrame(sensor_data)
    # 图像数据
    images = [cv2.imread(img) for img in image_data]
    # 特征提取
    features = extract_features(images)
    # 合并数据
    data = pd.concat([data, pd.DataFrame(features)], axis=1)
    return data

# 特征提取
def extract_features(images):
    features = []
    for image in images:
        # 使用OpenCV进行图像处理
        gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
        # 计算图像特征
        feature = np.mean(gray)
        features.append(feature)
    return features

# 模型训练
def train_model(data):
    # 划分训练集和测试集
    X_train, X_test, y_train, y_test = train_test_split(data, test_size=0.2, random_state=42)
    # 使用随机森林分类器进行训练
    model = RandomForestClassifier(n_estimators=100)
    model.fit(X_train, y_train)
    return model

# 预测和预警
def predict(model, new_data):
    # 预测病虫害发生概率
    probability = model.predict_proba(new_data)[0][1]
    # 判断是否需要预警
    if probability > 0.5:
        print("预警：农作物可能存在病虫害，请及时防治。")
    else:
        print("正常：农作物目前没有病虫害发生。")

# 实际应用
if __name__ == "__main__":
    # 传感器数据
    sensor_data = [[22, 27, 800], [23, 28, 800], [21, 26, 800]]
    # 图像数据
    image_data = ["leaf1.jpg", "leaf2.jpg", "leaf3.jpg"]
    # 数据采集
    data = collect_data(sensor_data, image_data)
    # 模型训练
    model = train_model(data)
    # 预测和预警
    predict(model, data)
```

2. **特征提取**：`extract_features`函数使用OpenCV对图像进行处理，提取出与病虫害相关的特征。

3. **模型训练**：`train_model`函数使用随机森林分类器对数据进行训练。

4. **预测和预警**：`predict`函数根据训练好的模型预测新数据的病虫害发生概率，并生成预警信息。
