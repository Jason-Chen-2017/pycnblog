import numpy as np
from scipy.signal import butter, lfilter

def butter_bandpass_filter(data, lowcut, highcut, fs, order=4):
    nyq = 0.5 * fs
    low = lowcut / nyq
    high = highcut / nyq
    b, a = butter(order, [low, high], btype='band')
    y = lfilter(b, a, data)
    return y

# 示例数据
data = np.random.rand(1000)
fs = 100  # 采样频率
lowcut = 1
highcut = 50
order = 4

# 应用带通滤波器

# 输出结果
print(filtered_data)
