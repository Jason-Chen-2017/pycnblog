from flask import Flask, request, jsonify
from goal import Goal
from person import Person
from society import Society
from culture import Culture

app = Flask(__name__)

@app.route('/analyze', methods=['POST'])
def analyze():
    user_data = request.json
    person = Person(user_data['name'], user_data['age'], user_data['background'])
    society = Society()
    culture = Culture()

    goal = person.determine_goal()

    return jsonify({'goal': goal.to_dict(), 'values': values.to_dict(), 'contribution': contribution.to_dict()})

from flask import Flask, request, jsonify
from goal import Goal
from person import Person
from society import Society
from culture import Culture

app = Flask(__name__)

@app.route('/analyze', methods=['POST'])
def analyze():
    user_data = request.json
    person = Person(user_data['name'], user_data['age'], user_data['background'])
    society = Society()
    culture = Culture()

    goal = person.determine_goal()
    values = person.determine_values(culture)
    contribution = person.contribute_to_society(society, values)

    return jsonify({'goal': goal.to_dict(), 'values': values.to_dict(), 'contribution': contribution.to_dict()})

if __name__ == '__main__':
    app.run(debug=True)
