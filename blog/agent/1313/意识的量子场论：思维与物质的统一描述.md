```python
# quantum_computer.py
from qiskit import QuantumCircuit, execute, Aer
from qiskit.visualization import plot_bloch_vector
from numpy import pi

class QuantumComputer:
    def __init__(self):
        self.qc = QuantumCircuit(2)

    def apply_quantum纠缠(self):
        self.qc.h(0)
        self.qc.cx(0, 1)

    def execute_quantum计算(self):
        self.qc.h(0)
        self.qc.s(0)
        self.qc.cx(0, 1)
        self.qc.s(0)
        self.qc.h(0)

    def measure(self):
        self.qc.measure_all()

    def run(self):
        sim = Aer.get_backend("qasm_simulator")
        job = execute(self.qc, sim, shots=1000)
        result = job.result()
        return result.get_counts(self.qc)

# user_interface.py
from quantum_computer import QuantumComputer
import matplotlib.pyplot as plt

class UserInterface:

        self.qc = QuantumComputer()

    def run_experiment(self):

        self.qc.execute_quantum计算()
        self.qc.measure()
        result = self.qc.run()
        print("Results:", result)

    def plot_bloch_vector(self):
        state = self.qc.qc.state()
        plot_bloch_vector(state)
        plt.show()
```
