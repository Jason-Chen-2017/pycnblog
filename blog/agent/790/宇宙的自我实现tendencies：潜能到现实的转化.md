```python
from flask import Flask, request, jsonify
import sqlite3
import numpy as np

app = Flask(__name__)

# 数据库连接
def get_connection():
    conn = sqlite3.connect('data.db')
    return conn

# 存储参数
@app.route('/input', methods=['POST'])
def input_data():
    data = request.get_json()
    conn = get_connection()
    cursor = conn.cursor()
    cursor.execute('''CREATE TABLE IF NOT EXISTS parameters (
                        universe_volume REAL,
                        star_system_mass REAL,
                        black_hole_mass REAL,
                        life_appearance_probability REAL,
                        self_realization_extent REAL)''')
    cursor.execute("INSERT INTO parameters VALUES (?, ?, ?, ?, ?)",
                   (data['universe_volume'], data['star_system_mass'],
                    data['black_hole_mass'], data['life_appearance_probability'],
                    data['self_realization_extent']))
    conn.commit()
    conn.close()
    return jsonify({"status": "success"})

# 执行算法
@app.route('/algorithm', methods=['GET'])
def execute_algorithm():
    time = request.args.get('time')
    conn = get_connection()
    cursor = conn.cursor()
    cursor.execute("SELECT * FROM parameters")
    param = cursor.fetchone()
    conn.close()

    # 宇宙大爆炸
    V0, M0, T0, P0, R0 = param
    k = 0.1
    r = 0.05
    s = 0.01
    p = 0.001
    q = 0.001

    V_t = V0 * np.exp(k * float(time))
    M_t = M0 * np.exp(r * float(time))
    T_t = T0 * np.exp(-s * float(time))
    P_t = P0 * np.exp(p * float(time))
    R_t = R0 * np.exp(q * float(time))

    return jsonify({
        "time": time,
        "universe_volume": V_t,
        "star_system_mass": M_t,
        "black_hole_mass": T_t,
        "life_appearance_probability": P_t,
        "self_realization_extent": R_t
    })

# 主函数
if __name__ == '__main__':
    app.run(debug=True)
```
