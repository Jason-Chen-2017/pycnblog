$$
f(x) = \frac{1}{n} \sum_{i=1}^{n} \frac{1}{n} \sum_{j=1}^{n} \frac{w_{ij}}{r_i + r_j}

1. 任务分配算法实现：

```python
import numpy as np
from sklearn.cluster import KMeans

def task_assignment(tasks, agents):
    # 构建目标函数
    def objective(x):
        total_cost = 0
        for i in range(len(tasks)):
            for j in range(len(agents)):
                total_cost += x[i][j] / (agents[j] + tasks[i])
        return total_cost

    # 选择优化算法
    optimizer = KMeans(n_clusters=len(agents), init='k-means++', max_iter=300)

    # 运行优化算法
    optimizer.fit(tasks)

    # 分配任务
    assignments = optimizer.predict(tasks)
    for i in range(len(assignments)):
        agents[assignments[i]] += tasks[i]

    return assignments
```

2. 信息传递算法实现：

```python
import tensorflow as tf
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Dense, LSTM

def information_transmission(data, actions):
    # 构建状态空间和动作空间
    state_space = tf.keras.utils.to_categorical(data, num_classes=10)
    action_space = tf.keras.utils.to_categorical(actions, num_classes=5)

    # 设计奖励机制
    rewards = []

    # 选取深度强化学习算法
    model = Sequential([
        LSTM(50, activation='relu', input_shape=(state_space.shape[1],)),
        Dense(50, activation='relu'),
        Dense(1, activation='sigmoid')
    ])

    model.compile(optimizer='adam', loss='binary_crossentropy', metrics=['accuracy'])

    # 训练模型
    for i in range(len(state_space)):
        model.fit(state_space[i], action_space[i], epochs=10, batch_size=1)

        # 计算奖励
        reward = action_space[i].dot(data[i])
        rewards.append(reward)

    return rewards
```

import numpy as np
from sklearn.naive_bayes import GaussianNB

def decision_making(data):
    # 构建贝叶斯网络
    features = data[:, :5]
    labels = data[:, 5]

    # 训练模型
    model = GaussianNB()
    model.fit(features, labels)

    # 预测
    predictions = model.predict(features)

    # 计算概率
    probabilities = model.predict_proba(features)
