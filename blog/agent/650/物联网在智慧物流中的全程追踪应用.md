Dijkstra算法的基本思想是从起点开始，逐步扩展到其他节点，计算到每个节点的最短路径。以下是Dijkstra算法的Python代码实现：

```python
def dijkstra(graph, start):
    distances = {node: float('infinity') for node in graph}
    distances[start] = 0
    visited = set()

    while True:
        next_node = min((node, distance) for node, distance in distances.items() if node not in visited)
        if next_node is None:
            break
        visited.add(next_node)

        for neighbor, weight in graph[next_node].items():
            old_distance = distances[neighbor]
            new_distance = distances[next_node] + weight
            if new_distance < old_distance:
                distances[neighbor] = new_distance

    return distances
```

K-means算法的基本思想是将数据集划分为K个簇，每个簇由一个聚类中心表示。以下是K-means算法的Python代码实现：

```python
import numpy as np

def k_means(data, k, max_iterations):
    centroids = data[np.random.choice(data.shape[0], k, replace=False)]
    for _ in range(max_iterations):
        distances = np.linalg.norm(data - centroids, axis=1)
        labels = np.argmin(distances, axis=1)
        new_centroids = np.array([data[labels == i].mean(axis=0) for i in range(k)])

        if np.all(centroids == new_centroids):
            break

        centroids = new_centroids

    return centroids, labels
```

```python
import numpy as np

def linear_regression(data, labels, learning_rate, max_iterations):
    weights = np.random.rand(data.shape[1])
    bias = 0

    for _ in range(max_iterations):
        predictions = np.dot(data, weights) + bias
        loss = np.mean((predictions - labels) ** 2)

        d_weights = 2 * (predictions - labels) * data
        d_bias = 2 * (predictions - labels)

        weights -= learning_rate * d_weights
        bias -= learning_rate * d_bias

    return weights, bias
```
