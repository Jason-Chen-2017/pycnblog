from pymongo import MongoClient

db = client['digital_twin']
collection = db['status']

def sync_state(df):
    # 更新虚拟模型状态
    for index, row in df.iterrows():
        collection.update_one({'id': row['id']}, {'$set': row.to_dict()}, upsert=True)

# 性能预测模块
from sklearn.linear_model import LinearRegression

def predict_performance(df):
    # 预测性能
    X = df[['timestamp', 'speed']]
    y = df['fuel_consumption']
    model = LinearRegression()
    model.fit(X, y)
    future_x = pd.DataFrame({'timestamp': [df['timestamp'].iloc[-1] + 1]})
    predicted_y = model.predict(future_x)
    return predicted_y

# 故障诊断模块
def diagnose_fault(df):
    # 诊断故障
    if df['temperature'] > 100:
        return '高温故障'
    elif df['pressure'] < 0.8:
        return '低压故障'
    else:
        return '正常'

def provide_maintenance_advice(df):
    # 提供维护建议
    if df['remaining_fuel'] < 10:
        return '加油'
    elif df['vibration'] > 5:
