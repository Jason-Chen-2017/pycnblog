import numpy as np

def accelerate_particle(particle):
    particle['velocity'] *= 2

def collision(particle1, particle2):
    # 计算碰撞后的速度
    new_velocity1 = particle1['velocity'] + particle2['velocity']
    new_velocity2 = -particle1['velocity'] + particle2['velocity']
    
    # 更新粒子速度
    particle1['velocity'] = new_velocity1
    particle2['velocity'] = new_velocity2

def simulate_collision(particle1, particle2, iterations):
    for _ in range(iterations):
        accelerate_particle(particle1)
        accelerate_particle(particle2)
        collision(particle1, particle2)

particle1 = {'velocity': 1.0}
particle2 = {'velocity': 1.0}

# 模拟粒子碰撞
simulate_collision(particle1, particle2, 10)

- `collision`函数用于模拟粒子碰撞的过程，计算碰撞后的速度。
- `simulate_collision`函数用于模拟多次粒子碰撞，重复执行加速和碰撞操作。
