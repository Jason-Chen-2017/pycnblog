#### 6.2.1 音乐情感分析核心代码

```python
import librosa
import numpy as np
from tensorflow.keras.models import load_model

def preprocess_audio(audio_path):
    audio, sample_rate = librosa.load(audio_path)
    mfccs = librosa.feature.mfcc(y=audio, sr=sample_rate, n_mfcc=13)
    return mfccs

def predict_emotion(mfccs):
    emotion_model = load_model('emotion_model.h5')
    emotion_prediction = emotion_model.predict(mfccs)
    return emotion_prediction

# 示例
audio_path = 'example.wav'
mfccs = preprocess_audio(audio_path)
emotion_prediction = predict_emotion(mfccs)
print(f'情感预测结果：{emotion_prediction}')
```

#### 6.2.2 情感识别核心代码

```python
import nltk
from nltk.sentiment import SentimentIntensityAnalyzer

def preprocess_text(text):
    tokens = nltk.word_tokenize(text)
    return tokens

def analyze_text(text):
    sia = SentimentIntensityAnalyzer()
    sentiment = sia.polarity_scores(text)
    return sentiment

# 示例
text = '我喜欢这首歌'
tokens = preprocess_text(text)
sentiment = analyze_text(' '.join(tokens))
print(f'情感分析结果：{sentiment}')
```
