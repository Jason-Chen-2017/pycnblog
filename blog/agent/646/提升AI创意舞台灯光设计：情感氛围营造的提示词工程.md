import tensorflow as tf
from tensorflow.keras.models import Sequential

# 生成器模型
generator = Sequential([
    Flatten(input_shape=(28, 28)),
    Dense(128),
    Dense(784, activation='tanh'),
    Reshape((28, 28, 1))
