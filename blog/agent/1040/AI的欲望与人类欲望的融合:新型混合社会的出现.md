```python
            weights[i] += learning_rate * error * data[i-1]

    return weights

# 预测未来价格
def predict_price(weights, future_data):
    predictions = [weights[0]]
    for i in range(1, len(future_data)):
        prediction = weights[i] * future_data[i-1]
        predictions.append(prediction)
    return predictions

# 测试数据
train_data = load_stock_data('train_data.txt')
future_data = load_stock_data('future_data.txt')

# 训练模型
model_weights = train_model(train_data)

# 预测未来价格
predictions = predict_price(model_weights, future_data)

# 绘制预测结果
plt.plot(predictions)
plt.xlabel('时间')
plt.ylabel('股票价格')
plt.title('股票价格预测')
plt.show()
```
