**GPT系列模型**

GPT（Generative Pre-trained Transformer）系列模型是OpenAI开发的一组基于Transformer架构的预训练模型。以下是一个使用PyTorch实现GPT模型的简单示例：

```python
import torch
import torch.nn as nn
import torch.optim as optim

class GPT(nn.Module):
    def __init__(self, vocab_size, d_model, n_layers, p):
        super(GPT, self).__init__()
        self.embedding = nn.Embedding(vocab_size, d_model)
        self.transformer = nn.Transformer(d_model, n_layers)
        self.fc = nn.Linear(d_model, vocab_size)
        self.dropout = nn.Dropout(p)

    def forward(self, x):
        x = self.dropout(self.embedding(x))
        x = self.transformer(x)
        x = self.fc(x)
        return x

# 模型参数设置
vocab_size = 10000
d_model = 512
n_layers = 2
p = 0.1

# 实例化模型
gpt = GPT(vocab_size, d_model, n_layers, p)

# 模型优化
optimizer = optim.Adam(gpt.parameters(), lr=0.001)
criterion = nn.CrossEntropyLoss()

# 训练模型
for epoch in range(10):
    for batch in data_loader:
        x, y = batch
        optimizer.zero_grad()
        output = gpt(x)
        loss = criterion(output, y)
        loss.backward()
        optimizer.step()

# 模型保存
torch.save(gpt.state_dict(), 'gpt_model.pth')
```
