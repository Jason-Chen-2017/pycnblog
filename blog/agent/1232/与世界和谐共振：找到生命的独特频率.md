```python
import numpy as np
import matplotlib.pyplot as plt

# 行为监测模块
class BehaviorMonitoringSystem:
    def __init__(self):
        self.movement_data = []
        self.sleep_quality = []

    def monitor_movement(self, data):
        self.movement_data.append(data)

    def monitor_sleep_quality(self, quality):
        self.sleep_quality.append(quality)

    def generate_report(self):
        movement_avg = np.mean(self.movement_data)
        sleep_avg = np.mean(self.sleep_quality)
        return f"运动数据平均值：{movement_avg}，睡眠质量平均值：{sleep_avg}"

# 环境调整模块
class EnvironmentalAdjustmentSystem:
    def __init__(self):
        self.light_intensity = 0
        self.temperature = 0
        self.sound_level = 0

    def adjust_light(self, intensity):
        self.light_intensity = intensity

    def adjust_temperature(self, temp):
        self.temperature = temp

    def adjust_sound(self, level):
        self.sound_level = level

    def generate_adjustment_report(self):
        return f"光线强度：{self.light_intensity}，温度：{self.temperature}℃，声音强度：{self.sound_level}"

# 心理平衡模块
class PsychologicalBalancingSystem:
    def __init__(self):
        self.behavior_analysis = []
        self.psychological_advice = []

    def analyze_behavior(self, behavior_data):
        self.behavior_analysis.append(behavior_data)

    def provide_psychological_advice(self, advice):
        self.psychological_advice.append(advice)

    def generate_analysis_report(self):
        return f"行为分析结果：{self.behavior_analysis}，心理建议：{self.psychological_advice}"

# 主函数
def main():
    # 初始化系统
    behavior_monitoring_system = BehaviorMonitoringSystem()
    environmental_adjustment_system = EnvironmentalAdjustmentSystem()
    psychological_balancing_system = PsychologicalBalancingSystem()

    # 监测行为
    behavior_monitoring_system.monitor_movement(100)
    behavior_monitoring_system.monitor_sleep_quality(0.8)

    # 调整环境
    environmental_adjustment_system.adjust_light(70)
    environmental_adjustment_system.adjust_temperature(22)
    environmental_adjustment_system.adjust_sound(30)

    # 分析行为
    psychological_balancing_system.analyze_behavior(["运动较多", "睡眠质量良好"])
    psychological_balancing_system.provide_psychological_advice("保持良好的生活习惯，继续努力！")

    # 输出报告
    print(behavior_monitoring_system.generate_report())
    print(environmental_adjustment_system.generate_adjustment_report())
    print(psychological_balancing_system.generate_analysis_report())

if __name__ == "__main__":
    main()
```
