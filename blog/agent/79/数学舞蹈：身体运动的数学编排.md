# 输入舞蹈动作

# 计算对称性
def calculate_symmetry(moves):
    symmetric_moves = []
    for move in moves:
        if move == moves[-1]:
            symmetric_moves.append(move)
    return symmetric_moves

symmetric_moves = calculate_symmetry(dance_moves)

# 输出结果
