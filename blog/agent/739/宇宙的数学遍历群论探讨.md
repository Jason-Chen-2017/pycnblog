               state = self.apply_group_operation(state)

import numpy as np

# 初始化参数

# 建立群结构
def group_operation(state):
    theta = np.random.uniform(0, 2 * np.pi)
    rotation_matrix = np.array([[np.cos(theta), -np.sin(theta)],
                                 [np.sin(theta), np.cos(theta)]])
    return np.dot(rotation_matrix, state)

# 计算群操作
def iterate_group(state, group_operation, N, timestep):
    for _ in range(N):
        state = group_operation(state)
        if np.linalg.norm(state) > 1.0:
