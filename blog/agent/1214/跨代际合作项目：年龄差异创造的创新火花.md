```python
def calculate_age_distribution(team):
    """
    计算团队年龄分布
    :param team: 团队成员列表
    :return: 年龄分布列表
    """
    age_distribution = []
    for member in team:
        age = member['age']
        age_distribution.append(age)
    return age_distribution

def detect_communication_issues(team):
    """
    检测团队沟通问题
    :param team: 团队成员列表
    :return: 沟通问题列表
    """
    communication_issues = []
    for i in range(len(team)):
        for j in range(i + 1, len(team)):
            if team[i]['communication_level'] < team[j]['communication_level']:
                communication_issues.append((team[i]['name'], team[j]['name']))
    return communication_issues

def calculate_innovation_score(tasks):
    """
    计算创新得分
    :param tasks: 任务列表
    :return: 创新得分
    """
    innovation_score = 0
    for task in tasks:
        if task['innovation_level'] == 'high':
            innovation_score += 1
    return innovation_score
```
