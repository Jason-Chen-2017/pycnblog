```python
import numpy as np
import matplotlib.pyplot as plt
from database import Database

class UniverseSimulation:
    def __init__(self, num_objects, scale):
        self.positions = np.random.uniform(-scale, scale, size=(num_objects, 2))
        self.velocities = np.random.uniform(-1, 1, size=(num_objects, 2))
        self.db = Database()

    def calculate_synchrony(self):
        synchrony_scores = []
        for i in range(len(self.positions)):
            sync_score = 0
            for j in range(len(self.positions)):
                if i != j:
                    distance = np.linalg.norm(self.positions[i] - self.positions[j])
                    speed_diff = np.linalg.norm(self.velocities[i] - self.velocities[j])
                    sync_score += 1 / (distance + speed_diff)
            synchrony_scores.append(sync_score / (len(self.positions) - 1))
        return synchrony_scores

    def adjust_positions_velocities(self, synchrony_scores):
        for i in range(len(self.positions)):
            if synchrony_scores[i] < 0.5:
                self.positions[i] += self.velocities[i] * 0.01
                self.velocities[i] += np.random.normal(0, 0.1, size=2)
            else:
                self.positions[i] -= self.velocities[i] * 0.01
                self.velocities[i] -= np.random.normal(0, 0.1, size=2)
        return self.positions, self.velocities

    def run_simulation(self, max_iterations):
        synchrony_scores = []
        for _ in range(max_iterations):
            synchrony_scores.append(self.calculate_synchrony())
            self.positions, self.velocities = self.adjust_positions_velocities(synchrony_scores)
            self.db.save_state(self.positions, self.velocities)
        
        self.db.save_final_state(self.positions, self.velocities)
        self.plot_synchrony(synchrony_scores)

    def plot_synchrony(self, synchrony_scores):
        plt.plot(synchrony_scores)
        plt.xlabel('Iterations')
        plt.ylabel('Synchrony Score')
        plt.title('Universe Synchronization Simulation')
        plt.show()

if __name__ == "__main__":
    simulation = UniverseSimulation(50, 10)
    simulation.run_simulation(100)
```
