```python
import datetime
import os
import json

MEDITATION_FILE = 'meditations.json'

def load_meditations():
    if os.path.exists(MEDITATION_FILE):
        with open(MEDITATION_FILE, 'r') as f:
            return json.load(f)
    else:
        return []

def save_meditations(meditations):
    with open(MEDITATION_FILE, 'w') as f:
        json.dump(meditations, f, indent=4)

def start_meditation():
    meditations = load_meditations()
    start_time = datetime.datetime.now()
    meditations.append({'start': start_time})
    save_meditations(meditations)
    print(f"Meditation started at {start_time}.")

def end_meditation():
    meditations = load_meditations()
    end_time = datetime.datetime.now()
    meditations[-1]['end'] = end_time
    save_meditations(meditations)
    duration = end_time - meditations[-1]['start']
    print(f"Meditation ended at {end_time}. Total duration: {duration}.")

def show_meditation_stats():
    meditations = load_meditations()
    total_duration = sum(end_time - start_time for start_time, end_time in meditations)
    print(f"Total meditation duration: {total_duration}")

if __name__ == '__main__':
    while True:
        print("\nMeditation Tracker")
        print("1. Start Meditation")
        print("2. End Meditation")
        print("3. Show Meditation Stats")
        print("4. Exit")
        choice = input("Enter your choice (1-4): ")
        
        if choice == '1':
            start_meditation()
        elif choice == '2':
            end_meditation()
        elif choice == '3':
            show_meditation_stats()
        elif choice == '4':
            break
        else:
            print("Invalid choice. Please try again.")
```

```python
    except FileNotFoundError:
        return []
```

```python
import datetime
import json

# 定义文件路径
LOG_FILE = "programming_log.json"

# 加载编程日志
def load_log():
    try:
        with open(LOG_FILE, "r") as file:
            return json.load(file)
    except FileNotFoundError:
        return []

# 保存编程日志
def save_log(log):
    with open(LOG_FILE, "w") as file:
        json.dump(log, file, indent=4)

# 开始编程
def start_programming():
    start_time = datetime.datetime.now()
    print(f"Programming started at {start_time}")
    return start_time

# 结束编程
def end_programming(start_time, log):
    end_time = datetime.datetime.now()
    duration = end_time - start_time
    log.append({
        "start_time": start_time.strftime("%Y-%m-%d %H:%M:%S"),
        "end_time": end_time.strftime("%Y-%m-%d %H:%M:%S"),
        "duration": int(duration.total_seconds())
    })
    print(f"Programming ended at {end_time}. Duration: {duration.total_seconds()} seconds.")
    save_log(log)

# 计算并打印编程时长统计
def print_stats(log):
    total_duration = sum(session["duration"] for session in log)
    average_duration = total_duration / len(log) if log else 0
    print(f"Total programming duration: {total_duration} seconds.")
    print(f"Average programming duration: {average_duration} seconds.")

# 主程序
if __name__ == "__main__":
    log = load_log()

    print_stats(log)
```
