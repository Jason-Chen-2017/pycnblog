import numpy as np
import tensorflow as tf

# 定义神经图灵机模型
class NeuralTuringMachine(tf.keras.Model):
    def __init__(self, input_dim, memory_size, output_dim):
        super(NeuralTuringMachine, self).__init__()
        self.input_dim = input_dim
        self.memory_size = memory_size
        self.output_dim = output_dim

        # 输入层、隐藏层和输出层的权重矩阵
        self.input_weights = self.add_weight(
            shape=(input_dim, memory_size),
            initializer="random_normal",
            trainable=True
        )
        self.hidden_weights = self.add_weight(
            shape=(memory_size, memory_size),
            initializer="random_normal",
            trainable=True
        )
        self.output_weights = self.add_weight(
            shape=(memory_size, output_dim),
            initializer="random_normal",
            trainable=True
        )

    @tf.function
    def call(self, inputs, memory=None):
        # 输入层计算
        input_layer = tf.matmul(inputs, self.input_weights)

        # 记忆网络计算
        if memory is None:
            memory = tf.zeros((self.memory_size, self.memory_size))
        read_head = tf.random.normal((self.memory_size, 1))
        memory_read = tf.matmul(read_head, memory)
        memory_write = tf.random.normal((self.memory_size, 1))

        # 输出层计算
        output_layer = tf.matmul(memory_read, self.output_weights)

        return output_layer

# 实例化神经图灵机模型
ntm = NeuralTuringMachine(input_dim=10, memory_size=10, output_dim=2)

# 输入数据
input_data = tf.random.normal((1, 10))

# 训练模型
ntm.compile(optimizer="adam", loss="mse")
ntm.fit(input_data, tf.zeros((1, 2)), epochs=10)

# 进行推理
output = ntm.predict(input_data)
print("输出结果：", output)
