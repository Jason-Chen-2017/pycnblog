```python
        out = self.relu(out)
        out = self.fc2(out)
        return out

# 实例化模型
model = NeuralNetwork(input_size=X_train.shape[1], hidden_size=50, output_size=1)

# 定义损失函数和优化器
loss_function = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=0.001)

# 训练模型
for epoch in range(100):
    model.train()
    optimizer.zero_grad()
    
    # 前向传播
    outputs = model(X_train)
    loss = loss_function(outputs, y_train)
    
    # 反向传播
    loss.backward()
    optimizer.step()
    
    if (epoch+1) % 10 == 0:
        print(f'Epoch [{epoch+1}/100], Loss: {loss.item():.4f}')
```
