data = pd.read_csv('tool_data.csv')

data = (data - data.min()) / (data.max() - data.min())

### 数学模型和数学公式

#### 信誉评分模型

$$
\text{信誉评分} = \alpha \times \text{行为评分} + (1 - \alpha) \times \text{历史评分}
$$

其中，$\alpha$ 为权重系数，用于平衡行为评分和历史评分的重要性。

#### 资源调度模型

$$
\min \sum_{i=1}^{n} (t_i - c_i)
$$

其中，$t_i$ 为第 $i$ 项任务的响应时间，$c_i$ 为第 $i$ 项任务的处理时间。

def allocate_tool(request):
    if request.method == 'POST':

        tool_id = data.get('tool_id')
        user_id = data.get('user_id')

        if validate_tool_reservation(tool_id, user_id):
            tool = Tool.objects.get(id=tool_id)
            user = User.objects.get(id=user_id)

            if allocate_resource(tool, user):
                return JsonResponse({'status': 'success', 'message': 'Tool allocated successfully.'})
            else:
                return JsonResponse({'status': 'error', 'message': 'Tool is not available.'})
        else:
            return JsonResponse({'status': 'error', 'message': 'Invalid tool reservation.'})

**algorithms.py**：

```python
from .models import Tool, Reservation
from django.db import transaction
from datetime import datetime

def allocate_resource(tool, user):
    with transaction.atomic():
        # 检查工具状态
        if tool.status != 'available':
            return False

        # 检查用户是否有未完成的预订
        if Reservation.objects.filter(user=user, status='pending').exists():
            return False

        # 更新工具状态
        tool.status = 'allocated'
        tool.save()

        # 创建预订记录
        reservation = Reservation(tool=tool, user=user, start_time=datetime.now(), end_time=datetime.now())
        reservation.save()

        return True
```
