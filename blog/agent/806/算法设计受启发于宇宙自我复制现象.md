from tensorflow.keras.models import Model
from tensorflow.keras.layers import Input, Dense, Reshape, Flatten
from tensorflow.keras.optimizers import Adam

def build_generator():
    input_layer = Input(shape=(100,))
    x = Dense(128, activation='relu')(input_layer)
    x = Dense(64, activation='relu')(x)
    x = Reshape((28, 28, 1))(x)
    output_layer = Flatten()(x)
    model = Model(inputs=input_layer, outputs=output_layer)
    return model

def build_discriminator():
    input_layer = Input(shape=(28, 28, 1))
    x = Flatten()(input_layer)
    x = Dense(64, activation='relu')(x)
    x = Dense(128, activation='relu')(x)
    output_layer = Dense(1, activation='sigmoid')(x)
    model = Model(inputs=input_layer, outputs=output_layer)
    return model

def build_gan(generator, discriminator):
    input_layer = Input(shape=(100,))
    generated_image = generator(input_layer)
    valid = discriminator(generated_image)
    model = Model(inputs=input_layer, outputs=valid)
    return model

def train_gan(generator, discriminator, real_images, fake_images, batch_size, epochs):
    gan_optimizer = Adam(0.0001, 0.5)
    for epoch in range(epochs):
        for _ in range(real_images.shape[0] // batch_size):
            real_images_batch = real_images[:batch_size]
            noise = np.random.normal(0, 1, (batch_size, 100))
            generated_images = generator.predict(noise)
            combined_images = np.concatenate([real_images_batch, generated_images])
            labels = np.concatenate([np.ones((batch_size, 1)), np.zeros((batch_size, 1))])
            discriminator.trainable = True
            discriminator.train_on_batch(combined_images, labels)
            discriminator.trainable = False
            noise = np.random.normal(0, 1, (batch_size, 100))
            valid = generator.predict(noise)
            labels = np.ones((batch_size, 1))
            gan_optimizer.train_on_batch(noise, labels)

def generate_fake_images(generator, real_images, batch_size):
    noise = np.random.normal(0, 1, (batch_size, 100))
    return generator.predict(noise)

def main():
    real_images = np.load("real_images.npy")
    generator = build_generator()
    discriminator = build_discriminator()
    gan = build_gan(generator, discriminator)
    train_gan(generator, discriminator, real_images, generate_fake_images(generator, real_images, batch_size=32), batch_size=32, epochs=100)
    generator.save("generator.h5")
    discriminator.save("discriminator.h5")

if __name__ == "__main__":
    main()
```

以下是一个简单的生成手写数字图像的代码示例。

```python
import tensorflow as tf
from tensorflow.keras.models import Model
from tensorflow.keras.layers import Input, Dense, Reshape, Flatten
from tensorflow.keras.optimizers import Adam

def build_generator():
    input_layer = Input(shape=(100,))
    x = Dense(128, activation='relu')(input_layer)
    x = Dense(64, activation='relu')(x)
    x = Reshape((28, 28, 1))(x)
    output_layer = Flatten()(x)
    model = Model(inputs=input_layer, outputs=output_layer)
    return model

def build_discriminator():
    input_layer = Input(shape=(28, 28, 1))
    x = Flatten()(input_layer)
    x = Dense(64, activation='relu')(x)
    x = Dense(128, activation='relu')(x)
    output_layer = Dense(1, activation='sigmoid')(x)
    model = Model(inputs=input_layer, outputs=output_layer)
    return model

def build_gan(generator, discriminator):
    input_layer = Input(shape=(100,))
    generated_image = generator(input_layer)
    valid = discriminator(generated_image)
    model = Model(inputs=input_layer, outputs=valid)
    return model

def train_gan(generator, discriminator, real_images, fake_images, batch_size, epochs):
    gan_optimizer = Adam(0.0001, 0.5)
    for epoch in range(epochs):
        for _ in range(real_images.shape[0] // batch_size):
            real_images_batch = real_images[:batch_size]
            noise = np.random.normal(0, 1, (batch_size, 100))
            generated_images = generator.predict(noise)
            combined_images = np.concatenate([real_images_batch, generated_images])
            labels = np.concatenate([np.ones((batch_size, 1)), np.zeros((batch_size, 1))])
            discriminator.trainable = True
            discriminator.train_on_batch(combined_images, labels)
            discriminator.trainable = False
            noise = np.random.normal(0, 1, (batch_size, 100))
            valid = generator.predict(noise)
            labels = np.ones((batch_size, 1))
            gan_optimizer.train_on_batch(noise, labels)

def generate_fake_images(generator, real_images, batch_size):
    noise = np.random.normal(0, 1, (batch_size, 100))
    return generator.predict(noise)

def main():
    real_images = np.load("real_images.npy")
    generator = build_generator()
    discriminator = build_discriminator()
    gan = build_gan(generator, discriminator)
    train_gan(generator, discriminator, real_images, generate_fake_images(generator, real_images, batch_size=32), batch_size=32, epochs=100)
    generator.save("generator.h5")
    discriminator.save("discriminator.h5")

if __name__ == "__main__":
    main()
```

在这个代码中，我们首先定义了生成器、判别器和生成对抗网络（GANs）的模型结构。然后，我们使用训练数据和噪声数据来训练生成对抗网络（GANs），最终生成逼真的手写数字图像。
