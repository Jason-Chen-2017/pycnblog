#### 4.6.2 系统核心实现源代码

以下是一个简单的Python源代码示例，用于生成一个基于涂鸦风格的图像：

```python
from PIL import Image, ImageDraw
import numpy as np

# 设置画布大小
width, height = 800, 600
canvas = Image.new('RGB', (width, height), 'white')
draw = ImageDraw.Draw(canvas)

# 生成随机涂鸦线条
def generate_stroke(start, end, thickness=5):
    x1, y1 = start
    x2, y2 = end
    dx = x2 - x1
    dy = y2 - y1
    distance = np.sqrt(dx**2 + dy**2)
    angle = np.arctan2(dy, dx) * 180 / np.pi
    points = [(x1 + thickness/2 * np.cos(angle), y1 + thickness/2 * np.sin(angle))]
    for _ in range(int(distance)):
        points.append((points[-1][0] + np.cos(angle), points[-1][1] + np.sin(angle)))
    return points

# 在画布上绘制多条随机线条
for _ in range(20):
    start = (np.random.randint(0, width), np.random.randint(0, height))
    end = (np.random.randint(0, width), np.random.randint(0, height))
    draw.line(generate_stroke(start, end), fill='black', width=3)

# 显示图像
canvas.show()

# 保存图像
canvas.save('guerrilla_art.png', format='PNG')
```
