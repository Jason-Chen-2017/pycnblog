```python
# 初始化数据

    '问题': ['你最近的生活状态如何？', '你对未来的职业规划是什么？', '你如何看待社会责任？'],
    '答案': ['忙碌而充实', '希望成为一名软件开发工程师', '认为个人应积极参与社会公益活动']
}

# 创建DataFrame
df = pd.DataFrame(data)

# 打印DataFrame
print(df)

# 分析答案
for index, row in df.iterrows():
    if row['答案'] == '忙碌而充实':
        print(f"{row['问题']}：状态良好，建议保持现有节奏。")
    elif row['答案'] == '希望成为一名软件开发工程师':
        print(f"{row['问题']}：目标明确，建议制定详细的职业发展计划。")
    else:
        print(f"{row['问题']}：社会责任感强烈，建议积极参与社会公益活动。")
```
