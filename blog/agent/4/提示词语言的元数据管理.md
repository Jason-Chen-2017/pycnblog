```python
import pymysql
import csv

# 连接数据库
connection = pymysql.connect(host='localhost', user='root', password='password', database='metadata_db')

# 采集数据库表结构
cursor = connection.cursor()
cursor.execute("SHOW TABLES")
tables = cursor.fetchall()

# 保存表结构到CSV文件
with open('metadata.csv', 'w', newline='') as file:
    writer = csv.writer(file)
    writer.writerow(['table_name', 'column_name', 'data_type', 'column_comment'])
    for table in tables:
        table_name = table[0]
        cursor.execute(f"SHOW COLUMNS FROM {table_name}")
        columns = cursor.fetchall()
        for column in columns:
            writer.writerow([table_name, column[0], column[1], column[2]])

# 关闭数据库连接
cursor.close()
connection.close()
```

```sql
);

LOAD DATA INFILE 'metadata.csv'
```
