### 1.2 提示词工程概述

提示词工程（Prompt Engineering）是一种专门针对自然语言处理（NLP）任务进行提示设计的工程实践。其目的是通过精心设计的提示，引导模型生成更准确、更符合预期的输出。

| 关键技术        | 提示设计、反馈机制 | 词嵌入、序列处理   | 神经网络架构       |

import cv2
import numpy as np

def classify_image(image_path, label_prompt):
    image = cv2.imread(image_path)
    image = cv2.resize(image, (224, 224))
    image = image / 255.0
    image = image.reshape(1, 224, 224, 3)

    prompt = f"label this image: {label_prompt}"
    response = openai.Completion.create(
        engine="text-davinci-002",
        prompt=prompt,
        max_tokens=50,
        n=1,
        stop=None,
        temperature=0.5,
    )
    label = response.choices[0].text.strip()
    return label

image_path = "path/to/image.jpg"
label_prompt = "cat"
label = classify_image(image_path, label_prompt)
