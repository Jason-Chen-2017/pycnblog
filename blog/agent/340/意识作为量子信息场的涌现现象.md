backend = Aer.get_backend('qasm_simulator')

# 运行量子电路

fidelity_00 = state_fidelity(state_00, result.state())
fidelity_11 = state_fidelity(state_11, result.state())
fidelity_01 = state_fidelity(state_01, result.state())
fidelity_10 = state_fidelity(state_10, result.state())

print("fidelity_00:", fidelity_00)
