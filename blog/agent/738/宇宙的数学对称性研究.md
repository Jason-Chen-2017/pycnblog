   $$ O(x) = x' $$

2. **对称空间**：宇宙中所有满足对称性操作的点构成的集合称为对称空间，用$S$表示。

   $$ \frac{\partial L}{\partial x} = 0 $$

在物理学中，对称性原理与物理定律密切相关。许多物理定律，如守恒定律，都与对称性原理有关。例如，能量守恒定律是时间对称性的表现，动量守恒定律是空间平移对称性的表现。对称性原理为我们提供了一种强大的工具，可以帮助我们理解物理系统的行为和性质。

### 2.2 宇宙演化模型

宇宙演化模型是描述宇宙从大爆炸到现在的演化过程的数学模型。这些模型基于物理定律和观测数据，帮助我们理解宇宙的结构和演化规律。宇宙演化模型包括多个层次，从宇宙的大尺度结构到微观的星系和星体的形成。

宇宙演化模型的核心是宇宙膨胀理论。根据宇宙膨胀理论，宇宙从大爆炸开始，经历了数十亿年的膨胀，形成了我们现在观测到的宇宙。宇宙膨胀理论包括哈勃定律、奥克威尔定律、宇宙背景辐射等。

#### 哈勃定律

哈勃定律描述了宇宙膨胀的速度与宇宙距离之间的关系。根据哈勃定律，宇宙的膨胀速度与宇宙距离成正比，即：

$$ v = H_0 d $$

其中，$v$ 是宇宙膨胀速度，$H_0$ 是哈勃常数，$d$ 是宇宙距离。

$$ \rho \propto \frac{1}{d^3} $$

其中，$\rho$ 是宇宙密度，$d$ 是宇宙距离。

#define BNO055_SAMPLERATE_DELAY_MS 1000

// 定义Adafruit_BNO055对象
Adafruit_BNO055 bno = Adafruit_BNO055(55);

// 声明全局变量
float heading;
float roll;
float pitch;

void setup() {
  // 初始化串口通信
  Serial.begin(9600);

  // 初始化传感器
  if (!bno.begin()) {
    Serial.println("Ooops, no BNO055 detected ... Check your wiring or I2C ADDR!");
    while (1);
  }

  // 配置传感器
  delay(1000);
  bno.setExtCrystalUse(true);
}

void loop() {
  // 读取传感器数据
  sensors_event_t event;
  bno.getEvent(&event);

  // 计算方向、倾斜角度
  heading = bno.getHeading();
  roll = bno.getRoll();
  pitch = bno.getPitch();

  // 输出数据
  Serial.print(" Heading: ");
  Serial.print(heading);
  Serial.print(" Roll: ");
  Serial.print(roll);
  Serial.print(" Pitch: ");
  Serial.println(pitch);

  delay(BNO055_SAMPLERATE_DELAY_MS);

        String target;

            this.temperature = temperature;
            this.pressure = pressure;
            this.humidity = humidity;
            this.target = target;
        }
    }

    public static List<Observation> readData(String filePath) throws IOException {
        List<Observation> observations = new ArrayList<>();
        try (BufferedReader br = new BufferedReader(new FileReader(filePath))) {
            String line;
            while ((line = br.readLine()) != null) {
                String[] values = line.split(",");
                observations.add(new Observation(Double.parseDouble(values[0]),
                                                 Double.parseDouble(values[1]),
                                                 Double.parseDouble(values[2]),
                                                 values[3]));
            }
        }
        return observations;
    }

            List<Observation> cosmicData = readData("cosmic_data.csv");

            // 分割数据为特征和目标

            List<String> targets = new ArrayList<>();

                features.add(feature);
                targets.add(obs.target);
            }

            // 划分训练集和测试集
