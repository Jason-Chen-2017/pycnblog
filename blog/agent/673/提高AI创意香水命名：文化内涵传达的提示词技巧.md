import numpy as np
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

# 基于规则的生成算法
def generate_names(rule_template, brand_features, cultural_elements):
    names = []
    for element in cultural_elements:
        name = rule_template.format(element=element, brand=brand_features['brand_name'])
        names.append(name)
    return names

# 基于评估函数的优化算法
def optimize_names(candidate_names, cultural_content, threshold):
    scores = []
    for name in candidate_names:
        score = calculate_similarity(name, cultural_content)
        scores.append(score)
    optimized_names = [name for name, score in zip(candidate_names, scores) if score > threshold]
    return optimized_names

# 基于语义相似度的评估算法
def calculate_similarity(name, cultural_content):
    vectorizer = TfidfVectorizer()
    content_vector = vectorizer.transform([cultural_content])
    name_vector = vectorizer.transform([name])
    similarity = cosine_similarity(content_vector, name_vector)
    return similarity[0][0]

# 示例数据
brand_features = {
    'brand_name': '雅诗兰黛',
    'cultural_elements': ['西湖', '浪漫', '典雅']

cultural_content = "雅诗兰黛的香水，充满了西湖的浪漫与典雅。"

# 生成候选命名
rule_template = "{}的{}香氛"
candidate_names = generate_names(rule_template, brand_features, brand_features['cultural_elements'])

# 优化命名
optimized_names = optimize_names(candidate_names, cultural_content, 0.5)

# 输出最优命名
print(optimized_names)

   - 在`app.py`中，定义命名生成函数，使用基于机器学习的算法生成命名候选：
     ```python
     from sklearn.feature_extraction.text import TfidfVectorizer
     from sklearn.metrics.pairwise import cosine_similarity

     def generate_names(data):
         vectorizer = TfidfVectorizer()
         vectors = vectorizer.fit_transform(data)
         similarities = cosine_similarity(vectors)
         names = []
         for i in range(len(similarities)):
             closest_names = similarities[i].argsort()[-10:]
             names.extend([data[idx] for idx in closest_names if idx != i])
         return names
     ```

3. **提示词设计模块**：

   - 在`app.py`中，定义提示词设计函数，提取与文化元素相关的词汇作为提示词：
     ```python
     import nltk

     nltk.download('wordnet')
     from nltk.corpus import wordnet

     def generate_hintWords(word):
         synsets = wordnet.synsets(word)
         hints = set()
         for synset in synsets:
             for lemma in synset.lemmas():
                 hints.add(lemma.name())
         return hints
     ```

4. **命名优化模块**：

   - 在`app.py`中，定义命名优化函数，使用评估函数对命名候选进行评分和筛选：
     ```python
     def optimize_names(names, cultural_content, threshold):
         scores = [calculate_similarity(name, cultural_content) for name in names]
         optimized_names = [name for name, score in zip(names, scores) if score > threshold]
         return optimized_names

     def calculate_similarity(name, cultural_content):
         vectorizer = TfidfVectorizer()
         content_vector = vectorizer.transform([cultural_content])
         name_vector = vectorizer.transform([name])
         similarity = cosine_similarity(content_vector, name_vector)
         return similarity[0][0]
     ```

5. **用户反馈处理模块**：

         with connection.cursor() as cursor:
             cursor.execute("INSERT INTO feedback (name, rating, comment) VALUES (%s, %s, %s)", (name, rating, comment))
         connection.commit()
         connection.close()
