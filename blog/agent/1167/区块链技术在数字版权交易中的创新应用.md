    // 构造函数，初始化版权信息

        title = _title;

}

cursor = conn.cursor()

# 创建表
cursor.execute('''CREATE TABLE IF NOT EXISTS copyright (id INTEGER PRIMARY KEY AUTOINCREMENT, title TEXT NOT NULL, owner TEXT NOT NULL, buyer TEXT, is_transferred BOOLEAN DEFAULT FALSE)''')

@click.group()
def cli():
    pass

@click.argument('title')
@click.argument('owner')
def register(title, owner):
    """Register a new copyright."""
    cursor.execute("INSERT INTO copyright (title, owner) VALUES (?, ?)", (title, owner))
    conn.commit()
    print(f"Copyright '{title}' registered successfully.")

@cli.command()
@click.argument('id')
@click.argument('buyer')
def transfer(id, buyer):
    """Transfer a copyright to a new buyer."""
    cursor.execute("UPDATE copyright SET buyer=?, is_transferred=True WHERE id=? AND buyer IS NULL", (buyer, id))
    conn.commit()
    if cursor.rowcount > 0:
        print(f"Copyright with ID {id} transferred to {buyer} successfully.")
    else:
        print("Error: Invalid ID or copyright is already transferred.")

@cli.command()
@click.argument('id')
def query(id):
    """Query a copyright by its ID."""
    cursor.execute("SELECT id, title, owner, buyer, is_transferred FROM copyright WHERE id=?", (id,))

        print(f"ID: {result[0]}, Title: {result[1]}, Owner: {result[2]}, Buyer: {result[3]}, Transferred: {result[4]}")
    else:
        print("Error: Invalid ID.")

if __name__ == '__main__':
    cli()

3. **可追溯性**：区块链的可追溯性使得数据交易的全过程都可以被记录和追溯。版权所有者可以随时查询自己的作品被用于何处，买家也可以确认所购买作品的版权合法性。

1. **数据生成**：交易数据通过节点生成，并打包成区块。每个区块包含一定数量的交易记录，并通过哈希算法生成区块头。

2. **数据传输**：新生成的区块通过节点在网络中传输，并广播给其他节点。

3. **数据验证**：其他节点接收区块后，对其进行验证。验证过程包括检查区块头的哈希值、检查交易的有效性等。

4. **数据确认**：验证通过后，区块将被添加到区块链中，成为永久记录。

1. **区块**：区块是区块链中的基本数据单元，包含一定数量的交易记录。每个区块都有一个唯一的标识，并通过哈希算法与前一个区块连接，形成链式结构。

2. **链**：链是区块链的核心组成部分，由多个区块按照特定顺序连接而成。链的结构确保了数据的不可篡改性，一旦数据上链，就难以被修改。

3. **节点**：节点是区块链网络中的参与者，负责数据的存储和验证。每个节点都保存一份完整的区块链数据，并通过网络与其他节点通信，共同维护区块链的运行。

#### 3.3.1 智能合约的应用

智能合约是区块链技术中的一个核心组件，它能够自动执行合同条款。以下是一个简单的智能合约示例，用于数字版权交易。

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract CopyrightManagement {
    // 定义版权结构体
    struct Copyright {
        string title;
        address owner;
        bool isSold;
    }

    // 字典，存储版权信息
    mapping(uint256 => Copyright) public copyrights;

    // 记录已注册的版权数量
    uint256 public totalCopyrights;

    // 事件，用于通知版权转让
    event CopyrightTransfer(uint256 id, address owner, address buyer);

    // 注册版权
    function registerCopyright(string memory title) public {
        require(!isSold(title), "Copyright is already sold.");
        copyrights[totalCopyrights] = Copyright({title: title, owner: msg.sender, isSold: false});
        totalCopyrights++;
    }

    // 检查版权是否已售出
    function isSold(string memory title) public view returns (bool) {
        for (uint256 i = 0; i < totalCopyrights; i++) {
            if (keccak256(abi.encodePacked(copyrights[i].title)) == keccak256(abi.encodePacked(title)) && copyrights[i].isSold) {
                return true;
            }
        }
        return false;
    }

    // 转让版权
    function transferCopyright(uint256 id, address buyer) public {
        require(!isSold(id), "Copyright is already sold.");
        require(msg.sender == copyrights[id].owner, "Only the owner can transfer the copyright.");
        copyrights[id].isSold = true;
        copyrights[id].owner = buyer;
        emit CopyrightTransfer(id, msg.sender, buyer);
    }
}
```

```solidity
// SPDX-License-Identifier: MIT

pragma solidity ^0.8.0;

contract Copyright {
    // 结构体定义版权信息
    struct CopyrightInfo {
        uint256 id;
        string title;
        address owner;
        bool isSold;
    }

    // 存储所有版权信息
    mapping(uint256 => CopyrightInfo) public copyrights;

    // 记录版权数量
    uint256 public totalCopyrights = 0;

    // 事件通知版权注册和转让
    event NewCopyrightRegistered(uint256 id);
    event CopyrightSold(uint256 id, address newOwner);

    // 注册版权
    function registerCopyright(string memory title) public {
        require(copyrights[totalCopyrights].isSold == false, "Copyright already exists.");
        copyrights[totalCopyrights] = CopyrightInfo(totalCopyrights, title, msg.sender, false);
        totalCopyrights++;
        emit NewCopyrightRegistered(totalCopyrights - 1);
    }

    // 转让版权
    function sellCopyright(uint256 id, address newOwner) public {
        require(copyrights[id].owner == msg.sender, "Not the owner of the copyright.");
        require(copyrights[id].isSold == false, "Copyright already sold.");
        copyrights[id].isSold = true;
        copyrights[id].owner = newOwner;
        emit CopyrightSold(id, newOwner);
    }

    // 查询版权信息
    function getCopyrightInfo(uint256 id) public view returns (CopyrightInfo memory) {
        return copyrights[id];
    }
}
```
