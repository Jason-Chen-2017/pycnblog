```python
import nltk
from nltk.tokenize import word_tokenize
from nltk.corpus import stopwords
from nltk.stem import WordNetLemmatizer

# Preprocessing the input text
def preprocess_text(text):
    # Tokenization
    tokens = word_tokenize(text)
    
    # Remove stop words
    stop_words = set(stopwords.words('english'))
    filtered_tokens = [token for token in tokens if token.lower() not in stop_words]
    
    # Lemmatization
    lemmatizer = WordNetLemmatizer()
    lemmatized_tokens = [lemmatizer.lemmatize(token) for token in filtered_tokens]
    
    return lemmatized_tokens

# Context extraction using NLP techniques
def extract_context(tokens):
    # Named entity recognition
    named_entities = nltk.ne_chunk(tokens)
    
    # Part-of-speech tagging
    pos_tags = nltk.pos_tag(tokens)
    
    # Relation extraction
    # (Note: This is a simplified example. In practice, more advanced techniques may be required)
    relations = []
    for i in range(len(tokens) - 1):
        if tokens[i].lower() in ['and', 'but', 'while', 'however']:
            relations.append((tokens[i], tokens[i+1]))
    
    return named_entities, pos_tags, relations

# Calculate coherence score and identify inconsistencies
def calculate_coherence(tokens):
    # (Note: This is a simplified example. In practice, more advanced methods may be required)
    coherence_score = 0
    for i in range(len(tokens) - 1):
        if tokens[i].lower() in ['and', 'but', 'while', 'however']:
            coherence_score += 1
    
    if coherence_score < len(tokens) / 2:
        print("Inconsistency detected in the text.")
    else:
        print("Text is consistent.")
```
