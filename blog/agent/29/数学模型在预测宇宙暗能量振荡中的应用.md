```python
import numpy as np
import matplotlib.pyplot as plt

# Constants
G = 6.67430e-11  # Gravitational constant
c = 299792458    # Speed of light
H_0 = 70         # Hubble constant

# Function to calculate the density of dark energy
def dark_energy_density(z):
    Omega_m = 0.3  # Matter density parameter
    Omega_de = 0.7  # Dark energy density parameter
    return Omega_de * (1 + z)**(-3)

# Function to calculate the acceleration due to dark energy
def acceleration(z):
    rho_de = dark_energy_density(z)
    a = np.sqrt(rho_de / (G * c**2))
    return a

# Function to simulate the oscillations of dark energy
def simulate_oscillations(z_range):
    accelerations = [acceleration(z) for z in z_range]
    plt.plot(z_range, accelerations)
    plt.xlabel('Redshift (z)')
    plt.ylabel('Acceleration (a)')
    plt.title('Dark Energy Oscillations')
    plt.show()

# Example: Simulate oscillations from z = 0 to z = 5
z_range = np.linspace(0, 5, 1000)
simulate_oscillations(z_range)
```
