import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from mermaid import Mermaid

# 匹配度计算函数
def calculate_matching_score(data):
    emotional_matching = 0.5 * (data['emotional_investment'] + data['emotional_satisfaction'])
    financial_matching = 0.5 * (data['financial_support'] + data['financial_satisfaction'])
    total_matching = emotional_matching + financial_matching
    return total_matching

# 数据读取与预处理
data = pd.read_csv('marital_data.csv')
data['total_matching_score'] = data.apply(calculate_matching_score, axis=1)

# 关系平衡分析
balanced_data = data[data['total_matching_score'] > 0.6]

# 绘制匹配度分布图
plt.hist(balanced_data['total_matching_score'], bins=10, edgecolor='black')
plt.xlabel('Total Matching Score')
plt.ylabel('Frequency')
plt.title('Distribution of Total Matching Scores')
plt.show()

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from mermaid import Mermaid

# 匹配度计算函数
def calculate_matching_score(data):
    emotional_matching = 0.5 * (data['emotional_investment'] + data['emotional_satisfaction'])
    financial_matching = 0.5 * (data['financial_support'] + data['financial_satisfaction'])
    total_matching = emotional_matching + financial_matching
    return total_matching

# 数据读取与预处理
data = pd.read_csv('marital_data.csv')

# 关系平衡分析
balanced_data = data[data['total_matching_score'] > 0.6]

plt.hist(balanced_data['total_matching_score'], bins=10, edgecolor='black')
plt.xlabel('Total Matching Score')
plt.ylabel('Frequency')
plt.title('Distribution of Total Matching Scores')

mermaid = Mermaid()
