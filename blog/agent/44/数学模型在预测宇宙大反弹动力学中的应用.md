import numpy as np

# 定义宇宙膨胀模型

    rho, Lambda, Omega_m = parameters
    a, da_dt = y
    H = np.sqrt(Lambda + Omega_m * a**-2)
    d2a_dt2 = - 2 * H * da_dt / a
    return [da_dt, d2a_dt2]
