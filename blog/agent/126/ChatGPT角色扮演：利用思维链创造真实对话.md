3. **角色匹配**：根据角色特征，对生成的对话进行筛选和匹配。

# 定义角色扮演函数
def chatgpt_role_playing(role_desc, context):
    # 对输入进行处理
    prompt = f"{context}，作为{role_desc}，你接下来会说什么？"
    
    # 调用ChatGPT API进行对话生成
    response = openai.Completion.create(
        engine="text-davinci-002",
        prompt=prompt,
        max_tokens=100
    )
    
    # 对角色进行匹配
    role_match = role_desc in response.choices[0].text
    
    # 输出结果
    if role_match:
        return response.choices[0].text
    else:
        return "角色匹配失败，请检查角色描述。"
