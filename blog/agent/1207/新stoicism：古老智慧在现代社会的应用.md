    }

    // 搜索文章
    function searchArticles(keyword) {
        fetch(`/api/articles?keyword=${keyword}`)
            .then(response => response.json())
            .then(data => setArticles(data));
    }

    // 阅读文章
    function readArticle(articleId) {
        fetch(`/api/articles/${articleId}`)
            .then(response => response.json())
            .then(article => {
                // 显示文章内容
                console.log(article.content);
            });
    }

    fetchArticles();

    return (
        <div>
            <input type="text" placeholder="搜索文章" onChange={e => searchArticles(e.target.value)} />
            <ul>
                {articles.map(article => (
                    <li key={article.id} onClick={() => readArticle(article.id)}>
                        {article.title}
                    </li>
                ))}

    );
