### 5.2 系统核心实现源代码

import numpy as np
from qiskit import QuantumCircuit, execute, Aer
from sklearn.model_selection import train_test_split
from sklearn.metrics import accuracy_score
from sklearn.linear_model import LogisticRegression
import matplotlib.pyplot as plt

# 生成模拟数据
def generate_data(num_samples):
    # 生成特征矩阵和标签向量
    X = np.random.rand(num_samples, 2)
    y = np.where(X[:, 0] > 0.5, 1, 0)
    return X, y

# 数据预处理
def preprocess_data(X):
    # 去除特征矩阵中的噪声
    X_noisy = X + 0.1 * np.random.randn(*X.shape)
    # 归一化特征矩阵
    X_normalized = (X_noisy - np.mean(X_noisy, axis=0)) / np.std(X_noisy, axis=0)
    return X_normalized

# 特征提取
def extract_features(X):
    # 从特征矩阵中提取两个特征：特征1和特征2
    feature1 = X[:, 0]
    feature2 = X[:, 1]
    return feature1, feature2

# 训练量子传感模型
def train_quantum_model(feature1, feature2, y):
    # 创建量子电路
    qc = QuantumCircuit(2)  # 假设有两个量子比特
    qc.h(0)
    qc.cx(0, 1)
    
    # 执行量子电路，获取测量结果
    backend = Aer.get_backend('qasm_simulator')
    job = execute(qc, backend, shots=1000)
    result = job.result()
    counts = result.get_counts(qc)
    
    # 训练经典机器学习模型
    model = LogisticRegression()
    model.fit(np.array(list(counts.keys())), y)
    
    return model

# 预测与评估
def predict_and_evaluate(model, X_test, y_test):
    y_pred = model.predict(X_test)
    accuracy = accuracy_score(y_test, y_pred)
    print(f"Accuracy: {accuracy:.2f}")
    
    return accuracy

# 主函数
def main():
    # 生成模拟数据
    num_samples = 100
    X, y = generate_data(num_samples)
    
    # 数据预处理
    X_processed = preprocess_data(X)
    
    # 特征提取
    feature1, feature2 = extract_features(X_processed)
    
    # 训练量子传感模型
    model = train_quantum_model(feature1, feature2, y)
    
    # 预测与评估
    X_test = np.random.rand(10, 2)
    y_test = np.where(X_test[:, 0] > 0.5, 1, 0)
    predict_and_evaluate(model, X_test, y_test)

if __name__ == '__main__':
    main()
