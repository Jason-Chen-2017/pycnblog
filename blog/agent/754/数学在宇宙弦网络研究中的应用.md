import random

# 初始化种群
def initialize_population(pop_size, param_range):
    population = []
    for _ in range(pop_size):
        individual = [random.uniform(*param_range) for _ in range(len(param_range))]
        population.append(individual)
    return population

# 计算适应度
def calculate_fitness(population, fitness_function):
    fitness_scores = []
    for individual in population:
        fitness_scores.append(fitness_function(individual))
    return fitness_scores

# 选择操作
def selection(population, fitness_scores, selection_rate):
    selected_individuals = random.choices(population, weights=fitness_scores, k=len(population) * selection_rate)
    return selected_individuals

# 交叉操作
def crossover(parent1, parent2, crossover_rate):
    if random.random() < crossover_rate:
        crossover_point = random.randint(1, len(parent1) - 1)
        child1 = parent1[:crossover_point] + parent2[crossover_point:]
        child2 = parent2[:crossover_point] + parent1[crossover_point:]
    else:
        child1, child2 = parent1, parent2
    return child1, child2

# 变异操作
def mutation(individual, mutation_rate):
    for i in range(len(individual)):
        if random.random() < mutation_rate:
            individual[i] = random.uniform(individual[i][0], individual[i][1])
    return individual

# 主函数
def genetic_algorithm(pop_size, param_range, fitness_function, selection_rate, crossover_rate, mutation_rate, generations):
    population = initialize_population(pop_size, param_range)
    for _ in range(generations):
        fitness_scores = calculate_fitness(population, fitness_function)
        selected_individuals = selection(population, fitness_scores, selection_rate)
        next_population = []
        for i in range(0, pop_size, 2):
            parent1, parent2 = selected_individuals[i], selected_individuals[i+1]
            child1, child2 = crossover(parent1, parent2, crossover_rate)
            next_population.append(mutation(child1, mutation_rate))
            next_population.append(mutation(child2, mutation_rate))
        population = next_population
    return population

# 示例：适应度函数
def fitness_function(individual):
    # 假设个体为三个参数，适应度函数为个体参数的平均值
    return sum(individual) / len(individual)

# 示例：参数设置
pop_size = 100
param_range = [(0, 10), (0, 10), (0, 10)]
selection_rate = 0.5
crossover_rate = 0.7
mutation_rate = 0.01
generations = 100

# 运行遗传算法
best_population = genetic_algorithm(pop_size, param_range, fitness_function, selection_rate, crossover_rate, mutation_rate, generations)
print("最优解：", best_population)

# 示例：适应度函数

    # 假设个体为三个参数，适应度函数为个体参数的平均值
