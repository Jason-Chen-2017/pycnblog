def initialize_database():
    conn = sqlite3.connect('qdlgip.db')
    cursor = conn.cursor()
    
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS quality (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            value REAL NOT NULL
        )
    ''')
    
    conn.commit()
    conn.close()

# 随机生成质
def generate_quality(n):
    return np.random.uniform(0, 1, n)

# 应用语言游戏规则
def apply_language_game(quality, factor=1.1):
    return quality * factor

# 存储质
def store_quality(quality):
    conn = sqlite3.connect('qdlgip.db')
    cursor = conn.cursor()
    
    cursor.executemany("INSERT INTO quality (value) VALUES (?)", quality.tolist())
    
    conn.commit()
    conn.close()

# 从数据库中获取质
def get_quality():
    conn = sqlite3.connect('qdlgip.db')
    cursor = conn.cursor()
    
    cursor.execute("SELECT * FROM quality")
    rows = cursor.fetchall()
    
    conn.close()
    
    return [row[1] for row in rows]

# 主函数
def main():
    # 初始化数据库
    initialize_database()
    
    # 生成100个质
    qualities = generate_quality(100)
    
    # 应用语言游戏规则并存储质
    for quality in qualities:
        updated_quality = apply_language_game(quality)
        store_quality([updated_quality])
    
    # 从数据库中获取质
    updated_qualities = get_quality()
    
    # 绘制质的变化曲线
    plt.plot(updated_qualities)
    plt.xlabel('迭代次数')
    plt.ylabel('质值')
    plt.title('质的变化曲线')
    plt.show()

if __name__ == "__main__":
    main()
