```python
import numpy as np
import cv2

# 获取医学成像数据
def get_imaging_data():
    # 示例代码，实际中需要连接医学成像设备
    return np.random.rand(256, 256, 3)

# 预处理医学成像数据
def preprocess_imaging_data(data):
    # 示例代码，实际中需要根据具体图像进行预处理
    return cv2.resize(data, (128, 128))

# 获取手术操作环境数据
def get_surgery_environment_data():
    # 示例代码，实际中需要连接AR系统获取数据
    return np.random.rand(128, 128, 3)

# 数据融合
def fuse_data(imaging_data, environment_data):
    # 示例代码，实际中需要根据具体算法进行数据融合
    return imaging_data * environment_data

# 生成导航信息
def generate_navigation_info(fused_data):
    # 示例代码，实际中需要根据具体算法生成导航信息
    return cv2.cvtColor(fused_data, cv2.COLOR_BGR2RGB)

# 显示导航信息
def display_navigation_info(info):
    # 示例代码，实际中需要在AR眼镜或屏幕上显示信息
    cv2.imshow('Navigation Info', info)
    cv2.waitKey(0)
    cv2.destroyAllWindows()

# 主函数
def main():
    imaging_data = get_imaging_data()
    environment_data = get_surgery_environment_data()
    preprocessed_data = preprocess_imaging_data(imaging_data)
    fused_data = fuse_data(preprocessed_data, environment_data)
    navigation_info = generate_navigation_info(fused_data)
    display_navigation_info(navigation_info)

if __name__ == '__main__':
    main()
```

```python
# main.py

import sys
import cv2
from PyQt5 import QtWidgets, QtGui, QtCore

class ARNavigationSystem(QtWidgets.QMainWindow):
    def __init__(self):
        super().__init__()
        self.initUI()

    def initUI(self):
        self.setWindowTitle('AR Surgery Navigation')
        self.setGeometry(100, 100, 800, 600)

        self.image_label = QtWidgets.QLabel(self)
        self.image_label.resize(640, 480)
        self.setCentralWidget(self.image_label)

        self.load_data()
        self.start_display()

    def load_data(self):
        # 获取医学成像数据和手术操作环境数据
        self.imaging_data = cv2.imread('imaging_data.jpg')
        self.environment_data = cv2.imread('environment_data.jpg')

    def start_display(self):
        # 数据预处理
        self.preprocessed_imaging_data = cv2.resize(self.imaging_data, (640, 480))
        self.preprocessed_environment_data = cv2.resize(self.environment_data, (640, 480))

        # 数据融合
        self.fused_data = self.preprocessed_imaging_data * self.preprocessed_environment_data

        # 导航信息生成
        self.navigation_info = cv2.cvtColor(self.fused_data, cv2.COLOR_BGR2RGB)

        # 显示导航信息
        self.update_image()

    def update_image(self):
        # 将导航信息显示在屏幕上
        qimg = QtGui.QImage(self.navigation_info.data, self.navigation_info.shape[1], self.navigation_info.shape[0], QtGui.QImage.Format_RGB888)
        self.image_label.setPixmap(QtGui.QPixmap.fromImage(qimg))
        self.image_label.show()

if __name__ == '__main__':
    app = QtWidgets.QApplication(sys.argv)
    ex = ARNavigationSystem()
    sys.exit(app.exec_())
```

2. **初始化UI**：设置窗口标题和大小，创建一个 QLabel 用于显示导航信息。
3. **加载数据**：通过`cv2.imread`函数加载医学成像数据和手术操作环境数据。
4. **预处理数据**：使用`cv2.resize`函数调整图像大小，以匹配窗口尺寸。
