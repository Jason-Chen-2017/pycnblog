import numpy as np
import matplotlib.pyplot as plt
from scipy.signal import find_peaks

# 数据采集模块
def data_collection(sensors):
    data = []
    for sensor in sensors:
        data_point = np.random.rand()  # 模拟传感器采集数据
        data.append(data_point)
    return data

# 数据预处理模块
def data_preprocessing(data):
    processed_data = (data - np.mean(data)) / np.std(data)  # 标准化处理
    return processed_data

# 特征提取模块
def feature_extraction(processed_data):
    peaks, _ = find_peaks(processed_data)  # 检测峰值
    peak_values = processed_data[peaks]
    return peak_values

# 异常检测模块
def anomaly_detection(peak_values, threshold):
    anomalies = peak_values[peak_values > threshold]  # 根据阈值检测异常
    return anomalies

# 主函数
def main():
    num_sensors = 10
    threshold = 0.5

    # 初始化传感器
    sensors = [np.random.rand() for _ in range(num_sensors)]

    # 采集数据
    data = data_collection(sensors)

    # 数据预处理
    processed_data = data_preprocessing(data)

    # 特征提取
    peak_values = feature_extraction(processed_data)

    # 异常检测
    anomalies = anomaly_detection(peak_values, threshold)

    # 结果展示
    plt.plot(processed_data)
    plt.scatter(peaks, processed_data[peaks], color='r')  # 标记异常值
    plt.show()

    # 输出异常值
    print("异常值：", anomalies)

if __name__ == "__main__":
    main()
