    import networkx as nx

    def dijkstra_path(G, source, target):
        path, _ = nx.single_source_dijkstra(G, source, target)

    G = nx.DiGraph()
    G.add_edges_from([(1, 2, {'weight': 3}), (1, 3, {'weight': 1}), (2, 4, {'weight': 1}), (3, 4, {'weight': 1})])
