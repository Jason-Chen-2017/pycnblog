```python
def preprocess_text(text):
    # 去除HTML标签
    text = re.sub('<[^>]*>', '', text)
    
    # 转小写
    text = text.lower()
    
    # 去除标点符号
    text = re.sub(r'[^\w\s]', '', text)
    
    # 分词
    tokens = word_tokenize(text)
    
    # 去除停用词
    from nltk.corpus import stopwords
    stop_words = set(stopwords.words('english'))
    tokens = [word for word in tokens if word not in stop_words]
    
    return tokens
```

**自动分析模块：**

```python
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

def analyze_text(text, model):
    # 预处理文本
    tokens = preprocess_text(text)
    
    # 向量化文本
    vectorizer = TfidfVectorizer()
    text_vector = vectorizer.fit_transform([' '.join(tokens)])
    
    # 计算文本相似度
    similarity_scores = cosine_similarity(text_vector, model['vectors'])
    
    # 选择相似度最高的结果
    top_similarity = similarity_scores[0].argsort()[-2:][::-1]
    
    # 返回相似度最高的标签和分数
    labels = [model['labels'][i] for i in top_similarity]
    scores = [similarity_scores[0][i] for i in top_similarity]
    
    return labels, scores
```
