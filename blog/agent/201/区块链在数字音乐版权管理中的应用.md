    def __init__(self, name, owner):
        self.name = name
        self.owner = owner
        self.status = "Unsold"

class SmartContract:
    def __init__(self):
        self.assets = []

    def add_asset(self, asset):
        self.assets.append(asset)

    def sell_asset(self, asset_name, buyer):
        for asset in self.assets:
            if asset.name == asset_name and asset.status == "Unsold":
                asset.owner = buyer
                asset.status = "Sold"
                return True
        return False

```python
# blockchain.py
from web3 import Web3
from solc import compile_source
from textwrap import dedent

# 编写智能合约代码
smart_contract_code = dedent("""
    // SPDX-License-Identifier: MIT
    pragma solidity ^0.8.0;

    contract MusicCopyright {
        struct Asset {
            string name;
            address owner;
            bool isSold;
        }

        mapping(string => Asset) public assets;

        function registerAsset(string memory _name, address _owner) public {
            assets[_name] = Asset(_name, _owner, false);
        }

        function sellAsset(string memory _name, address _buyer) public {
            Asset memory asset = assets[_name];
            require(asset.owner != address(0), "Asset not found");
            require(!asset.isSold, "Asset already sold");

            asset.owner = _buyer;
            asset.isSold = true;

            assets[_name] = asset;
        }
    }
""")

# 编译智能合约
compiled_sol = compile_source(smart_contract_code)
contract_interface = compiled_sol["contracts"]["MusicCopyright.json"]

# 部署智能合约
w3 = Web3(Web3.HTTPProvider('http://127.0.0.1:8545'))
contract = w3.eth.contract(abi=contract_interface['abi'])
bytecode = contract_interface['bin']

# 部署合约实例
contract_instance = contract.constructor().transact()
tx_hash = w3.toHex(contract_instance)
print(f"Contract deployed at address: {tx_hash}")
```
