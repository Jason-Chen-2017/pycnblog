启蒙运动的起源可以追溯到17世纪末期，当时法国哲学家笛卡尔（René Descartes）的哲学思想对欧洲学术界产生了深远影响。笛卡尔的“我思故我在”（Cogito, ergo sum）强调了个人的独立思考能力，为后来的启蒙运动奠定了基础。同时，英国经验主义者如约翰·洛克（John Locke）强调个人权利和社会契约理论，也为启蒙运动提供了重要的理论支持。

from flask import Flask, request, jsonify

from sqlalchemy.orm import sessionmaker

# 哲学思想类
class Philosophy:
    def __init__(self, name, description):
        self.name = name
        self.description = description

    def save_to_db(self):
        session.add(self)
        session.commit()

    def delete_from_db(self):
        session.delete(self)
        session.commit()

    def update_in_db(self, new_description):
        self.description = new_description
        session.commit()

# 哲学思想接口
@app.route('/philosophy', methods=['POST'])
def add_philosophy():
    data = request.json
    philosophy = Philosophy(data['name'], data['description'])
    philosophy.save_to_db()
    return jsonify({"message": "哲学思想添加成功"}), 201

@app.route('/philosophy', methods=['DELETE'])
def delete_philosophy():
    data = request.json
    philosophy = session.query(Philosophy).filter_by(name=data['name']).first()
    if philosophy:
        philosophy.delete_from_db()
        return jsonify({"message": "哲学思想删除成功"}), 200
    else:
        return jsonify({"error": "未找到指定的哲学思想"}), 404

@app.route('/philosophy', methods=['PUT'])
def update_philosophy():
    data = request.json
    philosophy = session.query(Philosophy).filter_by(name=data['name']).first()
    if philosophy:
        philosophy.update_in_db(data['description'])
        return jsonify({"message": "哲学思想更新成功"}), 200
    else:
        return jsonify({"error": "未找到指定的哲学思想"}), 404

if __name__ == '__main__':
    app.run(debug=True)
