```python
# 初始化数据
T = data

# 初始化参数

# 迭代优化
for i in range(num_iterations):
    error = T - A @ B @ C
    dA = -2 * (error @ B @ C).T
    dB = -2 * (A.T @ error @ C).T
    dC = -2 * (A.T @ B.T @ error).T
    A -= learning_rate * dA
    B -= learning_rate * dB
    C -= learning_rate * dC

print(f"Final C:\n{C}")
```
