**5.2 系统核心实现**  

以下是系统核心实现的源代码：

```python
import numpy as np
from scipy.optimize import minimize
import matplotlib.pyplot as plt
from mermaid import mermaid

def objective_function(x):
    return x[0]**2 + x[1]**2

def constraint1(x):
    return x[0] - x[1]

def constraint2(x):
    return x[0] + x[1] - 10

constraints = ({'type': 'eq', 'fun': constraint1}, {'type': 'eq', 'fun': constraint2})

def optimize_structure(demand):
    result = minimize(objective_function, demand, constraints=constraints)
    return result.x

def plot_solution(solution):
    x1, x2 = solution
    plt.scatter(x1, x2)
    plt.plot(x1, x2, 'r')
    plt.xlabel('长度')
    plt.ylabel('宽度')
    plt.show()

def main():
    demand = np.array([1, 2])
    solution = optimize_structure(demand)
    print(solution)
    plot_solution(solution)

if __name__ == '__main__':
    main()
```
