import numpy as np
import matplotlib.pyplot as plt

def calculate_contact_angle(material_property):
    liquid_tension = material_property['liquid_tension']
    solid_tension = material_property['solid_tension']
    return np.arccos((liquid_tension - solid_tension) / liquid_tension)

material_property = {
    'liquid_tension': 72,
    'solid_tension': 20
}

contact_angle = calculate_contact_angle(material_property)
print(f"接触角为：{np.degrees(contact_angle)}度")
```
