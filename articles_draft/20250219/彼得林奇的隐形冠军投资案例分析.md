import pandas as pd
import numpy as np

def calculate_invisible_champion_score(company_data):
    market_share = company_data['市场份额']
    profit_margin = company_data['净利润率']
    competitive_advantage = company_data['竞争优势']
    management_stability = company_data['管理稳定性']
    
    score = (market_share * profit_margin * competitive_advantage * management_stability)
    return score

# 示例数据
data = {
    '市场份额': 0.3,
    '净利润率': 0.2,
    '竞争优势': 0.8,
    '管理稳定性': 0.7
}

score = calculate_invisible_champion_score(data)
print(f"隐形冠军评分: {score}")
