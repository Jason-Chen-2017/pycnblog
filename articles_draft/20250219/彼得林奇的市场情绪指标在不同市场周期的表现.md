```python
import pandas as pd
import matplotlib.pyplot as plt

def calculate_market_mood(index_data):
    average_return = index_data['return'].mean()
    volatility = index_data['return'].std()
    mood_index = average_return / volatility
    return mood_index

# 示例数据
data = {
    'return': [0.02, 0.01, -0.03, 0.05, -0.02]
}
index_data = pd.DataFrame(data)

mood_index = calculate_market_mood(index_data)
print(f"市场情绪指数为：{mood_index}")

# 可视化
plt.plot(index_data['return'], label='回报率')
plt.xlabel('时间')
plt.ylabel('回报率')
plt.legend()
plt.show()
```
