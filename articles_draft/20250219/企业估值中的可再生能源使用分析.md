import numpy as np
import matplotlib.pyplot as plt

# 数据处理
data_processed = data.dropna()
data_processed['EnergyConsumption'] = data_processed['EnergyConsumption'].apply(lambda x: x.strip())

# 模型建立

# 模型计算
predicted_cost = np.polyval(model, data_processed['EnergyConsumption'])

# 结果输出

plt.plot(data_processed['EnergyConsumption'], predicted_cost, color='red')
plt.xlabel('Energy Consumption')
plt.ylabel('Cost')
plt.show()
