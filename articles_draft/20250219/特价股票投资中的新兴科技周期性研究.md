## 4.2 数据处理与分析

### 4.2.1 数据采集
```python
import pandas as pd
import requests
from datetime import datetime, timedelta

# 获取过去5年的历史数据
today = datetime.today().strftime('%Y-%m-%d')
five_years_ago = (datetime.today() - timedelta(days=5*365)).strftime('%Y-%m-%d')

# 下载数据
url = f'https://query1.finance.yahoo.com/ajax/finance/get-historical-data/{stock_ticker}?formatted=true&start={five_years_ago}&end={today}&crumb=latest'
response = requests.get(url)
data = response.json()
```

### 4.2.2 数据预处理
```python
df = pd.DataFrame(data['finance']['result'][0]['historical'])
df['Date'] = pd.to_datetime(df['date'])
df.set_index('Date', inplace=True)
df = df[['open', 'high', 'low', 'close', 'volume']]
df.head()
```

### 4.2.3 数据可视化
```python
df['close'].plot(figsize=(10,6))

## 4.3 模型训练与预测

### 4.3.1 数据分割
```python
train_size = int(len(df) * 0.7)
train_data, test_data = df[:train_size], df[train_size:]
```

### 4.3.2 使用LSTM进行预测
```python
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import LSTM, Dense

# 创建模型
model = Sequential()
model.add(LSTM(128, input_shape=(None, 5)))
model.add(Dense(1))
model.compile(loss='mean_squared_error', optimizer='adam')

# 训练模型
train_x = train_data[['open', 'high', 'low', 'close', 'volume']].values.reshape(-1, 5, 1)
train_y = train_data['close'].values.reshape(-1, 1)
model.fit(train_x, train_y, epochs=100, batch_size=32)
```

### 4.3.3 模型评估
```python
test_x = test_data[['open', 'high', 'low', 'close', 'volume']].values.reshape(-1, 5, 1)
test_y = test_data['close'].values.reshape(-1, 1)
predicted_prices = model.predict(test_x).reshape(-1)

# 可视化结果
plt.figure(figsize=(10,6))
plt.plot(test_data.index, test_y, label='实际价格')
plt.plot(test_data.index, predicted_prices, label='预测价格')
plt.title('股票价格预测结果')

### 4.4.2 模型优化
```python
from sklearn.preprocessing import MinMaxScaler

# 标准化处理
scaler = MinMaxScaler()
train_scaled = scaler.fit_transform(train_data)
test_scaled = scaler.transform(test_data)

# 创建数据集
def create_dataset(data, look_back=30):
    X, Y = [], []
    for i in range(len(data) - look_back):
        X.append(data[i:i+look_back])
        Y.append(data[i+look_back])
    return np.array(X), np.array(Y)

train_X, train_Y = create_dataset(train_scaled, look_back=30)
test_X, test_Y = create_dataset(test_scaled, look_back=30)

# 调整数据形状
train_X = train_X.reshape(train_X.shape[0], train_X.shape[1], 1)
test_X = test_X.reshape(test_X.shape[0], test_X.shape[1], 1)
```

### 4.4.3 模型预测与分析
```python
predicted = model.predict(test_X)
predicted = scaler.inverse_transform(predicted.reshape(-1, 1))
test_Y_actual = scaler.inverse_transform(test_Y.reshape(-1, 1))

# 计算MAE和RMSE
from sklearn.metrics import mean_absolute_error, mean_squared_error

mae = mean_absolute_error(test_Y_actual, predicted)
rmse = mean_squared_error(test_Y_actual, predicted)**0.5

print(f'MAE: {mae}')
print(f'RMSE: {rmse}')
