import pandas as pd
import numpy as np

def calculate_liquidity(stock_data):
    # 计算流动性指标
    volume = stock_data['volume']
    turnover = stock_data['turnover']
    liquidity_ratio = turnover / volume
    return liquidity_ratio

def graham_strategy(liquidity_ratio, stock_price):
    # 格雷厄姆策略实现
    intrinsic_value = stock_price * 2.5
    margin_of_safety = intrinsic_value * 0.75
    return margin_of_safety

# 示例数据
data = {
    'volume': [1000, 2000, 1500],
    'turnover': [50000, 100000, 75000],
    'stock_price': [50, 60, 40]
}

df = pd.DataFrame(data)
liquidity_ratio = calculate_liquidity(df)
margin_of_safety = graham_strategy(liquidity_ratio, df['stock_price'])
print("Margin of Safety:", margin_of_safety)
