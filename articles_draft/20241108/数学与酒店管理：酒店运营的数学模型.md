```python
# 需求预测项目代码示例
import pandas as pd
from statsmodels.tsa.arima.model import ARIMA

# 读取数据
data = pd.read_csv('customer_data.csv')
time_series = data['customer_demand']

# 建立ARIMA模型
model = ARIMA(time_series, order=(5, 1, 2))
model_fit = model.fit()

# 预测未来三个月的需求
forecast = model_fit.forecast(steps=3)

# 输出预测结果
print(forecast)
```
