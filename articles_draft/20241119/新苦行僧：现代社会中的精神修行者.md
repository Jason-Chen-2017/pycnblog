```python
import time
import random

def intro():
    print("欢迎来到冥想练习。请确保在安静的环境中开始练习。")

def focus_on_breath():
    print("专注于你的呼吸。吸气... 呼气...")

def check_for_distractinos():
    while True:
        distraction = random.choice(["手机响铃", "人声喧哗", "门窗响动"])
        if input(f"你被以下干扰了：{distraction}（输入‘继续’继续练习）\n") == "继续":
            break
        else:
            print("休息一会儿，然后继续练习。")

def end():
    print("冥想练习完成。请记得记录你的感受和效果。")

def main():
    intro()
    time.sleep(3)
    focus_on_breath()
    check_for_distractinos()
    end()

if __name__ == "__main__":
    main()
```

- `focus_on_breath()` 函数用于指导用户专注于呼吸。

- `end()` 函数用于结束冥想练习。
