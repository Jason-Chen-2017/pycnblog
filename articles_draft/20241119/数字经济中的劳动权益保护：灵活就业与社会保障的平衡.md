class Employee:
    def __init__(self, name, skills):

        self.skills = skills

    def apply_job(self, job):
        if self.skills == job.required_skills:
            return True
        else:
            return False

class Job:
    def __init__(self, name, required_skills):
        self.name = name
        self.required_skills = required_skills

    def hire_employee(self, employee):
        if employee.apply_job(self):
            return "Hired!"
        else:
            return "Not hired!"

# 实例化类
