#### Python 源代码：天体轨道计算

```python
import numpy as np

def compute_orbit(G, m1, m2, r1, v1):
    """
    计算天体轨道
    :param G: 万有引力常数
    :param m1: 天体1的质量
    :param m2: 天体2的质量
    :param r1: 天体1的初始位置
    :param v1: 天体1的初始速度
    :return: 天体轨道
    """
    r = r1
    v = v1
    t = 0
    dt = 0.01
    while t < 1000:
        a = -G * m2 * r / np.linalg.norm(r)**3
        v += a * dt
        r += v * dt
        t += dt
    return r, v

# 万有引力常数
G = 6.674 * 10**-11
# 天体1的质量（地球质量）
m1 = 5.972 * 10**24
# 天体2的质量（太阳质量）
m2 = 1.989 * 10**30
# 地球的初始位置
r1 = np.array([1.496 * 10**11, 0, 0])
# 地球的初始速度
v1 = np.array([-29000, 0, 0])

# 计算地球轨道
orbit, velocity = compute_orbit(G, m1, m2, r1, v1)

# 打印结果
print("地球轨道：", orbit)
print("地球速度：", velocity)
```

#### Python 源代码：最小二乘法拟合轨道

```python
import numpy as np
from scipy.optimize import curve_fit

def orbit_model(t, a, b, c, d):
    """
    天体轨道模型
    :param t: 时间
    :param a: 轨道倾角
    :param b: 升交点黄经
    :param c: 轨道离心率
    :param d: 近心距
    :return: 天体轨道
    """
    theta = 2 * np.pi * (t - a) / (b * 365.25)
    r = b * (1 - c * np.cos(theta)) / (1 + c * np.cos(theta))
    return r * np.array([np.cos(theta), np.sin(theta), 0])

def residual(params, t, y):
    """
    残差函数
    :param params: 模型参数
    :param t: 时间
    :param y: 观测数据
    :return: 残差
    """
    a, b, c, d = params
    y_fit = orbit_model(t, a, b, c, d)
    return y - y_fit

# 观测数据（示例）
t = np.array([1, 2, 3, 4, 5])
y = np.array([2.5, 3.2, 4.1, 3.9, 4.8])

# 拟合轨道参数
params_init = [0, 1, 0, 0]
params, cov = curve_fit(residual, t, y, p0=params_init)

# 打印拟合结果
print("拟合参数：", params)
print("参数协方差矩阵：", cov)

# 计算拟合轨道
t_fit = np.linspace(t.min(), t.max(), 100)
y_fit = orbit_model(t_fit, *params)

# 打印拟合轨道
print("拟合轨道：", y_fit)
```
