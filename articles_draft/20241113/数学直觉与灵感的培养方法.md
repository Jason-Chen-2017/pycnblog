```python
import numpy as np

# 牛顿迭代法解方程
def newton_method(a, b, c, x0):
    x = x0
    while abs(x - np.sqrt(a * x * x + b * x + c)) > 1e-5:
        x = (x + a / x) / 2
    return x

# 初始值
x0 = 1.0

# 解方程
solution = newton_method(1, -5, 6, x0)

print(f"The solution is: {solution}")
```
