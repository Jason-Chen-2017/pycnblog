function FFT(x):
    if length(x) <= 1:
        return x
    else:
        N = length(x)
        even = FFT(x[0:N/2])
        odd = FFT(x[N/2:N])
        return [2 * cos(2*pi*k/N) * even + 2 * sin(2*pi*k/N) * odd for k in 0:N/2]

# 最小二乘法（Least Squares）
function LeastSquares(x, y):
    A = [x^2 for x in x]
    b = y
    return (A.T * A).inv() * A.T * b
