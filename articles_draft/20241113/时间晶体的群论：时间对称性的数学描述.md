```python
G = {'E': np.eye(3), 'C2': np.array([[0, -1], [1, 0]])}
```

**开发环境搭建：**

```python
import numpy as np

# 定义时间晶体
lattice = np.array([[1, 0], [0, 1]])

# 定义对称性变换
def symmetry_transformation(symmetry_group):
    if symmetry_group == 'E':
        return np.eye(2)
    elif symmetry_group == 'C2':
        return np.array([[0, -1], [1, 0]])

# 应用对称性变换
def apply_transformation(lattice, symmetry_group):
    transformation = symmetry_transformation(symmetry_group)
    return np.dot(lattice, transformation)

# 测试
print(apply_transformation(lattice, 'C2'))
```
