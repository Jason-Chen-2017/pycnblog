def sigmoid(x):
    return 1 / (1 + np.exp(-x))

input_data = np.array([0.5, 0.7, 0.8])
weight = np.array([-0.1, 0.2, -0.3])
bias = 0.5

z = np.dot(input_data, weight) + bias
output = sigmoid(z)

print(f"Output: {output}")
