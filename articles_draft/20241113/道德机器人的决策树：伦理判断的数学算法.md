def build_tree(data, depth):
    if len(data) == 0 or depth == 0:
        return LeafNode
    else:
        best_feature = find_best_feature(data)
        node = TreeNode(best_feature)
        for value in best_feature.unique():
            subset = data[best_feature == value]
            node.children[value] = build_tree(subset, depth - 1)
        return node

# 决策树预测函数
def predict(node, sample):
    if isinstance(node, LeafNode):
        return node.label
    else:
        value = sample[node.feature]
        return predict(node.children[value], sample)
