#### 神经元放电模式模拟

```python
import numpy as np
import matplotlib.pyplot as plt

def simulate_neuron(frequency, noise_intensity, resting_potential, threshold_potential, time_step, total_time):
    times = np.arange(0, total_time, time_step)
    membrane_potentials = []

    for t in times:
        noise = noise_intensity * np.random.normal(0, 1)
        membrane_potential = resting_potential + noise
        if membrane_potential >= threshold_potential:
            print(f"{t}秒：动作电位")
            membrane_potentials.append(membrane_potential)
        else:
            print(f"{t}秒：静息状态")
            membrane_potentials.append(membrane_potential)

    plt.plot(times, membrane_potentials)
    plt.xlabel("Time (s)")
    plt.ylabel("Membrane Potential (mV)")
    plt.show()
```
