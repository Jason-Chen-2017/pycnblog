### 附录C: 净净股策略的Python实现示例

```python
import pandas as pd
import numpy as np

def calculate_intrinsic_value(financial_data):
    # 计算内在价值
    intrinsic_value = (financial_data['Earnings'] + financial_data['Cash Flow']) / (financial_data['Interest Rate'] + financial_data['Growth Rate'])
    return intrinsic_value

def net_net_strategy(stock_data, threshold=1.2):
    # 筛选市净率低于阈值的股票
    selected_stocks = stock_data[stock_data['P/B'] < threshold]
    return selected_stocks

# 示例数据加载
data = pd.read_csv('stock_data.csv')

# 应用净净股策略
selected = net_net_strategy(data)
print("筛选出的股票数量:", len(selected))
```
