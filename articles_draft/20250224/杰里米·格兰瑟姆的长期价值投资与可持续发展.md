```python
def calculate_esg_rating(company_data):
    # 计算环境评分
    environmental_score = company_data['CO2排放'] * -1 + 100
    # 计算社会评分
    social_score = company_data['员工满意度'] * 0.7 + company_data['社会责任'] * 0.3
    # 计算治理评分
    governance_score = company_data['管理层稳定性'] * 0.5 + company_data['董事会独立性'] * 0.5
    # 综合ESG评分
    esg_rating = (environmental_score + social_score + governance_score) / 3
    return esg_rating
```
