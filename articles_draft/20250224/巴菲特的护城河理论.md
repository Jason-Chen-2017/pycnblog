```python
from flask import Flask, request, jsonify

app = Flask(__name__)

def calculate_h护城河评分(企业数据):
    成本优势 = 企业数据['成本优势']
    网络效应 = 企业数据['网络效应']
    品牌影响力 = 企业数据['品牌影响力']
    法律壁垒 = 企业数据['法律壁垒']
    总评分 = (成本优势 * 0.3 + 网络效应 * 0.2 + 品牌影响力 * 0.3 + 法律壁垒 * 0.2)
    return 总评分

@app.route('/评估', methods=['POST'])
def 评估():
    data = request.json
    评分 = calculate_h护城河评分(data)
    return jsonify({'评分': 评分})

if __name__ == '__main__':
    app.run(debug=True)
```
