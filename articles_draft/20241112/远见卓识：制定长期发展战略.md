import pandas as pd

# SWOT分析
def swot_analysis(data):
    # 读取数据
    swot_data = pd.read_excel(data)

    # 分析数据
    strengths = swot_data['Strengths'].sum()
    weaknesses = swot_data['Weaknesses'].sum()

    threats = swot_data['Threats'].sum()

    # 输出结果
    return {
        'Strengths': strengths,
        'Weaknesses': weaknesses,
        'Opportunities': opportunities,
        'Threats': threats
