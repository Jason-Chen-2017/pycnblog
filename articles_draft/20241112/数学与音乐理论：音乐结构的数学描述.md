#### 源代码实现

```python
import numpy as np
import matplotlib.pyplot as plt

def generate_melody(frequencies, durations):
    """
    生成旋律。
    
    参数:
    frequencies: 音高频率列表。
    durations: 每个音符的持续时间列表。
    """
    # 使用正弦波生成每个音符
    waveforms = [np.sin(2 * np.pi * f * t) for f, t in zip(frequencies, durations)]
    # 求和所有波形
    melody = np.sum(waveforms, axis=0)
    return melody

def plot_melody(melody, frequencies):
    """
    绘制旋律图。
    
    参数:
    melody: 旋律信号。
    frequencies: 音高频率列表。
    """
    # 计算每个音符的时域波形
    durations = np.diff(np.argmax(melody, axis=0))
    time_axis = np.cumsum(durations)
    # 绘制时域波形图
    plt.plot(time_axis, melody)
    # 标注频率
    for i, f in enumerate(frequencies):
        plt.axvline(x=time_axis[i], label=f'{f} Hz')
    plt.xlabel('Time (s)')
    plt.ylabel('Amplitude')
    plt.legend()
    plt.show()

# 生成一个简单的旋律
frequencies = [440, 550, 660, 770]
durations = [1, 0.5, 1, 0.5]
melody = generate_melody(frequencies, durations)

# 绘制旋律
plot_melody(melody, frequencies)
```

- `generate_melody` 函数用于生成旋律信号。
- `plot_melody` 函数用于绘制旋律时域波形图。
