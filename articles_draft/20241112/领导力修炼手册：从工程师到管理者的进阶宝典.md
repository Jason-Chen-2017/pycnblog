import git
import slack

# 配置Git仓库
git.init()

# 配置Slack通知
slack_bot = slack.SlackBot(token='your-slack-token')

# 检查代码提交
def check_code_commit():
    commits = git.log()
    for commit in commits:
        if commit['message'].startswith('Bugfix'):
            slack_bot.send_message('bugfix-commit-alert', f"A bugfix commit was made: {commit['message']}")
        elif commit['message'].startswith('Feature'):
            slack_bot.send_message('feature-commit-alert', f"A new feature was added: {commit['message']}")
```
