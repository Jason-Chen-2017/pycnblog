```python
    qualified_suppliers = []
    for supplier in candidates:
        if evaluate_supplier(supplier):
            qualified_suppliers.append(supplier)
    return qualified_suppliers

def evaluate_supplier(supplier):
    score = 0
    if supplier.has_relevant_experience():
        score += 1
    if supplier.meets_quality_standards():
        score += 1
    return score >= 2
```

  - **源代码实现**：
    ```python
    import mysql.connector

    def filter_suppliers():
        connection = mysql.connector.connect(
            host="localhost",
            user="yourusername",
            password="yourpassword",
            database="yourdatabase"
        )

        cursor = connection.cursor()

        cursor.execute("SELECT * FROM suppliers")
        suppliers = cursor.fetchall()

        qualified_suppliers = []
        for supplier in suppliers:
            if evaluate_supplier(supplier):
                qualified_suppliers.append(supplier)

        return qualified_suppliers

    def evaluate_supplier(supplier):
        score = 0
        if supplier[1] == "Yes":
            score += 1
        if supplier[2] == "Yes":
            score += 1
        return score >= 2

    # 执行筛选流程
    qualified_suppliers = filter_suppliers()
    for supplier in qualified_suppliers:
        print(supplier)
    ```

    - 该脚本使用 MySQL 连接数据库，查询供应商表中的数据
