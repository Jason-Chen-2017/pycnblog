```python
import pandas as pd
import numpy as np

def calculate_intrinsic_value(eps, growth_rate, required_return):
    # 计算未来现金流现值
    terminal_value = eps * (1 + growth_rate) / (required_return - growth_rate)
    # 内在价值公式
    intrinsic_value = terminal_value / (1 + required_return)
    return intrinsic_value

# 示例数据
data = {
    '公司': ['公司A', '公司B', '公司C'],
    'EPS': [1.5, 2.0, 1.8],
    '增长率': [0.05, 0.07, 0.06],
    '要求回报率': [0.10, 0.12, 0.11]
}

df = pd.DataFrame(data)

# 计算内在价值
df['内在价值'] = df.apply(lambda x: calculate_intrinsic_value(x['EPS'], x['增长率'], x['要求回报率']), axis=1)

print(df)
```
