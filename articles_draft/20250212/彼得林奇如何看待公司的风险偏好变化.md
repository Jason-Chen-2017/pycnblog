### 11.2 系统核心实现

import pandas as pd
import numpy as np

def calculate_risk_preference(data):
    # 计算风险因素
    risk_factors = data['风险因素']
    # 量化风险
    risk_scores = pd.Series(risk_factors).rank()
    return risk_scores

# 示例数据
data = {
    '风险因素': ['高', '中', '低'],
    '预期收益': [10, 5, 2]
}
result = calculate_risk_preference(data)
print(result)
