def compute_intrinsic_value(financials):

def trading_strategy(data, threshold=0.1):
    # 判断价格是否低于价值
    value = compute_intrinsic_value(data)
    if data['price'][-1] < value * (1 - threshold):
        return 'BUY'
    elif data['price'][-1] > value * (1 + threshold):
        return 'SELL'
    else:
        return 'HOLD'
