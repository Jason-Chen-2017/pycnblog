```python
import pulp

def supply_chain_optimization(suppliers, manufacturers, products):
    # 初始化问题
    prob = pulp.LpProblem("Supply_Chain_Optimization", pulp.LpMinimize)
    
    # 定义变量
    x = pulp.LpVariable.dicts("x", suppliers, lowInclusive=False)
    
    # 定义目标函数
    prob += pulp.lpSum([products[i] * x[i] for i in suppliers]), "Total_Cost"
    
    # 添加约束
    for product in products:
        prob += pulp.lpSum([x[i] * suppliers[i][product] for i in suppliers]) >= manufacturers[product], f"Supply_{product}"
    
    # 求解问题
    prob.solve()
    
    # 输出结果
    return {i: x[i].value() for i in suppliers}

# 示例数据
suppliers = {
    'S1': {'P1': 10, 'P2': 20},
    'S2': {'P1': 15, 'P2': 25}
}
manufacturers = {'P1': 50, 'P2': 75}
products = ['P1', 'P2']

# 调用优化函数
result = supply_chain_optimization(suppliers, manufacturers, products)
print(result)
```
