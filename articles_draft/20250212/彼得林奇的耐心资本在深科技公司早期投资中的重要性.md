```python
# 耐心资本评估模型
def patience_capital_assessment(revenue, growth_rate, risk_factor):
    return (revenue * (1 + growth_rate) / risk_factor)

# 示例
revenue = 1000000
growth_rate = 0.2
risk_factor = 1.5
print(patience_capital_assessment(revenue, growth_rate, risk_factor))
```
