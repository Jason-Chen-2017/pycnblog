  import pandas as pd
  import numpy as np

  def calculate_pe(stock_price, eps):
      return stock_price / eps

  def calculate_dynamic_pe(stock_price, projected_eps):
      return stock_price / projected_eps

  # 示例数据
  stock_price = 50
  eps = 2.5
  projected_eps = 3

  pe = calculate_pe(stock_price, eps)
  dynamic_pe = calculate_dynamic_pe(stock_price, projected_eps)
