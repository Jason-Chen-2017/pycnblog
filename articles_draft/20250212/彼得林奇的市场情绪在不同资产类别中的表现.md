- 3.4 市场情绪的Python代码实现
  ```python
  import pandas as pd
  import numpy as np

  # 示例数据
  data = {
      '日期': ['2023-01-01', '2023-01-02', '2023-01-03'],
      '资产价格': [100, 105, 102],
      '市场情绪': [0.6, 0.8, 0.7]
  }
  df = pd.DataFrame(data)

  # 计算市场情绪指数
  def calculate_market_sentiment(dataframe):
      # 示例：计算移动平均数和相对强弱指数（RSI）
      dataframe['MA'] = dataframe['资产价格'].rolling(2).mean()
      dataframe['RSI'] = (dataframe['资产价格'] - dataframe['资产价格'].shift(1)) / dataframe['资产价格'].shift(1)
      dataframe['市场情绪指数'] = 0.5 * dataframe['MA'] + 0.5 * dataframe['RSI']
      return dataframe

  df = calculate_market_sentiment(df)
  print(df)
  ```

  import pandas as pd
  import matplotlib.pyplot as plt

  # 数据读取
  data = pd.read_csv('market_data.csv')

  # 计算市场情绪指数
  def calculate_sentiment(data):
      # 示例：计算移动平均数和标准差
      data['MA'] = data['资产价格'].rolling(5).mean()
      data['STD'] = data['资产价格'].rolling(5).std()
      data['市场情绪指数'] = (data['MA'] / data['STD']).rank(pct=True)
      return data

  df = calculate_sentiment(data)
  df.to_csv('result.csv', index=False)

  # 可视化
  plt.figure(figsize=(10, 6))
  plt.plot(df['日期'], df['市场情绪指数'], label='市场情绪指数')
