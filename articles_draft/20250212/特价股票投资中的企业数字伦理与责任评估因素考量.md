  ```python
      # 公司治理评分
      governance_score = company_info['governance'] * 0.3
      # 财务健康评分
      financial_score = company_info['financial_health'] * 0.4
      # 社会责任评分
      csr_score = company_info['csr'] * 0.3
      # 总分
      total_score = governance_score + financial_score + csr_score
      return total_score
  ```
