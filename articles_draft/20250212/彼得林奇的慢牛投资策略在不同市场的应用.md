```python
import pandas as pd
import numpy as np

def calculate_intrinsic_value(company_data, discount_rate):
    """
    计算公司股票的内在价值
    """
    E = company_data['future_cash_flows'].sum()
    k = discount_rate / 100
    V = E / k
    return V

def calculate_pe_ratio(company_data):
    """
    计算市盈率
    """
    price = company_data['stock_price']
    earnings = company_data['earnings_per_share']
    pe_ratio = price / earnings
    return pe_ratio

def calculate_pb_ratio(company_data):
    """
    计算市净率
    """
    price = company_data['stock_price']
    book_value = company_data['book_value_per_share']
    pb_ratio = price / book_value
    return pb_ratio
```

```python
def generate_investment_recommendation(company_data, threshold=0.2):
    """
    生成投资建议
    """
    pe_ratio = calculate_pe_ratio(company_data)
    pb_ratio = calculate_pb_ratio(company_data)
    if pe_ratio < threshold and pb_ratio < threshold:
        return "买入"
    elif pe_ratio > threshold * 2 and pb_ratio > threshold * 2:
        return "卖出"
    else:
        return "持有"
```
