```python
# 人才价值评估模型
def calculate_value_index(skills, potential, loyalty):
    return 0.5 * skills + 0.3 * potential + 0.2 * loyalty

# 数据处理
import pandas as pd

# 示例数据
data = {
    '员工': ['A', 'B', 'C'],
    '技能': [85, 90, 88],
    '潜力': [70, 80, 75],
    '忠诚度': [90, 85, 88]
}

df = pd.DataFrame(data)
df['价值指数'] = df.apply(lambda x: calculate_value_index(x['技能'], x['潜力'], x['忠诚度']), axis=1)

print(df)
```
