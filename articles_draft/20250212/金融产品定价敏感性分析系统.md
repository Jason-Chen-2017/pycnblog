  import numpy as np

  def monte_carlo_simulation(n_simulations, function, params):
      results = []
      for _ in range(n_simulations):
          random_params = {param: np.random.rand() for param in params}
          result = function(random_params)
          results.append(result)
      return results

  def example_function(params):
      return params['a'] * params['b'] + params['c']

  n_simulations = 1000
  params = ['a', 'b', 'c']
  results = monte_carlo_simulation(n_simulations, example_function, params)

  import matplotlib.pyplot as plt

  def calculate_sensitivity(prices, variables):
      # 计算敏感性指标
      results = {}
      for var in variables:
          results[var] = sum(abs(p - var)**2 for p in prices) / len(prices)
      return results

  prices = [100, 101, 98, 99, 102]
  variables = ['volatility', 'interest_rate']
