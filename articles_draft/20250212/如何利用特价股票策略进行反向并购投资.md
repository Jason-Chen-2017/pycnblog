import pandas as pd
import numpy as np
from sklearn.metrics import pairwise_distances

# 示例数据：目标公司列表
companies = pd.DataFrame({
    '公司名称': ['公司A', '公司B', '公司C'],
    '市盈率': [10, 15, 20],
    '市净率': [1.5, 2.0, 2.5]
})

# 计算相似性矩阵
distance_matrix = pairwise_distances(companies[['市盈率', '市净率']], metric='manhattan')

# 找出最相似的目标公司
def find_similarCompanies(target, distance_matrix, threshold=5):
    similar = []
    for i in range(len(distance_matrix)):
        if distance_matrix[target][i] < threshold:
            similar.append(i)
    return similar

# 示例：寻找与公司A相似的目标公司
similar = find_similarCompanies(0, distance_matrix)
print(similar)
```
