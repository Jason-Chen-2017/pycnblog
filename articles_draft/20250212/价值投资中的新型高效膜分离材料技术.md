from scipy.optimize import minimize

# 定义膜分离效率函数
def membrane_efficiency(x, data):
    params = x
    efficiency = 0
    for i in range(len(data)):
        if data[i] > params[0]:
            efficiency += 1
    return -efficiency

# 优化参数
