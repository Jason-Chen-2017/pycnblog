### 第2章：区块链技术在数字艺术品拍卖中的应用原理

#### 2.1 区块链技术的核心原理

- **区块链的基本定义与工作原理**：
  区块链是一种去中心化的分布式账本技术，通过密码学和共识机制确保数据的安全性和一致性。每个区块包含交易数据和时间戳，并通过哈希值连接成链，形成不可篡改的区块链。

- **区块链的核心特点**：
  - **去中心化**：数据存储在多个节点上，避免单点故障。
  - **不可篡改性**：每个区块的哈希值确保数据无法被修改。
  - **透明性**：所有交易记录在区块链上公开可查，但交易双方的身份可以匿名。

- **区块链在数字艺术品拍卖中的应用优势**：
  - 确保数字艺术品的 authenticity 和 ownership。
  - 提供透明的拍卖流程，增强市场信任。
  - 自动化执行智能合约，降低交易成本和风险。

  contract NFTAuction {
    struct Auction {
        uint256 tokenId;
        uint256 startPrice;
        uint256 currentPrice;
        address payable owner;
        bool isLive;
    }

    mapping(uint256 => Auction) public auctions;

    function createAuction(uint256 tokenId, uint256 startPrice) public {
        auctions[tokenId] = Auction({
            tokenId: tokenId,
            startPrice: startPrice,
            currentPrice: startPrice,
            owner: payable(msg.sender),
            isLive: true
        });
    }

    function bid(uint256 tokenId) public payable {
        Auction storage auction = auctions[tokenId];
        require(auction.isLive, "Auction is not live");
        require(msg.value > auction.currentPrice, "Bid must be higher than current price");

        auction.currentPrice = msg.value;
        auction.owner.transfer(msg.value - auction.startPrice);
        emit BidEvent(tokenId, msg.sender, msg.value);
    }

    function endAuction(uint256 tokenId) public {
        Auction storage auction = auctions[tokenId];
        require(auction.isLive, "Auction is not live");
        auction.isLive = false;
    }

    event BidEvent(uint256 indexed tokenId, address indexed bidder, uint256 bidAmount);
  }
