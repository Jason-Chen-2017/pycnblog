def calculate_innovation_index(patents):
    # 计算每个专利的被引用次数
    citations = {}
    for idx, row in patents.iterrows():
        if row['被引用专利ID'] not in citations:
            citations[row['被引用专利ID']] = 0
        citations[row['被引用专利ID']] += 1
    return citations
