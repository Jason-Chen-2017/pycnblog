##### 代码实现

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract ValueInvestment {
    // 投资者信息
    struct Investor {
        uint256 investmentAmount;
        uint256 shares;
        bool isAuthorized;
    }

    // 股票信息
    struct Stock {
        uint256 price;
        uint256 quantity;
        address[] investors;
    }

    // 系统角色
    address public owner;
    mapping(address => Investor) public investors;
    mapping(address => Stock) public stocks;

    constructor() {
        owner = msg.sender;
    }

    // 投资者授权
    modifier onlyAuthorized() {
        require(investors[msg.sender].isAuthorized, "Not authorized");
        _;
    }

    // 投资者存入资金
    function invest(address stockAddress, uint256 amount) external payable onlyAuthorized {
        Stock storage stock = stocks[stockAddress];
        require(stock.price > 0, "Stock not available");

        // 计算可购买的数量
        uint256 quantity = amount / stock.price;
        stock.quantity -= quantity;
        stock.investors.push(msg.sender);

        // 更新投资者信息
        investors[msg.sender].investmentAmount += amount;
        investors[msg.sender].shares += quantity;
    }

    // 自动卖出策略
    function sell(address stockAddress, uint256 quantity) external onlyAuthorized {
        Stock storage stock = stocks[stockAddress];
        require(stock.quantity >= quantity, "Not enough quantity");

        uint256 price = stock.price;
        uint256 total = price * quantity;
        stock.quantity -= quantity;
        stock.investors.pop(); // 假设 FIFO 模型

        // 分配收益
        // 这里可以进一步优化，例如将收益分配给投资者
        // 这里简写，实际实现中需要考虑更多细节
        payable(msg.sender).transfer(total);
    }

    // 管理员设置股票价格
    function setStockPrice(address stockAddress, uint256 price) external {
        stocks[stockAddress].price = price;
    }

    // 查询功能
    function getStockPrice(address stockAddress) external view returns (uint256) {
        return stocks[stockAddress].price;
    }

    // 查询投资者信息
    function getInvestorInfo(address investorAddress) external view returns (uint256, uint256, bool) {
        return (investors[investorAddress].investmentAmount,
                investors[investorAddress].shares,
                investors[investorAddress].isAuthorized);
    }
}
```

##### 功能解读

这段代码实现了一个简单的智能合约，用于管理投资者对某种股票的投资。以下是主要功能的解读：

1. **投资者授权**：只有授权的投资者才能进行投资操作。这通过`onlyAuthorized`修饰器实现，确保只有经过验证的地址可以调用相关函数。

2. **投资功能**：投资者可以将资金存入智能合约，并购买相应的股票数量。购买的数量由投资金额除以股票价格决定。购买后，投资者的份额和投资金额都会更新。

3. **自动卖出策略**：投资者可以卖出持有的股票。合约中假设使用FIFO（先进先出）模型，卖出最先进入的股票。卖出后，收益会自动Transfer给投资者。

4. **股票价格管理**：管理员可以设置股票的价格，这在实际应用中可能需要进一步的安全措施，比如多签授权，以防止价格被恶意篡改。

5. **查询功能**：投资者可以查询自己购买的股票价格以及自己的投资信息。
