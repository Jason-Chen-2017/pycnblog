import pandas as pd
import numpy as np

def calculate_intrinsic_value(earnings, yield_rate, growth_rate):
    return earnings / (yield_rate - growth_rate)

def calculate_margin_of_safety(intrinsic_value, market_price):
    return intrinsic_value * (1 - (market_price / intrinsic_value))

# 示例数据
data = {
    'Ticker': ['AAPL', 'MSFT', 'GOOGL'],
    'Earnings': [10, 15, 20],
    'Yield': [2, 3, 4],
    'Growth': [5, 4, 3],
    'Price': [300, 250, 280]
}

df = pd.DataFrame(data)

df['Intrinsic Value'] = df.apply(lambda x: calculate_intrinsic_value(x['Earnings'], x['Yield'], x['Growth']), axis=1)
df['Margin of Safety'] = df.apply(lambda x: calculate_margin_of_safety(x['Intrinsic Value'], x['Price']), axis=1)

print(df)
