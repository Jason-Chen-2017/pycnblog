```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# 示例数据
data = {'企业': ['A', 'B', 'C'], '碳排放': [100, 50, 20], '政策支持': [0.8, 0.6, 0.4]}
df = pd.DataFrame(data)

# 计算企业估值
beta = np.array([1, -2, 1])
X = df[['碳排放', '政策支持']]
y = np.dot(X, beta) + np.random.normal(0, 1, len(X))

plt.scatter(X['碳排放'], y)
plt.title('碳排放与企业估值关系')
plt.xlabel('碳排放')
plt.ylabel('企业估值')
plt.show()
```
