### 6.2 核心代码实现
```python
import pandas as pd
import numpy as np

def calculate_brand_value(brand_revenue, brand_strength, cost):
    return brand_revenue * brand_strength - cost

def normalize_brand_value(brand_value, market_size):
    return brand_value / market_size

# 示例数据
brand_revenue = 1000000  # 品牌收入
brand_strength = 0.8      # 品牌强度
cost = 200000            # 成本
market_size = 10000000   # 市场规模

# 计算品牌价值
brand_value = calculate_brand_value(brand_revenue, brand_strength, cost)
# 标准化品牌价值
normalized_value = normalize_brand_value(brand_value, market_size)

print(f"品牌价值：${brand_value}")
print(f"标准化品牌价值：${normalized_value}")
```
