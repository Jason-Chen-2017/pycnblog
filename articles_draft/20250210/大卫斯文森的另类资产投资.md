```python
import numpy as np
import pandas as pd

def optimize_portfolio(returns, weights):
    # 计算协方差矩阵
    covariance_matrix = np.cov(returns.T)
    # 均值-方差优化
    n = len(weights)
    ones = np.ones(n)
    inv_cov = np.linalg.inv(covariance_matrix)
    weights = inv_cov.dot(ones) / (ones.T.dot(inv_cov).dot(ones))
    return weights

# 示例数据
returns = pd.DataFrame({
    '股票': [0.1, 0.05, 0.08],
    '债券': [0.05, 0.04, 0.06],
    '房地产': [0.12, 0.08, 0.1]
})

weights = optimize_portfolio(returns, [1/3, 1/3, 1/3])
print(weights)
```
