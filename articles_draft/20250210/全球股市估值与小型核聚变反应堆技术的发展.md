def calculate_stock_value(stock_data):
    # 计算市盈率
    pe_ratio = stock_data['profit'] / stock_data['price']
    # 计算市净率
    pb_ratio = stock_data['book_value'] / stock_data['price']
    # 综合评估股票价值
    stock_value = (pe_ratio + pb_ratio) / 2
    return stock_value
