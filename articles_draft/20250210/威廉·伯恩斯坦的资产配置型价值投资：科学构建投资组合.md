import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# 示例代码：计算投资组合的期望收益和风险
def portfolio_optimizer(returns, weights):
    expected_return = np.dot(returns.mean(), weights)
    variance = np.dot(weights.T, np.dot(returns.cov(), weights))
    return expected_return, variance

# 数据获取与处理
data = pd.read_csv('stock_data.csv')
returns = data.pct_change().dropna()

# 资产配置权重示例
weights = np.array([0.6, 0.3, 0.1])
expected_return, variance = portfolio_optimizer(returns, weights)
print(f"期望收益: {expected_return:.2%}")
print(f"风险: {variance**0.5:.2%}")
