import pandas as pd
import numpy as np

def slow_bull_strategy(prices, slow_period=20, fast_period=5):
    # 计算慢均线和快均线
    slow_ma = prices.rolling(slow_period).mean()
    fast_ma = prices.rolling(fast_period).mean()
    
    # 生成交易信号
    signals = pd.Series(index=prices.index)
    signals[slow_ma > fast_ma] = 1
    signals[slow_ma < fast_ma] = -1
    signals[signals.isna()] = 0
    
    return signals

# 示例数据
data = pd.read_csv('market_data.csv')
prices = data['close']

# 生成信号
signals = slow_bull_strategy(prices)

# 执行交易
positions = pd.Series(index=prices.index, dtype='int64')
positions[0] = 0
for i in range(1, len(prices)):
    if signals.iloc[i] == 1:
        positions[i] = 1
    elif signals.iloc[i] == -1:
        positions[i] = 0
    else:
        positions[i] = positions[i-1]

# 计算收益
returns = (prices.shift(-1) - prices) * positions.shift(1)
cumulative_returns = (1 + returns).cumprod()
