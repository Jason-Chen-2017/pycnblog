def calculate_pe_ratio(stock_data):
    pe_ratio = stock_data['price'] / stock_data['earnings_per_share']
    return pe_ratio

data = {
    'price': [100, 200, 300],
    'earnings_per_share': [10, 20, 30]
}
df = pd.DataFrame(data)
