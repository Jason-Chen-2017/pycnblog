```python
def brand_intensity_index(brand_recognition, market_performance, financial_status):
    alpha = 0.5
    beta = 0.3
    gamma = 0.2
    return alpha * brand_recognition + beta * market_performance + gamma * financial_status

# 示例数据
brand_recognition = 0.8
market_performance = 0.7
financial_status = 0.9

index = brand_intensity_index(brand_recognition, market_performance, financial_status)
print(f"品牌强度指数: {index}")
```

```python
from dataclasses import dataclass

@dataclass
class BrandMetrics:
    recognition: float
    performance: float
    financial_health: float

def calculate_brand_intensity(metrics: BrandMetrics) -> float:
    alpha = 0.5
    beta = 0.3
    gamma = 0.2
    return alpha * metrics.recognition + beta * metrics.performance + gamma * metrics.financial_health

# 示例
metrics = BrandMetrics(0.85, 0.75, 0.9)
index = calculate_brand_intensity(metrics)
print(f"品牌强度指数: {index}")
```
