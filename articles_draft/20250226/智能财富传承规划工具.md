def generate_inheritance_plan(user_assets, beneficiary_relationship):
    if beneficiary_relationship == 'spouse':
        return {'spouse': user_assets * 0.5, 'children': user_assets * 0.5}
    elif beneficiary_relationship == 'children':
        return {'children': user_assets * 0.8, 'charity': user_assets * 0.2}
    else:
        return {'other': user_assets * 1.0}
```

#### 4.2 算法实现

import numpy as np
from sklearn import svm

# 示例数据
X = np.array([[500000, 0.5, 'spouse'], [300000, 0.7, 'children']])
y = np.array(['equity', 'trust'])

model = svm.SVC()
model.fit(X, y)
new_case = np.array([[400000, 0.6, 'spouse']])
prediction = model.predict(new_case)
print(prediction)
```
