# 数据发送到云端
def send_data(data):
    headers = {'Content-Type': 'application/json'}
    response = requests.post('http://localhost:5000/api', json=data, headers=headers)
    return response.status_code

# 实时监控
while True:
    data = get_sensor_data()
    send_data(data)
    time.sleep(60)
