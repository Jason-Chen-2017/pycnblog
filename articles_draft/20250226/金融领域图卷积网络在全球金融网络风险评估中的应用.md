from torch import nn
import torch.nn.functional as F

class GCN(nn.Module):
    def __init__(self, in_channels, out_channels, adj_matrix):
        super(GCN, self).__init__()
        self.adj = adj_matrix
        self.adj = self.adj.to_dense().to(torch.float32)

    def forward(self, x):

        output = torch.bmm(support, self.adj)
        return output
