```python
def decompose_task(main_task, max_depth):
    sub_tasks = []
    # 分解逻辑
    for task in main_task:
        if depth < max_depth:
            decompose_task(task, max_depth)
        else:
            sub_tasks.append(task)
    return sub_tasks
```

```python
def plan_task(tasks):
```
