def build_super_resolution_model(input_shape):
    input_image = Input(shape=input_shape)
    
    # 下采样
    conv1 = Conv2D(32, (3, 3), activation='relu', padding='same')(input_image)
    pool1 = MaxPooling2D(pool_size=(2, 2))(conv1)
    
    # 上采样
    up1 = UpSampling2D(size=(2, 2))(pool1)
    conv2 = Conv2D(32, (3, 3), activation='relu', padding='same')(up1)
    
    # 输出
    output_image = Conv2D(1, (3, 3), activation='sigmoid', padding='same')(conv2)
    
    model = Model(inputs=input_image, outputs=output_image)
