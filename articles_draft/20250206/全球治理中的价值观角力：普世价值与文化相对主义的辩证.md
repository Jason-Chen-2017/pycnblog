```python
def calculate_balance(diversity_scores):
    min_score = min(diversity_scores)
    max_score = max(diversity_scores)
    average = sum(diversity_scores) / len(diversity_scores)
    return (average - min_score) / (max_score - min_score)

# 示例
diversity_scores = [0.6, 0.7, 0.5, 0.8]
balance_point = calculate_balance(diversity_scores)
print(f"平衡点：{balance_point}")
```
