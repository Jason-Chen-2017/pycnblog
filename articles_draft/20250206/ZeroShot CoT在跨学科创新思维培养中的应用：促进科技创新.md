```python
def zero_shot_cot(problem):
    # 问题分析
    analysis = problem.split()
    # 假设生成
    hypotheses = [f"假设{i+1}：{hypothesis}" for i, hypothesis in enumerate(analysis)]
    # 验证推理
    valid_hypotheses = [h for h in hypotheses if "创新" in h]
    # 生成解决方案
    solution = "，".join(valid_hypotheses)
    return solution

print(zero_shot_cot("如何培养跨学科创新思维？"))
```
