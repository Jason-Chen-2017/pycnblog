```python
def user_resource_matching(users, resources, preferences):
    matched = {}
    for user in users:
        max_score = -1
        best_resource = None
        for resource in resources:
            score = calculate_preference_score(user, resource, preferences)
            if score > max_score:
                max_score = score
                best_resource = resource
        matched[user] = best_resource
    return matched

def calculate_preference_score(user, resource, preferences):
    score = 0
    for pref in preferences:
        if user[pref] == resource[pref]:
            score += 1
    return score
```
