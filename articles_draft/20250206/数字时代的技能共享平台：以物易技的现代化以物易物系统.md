```python
def evaluate_item_value(features):
    # 特征包括：品牌、年份、使用情况
    market_price = get_market_price(features['品牌'], features['年份'])
    usage_impact = 1 - features['使用情况']
    total_value = market_price * (1 + usage_impact * 0.5)
    return total_value

def get_market_price(brand, year):
    # 示例：品牌A的市场价每年下降10%
    base_price = 100  # 基准价
    return base_price * (0.9 ** (2023 - year)) if brand == 'A' else base_price
```
