from transformers import pipeline

def zero_shot_reasoning(question):
    model = pipeline("text-generation", model="gpt2")
    steps = question.split("\n")
    result = ""
    for step in steps:
        response = model(step)
        result += response[0]['generated_text'] + "\n"
    return result
