```python
import pandas as pd
import numpy as np
from statsmodels.tsa.arima.model import ARIMA

# 加载数据
data = pd.read_csv('technology_cycle.csv')

# 训练ARIMA模型
model = ARIMA(data['value'], order=(5, 1, 0))
model_fit = model.fit()

# 预测未来值
 forecast = model_fit.forecast(steps=10)
print(forecast)
```
