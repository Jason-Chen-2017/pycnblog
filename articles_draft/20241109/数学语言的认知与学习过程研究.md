// 输出：模型参数w

// 初始化w为0
w = 0

// 遍历数据集D
for each (x_i, y_i) in D do
    // 计算预测值
    y_pred = w * x_i + b
    
    // 计算损失函数
    loss = (y_pred - y_i)^2
    
    // 更新模型参数
    w = w - learning_rate * loss
end for

return w
