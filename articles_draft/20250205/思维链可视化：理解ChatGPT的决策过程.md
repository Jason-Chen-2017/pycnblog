def visualize_decision_process(input_text):
    tokens = tokenizer.encode(input_text)
    for i, token in enumerate(tokens):
        print(f"Step {i+1}: 处理 {tokenizer.decode([token])}")

visualize_decision_process("我喜欢吃苹果。")
