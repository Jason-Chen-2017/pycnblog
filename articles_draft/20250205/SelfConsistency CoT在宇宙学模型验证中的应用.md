import numpy as np
from scipy.optimize import minimize

# 定义宇宙膨胀模型
def cosmological_model(params, data):
    H0, Omega_m, Omega_lambda = params
    predicted = H0 * np.sqrt(Omega_m + (1 - Omega_m) * (1 + z)**2)
    return np.sum((data - predicted)**2)

# 数据点
z = np.array([0.5, 1.0, 1.5])
data = np.array([150, 200, 250])

# 初始猜测参数
initial_guess = [70, 0.3, 0.7]

# 最小化模型误差
result = minimize(cosmological_model, initial_guess, args=(data,))
print(result)
