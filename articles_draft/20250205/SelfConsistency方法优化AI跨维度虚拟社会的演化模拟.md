```python
class SelfConsistencyModel:
    def __init__(self, params):
        self.params = params

    def predict(self):
        # 模型预测
        return self.params * 2

    def consistency_check(self):
        # 自洽性检验
        return abs(self.params - self.predict()) < 0.1

    def adjust_params(self, learning_rate=0.1):
        if not self.consistency_check():
            self.params += learning_rate * (-self.predict() + self.params)
```
