**Python源代码**：

```python
import cv2
import numpy as np
from tensorflow.keras.models import load_model

model = load_model('emotion_model.h5')

cap = cv2.VideoCapture(0)

while True:
    ret, frame = cap.read()
    if not ret:
        break
    
    face = cv2.resize(frame, (48, 48))
    face = face / 255.0
    
    features = model.predict(np.expand_dims(face, axis=0))
    emotion = np.argmax(features)
    
    print(f"当前情感状态：{emotion}")
    
cap.release()
```
