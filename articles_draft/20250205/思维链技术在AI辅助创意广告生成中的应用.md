##### 3.2 Python源代码实现

```python
import tensorflow as tf
from tensorflow.keras import layers

def build_model(max_length):
    model = tf.keras.Sequential()
    model.add(layers.Embedding(input_dim=10000, output_dim=100, input_length=max_length))
    model.add(layers.LSTM(128, return_sequences=True))
    model.add(layers.Dense(256, activation='relu'))
    model.add(layers.Dropout(0.5))
    model.add(layers.Dense(1, activation='sigmoid'))
    model.compile(optimizer='adam', loss='binary_crossentropy', metrics=['accuracy'])
    return model

def generate_advertisement(topic, model, tokenizer):
    inputs = tokenizer.texts_to_sequences([topic])
    padded = tf.keras.utils.pad_sequences(inputs, maxlen=model.input_shape[1])
    prediction = model.predict(padded)
    return prediction
```

##### 3.3 数学模型与公式
广告生成模型的损失函数：
$$
\mathcal{L} = -\frac{1}{N}\sum_{i=1}^{N} y_i \log(p_i) + (1 - y_i)\log(1 - p_i)
$$
其中，\( p_i \) 是预测概率，\( y_i \) 是真实标签。
