import numpy as np

def attention(query, key, value):
    # 计算注意力权重
    scores = np.dot(query, key.T)
    scores = scores / np.sqrt(scores.shape[-1])
    weights = softmax(scores)
    return np.dot(weights, value)

def softmax(x):
    return np.exp(x) / np.sum(np.exp(x), axis=-1, keepdims=True)

def generate_text(model, input_text):
    while True:
        # 分解输入
        tokens = tokenize(input_text)
        # 生成新文本
        output = attention(tokens, model.key, model.value)
        yield output
```

class TextGenerator:
    def __init__(self, model, tokenizer):
        self.model = model
        self.tokenizer = tokenizer

    def generate(self, input_text, max_length=50):
        tokens = self.tokenizer(input_text)
        generated = input_text
        for _ in range(max_length):
            # 预测下一个词
            predictions = self.model.predict(tokens)
            next_word = sample(predictions)
            generated += next_word
            tokens.append(self.tokenizer(next_word))
        return generated
```
