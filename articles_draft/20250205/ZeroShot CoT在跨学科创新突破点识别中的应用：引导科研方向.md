```python
import numpy as np

def calculate_similarity(concepts):
    # 计算概念间相似度矩阵
    similarity = np.zeros((len(concepts), len(concepts)))
    for i in range(len(concepts)):
        for j in range(len(concepts)):
            if i != j:
                similarity[i][j] = np.exp(-np.sum(abs(concepts[i] - concepts[j])))
    return similarity

# 示例输入
concepts = [[1, 2], [2, 3], [3, 4]]
similarity_matrix = calculate_similarity(concepts)
print(similarity_matrix)
```
