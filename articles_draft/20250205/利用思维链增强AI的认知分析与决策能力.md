AI认知涉及数据感知、特征提取和语义理解。通过思维链，AI能更系统地处理信息，提升认知能力。

    model = LinearRegression()
    model.fit(data[['X1', 'X2', 'X3']], data['Y'])
    return model

def evaluate_model(model, data):
    return model.score(data[['X1', 'X2', 'X3']], data['Y'])

# 示例数据集
data = {
    'X1': [1, 4, 7],
    'X2': [2, 5, 8],
    'X3': [3, 6, 9],
    'Y': [2*1 + 3*4 -7, 2*2 +3*5 -8, 2*3 +3*6 -9]
}

processed_data = preprocess_data(data)
model = train_model(processed_data)
score = evaluate_model(model, processed_data)
print(f'模型得分：{score}')
