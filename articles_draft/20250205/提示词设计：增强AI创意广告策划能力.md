import requests
import numpy as np

def generate_prompts(topic, target_audience, brand_tone):
    url = "https://api.example.com/generate_prompts"
    payload = {
        "topic": topic,
        "target_audience": target_audience,
        "brand_tone": brand_tone
    }
    response = requests.post(url, json=payload)
    return response.json()

# Example usage
