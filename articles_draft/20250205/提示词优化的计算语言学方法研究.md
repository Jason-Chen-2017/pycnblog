        self.score = self.evaluate_prompt(self.prompt)
    
    def evaluate_prompt(self, prompt):
        length = len(prompt)
        diversity = len(set(prompt))
        return length * 0.5 + diversity * 0.5
    
    def optimize(self):
        while True:
            new_prompt = self.adjust_prompt()
            new_score = self.evaluate_prompt(new_prompt)
            if new_score > self.score:
                self.prompt = new_prompt
                self.score = new_score
            else:
                break
    
    def adjust_prompt(self):
