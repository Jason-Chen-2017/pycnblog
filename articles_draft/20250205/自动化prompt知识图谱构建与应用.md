# 示例代码：基于自动化prompt的知识图谱构建

class KnowledgeGraph:
    def __init__(self):
        self.graph = {}

    def add_entity(self, entity, properties):
        self.graph[entity] = properties

    def add_relation(self, entity1, relation, entity2):
        if entity1 not in self.graph:
            self.graph[entity1] = {}
        self.graph[entity1][relation] = entity2

def construct_knowledge_graph(data):
    kg = KnowledgeGraph()
    for entry in data:
        entity = entry['entity']
        properties = entry['properties']
        relations = entry['relations']
        kg.add_entity(entity, properties)
        for rel in relations:
            kg.add_relation(entity, rel, relations[rel])
    return kg

# 示例数据
data = [
    {
        'entity': '苹果',
        'properties': {'成立时间': '1971年', '创始人': '史蒂夫·乔布斯'},
        'relations': {'研发': 'iPhone', '收购': 'NeXT'}
    },
    # 更多数据...
]

# 调用构建函数
kg = construct_knowledge_graph(data)
