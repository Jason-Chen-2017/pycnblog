```python
def zero_shot_cot(environment_input):
    # 初始化推理链
    thought_process = []
    # 添加初始问题
    thought_process.append(f"问题：{environment_input}")
    # 推理规则匹配
    while True:
        last_thought = thought_process[-1]
        if "?" in last_thought:
            # 添加推理步骤
            thought_process.append(f"推理：{last_thought.split('?')[1]}")
        else:
            break
    # 生成决策结果
    decision = "最终决策："
    for thought in thought_process:
        if "推理" in thought:
            decision += f" {thought.split(':')[1]}"
    return decision
```

```python
class ZeroShotCOT:
    def __init__(self, rules):
        self.rules = rules

    def decide(self, input_data):
        thought_process = [f"问题：{input_data}"]
        for rule in self.rules:
            thought_process.append(f"推理：{rule(thought_process[-1])}")
        return f"最终决策：{thought_process[-1].split(':')[1]}"
```
