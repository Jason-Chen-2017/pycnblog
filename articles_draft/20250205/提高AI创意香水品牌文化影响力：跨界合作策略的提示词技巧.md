$$

import numpy as np
from sklearn.metrics.pairwise import cosine_similarity

def recommend_perfume(user_profile, perfume_profiles):
    return cosine_similarity(user_profile, perfume_profiles).argmax()

user_profile = np.array([0.7, 0.3])
perfume_profiles = np.array([[0.6, 0.4], [0.5, 0.5], [0.8, 0.2]])
recommendation = recommend_perfume(user_profile, perfume_profiles)
print(recommendation)
