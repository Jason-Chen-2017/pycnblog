```python
def preprocess_patent_text(text):
    # 分割段落
    paragraphs = text.split('\n')
    # 提取关键词
    keywords = []
    for paragraph in paragraphs:
        # 使用正则表达式提取技术术语
        import re
        matches = re.findall(r'\b\w+\b', paragraph)
        keywords.extend(matches)
    return keywords
```
