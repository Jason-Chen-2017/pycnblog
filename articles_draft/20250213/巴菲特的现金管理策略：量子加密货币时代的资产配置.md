import numpy as np
import matplotlib.pyplot as plt

# 假设的量子加密货币收益数据
returns = np.array([0.05, 0.03, 0.08, 0.02, 0.06])

# 计算投资组合的期望收益和风险
def portfolio_optimization(returns, weights):
    expected_return = np.sum(returns * weights)
    variance = np.sum(weights**2 * returns.std())
    return expected_return, variance

# 使用等权配置
weights = np.array([1/len(returns)] * len(returns))
expected_return, variance = portfolio_optimization(returns, weights)
print(f"Expected Return: {expected_return}, Variance: {variance}")
