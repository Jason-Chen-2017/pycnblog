import pandas as pd
import numpy as np

def calculate_market_sentiment(data):
    # 计算市场波动性
    market_volatility = data['价格'].std()
    # 统计市场参与者数量
    market_participants = data['成交量'].sum()
    # 计算市场情绪指标
    market_sentiment = market_volatility / market_participants
    return market_sentiment

# 示例数据
