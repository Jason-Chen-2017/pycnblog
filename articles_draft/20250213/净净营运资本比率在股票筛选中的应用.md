##### 7.2 净净营运资本比率计算的Python代码示例
```python
import pandas as pd
import numpy as np

# 假设我们有一个包含财务数据的DataFrame df，其中包含流动资产和流动负债的数据
def calculate_net_working_capital_ratio(df):
    # 计算流动资产和流动负债的总和
    current_assets = df['流动资产'].sum()
    current_liabilities = df['流动负债'].sum()
    # 计算净营运资本
    net_working_capital = current_assets - current_liabilities
    # 计算净营运资本比率
    net_working_capital_ratio = net_working_capital / current_assets
    return net_working_capital_ratio

# 示例数据
data = {
    '流动资产': [100, 150, 200, 250],
    '流动负债': [60, 70, 80, 90]
}
df = pd.DataFrame(data)
ratio = calculate_net_working_capital_ratio(df)
print(f"净营运资本比率: {ratio}")
```
