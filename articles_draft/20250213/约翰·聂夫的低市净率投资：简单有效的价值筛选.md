```python
import pandas as pd

def calculate_p_b(stock_price, book_value_per_share):
    return stock_price / book_value_per_share

def screen_low_p_b(stocks_data, threshold=1.5):
    low_p_b_stocks = []
    for stock in stocks_data:
        p_b = calculate_p_b(stock['price'], stock['book_value'])
        if p_b < threshold:
            low_p_b_stocks.append(stock['name'])
    return low_p_b_stocks
```
