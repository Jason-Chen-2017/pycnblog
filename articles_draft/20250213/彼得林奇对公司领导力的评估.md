  app = Flask(__name__)
  
  @app.route('/evaluate', methods=['POST'])
  def evaluate():
      data = request.json
      # 计算各项能力评分
      c1 = data['strategic_thinking']
      c2 = data['decision_making']
      c3 = data['team_building']

      app.run(debug=True)
