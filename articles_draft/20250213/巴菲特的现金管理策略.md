#### 3.2 巴菲特投资组合的数学公式

  $$ E(R_p) = \sum_{i=1}^{n} w_i E(R_i) $$
- 3.2.2 投资组合的风险（方差）
  $$ Var(R_p) = \sum_{i=1}^{n} \sum_{j=1}^{n} w_i w_j Cov(R_i, R_j) $$

import numpy as np
import pandas as pd

def calculate_npvcash_flows(cf, r):
    return np.npv(r, *cf)

def calculate_irr(cf):
    return np.irr(cf)

# 示例数据
cash_flows = [-100, 200, 300]
rate = 0.1

npv = calculate_npvcash_flows(cash_flows, rate)
irr = calculate_irr(cash_flows)

print(f"净现值: {npv}")
print(f"内部收益率: {irr}")
