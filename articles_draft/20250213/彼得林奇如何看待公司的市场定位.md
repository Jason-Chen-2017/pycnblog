import pandas as pd

# 数据收集
data = pd.read_csv('market.csv')

# 数据分析
def market_positioning_score(data):
    score = 0
    # 加权评分公式
    score = 0.4 * data['产品定位'] + 0.3 * data['价格定位'] + 0.2 * data['渠道定位'] + 0.1 * data['促销定位']
    return score

# 市场定位评分
data['市场定位评分'] = data.apply(market_positioning_score, axis=1)

# 结果输出
print(data[['公司', '市场定位评分']].head())
