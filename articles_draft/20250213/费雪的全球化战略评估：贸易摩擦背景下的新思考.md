import numpy as np
import pandas as pd
from sklearn.linear_model import LinearRegression

# 示例数据
data = {
    'market_access': [0.8, 0.6, 0.7, 0.9],
    'policy_risk': [0.5, 0.7, 0.6, 0.4],
    'supply_chain': [0.7, 0.8, 0.6, 0.5],
    'global_strategy_score': [0.6, 0.5, 0.7, 0.8]
}

df = pd.DataFrame(data)
X = df[['market_access', 'policy_risk', 'supply_chain']]
y = df['global_strategy_score']

model = LinearRegression()
model.fit(X, y)

# 预测结果
new_data = pd.DataFrame({
    'market_access': [0.7],
    'policy_risk': [0.6],
    'supply_chain': [0.6]
})
predicted_score = model.predict(new_data)
print("预测的全球化战略评分：", predicted_score[0])
