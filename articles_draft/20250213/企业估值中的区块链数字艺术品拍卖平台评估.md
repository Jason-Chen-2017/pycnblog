```solidity
pragma solidity ^0.8.0;

contract ArtNFT {
    struct Artwork {
        address owner;
        uint256 price;
        bool isListed;
    }

    mapping(uint256 => Artwork) public artworks;

    function listItem(uint256 tokenId, uint256 price) public {
        require(msg.sender == ownerOf(tokenId), "Only owner can list item");
        artworks[tokenId].owner = msg.sender;
        artworks[tokenId].price = price;
        artworks[tokenId].isListed = true;
    }

    function buyItem(uint256 tokenId) public payable {
        Artwork memory art = artworks[tokenId];
        require(art.isListed, "Item is not listed for sale");
        require(msg.sender != art.owner, "Cannot buy own item");
        require(msg.value >= art.price, "Insufficient funds");

        art.owner = msg.sender;
        artworks[tokenId] = art;
        payable(art.owner).transfer(msg.value);
    }
}
```
