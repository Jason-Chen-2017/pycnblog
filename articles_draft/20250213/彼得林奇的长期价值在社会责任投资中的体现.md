import pandas as pd

def calculate_sri_score(data):
    # 计算社会责任评分
    score = data['环境'] * 0.4 + data['社会'] * 0.3 + data['治理'] * 0.3
    return score

data = pd.DataFrame({
    '环境': [85, 90, 75],
    '社会': [80, 75, 90],
    '治理': [90, 85, 80]
})

sri_scores = data.apply(lambda x: calculate_sri_score(x), axis=1)
print(sri_scores)

import pandas as pd
import numpy as np

def evaluate_sri_factors(data):
    # 环境因素评分
    environmental_score = np.mean(data['环境'])
    # 社会因素评分
    social_score = np.mean(data['社会'])
    # 治理因素评分
    governance_score = np.mean(data['治理'])
    return environmental_score, social_score, governance_score

data = pd.DataFrame({
    '环境': [85, 90, 75],
    '社会': [80, 75, 90],
    '治理': [90, 85, 80]
})

env_score, social_score, gov_score = evaluate_sri_factors(data)
print(f"环境评分：{env_score}\n社会评分：{social_score}\n治理评分：{gov_score}")
