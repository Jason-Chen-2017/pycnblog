import numpy as np
from scipy.signal import fir_filter

def main():
    # 高频信号生成
    fs = 10e9  # 采样频率
    f = 1.5e12  # 信号频率
    t = np.arange(0, 1, 1/fs)
    signal = np.sin(2 * np.pi * f * t)
    
    # 信道模拟
    noise = np.random.normal(0, 0.1, len(signal))
    received_signal = signal + noise
    
    # 信号处理
    coefficients = fir_filter(order=10, cutoff=0.5*fs, fs=fs)
    filtered_signal = fir_filter(coefficients, received_signal)
    
    # 输出结果
    print("信号恢复效果：", np.mean(filtered_signal))

if __name__ == "__main__":
    main()
```
