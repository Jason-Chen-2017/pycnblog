def value_decomposition_learning():
    # 初始化各智能体的Q值
    Q = {i: defaultdict(float) for i in range(n_agents)}
    
    while True:
        state = environment.reset()
        agents_actions = [agent.act(state) for agent in agents]
        rewards = environment.get_rewards(agents_actions)
        for i in range(n_agents):
            Q[i][state][agents_actions[i]] += learning_rate * ( rewards[i] - Q[i][state][agents_actions[i]] )
        # 通信与更新策略
