```python
import pandas as pd
import matplotlib.pyplot as plt

# 示例：计算产品线市场占有率
data = {
    '产品线': ['A', 'B', 'C'],
    '销量': [100, 200, 150],
    '市场总销量': [500, 500, 500]
}
df = pd.DataFrame(data)

market_share = df['销量'] / df['市场总销量'] * 100
plt.bar(df['产品线'], market_share)
plt.title('产品线市场占有率')
plt.show()
```
