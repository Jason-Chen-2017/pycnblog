```python
import numpy as np

def calculate_optimal_portfolio(returns, cov_matrix, target_return):
    # 使用均值-方差优化
    n = len(returns)
    # 计算权重
    weights = np.linalg.solve(cov_matrix, returns) / np.sum(np.linalg.solve(cov_matrix, returns))
    # 验证目标收益
    portfolio_return = np.dot(returns, weights)
    if portfolio_return >= target_return:
        return weights
    else:
        return None
```
