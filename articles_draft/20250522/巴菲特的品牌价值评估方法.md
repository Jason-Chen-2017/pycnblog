```python
def calculate_brand_value(data):
    weights = [0.4, 0.3, 0.2, 0.1]
    scores = [data['市场占有率'], data['品牌忠诚度'], 
              data['品牌溢价'], data['财务表现']]
    brand_value = sum(w * s for w, s in zip(weights, scores))
    return brand_value

# 示例数据
data = {
    '市场占有率': 0.25,
    '品牌忠诚度': 0.8,
    '品牌溢价': 0.6,
    '财务表现': 0.7
}
print(calculate_brand_value(data))
```
