### 4. 免疫系统的概述

```python
# 导入必要的库
import numpy as np
import matplotlib.pyplot as plt

# 定义常微分方程
def model(N, I, R):
    r = 1.0
    a = 0.1
    b = 0.2
    g = 0.3
    c = 0.4
    d = 0.5
    dNdt = r * N - a * N * I
    dIdt = -b * I + g * I * N
    dRdt = c * I - d * R
    return dNdt, dIdt, dRdt

# 求解常微分方程
def solveODE(N0, I0, R0, t):
    N = np.zeros_like(t)
    I = np.zeros_like(t)
    R = np.zeros_like(t)
    N[0] = N0
    I[0] = I0
    R[0] = R0
    for i in range(1, len(t)):
        dNdt, dIdt, dRdt = model(N[i-1], I[i-1], R[i-1])
        N[i] = N[i-1] + dNdt * dt
        I[i] = I[i-1] + dIdt * dt
        R[i] = R[i-1] + dRdt * dt
    return N, I, R

# 设置参数
N0 = 1000
I0 = 100
R0 = 100
t = np.linspace(0, 100, 1000)

# 求解模型
N, I, R = solveODE(N0, I0, R0, t)

# 绘制结果
plt.plot(t, N, label='N')
plt.plot(t, I, label='I')
plt.plot(t, R, label='R')
plt.xlabel('Time')
plt.ylabel('Cell Concentration')
plt.legend()
plt.show()
```
