```python
# 跨文化领导力模型：文化适应与领导风格匹配算法

def cultural_adaptation(leader_style, cultural_factor):
    """
    文化适应与领导风格匹配算法

    :param leader_style: 领导风格（0: 权力导向，1: 人本主义，2: 任务导向）
    :param cultural_factor: 文化因素（0: 低语境，1: 高语境）
    :return: 适应后的领导风格
    """
    if cultural_factor == 0:  # 低语境文化
        if leader_style == 0:  # 权力导向
            return 1  # 转为人本主义领导风格
        elif leader_style == 1:  # 人本主义
            return 2  # 转为任务导向领导风格
        elif leader_style == 2:  # 任务导向
            return 1  # 转为人本主义领导风格
    else:  # 高语境文化
        if leader_style == 0:  # 权力导向
            return 2  # 转为任务导向领导风格
        elif leader_style == 1:  # 人本主义
            return 0  # 转为权力导向领导风格
        elif leader_style == 2:  # 任务导向
            return 0  # 转为权力导向领导风格
```
