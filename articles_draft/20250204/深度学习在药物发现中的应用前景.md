# Python代码示例：前向传播计算
def forward_propagation(X, weights, biases):
    # 输入数据处理
    z = np.dot(X, weights) + biases
    # 激活函数
    a = sigmoid(z)
    return a

# sigmoid函数
def sigmoid(x):
    return 1 / (1 + np.exp(-x))

# 测试代码

print(output)
