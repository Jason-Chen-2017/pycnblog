```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

def calculate_tax_after_capital_gains(taxable_income, holding_period, tax_rate):
    # 计算资本利得税后的收益
    tax = taxable_income * tax_rate
    return taxable_income - tax

def optimize_portfolio(tax_rates, holding_periods):
    # 简单优化算法：寻找最优持有期限和税率组合
    max_tax_saving = 0
    best_period = 0
    best_tax_rate = 0
    for period in holding_periods:
        for rate in tax_rates:
            tax_saving = 1 - rate
            if tax_saving > max_tax_saving:
                max_tax_saving = tax_saving
                best_period = period
                best_tax_rate = rate
    return best_period, best_tax_rate

# 示例数据
tax_rates = [0.15, 0.2, 0.25]
holding_periods = [1, 2, 3, 5, 10]

best_period, best_tax_rate = optimize_portfolio(tax_rates, holding_periods)
print(f"最优持有期限：{best_period}年，最优税率：{best_tax_rate}")
```
