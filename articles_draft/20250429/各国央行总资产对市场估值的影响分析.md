```python
import pandas as pd
import numpy as np
from sklearn.linear_model import LinearRegression
import matplotlib.pyplot as plt

# 数据加载
df = pd.read_csv('central_bank_assets.csv')
market_val = pd.read_csv('market_valuation.csv')

# 数据预处理
merged_data = pd.merge(df, market_val, on='date')

# 模型训练
model = LinearRegression()
X = merged_data[['central_bank_assets']]
y = merged_data['market_valuation']
model.fit(X, y)

# 模型预测
predictions = model.predict(X)

# 可视化
plt.scatter(X, y, color='blue')
plt.plot(X, predictions, color='red')
plt.xlabel('Central Bank Assets')
plt.ylabel('Market Valuation')
plt.title('Central Bank Assets vs Market Valuation')
plt.show()

# 模型评估
print('Coefficients:', model.coef_)
print('Intercept:', model.intercept_)
```
