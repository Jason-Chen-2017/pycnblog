### 3.3 Python实现示例
```python
def calculate_probability(factor):
    # 假设factor是一个风险因素
    # 返回该因素的发生概率
    return 0.2  # 示例值

def calculate_impact(factor):
    # 返回该因素的影响程度（1-5分）
    return 3

def risk_assessment(factors):
    probability = {}
    impact = {}
    score = {}
    for factor in factors:
        probability[factor] = calculate_probability(factor)
        impact[factor] = calculate_impact(factor)
        score[factor] = probability[factor] * impact[factor]
    return score

# 示例使用
factors = ['市场波动', '经济周期', '政策变化']
result = risk_assessment(factors)
print(result)
```
