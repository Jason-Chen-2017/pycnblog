```python
import pandas as pd
import numpy as np

def assess_competitive_edge(financial_data, market_data):
    # 计算财务指标
    roe = financial_data['净利润'] / financial_data['股东权益']
    gross_margin = financial_data['毛利润'] / financial_data['营业收入']
    # 分析市场地位
    market_share = market_data['市场份额']
    # 综合评估
    competitive_edge_score = 0.5 * roe + 0.3 * gross_margin + 0.2 * market_share
    return competitive_edge_score

# 示例数据
financial_data = {
    '净利润': 100,
    '股东权益': 500,
    '毛利润': 200,
    '营业收入': 1000
}
market_data = {
    '市场份额': 0.3
}

score = assess_competitive_edge(financial_data, market_data)
print(f"竞争优势得分：{score}")
```
