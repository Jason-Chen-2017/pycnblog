##### 3.4 套利算法的Python实现

```python
def calculate_profit(buy_price, sell_price):
    return sell_price - buy_price

def cross_chain_arbitrage(price_data):
    for chain in price_data:
        for asset in chain.assets:
            buy_price = asset.buy_price
            sell_price = asset.sell_price
            if sell_price > buy_price:
                profit = calculate_profit(buy_price, sell_price)
                print(f"Detect arbitrage opportunity: Buy at {buy_price}, Sell at {sell_price}, Profit: {profit}")
    return

# 示例用法
price_data = [
    {"chain_id": 1, "assets": [{"asset_id": 1, "buy_price": 100, "sell_price": 110}]},
    {"chain_id": 2, "assets": [{"asset_id": 2, "buy_price": 90, "sell_price": 100}]}
]
cross_chain_arbitrage(price_data)
```

class Asset:
    def __init__(self, buy_price, sell_price):
        self.buy_price = buy_price
        self.sell_price = sell_price

class CrossChainArbitrage:
    def __init__(self, assets):
        self.assets = assets

    def detect_arbitrage(self):
        for asset in self.assets:
            if asset.sell_price > asset.buy_price:
                print(f"Arbitrage detected: Buy at {asset.buy_price}, Sell at {asset.sell_price}")

# 示例用法
assets = [
    Asset(100, 110),
    Asset(90, 100)
]
arbitrage_system = CrossChainArbitrage(assets)
arbitrage_system.detect_arbitrage()
