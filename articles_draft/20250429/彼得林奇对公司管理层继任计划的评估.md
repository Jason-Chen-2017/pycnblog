import pandas as pd
import numpy as np

def evaluate_management团队(管理层信息):
    管理能力 = 管理层信息['管理能力']
    领导力 = 管理层信息['领导力']
    战略规划能力 = 管理层信息['战略规划能力']
    总分 = (管理能力 + 领导力 + 战略规划能力) / 3
    return 总分

管理层信息 = pd.DataFrame({
    '管理能力': [8, 7, 9],
    '领导力': [8, 8, 7],
    '战略规划能力': [7, 9, 8]
})

评估结果 = evaluate_management团队(管理层信息)
print(评估结果)
