```python
import numpy as np

              [80, 80],   # 蛋白质和碳水化合物含量
              [20, 120],  # 蛋白质和碳水化合物含量
              [70, 70]])  # 蛋白质和碳水化合物含量

y = np.array([0, 1, 1, 0])  # 标签：0代表不均衡，1代表均衡

# 训练模型
model = svm.SVC()
model.fit(X, y)

# 预测新数据
new_data = np.array([[50, 80]])
print("预测结果：", model.predict(new_data))
```
