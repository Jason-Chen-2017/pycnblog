# 示例代码：使用机器学习模型进行索赔分类
import pandas as pd
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.svm import SVC
from sklearn.pipeline import Pipeline

# 加载数据
data = pd.read_csv('claims.csv')
X = data['description']
y = data['claim_type']

# 特征提取
vectorizer = TfidfVectorizer()
X_features = vectorizer.fit_transform(X)

# 模型训练
model = SVC()
pipeline = Pipeline([('tfidf', vectorizer), ('svc', model)])
pipeline.fit(X, y)

# 预测
new_claim = "车辆碰撞"
predicted_claim_type = pipeline.predict([new_claim])[0]
print(predicted_claim_type)
