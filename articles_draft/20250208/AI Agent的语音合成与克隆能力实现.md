```python
import tensorflow as tf
from tensorflow.keras import layers

def build_model():
    input_layer = layers.Input(shape=(None, 256))
    embedding_layer = layers.Dense(256, activation='relu')(input_layer)
    lstm_layer = layers.LSTM(128, return_sequences=True)(embedding_layer)
    dense_layer = layers.Dense(128, activation='relu')(lstm_layer)
    output_layer = layers.Dense(8000, activation='sigmoid')(dense_layer)
    model = tf.keras.Model(inputs=input_layer, outputs=output_layer)
    return model
```
