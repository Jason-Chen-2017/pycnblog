## 5.2 模型实现

```python
import torch
import torch.nn as nn
import torch.nn.functional as F

class VisualLanguageModel(nn.Module):
    def __init__(self, visual_dim, language_dim):
        super().__init__()
        self.visual_proj = nn.Linear(visual_dim, language_dim)
        self.language_proj = nn.Linear(language_dim, visual_dim)

    def forward(self, visual_features, language_features):
        visual_aligned = self.visual_proj(visual_features)
        language_aligned = self.language_proj(language_features)
        aligned = (visual_aligned + language_aligned) / 2
        return aligned
```
