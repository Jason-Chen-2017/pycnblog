### 2.5.2 核心代码实现

```python
import numpy as np

def optimize_space(items, available_space):
    # 生成排列组合
    permutations = np.permutations(items)
    # 计算最优解
    best_permutation = None
    max_utilization = 0
    for p in permutations:
        utilization = sum(p[:available_space])
        if utilization > max_utilization:
            max_utilization = utilization
            best_permutation = p
    return best_permutation

# 示例数据
items = [5, 4, 3, 2, 1]
available_space = 3
print(optimize_space(items, available_space))
```
