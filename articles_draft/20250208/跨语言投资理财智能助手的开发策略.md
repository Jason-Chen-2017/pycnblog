### 6.2 系統核心实现源代码
```python
from transformers import AutoTokenizer, AutoModelForSeq2Seq
import numpy as np

class CrossLanguageInvestmentHelper:
    def __init__(self):
        self.tokenizer = AutoTokenizer.from_pretrained("microsoft/mdeberta-v2-base")
        self.model = AutoModelForSeq2Seq.from_pretrained("facebook/m2m-100")

    def translate(self, text, src_lang, tgt_lang):
        inputs = self.tokenizer(text, return_tensors="np")
        inputs["src_lang"] = np.array([src_lang])
        inputs["tgt_lang"] = np.array([tgt_lang])
        outputs = self.model.generate(**inputs)
        return self.tokenizer.decode(outputs[0], skip_special_tokens=True)
```
