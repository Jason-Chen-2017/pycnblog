```python
# 知识图谱构建
def build_knowledge_graph(data):
    graph = {}
    for entity in data['entities']:
        graph[entity] = []
    for relation in data['relations']:
        graph[relation['head']].append((relation['tail'], relation['label']))
    return graph

# 动态更新
def update_graph(graph, new_data):
    for entity in new_data['entities']:
        graph[entity] = []
    for relation in new_data['relations']:
        graph[relation['head']].append((relation['tail'], relation['label']))
    return graph

# 实时推理
def real_time_reasoning(graph, query):
    result = []
    for head in graph:
        if head == query['head']:
            for tail, label in graph[head]:
                result.append((tail, label))
    return result
```
