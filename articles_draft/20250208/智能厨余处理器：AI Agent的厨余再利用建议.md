```python
from sklearn import svm
import numpy as np

def main():
    # 数据集
    features = np.array([[1, 2], [3, 4], [5, 6]])
    labels = np.array([0, 1, 0])
    
    # 模型训练
    model = svm.SVC()
    model.fit(features, labels)
    
    # 预测
    new_data = np.array([[2, 3]])
    prediction = model.predict(new_data)
    print(f"预测结果：{prediction}")

if __name__ == "__main__":
    main()
```
