#### 3.3 Python实现代码
```python
import numpy as np

class AI-Agent:
    def __init__(self):
        self.state = 'closed'
    
    def perceive(self, light_intensity):
        self.current_light = light_intensity
    
    def decide_action(self):
        if self.current_light < 50:
            return 'open'
        else:
            return 'close'
    
    def execute_action(self, action):
        self.state = action
        return self.state

# 示例
agent = AI-Agent()
agent.perceive(30)
action = agent.decide_action()
agent.execute_action(action)
print(agent.state)
```
