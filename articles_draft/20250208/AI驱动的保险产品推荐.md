```python
import pandas as pd
from sklearn.metrics.pairwise import cosine_similarity

# 数据加载
products = pd.read_csv('products.csv')
users = pd.read_csv('users.csv')

# 数据预处理
user_features = users[['age', 'income', 'occupation']]
product_features = products[['premium', 'coverage', 'deductible']]

# 计算相似性
similarity_matrix = cosine_similarity(user_features, product_features)

# 推荐结果
recommendations = similarity_matrix.argsort()[:, ::-1]

print(recommendations)
```
