## 附录：完整的Python代码示例
```python
import networkx as nx
from sklearn.metrics.pairwise import cosine_similarity

def build_knowledge_graph(data):
    G = nx.Graph()
    for entry in data:
        G.add_node(entry['id'])
        for relation in entry['relations']:
            G.add_edge(entry['id'], relation['target'])
    return G

def risk_assessment(graph):
    # 假设节点特征为向量表示
    node_features = {node: [0.5, 0.3, 0.2] for node in graph.nodes()}
    similarities = cosine_similarity([node_features[node] for node in graph.nodes()])
    # 计算风险评分
    risk_scores = {}
    for node in graph.nodes():
        score = sum(similarities[i][j] for i, j in graph.edges())
        risk_scores[node] = score
    return risk_scores
```
