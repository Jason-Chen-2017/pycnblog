```python
import pandas as pd
import numpy as np

# 示例数据
data = {
    '股票代码': ['A', 'B', 'C'],
    '市盈率': [10, 15, 20],
    '股息率': [2%, 3%, 4%],
    '通胀预期': [5%, 3%, 1%]
}

df = pd.DataFrame(data)

# 计算调整后的内在价值
def calculate_adjusted_value(dividend_yield, inflation_rate):
    return dividend_yield / (1 + inflation_rate)

df['调整后的内在价值'] = df.apply(lambda x: calculate_adjusted_value(x['股息率'], x['通胀预期']), axis=1)

print(df)
```
