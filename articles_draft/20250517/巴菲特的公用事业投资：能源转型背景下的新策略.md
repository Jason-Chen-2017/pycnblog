import pandas as pd
import numpy as np

# 示例代码：计算股票的内在价值
def calculate_intrinsic_value(stock_price, growth_rate, discount_rate):
    return stock_price * (1 + growth_rate) / (1 + discount_rate)

# 示例数据
data = {
    '公司': ['公司A', '公司B', '公司C'],
    '股价': [50, 60, 70],
    '增长率': [0.05, 0.03, 0.04],
    '贴现率': [0.08, 0.07, 0.06]
}

df = pd.DataFrame(data)
df['内在价值'] = df.apply(lambda x: calculate_intrinsic_value(x['股价'], x['增长率'], x['贴现率']), axis=1)
print(df)
