###### 3.3.1 Python代码实现
```python
import numpy as np

def calculate_enterprise_value(fcf, wacc, terminal_value, n=5):
    # 现金流贴现计算
    discounted_cash_flows = []
    for i in range(n):
        discounted_cash_flows.append(fcf * (1 + wacc)**i)
    discounted_cash_flows = [fcf * (1 + wacc)**i for i in range(n)]
    present_value = sum(discounted_cash_flows) + terminal_value / (1 + wacc)**n
    return present_value

# 示例数据
fcf = 100  # 自由现金流
wacc = 0.10  # 加权平均资本成本
terminal_value = 500  # 终值
n = 5  # 预测期

enterprise_value = calculate_enterprise_value(fcf, wacc, terminal_value, n)
print(f"企业价值：{enterprise_value}")
```
