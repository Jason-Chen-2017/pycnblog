```python
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.naive_bayes import MultinomialNB
from sklearn.metrics import accuracy_score
import matplotlib.pyplot as plt

# 数据集加载与预处理
data = pd.read_csv('social_media_data.csv')
data['text'] = data['text'].str.lower().str.replace('[^a-zA-Z0-9\s]', '', regex=True)

# 特征提取
vectorizer = TfidfVectorizer(max_features=1000)
X = vectorizer.fit_transform(data['text'])

# 标签编码
y = data['emotion']
y_encoded = pd.get_dummies(y).values

# 模型训练
X_train, X_test, y_train, y_test = train_test_split(X, y_encoded, test_size=0.2, random_state=42)
model = MultinomialNB()
model.fit(X_train, y_train)

# 模型评估
y_pred = model.predict(X_test)
accuracy = accuracy_score(y_test, y_pred)
print(f"Accuracy: {accuracy}")

# 互动算法实现
def interact_user(user_message, model, vectorizer):
    user_vector = vectorizer.transform([user_message])
    predicted_emotion = model.predict(user_vector)[0]
    return predicted_emotion

# 代码解读与分析
# 假设我们有一个用户的留言："I had a great day at the park with my friends!"
predicted_emotion = interact_user("I had a great day at the park with my friends!", model, vectorizer)
print(f"Predicted emotion: {predicted_emotion}")
