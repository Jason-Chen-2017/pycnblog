# 计算生命意义
def calculate_life_meaning(hours_worked, hours_recreation, hours_health):
    work_ratio = hours_worked / (hours_worked + hours_recreation + hours_health)
    health_ratio = hours_health / (hours_worked + hours_recreation + hours_health)
    meaning = work_ratio * 0.6 + health_ratio * 0.4
    return meaning

# 示例
meanings = [calculate_life_meaning(8, 4, 4) for i in range(5)]
print(meanings)
