# cognitive_reconstruction.py

def cognitive_reconstruction(thought, alternative_thought):
    """
    认知重构函数，将负面思维转变为积极思维
    :param thought: 负面思维
    :param alternative_thought: 积极思维
    :return: 重构后的积极思维
    """
    # 将负面思维替换为积极思维
    return alternative_thought

def main():
    # 输入负面思维
    negative_thought = input("请输入你的负面思维：")

    # 提供积极思维的选项
    positive_thought_options = [
        "尽管目前面临困难，但我可以通过努力找到生活的意义。",
        "每一次挫折都是成长的机会。",
        "我会找到解决问题的方法。"
    ]

    # 选择一个积极思维
    print("以下是一个积极思维的选项，请选择其中一个：")
    for i, option in enumerate(positive_thought_options, start=1):
        print(f"{i}. {option}")

    choice = int(input("请输入选项的编号："))
    positive_thought = positive_thought_options[choice - 1]

    # 实现认知重构
    reconstructed_thought = cognitive_reconstruction(negative_thought, positive_thought)
    print("重构后的思维：", reconstructed_thought)

if __name__ == "__main__":
    main()
