- 分布式：多个智能体独立运行。

- 反应性：能感知环境并做出实时反应。

```python
import numpy as np

def optimize_portfolio(assets, returns):
    n = len(assets)
    returns = np.array(returns)
    covariance = np.cov(returns)
    
    # 使用分布式优化算法
    def objective(weights):
        return np.dot(weights.T, covariance).dot(weights)
    
    # 初始化权重
    weights = np.array([1.0/n]*n)
    
    # 迭代优化
    for _ in range(100):
        gradient = 2 * covariance.dot(weights)
        weights = weights - 0.01 * gradient
    return weights
```
