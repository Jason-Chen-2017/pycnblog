# 示例代码：构建影响指数
import pandas as pd
import numpy as np

# 读取数据
market_data = pd.read_csv('market.csv')
tide_data = pd.read_csv('tide.csv')

# 数据合并
merged_data = pd.merge(market_data, tide_data, on='date')

# 构建影响指数
merged_data['impact_score'] = 0.5 * merged_data['market_valuation'] + 0.5 * merged_data['tide_power']

# 可视化
import matplotlib.pyplot as plt
plt.plot(merged_data['date'], merged_data['impact_score'])
plt.xlabel('日期')
plt.ylabel('影响指数')
plt.title('全球股市估值与潮汐能发电技术的影响指数')
plt.show()
