#### 5.2 核心实现
实现协同过滤算法，代码如下：
```python
def compute_similarity(matrix):
    n_users, n_items = matrix.shape
    sim = np.zeros((n_users, n_users))
    for i in range(n_users):
        for j in range(n_users):
            if i == j:
                sim[i][j] = 0
                continue
            numerator = np.sum((matrix[i] - matrix[i].mean()) * (matrix[j] - matrix[j].mean()))
            denominator = np.sqrt(np.sum((matrix[i] - matrix[i].mean())**2)) * np.sqrt(np.sum((matrix[j] - matrix[j].mean())**2))
            if denominator == 0:
                sim[i][j] = 0
            else:
                sim[i][j] = numerator / denominator
    return sim
```
