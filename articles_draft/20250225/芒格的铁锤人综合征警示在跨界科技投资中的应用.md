def assess_risk(weights):
    # 权重为各领域的投资比例
    risk = sum(w * (1 - w) for w in weights)
    return risk

def diversified_strategy(weights, threshold=0.6):
    # 判断单一领域投资比例是否超过阈值
    for w in weights:
        if w > threshold:
            return "存在铁锤人综合征风险"
    return "风险较低，建议继续分散投资"
```
