```python
def statistical_outlier_detection(data):
    mean = np.mean(data)
    std = np.std(data)
    threshold = mean + 3 * std
    return [x > threshold for x in data]
```

```python
import tensorflow as tf
from tensorflow.keras import layers

def autoencoder_detection(data):
    input_dim = data.shape[1]
    encoder_dim = 128
    input_layer = layers.Input(shape=(input_dim,))
    encoder_layer = layers.Dense(encoder_dim, activation='relu')(input_layer)
    decoder_layer = layers.Dense(input_dim, activation='sigmoid')(encoder_layer)
    autoencoder = tf.keras.Model(inputs=input_layer, outputs=decoder_layer)
    autoencoder.compile(optimizer='adam', loss='binary_crossentropy')
    autoencoder.fit(data, data, epochs=10, batch_size=32)
    return autoencoder.predict(data)
```
