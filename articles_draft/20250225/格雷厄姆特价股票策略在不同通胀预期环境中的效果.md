import pandas as pd
import numpy as np

def graham_strategy(df, inflation_rate):
    # 计算市盈率
    df['PE'] = df['Price'] / df['Earnings']
    # 计算市净率
    df['PB'] = df['Price'] / df['Book Value']
    # 计算通胀调整后的市盈率
    df['Adjusted_PE'] = df['PE'] / (1 + inflation_rate)
    # 格雷厄姆筛选标准：PE < 15，PB < 1.5
    df['Pass_Graham'] = (df['Adjusted_PE'] < 15) & (df['PB'] < 1.5)
    return df

# 示例数据
data = {
    'Ticker': ['AAPL', 'MSFT', 'GOOGL'],
    'Price': [150, 300, 250],
    'Earnings': [10, 15, 12],
    'Book Value': [50, 80, 60],
    'Inflation Rate': [0.02, 0.02, 0.02]
}

df = pd.DataFrame(data)
result = graham_strategy(df, 0.02)
print(result)
