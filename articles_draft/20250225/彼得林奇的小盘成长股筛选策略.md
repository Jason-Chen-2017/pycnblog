### 6.2 核心代码实现
```python
import pandas as pd
import numpy as np

def calculate_growth_score(revenue_growth, net_profit_growth):
    return (revenue_growth + net_profit_growth) / 2

def calculate_valuation_multiple(pe, pb):
    return (pe + pb) / 2

def calculate_risk_score(volatility, beta):
    return (volatility + beta) / 2

# 示例数据
company_data = {
    '收入增长率': [0.3, 0.25, 0.4],
    '净利润增长率': [0.2, 0.15, 0.3],
    '市盈率': [15, 20, 25],
    '市净率': [2, 3, 4],
    '波动率': [0.2, 0.3, 0.1],
    '贝塔系数': [1.2, 1.5, 0.8]
}

df = pd.DataFrame(company_data)

# 计算评分
df['成长性评分'] = df.apply(lambda x: calculate_growth_score(x['收入增长率'], x['净利润增长率']), axis=1)
df['估值倍数'] = df.apply(lambda x: calculate_valuation_multiple(x['市盈率'], x['市净率']), axis=1)
df['风险评分'] = df.apply(lambda x: calculate_risk_score(x['波动率'], x['贝塔系数']), axis=1)

print(df)
```
