### 5.2 核心代码实现

import pandas as pd
from cryptography.hazmat.primitives.asymmetric import padding

# 数据加密函数
def encrypt_data(data):
    cipher = padding.OAEP(
        hash_algorithm=padding.HashAlgorithm.SHA256()
    )
    encrypted_data = []
    for item in data:
        encrypted_item = cipher.encrypt(item.encode())
        encrypted_data.append(encrypted_item.decode())
    return encrypted_data

# 数据匿名化处理函数
def anonymize_data(dataframe):
    dataframe['user_id'] = dataframe['user_id'].apply(lambda x: 'ID_' + str(x))
    return dataframe
