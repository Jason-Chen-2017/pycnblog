# 伪代码：AES加密算法
def AES_encrypt(plaintext, key):
    # 初始化密钥
    expanded_key = KeyExpansion(key)
    ciphertext = []

    # 分块加密
    for block in blocks(plaintext):
        state = AddRoundKey(block, expanded_key[0])
        
        for i in range(1, 10):
            state = SubBytes(state)
            state = ShiftRows(state)
            state = MixColumns(state)
            if i != 10:
                state = AddRoundKey(state, expanded_key[i])

        ciphertext.append(state)

    return ciphertext

# 伪代码：AES解密算法
def AES_decrypt(ciphertext, key):
    # 初始化密钥
    expanded_key = KeyExpansion(key)
    plaintext = []

    # 分块解密
    for block in blocks(ciphertext):
        state = AddRoundKey(block, expanded_key[10])
        
        for i in range(9, 0, -1):
            state = InvShiftRows(state)
            state = InvSubBytes(state)
            state = AddRoundKey(state, expanded_key[i])
            state = InvMixColumns(state)

        plaintext.append(state)

    return concatenate(plaintext)
