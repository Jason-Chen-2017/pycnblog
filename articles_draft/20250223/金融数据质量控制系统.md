    ```python
    from datetime import datetime

    def monitor_data_quality(data):
        # 检查数据完整性
        if data.isnull().any():
            print("存在缺失值")
        # 检查数据一致性
        if data.duplicated().any():
            print("存在重复值")
        # 检查数据及时性
        if max(data['timestamp']) < datetime.now():
            print("数据延迟")
    ```
