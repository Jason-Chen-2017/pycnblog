  ```python
  import numpy as np
  from scipy.optimize import minimize
  
  def portfolio_return(weights, returns):
      return np.dot(weights.T, returns)
  
  def portfolio_variance(weights, covariance_matrix):
      return np.dot(weights.T, np.dot(covariance_matrix, weights))
  
  def optimize_portfolio(returns, covariance_matrix, target_return):
      n = len(weights)
      constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 1},
                     {'type': 'eq', 'fun': lambda x: portfolio_return(x, returns) - target_return}]
      result = minimize(portfolio_variance, np.ones(n)/n, 
                        constraints=constraints, method='SLSQP')
      return result.x
  ```
