import pandas as pd
import matplotlib.pyplot as plt

# 数据获取与清洗
df_stock = pd.read_csv('global_stock_indices.csv')
df_tourism = pd.read_csv('space_tourism_data.csv')

# 数据合并与相关性计算
merged_data = pd.merge(df_stock['index_value'], df_tourism['revenue'], on='date')
correlation = merged_data.corr()

# 可视化
plt.scatter(merged_data['index_value'], merged_data['revenue'])
plt.xlabel('Stock Index Value')
plt.ylabel('Tourism Revenue')
plt.title('Correlation between Stock Valuation and Tourism Revenue')
plt.show()
