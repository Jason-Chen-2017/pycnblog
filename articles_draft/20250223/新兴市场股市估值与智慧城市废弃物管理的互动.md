#### 3.3 代码实现（Python）

```python
import numpy as np
import pandas as pd
from sklearn.linear_model import LinearRegression
from sklearn.metrics import r2_score

# 数据加载与预处理
data = pd.DataFrame({
    'valuation': [100, 150, 200, 250],
    'waste_management_cost': [10, 15, 20, 25]
})

# 模型训练
model = LinearRegression()
model.fit(data[['valuation']], data['waste_management_cost'])

# 结果分析
predictions = model.predict(data[['valuation']])
print("R²:", r2_score(data['waste_management_cost'], predictions))
```

import numpy as np
import pandas as pd
from sklearn.linear_model import LinearRegression

# 数据加载
data = pd.DataFrame({
    'valuation': [100, 150, 200, 250],
    'waste_management_cost': [10, 15, 20, 25]
})

model.fit(data[['valuation']], data['waste_management_cost'])

# 预测与评估
predictions = model.predict([[200]])
print("预测处理成本:", predictions[0])
print("R²值:", model.score(data[['valuation']], data['waste_management_cost']))
