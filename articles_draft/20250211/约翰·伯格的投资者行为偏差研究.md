#### 5.2 系统核心实现
以下是投资者行为偏差识别与分析的核心代码实现：

```python
import pandas as pd
import numpy as np
from sklearn.tree import DecisionTreeClassifier
import statsmodels.api as sm

# 数据预处理
data = pd.read_csv('investor_behavior.csv')
data = data.dropna()

# 特征提取
features = ['交易频率', '持仓时间', '收益波动']
target = '行为偏差'

# 模型训练
model = DecisionTreeClassifier()
model.fit(data[features], data[target])

# 模型解读
X = data[['交易频率', '持仓时间', '收益波动']]
y = data['行为偏差']
model_ols = sm.OLS(y, X)
results = model_ols.fit()
print(results.summary())

# 偏差识别
def identify_bias(investor_id):
    return model.predict([[investor_id, 0.5, 10, 0.2]])

identify_bias(1)
```
