```python
import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestRegressor

def load_data():
    # 加载数据
    data = pd.read_csv('governance_data.csv')
    return data

def preprocess(data):
    # 数据预处理
    data = data.dropna()
    data = (data - data.mean()) / data.std()
    return data

def train_model(data, target):
    # 训练模型
    features = data.drop(target, axis=1)
    model = RandomForestRegressor(n_estimators=100, random_state=42)
    model.fit(features, target)
    return model

def calculate_score(model, features, weights):
    # 计算评分
    predicted = model.predict(features)
    weighted_scores = predicted * weights
    total_score = weighted_scores.sum()
    return total_score

def main():
    data = load_data()
    processed_data = preprocess(data)
    target = processed_data['score']
    model = train_model(processed_data, target)
    weights = np.array([0.4, 0.3, 0.2, 0.1])
    final_score = calculate_score(model, processed_data.drop('score', axis=1), weights)
    print(f'最终评分：{final_score}')

if __name__ == '__main__':
    main()
```
