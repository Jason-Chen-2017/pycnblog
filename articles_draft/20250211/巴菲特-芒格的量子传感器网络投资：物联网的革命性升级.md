  import numpy as np
  from qiskit import QuantumCircuit, Aer, execute

  def quantum_optimization(n_qubits):
      # 初始化量子电路
      qc = QuantumCircuit(n_qubits)
      # 应用哈密顿量
      qc.h(range(n_qubits))
      qc.z(range(n_qubits))
      qc.measure_all()
      # 执行模拟
      backend = Aer.get_backend('qasm_simulator')
      result = execute(qc, backend).result()
      return result.get_counts()

  # 示例：优化投资组合
  n = 3  # 假设3个资产
  counts = quantum_optimization(n)
  print(counts)
