from qiskit import QuantumCircuit, execute, Aer

def optimize_portfolio(n_assets, risk_tolerance):
    qc = QuantumCircuit(n_assets)
    # 量子计算部分
    qc.h(range(n_assets))
    qc.measure_all()
    # 执行计算
    backend = Aer.get_backend('qasm_simulator')
    job = execute(qc, backend)
    result = job.result()
    return result.get_counts()

# 示例运行
n_assets = 3
risk_tolerance = 0.05
result = optimize_portfolio(n_assets, risk_tolerance)
