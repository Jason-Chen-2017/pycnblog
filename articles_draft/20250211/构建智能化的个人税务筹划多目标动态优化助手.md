def multi_objective_dynamic_optimization():
    import numpy as np
    import random

    # 目标函数
    def objective_function(x):
        return (x[0]**2 + x[1]**2) / 2

    # 约束条件
    def constraint(x):
        return x[0] + x[1] <= 1

    # 初始化参数
    x = np.array([0.5, 0.5])
    iteration = 0
    max_iterations = 1000

    for _ in range(max_iterations):
        iteration += 1
        # 计算目标函数值
        current_value = objective_function(x)
        # 检查约束条件
        if constraint(x):
            print(f"Iteration {iteration}: Optimal solution found: {x}")
            return x
        # 随机调整参数
        delta = random.uniform(0, 1)
        x = x + delta * (1 - iteration / max_iterations)

    print("Optimization completed without finding an exact solution.")
    return x

multi_objective_dynamic_optimization()
