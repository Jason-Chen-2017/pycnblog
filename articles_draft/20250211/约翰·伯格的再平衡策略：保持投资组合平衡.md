```python
import numpy as np

def calculate_rebalancement(current_weights, target_weights):
    # 计算差异
    differences = target_weights - current_weights
    # 计算调整金额
    adjustment_amount = differences * np.sum(current_weights)
    return adjustment_amount

current_weights = np.array([0.5, 0.3, 0.2])
target_weights = np.array([0.4, 0.4, 0.2])
adjustment_amount = calculate_rebalancement(current_weights, target_weights)
print(adjustment_amount)
```
