#### 5.2 核心代码实现
```python
import numpy as np
from sklearn import linear_model

# 智能体类
class Agent:
    def __init__(self, data):
        self.data = data
        self.model = linear_model.LinearRegression()

    def train(self):
        X = self.data[['收入', '支出']]
        y = self.data['现金流']
        self.model.fit(X, y)

    def predict(self, X):
        return self.model.predict(X)

# 多智能体协同
class MultiAgentSystem:
    def __init__(self, data):
        self.agents = [Agent(data), Agent(data)]

    def evaluate(self, X):
        results = []
        for agent in self.agents:
            agent.train()
            results.append(agent.predict(X))
        return np.mean(results, axis=0)
```
