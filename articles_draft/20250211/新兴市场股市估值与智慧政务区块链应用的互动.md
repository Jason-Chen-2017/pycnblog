### 3.3.1 新兴市场股市估值的Python实现
```python
import pandas as pd
import numpy as np
from sklearn.linear_model import LinearRegression

# 数据加载
data = pd.read_csv('em_stocks.csv')

# 数据预处理
data['Date'] = pd.to_datetime(data['Date'])
data.set_index('Date', inplace=True)

# 模型训练
model = LinearRegression()
X = data[['PE', 'PS', 'ROE']]
y = data['Price']
model.fit(X, y)

# 预测
new_data = pd.DataFrame({
    'PE': [15],
    'PS': [3],
    'ROE': [12]
})
predicted_price = model.predict(new_data)
print(f"预测价格为: {predicted_price[0]:.2f}")
```

### 3.3.2 智慧政务区块链应用的Python实现
```python
from hashlib import sha256
import json

def compute_hash(data):
    return sha256(json.dumps(data).encode()).hexdigest()

# 示例数据
government_data = {
    'record_id': 123,
    'timestamp': '2023-10-01',
    'action': 'approval',
    'status': 'completed'
}

# 计算哈希
data_hash = compute_hash(government_data)
print(f"数据哈希值为: {data_hash}")
```
