```python
import numpy as np
import matplotlib.pyplot as plt

def calculate_efficient_frontier(returns, cov_matrix, num_portfolios=1000):
    # 计算有效前沿
    n = len(returns)
    means = returns.mean()
    variances = returns.var()

    # 生成随机的组合权重
    weights = np.random.dirichlet(np.ones(n), num_portfolios)
    # 计算组合收益和风险
    port_returns = np.dot(weights, returns)
    port_variances = np.zeros(num_portfolios)
    for i in range(num_portfolios):
        port_variances[i] = np.dot(weights[i], np.dot(cov_matrix, weights[i]))

    # 绘制有效前沿
    plt.scatter(port_variances, port_returns, c=port_returns, cmap='viridis')
    plt.colorbar()
    plt.xlabel('Risk')
    plt.ylabel('Return')
    plt.title('Efficient Frontier')
    plt.show()

# 示例数据
returns = np.array([0.05, 0.07, 0.06, 0.04])
cov_matrix = np.array([
    [0.02, 0.01, 0.015, 0.01],
    [0.01, 0.03, 0.02, 0.015],
    [0.015, 0.02, 0.035, 0.02],
    [0.01, 0.015, 0.02, 0.04]
])

calculate_efficient_frontier(returns, cov_matrix)
```
