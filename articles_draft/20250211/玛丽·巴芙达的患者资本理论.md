#### 4.2 患者资本理论的Python实现
```python
import numpy as np

def calculate_return_rate(initial_investment, capital_gain):
    return (capital_gain / initial_investment) * 100

def calculate_risk(standard_deviation, beta_coefficient):
    return np.sqrt(standard_deviation**2 + beta_coefficient**2)

# 示例
initial_investment = 1000000
capital_gain = 200000
standard_deviation = 0.12
beta_coefficient = 1.5

return_rate = calculate_return_rate(initial_investment, capital_gain)
risk = calculate_risk(standard_deviation, beta_coefficient)

print(f"回报率: {return_rate}%")
print(f"风险系数: {risk}")
```
