  - 若护城河评分 < 4，则不具备宽护城河。

import pandas as pd
import numpy as np

def calculate_h护城河评分(row):
    return (row['品牌优势'] + row['成本优势'] + row['竞争优势'] + row['管理层能力']) / 4

def is宽护城河评分(score):
