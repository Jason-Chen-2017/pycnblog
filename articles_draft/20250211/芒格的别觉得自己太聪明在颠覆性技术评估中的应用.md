def assess_technology(technologies):
    weights = [0.4, 0.3, 0.2, 0.1]
    result = []
    for tech in technologies:
        total = 0
        for i in range(len(tech)):
            total += weights[i] * tech[i]
        result.append(total)
    return result

technologies = [
    [0.8, 0.7, 0.6, 0.5],
    [0.6, 0.5, 0.7, 0.8]
]
print(assess_technology(technologies))
