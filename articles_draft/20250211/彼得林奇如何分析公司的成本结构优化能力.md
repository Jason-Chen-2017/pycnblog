from pulp import *

def optimize_cost(costs, constraints):
    # 创建问题
    prob = LpProblem("Cost_Optimization", LpMinimize)
    
    # 定义变量
    x = [LpVariable(f'x{i}', 0, None) for i in range(len(costs))]
    
    # 定义目标函数
    objective = sum(c * x[i] for i, c in enumerate(costs))
    prob += objective, "Total Cost"
    
    # 定义约束
    prob += sum(x[i] for i in range(len(costs))) <= constraints, "Total"
    
    # 求解问题
    prob.solve()
    
    # 输出结果
    return [value(x[i]) for i in range(len(costs))]

# 示例数据
costs = [10, 15, 20]
constraints = 100

# 调用函数
result = optimize_cost(costs, constraints)
print("最优解为：", result)
