#### 8.2 核心代码实现

import numpy as np
import pandas as pd

# 数据准备
data = pd.DataFrame({'x': [1, 2, 3, 4, 5], 'y': [2, 4, 5, 4, 5]})

# 模型训练
X = data[['x']]
y = data['y']

from sklearn.linear_model import LinearRegression
model = LinearRegression()
model.fit(X, y)

# 预测
predicted = model.predict(X)
print(predicted)
