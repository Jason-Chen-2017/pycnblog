from django.http import JsonResponse
from .models import UniverseInfo

def get_universe_info(request):
    universe_info = UniverseInfo.objects.all()
    data = [
        {
            'id': info.id,
            'name': info.name,
            'description': info.description,
        }
        for info in universe_info
    ]
    return JsonResponse({'universe_info': data})
