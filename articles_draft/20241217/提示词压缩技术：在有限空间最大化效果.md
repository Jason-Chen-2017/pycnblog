##### 3.2 Python实现示例

```python
import heapq
from collections import defaultdict

# 哈夫曼编码实现
def huffman_encoding(data):
    # 统计字符频率
    frequency = defaultdict(int)
    for char in data:
        frequency[char] += 1

    # 构建优先队列
    heap = [[weight, [symbol, ""]] for symbol, weight in frequency.items()]
    heapq.heapify(heap)

    # 构建哈夫曼树
    while len(heap) > 1:
        lo = heapq.heappop(heap)
        hi = heapq.heappop(heap)
        for pair in lo[1:]:
            pair[1] = '0' + pair[1]
        for pair in hi[1:]:
            pair[1] = '1' + pair[1]
        heapq.heappush(heap, [lo[0] + hi[0]] + lo[1:] + hi[1:])

    # 生成编码表
    huffman_tree = heap[0]
    encoding_table = defaultdict(str)
    for pair in huffman_tree[1:]:
        symbol, code = pair
        encoding_table[symbol] = code

    # 编码提示词
    encoded_data = ""
    for char in data:
        encoded_data += encoding_table[char]

    return encoded_data

# 解码提示词
def huffman_decoding(encoded_data, encoding_table):
    reverse_encoding_table = {v: k for k, v in encoding_table.items()}
    decoded_data = ""
    current_code = ""

    for bit in encoded_data:
        current_code += bit
        if current_code in reverse_encoding_table:
            decoded_data += reverse_encoding_table[current_code]
            current_code = ""

    return decoded_data

# 示例
data = "this is an example for huffman encoding"
encoded_data = huffman_encoding(data)
decoded_data = huffman_decoding(encoded_data, encoding_table)

print("Encoded data:", encoded_data)
print("Decoded data:", decoded_data)
```
