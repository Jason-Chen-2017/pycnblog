**源代码实现：**

以下是一个简单的Python代码示例，用于模拟黑洞的形成过程：

```python
import numpy as np
import matplotlib.pyplot as plt

# 黑洞质量
M = 1e10

# 黑洞半径
R = 3 * M / np.pi

# 模拟空间
x = np.linspace(-10 * R, 10 * R, 1000)
y = np.linspace(-10 * R, 10 * R, 1000)

# 引力势能计算
def potential(x, y, M):
    r = np.sqrt(x**2 + y**2)
    return -M / r

# 黑洞边界计算
def event_horizon(x, y, R):
    r = np.sqrt(x**2 + y**2)
    return r <= R

# 绘制引力势能图
fig, ax = plt.subplots()
contour = ax.contour(x, y, potential(x, y, M), levels=[0])
plt.colorbar(contour)
plt.title('Gravitational Potential around a Black Hole')
plt.xlabel('x')
plt.ylabel('y')

# 绘制黑洞边界
event_horizon_mask = event_horizon(x, y, R)
ax.plot(x[event_horizon_mask], y[event_horizon_mask], color='red')
plt.show()
```
