```python
import networkx as nx
from sklearn.metrics import pairwise_distances
import numpy as np

def fisher_network_analysis(data):
    # 创建图
    G = nx.Graph()
    # 添加节点
    G.add_nodes_from(data['node'])
    # 添加边
    G.add_edges_from(data['edge'])
    # 计算节点度
    degrees = nx.degree(G)
    return degrees

# 示例数据
data = {
    'node': ['A', 'B', 'C'],
    'edge': [('A', 'B'), ('B', 'C')]
}

result = fisher_network_analysis(data)
print(result)
```
