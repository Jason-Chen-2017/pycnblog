    # 初始化适应性得分
    adaptability_score = 0

    # 根据团队成熟度计算适应性得分
    if team_maturity == "新手":
        adaptability_score += 10
    elif team_maturity == "成长":
        adaptability_score += 20
    elif team_maturity == "熟练":
        adaptability_score += 30

    # 根据适应性得分评估领导风格适应性
    if adaptability_score >= 30:
        return "高度适应"
    elif adaptability_score >= 20:
        return "中等适应"
    else:
        return "低度适应"

# 应用示例
