```python
# 示例代码：设计一个简单的ChatGPT提示词

import openai

# 设计提示词
prompt = "请根据以下问题给出详细回答：什么是人工智能？"

# 调用ChatGPT接口
response = openai.Completion.create(

  prompt=prompt,
  max_tokens=150
)

# 输出答案
print(response.choices[0].text.strip())
```
