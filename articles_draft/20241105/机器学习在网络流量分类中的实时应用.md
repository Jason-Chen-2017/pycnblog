#### C.1 k-均值聚类算法

```python
def k_means(data, k):
    # 初始化聚类中心
    centers = initialize_centers(data, k)
    
    while not convergence(centers):
        # 为每个数据点分配最近的聚类中心
        for data_point in data:
            closest_center = find_closest_center(data_point, centers)
            assign_to_cluster(data_point, closest_center)
        
        # 更新聚类中心
        new_centers = update_centers(centers, data)
        
        # 判断是否收敛
        if is_converged(centers, new_centers):
            break
    
    return centers
```

```python
def build_decision_tree(data):

        return most_common_label(data)
    
    # 找到最佳特征
    best_feature = find_best_feature(data)
    
    # 根据最佳特征划分数据集
    yes_data, no_data = split_data(data, best_feature)
    
    # 递归构建子树
    left_tree = build_decision_tree(yes_data)
    right_tree = build_decision_tree(no_data)
    
    # 创建决策树节点
    node = Node(best_feature, left_tree, right_tree)
    
    return node
```
