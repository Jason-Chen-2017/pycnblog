```python
    # 分配任务
    for member in team_members:
        member['task'] = tasks.pop(0)
    # 监督进度
    for member in team_members:
        if member['task']['status'] == 'in_progress':
            print(f"{member['name']}的任务进度：{member['task']['progress']}")
    # 反馈结果
    return [member['task']['result'] for member in team_members]
```
