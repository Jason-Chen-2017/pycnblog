```python
def calculate_innovation_capacity(technology, market, management):
    # 权重系数
    alpha = 0.4
    beta = 0.3
    gamma = 0.3
    # 计算创新能力
    capacity = alpha * technology + beta * market + gamma * management
    return capacity

# 示例数据

```
