      return cash_flow / (discount_rate - growth_rate)

- 投资组合优化的代码
  ```python
  import numpy as np

  def optimize_portfolio(returns, cov_matrix):
      n = len(returns)
      # 初始化权重向量
      weights = np.ones(n) / n
      # 使用拉格朗日乘数法优化
      while True:
          # 计算夏普比率
          port_ret = np.dot(weights, returns)
          port_vol = np.sqrt(np.dot(weights.T, np.dot(cov_matrix, weights)))
          # 检查收敛条件
          if port_vol < 0.01:
              break
          # 调整权重以降低波动率
          weights = weights / sum(weights)
      return weights
  
  # 示例
  returns = np.array([0.1, 0.05, 0.08])
  cov_matrix = np.cov(returns)
  weights = optimize_portfolio(returns, cov_matrix)
  print(f"最优权重: {weights}")
  ```
