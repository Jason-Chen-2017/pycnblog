    return health_score

from sklearn import linear_model

# 示例代码：企业价值预测模型
data = pd.DataFrame({
    'assets': [100, 200, 300],
    'liabilities': [80, 160, 240],
    'equity': [20, 40, 60],
    'value': [120, 240, 360]
})

model = linear_model.LinearRegression()
model.fit(data[['assets', 'liabilities', 'equity']], data['value'])

print(f"模型系数：{model.coef_}")
print(f"模型截距：{model.intercept_}")
