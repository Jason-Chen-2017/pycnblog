  def reverse_thinking_algorithm(c特征, r逻辑):
      # 计算影响程度
      影响 = 0
      for i in range(len(c特征)):
          影响 += c特征[i] * r逻辑[i]
      return 影响

  # 示例
  c特征 = [0.8, 0.6, 0.7]
  r逻辑 = [0.5, 0.4, 0.6]
  结果 = reverse_thinking_algorithm(c特征, r逻辑)
