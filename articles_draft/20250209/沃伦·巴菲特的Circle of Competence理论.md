def calculate_circle_boundaries(data):
    # 计算内边界和外边界
    inner_boundary = min(data)
    outer_boundary = max(data)
    return inner_boundary, outer_boundary

def dynamic_adjustment(data, inner, outer):
    # 动态调整能力圈
    new_inner, new_outer = calculate_circle_boundaries(data)
    if inner <= new_inner and new_outer <= outer:
        return new_inner, new_outer
    else:
        return inner, outer
