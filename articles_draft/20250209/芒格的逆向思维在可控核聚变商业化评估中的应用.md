import pandas as pd
import matplotlib.pyplot as plt

# 数据采集
data = pd.read_csv('fusion_data.csv')

# 数据处理
data_processed = data.dropna()

# 模型计算
def calculate_npv(data, discount_rate):
    npv = 0
    for t in range(len(data)):
        cash_flow = data.iloc[t]
        npv += cash_flow / (1 + discount_rate) ** t
    return npv

discount_rate = 0.1
npv = calculate_npv(data_processed, discount_rate)

# 结果分析
print(f'净现值：{npv}')

plt.plot(range(len(data_processed)), data_processed)
plt.title('现金流折现分析')
plt.xlabel('时间（年）')
plt.ylabel('现金流（万元）')
plt.show()
