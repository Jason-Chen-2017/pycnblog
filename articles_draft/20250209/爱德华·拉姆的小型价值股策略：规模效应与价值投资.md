         return price / earnings

     def calculate_pb_ratio(price, book_value):
         return price / book_value

     def calculate_roe(net_profit, total_equity):
         return net_profit / total_equity

     # 示例数据
     price = 100
     earnings = 20
     book_value = 50
     net_profit = 10
     total_equity = 50

     pe_ratio = calculate_pe_ratio(price, earnings)
     pb_ratio = calculate_pb_ratio(price, book_value)
     roe = calculate_roe(net_profit, total_equity)

     print(f"PE Ratio: {pe_ratio}")
     print(f"PB Ratio: {pb_ratio}")
     print(f"ROE: {roe}")

         selected_stocks = []
         for stock in stocks:
             pe = calculate_pe_ratio(stock['price'], stock['earnings'])
             pb = calculate_pb_ratio(stock['price'], stock['book_value'])
             roe = calculate_roe(stock['net_profit'], stock['total_equity'])
             if pe < 15 and pb < 2 and roe > 8:
                 selected_stocks.append(stock)
         return selected_stocks
