import pandas as pd

import matplotlib.pyplot as plt

data = pd.read_csv('user_evolution.csv')

# 用户画像特征提取

y = data['user_evolvement']

# 模型训练

model.fit(X, y)

# 预测结果
y_pred = model.predict(X)

# 可视化
plt.scatter(y, y_pred)

plt.title('用户画像演变预测模型')
plt.show()
