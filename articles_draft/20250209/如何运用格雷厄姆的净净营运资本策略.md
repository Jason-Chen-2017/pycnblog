import pandas as pd

def calculate_net_operating_assets(df):
    net_operating_assets = df['流动资产'] - df['流动负债']
    return net_operating_assets

# 示例数据
data = {
    '公司': ['公司A', '公司B', '公司C'],
    '流动资产': [100, 150, 200],
    '流动负债': [60, 70, 80]
}
df = pd.DataFrame(data)

# 计算净净营运资本
net_assets = calculate_net_operating_assets(df)
print(net_assets)
