import numpy as np

def宇宙膨胀模型(宇宙观测数据，时间步数):
    膨胀速率 = 宇宙观测数据['膨胀速率']
    时间步 = np.linspace(0, 时间步数， 时间步数)

    膨胀距离 = 膨胀速率 * 时间步

宇宙观测数据 = {'膨胀速率': [2.5, 3.0, 3.5]}

膨胀距离 = 宇宙膨胀模型(宇宙观测数据，时间步数)
