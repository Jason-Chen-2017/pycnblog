**Python源代码讲解**：

```python
import re

def filter_keywords(input_text, keyword_list):
    for keyword in keyword_list:
        if re.search(keyword, input_text):
            return "过滤失败：包含敏感词"
    return "过滤成功：无敏感词"

# 示例
keyword_list = ["恶意代码", "黑客攻击"]
input_text = "请输入您的查询：我是一个正常的用户，想查询天气预报"
print(filter_keywords(input_text, keyword_list))
```
