# 需求分析模块
def need_analysis(traditional_demand, innovative_demand):
    # 根据传统需求和创新需求的权重，选择葬礼风格
    if traditional_demand > innovative_demand:
        return "传统"
    else:
        return "创新"

# 环境影响评估模块
def environmental_assessment(ceremony_style):
    # 根据葬礼风格评估环境影响
    # 返回一个字典，包含"适应性"和"环境影响"两个键
    if ceremony_style == "传统":
        return {'适应性': False, '环境影响': 1}
    else:
        return {'适应性': True, '环境影响': 0}

# 环保葬礼设计模块
def eco_funeral_design(need_analysis):
    ceremony_style = need_analysis()
    assessment_result = environmental_assessment(ceremony_style)
    if not assessment_result['适应性']:
        ceremony_style = "创新"
        assessment_result = environmental_assessment(ceremony_style)
    return ceremony_style, assessment_result

# 主程序
if __name__ == "__main__":
    traditional_demand = 0.6
    innovative_demand = 0.4
    ceremony_style, assessment_result = eco_funeral_design(need_analysis, traditional_demand, innovative_demand)
    print("葬礼风格：", ceremony_style)
    print("环境影响评估结果：", assessment_result)
