import matplotlib.pyplot as plt
import numpy as np

def koch_curve(points, levels):
    if levels == 0:
        return [points]
    midpoints = [(points[0] + points[1]) / 2, (points[1] + points[2]) / 2, (points[2] + points[0]) / 2]
    new_points = koch_curve(points[:2] + midpoints, levels - 1) + koch_curve(midpoints + points[2:], levels - 1)
    return new_points

points = np.array([[0, 0], [1, 0], [1, 1], [0, 1]])
levels = 3
koch_curve_points = koch_curve(points, levels)

plt.plot(koch_curve_points[:, 0], koch_curve_points[:, 1])
plt.show()
