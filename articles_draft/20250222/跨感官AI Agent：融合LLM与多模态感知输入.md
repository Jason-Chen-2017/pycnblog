```python
import torch
import torch.nn as nn

class CrossModalFusion(nn.Module):
    def __init__(self, input_dim_v, input_dim_a, output_dim):
        super(CrossModalFusion, self).__init__()
        self.encoder_v = nn.Linear(input_dim_v, output_dim)
        self.encoder_a = nn.Linear(input_dim_a, output_dim)
        self.decoder = nn.Linear(output_dim, output_dim)
    
    def forward(self, x_v, x_a):
        h_v = self.encoder_v(x_v)
        h_a = self.encoder_a(x_a)
        h = h_v + h_a
        output = self.decoder(h)
        return output
```
