  import pandas as pd
  import numpy as np

  # 示例代码：计算企业价值
  def calculate_enterprise_value(cash_flows, discount_rate):
      present_value = sum(cash_flow / (1 + discount_rate)**i for i, cash_flow in enumerate(cash_flows))
      return present_value

  # 示例数据
  cash_flows = [100, 150, 200, 250]
  discount_rate = 0.1
