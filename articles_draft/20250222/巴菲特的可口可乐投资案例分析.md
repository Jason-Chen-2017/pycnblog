## 3.3 代码实现

```python
def calculate_intrinsic_value(future_cash_flows, discount_rate):
    present_value = sum(flow / (1 + discount_rate)**i for i, flow in enumerate(future_cash_flows))
    return present_value / discount_rate

def calculate_margin_of_safety(intrinsic_value, market_price):
    return intrinsic_value - market_price

# 示例数据
future_cash_flows = [100, 120, 140, 160]  # 每年的现金流
discount_rate = 0.08

intrinsic_value = calculate_intrinsic_value(future_cash_flows, discount_rate)
market_price = 50  # 市场价格
margin_of_safety = calculate_margin_of_safety(intrinsic_value, market_price)

print(f"内在价值: {intrinsic_value}")
print(f"市场价格: {market_price}")
print(f"安全边际: {margin_of_safety}")
```
