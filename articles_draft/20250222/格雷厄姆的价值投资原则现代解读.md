### 3.2.2 Python代码实现DCF模型  
```python
import numpy as np
import pandas as pd

def calculate_intrinsic_value(cash_flows, discount_rate, terminal_value):
    present_value = 0
    for year in range(len(cash_flows)):
        present_value += cash_flows[year] / (1 + discount_rate) ** year
    present_value += terminal_value / (1 + discount_rate) ** len(cash_flows)
    return present_value

# 示例数据
cash_flows = [100, 120, 150, 180]  # 每年的自由现金流
discount_rate = 0.10  # 折现率
terminal_value = 200  # 终值

intrinsic_value = calculate_intrinsic_value(cash_flows, discount_rate, terminal_value)
print(f"内在价值：{intrinsic_value:.2f}")
```
