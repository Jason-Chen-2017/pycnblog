from sklearn.metrics import mean_squared_error

def calculate_competitiveness(company_data, industry_avg_cost):
    company_cost = company_data['cost']
    cost_advantage = company_cost / industry_avg_cost
    return cost_advantage

# 示例数据

    '供应商': ['供应商A', '供应商B', '供应商C'],
    '生产商': ['生产商X', '生产商Y', '生产商Z'],
    '分销商': ['分销商1', '分销商2', '分销商3'],

    'industry_avg_cost': 15
