```python
import numpy as np

def optimize_portfolio(cov_matrix, returns):
    n = len(returns)
    # 定义目标函数
    def objective(w):
        return (w.T @ cov_matrix @ w)[0,0]
    # 定义约束条件
    constraints = ({'type': 'eq', 'fun': lambda w: np.sum(w) - 1})
    # 使用SLSQP求解
    from scipy.optimize import minimize
    result = minimize(objective, np.ones(n)/n, method='SLSQP', constraints=constraints)
    return result.x
```
