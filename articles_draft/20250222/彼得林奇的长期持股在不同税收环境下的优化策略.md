```python
def optimize_portfolio(tax_rate, holding_period):
    # 计算税后收益
    pre_tax_return = 1.05  # 假设年化收益率为5%
    post_tax_return = pre_tax_return * (1 - tax_rate)
    # 计算持有期的总收益
    total_return = post_tax_return ** holding_period
    return total_return

# 示例：在10%税率下的5年持有期
print(optimize_portfolio(0.10, 5))
```

#### 5.1 税后收益计算公式
$$ \text{税后收益} = \text{投资收益} \times (1 - \text{税率}) $$

#### 5.2 投资组合优化的数学模型
$$ \max_{w} \left( \mu^T w - \lambda \times \text{Tax}(w) \right) $$
其中，$\mu$ 是资产的预期收益率向量，$w$ 是投资权重向量，$\lambda$ 是风险厌恶系数，$\text{Tax}(w)$ 是税收函数。

import numpy as np
import pandas as pd

def calculate_tax_optimized_returns(tax_rates, holding_periods):
    returns = []
    for rate in tax_rates:
        for period in holding_periods:
            pre_tax = 1.05 ** period  # 假设年化收益率为5%
            post_tax = pre_tax * (1 - rate)
            returns.append(post_tax)
    return returns

# 示例：计算不同税率和持有期的税后收益
tax_rates = [0.10, 0.15, 0.20]
holding_periods = [1, 2, 3, 4, 5]
returns = calculate_tax_optimized_returns(tax_rates, holding_periods)
print(returns)
