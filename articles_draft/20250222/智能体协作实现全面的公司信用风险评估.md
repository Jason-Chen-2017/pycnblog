  import random

  class Agent:
      def __init__(self, id):
          self.id = id
          self.data = None

      def receive_data(self, data):
          self.data = data

      def send_data(self):
          return self.data

  # 初始化智能体
  agents = [Agent(i) for i in range(10)]

  # 分配任务
  def distribute_task(agents, data):
      agent = random.choice(agents)
      agent.receive_data(data)
      return agent

  # 协作计算
  def collaborate(agents):
      results = []
      for agent in agents:
          results.append(agent.send_data())
      return results

  # 示例数据
  data = {"credit_score": 700, "income": 50000}
  # 分配任务
  selected_agent = distribute_task(agents, data)
  # 协作计算
  results = collaborate(agents)
  print(results)
