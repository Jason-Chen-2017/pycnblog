import numpy as np
import pandas as pd
from datetime import datetime

def generate_signals(data, window=20):
    # 计算短期移动平均线
    short_ma = data['close'].rolling(window=5).mean()
    # 计算长期移动平均线
    long_ma = data['close'].rolling(window=20).mean()
    # 生成交易信号
    signals = short_ma - long_ma
    return signals
