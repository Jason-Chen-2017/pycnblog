import numpy as np
def monte_carlo_simulation(prices, iterations=1000):
    returns = np.log(prices / prices.shift(1)).dropna()
    simulations = np.random.normal(returns.mean(), returns.std(), size=(iterations, len(prices)))
    return simulations

# 示例数据
prices = [100, 150, 200]
simulations = monte_carlo_simulation(prices)
print(simulations)
