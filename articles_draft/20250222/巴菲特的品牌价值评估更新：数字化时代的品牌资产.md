```python
# 示例代码
import pandas as pd
from sklearn.decomposition import FactorAnalyzer

# 数据加载
data = pd.read_csv('brand_data.csv')

# 数据清洗
data.dropna(inplace=True)

# 特征提取
features = data[['feature1', 'feature2', 'feature3']]

# 模型训练
fa = FactorAnalyzer(n_factors=3, random_state=42)
fa.fit(features)

# 提取因子得分
factor_scores = pd.DataFrame(fa.components_, columns=['factor1', 'factor2', 'factor3'])

# 品牌价值评估
brand_value = factor_scores.mean(axis=1)
print(brand_value)
```
