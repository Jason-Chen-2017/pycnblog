```python
import pandas as pd

def calculate_finance_health_index(finance_data):
    # 计算关键财务比率
    roe = finance_data['净利润'] / finance_data['股东权益']
    debt_ratio = finance_data['负债总额'] / finance_data['资产总额']
    growth_rate = (finance_data['净利润'] - finance_data['去年同期净利润']) / finance_data['去年同期净利润']
    
    # 生成财务健康指数
    health_index = (roe * 0.4) + (debt_ratio * 0.3) + (growth_rate * 0.3)
    return health_index

# 示例数据
data = {
    '净利润': [100, 120, 150],
    '股东权益': [500, 600, 700],
    '负债总额': [300, 350, 400],
    '资产总额': [800, 900, 1000],
    '去年同期净利润': [80, 90, 100]
}

finance_data = pd.DataFrame(data)
health_index = calculate_finance_health_index(finance_data)
print("财务健康指数:", health_index)
```
