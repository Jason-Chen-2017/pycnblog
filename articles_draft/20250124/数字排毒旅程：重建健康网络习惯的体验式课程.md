```python
import time

def reduce_screen_time():
    """
    减少屏幕使用时间的方法
    """
    print("开始减少屏幕使用时间...")
    start_time = time.time()
    while True:
        current_time = time.time()
        if current_time - start_time >= 3600:  # 每小时提醒一次
            print("已经达到一小时屏幕使用时间，请休息一下。")
            start_time = current_time

def main():
    """
    数字排毒主程序
    """
    print("欢迎来到数字排毒实践！")
    reduce_screen_time()

if __name__ == "__main__":
    main()
```
