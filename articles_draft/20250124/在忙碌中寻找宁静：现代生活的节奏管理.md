```python
# Python代码实现时间管理算法
import time

def time_management(tasks, time_allotment):
    start_time = time.time()
    for task in tasks:
        start = time.time()
        print(f"开始执行任务：{task['name']}")
        time.sleep(task['duration'])
        end = time.time()
        print(f"完成任务：{task['name']}，耗时：{end - start}秒")
    end_time = time.time()
    print(f"总耗时：{end_time - start_time}秒")

tasks = [
    {"name": "任务1", "duration": 5},
    {"name": "任务2", "duration": 10},
    {"name": "任务3", "duration": 3}
]

time_management(tasks, 20)
```

```python
# Python代码实现时间管理系统
import sqlite3
from PyQt5 import QtWidgets, uic

class TimeManager:
    def __init__(self):
        self.conn = sqlite3.connect('tasks.db')
        self.cursor = self.conn.cursor()
        self.cursor.execute('''CREATE TABLE IF NOT EXISTS tasks
                            (id INTEGER PRIMARY KEY, name TEXT, duration INTEGER, status TEXT)''')
        self.conn.commit()

    def add_task(self, task):
        self.cursor.execute('''INSERT INTO tasks (name, duration, status) VALUES (?, ?, ?)''', (task['name'], task['duration'], '未完成'))
        self.conn.commit()

    def remove_task(self, task_id):
        self.cursor.execute('''DELETE FROM tasks WHERE id = ?''', (task_id,))
        self.conn.commit()

    def execute_tasks(self):
        self.cursor.execute('''SELECT * FROM tasks WHERE status = '未完成' ORDER BY id''')
        tasks = self.cursor.fetchall()
        for task in tasks:
            print(f"执行任务：{task[1]}, 耗时：{task[2]}分钟")
            time.sleep(task[2])
            self.cursor.execute('''UPDATE tasks SET status = '已完成' WHERE id = ?''', (task[0],))
        self.conn.commit()

app = QtWidgets.QApplication([])
window = QtWidgets.QWidget()
uic.loadUi('time_manager.ui', window)
time_manager = TimeManager()

window.add_task_button.clicked.connect(lambda: time_manager.add_task(window.add_task_form))
window.remove_task_button.clicked.connect(lambda: time_manager.remove_task(window.remove_task_form))
window.execute_tasks_button.clicked.connect(lambda: time_manager.execute_tasks())

window.show()
app.exec_()
```
