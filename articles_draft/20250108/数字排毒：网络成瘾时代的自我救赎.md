#### 2. 系统核心实现源代码

以下是数字排毒系统的核心实现源代码：

```python
# 数据收集模块
import pandas as pd

def collect_data():
    # 读取数据文件
    data = pd.read_csv('social_media_usage.csv')
    # 计算平均使用时间
    avg_usage = data['usage_time'].mean()
    return avg_usage

# 行为分析模块
def analyze_behavior(data):
    # 计算使用频率
    freq = data['usage_time'].value_counts().sort_index()
    return freq

# 干预建议模块
def suggest_interventions(data):
    # 计算成瘾程度
    addiction_level = analyze_behavior(data)
    if addiction_level < 3:
        suggestion = "继续保持，适当增加使用频率"
    elif addiction_level < 6:
        suggestion = "减少使用时间，增加使用频率"
    else:
        suggestion = "立即停止使用，寻求专业帮助"
    return suggestion

# 主函数
def main():
    # 收集数据
    avg_usage = collect_data()
    # 分析行为
    freq = analyze_behavior(avg_usage)
    # 提出干预建议
    suggestion = suggest_interventions(freq)
    # 输出结果
    print(f"平均使用时间：{avg_usage}小时/天")
    print(f"使用频率：{freq}")
    print(f"干预建议：{suggestion}")

if __name__ == '__main__':
    main()
```
