#### 3.2 Python源代码实现

```python
import numpy as np
import pandas as pd

def calculate_long_term_return(data):
    # 数据预处理
    processed_data = data['return'].rolling(24).mean().dropna()
    # 模型训练
    model = np.polyfit(range(len(processed_data)), processed_data, 1)
    # 预测回报
    predicted_return = np.polyval(model, len(processed_data))
    return predicted_return

# 示例数据
data = pd.DataFrame({'return': np.random.randn(100)})
print("预测的长期回报:", calculate_long_term_return(data))
```

```python
from sklearn.linear_model import LinearRegression
import pandas as pd
import numpy as np

# 示例数据
data = pd.DataFrame({
    'month': range(1, 61),
    'return': np.random.randn(60) * 0.05 + 0.02
})

# 模型训练
model = LinearRegression()
model.fit(data[['month']], data['return'])

# 预测未来回报
future_months = np.array([[61]])
predicted_return = model.predict(future_months)
print("预测的长期回报:", predicted_return[0])
```
