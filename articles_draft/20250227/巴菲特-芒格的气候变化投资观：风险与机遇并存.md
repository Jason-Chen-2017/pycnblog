def monte_carlo_simulation(returns, num_simulations=10000):
    simulations = np.random.normal(returns.mean(), returns.std(), size=(num_simulations, len(returns)))
    return simulations

# 示例：情景分析法评估气候变化影响
def scenario_analysis(climate_data, investment_data):
    scenarios = climate_data['scenario'].unique()
    results = {}
    for scenario in scenarios:
        filtered_data = investment_data[climate_data['scenario'] == scenario]
        results[scenario] = filtered_data.mean()
    return results

# 示例：绘制气候变化影响的图表
def plot_climate_impact(data):
    plt.figure(figsize=(10,6))
    plt.plot(data['year'], data['impact'], label='Climate Impact')
    plt.xlabel('Year')
    plt.ylabel('Impact')
    plt.title('Climate Change Impact Over Time')
    plt.legend()
