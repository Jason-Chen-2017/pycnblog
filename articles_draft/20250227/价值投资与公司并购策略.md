import numpy as np
import pandas as pd

def discounted_cash_flow(cash_flows, discount_rate):
    return sum(cash_flows / (1 + discount_rate) ** i for i, cash_flows in enumerate(cash_flows))

# 示例数据
cash_flows = [100, 150, 200]
discount_rate = 0.08

value = discounted_cash_flow(cash_flows, discount_rate)
