import numpy as np
import matplotlib.pyplot as plt

def visualize_multi_perspective(data):
    plt.figure(figsize=(10, 6))
    for i, perspective in enumerate(data):
        plt.subplot(2, 2, i+1)
        plt.title(f'视角{i+1}')
        plt.plot(perspective)
    plt.tight_layout()
    plt.show()

# 示例数据
data = [
    np.random.rand(10),
    np.random.rand(10),
    np.random.rand(10),
