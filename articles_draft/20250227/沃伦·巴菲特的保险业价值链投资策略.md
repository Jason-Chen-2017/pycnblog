import pandas as pd

def calculate_roe(net_profit, equity):
    return net_profit / equity

def calculate_shielding(net_profit, equity, margin):
    return net_profit * (1 - margin)

# 示例数据
data = {
    '公司': ['平安保险', '中国人寿'],
    '净利润': [100, 150],
    '股东权益': [500, 600],
    '安全边际率': [0.2, 0.15]
}

df = pd.DataFrame(data)

# 计算ROE和投资价值
df['ROE'] = df.apply(lambda x: calculate_roe(x['净利润'], x['股东权益']), axis=1)
df['投资价值'] = df.apply(lambda x: x['净利润'] * (1 - x['安全边际率']), axis=1)

print(df)
