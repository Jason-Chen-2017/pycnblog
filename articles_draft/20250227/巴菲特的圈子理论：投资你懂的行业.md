```python
import pandas as pd
import numpy as np

def preprocess_data(df):
    # 删除缺失值
    df = df.dropna()
    # 标准化数据
    df['Price'] = (df['Price'] - df['Price'].mean()) / df['Price'].std()
    return df

data = fetch_data('AAPL')
processed_data = preprocess_data(data)
print(processed_data)
```
