### 7.2 核心代码实现

```python
import networkx as nx
import numpy as np

def analyze_social_behavior():
    # 创建社交网络图
    G = nx.Graph()
    G.add_nodes_from([1, 2, 3, 4])
    G.add_edges_from([(1,2), (2,3), (3,4)])
    # 计算中心性
    centrality = nx.betweenness_centrality(G)
    return centrality

result = analyze_social_behavior()
print(result)
```
